def justify(header, rows, row):
    minimise_cols = [c for c in header if c.endswith('-')]
    maximise_cols = [c for c in header if c.endswith('+')]

    col_idx = {name: i for i, name in enumerate(header)}

    def score(r):
        s = 0.0
        for col in minimise_cols:
            val = r[col_idx[col]]
            try:
                fval = float(val) if str(val) != '?' else None
            except (ValueError, TypeError):
                continue
            if fval is not None:
                s -= fval
        for col in maximise_cols:
            val = r[col_idx[col]]
            try:
                fval = float(val) if str(val) != '?' else None
            except (ValueError, TypeError):
                continue
            if fval is not None:
                s += fval
        return s

    row_score = score(row)

    scores = [score(r) for r in rows]
    n = len(scores)
    p = sum(1 for s in scores if s < row_score) / n

    reason = (
        "This setup beats {:.0%} of all {} setups evaluated on the training data.".format(p, n)
    )
    return p, reason


if __name__ == "__main__":
    import csv

    with open('Health-ClosedIssues0000.train.csv', 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        rows = []
        for row in reader:
            converted = []
            for cell in row:
                if cell.strip() == '?':
                    converted.append('?')
                else:
                    try:
                        converted.append(float(cell))
                    except ValueError:
                        converted.append(cell)
            rows.append(converted)

    p, reason = justify(header, rows, rows[0])
    print(reason)
