def remember(header, rows):
    minimise_cols = [c for c in header if c.endswith('-')]
    maximise_cols = [c for c in header if c.endswith('+')]

    best_positions = []
    scores = []

    for i, row in enumerate(rows):
        score = 0.0
        for col in minimise_cols:
            val = row[header.index(col)]
            try:
                fval = float(val) if str(val) != '?' else None
            except (ValueError, TypeError):
                continue
            if fval is not None:
                score -= fval  # lower MRE → better
        for col in maximise_cols:
            val = row[header.index(col)]
            try:
                fval = float(val) if str(val) != '?' else None
            except (ValueError, TypeError):
                continue
            if fval is not None:
                score += fval  # higher ACC/PRED40 → better

        scores.append(score)

    n = len(rows)
    max_positions = max(1, int(n * 0.05))

    indexed_scores = [(i, s) for i, s in enumerate(scores)]
    indexed_scores.sort(key=lambda x: -x[1])

    best_positions = [idx for idx, _ in indexed_scores[:max_positions]]
    return best_positions


if __name__ == "__main__":
    import csv

    with open('Health-ClosedIssues0000.train.csv', 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        rows = []
        for row in reader:
            converted = []
            for cell in row:
                if cell.strip() == '?':
                    converted.append('?')
                else:
                    try:
                        converted.append(float(cell))
                    except ValueError:
                        converted.append(cell)
            rows.append(converted)

    result = remember(header, rows)
    print(f"Best {len(result)} of {len(rows)} setups:")
    for pos in result:
        print(f"  Row {pos}")
