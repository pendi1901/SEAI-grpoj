def fix(header, row, levers):
    minimise_cols = [c for c in header if c.endswith('-')]
    maximise_cols = [c for c in header if c.endswith('+')]

    col_idx = {name: i for i, name in enumerate(header)}

    def score(r):
        s = 0.0
        for col in minimise_cols:
            val = r[col_idx[col]]
            try:
                fval = float(val) if str(val) != '?' else None
            except (ValueError, TypeError):
                continue
            if fval is not None:
                s -= fval
        for col in maximise_cols:
            val = r[col_idx[col]]
            try:
                fval = float(val) if str(val) != '?' else None
            except (ValueError, TypeError):
                continue
            if fval is not None:
                s += fval
        return s

    orig_score = score(row)

    best_change = None
    best_score = -float('inf')

    # Try single-lever changes first
    for col, values in levers.items():
        idx = col_idx[col]
        if isinstance(values[0], tuple):  # numeric: (low, high)
            low, high = values
            try:
                orig_val = float(row[idx])
            except (ValueError, TypeError):
                continue
            for v in [low, high]:
                new_row = row[:]
                new_row[idx] = v
                sc = score(new_row)
                if sc > best_score:
                    best_score = sc
                    best_change = {col: v}
        else:  # symbolic: list of strings
            for v in values:
                new_row = row[:]
                new_row[idx] = v
                sc = score(new_row)
                if sc > best_score:
                    best_score = sc
                    best_change = {col: v}

    # If no single lever improved things, try pairs of levers
    if best_change is None or best_score <= orig_score:
        lever_names = list(levers.keys())
        for i in range(len(lever_names)):
            for j in range(i + 1, len(lever_names)):
                col_a, col_b = lever_names[i], lever_names[j]
                vals_a = levers[col_a]
                vals_b = levers[col_b]

                if isinstance(vals_a[0], tuple):
                    low_a, high_a = vals_a
                    for va in [low_a, high_a]:
                        for vb in vals_b:
                            new_row = row[:]
                            new_row[col_idx[col_a]] = va
                            new_row[col_idx[col_b]] = vb
                            sc = score(new_row)
                            if sc > best_score:
                                best_score = sc
                                best_change = {col_a: va, col_b: vb}
                else:
                    for va in vals_a:
                        for vb in vals_b:
                            new_row = row[:]
                            new_row[col_idx[col_a]] = va
                            new_row[col_idx[col_b]] = vb
                            sc = score(new_row)
                            if sc > best_score:
                                best_score = sc
                                best_change = {col_a: va, col_b: vb}

    return best_change
