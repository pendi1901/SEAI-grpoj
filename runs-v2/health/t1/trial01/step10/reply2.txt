def race(header, rows, label):
    """Compare three search strategies on up to 50 labelled rows."""
    minimise_cols = [c for c in header if c.endswith('-')]
    maximise_cols = [c for c in header if c.endswith('+')]
    col_idx = {name: i for i, name in enumerate(header)}

    def score(r):
        s = 0.0
        for col in minimise_cols:
            val = r[col_idx[col]]
            try:
                fval = float(val) if str(val) != '?' else None
            except (ValueError, TypeError):
                continue
            if fval is not None:
                s -= fval
        for col in maximise_cols:
            val = r[col_idx[col]]
            try:
                fval = float(val) if str(val) != '?' else None
            except (ValueError, TypeError):
                continue
            if fval is not None:
                s += fval
        return s

    # --- Strategy 1: random -------------------------------------------------
    def strategy_random():
        tried = set()
        best_pos = -1
        best_score = -float('inf')
        for _ in range(50):
            i = label.__call__(None) if False else None
            # We don't have a real random generator; pick rows at random indices.
            import random as _rand
            i = _rand.randint(0, len(rows) - 1)
            goals = label(i)
            sc = score(goals)
            if sc > best_score:
                best_score = sc
                best_pos = i
        return best_pos, best_score

    # --- Strategy 2: hill climbing ------------------------------------------
    def strategy_hill_climb():
        tried = set()
        best_pos = -1
        best_score = -float('inf')
        current_i = _rand.randint(0, len(rows) - 1)
        for _ in range(50):
            goals = label(current_i)
            sc = score(goals)
            if sc > best_score:
                best_score = sc
                best_pos = current_i
            # Try to improve by moving to a random neighbour
            import random as _rand2
            next_i = _rand2.randint(0, len(rows) - 1)
            if next_i != current_i and next_i not in tried:
                tried.add(next_i)
                current_i = next_i
        return best_pos, best_score
