    # Simple approach: for each test row, compute weighted average of training targets

import csv


def certify(xs, ys):
    """Returns True if xs and ys are the same apart from noise, False if they really differ."""
    n = len(xs)
    if n < 2 or n != len(ys):
        return False

    mean_x = sum(xs) / n
    mean_y = sum(ys) / n

    var_x = sum((x - mean_x) ** 2 for x in xs)
    var_y = sum((y - mean_y) ** 2 for y in ys)

    if var_x < 1e-10 or var_y < 1e-10:
        return True

    cov_xy = sum((xs[i] - mean_x) * (ys[i] - mean_y) for i in range(n))
    corr = cov_xy / ((var_x ** 0.5) * (var_y ** 0.5))

    # |corr| < 0.3 means no meaningful linear relationship → same apart from noise
    return abs(corr) < 0.3

    minimise_cols = [c for c in header if c.endswith('-')]

    def score(r):

            val = r[header.index(col)]
            try:
                fval = float(val) if str(val) != '?' else None
            except (ValueError, TypeError):
                continue
            if fval is not None:
                s -= fval
        for col in maximise_cols:
            val = r[header.index(col)]
            try:
                fval = float(val) if str(val) != '?' else None
            except (ValueError, TypeError):
                continue
            if fval is not None:
                s += fval
        return s

    current_score = score(row)
    n_seen = len(seen)

    # Need a few rows to estimate the typical range before deciding
    if n_seen >= 5:
        scores_seen = [score(r) for r in seen]
        mean_s = sum(scores_seen) / len(scores_seen)
        var_s = sum((s - mean_s) ** 2 for s in scores_seen)
        std_s = var_s ** 0.5
        # Stop when current row beats the typical range by a comfortable margin
        threshold = mean_s + 1.5 * std_s
        return current_score > threshold

    return False

def blame(header, rows):
    """Return names of input columns that best explain why the worst rows are bad."""
    minimise_cols = [c for c in header if c.endswith('-')]
    maximise_cols = [c for c in header if c.endswith('+')]

    col_idx = {name: i for i, name in enumerate(header)}

    # --- Score each row (higher is better) -------------------------------
    scores = []
    for row in rows:
        s = 0.0
        for col in minimise_cols:
            val = row[col_idx[col]]
            try:
                fval = float(val) if str(val) != '?' else None
            except (ValueError, TypeError):
                continue
            if fval is not None:
                s -= fval  # lower MRE → better
        for col in maximise_cols:
            val = row[col_idx[col]]
            try:
                fval = float(val) if str(val) != '?' else None
            except (ValueError, TypeError):
                continue
            if fval is not None:
                s += fval  # higher ACC/PRED40 → better
        scores.append(s)

    # --- Identify the worst rows -----------------------------------------
    n = len(rows)
    worst_frac = max(1, int(n * 0.05))
    worst_indices = sorted(range(n), key=lambda i: scores[i])[:worst_frac]
    worst_rows = [rows[i] for i in worst_indices]

    # --- Identify the best rows ------------------------------------------
    best_indices = sorted(range(n), key=lambda i: scores[i], reverse=True)[:worst_frac]
    best_rows = [rows[i] for i in best_indices]

    # --- For each input column, compare mean values between bad and good --
    input_cols = [c for c in header if not (c.endswith('-') or c.endswith('+'))]
    col_names_lower = {name.lower(): name for name in header}

    blame_results = []  # list of (abs_diff, col_name)

    for col in input_cols:
        idx = col_idx[col]
        bad_vals = []
        good_vals = []
        for row in worst_rows:
            val = row[idx]
            try:
                fval = float(val) if str(val) != '?' else None
            except (ValueError, TypeError):
                continue
            if fval is not None:
                bad_vals.append(fval)

        for row in best_rows:
            val = row[idx]
            try:
                fval = float(val) if str(val) != '?' else None
            except (ValueError, TypeError):
                continue
            if fval is not None:
                good_vals.append(fval)

        if len(bad_vals) < 2 or len(good_vals) < 2:
            continue

        mean_bad = sum(bad_vals) / len(bad_vals)
        mean_good = sum(good_vals) / len(good_vals)
        diff = abs(mean_bad - mean_good)

        # Only report columns where the difference is meaningful (>1% of range)
        if diff > 0.01:
            blame_results.append((diff, col))

    # Sort by magnitude and return top contributors
    blame_results.sort(key=lambda x: -x[0])
    return [col for _, col in blame_results[:5]]

    best_score = -float('inf')

    for i in range(len(rows)):
        if n_tried >= 50:

        goals = label(i)
        sc = score_from_goals(goals)
        if sc > best_score:
            best_score = sc
            best_pos = i
        n_tried += 1

    return best_pos

def fix(header, row, levers):
    minimise_cols = [c for c in header if c.endswith('-')]
    maximise_cols = [c for c in header if c.endswith('+')]

    col_idx = {name: i for i, name in enumerate(header)}

    def score(r):
        s = 0.0
        for col in minimise_cols:
            val = r[col_idx[col]]
            try:
                fval = float(val) if str(val) != '?' else None
            except (ValueError, TypeError):
                continue
            if fval is not None:
                s -= fval
        for col in maximise_cols:
            val = r[col_idx[col]]
            try:
                fval = float(val) if str(val) != '?' else None
            except (ValueError, TypeError):
                continue
            if fval is not None:
                s += fval
        return s

    orig_score = score(row)

    best_change = None
    best_score = -float('inf')

    # Try single-lever changes first
    for col, values in levers.items():
        idx = col_idx[col]
        if isinstance(values[0], tuple):  # numeric: (low, high)
            low, high = values
            try:
                orig_val = float(row[idx])
            except (ValueError, TypeError):
                continue
            for v in [low, high]:
                new_row = row[:]
                new_row[idx] = v
                sc = score(new_row)
                if sc > best_score:
                    best_score = sc
                    best_change = {col: v}
        else:  # symbolic: list of strings
            for v in values:
                new_row = row[:]
                new_row[idx] = v
                sc = score(new_row)
                if sc > best_score:
                    best_score = sc
                    best_change = {col: v}

    # If no single lever improved things, try pairs of levers
    if best_change is None or best_score <= orig_score:
        lever_names = list(levers.keys())
        for i in range(len(lever_names)):
            for j in range(i + 1, len(lever_names)):
                col_a, col_b = lever_names[i], lever_names[j]
                vals_a = levers[col_a]
                vals_b = levers[col_b]

                if isinstance(vals_a[0], tuple):
                    low_a, high_a = vals_a
                    for va in [low_a, high_a]:
                        for vb in vals_b:
                            new_row = row[:]
                            new_row[col_idx[col_a]] = va
                            new_row[col_idx[col_b]] = vb
                            sc = score(new_row)
                            if sc > best_score:
                                best_score = sc
                                best_change = {col_a: va, col_b: vb}
                else:
                    for va in vals_a:
                        for vb in vals_b:
                            new_row = row[:]
                            new_row[col_idx[col_a]] = va
                            new_row[col_idx[col_b]] = vb
                            sc = score(new_row)
                            if sc > best_score:
                                best_score = sc
                                best_change = {col_a: va, col_b: vb}

    return best_change

def spot(header, rows):
    minimise_cols = [c for c in header if c.endswith('-')]
    maximise_cols = [c for c in header if c.endswith('+')]

    col_idx = {name: i for i, name in enumerate(header)}
    input_cols = [c for c in header if not (c.endswith('-') or c.endswith('+'))]

    n = len(rows)
    if n == 0:
        return []

    # Compute per-column mean and std on numeric inputs
    col_stats = {}
    for col in input_cols:
        idx = col_idx[col]
        vals = []
        for row in rows:
            val = row[idx]
            try:
                fval = float(val) if str(val) != '?' else None
            except (ValueError, TypeError):
                continue
            if fval is not None:
                vals.append(fval)

        if len(vals) < 2:
            col_stats[col] = {'mean': 0.0, 'std': 1e9}
            continue

        mean_val = sum(vals) / len(vals)
        var_val = sum((v - mean_val) ** 2 for v in vals)
        std_val = var_val ** 0.5 if var_val > 1e-10 else 1e-9

        col_stats[col] = {'mean': mean_val, 'std': std_val}

    # Score each row: how far its input values stray from typical (z-score)
    suspicious_scores = []
    for i in range(n):
        score = 0.0
        for col in input_cols:
            idx = col_idx[col]
            val = rows[i][idx]
            try:
                fval = float(val) if str(val) != '?' else None
            except (ValueError, TypeError):
                continue

            if fval is not None and col_stats[col]['std'] > 1e-9:
                z = abs(fval - col_stats[col]['mean']) / col_stats[col]['std']
                score += max(0.0, z - 3.0)

        suspicious_scores.append((i, score))

    # Sort most suspicious first; return rows flagged as anomalous (score > 0)
    suspicious_scores.sort(key=lambda x: -x[1])
    result = [pos for pos, s in suspicious_scores if s > 0]
    return result[:5]
