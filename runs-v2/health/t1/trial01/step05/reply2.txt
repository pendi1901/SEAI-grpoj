def blame(header, rows):
    """Return names of input columns that best explain why the worst rows are bad."""
    minimise_cols = [c for c in header if c.endswith('-')]
    maximise_cols = [c for c in header if c.endswith('+')]

    col_idx = {name: i for i, name in enumerate(header)}

    # --- Score each row (higher is better) -------------------------------
    scores = []
    for row in rows:
        s = 0.0
        for col in minimise_cols:
            val = row[col_idx[col]]
            try:
                fval = float(val) if str(val) != '?' else None
            except (ValueError, TypeError):
                continue
            if fval is not None:
                s -= fval  # lower MRE → better
        for col in maximise_cols:
            val = row[col_idx[col]]
            try:
                fval = float(val) if str(val) != '?' else None
            except (ValueError, TypeError):
                continue
            if fval is not None:
                s += fval  # higher ACC/PRED40 → better
        scores.append(s)

    # --- Identify the worst rows -----------------------------------------
    n = len(rows)
    worst_frac = max(1, int(n * 0.05))
    worst_indices = sorted(range(n), key=lambda i: scores[i])[:worst_frac]
    worst_rows = [rows[i] for i in worst_indices]

    # --- Identify the best rows ------------------------------------------
    best_indices = sorted(range(n), key=lambda i: scores[i], reverse=True)[:worst_frac]
    best_rows = [rows[i] for i in best_indices]

    # --- For each input column, compare mean values between bad and good --
    input_cols = [c for c in header if not (c.endswith('-') or c.endswith('+'))]
    col_names_lower = {name.lower(): name for name in header}

    blame_results = []  # list of (abs_diff, col_name)

    for col in input_cols:
        idx = col_idx[col]
        bad_vals = []
        good_vals = []
        for row in worst_rows:
            val = row[idx]
            try:
                fval = float(val) if str(val) != '?' else None
            except (ValueError, TypeError):
                continue
            if fval is not None:
                bad_vals.append(fval)

        for row in best_rows:
            val = row[idx]
            try:
                fval = float(val) if str(val) != '?' else None
            except (ValueError, TypeError):
                continue
            if fval is not None:
                good_vals.append(fval)

        if len(bad_vals) < 2 or len(good_vals) < 2:
            continue

        mean_bad = sum(bad_vals) / len(bad_vals)
        mean_good = sum(good_vals) / len(good_vals)
        diff = abs(mean_bad - mean_good)

        # Only report columns where the difference is meaningful (>1% of range)
        if diff > 0.01:
            blame_results.append((diff, col))

    # Sort by magnitude and return top contributors
    blame_results.sort(key=lambda x: -x[0])
    return [col for _, col in blame_results[:5]]
