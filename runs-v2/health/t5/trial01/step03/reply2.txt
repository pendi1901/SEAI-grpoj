import ezr


def certify(xs, ys):
    """Return True if xs and ys agree up to noise; False otherwise."""
    n = max(len(xs), len(ys))
    if n == 0:
        return True
    # Use a tolerance that scales with the data range.
    diffs = [abs(xs[i] - ys[i]) for i in range(n)]
    avg_diff = sum(diffs) / n
    max_diff = max(diffs)
    # If average gap is small relative to the spread, treat as noise.
    if max_diff < 1e-6:
        return True
    # Relative tolerance based on observed variation in xs and ys.
    var_x = sum((x - avg(xs)) ** 2 for x in xs) / n if n else 0
    var_y = sum((y - avg(ys)) ** 2 for y in ys) / n if n else 0
    scale = (var_x + var_y) ** 0.5
    if scale < 1e-32:
        return True
    tol = 4 * scale
    return all(d <= tol for d in diffs)


if __name__ == "__main__":
    with open("Health-ClosedIssues0000.train.csv", encoding="utf-8-sig") as f:
        lines = [line.strip() for line in f if line.strip()]
    header = list(lines[0].split(","))
    rows = [[ezr.atom(s) for s in line.split(",")] for line in lines[1:]]

    # Identify goal columns and input columns.
    goals = [c for c in header if c[-1] in "+-"]
    inputs = [c for i, c in enumerate(header) if c not in goals and c[-1] != "X"]

    # For each row compute ydist (distance from best).
    tbl = ezr.Tbl(rows)
    ranker_fn = ezr.ranker(tbl)
    wins_fn = ezr.wins(tbl)

    scores = [ranker_fn(r) for r in rows]
    score_best = min(scores)
    score_avg  = sum(scores) / len(scores)

    # Group rows into "big" (>=50%) and "small" (<50%) setups.
    big_rows = [r for i, r in enumerate(rows) if scores[i] >= score_best - (score_avg - score_best) * 0.3]
    small_rows = [r for r in rows if r not in big_rows]

    # Compare the two groups on their goal values.
    def extract_goals(row):
        return [row[header.index(c)] for c in goals]

    big_vals = [extract_goals(r) for r in big_rows]
    small_vals = [extract_goals(r) for r in small_rows]

    # Flatten and compare numerically (skip non-numeric cells).
    def to_floats(vals):
        out = []
        for v in vals:
            if isinstance(v, float):
                out.append(v)
        return out

    big_f = to_floats(big_vals)
    small_f = to_floats(small_vals)

    # If both groups have data, check whether they agree.
    if len(big_f) > 0 and len(small_f) > 0:
        result = certify(big_f, small_f)
    else:
        result = True

    print(f"Big vs Small agreement: {result}")
