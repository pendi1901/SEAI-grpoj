import ezr


def blame(header, rows):
    """Return a list of strings naming the input columns that best explain
    why the worst rows are bad."""
    goals = [c for c in header if c[-1] in "+-"]
    inputs = [c for i, c in enumerate(header) if c not in goals and c[-1] != "X"]

    tbl = ezr.Tbl(rows)

    def ydist_row(row):
        return ezr.ydist(tbl, row)

    scores = [ydist_row(r) for r in rows]
    best_score = min(scores)
    worst_pct = 0.25
    threshold = best_score + (max(scores) - best_score) * worst_pct
    worst_rows = [r for i, r in enumerate(rows) if scores[i] >= threshold]

    results = []
    for col_name in inputs:
        col_idx = header.index(col_name)
        col_vals = [r[col_idx] if isinstance(r[col_idx], float) else 0.0 for r in rows]

        n = len(rows)
        mean_x = sum(col_vals) / n
        mean_y = sum(scores) / n

        num = sum((col_vals[i] - mean_x) * (scores[i] - mean_y) for i in range(n))
        den_x = (sum((v - mean_x) ** 2 for v in col_vals)) ** 0.5
        den_y = (sum((s - mean_y) ** 2 for s in scores)) ** 0.5

        if den_x < 1e-32 or den_y < 1e-32:
            corr = 0.0
        else:
            corr = num / (den_x * den_y)

        results.append((col_name, corr))

    results.sort(key=lambda x: abs(x[1]), reverse=True)

    return [f"{name} correlates with poor ydist ({corr:+.3f})" for name, corr in results[:5]]


if __name__ == "__main__":
    with open("Health-ClosedIssues0000.train.csv", encoding="utf-8-sig") as f:
        lines = [line.strip() for line in f if line.strip()]
    header = list(lines[0].split(","))
    rows = [[ezr.atom(s) for s in line.split(",")] for line in lines[1:]]

    bad_cols = blame(header, rows)
    print(f"Bad setups miss because of: {', '.join(bad_cols)}")
