import ezr


def flow(header, row, seen):
    tbl = ezr.Tbl([header] + seen + [row])
    ranker_fn = ezr.ranker(tbl)
    cur_score = ranker_fn(row)

    if not seen:
        return True

    best_score = min(ranker_fn(r) for r in seen)

    if cur_score <= best_score * 1.05:
        return True
    return False


if __name__ == "__main__":
    with open("Health-ClosedIssues0000.train.csv", encoding="utf-8-sig") as f:
        lines = [line.strip() for line in f if line.strip()]
    header = list(lines[0].split(","))
    rows = [[ezr.atom(s) for s in line.split(",")] for line in lines[1:]]

    seen = []
    for row in rows:
        if flow(header, row, seen):
            print(f"Chose row {len(seen)}")
            break
        seen.append(row)
    else:
        print("No good row found")
