import ezr


def justify(header, rows, row):
    """Return (p, reason) where p is the fraction of complete rows that `row` beats."""
    tbl = ezr.Tbl(rows)
    rank_fn = ezr.ranker(tbl)
    scores = [rank_fn(r) for r in rows]
    n = len(scores)

    node = ezr.tree(tbl, tbl.rows)

    def estimate(row):
        cur = node
        while cur.kids:
            if cur.go(row):
                cur = cur.kids[0]
            else:
                cur = cur.kids[1]
        return cur.mu

    est = estimate(row)
    better = sum(1 for s in scores if s >= est - 1e-32)
    p = better / n
    reason = f"Better than {p:.0%} of the {n} reference rows."
    return (p, reason)


if __name__ == "__main__":
    with open("Health-ClosedIssues0000.train.csv", encoding="utf-8-sig") as f:
        lines = [line.strip() for line in f if line.strip()]
    header = list(lines[0].split(","))
    rows = [[ezr.atom(s) for s in line.split(",")] for line in lines[1:]]

    row = rows[-1]
    p, reason = justify(header, rows, row)
    print(f"{reason} ({p:.4f})")
