def certify(xs, ys):
    """Return True if xs and ys are the same apart from noise."""
    n_x = len(xs) or 1
    n_y = len(ys) or 1
    mean_x = sum(xs) / n_x
    mean_y = sum(ys) / n_y

    def std(vals):
        m = sum(vals) / len(vals)
        return (sum((v - m) ** 2 for v in vals) / len(vals)) ** 0.5

    sx = std(xs) if xs else 1.0
    sy = std(ys) if ys else 1.0

    threshold = (sx + sy) * 2.0
    return abs(mean_x - mean_y) <= threshold

    # First row: nothing to compare against yet.

        return False

    # Best goals so far (minimise → low, maximise → high).

    for col in minimise + maximise:
        vals = [_to_float(r[header.index(col)]) for r in seen]
        best[col] = min(vals) if col in minimise else max(vals)

    cur = {col: _to_float(row[header.index(col)]) for col in minimise + maximise}

    # Within tolerance of the best seen so far.
    tol = 0.15
    return all(
        (cur[col] - best[col]) <= tol if col in minimise else (best[col] - cur[col]) <= tol
        for col in minimise + maximise

        diff = abs(avg_worst - avg_all)

def fix(header, row, levers):
    """Return {column: new_value} for the single best lever change."""
    minimise = [c for c in header if c.endswith('-')]
    maximise = [c for c in header if c.endswith('+')]

    # Current goal values (handle ?)
    cur_goals = {}
    for col in minimise + maximise:
        v = _to_float(row[header.index(col)])
        cur_goals[col] = v

    def score(goals):
        s = sum(goals[c] for c in minimise) - sum(goals[c] for c in maximise)
        return s

    # Best possible score from any real row (excluding this one).
    best_score = float('inf')
    n_rows = len(row)  # placeholder; we'll read the file properly below
    # We don't have access to all rows here, so use a tolerance-based approach:
    # try each lever change and pick the one that improves score most.

    def fill_row(cells):
        out = list(cells)
        for i, cell in enumerate(out):
            if cell == '?':
                out[i] = 0.0
        return out

    row_filled = fill_row(row)

    # For each lever column, try each allowed value and pick the best single change.
    best_change = None
    best_improvement = -float('inf')

    for col, values in levers.items():
        idx = header.index(col)
        cur_val = _to_float(row[idx])
        if isinstance(values[0], tuple):
            # numeric: (low, high) — try both endpoints and midpoint
            lo, hi = values
            candidates = [lo, hi]
            mid = (lo + hi) / 2.0
            candidates.append(mid)
        else:
            candidates = list(values)

        for val in candidates:
            new_row = fill_row(list(row_filled))
            new_row[idx] = str(val) if isinstance(val, float) else val
            new_goals = {}
            for c in minimise + maximise:
                v = _to_float(new_row[header.index(c)])
                new_goals[c] = v

            s = score(new_goals)
            improvement = s - score(cur_goals)
            if improvement > best_improvement:
                best_improvement = improvement
                best_change = {col: val}

    # If no lever was given, we still need to change at least one — pick the column

                lo = 0.0

from step01 import _to_float


def spot(header, rows):
    minimise = [c for c in header if c.endswith('-')]
    maximise = [c for c in header if c.endswith('+')]
    goal_cols = set(minimise + maximise)

    # For each row, compute a "goal score" — lower is better.
    def goal_score(row):
        s = 0.0
        for col in minimise:
            v = _to_float(row[header.index(col)])
            s += v
        for col in maximise:
            v = _to_float(row[header.index(col)])
            s -= v
        return s

    # For each row, compute a "feature score" — how well the non-goal numeric columns align.
    def feature_score(row):
        feat_cols = [c for c in header if c not in goal_cols and _to_float("") != 0.0]
        if not feat_cols:
            return 0.0
        s = 0.0
        n = len(feat_cols)
        for col in feat_cols:
            v = _to_float(row[header.index(col)])
            s += v
        return s / n

    # Compute scores for all rows.
    row_scores = []
    for i, row in enumerate(rows):
        gs = goal_score(row)
        fs = feature_score(row)
        row_scores.append((gs, fs, i))

    # Find the best goal score among real-looking rows (no "?" in goals).
    clean_rows = [i for i, row in enumerate(rows) if all(
        _to_float(row[header.index(c)]) != 0.0 or c not in minimise + maximise
        for c in minimise + maximise
    )]

    # Actually, let's be more precise: a clean row has no "?" in goal columns.
    def is_clean(i):
        return all(
            row[header.index(c)] != '?'
            for row in [rows[i]]
            for c in minimise + maximise
        )

    clean_indices = [i for i in range(len(rows)) if is_clean(i)]

    # Best goal score among clean rows.
    best_goal = min(goal_score(rows[i]) for i in clean_indices) if clean_indices else 0.0

    # Now, find rows where the feature score is unusually low compared to others — these are likely "failed runs".
    feat_scores_clean = [feature_score(rows[i]) for i in clean_indices]
    if not feat_scores_clean:
        return []

    mean_feat = sum(feat_scores_clean) / len(feat_scores_clean)
    std_feat = (sum((v - mean_feat) ** 2 for v in feat_scores_clean) / len(feat_scores_clean)) ** 0.5
    if std_feat == 0:
        std_feat = 1.0

    # Rows with feature score more than 2 standard deviations below the mean are suspicious.
    threshold = mean_feat - 2 * std_feat
    suspicious = [i for i in clean_indices if feature_score(rows[i]) < threshold]

    # Also check rows that have "?" in any column — these might be failed runs too.
    question_mark_rows = []
    for i, row in enumerate(rows):
        if '?' in row:
            question_mark_rows.append(i)

    all_suspicious = suspicious + question_mark_rows
    all_suspicious.sort(key=lambda i: feature_score(rows[i]) if is_clean(i) else float('inf'))

    return all_suspicious[:10]
