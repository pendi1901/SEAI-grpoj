from step01 import _to_float, remember


def blame(header, rows):
    """Return names of input columns (not goals) that explain why worst rows are bad."""
    minimise = [c for c in header if c.endswith('-')]
    maximise = [c for c in header if c.endswith('+')]
    goal_cols = minimise + maximise

    # Score each row: low is good for minimise, high is good for maximise.
    scored = []
    for i, row in enumerate(rows):
        score = 0.0
        for col in minimise:
            v = _to_float(row[header.index(col)])
            score += v
        for col in maximise:
            v = _to_float(row[header.index(col)])
            score -= v
        scored.append((score, i))

    scored.sort()
    worst_n = max(1, int(len(scored) * 0.15))
    worst_indices = [pos for _, pos in scored[:worst_n]]

    # Input columns: numeric non-goal columns
    input_cols = []
    for col in header:
        if col not in goal_cols and _to_float("") != 0.0:
            idx = header.index(col)
            sample_val = _to_float(rows[0][idx])
            if isinstance(sample_val, float):
                input_cols.append((col, idx))

    # For each input column, compare average in worst vs all rows.
    blame_results = []
    for col_name, col_idx in input_cols:
        vals_all = [_to_float(row[col_idx]) for row in rows]
        vals_worst = [_to_float(rows[pos][col_idx]) for pos in worst_indices]

        avg_all = sum(vals_all) / len(vals_all) if vals_all else 0.0
        avg_worst = sum(vals_worst) / len(vals_worst) if vals_worst else 0.0

        # The column "blames" bad rows if its value differs significantly in worst rows.
        diff = abs(avg_worst - avg_all)
        blame_results.append((diff, col_name))

    blame_results.sort(reverse=True)
    return [col for _, col in blame_results[:5]]


if __name__ == "__main__":
    import csv

    with open("Health-ClosedIssues0000.train.csv", "r") as f:
        reader = csv.reader(f)
        header = next(reader)
        rows = []
        for row in reader:
            rows.append(row)

    bad_cols = blame(header, rows)
    print(f"The worst rows are blamed on these input columns: {bad_cols}")
