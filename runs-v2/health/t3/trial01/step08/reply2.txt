import pandas as pd
import numpy as np


def fix(header, row, levers):
    goal_cols = [c for c in header if c.endswith('-') or c.endswith('+')]
    input_cols = [c for c in header if not c.endswith('-') and not c.endswith('+')]

    # Build a DataFrame from the training rows to get real goal values.
    df = pd.read_csv("Health-ClosedIssues0000.train.csv")
    train_rows = [list(r) for r in df.values]

    # Compute best/worst per goal across all training rows.
    best = {}
    worst = {}
    for col in goal_cols:
        vals = [float(r[header.index(col)]) for r in train_rows]
        if col.endswith('-'):
            best[col], worst[col] = min(vals), max(vals)
        else:
            best[col], worst[col] = max(vals), min(vals)

    # Score a row by summing normalised goal values.
    def score_row(r):
        s = 0.0
        for col in goal_cols:
            v = float(r[header.index(col)])
            span = best[col] - worst[col]
            if span > 1e-9:
                normed = (v - worst[col]) / span
            else:
                normed = 0.5
            s += normed
        return s

    # Score the bad row.
    bad_score = score_row(row)

    # Try each lever in order of preference: pick the value that gives the best score.
    for col, values in levers.items():
        if col not in goal_cols and col not in input_cols:
            continue
        # For numeric levers, try a few candidate values including extremes.
        is_numeric = True
        for v in values:
            try:
                float(v)
            except (ValueError, TypeError):
                is_numeric = False
                break

        if is_numeric:
            candidates = list(values)
            # Also include the current value and a midpoint.
            cur = row[header.index(col)]
            try:
                cur_f = float(cur)
                mid = (cur_f + values[-1]) / 2 if len(values) > 0 else cur_f
                candidates.append(mid)
            except (ValueError, TypeError):
                pass

        # For symbolic levers, use all provided values.
        else:
            candidates = list(values)

        best_score_for_col = -np.inf
        best_val = None
        for cand in candidates:
            trial_row = list(row)
            trial_row[header.index(col)] = str(cand) if not is_numeric else float(cand)
            s = score_row(trial_row)
            if s > best_score_for_col:
                best_score_for_col = s
                best_val = cand

        # If changing this lever improves the score, use it.
        if best_score_for_col > bad_score + 1e-9:
            row[header.index(col)] = best_val
            bad_score = best_score_for_col
            continue

    return {col: row[header.index(col)] for col in levers}


if __name__ == "__main__":
    df = pd.read_csv("Health-ClosedIssues0000.train.csv")
    header = list(df.columns)
    rows = [list(r) for r in df.values]

    # Use the worst row as the "bad" setup.
    goal_cols = [c for c in header if c.endswith('-') or c.endswith('+')]
    scores = []
    for i, row in enumerate(rows):
        s = 0.0
        for col in goal_cols:
            v = float(row[header.index(col)])
            span = best[col] - worst[col]
            if span > 1e-9:
                normed = (v - worst[col]) / span
            else:
                normed = 0.5
            s += normed
        scores.append(s)

    worst_idx = int(np.argmin(scores))
    bad_row = rows[worst_idx]

    # Determine levers and their allowed values from the training data.
    input_cols = [c for c in header if not c.endswith('-') and not c.endswith('+')]
    levers = {}
    for col in input_cols:
        vals = []
        for r in rows:
            v = r[header.index(col)]
            try:
                fv = float(v)
                vals.append(fv)
            except (ValueError, TypeError):
                pass

        if len(vals) > 0:
            low = min(vals)
            high = max(vals)
            levers[col] = [low, high]
        else:
            # Fallback to unique string values.
            uniq = sorted(set(r[header.index(col)] for r in rows))
            levers[col] = uniq

    result = fix(header, bad_row, levers)
    print(f"Fixed setup #{worst_idx + 1}: {result}")
