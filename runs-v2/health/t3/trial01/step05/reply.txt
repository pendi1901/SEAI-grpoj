import numpy as np


def blame(header, rows):
    """Return input column names that best explain why the worst setups are bad."""
    goal_cols = [c for c in header if c.endswith('-') or c.endswith('+')]
    input_cols = [c for c in header if not c.endswith('-') and not c.endswith('+')]

    # Compute per-goal best / worst across all rows.
    best = {}
    worst = {}
    for i, row in enumerate(rows):
        for col in goal_cols:
            v = float(row[header.index(col)])
            if col not in best:
                best[col] = v
                worst[col] = v
            else:
                if col.endswith('-'):
                    best[col] = min(best[col], v)
                    worst[col] = max(worst[col], v)
                else:
                    best[col] = max(best[col], v)
                    worst[col] = min(worst[col], v)

    # Normalised score for each row (sum of per-goal normalised values).
    scores = []
    for i, row in enumerate(rows):
        s = 0.0
        for col in goal_cols:
            span = best[col] - worst[col]
            if span > 1e-9:
                normed = (float(row[header.index(col)]) - worst[col]) / span
            else:
                normed = 0.5
            s += normed
        scores.append(s)

    # Fit a linear model on the input columns to predict score.
    X = np.array([[float(row[header.index(c)]) for c in input_cols] for row in rows])
    y = np.array(scores)

    from sklearn.linear_model import LinearRegression
    model = LinearRegression()
    model.fit(X, y)

    # Rank features by absolute coefficient magnitude.
    coef_abs = sorted(enumerate(model.coef_), key=lambda x: abs(x[1]), reverse=True)
    n_top = max(1, len(input_cols) // 2)
    return [input_cols[i] for i, _ in coef_abs[:n_top]]


if __name__ == "__main__":
    import pandas as pd

    df = pd.read_csv("Health-ClosedIssues0000.train.csv")
    header = list(df.columns)
    rows = [list(r) for r in df.values]

    bad_cols = blame(header, rows)
    print(f"Top {len(bad_cols)} input columns driving poor setups: {bad_cols}")
