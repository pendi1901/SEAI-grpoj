import pandas as pd
import numpy as np


def certify(xs, ys):
    xs = np.array(xs)
    ys = np.array(ys)

    n1, n2 = len(xs), len(ys)
    mean_diff = abs(float(np.mean(xs)) - float(np.mean(ys)))

    var1 = float(np.var(xs, ddof=1)) if n1 > 1 else 0.0
    var2 = float(np.var(ys, ddof=1)) if n2 > 1 else 0.0

    se_sq = var1 / n1 + var2 / n2
    se = np.sqrt(se_sq) if se_sq > 0 else 0.0

    if se == 0:
        return True

    z = mean_diff / se
    return bool(z < 1.96)


if __name__ == "__main__":
    df = pd.read_csv("Health-ClosedIssues0000.train.csv")
    header = list(df.columns)
    rows = [list(r) for r in df.values]

    goal_cols = {}
    for c in header:
        if c.endswith('-'):
            goal_cols[c] = 'min'
        elif c.endswith('+'):
            goal_cols[c] = 'max'

    best = {c: float(df[c].min()) for c, d in goal_cols.items() if d == 'min'}
    best.update({c: float(df[c].max()) for c, d in goal_cols.items() if d == 'max'})
    worst = {c: float(df[c].max()) for c, d in goal_cols.items() if d == 'min'}
    worst.update({c: float(df[c].min()) for c, d in goal_cols.items() if d == 'max'})

    scores = []
    for i in range(len(df)):
        s = 0.0
        for col, direction in goal_cols.items():
            v = float(df.loc[i, col])
            span = best[col] - worst[col]
            if span > 1e-9:
                normed = (v - worst[col]) / span
            else:
                normed = 0.5
            s += normed
        scores.append(s)

    n_top = max(1, int(len(scores) * 0.05))
    top_idx = np.argsort(scores)[::-1][:n_top]
    best_positions = sorted(top_idx.tolist())

    big_indices = [i for i in range(len(df)) if float(df.loc[i, 'N_estimators']) >= 50]
    small_indices = [i for i in range(len(df)) if float(df.loc[i, 'N_estimators']) < 50]

    xs = [float(df.loc[i, 'MRE-']) for i in big_indices]
    ys = [float(df.loc[i, 'MRE-']) for i in small_indices]

    result = certify(xs, ys)

    print(f"Big setups (N_estimators >= 50): mean MRE- = {np.mean(xs):.4f}")
    print(f"Small setups (N_estimators < 50): mean MRE- = {np.mean(ys):.4f}")
    print(f"Their difference is {'real' if not result else 'noise'}: certify returned {result}")
