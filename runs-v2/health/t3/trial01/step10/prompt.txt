def certify(xs, ys):
    xs = np.array(xs)
    ys = np.array(ys)

    n1, n2 = len(xs), len(ys)
    mean_diff = abs(float(np.mean(xs)) - float(np.mean(ys)))

    var1 = float(np.var(xs, ddof=1)) if n1 > 1 else 0.0
    var2 = float(np.var(ys, ddof=1)) if n2 > 1 else 0.0

    se_sq = var1 / n1 + var2 / n2
    se = np.sqrt(se_sq) if se_sq > 0 else 0.0

    if se == 0:
        return True

    z = mean_diff / se
    return bool(z < 1.96)

def flow(header, row, seen):
    """Decide whether to stop on `row` given all previously seen rows."""
    goal_cols = [c for c in header if c.endswith('-') or c.endswith('+')]

    # Compute best/worst from the union of seen + current row.
    candidates = seen + [[row]]
    df = pd.DataFrame(candidates, columns=header)

    best = {}
    worst = {}
    for c in goal_cols:
        if c.endswith('-'):
            best[c] = float(df[c].min())
            worst[c] = float(df[c].max())
        else:  # ends with '+'
            best[c] = float(df[c].max())
            worst[c] = float(df[c].min())

    span = {c: best[c] - worst[c] for c in goal_cols}

    def score(r):
        s = 0.0
        for col, d in zip(goal_cols, ['min' if c.endswith('-') else 'max' for c in goal_cols]):
            v = float(r[header.index(col)])
            sp = span[col]
            if sp > 1e-9:
                normed = (v - worst[col]) / sp
            else:
                normed = 0.5
            s += normed
        return s

    current_score = score(row)
    best_so_far = max(score(r) for r in seen) if seen else current_score

    # Stop as soon as the current row is at least as good as any previous one,
    # and within a tight margin of the theoretical optimum.
    threshold = 0.98 * (best_so_far - worst[goal_cols[0]]) + worst[goal_cols[0]]
    return bool(current_score >= best_so_far)

import numpy as np


def blame(header, rows):
    """Return input column names that best explain why the worst setups are bad."""
    goal_cols = [c for c in header if c.endswith('-') or c.endswith('+')]
    input_cols = [c for c in header if not c.endswith('-') and not c.endswith('+')]

    # Compute per-goal best / worst across all rows.
    best = {}
    worst = {}
    for i, row in enumerate(rows):
        for col in goal_cols:
            v = float(row[header.index(col)])
            if col not in best:
                best[col] = v
                worst[col] = v
            else:
                if col.endswith('-'):
                    best[col] = min(best[col], v)
                    worst[col] = max(worst[col], v)
                else:
                    best[col] = max(best[col], v)
                    worst[col] = min(worst[col], v)

    # Normalised score for each row (sum of per-goal normalised values).
    scores = []
    for i, row in enumerate(rows):
        s = 0.0
        for col in goal_cols:
            span = best[col] - worst[col]
            if span > 1e-9:
                normed = (float(row[header.index(col)]) - worst[col]) / span
            else:
                normed = 0.5
            s += normed
        scores.append(s)

    # Fit a linear model on the input columns to predict score.
    X = np.array([[float(row[header.index(c)]) for c in input_cols] for row in rows])
    y = np.array(scores)

    from sklearn.linear_model import LinearRegression
    model = LinearRegression()
    model.fit(X, y)

    # Rank features by absolute coefficient magnitude.
    coef_abs = sorted(enumerate(model.coef_), key=lambda x: abs(x[1]), reverse=True)
    n_top = max(1, len(input_cols) // 2)
    return [input_cols[i] for i, _ in coef_abs[:n_top]]

def justify(header, rows, row):
    goal_cols = [c for c in header if c.endswith('-') or c.endswith('+')]
    input_cols = [c for c in header if not c.endswith('-') and not c.endswith('+')]

    # Build training data from the complete rows (which have real goals)
    X_train = np.array([[float(row[input_cols.index(c)]) for c in input_cols] for row in rows])

    # Train a separate model per goal column to predict what the test row's goals would be
    models = {}
    for col in goal_cols:
        y_col = [float(r[header.index(col)]) for r in rows]
        m = LinearRegression()
        m.fit(X_train, np.array(y_col))
        models[col] = m

    # Predict the hidden goals of the test row
    x_test = [float(row[input_cols.index(c)]) for c in input_cols]
    predicted_goals = {col: float(models[col].predict([x_test])[0]) for col in goal_cols}

    # Compute best / worst per goal across all training rows (for normalisation)
    best = {}
    worst = {}
    for col in goal_cols:
        vals = [float(r[header.index(col)]) for r in rows]
        if col.endswith('-'):
            best[col], worst[col] = min(vals), max(vals)
        else:
            best[col], worst[col] = max(vals), min(vals)

    # Score a row by summing normalised goal values (0.5 when span is zero)
    def score(goals):
        s = 0.0
        for col in goal_cols:
            v = goals[col]
            span = best[col] - worst[col]
            if span > 1e-9:
                normed = (v - worst[col]) / span
            else:
                normed = 0.5
            s += normed
        return s

    # Score each training row and the test row
    train_scores = [score({col: float(r[header.index(col)]) for col in goal_cols}) for r in rows]
    test_score = score(predicted_goals)

    # Fraction of training rows that are worse than the test row
    n_better = sum(1 for s in train_scores if s < test_score)
    p = n_better / len(rows)

    reason = f"This setup beats {p:.0%} of all other setups."
    return (p, reason)

def fix(header, row, levers):
    goal_cols = [c for c in header if c.endswith('-') or c.endswith('+')]
    input_cols = [c for c in header if not c.endswith('-') and not c.endswith('+')]

    # Build a DataFrame from the training rows to get real goal values.
    df = pd.read_csv("Health-ClosedIssues0000.train.csv")
    train_rows = [list(r) for r in df.values]

    # Compute best/worst per goal across all training rows.
    best = {}
    worst = {}
    for col in goal_cols:
        vals = [float(r[header.index(col)]) for r in train_rows]
        if col.endswith('-'):
            best[col], worst[col] = min(vals), max(vals)
        else:
            best[col], worst[col] = max(vals), min(vals)

    # Score a row by summing normalised goal values.
    def score_row(r):
        s = 0.0
        for col in goal_cols:
            v = float(r[header.index(col)])
            span = best[col] - worst[col]
            if span > 1e-9:
                normed = (v - worst[col]) / span
            else:
                normed = 0.5
            s += normed
        return s

    # Score the bad row.
    bad_score = score_row(row)

    # Try each lever in order of preference: pick the value that gives the best score.
    for col, values in levers.items():
        if col not in goal_cols and col not in input_cols:
            continue
        # For numeric levers, try a few candidate values including extremes.
        is_numeric = True
        for v in values:
            try:
                float(v)
            except (ValueError, TypeError):
                is_numeric = False
                break

        if is_numeric:
            candidates = list(values)
            # Also include the current value and a midpoint.
            cur = row[header.index(col)]
            try:
                cur_f = float(cur)
                mid = (cur_f + values[-1]) / 2 if len(values) > 0 else cur_f
                candidates.append(mid)
            except (ValueError, TypeError):
                pass

        # For symbolic levers, use all provided values.
        else:
            candidates = list(values)

        best_score_for_col = -np.inf
        best_val = None
        for cand in candidates:
            trial_row = list(row)
            trial_row[header.index(col)] = str(cand) if not is_numeric else float(cand)
            s = score_row(trial_row)
            if s > best_score_for_col:
                best_score_for_col = s
                best_val = cand

        # If changing this lever improves the score, use it.
        if best_score_for_col > bad_score + 1e-9:
            row[header.index(col)] = best_val
            bad_score = best_score_for_col
            continue

    return {col: row[header.index(col)] for col in levers}
