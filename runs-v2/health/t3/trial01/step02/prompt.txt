import pandas as pd
import numpy as np


def remember(header, rows):
    df = pd.DataFrame(rows, columns=header)

    goal_cols = {}
    for c in header:
        if c.endswith('-'):
            goal_cols[c] = 'min'
        elif c.endswith('+'):
            goal_cols[c] = 'max'

    best = {c: float(df[c].min()) for c, d in goal_cols.items() if d == 'min'}
    best.update({c: float(df[c].max()) for c, d in goal_cols.items() if d == 'max'})
    worst = {c: float(df[c].max()) for c, d in goal_cols.items() if d == 'min'}
    worst.update({c: float(df[c].min()) for c, d in goal_cols.items() if d == 'max'})

    scores = []
    for i in range(len(df)):
        s = 0.0
        for col, direction in goal_cols.items():
            v = float(df.loc[i, col])
            span = best[col] - worst[col]
            if span > 1e-9:
                normed = (v - worst[col]) / span
            else:
                normed = 0.5
            s += normed
        scores.append(s)

    n_top = max(1, int(len(scores) * 0.05))
    top_idx = np.argsort(scores)[::-1][:n_top]
    return sorted(top_idx.tolist())
