import numpy as np
from sklearn.linear_model import LinearRegression


def choose(header, rows, label):
    goal_cols = [c for c in header if c.endswith('-') or c.endswith('+')]
    input_cols = [c for c in header if not c.endswith('-') and not c.endswith('+')]

    n_rows = len(rows)
    X_train = np.array([[float(row[input_cols.index(c)]) for c in input_cols] for row in rows])

    models = {}
    for col in goal_cols:
        y_col = [float(r[header.index(col)]) for r in rows]
        m = LinearRegression()
        m.fit(X_train, np.array(y_col))
        models[col] = m

    def predict_score(row):
        x = [float(row[input_cols.index(c)]) for c in input_cols]
        pred_goals = {col: float(models[col].predict([x])[0]) for col in goal_cols}
        best = {}
        worst = {}
        for col in goal_cols:
            vals = [float(r[header.index(col)]) for r in rows]
            if col.endswith('-'):
                best[col], worst[col] = min(vals), max(vals)
            else:
                best[col], worst[col] = max(vals), min(vals)
        s = 0.0
        for col in goal_cols:
            v = pred_goals[col]
            span = best[col] - worst[col]
            if span > 1e-9:
                normed = (v - worst[col]) / span
            else:
                normed = 0.5
            s += normed
        return s

    scores = [predict_score(row) for row in rows]
    top_k = min(50, n_rows)
    top_idx = np.argsort(scores)[::-1][:top_k].tolist()

    best_row_idx = None
    best_real_score = -np.inf
    for i in top_idx:
        real_goals = label(i)
        s = 0.0
        for col in goal_cols:
            v = float(real_goals[header.index(col)])
            span = best[col] - worst[col]
            if span > 1e-9:
                normed = (v - worst[col]) / span
            else:
                normed = 0.5
            s += normed
        if s >= best_real_score:
            best_real_score = s
            best_row_idx = i

    return int(best_row_idx)


if __name__ == "__main__":
    import pandas as pd

    df = pd.read_csv("Health-ClosedIssues0000.train.csv")
    header = list(df.columns)
    rows = [list(r) for r in df.values]

    chosen = choose(header, rows, label)
    print(f"Chose setup #{chosen + 1} out of {len(rows)}")
