import pandas as pd
import numpy as np


def remember(header, rows):
    df = pd.DataFrame(rows, columns=header)

    goal_cols = {}
    for c in header:
        if c.endswith('-'):
            goal_cols[c] = 'min'
        elif c.endswith('+'):
            goal_cols[c] = 'max'

    best = {c: float(df[c].min()) for c, d in goal_cols.items() if d == 'min'}
    best.update({c: float(df[c].max()) for c, d in goal_cols.items() if d == 'max'})
    worst = {c: float(df[c].max()) for c, d in goal_cols.items() if d == 'min'}
    worst.update({c: float(df[c].min()) for c, d in goal_cols.items() if d == 'max'})

    scores = []
    for i in range(len(df)):
        s = 0.0
        for col, direction in goal_cols.items():
            v = float(df.loc[i, col])
            span = best[col] - worst[col]
            if span > 1e-9:
                normed = (v - worst[col]) / span
            else:
                normed = 0.5
            s += normed
        scores.append(s)

    n_top = max(1, int(len(scores) * 0.05))
    top_idx = np.argsort(scores)[::-1][:n_top]
    return sorted(top_idx.tolist())

import numpy as np


def guess(header, train, test):
    goal_cols = [c for c in header if c.endswith('-') or c.endswith('+')]
    first_goal = goal_cols[0]  # MRE-

    input_cols = [c for c in header if not c.endswith('-') and not c.endswith('+')]

    numeric_input_cols = []
    symbolic_input_cols = []

    for c in input_cols:
        is_numeric = True
        for row in train:
            val = row[input_cols.index(c)]
            try:
                float(val)
            except (ValueError, TypeError):
                is_numeric = False
                break
        if is_numeric:
            numeric_input_cols.append(c)
        else:
            symbolic_input_cols.append(c)

    X_train = []
    for row in train:
        x = [float(row[input_cols.index(c)]) for c in numeric_input_cols]
        X_train.append(x)
    X_train = np.array(X_train)

    y_train = np.array([float(row[header.index(first_goal)]) for row in train])

    from sklearn.linear_model import LinearRegression
    model = LinearRegression()
    model.fit(X_train, y_train)

    predictions = []
    for row in test:
        x = [float(row[input_cols.index(c)]) for c in numeric_input_cols]
        pred = float(model.predict([x])[0])
        predictions.append(pred)

    return predictions

import pandas as pd
import numpy as np


def certify(xs, ys):
    xs = np.array(xs)
    ys = np.array(ys)

    n1, n2 = len(xs), len(ys)
    mean_diff = abs(float(np.mean(xs)) - float(np.mean(ys)))

    var1 = float(np.var(xs, ddof=1)) if n1 > 1 else 0.0
    var2 = float(np.var(ys, ddof=1)) if n2 > 1 else 0.0

    se_sq = var1 / n1 + var2 / n2
    se = np.sqrt(se_sq) if se_sq > 0 else 0.0

    if se == 0:
        return True

    z = mean_diff / se
    return bool(z < 1.96)
