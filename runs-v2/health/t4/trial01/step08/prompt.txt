def certify(xs, ys):
    """Return True if xs and ys are the same apart from noise."""
    diff = [abs(x - y) for x, y in zip(xs, ys)]
    avg_diff = sum(diff) / len(diff) if diff else 0.0
    return avg_diff < 0.1


def main():
    df = pd.read_csv("Health-ClosedIssues0000.train.csv")

    rows = [list(map(float, row)) for _, row in df.iterrows()]

    best_positions = remember(header, rows)

    big_rows = []
    small_rows = []
    for pos in best_positions:
        if rows[pos][header.index('n_estimators')] > 50:
            big_rows.append(pos)
        else:
            small_rows.append(pos)

    small_mre = [rows[p][header.index('MRE-')] for p in small_rows]

    result = certify(big_mre, small_mre)

def blame(header, rows):
    goal_cols = [c for c in header if c.endswith('-') or c.endswith('+')]

    # Score each row: sum of absolute deviations from best on goals
    scores = []
    min_vals = {}
    max_vals = {}
    for col in goal_cols:
        vals = []
        for row in rows:
            cell = row[header.index(col)]
            if cell == "?" or (isinstance(cell, float) and np.isnan(cell)):
                continue
            v = float(cell)
            vals.append(v)
        if '-' in col:
            min_vals[col] = min(vals) if vals else 0.0
        else:
            max_vals[col] = max(vals) if vals else 0.0

    for i, row in enumerate(rows):
        s = 0.0
        for col in goal_cols:
            cell = row[header.index(col)]
            if cell == "?" or (isinstance(cell, float) and np.isnan(cell)):
                continue
            v = float(cell)
            if '-' in col:
                s += abs(v - min_vals[col])
            else:
                s += abs(v - max_vals[col])
        scores.append(s)

    # Worst rows (highest score = furthest from best performance)
    worst_indices = np.argsort(scores)[::-1][:max(1, int(len(rows) * 0.2))]
    worst_set = set(worst_indices.tolist())

    # Input columns (not goals)
    input_cols = [c for c in header if c not in goal_cols]

    # For each input col: compare average value in worst vs best rows
    explanations = []
    for col in input_cols:
        vals_worst = []
        vals_best = []
        for i, row in enumerate(rows):
            cell = row[header.index(col)]
            if isinstance(cell, (int, float)):
                v = float(cell)
                if i in worst_set:
                    vals_worst.append(v)
                elif scores[i] <= np.percentile(scores, 20):
                    vals_best.append(v)

        if vals_worst and vals_best:
            avg_worst = sum(vals_worst) / len(vals_worst)
            avg_best = sum(vals_best) / len(vals_best)
            diff = abs(avg_worst - avg_best)
            explanations.append((col, diff))

    # Sort by magnitude of difference (most explanatory first)
    explanations.sort(key=lambda x: -x[1])

    return [c for c, _ in explanations[:5]]

def choose(header, rows, label):
    goal_names = [c for c in header if c.endswith('-') or c.endswith('+')]

    best_pos = 0
    best_score = float('inf')

    for i in range(50):
        goals = label(i)
        score = 0.0
        for j, g_name in enumerate(goal_names):
            v = float(goals[j]) if isinstance(goals[j], str) else goals[j]
            if '-' in g_name:
                score += v          # minimise MRE-
            else:
                score -= v          # maximise ACC+, PRED40+

        if score < best_score:
            best_score = score
            best_pos = i

    return best_pos
