import pandas as pd
from step01 import remember


def certify(xs, ys):
    """Return True if xs and ys are the same apart from noise."""
    diff = [abs(x - y) for x, y in zip(xs, ys)]
    avg_diff = sum(diff) / len(diff) if diff else 0.0
    return avg_diff < 0.1


def main():
    df = pd.read_csv("Health-ClosedIssues0000.train.csv")
    header = list(df.columns)
    rows = [list(map(float, row)) for _, row in df.iterrows()]

    best_positions = remember(header, rows)

    big_rows = []
    small_rows = []
    for pos in best_positions:
        if rows[pos][header.index('n_estimators')] > 50:
            big_rows.append(pos)
        else:
            small_rows.append(pos)

    big_mre = [rows[p][header.index('MRE-')] for p in big_rows]
    small_mre = [rows[p][header.index('MRE-')] for p in small_rows]

    result = certify(big_mre, small_mre)
    print(f"Big vs Small n_estimators: {'Yes' if result else 'No'}, they differ by noise")


if __name__ == "__main__":
    main()
