import ezr

def justify(header, rows, row):
    """Compare a query setup against all training setups."""
    tbl = ezr.Tbl([header] + rows)
    node = ezr.tree(tbl, tbl.rows)

    scores = []
    for r in rows:
        leaf = ezr.leaf(node, r)
        scores.append(leaf.mu)

    qlf = ezr.leaf(node, row)
    qscore = qlf.mu

    n_worse = sum(1 for s in scores if s > qscore)
    p = n_worse / len(scores)

    reason = f"This setup is better than {p:.0%} of all setups."
    return (p, reason)


if __name__ == "__main__":
    import os
    path = os.path.join(os.path.dirname(__file__) or ".", "Health-ClosedIssues0000.train.csv")
    with open(path, encoding="utf-8-sig") as f:
        lines = [l.strip() for l in f if l.strip()]
    header = lines[0].split(",")
    rows = []
    for line in lines[1:]:
        rows.append([ezr.atom(s) for s in line.split(",")])

    p, reason = justify(header, rows, rows[-1])
    print(reason)
