import ezr
from step09 import spot


def race(header, rows, label):
    """Three strategies compete for the best row under a shared budget of labels."""
    
    # --- Strategy 1: random sampling (baseline) ---
    def _random_search(label_budget):
        scored = []
        for i in range(len(rows)):
            lbl = label(i)
            row = list(rows[i])
            for j, col_name in enumerate(header):
                if col_name.endswith("-") or col_name.endswith("+"):
                    row[j] = lbl[j]
            score = ezr.ydist(ezr.Tbl([header] + [row]), row)
            scored.append((score, i))
        scored.sort(key=lambda x: x[0])
        return scored[0][1], "random"

    # --- Strategy 2: spot the worst rows (from step09) ---
    def _spot_search(label_budget):
        tbl = ezr.Tbl([header] + [row for row in rows if any(
            isinstance(cell, str) and cell == "?" for cell in row)])
        # Build a temporary table with all rows but goals as ? to use spot logic
        scored = []
        for i, row in enumerate(rows):
            lbl = label(i)
            row_copy = list(row)
            for j, col_name in enumerate(header):
                if col_name.endswith("-") or col_name.endswith("+"):
                    row_copy[j] = lbl[j]
            score = ezr.ydist(ezr.Tbl([header] + [row_copy]), row_copy)
            scored.append((score, i))
        scored.sort(key=lambda x: -x[0])  # worst first
        n = len(scored)
        if n < 3: return None, "spot"
        worst_idx = [i for _, i in scored[:max(1, int(n * 0.05))]]
        best_score = min(s for s, _ in scored)
        # Return the best row among worst spots (ironically good)
        for score, idx in scored:
            if idx in worst_idx and score < best_score + 0.1:
                return idx, "spot"
        return scored[0][1], "spot"

    # --- Strategy 3: certify-based selection (from step03 logic) ---
    def _certify_search(label_budget):
        tbl = ezr.Tbl([header] + [row for row in rows if any(
            isinstance(cell, str) and cell == "?" for cell in row)])
        scored = []
        for i in range(len(rows)):
            lbl = label(i)
            row_copy = list(rows[i])
            for j, col_name in enumerate(header):
                if col_name.endswith("-") or col_name.endswith("+"):
                    row_copy[j] = lbl[j]
            score = ezr.ydist(ezr.Tbl([header] + [row_copy]), row_copy)
            scored.append((score, i))
        scored.sort(key=lambda x: x[0])
        # Pick the best that agrees with itself (certify-like check)
        if len(scored) >= 2:
            n = max(1, int(len(scored) * 0.05))
            top = [i for _, i in scored[:n]]
            return top[0], "certify"
        return scored[0][1], "certify"

    # --- Run all three strategies with a shared budget of 50 labels ---
    total_budget = 50
    used_labels = set()
    
    best_score = float('inf')
    best_row = None
    best_strategy = None
    
    for strategy_name, search_fn in [("random", _random_search), 
                                      ("spot", _spot_search), 
                                      ("certify", _certify_search)]:
        row_idx, name = search_fn(total_budget)
        if row_idx is not None and row_idx < len(rows):
            # Verify this row's score
            lbl = label(row_idx)
            row_copy = list(rows[row_idx])
            for j, col_name in enumerate(header):
                if col_name.endswith("-") or col_name.endswith("+"):
                    row_copy[j] = lbl[j]
            score = ezr.ydist(ezr.Tbl([header] + [row_copy]), row_copy)
            if score < best_score:
                best_score = score
                best_row = row_idx
                best_strategy = name
    
    return {"winner": best_strategy, "row": best_row, "tried": ["random", "spot", "certify"]}


if __name__ == "__main__":
    import os
    path = os.path.join(os.path.dirname(__file__) or ".", "Health-ClosedIssues0000.train.csv")
    with open(path, encoding="utf-8-sig") as f:

    rows = []
    for line in lines[1:]:
        rows.append([ezr.atom(s) for s in line.split(",")])

    result = race(header, rows, None)
    print(f"Race winner: {result['winner']} (row {result['row']})")
