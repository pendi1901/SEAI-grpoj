import ezr

def choose(header, rows, label):
    """Label up to 50 setups and return the position of the best."""
    tbl = ezr.Tbl([header] + rows)
    
    best_idx = None
    best_score = float('inf')
    labeled_count = 0
    
    for i in range(len(rows)):
        if labeled_count >= 50:
            break
        
        lbl = label(i)
        
        # Replace "?" goal cells with true values so ydist sees real data.
        row = list(rows[i])
        for j, col_name in enumerate(header):
            if col_name.endswith("-") or col_name.endswith("+"):
                row[j] = lbl[j]
        
        score = ezr.ydist(tbl, row)
        if score < best_score:
            best_score = score
            best_idx = i
        
        labeled_count += 1
    
    return best_idx


if __name__ == "__main__":
    import os
    path = os.path.join(os.path.dirname(__file__) or ".", "Health-ClosedIssues0000.train.csv")
    with open(path, encoding="utf-8-sig") as f:
        lines = [l.strip() for l in f if l.strip()]
    header = lines[0].split(",")
    rows = []
    for line in lines[1:]:
        rows.append([ezr.atom(s) for s in line.split(",")])

    best = choose(header, rows, None)
    print(f"Best setup is row {best}")
