import ezr

def flow(header, row, seen):
    """Return True when `row` is good enough to ship; False otherwise."""
    all_rows = seen + [row]
    tbl = ezr.Tbl([header] + all_rows)

    # Score each row by ydist (lower = closer to ideal goals).
    scores = [(ezr.ydist(tbl, r), i) for i, r in enumerate(all_rows)]
    scores.sort(key=lambda x: x[0])

    # Ship when the current row is the best seen so far.
    if all_rows[-1] == row and scores[0][1] == len(seen):
        return True

    return False


if __name__ == "__main__":
    import os
    path = os.path.join(os.path.dirname(__file__) or ".", "Health-ClosedIssues0000.train.csv")
    with open(path, encoding="utf-8-sig") as f:
        lines = [l.strip() for l in f if l.strip()]
    header = lines[0].split(",")
    rows = []
    for line in lines[1:]:
        rows.append([ezr.atom(s) for s in line.split(",")])

    best_idx = None
    for i, r in enumerate(rows):
        result = flow(header, r, rows[:i] if best_idx is not None else [])
        if result:
            print(f"Shipped setup {i} (MRE-={r[0]}, ACC+={r[1]}, PRED40+={r[2]})")
            break
    else:
        print("No good enough setup found.")
