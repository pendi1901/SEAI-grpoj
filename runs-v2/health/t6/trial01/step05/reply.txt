import ezr

def blame(header, rows):
    """Return names of input columns (not goals) that best explain why worst rows are bad."""
    tbl = ezr.Tbl([header] + rows)

    # Score every row by ydist; lower is better.
    scored = [(ezr.ydist(tbl, r), i) for i, r in enumerate(rows)]
    scored.sort(key=lambda x: x[0])  # ascending → best first

    n_bad = max(1, int(len(scored) * 0.1))
    bad_idx = [i for _, i in scored[:n_bad]]
    good_idx = [i for _, _ in scored[-n_bad:]]

    blame_scores = []
    for col in tbl.x:
        if type(col) is ezr.Sym:
            bad_set = set(rows[i][col.at] for i in bad_idx)
            good_set = set(rows[j][col.at] for j in good_idx)
            union = len(bad_set | good_set)
            diff = 1 - len(bad_set & good_set) / union if union else 0.5
        else:
            bad_mean = sum(float(rows[i][col.at]) for i in bad_idx) / max(1, len(bad_idx))
            good_mean = sum(float(rows[j][col.at]) for j in good_idx) / max(1, len(good_idx))
            sd = ezr.div(col) if ezr.div(col) > 0 else 1e-9
            bad_norm = (bad_mean - col.mu) / sd
            good_norm = (good_mean - col.mu) / sd
            diff = abs(bad_norm - good_norm)

        blame_scores.append((diff, col.txt))

    blame_scores.sort(key=lambda x: x[0], reverse=True)
    return [name for _, name in blame_scores]


if __name__ == "__main__":
    import os
    path = os.path.join(os.path.dirname(__file__) or ".", "Health-ClosedIssues0000.train.csv")
    with open(path, encoding="utf-8-sig") as f:
        lines = [l.strip() for l in f if l.strip()]

    header = lines[0].split(",")
    rows = []
    for line in lines[1:]:
        rows.append([ezr.atom(s) for s in line.split(",")])

    bad_cols = blame(header, rows)
    print(f"Worst setups are most damaged by these columns:")
    for c in bad_cols:
        print(f"  {c}")
