import pandas as pd
import numpy as np


def remember(header, rows):
    goals = {}
    for col in header:
        if col.endswith('+'):
            goals[col] = 'max'
        elif col.endswith('-'):
            goals[col] = 'min'

    goal_cols = list(goals.keys())
    max_vals = {col: 1.0 for col in goal_cols}
    for col in goal_cols:
        vals = [float(r[header.index(col)]) for r in rows if not np.isnan(float(r[header.index(col)]))]
        if vals:
            max_vals[col] = max(vals)

    scores = []
    for row in rows:
        total = 0.0
        for col in goal_cols:
            val = float(row[header.index(col)])
            if goals[col] == 'max':
                total += val / max_vals[col]
            else:
                total += 1 - (val / max_vals[col])
        scores.append(total)

    n = len(rows)
    k = max(1, int(n * 0.05))
    ranked = sorted(range(len(scores)), key=lambda i: -scores[i])
    return ranked[:k]


if __name__ == "__main__":
    df = pd.read_csv('WA_Fn-UseC_-HR-Employee-Attrition.train.csv')
    header = list(df.columns)

    rows = []
    for _, row in df.iterrows():
        r = []
        for col, val in zip(header, row):
            if pd.isna(val):
                r.append(float('nan'))
            elif isinstance(val, (int, float)):
                r.append(float(val))
            else:
                r.append(str(val))
        rows.append(r)

    result = remember(header, rows)
    print(f"Top {len(result)} employees who stay unusually long:")
    for pos in result:
        emp = df.iloc[pos]
        print(
            f"  Employee #{emp['EmployeeNumber']}: "
            f"AtCompany+={int(emp['YearsAtCompany+'])}y, "
            f"InCurrentRole+={int(emp['YearsInCurrentRole+'])}y, "
            f"SinceLastPromotion={int(emp['YearsSinceLastPromotion'])}y, "
            f"WithCurrManager+={int(emp['YearsWithCurrManager+'])}y"
        )
