import pandas as pd
import numpy as np


def guess(header, train, test):
    """Predict attrition (first goal column) for each test row."""
    # Build feature matrix from all non-goal columns in train
    X_train = []
    y_train = []
    for row in train:
        x = []
        y = None
        for i, col in enumerate(header):
            if col == 'attrition':
                y = 0.0 if str(row[i]) == 'No' else 1.0
            elif col.endswith('+'):
                continue  # skip goal columns used only for scoring
            else:
                val = row[i]
                x.append(float(val) if isinstance(val, (int, float)) else 0.0)
        X_train.append(x)

    X_test = []
    for row in test:
        x = []
        for i, col in enumerate(header):
            if col.endswith('+'):
                continue
            val = row[i]
            if val == '?':
                x.append(0.0)
            elif isinstance(val, (int, float)):
                x.append(float(val))
            else:
                x.append(0.0)
        X_test.append(x)

    from sklearn.linear_model import LogisticRegression
    model = LogisticRegression(max_iter=1000, solver='lbfgs')
    model.fit(X_train, y_train)
    probs = model.predict_proba(X_test)[:, 1]
    return [float(p) for p in probs]


if __name__ == "__main__":
    df = pd.read_csv('WA_Fn-UseC_-HR-Employee-Attrition.train.csv')
    header = list(df.columns)

    rows = []
    for _, row in df.iterrows():
        r = []
        for col, val in zip(header, row):
            if pd.isna(val):
                r.append(float('nan'))
            elif isinstance(val, (int, float)):
                r.append(float(val))
            else:
                r.append(str(val))
        rows.append(r)

    result = guess(header, rows, rows[:5])  # sanity check on first 5
    print(f"Attrition probability estimates for first 5 rows:")
    for i, p in enumerate(result):
        print(f"  Row {i}: P(attrition=Yes) = {p:.4f}")

    # Now predict attrition for the whole dataset and report overall rate
    preds = guess(header, rows, rows)
    predicted_attrition = ['No' if p < 0.5 else 'Yes' for p in preds]
    actual_attrition = [str(r[header.index('attrition')]) for r in rows]

    correct = sum(1 for a, p in zip(actual_attrition, predicted_attrition) if a == p)
    print(f"\nOverall accuracy: {correct}/{len(rows)}")
