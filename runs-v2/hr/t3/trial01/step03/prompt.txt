import pandas as pd
import numpy as np


def remember(header, rows):
    goals = {}
    for col in header:
        if col.endswith('+'):
            goals[col] = 'max'
        elif col.endswith('-'):
            goals[col] = 'min'

    goal_cols = list(goals.keys())
    max_vals = {col: 1.0 for col in goal_cols}
    for col in goal_cols:
        vals = [float(r[header.index(col)]) for r in rows if not np.isnan(float(r[header.index(col)]))]
        if vals:
            max_vals[col] = max(vals)

    scores = []
    for row in rows:
        total = 0.0
        for col in goal_cols:
            val = float(row[header.index(col)])
            if goals[col] == 'max':
                total += val / max_vals[col]
            else:
                total += 1 - (val / max_vals[col])
        scores.append(total)

    n = len(rows)
    k = max(1, int(n * 0.05))
    ranked = sorted(range(len(scores)), key=lambda i: -scores[i])
    return ranked[:k]

import pandas as pd

def guess(header, train, test):
    """Predict attrition (first goal column) for each test row."""
    # Build feature matrix from all non-goal columns in train
    X_train = []
    y_train = []
    for row in train:
        x = []
        y = None
        for i, col in enumerate(header):
            if col == 'attrition':
                y = 0.0 if str(row[i]) == 'No' else 1.0
            elif col.endswith('+'):
                continue  # skip goal columns used only for scoring
            else:
                val = row[i]
                x.append(float(val) if isinstance(val, (int, float)) else 0.0)
        X_train.append(x)

    X_test = []
    for row in test:
        x = []
        for i, col in enumerate(header):
            if col.endswith('+'):
                continue
            val = row[i]
            if val == '?':
                x.append(0.0)
            elif isinstance(val, (int, float)):
                x.append(float(val))
            else:
                x.append(0.0)
        X_test.append(x)

    from sklearn.linear_model import LogisticRegression
    model = LogisticRegression(max_iter=1000, solver='lbfgs')
    model.fit(X_train, y_train)
    probs = model.predict_proba(X_test)[:, 1]
    return [float(p) for p in probs]
