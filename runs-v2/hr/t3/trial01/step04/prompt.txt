import pandas as pd
import numpy as np


def remember(header, rows):
    goals = {}
    for col in header:
        if col.endswith('+'):
            goals[col] = 'max'
        elif col.endswith('-'):
            goals[col] = 'min'

    goal_cols = list(goals.keys())
    max_vals = {col: 1.0 for col in goal_cols}
    for col in goal_cols:
        vals = [float(r[header.index(col)]) for r in rows if not np.isnan(float(r[header.index(col)]))]
        if vals:
            max_vals[col] = max(vals)

    scores = []
    for row in rows:
        total = 0.0
        for col in goal_cols:
            val = float(row[header.index(col)])
            if goals[col] == 'max':
                total += val / max_vals[col]
            else:
                total += 1 - (val / max_vals[col])
        scores.append(total)

    n = len(rows)
    k = max(1, int(n * 0.05))
    ranked = sorted(range(len(scores)), key=lambda i: -scores[i])
    return ranked[:k]

import pandas as pd
import numpy as np


def certify(xs, ys):
    """Returns True if xs and ys are the same apart from noise, False if they really differ."""
    n_x = len(xs)
    n_y = len(ys)
    
    mean_x = sum(xs) / n_x if n_x else 0.0
    mean_y = sum(ys) / n_y if n_y else 0.0
    
    var_x = sum((x - mean_x)**2 for x in xs) / (n_x - 1) if n_x > 1 else 0.0
    var_y = sum((y - mean_y)**2 for y in ys) / (n_y - 1) if n_y > 1 else 0.0
    
    pooled_var = ((n_x - 1) * var_x + (n_y - 1) * var_y) / (n_x + n_y - 2)
    pooled_std = pooled_var ** 0.5
    
    se = pooled_std * (1.0 / n_x + 1.0 / n_y) ** 0.5 if pooled_std > 0 else 1.0
    t_stat = abs(mean_x - mean_y) / se if se > 0 else 999.0
    
    return t_stat < 2.0
