import csv


def remember(header, rows):
    attrition_idx = header.index('attrition')
    years_cols = ['YearsAtCompany+', 'YearsInCurrentRole+',
                  'YearsSinceLastPromotion', 'YearsWithCurrManager+']
    year_indices = [header.index(c) for c in years_cols]

    candidates = []
    for i, row in enumerate(rows):
        if row[attrition_idx] != 'No':
            continue
        skip = any(row[idx] == '?' for idx in year_indices)
        if skip:
            continue
        score = sum(float(row[idx]) for idx in year_indices)
        candidates.append((score, i))

    candidates.sort(reverse=True)
    max_count = max(1, int(len(rows) * 0.05))
    return [pos for _, pos in candidates[:max_count]]


if __name__ == '__main__':

        rows = list(reader)

    positions = remember(header, rows)
    print(
        f"Out of {len(rows)} staff records, "
        f"{len(positions)} stayed unusually long."
    )

import csv


def guess(header, train, test):
    goal_cols = [c for c in header if c.endswith('+')]
    attrition_idx = header.index('attrition')
    years_idx = header.index(goal_cols[0])  # YearsAtCompany+

    n_train = len(train)
    year_vals = [float(row[years_idx]) for row in train]
    overall_mean = sum(year_vals) / n_train
    year_std = (sum((v - overall_mean) ** 2 for v in year_vals) / n_train) ** 0.5

    numeric_features = []
    for i, col in enumerate(header):
        if col[0].isupper() and col != 'attrition' and not any(
            goal_cols[i] == c for c in header
        ):
            pass
        # simpler: just collect uppercase-starting non-goal columns
    numeric_features = []
    seen = set(goal_cols)
    for i, col in enumerate(header):
        if col[0].isupper() and col not in seen:
            numeric_features.append((i, col))

    feat_means = {}
    feat_stds = {}
    feat_covs = {}  # cov with yearsAtCompany+

    for idx, col in numeric_features:
        vals = [float(row[idx]) for row in train]
        mean_v = sum(vals) / n_train
        std_v = (sum((v - mean_v) ** 2 for v in vals) / n_train) ** 0.5
        feat_means[col] = mean_v
        feat_stds[col] = std_v

    year_mean = overall_mean
    year_var = sum((v - year_mean) ** 2 for v in year_vals) / n_train
    year_std_sq = year_var if year_var > 0 else 1.0

    for idx, col in numeric_features:
        vals = [float(row[idx]) for row in train]
        mean_v = sum(vals) / n_train
        std_v = (sum((v - mean_v) ** 2 for v in vals) / n_train) ** 0.5
        cov = sum((vals[i] - mean_v) * (year_vals[i] - year_mean) for i in range(n_train)) / n_train
        feat_covs[col] = cov

    result = []
    for row in test:
        pred = overall_mean
        for idx, col in numeric_features:
            v = float(row[idx])
            mean_v = feat_means[col]
            std_v = feat_stds[col] if feat_stds[col] > 0 else 1.0
            z = (v - mean_v) / std_v
            pred += feat_covs[col] * z / year_std_sq

        # attrition: Yes means early departure → reduce prediction
        if row[attrition_idx] == 'Yes':
            pred *= 0.5

        result.append(pred)

    return result


if __name__ == '__main__':
    with open('WA_Fn-UseC_-HR-Employee-Attrition.train.csv', 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        rows = list(reader)

    preds = guess(header, rows, [])
    print(
        f"Predicted tenure for {len(preds)} staff records "
        f"(range: {min(preds):.1f} – {max(preds):.1f} years)."
    )
