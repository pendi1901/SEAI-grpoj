import csv


def remember(header, rows):
    attrition_idx = header.index('attrition')
    years_cols = ['YearsAtCompany+', 'YearsInCurrentRole+',
                  'YearsSinceLastPromotion', 'YearsWithCurrManager+']
    year_indices = [header.index(c) for c in years_cols]

    candidates = []
    for i, row in enumerate(rows):
        if row[attrition_idx] != 'No':
            continue
        skip = any(row[idx] == '?' for idx in year_indices)
        if skip:
            continue
        score = sum(float(row[idx]) for idx in year_indices)
        candidates.append((score, i))

    candidates.sort(reverse=True)
    max_count = max(1, int(len(rows) * 0.05))
    return [pos for _, pos in candidates[:max_count]]

import csv


def guess(header, train, test):
    goal_cols = [c for c in header if c.endswith('+')]
    attrition_idx = header.index('attrition')
    years_idx = header.index(goal_cols[0])  # YearsAtCompany+

    n_train = len(train)
    year_vals = [float(row[years_idx]) for row in train]
    overall_mean = sum(year_vals) / n_train
    year_std = (sum((v - overall_mean) ** 2 for v in year_vals) / n_train) ** 0.5

    numeric_features = []
    for i, col in enumerate(header):
        if col[0].isupper() and col != 'attrition' and not any(
            goal_cols[i] == c for c in header
        ):
            pass
        # simpler: just collect uppercase-starting non-goal columns
    numeric_features = []
    seen = set(goal_cols)
    for i, col in enumerate(header):
        if col[0].isupper() and col not in seen:
            numeric_features.append((i, col))

    feat_means = {}
    feat_stds = {}
    feat_covs = {}  # cov with yearsAtCompany+

    for idx, col in numeric_features:
        vals = [float(row[idx]) for row in train]
        mean_v = sum(vals) / n_train
        std_v = (sum((v - mean_v) ** 2 for v in vals) / n_train) ** 0.5
        feat_means[col] = mean_v
        feat_stds[col] = std_v

    year_mean = overall_mean
    year_var = sum((v - year_mean) ** 2 for v in year_vals) / n_train
    year_std_sq = year_var if year_var > 0 else 1.0

    for idx, col in numeric_features:
        vals = [float(row[idx]) for row in train]
        mean_v = sum(vals) / n_train
        std_v = (sum((v - mean_v) ** 2 for v in vals) / n_train) ** 0.5
        cov = sum((vals[i] - mean_v) * (year_vals[i] - year_mean) for i in range(n_train)) / n_train
        feat_covs[col] = cov

    result = []

        pred = overall_mean

            v = float(row[idx])

import csv


def certify(xs, ys):
    """Return True if xs and ys are essentially the same apart from noise."""
    n = len(xs) + len(ys)
    if n == 0:
        return False
    mean_x = sum(xs) / len(xs)
    mean_y = sum(ys) / len(ys)
    all_vals = xs + ys
    overall_mean = sum(all_vals) / n
    var = sum((v - overall_mean) ** 2 for v in all_vals) / n
    std = var ** 0.5
    diff = abs(mean_x - mean_y)
    if std > 0:
        return diff < 1.5 * std
    return False

def flow(header, row, seen):
    """Return True if this employee needs a retention conversation."""
    attrition_idx = header.index('attrition')
    years_col = 'YearsAtCompany+'
    years_idx = header.index(years_col)

    n_seen = len(seen)
    if n_seen == 0:
        return False

    # Compute tenure stats from seen rows
    year_vals = []
    for r in seen:
        try:
            year_vals.append(float(r[years_idx]))
        except (ValueError, TypeError):
            pass

    mean_y = sum(year_vals) / len(year_vals) if year_vals else 0.0
    var_y = sum((v - mean_y) ** 2 for v in year_vals) / len(year_vals) if year_vals else 1.0
    std_y = var_y ** 0.5

    # Current row's tenure
    try:
        cur_y = float(row[years_idx])
    except (ValueError, TypeError):
        return False

    # Flag attrition cases and low-tenure at-risk cases
    if row[attrition_idx] == 'Yes':
        return True

    if std_y > 0 and mean_y - 2 * std_y < cur_y < mean_y + 1.5 * std_y:
        return True

    return False

import csv


def blame(header, rows):
    """Return a list of input column names that best explain why the worst rows are bad."""
    attrition_idx = header.index('attrition')
    years_col = 'YearsAtCompany+'
    years_idx = header.index(years_col)

    # Score each row: attrition=Yes is very bad, low tenure is also bad.
    scores = []
    for row in rows:
        if row[attrition_idx] == 'Yes':
            scores.append(-10.0)
        else:
            try:
                scores.append(float(row[years_idx]))
            except (ValueError, TypeError):
                scores.append(0.0)

    n = len(scores)
    if n == 0:
        return []

    # Separate numeric and symbolic input columns.
    goal_cols = {c for c in header if c.endswith('+')}
    attrition_col = 'attrition'
    inputs = [c for i, c in enumerate(header)
              if c[0].isupper() and c not in goal_cols and c != attrition_col]

    numeric_inputs = []
    symbolic_inputs = []
    for col in inputs:
        # Check if column is numeric by sampling a few rows.
        try:
            float(rows[0][header.index(col)])
            numeric_inputs.append(col)
        except (ValueError, TypeError):
            symbolic_inputs.append(col)

    results = {}  # col -> strength of association with badness

    for col in numeric_inputs:
        vals = []
        for row in rows:
            try:
                vals.append(float(row[header.index(col)]))
            except (ValueError, TypeError):
                vals.append(0.0)
        # Pearson correlation between this column and the score.
        mean_v = sum(vals) / len(vals)
        mean_s = sum(scores) / n
        num = sum((vals[i] - mean_v) * (scores[i] - mean_s) for i in range(n))
        den = ((sum((v - mean_v) ** 2 for v in vals)) *
               sum((s - mean_s) ** 2 for s in scores)) ** 0.5)
        if den > 0:
            results[col] = num / den
        else:
            results[col] = 0.0

    # For symbolic inputs, measure how much the average tenure differs
    # between attrition=Yes and attrition=No within each group of that column.
    for col in symbolic_inputs:
        groups = {}
        for row in rows:
            key = row[header.index(col)]
            if key not in ('?', ''):
                groups.setdefault(key, []).append(row)

        # Compute mean YearsAtCompany+ per group, then difference between Yes and No.
        yes_vals = []
        no_vals = []
        for row in rows:
            if row[attrition_idx] == 'Yes':
                try:
                    yes_vals.append(float(row[years_idx]))
                except (ValueError, TypeError):
                    pass
            else:
                try:
                    no_vals.append(float(row[years_idx]))
                except (ValueError, TypeError):
                    pass

        if not groups or not yes_vals or not no_vals:
            results[col] = 0.0
            continue

        mean_yes = sum(yes_vals) / len(yes_vals)
        mean_no = sum(no_vals) / len(no_vals)
        # A large positive difference means attrition=Yes group has much lower tenure → bad column.
        diff = mean_yes - mean_no
        results[col] = diff

    # Rank by absolute strength and return top explanatory columns.
    ranked = sorted(results.items(), key=lambda x: abs(x[1]), reverse=True)
    return [col for col, _ in ranked[:5]]

import csv


def justify(header, rows, row):
    """Return (p, reason) where p is fraction of rows that `row` beats."""

    goal_cols = [c for c in header if c.endswith('+')]

    # Compute mean tenure for each goal column from the training data.
    goal_means = {}
    for col in goal_cols:
        vals = []
        for r in rows:
            try:
                vals.append(float(r[header.index(col)]))
            except (ValueError, TypeError):
                pass
        if vals:
            goal_means[col] = sum(vals) / len(vals)
        else:
            goal_means[col] = 0.0

    # Predict the hidden goals for `row` using these averages.
    predicted_goals = {col: goal_means.get(col, 0.0) for col in goal_cols}

    n_total = len(rows)
    count_better = 0

    for r in rows:
        is_better = True
        for col in goal_cols:
            try:
                actual_val = float(r[header.index(col)])
            except (ValueError, TypeError):
                continue
            if actual_val > predicted_goals[col]:
                is_better = False
                break
        if is_better:
            count_better += 1

    p = count_better / n_total if n_total else 0.0

    question_marks = sum(1 for col in goal_cols if row[header.index(col)] == '?')
    if question_marks == len(goal_cols):
        reason = (
            f"Based on average tenure patterns, this employee ranks better "
            f"than {p:.0%} of staff."
        )
    else:
        reason = (
            f"With estimated tenure metrics, this employee ranks better "
            f"than {p:.0%} of staff."
        )

    return (p, reason)

def choose(header, rows, label):
    """Interview up to 50 staff and return the position of the best one."""
    goal_cols = [c for c in header if c.endswith('+')]
    
    attrition_idx = None
    for i, col in enumerate(header):
        if col == 'attrition':
            attrition_idx = i
            break
    
    # Pick up to 50 employees who haven't left yet.
    candidates = []
    for i, row in enumerate(rows):
        if attrition_idx is not None and row[attrition_idx] == 'No':
            candidates.append(i)
    
    n_to_interview = min(50, len(candidates))
    chosen = candidates[:n_to_interview]
    
    best_pos = -1
    best_score = float('-inf')
    
    for pos in chosen:
        goals = label(pos)  # list of true goal values (len == len(goal_cols))
        score = sum(goals)
        
        if score > best_score:
            best_score = score
            best_pos = pos
    
    return best_pos
