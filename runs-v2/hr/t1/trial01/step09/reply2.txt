import csv


def spot(header, rows):
    """Return row indices whose input values look wrong."""
    attrition_idx = header.index('attrition')
    goal_cols = [c for c in header if c.endswith('+')]
    years_col = 'YearsAtCompany+'
    years_idx = header.index(years_col)

    # Build a per-row score: sum of goals (higher is better).
    scores = []
    for row in rows:
        s = 0.0
        for col in goal_cols:
            cell = row[header.index(col)]
            if cell == '?':
                continue
            try:
                v = float(cell)
            except ValueError:
                v = 0.0
            s += v
        scores.append(s)

    # Per-column statistics from the whole dataset (excluding goal columns).
    numeric_inputs = []
    for i, col in enumerate(header):
        if not col[0].isupper():
            continue
        if col == 'attrition' or any(col.endswith('+') for _ in [1]):
            if col.endswith('+'):
                continue
        try:
            float(rows[0][i])
            numeric_inputs.append((i, col))
        except ValueError:
            pass

    col_stats = {}  # col -> (mean, std)
    for idx, col in numeric_inputs:
        vals = []
        for row in rows:
            try:
                vals.append(float(row[idx]))
            except ValueError:
                pass
        if not vals:
            continue
        mean_v = sum(vals) / len(vals)
        var_v = sum((v - mean_v) ** 2 for v in vals) / len(vals)
        std_v = var_v ** 0.5
        col_stats[col] = (mean_v, std_v)

    # Per-row anomaly score: how far the row's numeric inputs are from their column means.
    row_anomalies = []
    for i, row in enumerate(rows):
        a = 0.0
        for idx, col in numeric_inputs:
            if col == years_col:
                continue
            try:
                v = float(row[idx])
            except ValueError:
                continue
            mean_v, std_v = col_stats.get(col, (0.0, 1.0))
            if std_v > 0:
                a += abs(v - mean_v) / std_v
        row_anomalies.append(a)

    # Per-row goal score deviation from the dataset average.
    goal_means = {}
    for col in goal_cols:
        vals = []
        for r in rows:
            cell = r[header.index(col)]
            if cell == '?':
                continue
            try:
                vals.append(float(cell))
            except ValueError:
                pass
        if vals:
            goal_means[col] = sum(vals) / len(vals)

    row_goal_devs = []
    for i, row in enumerate(rows):
        d = 0.0
        for col in goal_cols:
            cell = row[header.index(col)]
            if cell == '?':
                continue
            try:
                v = float(cell)
            except ValueError:
                continue
            mean_v = goal_means.get(col, 0.0)
            d += abs(v - mean_v) / (1.0 + abs(mean_v))
        row_goal_devs.append(d)

    # Combine anomaly and goal deviation into a single suspiciousness score.
    combined = []
    for i in range(len(rows)):
        c = row_anomalies[i] + row_goal_devs[i]
        combined.append(c)

    # Rows with the highest combined score are most suspicious.
    ranked = sorted(range(len(combined)), key=lambda k: combined[k], reverse=True)
    return ranked[:5]


if __name__ == '__main__':
    with open('WA_Fn-UseC_-HR-Employee-Attrition.train.csv', 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        rows = list(reader)

    suspicious = spot(header, rows)
    print(
        "These row positions look wrong:\n"
        + "\n".join(f"  • Row {pos}" for pos in suspicious)
    )
