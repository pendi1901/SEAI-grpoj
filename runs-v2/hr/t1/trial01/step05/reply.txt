import csv


def blame(header, rows):
    """Return a list of input column names that best explain why the worst rows are bad."""
    attrition_idx = header.index('attrition')
    years_col = 'YearsAtCompany+'
    years_idx = header.index(years_col)

    # Score each row: attrition=Yes is very bad, low tenure is also bad.
    scores = []
    for row in rows:
        if row[attrition_idx] == 'Yes':
            scores.append(-10.0)
        else:
            try:
                scores.append(float(row[years_idx]))
            except (ValueError, TypeError):
                scores.append(0.0)

    n = len(scores)
    if n == 0:
        return []

    # Separate numeric and symbolic input columns.
    goal_cols = {c for c in header if c.endswith('+')}
    attrition_col = 'attrition'
    inputs = [c for i, c in enumerate(header)
              if c[0].isupper() and c not in goal_cols and c != attrition_col]

    numeric_inputs = []
    symbolic_inputs = []
    for col in inputs:
        # Check if column is numeric by sampling a few rows.
        try:
            float(rows[0][header.index(col)])
            numeric_inputs.append(col)
        except (ValueError, TypeError):
            symbolic_inputs.append(col)

    results = {}  # col -> strength of association with badness

    for col in numeric_inputs:
        vals = []
        for row in rows:
            try:
                vals.append(float(row[header.index(col)]))
            except (ValueError, TypeError):
                vals.append(0.0)
        # Pearson correlation between this column and the score.
        mean_v = sum(vals) / len(vals)
        mean_s = sum(scores) / n
        num = sum((vals[i] - mean_v) * (scores[i] - mean_s) for i in range(n))
        den = ((sum((v - mean_v) ** 2 for v in vals)) *
               sum((s - mean_s) ** 2 for s in scores)) ** 0.5)
        if den > 0:
            results[col] = num / den
        else:
            results[col] = 0.0

    # For symbolic inputs, measure how much the average tenure differs
    # between attrition=Yes and attrition=No within each group of that column.
    for col in symbolic_inputs:
        groups = {}
        for row in rows:
            key = row[header.index(col)]
            if key not in ('?', ''):
                groups.setdefault(key, []).append(row)

        # Compute mean YearsAtCompany+ per group, then difference between Yes and No.
        yes_vals = []
        no_vals = []
        for row in rows:
            if row[attrition_idx] == 'Yes':
                try:
                    yes_vals.append(float(row[years_idx]))
                except (ValueError, TypeError):
                    pass
            else:
                try:
                    no_vals.append(float(row[years_idx]))
                except (ValueError, TypeError):
                    pass

        if not groups or not yes_vals or not no_vals:
            results[col] = 0.0
            continue

        mean_yes = sum(yes_vals) / len(yes_vals)
        mean_no = sum(no_vals) / len(no_vals)
        # A large positive difference means attrition=Yes group has much lower tenure → bad column.
        diff = mean_yes - mean_no
        results[col] = diff

    # Rank by absolute strength and return top explanatory columns.
    ranked = sorted(results.items(), key=lambda x: abs(x[1]), reverse=True)
    return [col for col, _ in ranked[:5]]


if __name__ == '__main__':
    with open('WA_Fn-UseC_-HR-Employee-Attrition.train.csv', 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        rows = list(reader)

    bad_cols = blame(header, rows)
    print(
        "These input columns best explain why some employees leave early:\n"
        + "\n".join(f"  • {c}" for c in bad_cols)
    )
