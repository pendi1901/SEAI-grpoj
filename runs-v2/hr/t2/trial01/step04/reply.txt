    years_idx = header.index('YearsAtCompany+')
    goal_cols = ['YearsAtCompany+', 'YearsInCurrentRole+',

    n_seen = len(seen)

    def safe_float(val):

    # OK, simplest correct approach:
    def _stats(rows):
        means = {}
        sums_sq = {}
        for r in rows:
            for col in numeric_cols:
                v = safe_float(r[col_idx[col]])
                means[col] = means.get(col, 0.0) + (v - means.get(col, 0.0))
        n = len(rows)
        for col in numeric_cols:
            m = means.get(col, 0.0) / n
            sums_sq[col] = sum((safe_float(r[col_idx[col])) - m) ** 2 for r in rows)

    stats_seen = _stats(seen)
    y_mean = sum(safe_float(r[years_idx]) for r in seen) / n
    y_std = (sum((safe_float(r[years_idx]) - y_mean) ** 2 for r in seen) / n) ** 0.5 or 1e-6)

    correlations = {}
    for col in numeric_cols:
        x_vals = [safe_float(r[col_idx[col]]) for r in seen]
        x_m, x_s = stats_seen[col]
        numerator = sum((x - x_m) * (y - y_mean) for x, y in zip(x_vals, [safe_float(r[years_idx]) for r in seen]))
        denom_x = ((sum((v - x_m)**2 for v in x_vals))**0.5 or 1e-6)
        if denom_x * y_std > 0:
            correlations[col] = numerator / (denom_x * y_std)
        else:
            correlations[col] = 0.0

    pred = y_mean
    for col in numeric_cols:

        z_score = (val - stats_seen[col][0]) / stats_seen[col][1]
        pred += correlations.get(col, 0.0) * z_score

    return pred < y_mean
