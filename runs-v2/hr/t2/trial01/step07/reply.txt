import csv


def choose(header, rows, label):
    """Find the working arrangement that keeps people longest."""

    years_idx = header.index('YearsAtCompany+')
    goal_cols = ['YearsAtCompany+', 'YearsInCurrentRole+',
                 'YearsSinceLastPromotion', 'YearsWithCurrManager+']

    def safe_float(val):
        if val == '?':
            return 0.0
        try:
            return float(val)
        except (ValueError, TypeError):
            return 0.0

    # Identify input numeric columns
    numeric_cols = []
    for col in header:
        if col not in goal_cols and '+' not in col \
                and col != 'EmployeeNumber' and col != 'EmployeeCount':
            numeric_cols.append(col)

    n_rows = len(rows)

    # Compute correlations of each input feature with YearsAtCompany+
    y_vals = [safe_float(r[years_idx]) for r in rows]
    y_mean = sum(y_vals) / n_rows if n_rows else 0.0
    y_var = sum((v - y_mean)**2 for v in y_vals) / max(n_rows, 1)
    y_std = (y_var**0.5 or 1e-6)

    col_means = {}
    col_vars = {}
    for col in numeric_cols:
        vals = [safe_float(r[header.index(col)]) for r in rows]
        m = sum(vals) / len(vals) if vals else 0.0
        v = sum((x - m)**2 for x in vals) / max(len(vals), 1)
        col_means[col] = m
        col_vars[col] = (v**0.5 or 1e-6)

    # Rank rows by predicted YearsAtCompany+ score using correlations
    scored = []
    for i, row in enumerate(rows):
        pred = y_mean
        for col in numeric_cols:
            val = safe_float(row[header.index(col)])
            z = (val - col_means[col]) / col_vars[col] if col_vars[col] > 0 else 0.0
            x_vals = [safe_float(r[header.index(col)]) for r in rows]
            x_mean = col_means[col]
            numerator = sum((x - x_mean) * (y - y_mean) for x, y in zip(x_vals, y_vals))
            denom_x = ((sum((v - x_mean)**2 for v in x_vals))**0.5 or 1e-6)
            corr = numerator / (denom_x * y_std) if denom_x * y_std > 0 else 0.0
            pred += corr * z
        scored.append((pred, i))

    # Pick top candidates for interview (up to 50)
    scored.sort(reverse=True)
    top_n = min(50, n_rows)
    candidates = [pos for _, pos in scored[:top_n]]

    # Interview them and get true goals
    best_pos = None
    best_score = -float('inf')

    for i in candidates:
        goals = label(i)
        score = 0.0
        for col in goal_cols:
            v = safe_float(goals[header.index(col)])
            if col == 'YearsAtCompany+':
                score += v
            elif col == 'YearsInCurrentRole+':
                score += v
            elif col == 'YearsSinceLastPromotion':
                score += v
            elif col == 'YearsWithCurrManager+':
                score += v

        if score > best_score:
            best_score = score
            best_pos = i

    return best_pos


if __name__ == "__main__":
    path = 'WA_Fn-UseC_-HR-Employee-Attrition.train.csv'
    with open(path, newline='') as f:
        reader = csv.reader(f)
        header = next(reader)
        rows = list(reader)

    best = choose(header, rows, label)
    print(f"Best employee to interview: row {best}")
