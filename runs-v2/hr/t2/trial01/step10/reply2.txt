import csv


def race(header, rows, label):
    """Compare three strategies to find the best employee row."""

    def safe_float(val):
        if val == '?':
            return 0.0
        try:
            return float(val)
        except (ValueError, TypeError):
            return 0.0

    goal_cols = ['YearsAtCompany+', 'YearsInCurrentRole+',
                 'YearsSinceLastPromotion', 'YearsWithCurrManager+']

    numeric_cols = []
    for col in header:
        if col not in goal_cols and '+' not in col \
                and col != 'EmployeeNumber' and col != 'EmployeeCount':
            numeric_cols.append(col)

    n_rows = len(rows)
    col_idx = {col: header.index(col) for col in numeric_cols}

    # Precompute per-column mean, std, and correlation with YearsAtCompany+
    y_vals = [safe_float(r[header.index('YearsAtCompany+')]) for r in rows]
    y_mean = sum(y_vals) / n_rows if n_rows else 0.0
    y_var = sum((v - y_mean) ** 2 for v in y_vals) / max(n_rows, 1)
    y_std = (y_var ** 0.5 or 1e-6)

    col_means = {}
    col_vars = {}
    col_corrs = {}
    for col in numeric_cols:
        vals = [safe_float(r[col_idx[col]]) for r in rows]
        m = sum(vals) / len(vals) if vals else 0.0
        v = sum((x - m) ** 2 for x in vals) / max(len(vals), 1)
        col_means[col] = m
        col_vars[col] = (v ** 0.5 or 1e-6)

        numerator = sum((x - m) * (y - y_mean) for x, y in zip(vals, y_vals))
        denom_x = ((sum((x - m) ** 2 for x in vals)) ** 0.5 or 1e-6)
        col_corrs[col] = numerator / (denom_x * y_std) if denom_x * y_std > 0 else 0.0

    def predict(row):
        pred = y_mean
        for col in numeric_cols:
            val = safe_float(row[col_idx[col]])
            z = (val - col_means[col]) / col_vars[col] if col_vars[col] > 0 else 0.0
            pred += col_corrs[col] * z
        return pred

    def score_row(row):
        s = 0.0
        for col in goal_cols:
            v = safe_float(row[header.index(col)])
            if v < 0:
                s -= abs(v)
            else:
                s += v
        return s

    # Strategy 1: pick the row with highest predicted YearsAtCompany+
    def strategy_predict(rows):
        scored = [(predict(r), i) for i, r in enumerate(rows)]
        scored.sort(reverse=True)
        top_n = min(50, len(scored))
        return [pos for _, pos in scored[:top_n]]

    # Strategy 2: pick rows at random (up to 50)
    def strategy_random(rows):
        import random
        n = min(50, len(rows))
        indices = list(range(len(rows)))
        random.shuffle(indices)
        return indices[:n]

    # Strategy 3: pick rows with highest predicted score (all goals combined)
    def strategy_scored(rows):
        scored = [(score_row(r), i) for i, r in enumerate(rows)]
        scored.sort(reverse=True)
        top_n = min(50, len(scored))
        return [pos for _, pos in scored[:top_n]]

    # Run all three strategies and collect labelled rows (at most 50 total)
    tried = []
    seen_labels = set()
    best_strategy = None
    best_row = None
    best_score = -float('inf')

    for name, strategy in [('predict', strategy_predict), ('random', strategy_random),
                            ('scored', strategy_scored)]:
        candidates = strategy(rows)
        tried.append(name)
        for pos in candidates:
            if pos not in seen_labels and len(seen_labels) < 50:
                seen_labels.add(pos)
                true_goals = label(pos)
                s = score_row(true_goals)
                if s > best_score:
                    best_score = s
                    best_strategy = name
                    best_row = pos

    return {
        'winner': best_strategy,
        'row': best_row,
        'tried': tried
    }


if __name__ == '__main__':
    path = 'WA_Fn-UseC_-HR-Employee-Attrition.train.csv'
    with open(path, newline='') as f:
        reader = csv.reader(f)
        header = next(reader)
        rows = list(reader)

    result = race(header, rows, label)
    print(f"Winner: {result['winner']}, "
          f"Best row index: {result['row']}, "
          f"Strategies tried: {', '.join(result['tried'])}")
