import csv


def spot(header, rows):
    """Return positions (row indices) whose input cells look wrong, most suspicious first."""

    def safe_float(val):
        if val == '?':
            return 0.0
        try:
            return float(val)
        except (ValueError, TypeError):
            return 0.0

    goal_cols = ['YearsAtCompany+', 'YearsInCurrentRole+',
                 'YearsSinceLastPromotion', 'YearsWithCurrManager+']

    numeric_cols = []
    for col in header:
        if col not in goal_cols and '+' not in col \
                and col != 'EmployeeNumber' and col != 'EmployeeCount':
            numeric_cols.append(col)

    n_rows = len(rows)
    col_idx = {col: header.index(col) for col in numeric_cols}

    # Compute per-column mean and std
    col_means = {}
    col_vars = {}
    for col in numeric_cols:
        vals = [safe_float(r[col_idx[col]]) for r in rows]
        m = sum(vals) / n_rows if n_rows else 0.0
        v = sum((x - m) ** 2 for x in vals) / max(n_rows, 1)
        col_means[col] = m
        col_vars[col] = (v ** 0.5 or 1e-6)

    # Score each row by how many cells are far from expected
    scored = []
    for i, row in enumerate(rows):
        suspicious = 0.0
        worst_dev = 0.0
        for col in numeric_cols:
            val = safe_float(row[col_idx[col]])
            dev = abs(val - col_means[col]) / col_vars[col]
            if dev > worst_dev:
                worst_dev = dev
            if dev > 3.0:
                suspicious += 1
        scored.append((worst_dev, i))

    # Sort by worst deviation descending (most anomalous first)
    scored.sort(reverse=True)
    return [pos for _, pos in scored]


if __name__ == "__main__":
    path = 'WA_Fn-UseC_-HR-Employee-Attrition.train.csv'
    with open(path, newline='') as f:
        reader = csv.reader(f)
        header = next(reader)
        rows = list(reader)

    positions = spot(header, rows)
    print(f"Found {len(positions)} suspicious row(s):")
    for pos in positions[:10]:
        print(f"  Row #{pos}")
