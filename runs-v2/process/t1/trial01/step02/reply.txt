import csv


def guess(header, train, test):
    goal_cols = ['Cost-', 'Score-', 'Idle-']
    input_cols = [c for c in header if c not in goal_cols]

    def parse_row(row):
        result = {}
        ok = True
        for col in input_cols + ['Cost-']:
            idx = header.index(col)
            val = row[idx]
            if val == '?':
                return None
            try:
                result[col] = float(val)
            except ValueError:
                return None
        return result

    # Parse training rows (all complete, no "?" cells)
    train_parsed = []
    for row in train:
        parsed = parse_row(row)
        if parsed is not None:
            train_parsed.append(parsed)

    results = []
    for test_row in test:
        test_parsed = parse_row(test_row)
        if test_parsed is None:
            # Fallback: average Cost- across training data
            avg_cost = sum(t['Cost-'] for t in train_parsed) / len(train_parsed)
            results.append(avg_cost)
            continue

        # Nearest-neighbour on input features (excluding Cost-)
        best_idx = 0
        min_dist = float('inf')
        for i, tp in enumerate(train_parsed):
            dist = sum((test_parsed[c] - tp[c]) ** 2 for c in input_cols)
            if dist < min_dist:
                min_dist = dist
                best_idx = i

        results.append(train_parsed[best_idx]['Cost-'])

    return results


if __name__ == "__main__":
    with open('pom3a.train.csv', 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        rows = [row for row in reader]

    predictions = guess(header, rows, [])
    print("guess() function defined and ready.")
