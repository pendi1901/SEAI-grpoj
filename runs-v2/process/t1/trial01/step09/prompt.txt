import csv


def certify(xs, ys):
    """Return True if xs and ys are the same apart from noise, False if they really differ."""
    def mean(data):
        return sum(data) / len(data)

    def var(data, m):
        return sum((x - m) ** 2 for x in data) / (len(data) - 1)

    mx = mean(xs)
    my = mean(ys)
    sx = var(xs, mx)
    sy = var(ys, my)
    n = len(xs) + len(ys)
    se = ((sx / n) + (sy / n)) ** 0.5
    t = abs(mx - my) / se if se > 0 else 0
    return t < 2.0

import csv


def flow(header, row, seen):
    """Approve or reject each arriving proposal; cannot go back."""
    goal_cols = ['Cost-', 'Score-', 'Idle-']

    def parse_goals(row):
        goals = {}
        for col in goal_cols:
            idx = header.index(col)
            val = row[idx]
            if val == '?':
                return None
            try:
                goals[col] = float(val)
            except ValueError:
                return None
        return goals

    # Parse the current row's goals (skip if any goal is hidden)
    cur_goals = parse_goals(row)
    if cur_goals is None:
        return False  # can't evaluate, reject

    n_seen = len(seen) + 1
    seen.append(row)

    # If we have very few rows so far, be generous and approve anything decent.
    if n_seen <= 3:
        return True

    # Compute the best possible score for each goal from all rows seen so far.
    best = {}
    worst = {}
    for col in goal_cols:
        vals = [parse_goals(r)[col] for r in seen if parse_goals(r) is not None]
        best[col] = min(vals)   # lower is better (all goals end in -)
        worst[col] = max(vals)

    # Score the current row relative to what we've seen.
    score = 0.0
    for col in goal_cols:
        g = cur_goals[col]
        if g <= best[col]:
            score += 1.0   # at least as good as the best seen so far
        elif g >= worst[col]:
            score += 0.5   # between best and worst

    # If we've already approved something, only approve new rows that are clearly better.
    if len(seen) > 1:
        return False  # conservative: don't switch once we start rejecting
    else:
        # First row or early rows: approve anything that's not terrible.
        return True


if __name__ == "__main__":
    with open('pom3a.train.csv', 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        rows = [row for row in reader]

    seen = []
    approved = 0
    rejected = 0
    for row in rows:
        if flow(header, row, seen):
            approved += 1
        else:
            rejected += 1
        seen.append(row)

    print(f"Approved {approved}, rejected {rejected} out of {len(rows)} proposals.")

def blame(header, rows):
    goal_cols = ['Cost-', 'Score-', 'Idle-']
    input_cols = [c for c in header if c not in goal_cols]

    def parse_row(row):
        result = {}
        ok = True
        for col in input_cols + goal_cols:
            idx = header.index(col)
            val = row[idx]
            if val == '?':
                return None
            try:
                result[col] = float(val)
            except ValueError:
                return None
        return result

    parsed = []
    for row in rows:
        p = parse_row(row)
        if p is not None:
            parsed.append(p)

    # Score each row: lower is better (all goals end in -)
    def score(rec):
        s = 0.0
        for col in goal_cols:
            s += rec[col]
        return s

    scored = [(score(r), r) for r in parsed]
    scored.sort(key=lambda x: x[0])

    # Worst rows are those with highest total cost (sum of goals, since all end in -)
    worst_rows = [r for _, r in scored[len(scored)//2:]]

    if not worst_rows:
        return []

    # For each input column, compute correlation between its value and the row's score
    # Higher score = worse. We want columns that are high when rows are bad.
    n_worst = len(worst_rows)
    n_total = len(parsed)

    col_names = header[:]
    correlations = []
    for col in input_cols:
        vals = [r[col] for r in parsed if isinstance(r[col], (int, float))]
        worst_vals = [r[col] for r in worst_rows if isinstance(r[col], (int, float))]

        mean_all = sum(vals) / len(vals)
        mean_worst = sum(worst_vals) / len(worst_vals)

        num = sum((v - mean_all) * (w - mean_worst) for v, w in zip(vals, worst_vals))
        den = ((sum((v - mean_all)**2 for v in vals)) *
               (sum((w - mean_worst)**2 for w in worst_vals))) ** 0.5)

        corr = num / den if den > 0 else 0.0
        correlations.append((col, corr))

    # Sort by absolute correlation descending; pick top contributors
    correlations.sort(key=lambda x: abs(x[1]), reverse=True)
    return [c for c, _ in correlations[:3]]
