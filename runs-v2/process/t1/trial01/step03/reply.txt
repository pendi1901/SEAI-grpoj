import csv


def certify(xs, ys):
    """Return True if xs and ys are the same apart from noise, False if they really differ."""
    def mean(data):
        return sum(data) / len(data)

    def var(data, m):
        return sum((x - m) ** 2 for x in data) / (len(data) - 1)

    mx = mean(xs)
    my = mean(ys)
    sx = var(xs, mx)
    sy = var(ys, my)
    n = len(xs) + len(ys)
    se = ((sx / n) + (sy / n)) ** 0.5
    t = abs(mx - my) / se if se > 0 else 0
    return t < 2.0


def remember(header, rows):
    goal_cols = ['Cost-', 'Score-', 'Idle-']

    valid_rows = []
    for i, row in enumerate(rows):
        ok = True
        goals = {}
        for col in goal_cols:
            idx = header.index(col)
            val = row[idx]
            if val == '?':
                ok = False
                break
            try:
                goals[col] = float(val)
            except ValueError:
                ok = False
                break
        if ok:
            valid_rows.append((i, goals))

    if not valid_rows:
        return []

    n = len(valid_rows)

    goal_ranks = {col: {} for col in goal_cols}
    for col in goal_cols:
        vals = [(valid_rows[j][1][col], j) for j in range(n)]
        vals.sort(key=lambda x: x[0])
        for new_rank, (_, idx_in_valid) in enumerate(vals, 1):
            goal_ranks[col][idx_in_valid] = new_rank

    max_count = int(n * 0.05)
    threshold = n - max_count

    result = []
    for j, (orig_idx, goals) in enumerate(valid_rows):
        ranks = [goal_ranks[col][j] for col in goal_cols]
        if all(r <= threshold for r in ranks):
            result.append(orig_idx)

    return result


def guess(header, train, test):
    goal_cols = ['Cost-', 'Score-', 'Idle-']
    input_cols = [c for c in header if c not in goal_cols]

    def parse_row(row):
        result = {}
        ok = True
        for col in input_cols + ['Cost-']:
            idx = header.index(col)
            val = row[idx]
            if val == '?':
                return None
            try:
                result[col] = float(val)
            except ValueError:
                return None
        return result

    train_parsed = []
    for row in train:
        parsed = parse_row(row)
        if parsed is not None:
            train_parsed.append(parsed)

    results = []
    for test_row in test:
        test_parsed = parse_row(test_row)
        if test_parsed is None:
            avg_cost = sum(t['Cost-'] for t in train_parsed) / len(train_parsed)
            results.append(avg_cost)
            continue

        best_idx = 0
        min_dist = float('inf')
        for i, tp in enumerate(train_parsed):
            dist = sum((test_parsed[c] - tp[c]) ** 2 for c in input_cols)
            if dist < min_dist:
                min_dist = dist
                best_idx = i

        results.append(train_parsed[best_idx]['Cost-'])

    return results


def analyse(header, rows):
    goal_cols = ['Cost-', 'Score-', 'Idle-']
    input_cols = [c for c in header if c not in goal_cols]

    def parse_row(row):
        result = {}
        ok = True
        for col in input_cols + goal_cols:
            idx = header.index(col)
            val = row[idx]
            if val == '?':
                return None
            try:
                result[col] = float(val)
            except ValueError:
                return None
        return result

    parsed = []
    for row in rows:
        p = parse_row(row)
        if p is not None:
            parsed.append(p)

    small_cost, big_cost = [], []
    for rec in parsed:
        ts = rec['Team Size']
        c = rec['Cost-']
        if ts <= 0.5 * max(ts for _ in parsed):
            small_cost.append(c)
        else:
            big_cost.append(c)

    return certify(small_cost, big_cost), (small_cost, big_cost)
