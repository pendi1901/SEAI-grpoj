import pandas as pd
import numpy as np


def race(header, rows, label):
    goal_cols = [h for h in header if h.endswith('-')]
    input_cols = [h for h in header if not h.endswith('-')]

    df = pd.DataFrame(rows, columns=header)
    for col in goal_cols:
        df[col] = df[col].replace('?', np.nan).astype(float)

    means = {}
    stds = {}
    for col in goal_cols:
        vals = df[col].dropna()
        if len(vals) > 0:
            means[col] = vals.mean()
            stds[col] = vals.std()
        else:
            means[col] = 0.0
            stds[col] = 1.0

    def score_row(row):
        scores = []
        for col in goal_cols:
            val = row[header.index(col)] if header.index(col) < len(row) else '?'
            if pd.isna(val) or val == '?':
                return 0.0
            try:
                v = float(val)
            except (ValueError, TypeError):
                return 0.0
            if stds[col] == 0:
                z = 0.0
            else:
                z = (v - means[col]) / stds[col]
            scores.append(z)
        return np.mean(scores)

    def strategy_random(trials):
        best_idx = None
        best_score = -np.inf
        for _ in range(trials):
            i = np.random.randint(0, len(rows))
            s = score_row(rows[i])
            if s > best_score:
                best_score = s
                best_idx = i
        return best_idx

    def strategy_random_search(trials):
        best_idx = None
        best_score = -np.inf
        for _ in range(trials):
