import pandas as pd
import numpy as np
from sklearn.linear_model import LinearRegression


def guess(header, train, test):
    df = pd.DataFrame(train, columns=header)

    goal_cols = [h for h in header if h.endswith('-')]
    input_cols = [h for h in header if not h.endswith('-')]

    for col in goal_cols:
        df[col] = df[col].replace('?', np.nan).astype(float)

    models = {}
    for col in goal_cols:
        X = df[input_cols].values
        y = df[col].dropna().values
        mask = ~df[col].isna()
        if len(y) > 0:
            model = LinearRegression()
            model.fit(X[mask], y)
            models[col] = model

    results = []
    for row in test:
        X_pred = np.array([row[c] for c in input_cols])
        pred = models['Cost-'].predict(X_pred)[0]
        results.append(float(pred))

    return results

import pandas as pd
from scipy import stats


def certify(xs, ys):
    """Return True if xs and ys are statistically indistinguishable (same apart from noise)."""
    try:
        _, p = stats.ttest_ind(xs, ys)
        return p > 0.05
    except Exception:
        return False

import pandas as pd
import numpy as np


def blame(header, rows):
    goal_cols = [h for h in header if h.endswith('-')]
    input_cols = [h for h in header if not h.endswith('-')]

    df = pd.DataFrame(rows, columns=header)
    for col in goal_cols:
        df[col] = df[col].replace('?', np.nan).astype(float)

    # Normalize each goal to z-scores (low is good for '-', high is good for '+')
    scores = {}
    for col in goal_cols:
        vals = df[col].dropna()
        if len(vals) == 0:
            continue
        mean = vals.mean()
        std = vals.std()
        if std == 0:
            z_scores = np.zeros(len(df))
        else:
            z_scores = (df[col] - mean) / std
        scores[col] = z_scores

    # Composite score: average across all goals
    avg_score = np.mean([scores[c] for c in scores], axis=1)

    # Identify worst rows (lowest composite score)
    n_worst = max(1, int(len(rows) * 0.1))
    worst_idx = np.argsort(avg_score)[:n_worst]
    worst_mask = np.zeros(len(df), dtype=bool)
    worst_mask[worst_idx] = True

    # For each input column, measure how well it predicts being in the worst set
    correlations = {}
    for col in input_cols:
        vals = df[col].values
        if vals.dtype.kind == 'O':  # symbolic / categorical
            encoded = np.zeros(len(df))
            for i in range(len(df)):
                v = vals[i]
                if pd.isna(v):
                    continue
                if isinstance(v, str) and '?' not in v:
                    encoded[i] = 1.0
            correlations[col] = np.corrcoef(encoded, avg_score)[0, 1]
        else:
            correlations[col] = np.corrcoef(vals, avg_score)[0, 1]

    # Sort by absolute correlation descending and return top input columns
    sorted_cols = sorted(correlations.keys(), key=lambda c: abs(correlations[c]), reverse=True)
    return sorted_cols[:5]

    print(f"Top {len(result)} input columns explaining worst setups: {result}")

import pandas as pd
import numpy as np


def choose(header, rows, label):
    goal_cols = [h for h in header if h.endswith('-')]

    n_try = min(50, len(rows))

    best_idx = None
    best_score = -np.inf

    for i in range(n_try):
        goals = label(i)

        scores = {}
        for col in goal_cols:
            val = goals[col] if col in goals else '?'
            if isinstance(val, (int, float)):
                scores[col] = float(val)
            else:
                scores[col] = 0.0

        avg_score = np.mean(list(scores.values())) if len(scores) > 0 else -np.inf

        if avg_score > best_score:
            best_score = avg_score
            best_idx = i

    return best_idx

import pandas as pd
import numpy as np


def fix(header, row, levers):
    goal_cols = [h for h in header if h.endswith('-')]
    input_cols = [h for h in header if not h.endswith('-')]

    # Build a baseline DataFrame from the training data to compute stats
    df = pd.read_csv('pom3a.train.csv')
    for col in goal_cols:
        df[col] = df[col].replace('?', np.nan).astype(float)

    means = {}
    stds = {}
    for col in goal_cols:
        vals = df[col].dropna()
        if len(vals) > 0:
            means[col] = vals.mean()
            stds[col] = vals.std()
        else:
            means[col] = 0.0
            stds[col] = 1.0

    # Normalize goals to z-scores (low is good for '-', high is good for '+')
    def normalize(val, mean, std):
        if pd.isna(val) or val == '?':
            return 0.0
        try:
            v = float(val)
        except (ValueError, TypeError):
            return 0.0
        if std == 0:
            return 0.0
        return (v - mean) / std

    # Compute baseline score for the given row
    def compute_score(row_dict):
        scores = []
        for col in goal_cols:
            val = row_dict.get(col, '?')
            z = normalize(val, means[col], stds[col])
            scores.append(z)
        return np.mean(scores) if scores else 0.0

    # Try each lever combination to find the best one
    best_row = dict(row)
    best_score = compute_score(best_row)

    for col in levers:
        allowed = levers[col]
        if isinstance(allowed, (list, tuple)) and len(allowed) >= 2 and not isinstance(allowed[0], str):
            # Numeric lever: try low and high
            candidates = [allowed[0], allowed[-1]]
        else:
            # Symbolic lever: try all values
            candidates = list(allowed)

        for val in candidates:
            test_row = dict(best_row)
            if col.endswith('-'):
                test_row[col] = val
            else:
                test_row[col] = val
            score = compute_score(test_row)
            if score > best_score:
                best_score = score
                best_row = test_row

    return best_row
