import pandas as pd
import numpy as np


def justify(header, rows, row):
    goal_cols = [h for h in header if h.endswith('-')]
    input_cols = [h for h in header if not h.endswith('-')]

    df = pd.DataFrame(rows, columns=header)
    for col in goal_cols:
        df[col] = df[col].replace('?', np.nan).astype(float)

    scores = {}
    for col in goal_cols:
        vals = df[col].dropna()
        if len(vals) == 0:
            continue
        mean = vals.mean()
        std = vals.std()
        if std == 0:
            z_scores = np.zeros(len(df))
        else:
            z_scores = (df[col] - mean) / std
        scores[col] = z_scores

    avg_score = np.mean([scores[c] for c in scores], axis=1)

    target_row_idx = None
    for i, r in enumerate(rows):
        if all(r[j] == '?' or pd.isna(df.iloc[i][j]) for j in range(len(goal_cols))):
            # Check if this row's goal cells are "?"
            pass
    
    # Find the index of the query row (where goals are "?")
    target_idx = None
    for i, r in enumerate(rows):
        is_target = True
        for col in goal_cols:
            cell_val = df.iloc[i][col]
            if pd.notna(cell_val) and str(cell_val).strip() != '?':
                is_target = False
                break
        if is_target:
            target_idx = i
            break

    if target_idx is None:
        # If no row has "?" goals, treat the last row or compute normally
        return (0.5, "Could not identify a query setup with hidden goal values.")

    target_score = avg_score[target_idx]
    better_mask = avg_score > target_score
    p = float(np.mean(better_mask))

    rank = int(np.sum(avg_score >= target_score)) / len(rows) if len(rows) > 0 else 0.5
    
    reason = (
        f"This setup is better than {p:.1%} of all setups, "
        f"ranking in the top {int((1 - p) * 100)} percentile."
    )

    return (p, reason)


if __name__ == "__main__":
    df = pd.read_csv('pom3a.train.csv')
    header = list(df.columns)
    rows = [list(row) for row in df.values]

    # Find a representative query row (one with "?" goals) or use the first row
    query_row = None
    for r in rows:
        has_question = any(str(cell).strip() == '?' for cell in r)
        if has_question:
            query_row = r
            break

    if query_row is None:
        # Use a synthetic query with "?" goals to demonstrate the function
        sample_row = [0.5] * len(header)
        for col in header:
            if col.endswith('-'):
                sample_row[header.index(col)] = '?'
        query_row = sample_row

    p, reason = justify(header, rows, query_row)
    print(f"Result: {p:.1%} of setups are better. {reason}")
