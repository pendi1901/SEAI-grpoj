```python
import pandas as pd
from sklearn.linear_model import LinearRegression
import numpy as np


_df = pd.read_csv('pom3a.train.csv')
_header = list(_df.columns)
_rows = [list(row) for row in _df.values]

_goal_cols = [h for h in _header if h.endswith('-')]
_input_cols = [h for h in _header if not h.endswith('-')]

for col in _goal_cols:
    _df[col] = _df[col].replace('?', np.nan).astype(float)

_models = {}
_means = {}
_stds = {}
for col in _goal_cols:
    vals = _df[col].dropna()
    if len(vals) > 0:
        mean = vals.mean()
        std = vals.std()
        _means[col] = mean
        _stds[col] = std
        model = LinearRegression()
        X = _df[_input_cols].values
        y = vals.values
        mask = ~_df[col].isna()
        model.fit(X[mask], y)
        _models[col] = model


def flow(header, row, seen):
    df_row = pd.DataFrame([row], columns=header)
    for col in _goal_cols:
        df_row[col] = df_row[col].replace('?', np.nan).astype(float)

    preds = {}
    for col in _goal_cols:
        if col in _models:
            X_pred = np.array([row[c] for c in _input_cols])
            pred = _models[col].predict(X_pred)[0]
            preds[col] = float(pred)

    approve = True
    for col, mean in _means.items():
        if col not in preds:
            continue
        if preds[col] > mean:
            approve = False

    return approve


if __name__ == "__main__":
    result = []
    seen = []
    for row in _rows:
        approved = flow(_header, row, seen)
        if approved:
            result.append(row)
            print(f"Approved row {len(result)}")
        else:
            print(f"Rejected row {len(seen)+1}")

    print(f"\nFinal: Approved {len(result)} out of {len(_rows)} rows")
```
