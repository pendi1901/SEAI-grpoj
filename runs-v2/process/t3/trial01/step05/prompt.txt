import pandas as pd
import numpy as np


def remember(header, rows):
    goal_cols = [h for h in header if h.endswith('-')]

    df = pd.DataFrame(rows, columns=header)
    for col in goal_cols:
        df[col] = df[col].replace('?', np.nan).astype(float)

    scores = {}
    for col in goal_cols:
        vals = df[col].dropna()
        if len(vals) == 0:
            continue
        mean = vals.mean()
        std = vals.std()
        if std == 0:
            z_scores = np.zeros(len(df))
        else:
            z_scores = (df[col] - mean) / std
        scores[col] = z_scores

    if len(scores) > 0:
        avg_score = np.mean([scores[c] for c in scores], axis=1)
    else:
        return []

    n_rows = len(rows)
    max_return = max(1, int(n_rows * 0.05))
    top_indices = np.argsort(avg_score)[:max_return]

    return sorted(top_indices.tolist())


if __name__ == "__main__":
    df = pd.read_csv('pom3a.train.csv')
    header = list(df.columns)
    rows = [list(row) for row in df.values]
    result = remember(header, rows)
    print(f"Top {len(result)} out of {len(rows)} setups:")
    for idx in result:

            f"Score-={df.iloc[idx]['Score-']:.4f}, Idle-={df.iloc[idx]['Idle-']:.4f}"
        )

import pandas as pd
import numpy as np
from sklearn.linear_model import LinearRegression


def guess(header, train, test):
    df = pd.DataFrame(train, columns=header)

    goal_cols = [h for h in header if h.endswith('-')]
    input_cols = [h for h in header if not h.endswith('-')]

    for col in goal_cols:
        df[col] = df[col].replace('?', np.nan).astype(float)

    models = {}
    for col in goal_cols:
        X = df[input_cols].values
        y = df[col].dropna().values
        mask = ~df[col].isna()
        if len(y) > 0:
            model = LinearRegression()
            model.fit(X[mask], y)
            models[col] = model

    results = []
    for row in test:
        X_pred = np.array([row[c] for c in input_cols])
        pred = models['Cost-'].predict(X_pred)[0]
        results.append(float(pred))

    return results

import pandas as pd
from scipy import stats


def certify(xs, ys):
    """Return True if xs and ys are statistically indistinguishable (same apart from noise)."""
    try:
        _, p = stats.ttest_ind(xs, ys)
        return p > 0.05
    except Exception:
        return False
