#!/usr/bin/env python3 -B
"""step10.py: race three consultants on the same budget."""
import ezr


def race(header, rows, label):
    """Compare three strategies; return winner dict."""
    # Build a table from the hidden rows so column means are computed.
    tbl = ezr.Tbl(zip(header, rows))

    def fill(row):
        r = list(row)
        for i, h in enumerate(header):
            if h.endswith("-") and r[i] == "?":
                col_obj = tbl.cols[i]
                if hasattr(col_obj, "mu"):
                    r[i] = col_obj.mu
        return r

    # --- Strategy 1: random (must be one of them) -----------------
    def _random():
        n = len(rows)
        chosen = list(range(n))
        random.shuffle(chosen)
        best_idx, best_y = -1, 1e30
        for i in chosen:
            r = fill(rows[i])
            y = ezr.ranker(tbl)(r)
            if y < best_y:
                best_y = y
                best_idx = i
        return best_idx

    # --- Strategy 2: hill-climb (greedy, one step at a time) -----
    def _hillclimb():
        n = len(rows)
        chosen = list(range(n))
        random.shuffle(chosen)
        cur = fill(rows[chosen[0]])
        best_idx, best_y = chosen[0], ezr.ranker(tbl)(cur)
        for i in range(1, len(chosen)):
            nxt = fill(rows[chosen[i]])
            y = ezr.ranker(tbl)(nxt)
            if y < best_y:
                best_y = y
                best_idx = chosen[i]
                cur = nxt
        return best_idx

    # --- Strategy 3: random forest of small trees (diverse starts) -
    def _forest():
        n = len(rows)
        chosen = list(range(n))
        random.shuffle(chosen)
        best_idx, best_y = -1, 1e30
        for start in range(min(5, n)):
            seed_row = fill(rows[chosen[start]])
            # Grow a tree on the current set and score each candidate.
            sub = [rows[j] for j in chosen if j <= start + 20]
            tbl_sub = ezr.Tbl(zip(header, sub))
            ranker_sub = ezr.ranker(tbl_sub)
            for i in range(start + 1, min(start + 50, n)):
                r = fill(rows[i])
                y = ranker_sub(r)
                if y < best_y:
                    best_y = y
                    best_idx = i
        return best_idx

    # Run all three strategies.
    tried = ["random", "hillclimb", "forest"]
    results = {name: strategy() for name, strategy in zip(tried, [_random, _hillclimb, _forest])}

    # Pick the winner by lowest ydist (best row).
    best_name = min(results.keys(), key=lambda k: results[k])
    return {"winner": best_name, "row": results[best_name], "tried": tried}


if __name__ == "__main__":
    import os
    path = os.path.join(os.path.dirname(__file__) or ".", "pom3a.train.csv")
    with open(path, encoding="utf-8-sig") as f:
        lines = [l.strip() for l in f if l.strip()]
    header = [s for s in lines[0].split(",")]
    rows = [[ezr.atom(s) for s in line.split(",")] for line in lines[1:]]

    # Mask goal cells with "?" so the function sees hidden data.
    for row in rows:
        for j, col_name in enumerate(header):
            if col_name.endswith("-") and row[j] != "?":
                row[j] = "?"

    result = race(header, rows, label)
    print(f"Winner: {result['winner']} (row {result['row']+1} of {len(rows)})")
