def choose(header, rows, label):
    """Find the best setup among up to 50 trials."""
    goals = [c for c in header if c.endswith('-') and c[0].isupper()]

    n = len(rows)
    if n == 0:
        return -1

    import random
    indices = set(random.sample(range(n), min(50, n)))

    best_pos = None
    best_score = float('inf')

    for i in indices:
        true_row = label(i)
        score = sum(true_row[header.index(g)] for g in goals)
        if score < best_score:
            best_score = score
            best_pos = i

    return best_pos if best_pos is not None else -1


if __name__ == "__main__":
    import csv
    with open('pom3a.train.csv', 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        rows = list(reader)

    result = choose(header, rows, lambda i: [float(x) for x in rows[i]])
    print(f"Best setup position: {result}")
