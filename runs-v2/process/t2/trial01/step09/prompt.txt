def guess(header, train, test):
    """Predict Cost- (first goal column) for each test row."""
    goals = [c for c in header if c.endswith('-') and c[0].isupper()]
    first_goal = goals[0]  # "Cost-"

    n_train = len(train)
    if n_train == 0:
        return ["?"] * len(test)

    goal_idx = header.index(first_goal)
    input_cols = [c for c in header if c not in goals and c != first_goal]

    # Compute means of inputs and target from training data.
    m_target = sum(row[goal_idx] for row in train) / n_train
    m_inputs = {col: sum(row[header.index(col)] for row in train) / n_train for col in input_cols}

    # Build OLS coefficients using normal equations (manual implementation).
    p = len(input_cols)
    A = [[0.0] * (p + 1) for _ in range(p + 1)]
    b = [0.0] * (p + 1)

    for row in train:
        x = [row[header.index(col)] - m_inputs[col] for col in input_cols]
        y = row[goal_idx] - m_target
        for i in range(p):
            b[i] += x[i] * y
        for i in range(p):
            for j in range(p):
                A[i][j] += x[i] * x[j]
        A[i][p] = 0.0

    # Solve via Gaussian elimination on the augmented matrix [A | b].
    aug = [row[:] + [b[i]] for i, row in enumerate(A)]
    n = p + 1
    for col in range(n):
        max_row = col
        for r in range(col + 1, n):
            if abs(aug[r][col]) > abs(aug[max_row][col]):
                max_row = r
        aug[col], aug[max_row] = aug[max_row], aug[col]
        pivot = aug[col][col]
        if abs(pivot) < 1e-12:
            continue
        for r in range(col + 1, n):
            factor = aug[r][col] / pivot
            for c in range(col, n + 1):
                aug[r][c] -= factor * aug[col][c]

    coeffs = [aug[i][n] / aug[i][i] if abs(aug[i][i]) > 1e-12 else 0.0 for i in range(n)]

    predictions = []
    for row in test:
        pred = m_target + sum(coeffs[i] * (row[header.index(input_cols[i])] - m_inputs[input_cols[i]]) for i in range(p))
        predictions.append(round(pred, 6))

    return predictions

def certify(xs, ys):
    """Returns True if xs and ys are the same apart from noise."""
    n = min(len(xs), len(ys))
    if n == 0:
        return False

    mean_x = sum(xs[:n]) / n
    mean_y = sum(ys[:n]) / n

    num = sum((xs[i] - mean_x) * (ys[i] - mean_y) for i in range(n))
    den_x = sum((x - mean_x) ** 2 for x in xs[:n]) ** 0.5
    den_y = sum((y - mean_y) ** 2 for y in ys[:n]) ** 0.5

    if den_x == 0 or den_y == 0:
        return False

    r = num / (den_x * den_y)
    return abs(r) > 0.9


def certify_team_size_vs_cost(header, rows):
    """Test whether Team Size genuinely predicts Cost-."""
    team_idx = header.index('Team Size')
    cost_idx = header.index('Cost-')

    n = len(rows)
    if n == 0:
        return False

    x_vals = [row[team_idx] for row in rows]
    y_vals = [row[cost_idx] for row in rows]

    mean_x = sum(x_vals) / n
    mean_y = sum(y_vals) / n

    num = sum((x_vals[i] - mean_x) * (y_vals[i] - mean_y) for i in range(n))
    den_x = sum((x_vals[i] - mean_x) ** 2 for i in range(n)) ** 0.5
    den_y = sum((y_vals[i] - mean_y) ** 2 for i in range(n)) ** 0.5

    if den_x == 0 or den_y == 0:
        return False

    r = num / (den_x * den_y)

    # |r| > 0.9 means a real relationship; otherwise noise/luck
    return abs(r) > 0.9

def blame(header, rows):
    """Return names of input columns (not goals) that best explain why worst rows are bad."""
    goals = [c for c in header if c.endswith('-') and c[0].isupper()]
    
    n = len(rows)
    if n == 0:
        return []

    # Score each row; lower is better (goals end in -).
    scored = [(sum(row[header.index(g)] for g in goals), i) for i, row in enumerate(rows)]
    scored.sort(reverse=True)  # worst first

    k = max(1, int(n * 0.05))
    bad_indices = set(pos for _, pos in scored[:k])

    input_cols = [c for c in header if c not in goals]

    results = []
    for col in input_cols:
        x_vals = [row[header.index(col)] for row in rows]
        y_vals = [1.0 if i in bad_indices else 0.0 for i in range(n)]

        mean_x = sum(x_vals) / n
        mean_y = sum(y_vals) / n

        num = sum((x_vals[i] - mean_x) * (y_vals[i] - mean_y) for i in range(n))
        den_x = sum((x_vals[i] - mean_x) ** 2 for i in range(n)) ** 0.5
        den_y = sum((y_vals[i] - mean_y) ** 2 for i in range(n)) ** 0.5

        if den_x == 0 or den_y == 0:
            r = 0.0
        else:
            r = num / (den_x * den_y)

        results.append((abs(r), col, r))

    # Sort by absolute correlation descending; return column names in that order.
    results.sort(reverse=True)
    return [col for _, col, _ in results]

def choose(header, rows, label):
    """Find the best setup among up to 50 trials."""
    goals = [c for c in header if c.endswith('-') and c[0].isupper()]

    n = len(rows)
    if n == 0:
        return -1

    import random
    indices = set(random.sample(range(n), min(50, n)))

    best_pos = None
    best_score = float('inf')

    for i in indices:
        true_row = label(i)
        score = sum(true_row[header.index(g)] for g in goals)
        if score < best_score:
            best_score = score
            best_pos = i

    return best_pos if best_pos is not None else -1

import csv


def fix(header, row, levers):
    """Return {column: new_value} for the smallest change that rescues a bad row."""
    goals = [c for c in header if c.endswith('-') and c[0].isupper()]

    # Identify which goal columns are currently bad (above their best).
    n = len(row)
    current_score = sum(float(row[header.index(g)]) for g in goals)

    # For each lever, compute how much the score improves if we pick the extreme value.
    improvements = {}  # {lever: max improvement}
    for col, values in levers.items():
        best_improve = -float('inf')
        for v in values:
            new_row = list(row)
            idx = header.index(col)
            if isinstance(v, (int, float)):
                new_row[idx] = v
            else:
                new_row[idx] = v
            new_score = sum(float(new_row[header.index(g)]) for g in goals)
            improve = current_score - new_score  # positive means better
            best_improve = max(best_improve, improve)
        improvements[col] = best_improve

    if not improvements:
        return {}

    # Pick the lever with the largest improvement.
    chosen = max(improvements, key=improvements.get)
    new_val = None
    for v in levers[chosen]:
        new_row = list(row)
        idx = header.index(chosen)
        if isinstance(v, (int, float)):
            new_row[idx] = v
        else:
            new_row[idx] = v
        new_score = sum(float(new_row[header.index(g)]) for g in goals)
        improve = current_score - new_score
        if improve > 0:
            new_val = v
            break

    return {chosen: new_val}
