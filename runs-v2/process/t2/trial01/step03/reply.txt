import csv
from step01 import remember
from step02 import guess


def certify(xs, ys):
    """Returns True if xs and ys are the same apart from noise."""
    n = min(len(xs), len(ys))
    if n == 0:
        return False

    mean_x = sum(xs[:n]) / n
    mean_y = sum(ys[:n]) / n

    num = sum((xs[i] - mean_x) * (ys[i] - mean_y) for i in range(n))
    den_x = sum((x - mean_x) ** 2 for x in xs[:n]) ** 0.5
    den_y = sum((y - mean_y) ** 2 for y in ys[:n]) ** 0.5

    if den_x == 0 or den_y == 0:
        return False

    r = num / (den_x * den_y)
    return abs(r) > 0.9


def certify_team_size_vs_cost(header, rows):
    """Test whether Team Size genuinely predicts Cost-."""
    team_idx = header.index('Team Size')
    cost_idx = header.index('Cost-')

    n = len(rows)
    if n == 0:
        return False

    x_vals = [row[team_idx] for row in rows]
    y_vals = [row[cost_idx] for row in rows]

    mean_x = sum(x_vals) / n
    mean_y = sum(y_vals) / n

    num = sum((x_vals[i] - mean_x) * (y_vals[i] - mean_y) for i in range(n))
    den_x = sum((x_vals[i] - mean_x) ** 2 for i in range(n)) ** 0.5
    den_y = sum((y_vals[i] - mean_y) ** 2 for i in range(n)) ** 0.5

    if den_x == 0 or den_y == 0:
        return False

    r = num / (den_x * den_y)

    # |r| > 0.9 means a real relationship; otherwise noise/luck
    return abs(r) > 0.9


if __name__ == "__main__":
    with open('pom3a.train.csv', 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        rows = list(reader)

    best = remember(header, rows)
    print(f"Top {len(best)} of {len(rows)} setups:")
    for pos in best:
        print(f"  Row {pos}")

    preds = guess(header, rows, [])
    print(f"\nPredicted Cost- range: {min(preds):.4f} to {max(preds):.4f}")

    result = certify_team_size_vs_cost(header, rows)
    if result is True:
        print("Small-team setups are genuinely cheaper than big-team ones.")
    else:
        print("The apparent cost difference between team sizes was just noise/luck.")
