import ezr

def remember(header, rows):
    tbl = ezr.Tbl([header] + rows)
    scored = [(ezr.ydist(tbl, r), i) for i, r in enumerate(rows)]
    scored.sort(key=lambda x: x[0])
    n = len(rows)
    cap = max(1, int(n * 0.05))
    return [idx for _, idx in scored[:cap]]

if __name__ == "__main__":
    import os
    path = 'pom3a.train.csv'
    with open(path) as f:
        lines = [l.strip() for l in f if l.strip()]
    header = lines[0].split(',')
    rows = ezr.csv(path)
    result = remember(header, rows)
    print(f"Top {len(result)} of {len(rows)} setups:")
    for i in result:
        print(f"  Row {i}")
