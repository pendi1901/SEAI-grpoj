import ezr

def spot(header, rows):
    """Return row indices whose inputs look mistyped or made up."""
    # Per-input column stats (skip goal columns)
    n = len(rows)
    suspicious = []
    
    for at in range(len(header)):
        col_name = header[at]
        if not col_name[0].isupper():  # skip symbolic goals
            continue
        
        vals = [float(r[at]) for r in rows if isinstance(r[at], (int, float))]
        if len(vals) < 3:
            continue
        
        mu = sum(vals) / len(vals)
        m2 = sum((v - mu)**2 for v in vals)
        sd = (m2 / (len(vals) - 1)) ** 0.5 if len(vals) > 1 else 0
        
        # Flag rows where this input is far from the norm (>3 SD)
        for i, r in enumerate(rows):
