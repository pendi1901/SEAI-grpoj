import ezr

def justify(header, rows, row):
    tbl = ezr.Tbl([header] + [list(r) for r in rows])
    rank_fn = ezr.ranker(tbl)
    scored = [(ezr.ydist(tbl, list(r)), i) for i, r in enumerate(rows)]
    n = len(rows)
    pred = rank_fn(list(row))
    better = sum(1 for _, i in scored if ezr.ydist(tbl, list(rows[i])) > pred)
    p = better / n
    return (p, f"This setup is better than {int(p*100)}% of all setups.")

if __name__ == "__main__":
    import os
    path = 'pom3a.train.csv'
    with open(path) as f:
        lines = [l.strip() for l in f if l.strip()]
    header = lines[0].split(',')
    rows = ezr.csv(path)
    row = list(rows[0])
    p, reason = justify(header, rows, row)
    print(f"{reason} (p={p:.3f})")
