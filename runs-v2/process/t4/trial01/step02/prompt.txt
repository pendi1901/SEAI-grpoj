import pandas as pd
import numpy as np


def remember(header, rows):
    goal_cols = [c for c in header if c.endswith('-')]

    df = pd.DataFrame(rows, columns=header)

    # For each goal column (lower is better), compute percentile rank:
    #   smallest value → score 1.0, largest → score ~0.0
    scores_list = []
    for col in goal_cols:
        vals = np.array(df[col])
        sorted_idx = np.argsort(vals)
        ranks = np.zeros(len(vals))
        ranks[sorted_idx] = np.arange(len(vals), 0, -1) / len(vals)
        scores_list.append(ranks)

    total_score = sum(scores_list).sum(axis=0)

    # Top ~5% of rows (at least one) by combined goal score
    threshold = np.percentile(total_score, 100 - 5)
    best_positions = np.where(total_score >= threshold)[0].tolist()

    return list(best_positions)
