import pandas as pd
import numpy as np


def remember(header, rows):
    goal_cols = [c for c in header if c.endswith('-')]

    df = pd.DataFrame(rows, columns=header)

    # For each goal column (lower is better), compute percentile rank:
    #   smallest value → score 1.0, largest → score ~0.0
    scores_list = []
    for col in goal_cols:
        vals = np.array(df[col])
        sorted_idx = np.argsort(vals)
        ranks = np.zeros(len(vals))
        ranks[sorted_idx] = np.arange(len(vals), 0, -1) / len(vals)
        scores_list.append(ranks)

    total_score = sum(scores_list).sum(axis=0)

    # Top ~5% of rows (at least one) by combined goal score
    threshold = np.percentile(total_score, 100 - 5)
    best_positions = np.where(total_score >= threshold)[0].tolist()

    return list(best_positions)

import pandas as pd
import numpy as np


def certify(xs, ys):
    xs = np.array(xs)
    ys = np.array(ys)
    n_x, n_y = len(xs), len(ys)
    mean_x = xs.mean() if n_x > 0 else 0.0
    mean_y = ys.mean() if n_y > 0 else 0.0
    var_x = xs.var(ddof=1) if n_x > 1 else 0.0
    var_y = ys.var(ddof=1) if n_y > 1 else 0.0
    se = np.sqrt(var_x / n_x + var_y / n_y) if (var_x / n_x + var_y / n_y) > 0 else 0.0
    t_stat = abs(mean_x - mean_y) / se if se > 0 else float('inf')
    return t_stat < 2.0

def flow(header, row, seen):
    """Decide whether to approve the current proposal."""
    goal_cols = [c for c in header if c.endswith('-')]

    # Build a quick score from goals: lower is better.
    def goal_score(cells):
        s = 0.0
        for col in goal_cols:
            v = float(cells[header.index(col)])
            s += -v  # minimise each '-' column
        return s

    row_s = goal_score(row)
    seen_scores = [goal_score(r) for r in seen] if seen else []

    # If we have a history, use the best historical score as a benchmark.
    if seen_scores:
        best_seen = min(seen_scores)
        # Approve only when this row is at least as good as any seen so far.
        return row_s <= best_seen + 1e-9

    # No history yet — approve the first one (or use a simple heuristic).
    return True

def blame(header, rows):
    goal_cols = [c for c in header if c.endswith('-')]

    n = len(rows)
    scores = np.zeros(n)
    for col in goal_cols:
        vals = np.array([float(r[header.index(col)]) for r in rows])
        sorted_idx = np.argsort(vals)
        ranks = np.zeros(n)
        ranks[sorted_idx] = np.arange(n, 0, -1) / n
        scores += ranks

    # Worst positions (highest combined score = worst performance)
    threshold = np.percentile(scores, 95)
    worst_positions = [i for i in range(n) if scores[i] >= threshold]

    input_cols = [c for c in header if not c.endswith('-')]

    results = []
    for col in input_cols:
        vals_worst = []
        vals_rest = []
        for i, r in enumerate(rows):
            val = float(r[col])
            if i in worst_positions:
                vals_worst.append(val)
            else:
                vals_rest.append(val)

        mean_w = np.mean(vals_worst) if vals_worst else 0.0
        mean_r = np.mean(vals_rest) if vals_rest else 0.0

        diff = abs(mean_w - mean_r)
        results.append((col, diff))

    results.sort(key=lambda x: -x[1])

    return [f"{c}: avg worst={m:.3f}, avg rest={r:.3f}" for c, (m, r) in zip([r[0] for r in results], results)]

    # For each input column, compute the average value among top-ranked rows.
    n_ref = len(ref_rows)
    col_means_top = {}
    for col in input_cols:
        vals_top = []

            if i in best_positions:
                vals_top.append(v)
            else:
                vals_rest.append(v)
        col_means_top[col] = (np.mean(vals_top), np.mean(vals_rest))

    # Now evaluate each candidate row.
    best_pos = -1

            # Penalise deviation from the top-ranked average.

            best_score = score
            best_pos = i

    return int(best_pos)

def fix(header, row, levers):

    # Load training data to find good configurations.
    df = __import__('pandas').read_csv('pom3a.train.csv')
    train_rows = [list(r) for _, r in df.iterrows()]

    best_positions = remember(header, train_rows)

    input_cols = [c for c in header if not c.endswith('-')]

    # For each lever, pick the value from top training setups that is closest to the current row.
    results = {}
    for col in levers:
        allowed = levers[col]

        candidates = []
        for i in best_positions[:20]:
            val = float(train_rows[i][header.index(col)])
            if isinstance(allowed, tuple):
                low, high = allowed
                if low <= val <= high:
                    candidates.append(val)

        if candidates:
            # Smallest change: pick the candidate closest to current value.
            results[col] = min(candidates, key=lambda v: abs(v - float(row[header.index(col)])))

    # Must have changed at least one lever; fallback to midpoint of allowed range.
    if not any(results.values()):
        for col in levers:
            allowed = levers[col]
            if isinstance(allowed, tuple):
                low, high = allowed
                results[col] = (low + high) / 2.0
                break

    return results
