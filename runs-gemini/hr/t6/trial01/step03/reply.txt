```python
import ezr

def certify(xs, ys):
    """
    Returns True if xs and ys are statistically indistinguishable (noise),
    False if they are significantly different.
    """
    # ezr.same uses cliffs and ks tests to determine if two distributions are the same.
    return ezr.same(xs, ys)

if __name__ == "__main__":
    # Load the data
    data = list(ezr.csv("WA_Fn-UseC_-HR-Employee-Attrition.train.csv"))
    header = data[0]
    rows = data[1:]
    
    # Identify columns
    # 'overTime' is symbolic, 'YearsInCurrentRole+' is numeric
    idx_overtime = header.index("overTime")
    idx_tenure = header.index("YearsInCurrentRole+")
    
    # Split data into two groups: those who work overtime and those who don't
    yes_overtime = []
    no_overtime = []
    
    for row in rows:
        val = row[idx_tenure]
        if val != "?":
            if row[idx_overtime] == "Yes":
                yes_overtime.append(val)
            elif row[idx_overtime] == "No":
                no_overtime.append(val)
    
    # Perform the statistical test
    is_same = certify(yes_overtime, no_overtime)
    
    # Output results
    print(f"Analyzed {len(rows)} records.")
    print(f"Average tenure (YearsInCurrentRole+):")
    print(f" - Overtime: {sum(yes_overtime)/len(yes_overtime):.2f} years")
    print(f" - No Overtime: {sum(no_overtime)/len(no_overtime):.2f} years")
    
    if is_same:
        print("\nConclusion: The difference is statistically insignificant (likely noise).")
    else:
        print("\nConclusion: The difference is statistically significant (not just noise).")
    
    exit(0)
```
