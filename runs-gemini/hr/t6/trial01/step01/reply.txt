import ezr
import random

def remember(header, rows):
    """
    Find rows that stay 'unusually long'.
    'YearsAtCompany+' and 'YearsInCurrentRole+' are goals to maximize.
    We define 'best' as those with the highest values in these columns.
    """
    # Identify indices for the goal columns
    idx_years_at = header.index("YearsAtCompany+")
    idx_years_role = header.index("YearsInCurrentRole+")
    
    # Calculate a score: sum of normalized goals (or just raw sum if scales are similar)
    # Since we want to maximize, we sort by the sum of these two columns.
    scores = []
    for i, row in enumerate(rows):
        # Handle potential "?" if necessary, though problem implies numeric data
        val1 = row[idx_years_at] if row[idx_years_at] != "?" else 0
        val2 = row[idx_years_role] if row[idx_years_role] != "?" else 0
        scores.append((val1 + val2, i))
    
    # Sort descending by score
    scores.sort(key=lambda x: x[0], reverse=True)
    
    # Return top 5% (approx 58 rows for 1176 data rows)
    limit = max(1, int(len(rows) * 0.05))
    return [item[1] for item in scores[:limit]]

if __name__ == "__main__":
    # Load the data
    data = list(ezr.csv("WA_Fn-UseC_-HR-Employee-Attrition.train.csv"))
    header = data[0]
    rows = data[1:]
    
    # Get the indices of the best rows
    best_indices = remember(header, rows)
    
    # Print a summary for the customer
    print(f"Analyzed {len(rows)} employee records.")
    print(f"Identified {len(best_indices)} employees with the longest tenure.")
    print("\nTop 5 examples of long-tenured staff (YearsAtCompany, YearsInCurrentRole):")
    
    idx_at = header.index("YearsAtCompany+")
    idx_role = header.index("YearsInCurrentRole+")
    
    for i in best_indices[:5]:
        row = rows[i]
        print(f" - Employee at company for {row[idx_at]} years, "
              f"in current role for {row[idx_role]} years.")
    
    exit(0)
