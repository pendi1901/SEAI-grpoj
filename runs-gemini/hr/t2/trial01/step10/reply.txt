import csv
import random
import step01
import step07

def race(header, rows, label):
    """
    Compares three strategies for finding the best employee (highest tenure).
    Budget: 50 labels total.
    """
    n = len(rows)
    goal_idx = next(i for i, h in enumerate(header) if h.endswith('+'))
    
    # Strategies
    def strategy_random(budget):
        indices = random.sample(range(n), budget)
        best_i, best_val = -1, -1
        for i in indices:
            val = label(i)[0]
            if val > best_val:
                best_val, best_i = val, i
        return best_i, best_val

    def strategy_top_tenure_guess(budget):
        # Based on step01 logic: look at top 5% of the whole dataset
        # We use our budget to sample from the top 5% of the dataset
        indices = step01.remember(header, rows)
        sample_size = min(budget, len(indices))
        chosen = random.sample(indices, sample_size)
        best_i, best_val = -1, -1
        for i in chosen:
            val = label(i)[0]
            if val > best_val:
                best_val, best_i = val, i
        return best_i, best_val

    def strategy_step07_sample(budget):
        # Uses the logic from step07 (strided sampling)
        step = max(1, n // budget)
        indices = [i for i in range(0, n, step)][:budget]
        best_i, best_val = -1, -1
        for i in indices:
            val = label(i)[0]
            if val > best_val:
                best_val, best_i = val, i
        return best_i, best_val

    # Run race
    results = {
        "random": strategy_random(50),
        "tenure_focus": strategy_top_tenure_guess(50),
        "strided_sample": strategy_step07_sample(50)
    }

    winner = max(results, key=lambda k: results[k][1])
    
    return {
        "winner": winner,
        "row": results[winner][0],
        "tried": list(results.keys())
    }

if __name__ == "__main__":
    filename = "WA_Fn-UseC_-HR-Employee-Attrition.train.csv"
    rows = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = []
            for i, val in enumerate(row):
                if header[i][0].isupper():
                    try:
                        processed_row.append(float(val))
                    except ValueError:
                        processed_row.append(val)
                else:
                    processed_row.append(val)
            rows.append(processed_row)

    true_goals = []
    goal_indices = [i for i, h in enumerate(header) if h.endswith('+') or h.endswith('-')]
    for row in rows:
        true_goals.append([row[idx] for idx in goal_indices])
        for idx in goal_indices:
            row[idx] = "?"

    def label(i):
        return true_goals[i]

    outcome = race(header, rows, label)
    
    print(f"Race complete. The winning strategy was: {outcome['winner']}")
    print(f"Best employee found at index {outcome['row']} with tenure {true_goals[outcome['row']][0]} years.")
    print(f"Strategies evaluated: {', '.join(outcome['tried'])}")
