def remember(header, rows):
    """
    Identifies staff who stay 'unusually long'.
    'YearsAtCompany+' is a goal to maximize.
    We define 'unusually long' as the top 5% of employees by tenure.
    """
    # Find index for YearsAtCompany+
    try:
        idx = header.index("YearsAtCompany+")
    except ValueError:
        return [0]

    # Extract tenure and original index
    tenures = []
    for i, row in enumerate(rows):
        try:
            val = float(row[idx])
            tenures.append((val, i))
        except (ValueError, TypeError):
            continue

    # Sort by tenure descending
    tenures.sort(key=lambda x: x[0], reverse=True)

    # Return top 5% (or at least 1)
    limit = max(1, len(rows) // 20)
    return [item[1] for item in tenures[:limit]]

import csv

def certify(xs, ys):
    """
    Performs a simple two-sample t-test approximation.
    Returns True if the means are close (noise), False if they differ significantly.
    """
    n1, n2 = len(xs), len(ys)
    if n1 < 2 or n2 < 2:
        return True
    
    m1, m2 = sum(xs) / n1, sum(ys) / ys
    v1 = sum((x - m1) ** 2 for x in xs) / (n1 - 1)
    v2 = sum((y - m2) ** 2 for y in ys) / (n2 - 1)
    
    # Standard error of the difference
    se = ((v1 / n1) + (v2 / n2)) ** 0.5
    if se == 0:
        return m1 == m2
    
    # t-statistic
    t = abs(m1 - m2) / se
    # Threshold for 95% confidence (approx 1.96 for large samples)
    return t < 1.96

def flow(header, row, seen):
    """
    Decides if the current employee needs a 'what would keep you here' conversation.
    We identify 'at-risk' employees as those with low tenure compared to the 
    average of the historical data (the 'seen' rows).
    """
    # Identify goal columns (those ending in +)
    goal_indices = [i for i, h in enumerate(header) if h.endswith('+')]
    if not goal_indices:
        return False
    
    # Use the first goal column: YearsAtCompany+
    target_idx = goal_indices[0]
    
    # If this is the first row, we have no baseline, so don't trigger
    if not seen:
        return False
    
    # Calculate baseline from seen data
    tenures = [r[target_idx] for r in seen if isinstance(r[target_idx], (int, float))]
    if not tenures:
        return False
    
    avg_tenure = sum(tenures) / len(tenures)
    
    # Current employee's tenure
    current_val = row[target_idx]
    
    # If current tenure is significantly lower than the average of those seen,
    # flag them for a retention conversation.
    return current_val < (avg_tenure * 0.5)

def blame(header, rows):
    """
    Identifies input columns that correlate with short tenure (YearsAtCompany+).
    'Worst' rows are those with the lowest tenure.
    We compare the average of input columns in the bottom 20% of tenure
    against the rest of the population.
    """
    tenure_idx = header.index("YearsAtCompany+")
    
    # Sort rows by tenure to find the 'worst' (shortest stayers)
    sorted_rows = sorted(rows, key=lambda r: r[tenure_idx])
    cutoff = max(1, len(rows) // 5)
    worst = sorted_rows[:cutoff]
    rest = sorted_rows[cutoff:]
    
    # Identify input columns (exclude goals)
    input_indices = [i for i, h in enumerate(header) if not (h.endswith('+') or h.endswith('-'))]
    
    differences = []
    for i in input_indices:
        col_name = header[i]
        # Only analyze numeric inputs for simple difference comparison
        if isinstance(worst[0][i], (int, float)):
            avg_worst = sum(r[i] for r in worst) / len(worst)
            avg_rest = sum(r[i] for r in rest) / len(rest)
            # We look for features where the 'worst' group has significantly lower values
            # than the rest, suggesting a potential cause for short stays.
            diff = avg_rest - avg_worst
            differences.append((diff, col_name))
            
    # Sort by the largest positive difference (where 'worst' are significantly lower)
    differences.sort(key=lambda x: x[0], reverse=True)
    
    return [d[1] for d in differences[:3]]

import csv

def justify(header, rows, row):
    """
    Ranks a row's performance against historical data based on goal columns.
    A row is 'better' if its goal values are higher for '+' columns 
    and lower for '-' columns.
    """
    goal_indices = [i for i, h in enumerate(header) if h.endswith('+') or h.endswith('-')]
    
    better_count = 0
    total_rows = len(rows)
    
    for r in rows:
        is_better = True
        for idx in goal_indices:
            col_name = header[idx]
            val_r = r[idx]
            val_row = row[idx]
            
            # Skip if the row being tested has unknown goal values
            if val_row == "?":
                is_better = False
                break
            
            if col_name.endswith('+'):
                if not (val_row >= val_r):
                    is_better = False
                    break
            elif col_name.endswith('-'):
                if not (val_row <= val_r):
                    is_better = False
                    break
        
        if is_better:
            better_count += 1
            
    p = better_count / total_rows if total_rows > 0 else 0
    reason = f"This team's arrangement outperforms {p:.1%} of historical records based on tenure and retention goals."
    
    return (p, reason)

def choose(header, rows, label):
    """
    Finds the best working arrangement by sampling 50 rows.
    We pick 50 rows that are diverse or likely to be high-performing,
    label them, and return the index of the one with the highest tenure.
    """
    # Identify goal indices
    goal_indices = [i for i, h in enumerate(header) if h.endswith('+') or h.endswith('-')]
    tenure_idx = header.index("YearsAtCompany+")
    
    # We have a budget of 50 labels.
    # We'll sample 50 rows spread across the dataset to get a representative view.
    n = len(rows)
    step = max(1, n // 50)
    indices_to_sample = [i for i in range(0, n, step)][:50]
    
    best_idx = -1
    max_tenure = -1.0
    
    for i in indices_to_sample:
        # label(i) returns the true goal values for rows[i]
        goals = label(i)
        # The first goal is YearsAtCompany+
        tenure = goals[0]
        
        if tenure > max_tenure:
            max_tenure = tenure
            best_idx = i
            
    return best_idx
