import pandas as pd
import numpy as np

def remember(header, rows):
    """
    Identifies staff who stay 'unusually long'.
    We define 'unusually long' as those in the top 5% of 'YearsAtCompany+'
    among those who did not leave (attrition == 'No').
    """
    df = pd.DataFrame(rows, columns=header)
    
    # Filter for those who stay (attrition == 'No')
    stayed = df[df['attrition'] == 'No'].copy()
    
    # Convert goal column to numeric
    stayed['YearsAtCompany+'] = pd.to_numeric(stayed['YearsAtCompany+'])
    
    # Find the threshold for the top 5%
    threshold = stayed['YearsAtCompany+'].quantile(0.95)
    
    # Identify rows that meet the criteria
    best_rows = stayed[stayed['YearsAtCompany+'] >= threshold]
    
    # Return the original 0-based indices
    return best_rows.index.tolist()

def certify(xs, ys):
    """
    Returns True if the difference between two groups is statistically 
    insignificant (noise), False if they really differ (p-value < 0.05).
    """
    t_stat, p_value = stats.ttest_ind(xs, ys, equal_var=False)
    return p_value >= 0.05

def flow(header, row, seen):
    """
    Decides if the current row needs a 'what would keep you here' conversation.
    We identify 'high-value' employees as those who stay long (based on 
    the logic in step01) and are currently at risk of attrition.
    
    Since we cannot go back, we use the threshold established by the 
    historical data (seen) to decide if the current row is a 'top performer' 
    in terms of tenure who might be worth retaining.
    """
    df_seen = pd.DataFrame(seen, columns=header)
    
    # Identify goal columns (ending in +)
    # We want to retain people with high 'YearsAtCompany+'
    tenure_idx = header.index('YearsAtCompany+')
    attrition_idx = header.index('attrition')
    
    # If we haven't seen enough data, wait
    if len(seen) < 50:
        return False
    
    # Calculate threshold from seen data for those who stayed
    stayed = df_seen[df_seen['attrition'] == 'No']
    if len(stayed) < 10:
        return False
        
    threshold = stayed['YearsAtCompany+'].quantile(0.95)
    
    # Current row values
    current_tenure = row[tenure_idx]
    current_attrition = row[attrition_idx]
    
    # Logic: If they are a long-tenured employee (top 5%) 
    # and they are currently marked as 'Yes' for attrition, 
    # flag them for a conversation.
    if current_tenure >= threshold and current_attrition == 'Yes':
        return True
        
    return False
