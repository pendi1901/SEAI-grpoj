import pandas as pd

def choose(header, rows, label):
    """
    Finds the working arrangement that keeps people longest by sampling 50 rows.
    We use the first 50 rows to label, then identify the best working arrangement
    (BusinessTravel, OverTime, WorkLifeBalance) based on the highest 'YearsAtCompany+'.
    """
    # We have a budget of 50 interviews
    budget = 50
    
    # Identify goal column
    goal_col = [c for c in header if c.endswith('+')][0]
    goal_idx = header.index(goal_col)
    
    # Label the first 50 rows to get ground truth
    labeled_data = []
    for i in range(min(len(rows), budget)):
        true_goals = label(i)
        row = rows[i][:]
        row[goal_idx] = true_goals[0]
        labeled_data.append(row)
    
    # Convert to DataFrame for analysis
    df_labeled = pd.DataFrame(labeled_data, columns=header)
    
    # Group by working arrangements and find the mean tenure
    arrangements = ['businessTravel', 'overTime', 'WorkLifeBalance']
    best_group = df_labeled.groupby(arrangements)[goal_col].mean().idxmax()
    
    # Find the row index that matches the best arrangement and has the highest tenure
    best_idx = -1
    max_tenure = -1
    
    for i in range(len(labeled_data)):
        row = labeled_data[i]
        match = all(row[header.index(col)] == val for col, val in zip(arrangements, best_group))
        if match and row[goal_idx] > max_tenure:
            max_tenure = row[goal_idx]
            best_idx = i
            
    return best_idx

    df = pd.read_csv('WA_Fn-UseC_-HR-Employee-Attrition.train.csv')

    def label(i):
        # Return the goal columns for row i

        return [rows[i][header.index(g)] for g in goals]
    
    # Mask goals for the rows passed to choose
    for row in rows:
        for i, col in enumerate(header):
            if col.endswith('+') or col.endswith('-'):
                row[i] = "?"
                
    best_row_idx = choose(header, rows, label)
    
    # Retrieve the best row details
    best_row = dict(zip(header, rows[best_row_idx]))
    arrangements = {k: best_row[k] for k in ['businessTravel', 'overTime', 'WorkLifeBalance']}

    print(f"The optimal arrangement found is: {arrangements}")
