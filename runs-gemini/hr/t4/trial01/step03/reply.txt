import pandas as pd
import numpy as np
from scipy import stats
from step01 import remember

def certify(xs, ys):
    """
    Returns True if the difference between two groups is statistically 
    insignificant (noise), False if they really differ (p-value < 0.05).
    """
    t_stat, p_value = stats.ttest_ind(xs, ys, equal_var=False)
    return p_value >= 0.05

if __name__ == "__main__":
    df = pd.read_csv('WA_Fn-UseC_-HR-Employee-Attrition.train.csv')
    
    # We look at 'YearsInCurrentRole+' as a proxy for how long they stay in a role.
    # We compare those who work overtime vs those who do not.
    overtime_yes = df[df['overTime'] == 'Yes']['YearsInCurrentRole+']
    overtime_no = df[df['overTime'] == 'No']['YearsInCurrentRole+']
    
    is_noise = certify(overtime_yes.tolist(), overtime_no.tolist())
    
    print("Analysis: Does working overtime cause people to leave their role sooner?")
    if is_noise:
        print("Result: The observed difference is likely just noise (statistically insignificant).")
    else:
        print("Result: There is a statistically significant difference in how long people stay in their roles based on overtime.")
        
    avg_yes = overtime_yes.mean()
    avg_no = overtime_no.mean()
    print(f"Average years in current role (Overtime): {avg_yes:.2f}")
    print(f"Average years in current role (No Overtime): {avg_no:.2f}")
