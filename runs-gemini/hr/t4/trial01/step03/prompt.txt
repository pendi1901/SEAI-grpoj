import pandas as pd
import numpy as np

def remember(header, rows):
    """
    Identifies staff who stay 'unusually long'.
    We define 'unusually long' as those in the top 5% of 'YearsAtCompany+'
    among those who did not leave (attrition == 'No').
    """
    df = pd.DataFrame(rows, columns=header)
    
    # Filter for those who stay (attrition == 'No')
    stayed = df[df['attrition'] == 'No'].copy()
    
    # Convert goal column to numeric
    stayed['YearsAtCompany+'] = pd.to_numeric(stayed['YearsAtCompany+'])
    
    # Find the threshold for the top 5%
    threshold = stayed['YearsAtCompany+'].quantile(0.95)
    
    # Identify rows that meet the criteria
    best_rows = stayed[stayed['YearsAtCompany+'] >= threshold]
    
    # Return the original 0-based indices
    return best_rows.index.tolist()

import pandas as pd

from sklearn.ensemble import RandomForestRegressor
from sklearn.preprocessing import LabelEncoder

def guess(header, train, test):
    """
    Estimates the first goal column ('YearsAtCompany+') for test rows
    using a Random Forest Regressor.
    """
    df_train = pd.DataFrame(train, columns=header)
    df_test = pd.DataFrame(test, columns=header)
    
    # Identify the target column (first column ending in + or -)
    # Based on the prompt, 'YearsAtCompany+' is the first goal column.
    target_col = [c for c in header if c.endswith('+') or c.endswith('-')][0]
    
    # Prepare data: Encode symbolic columns
    df_all = pd.concat([df_train, df_test], axis=0)
    
    le_map = {}
    for col in df_all.columns:
        if df_all[col].dtype == 'object':
            le = LabelEncoder()
            # Convert to string to handle potential mixed types
            df_all[col] = le.fit_transform(df_all[col].astype(str))
            le_map[col] = le
            
    # Split back
    train_encoded = df_all.iloc[:len(df_train)]
    test_encoded = df_all.iloc[len(df_train):]
    
    X_train = train_encoded.drop(columns=[target_col])
    y_train = train_encoded[target_col].astype(float)
    X_test = test_encoded.drop(columns=[target_col])
    
    # Train model
    model = RandomForestRegressor(n_estimators=100, random_state=42)
    model.fit(X_train, y_train)
    
    # Predict
    predictions = model.predict(X_test)
    return predictions.tolist()
