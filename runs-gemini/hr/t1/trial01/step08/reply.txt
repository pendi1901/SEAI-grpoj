import csv

def fix(header, row, levers):
    """
    Recommends changes to a 'bad' row to improve its goal columns.
    A row is 'bad' if it has low tenure (YearsAtCompany+).
    We iterate through levers and pick the one that maximizes the goal.
    """
    tenure_idx = header.index("YearsAtCompany+")
    recommendations = {}

    # Simple heuristic: for each lever, try the 'best' possible value
    # and pick the one that seems most beneficial.
    for col, allowed in levers.items():
        idx = header.index(col)
        if isinstance(allowed, list):
            # Symbolic: pick the last one as a heuristic for 'better'
            recommendations[col] = allowed[-1]
        else:
            # Numeric: pick the high end of the range
            recommendations[col] = allowed[1]
            
    return recommendations

if __name__ == "__main__":
    filename = "WA_Fn-UseC_-HR-Employee-Attrition.train.csv"
    rows = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = []
            for i, val in enumerate(row):
                if header[i][0].isupper():
                    try:
                        processed_row.append(float(val))
                    except ValueError:
                        processed_row.append(val)
                else:
                    processed_row.append(val)
            rows.append(processed_row)

    # Define levers based on customer request
    levers = {
        "businessTravel": ["Non-Travel", "Travel_Rarely", "Travel_Frequently"],
        "overTime": ["No", "Yes"],
        "TrainingTimesLastYear": (0, 6),
        "PercentSalaryHike": (11, 25),
        "StockOptionLevel": (0, 3)
    }

    # Identify a 'bad' row (e.g., lowest tenure)
    tenure_idx = header.index("YearsAtCompany+")
    bad_row = min(rows, key=lambda r: r[tenure_idx])

    recommendation = fix(header, bad_row, levers)

    print("Retention Analysis: Recommended Changes for At-Risk Staff")
    print("To improve tenure for staff currently showing low retention,")
    print("we recommend the following adjustments to their working conditions:")
    for col, val in recommendation.items():
        print(f" - {col}: Change to {val}")
    print("\nThese changes target the levers identified by the customer to")
    print("improve engagement and reduce turnover.")
