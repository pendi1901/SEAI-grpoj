import csv
import random

def choose(header, rows, label):
    """
    Finds the best working arrangement by sampling 50 rows,
    labeling them to find their true tenure, and selecting the index
    of the row with the highest 'YearsAtCompany+'.
    """
    # Identify goal index
    try:
        goal_idx = next(i for i, h in enumerate(header) if h.endswith('+'))
    except StopIteration:
        return 0

    # We can label at most 50 rows.
    # To find the best, we sample 50 random indices.
    n = len(rows)
    sample_size = min(50, n)
    indices = random.sample(range(n), sample_size)
    
    best_val = -1.0
    best_idx = indices[0]
    
    for i in indices:
        # label(i) returns the true goal values for rows[i]
        goals = label(i)
        # Assuming the first goal returned corresponds to the one we identified
        val = float(goals[0])
        
        if val > best_val:
            best_val = val
            best_idx = i
            
    return best_idx

if __name__ == "__main__":
    filename = "WA_Fn-UseC_-HR-Employee-Attrition.train.csv"
    rows = []
    header = []
    
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = []
            for i, val in enumerate(row):
                if header[i][0].isupper():
                    try:
                        processed_row.append(float(val))
                    except ValueError:
                        processed_row.append(val)
                else:
                    processed_row.append(val)
            rows.append(processed_row)

    # Store original goal values to simulate the 'label' function
    goal_indices = [i for i, h in enumerate(header) if h.endswith('+') or h.endswith('-')]
    true_goals = [[row[i] for i in goal_indices] for row in rows]
    
    # Mask goals in the rows passed to choose
    for row in rows:
        for i in goal_indices:
            row[i] = "?"

    def label(i):
        return true_goals[i]

    best_idx = choose(header, rows, label)
    
    # Retrieve the row data (re-reading or using the original rows)
    # Since we masked them, we look at the original data structure
    print(f"Analysis complete. The best employee found (index {best_idx})")
    print(f"has a tenure of {true_goals[best_idx][0]} years.")
    print("This employee represents the most effective working arrangement found.")
