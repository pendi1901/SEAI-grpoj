    """
    Identifies staff who stay 'unusually long'.
    'YearsAtCompany+' is a goal to maximize.
    We define 'unusually long' as the top 5% of employees by tenure.
    """
    # Find index for YearsAtCompany+
    try:
        idx = header.index("YearsAtCompany+")
    except ValueError:
        return [0]

    # Extract tenure and original index
    tenures = []
    for i, row in enumerate(rows):
        try:
            val = float(row[idx])
            tenures.append((val, i))
        except (ValueError, TypeError):
            continue

    # Sort by tenure descending
    tenures.sort(key=lambda x: x[0], reverse=True)

    # Return top 5% (or at least 1)
    limit = max(1, len(rows) // 20)
    return [item[1] for item in tenures[:limit]]

import csv

def guess(header, train, test):
    """
    Estimates the first goal column (YearsAtCompany+) for test rows
    by finding the average of the goal column in the training set.
    """
    # Identify the first goal column (ends with + or -)
    goal_idx = -1
    for i, col in enumerate(header):
        if col.endswith('+') or col.endswith('-'):
            goal_idx = i
            break
    
    if goal_idx == -1:
        return [0.0] * len(test)

    # Calculate the mean of the goal column from training data
    total = 0.0
    count = 0
    for row in train:
        val = row[goal_idx]
        if isinstance(val, (int, float)):
            total += val
            count += 1
    
    mean_val = total / count if count > 0 else 0.0
    
    # Return the mean for every test row
    return [float(mean_val)] * len(test)

import csv

def certify(xs, ys):
    """
    Performs a simple two-sample t-test (Welch's) to check if two groups
    have significantly different means. Returns True if they are the same
    (null hypothesis not rejected), False if they differ.
    """
    n1, n2 = len(xs), len(ys)
    if n1 < 2 or n2 < 2:
        return True
    
    m1, m2 = sum(xs) / n1, sum(ys) / n2
    v1 = sum((x - m1) ** 2 for x in xs) / (n1 - 1)
    v2 = sum((y - m2) ** 2 for y in ys) / (n2 - 1)
    
    # Standard error
    se = ((v1 / n1) + (v2 / n2)) ** 0.5
    if se == 0:
        return True
    
    t_stat = abs(m1 - m2) / se
    # Critical value for t-distribution at alpha=0.05 (approx 1.96 for large samples)
    return t_stat < 1.96

import csv

def flow(header, row, seen):
    """
    Decides if the current employee needs a 'what would keep you here' conversation.
    We prioritize employees who have high 'YearsAtCompany+' but are currently
    experiencing 'OverTime' (Yes) or low 'JobSatisfaction', as these are 
    indicators of potential attrition for high-tenure staff.
    
    This is an online decision problem: we compare the current row against 
    the distribution of the 'seen' rows.
    """
    # Indices for decision criteria
    try:
        tenure_idx = header.index("YearsAtCompany+")
        ot_idx = header.index("overTime")
        sat_idx = header.index("JobSatisfaction")
    except ValueError:
        return False

    # If this is the first row, we can't compare, so let it go
    if not seen:
        return False

    # Extract current values
    curr_tenure = float(row[tenure_idx])
    curr_ot = row[ot_idx]
    curr_sat = float(row[sat_idx])

    # Calculate statistics from 'seen' to define 'high tenure' and 'low satisfaction'
    tenures = [float(r[tenure_idx]) for r in seen]
    sats = [float(r[sat_idx]) for r in seen]
    
    avg_tenure = sum(tenures) / len(tenures)
    avg_sat = sum(sats) / len(sats)

    # Logic: If they have been here longer than average AND (they work overtime OR 
    # have lower than average job satisfaction), they are a priority for retention.
    if curr_tenure > avg_tenure:
        if curr_ot == "Yes" or curr_sat < avg_sat:
            return True

    return False

import csv

def blame(header, rows):
    """
    Identifies input columns that correlate with short stays (low YearsAtCompany+).
    We define 'worst rows' as the bottom 25% of employees by tenure.
    We then compare the average of inputs in the 'worst' group vs the 'best' group.
    The columns with the largest normalized differences are returned as the 'blame'.
    """
    tenure_idx = header.index("YearsAtCompany+")
    
    # Filter out rows with missing goal data
    valid_rows = [r for r in rows if isinstance(r[tenure_idx], (int, float))]
    valid_rows.sort(key=lambda x: x[tenure_idx])
    
    # Split into bottom 25% (short stay) and top 25% (long stay)
    n = len(valid_rows) // 4
    worst = valid_rows[:n]
    best = valid_rows[-n:]
    
    # Identify input columns (not goals)
    input_indices = [
        i for i, h in enumerate(header) 
        if not (h.endswith('+') or h.endswith('-'))
    ]
    
    scores = []
    for i in input_indices:
        # Only analyze numeric inputs for correlation
        if isinstance(worst[0][i], (int, float)):
            w_vals = [r[i] for r in worst]
            b_vals = [r[i] for r in best]
            
            w_avg = sum(w_vals) / len(w_vals)
            b_avg = sum(b_vals) / len(b_vals)
            
            # Calculate absolute difference normalized by range
            all_vals = [r[i] for r in valid_rows]
            rng = max(all_vals) - min(all_vals)
            if rng > 0:
                diff = abs(w_avg - b_avg) / rng
                scores.append((diff, header[i]))
    
    # Return top 3 columns that explain the difference
    scores.sort(key=lambda x: x[0], reverse=True)
    return [s[1] for s in scores[:3]]
