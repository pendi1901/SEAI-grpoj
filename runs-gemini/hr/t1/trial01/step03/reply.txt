import csv

def certify(xs, ys):
    """
    Performs a simple two-sample t-test (Welch's) to check if two groups
    have significantly different means. Returns True if they are the same
    (null hypothesis not rejected), False if they differ.
    """
    n1, n2 = len(xs), len(ys)
    if n1 < 2 or n2 < 2:
        return True
    
    m1, m2 = sum(xs) / n1, sum(ys) / n2
    v1 = sum((x - m1) ** 2 for x in xs) / (n1 - 1)
    v2 = sum((y - m2) ** 2 for y in ys) / (n2 - 1)
    
    # Standard error
    se = ((v1 / n1) + (v2 / n2)) ** 0.5
    if se == 0:
        return True
    
    t_stat = abs(m1 - m2) / se
    # Critical value for t-distribution at alpha=0.05 (approx 1.96 for large samples)
    return t_stat < 1.96

if __name__ == "__main__":
    filename = "WA_Fn-UseC_-HR-Employee-Attrition.train.csv"
    rows = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = []
            for i, val in enumerate(row):
                if header[i][0].isupper():
                    try:
                        processed_row.append(float(val))
                    except ValueError:
                        processed_row.append(val)
                else:
                    processed_row.append(val)
            rows.append(processed_row)

    ot_idx = header.index("overTime")
    # YearsInCurrentRole+ is a good proxy for "leaving their role"
    tenure_idx = header.index("YearsInCurrentRole+")
    
    yes_ot = [r[tenure_idx] for r in rows if r[ot_idx] == "Yes"]
    no_ot = [r[tenure_idx] for r in rows if r[ot_idx] == "No"]
    
    is_same = certify(yes_ot, no_ot)
    
    print("Analysis: Does working overtime affect tenure in current role?")
    if not is_same:
        print("Result: The difference is statistically significant.")
        print("Conclusion: It is likely real, not just noise.")
    else:
        print("Result: The difference is not statistically significant.")
        print("Conclusion: The observed difference is likely just noise.")
