import csv

def remember(header, rows):
    """
    Identifies staff who stay 'unusually long'.
    'YearsAtCompany+' is a goal to maximize.
    We define 'unusually long' as the top 5% of employees by tenure.
    """
    # Find index for YearsAtCompany+
    try:
        idx = header.index("YearsAtCompany+")
    except ValueError:
        return [0]

    # Extract tenure and original index
    tenures = []
    for i, row in enumerate(rows):
        try:
            val = float(row[idx])
            tenures.append((val, i))
        except (ValueError, TypeError):
            continue

    # Sort by tenure descending
    tenures.sort(key=lambda x: x[0], reverse=True)

    # Return top 5% (or at least 1)
    limit = max(1, len(rows) // 20)
    return [item[1] for item in tenures[:limit]]

import csv

def guess(header, train, test):
    """
    Estimates the first goal column (YearsAtCompany+) for test rows
    by finding the average of the goal column in the training set.
    """
    # Identify the first goal column (ends with + or -)
    goal_idx = -1
    for i, col in enumerate(header):
        if col.endswith('+') or col.endswith('-'):
            goal_idx = i
            break
    
    if goal_idx == -1:
        return [0.0] * len(test)

    # Calculate the mean of the goal column from training data
    total = 0.0
    count = 0
    for row in train:
        val = row[goal_idx]
        if isinstance(val, (int, float)):
            total += val
            count += 1
    
    mean_val = total / count if count > 0 else 0.0
    
    # Return the mean for every test row
    return [float(mean_val)] * len(test)
