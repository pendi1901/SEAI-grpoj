import csv

def spot(header, rows):
    """
    Identifies suspicious rows by checking for outliers in numeric columns.
    A row is suspicious if it contains values that are statistically 
    improbable (e.g., outside 3 standard deviations from the mean).
    """
    suspicious = {}
    
    # Identify numeric columns
    numeric_indices = [i for i, h in enumerate(header) if h[0].isupper()]
    
    # Calculate mean and std dev for each numeric column
    stats = {}
    for i in numeric_indices:
        vals = [r[i] for r in rows if isinstance(r[i], (int, float))]
        if not vals:
            continue
        mean = sum(vals) / len(vals)
        var = sum((x - mean) ** 2 for x in vals) / len(vals)
        std = var ** 0.5
        stats[i] = (mean, std)
    
    # Score rows based on how many standard deviations they are from the mean
    for row_idx, row in enumerate(rows):
        score = 0
        for i in numeric_indices:
            if i in stats and isinstance(row[i], (int, float)):
                mean, std = stats[i]
                if std > 0:
                    z_score = abs(row[i] - mean) / std
                    if z_score > 3:
                        score += z_score
        
        if score > 0:
            suspicious[row_idx] = score
            
    # Return row indices sorted by suspicion score descending
    return sorted(suspicious, key=suspicious.get, reverse=True)

if __name__ == "__main__":
    filename = "WA_Fn-UseC_-HR-Employee-Attrition.train.csv"
    rows = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = []
            for i, val in enumerate(row):
                if header[i][0].isupper():
                    try:
                        processed_row.append(float(val))
                    except ValueError:
                        processed_row.append(val)
                else:
                    processed_row.append(val)
            rows.append(processed_row)

    suspects = spot(header, rows)
    
    print("Data Integrity Analysis: Suspicious Records")
    print("The following rows contain values that are statistically anomalous:")
    if not suspects:
        print("No suspicious records found.")
    else:
        for i in suspects[:10]:
            print(f" - Row {i} (Index {i}) shows highly unusual numeric values.")
    print("\nThese records should be manually verified for data entry errors.")
