import ezr
import random

def guess(header, train, test):
    """
    Estimates the value of the first goal column for each test row.
    Uses a tree-based approach: train a tree on the labeled data,
    then predict the mean of the leaf node for each test row.
    """
    # Create a table from training data
    train_tbl = ezr.Tbl([header] + train)
    
    # Build a tree using the training data
    # The tree splits based on minimizing ydist (distance to goals)
    root = ezr.tree(train_tbl, train_tbl.rows)
    
    # Identify the first goal column index
    # Goals are columns ending in '+' or '-'
    goal_indices = [i for i, h in enumerate(header) if h.endswith('+') or h.endswith('-')]
    first_goal_idx = goal_indices[0]
    
    results = []
    for row in test:
        # Walk the tree to find the leaf node for this test row
        # Note: We use the tree structure to find the "neighborhood"
        node = ezr.leaf(root, row)
        
        # Estimate the goal value as the average of that goal in the leaf
        # We calculate the mean of the first goal column for all rows in this leaf
        vals = [r[first_goal_idx] for r in node.rows if r[first_goal_idx] != "?"]
        if vals:
            results.append(sum(vals) / len(vals))
        else:
            # Fallback to global mean if leaf is empty or has no data
            all_vals = [r[first_goal_idx] for r in train if r[first_goal_idx] != "?"]
            results.append(sum(all_vals) / len(all_vals))
            
    return results

if __name__ == "__main__":
    # Load data
    filename = "WA_Fn-UseC_-HR-Employee-Attrition.train.csv"
    data = list(ezr.csv(filename))
    header = data[0]
    rows = data[1:]
    
    # Split data: 80% train, 20% test
    random.seed(1234567891)
    random.shuffle(rows)
    split = int(len(rows) * 0.8)
    train = rows[:split]
    test_full = rows[split:]
    
    # Mask goals in test set to simulate "day one" knowledge
    goal_indices = [i for i, h in enumerate(header) if h.endswith('+') or h.endswith('-')]
    test = []
    for row in test_full:
        new_row = list(row)
        for idx in goal_indices:
            new_row[idx] = "?"
        test.append(new_row)
        
    # Get predictions
    predictions = guess(header, train, test)
    
    # Print summary
    print(f"Analysis: Estimated tenure (first goal column) for {len(test)} new hires.")
    print(f"Sample predictions (first 5): {[round(p, 2) for p in predictions[:5]]}")
    print("These estimates are based on the patterns found in existing employee records.")
