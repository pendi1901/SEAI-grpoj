```python
import ezr
import random

def certify(xs, ys):
    """
    Returns True if the distributions of xs and ys are statistically 
    indistinguishable (same apart from noise), False if they differ.
    """
    # ezr.same uses cliffs delta and KS test to check for differences
    return ezr.same(xs, ys)

if __name__ == "__main__":
    # Load data
    filename = "WA_Fn-UseC_-HR-Employee-Attrition.train.csv"
    data = list(ezr.csv(filename))
    header = data[0]
    rows = data[1:]

    # Identify columns
    # 'overTime' is the symbolic column to split by
    # 'YearsInCurrentRole+' is the goal column to measure
    ot_idx = header.index("overTime")
    yr_idx = header.index("YearsInCurrentRole+")

    # Separate data into two groups based on overtime
    yes_overtime = []
    no_overtime = []

    for row in rows:
        val = row[yr_idx]
        if val != "?":
            if row[ot_idx] == "Yes":
                yes_overtime.append(val)
            elif row[ot_idx] == "No":
                no_overtime.append(val)

    # Perform the statistical test
    # same() returns True if they are the same (noise), False if they differ
    are_same = certify(yes_overtime, no_overtime)

    # Output results
    print("Analysis: Does working overtime affect years in current role?")
    print(f"Group 'Yes' (Overtime): {len(yes_overtime)} employees, mean={sum(yes_overtime)/len(yes_overtime):.2f}")
    print(f"Group 'No' (Overtime): {len(no_overtime)} employees, mean={sum(no_overtime)/len(no_overtime):.2f}")
    
    if not are_same:
        print("Result: The difference is statistically significant. Overtime is linked to tenure.")
    else:
        print("Result: The difference is not statistically significant; it appears to be noise.")
```
