import ezr

def justify(header, rows, row):
    """
    Returns (p, reason): p is the fraction of rows that 'row' is better than,
    and a human-readable reason based on the most influential column.
    """
    tbl = ezr.Tbl([header] + rows)
    
    # Calculate ydist for all rows
    # A lower ydist is better (closer to goals)
    all_ydists = [ezr.ydist(tbl, r) for r in rows]
    
    # We need to estimate the ydist of the target 'row'
    # Since goals are '?', we use the mean of the leaf node
    root = ezr.tree(tbl, rows)
    leaf_node = ezr.leaf(root, row)
    target_ydist = leaf_node.mu
    
    # Calculate fraction of rows better than target
    # (rows with ydist < target_ydist)
    better_count = sum(1 for y in all_ydists if y < target_ydist)
    p = better_count / len(rows)
    
    # Find the most influential column (the one at the root of the tree)
    # This explains "why" the row is ranked here
    col_name = header[root.at] if root.at is not None else "general company trends"
    
    reason = (f"This arrangement ranks better than {int(p*100)}% of staff, "
              f"primarily influenced by {col_name}.")
    
    return p, reason

if __name__ == "__main__":
    filename = "WA_Fn-UseC_-HR-Employee-Attrition.train.csv"
    data = list(ezr.csv(filename))
    header = data[0]
    rows = data[1:]
    
    # Pick a sample row and hide its goals to simulate the customer's query
    test_row = list(rows[100])
    goal_indices = [i for i, h in enumerate(header) if h.endswith('+') or h.endswith('-')]
    for idx in goal_indices:
        test_row[idx] = "?"
        
    p, reason = justify(header, rows, test_row)
    
    print("Analysis: Justification for employee arrangement ranking.")
    print(reason)
    print(f"Rank percentile: {p:.2f}")
