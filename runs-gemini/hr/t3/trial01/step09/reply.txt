```python
import pandas as pd
import numpy as np

def spot(header, rows):
    """
    Identifies rows that look 'wrong' by detecting outliers in numeric columns.
    We use the Interquartile Range (IQR) method to find rows with values 
    that are statistically improbable given the distribution of the data.
    """
    df = pd.DataFrame(rows, columns=header)
    
    # Identify numeric columns (those with uppercase first letter)
    # Exclude goal columns to focus on input data integrity
    numeric_cols = [c for c in header if c[0].isupper() and not (c.endswith('+') or c.endswith('-'))]
    
    suspicion_scores = np.zeros(len(df))
    
    for col in numeric_cols:
        series = pd.to_numeric(df[col], errors='coerce')
        q1 = series.quantile(0.25)
        q3 = series.quantile(0.75)
        iqr = q3 - q1
        
        # Define bounds for outliers
        lower_bound = q1 - 3 * iqr
        upper_bound = q3 + 3 * iqr
        
        # Add to suspicion score if value is outside the 3*IQR range
        outliers = (series < lower_bound) | (series > upper_bound)
        suspicion_scores += outliers.astype(int)
        
    # Get indices of rows with at least one outlier, sorted by suspicion score
    indices = np.where(suspicion_scores > 0)[0]
    sorted_indices = sorted(indices, key=lambda i: suspicion_scores[i], reverse=True)
    
    return sorted_indices

if __name__ == "__main__":
    # Load the data
    df = pd.read_csv('WA_Fn-UseC_-HR-Employee-Attrition.train.csv')
    header = df.columns.tolist()
    rows = df.values.tolist()
    
    # Identify suspicious rows
    suspicious_indices = spot(header, rows)
    
    print("Data Integrity Analysis: Identifying potentially erroneous entries")
    print("-" * 65)
    
    if not suspicious_indices:
        print("No suspicious data entries found.")
    else:
        print(f"Found {len(suspicious_indices)} rows with potential input errors.")
        print("Top 5 most suspicious rows (by index):")
        for idx in suspicious_indices[:5]:
            print(f"  Row {idx}: Contains extreme outliers in numeric input fields.")
            
    print("-" * 65)
    print("Note: These rows were flagged due to values significantly outside the expected range.")
    exit(0)
```
