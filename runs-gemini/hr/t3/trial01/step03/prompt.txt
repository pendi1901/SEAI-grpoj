import pandas as pd
import numpy as np

def remember(header, rows):
    """
    Identifies staff who stay 'unusually long'.
    Goals ending in '+' are maximized.
    We look for rows with high values in 'YearsAtCompany+', 'YearsInCurrentRole+', 
    and 'YearsWithCurrManager+'.
    """
    df = pd.DataFrame(rows, columns=header)
    
    # Identify goal columns ending in '+'
    goal_cols = [c for c in header if c.endswith('+')]
    
    # Normalize the goal columns to calculate a composite score
    # We use Z-score normalization to handle different scales
    scores = pd.DataFrame(index=df.index)
    for col in goal_cols:
        col_data = df[col].astype(float)
        scores[col] = (col_data - col_data.mean()) / col_data.std()
    
    # Calculate total score (higher is better)
    df['total_score'] = scores.sum(axis=1)
    
    # Get top 5% of rows (or at least 1)
    n_rows = len(df)
    n_select = max(1, int(n_rows * 0.05))
    
    top_indices = df.nlargest(n_select, 'total_score').index.tolist()
    return top_indices

import pandas as pd
import numpy as np
from sklearn.ensemble import RandomForestRegressor
from sklearn.preprocessing import LabelEncoder

def guess(header, train, test):
    """
    Estimates the first goal column (YearsAtCompany+) for test rows
    based on input columns (non-goal columns).
    """
    df_train = pd.DataFrame(train, columns=header)
    df_test = pd.DataFrame(test, columns=header)
    
    # Identify goal columns
    goal_cols = [c for c in header if c.endswith('+') or c.endswith('-')]
    target_col = goal_cols[0]
    
    # Identify input columns (exclude goals)
    input_cols = [c for c in header if c not in goal_cols]
    
    # Preprocess: Encode symbolic columns
    combined = pd.concat([df_train[input_cols], df_test[input_cols]], axis=0)
    encoded_df = combined.copy()
    
    for col in input_cols:
        if encoded_df[col].dtype == 'object':
            le = LabelEncoder()
            encoded_df[col] = le.fit_transform(encoded_df[col].astype(str))
            
    X_train = encoded_df.iloc[:len(df_train)]
    X_test = encoded_df.iloc[len(df_train):]
    y_train = df_train[target_col].astype(float)
    
    # Train a simple regressor
    model = RandomForestRegressor(n_estimators=100, random_state=42)
    model.fit(X_train, y_train)
    
    predictions = model.predict(X_test)
    return predictions.tolist()
