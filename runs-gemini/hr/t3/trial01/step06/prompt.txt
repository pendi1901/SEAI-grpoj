import pandas as pd
import numpy as np

def remember(header, rows):
    """
    Identifies staff who stay 'unusually long'.
    Goals ending in '+' are maximized.
    We look for rows with high values in 'YearsAtCompany+', 'YearsInCurrentRole+', 
    and 'YearsWithCurrManager+'.
    """
    df = pd.DataFrame(rows, columns=header)
    
    # Identify goal columns ending in '+'
    goal_cols = [c for c in header if c.endswith('+')]
    
    # Normalize the goal columns to calculate a composite score
    # We use Z-score normalization to handle different scales
    scores = pd.DataFrame(index=df.index)
    for col in goal_cols:
        col_data = df[col].astype(float)
        scores[col] = (col_data - col_data.mean()) / col_data.std()
    
    # Calculate total score (higher is better)
    df['total_score'] = scores.sum(axis=1)
    
    # Get top 5% of rows (or at least 1)
    n_rows = len(df)
    n_select = max(1, int(n_rows * 0.05))
    
    top_indices = df.nlargest(n_select, 'total_score').index.tolist()
    return top_indices

import pandas as pd
import numpy as np
from scipy import stats

def certify(xs, ys):
    """
    Returns True if the difference between two groups is statistically 
    insignificant (noise), False if they really differ.
    Uses Welch's t-test.
    """
    t_stat, p_value = stats.ttest_ind(xs, ys, equal_var=False)
    # If p-value > 0.05, we fail to reject the null hypothesis (they are the same)
    return p_value > 0.05

import pandas as pd
import numpy as np

def flow(header, row, seen):
    """
    Decides if the current row needs a 'what would keep you here' conversation.
    We prioritize employees who have high potential for retention (high tenure goals)
    but are currently showing signs of attrition risk (e.g., OverTime=Yes).
    
    Since we cannot revisit the pile, we use a threshold-based approach 
    relative to the 'seen' data to identify high-value individuals.
    """
    # Identify indices for relevant columns
    try:
        idx_overtime = header.index('overTime')
        idx_tenure = header.index('YearsAtCompany+')
        idx_attrition = header.index('attrition')
    except ValueError:
        return False

    # If this is the first row, we can't compare, so let it go
    if not seen:
        return False

    # Extract current row values
    current_overtime = row[idx_overtime]
    current_tenure = float(row[idx_tenure])
    current_attrition = row[idx_attrition]

    # Logic: If they are at risk (attrition=Yes) and have significant tenure 
    # (above the average of what we've seen so far), they are a priority.
    seen_df = pd.DataFrame(seen, columns=header)
    avg_tenure = seen_df['YearsAtCompany+'].mean()

    if current_attrition == 'Yes' and current_tenure >= avg_tenure:
        return True
    
    return False

if __name__ == "__main__":
    # Load the data
    df = pd.read_csv('WA_Fn-UseC_-HR-Employee-Attrition.train.csv')
    header = df.columns.tolist()
    rows = df.values.tolist()
    
    seen = []
    selected_count = 0
    
    print("Processing annual reviews one by one...")

    for i, row in enumerate(rows):
        if flow(header, row, seen):

            selected_count += 1
        seen.append(row)
        
    print("-" * 50)
    print(f"Total employees flagged for retention conversation: {selected_count}")
    print("Process complete.")

import pandas as pd
import numpy as np
from sklearn.ensemble import RandomForestRegressor

def blame(header, rows):
    """
    Identifies input columns that best explain why employees have short stays.
    We define 'bad' rows as those with the lowest values in the goal columns 
    (YearsAtCompany+, YearsInCurrentRole+, YearsWithCurrManager+).
    """
    df = pd.DataFrame(rows, columns=header)
    
    # Identify goal columns and input columns
    goal_cols = [c for c in header if c.endswith('+') or c.endswith('-')]
    input_cols = [c for c in header if c not in goal_cols]
    
    # Create a composite 'badness' score: lower goal values = worse
    # Normalize goals to 0-1 range
    scores = pd.DataFrame(index=df.index)
    for col in goal_cols:
        col_data = df[col].astype(float)
        scores[col] = (col_data - col_data.min()) / (col_data.max() - col_data.min() + 1e-9)
    
    # Average score: lower is worse
    df['badness'] = scores.mean(axis=1)
    
    # Focus on the bottom 25% (the 'worst' rows)
    threshold = df['badness'].quantile(0.25)
    worst_rows = df[df['badness'] <= threshold].copy()
    
    # Use a Random Forest to find feature importance for predicting 'badness'
    # Encode symbolic columns
    X = df[input_cols].copy()
    for col in input_cols:
        if X[col].dtype == 'object':
            X[col] = X[col].astype('category').cat.codes
            
    y = df['badness']
    
    model = RandomForestRegressor(n_estimators=50, random_state=42)
    model.fit(X, y)
    
    # Get feature importances
    importances = pd.Series(model.feature_importances_, index=input_cols)
    top_features = importances.nlargest(3).index.tolist()
    
    return top_features
