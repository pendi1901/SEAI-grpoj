```python
import pandas as pd
import numpy as np

def justify(header, rows, row):
    """
    Calculates the percentile rank of a given row relative to the dataset
    based on goal columns and provides a justification.
    """
    df = pd.DataFrame(rows, columns=header)
    
    # Identify goal columns
    goal_cols = [c for c in header if c.endswith('+') or c.endswith('-')]
    
    # Calculate a composite score for the dataset
    # Normalize goals: for '+', higher is better; for '-', lower is better
    scores = pd.DataFrame(index=df.index)
    for col in goal_cols:
        col_data = df[col].astype(float)
        if col.endswith('+'):
            scores[col] = (col_data - col_data.min()) / (col_data.max() - col_data.min() + 1e-9)
        else:
            scores[col] = (col_data.max() - col_data) / (col_data.max() - col_data.min() + 1e-9)
    
    # Global average score for comparison
    df['composite'] = scores.mean(axis=1)
    
    # Calculate the score for the input row
    # Since the row has '?', we must infer its performance based on its non-goal inputs.
    # We use the average composite score of rows that share the same categorical inputs.
    input_cols = [c for c in header if c not in goal_cols]
    
    # Find rows with matching categorical/numeric inputs
    match_mask = pd.Series(True, index=df.index)
    for col in input_cols:
        val = row[header.index(col)]
        if val != '?':
            match_mask &= (df[col] == val)
    
    matches = df[match_mask]
    if not matches.empty:
        row_score = matches['composite'].mean()
    else:
        row_score = df['composite'].mean()
        
    # Calculate percentile (fraction of rows that this row is better than)
    p = (df['composite'] < row_score).mean()
    
    reason = f"This team's arrangement ranks in the {p:.0%} percentile because its performance metrics across {len(goal_cols)} key indicators outperform the majority of other recorded team configurations."
    
    return (p, reason)

if __name__ == "__main__":
    df = pd.read_csv('WA_Fn-UseC_-HR-Employee-Attrition.train.csv')
    header = df.columns.tolist()
    rows = df.values.tolist()
    
    # Test with a sample row (the first one)
    sample_row = rows[0].copy()
    # Mask the goals
    goal_indices = [i for i, col in enumerate(header) if col.endswith('+') or col.endswith('-')]
    for idx in goal_indices:
        sample_row[idx] = "?"
        
    p, reason = justify(header, rows, sample_row)
    
    print(f"Rank: {p:.2f}")
    print(f"Justification: {reason}")
    
    exit(0)
```
