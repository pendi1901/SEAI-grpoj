```python
import ezr
import random

def flow(header, row, seen):
    """
    Decide whether to accept a row immediately.
    We compare the row to the 'seen' data. If the row is in the top 
    percentile of performance relative to what we have seen so far,
    we accept it.
    """
    if not seen:
        return True
    
    # Create a table from the seen rows to calculate ydist
    # We use the header to define the columns
    tbl = ezr.Tbl([header] + seen)
    
    # Calculate ydist for the current row
    # ydist is 0 at the ideal, higher as it gets worse
    current_ydist = ezr.ydist(tbl, row)
    
    # Calculate ydist for all seen rows
    seen_ydists = [ezr.ydist(tbl, r) for r in seen]
    
    # Accept if the current row is better than the median of what we've seen
    # This is a simple, greedy heuristic for "on the spot" decisions
    median_ydist = sorted(seen_ydists)[len(seen_ydists) // 2]
    
    return current_ydist <= median_ydist

if __name__ == "__main__":
    filename = "pom3a.train.csv"
    rows_iter = ezr.csv(filename)
    header = next(rows_iter)
    all_rows = list(rows_iter)
    
    # Simulate the customer's stream
    seen = []
    accepted = []
    
    # We process a subset to demonstrate the flow
    # The customer wants a decision on the spot
    for i, row in enumerate(all_rows[:500]):
        decision = flow(header, row, seen)
        if decision:
            accepted.append(row)
        seen.append(row)
        
    print(f"Processed 500 proposals.")
    print(f"Accepted {len(accepted)} proposals based on real-time evaluation.")
    
    if accepted:
        # Show the average quality of accepted vs seen
        tbl = ezr.Tbl([header] + seen)
        avg_seen = sum(ezr.ydist(tbl, r) for r in seen) / len(seen)
        avg_acc = sum(ezr.ydist(tbl, r) for r in accepted) / len(accepted)
        print(f"Average ydist of all seen: {avg_seen:.4f}")
        print(f"Average ydist of accepted: {avg_acc:.4f}")
        print("Lower ydist is better.")
    
    exit(0)
```
