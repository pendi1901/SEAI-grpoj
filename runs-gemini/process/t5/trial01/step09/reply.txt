import ezr
import random

def spot(header, rows):
    """
    Returns the indices of rows that are most suspicious.
    A row is suspicious if its input values are 'far' from the 
    expected distribution of the rest of the data.
    """
    # Create a table to get column summaries
    tbl = ezr.Tbl([header] + rows)
    
    # Calculate the distance of each row from the 'middle' of the data.
    # We use xdist to measure how far a row's inputs are from the 
    # centroid (mids) of the entire dataset.
    centroid = ezr.mids(tbl)
    
    scores = []
    for i, row in enumerate(rows):
        # Calculate distance from the centroid
        d = ezr.xdist(tbl, row, centroid)
        scores.append((d, i))
    
    # Sort by distance descending: the furthest rows are the most suspicious
    scores.sort(key=lambda x: x[0], reverse=True)
    
    # Return the indices of the top 1% most suspicious rows
    limit = max(1, len(rows) // 100)
    return [item[1] for item in scores[:limit]]

if __name__ == "__main__":
    filename = "pom3a.train.csv"
    rows_iter = ezr.csv(filename)
    header = next(rows_iter)
    rows = list(rows_iter)
    
    # Identify suspicious rows
    suspicious_indices = spot(header, rows)
    
    print("Anomaly Detection Analysis:")
    print(f"Identified {len(suspicious_indices)} rows that deviate significantly from the norm.")
    print("These rows may contain mistyped or erroneous input data:")
    
    for idx in suspicious_indices[:10]:  # Show top 10
        print(f"- Row index {idx} is highly suspicious.")
        
    print("\nThese rows should be reviewed for potential data entry errors.")
    exit(0)
