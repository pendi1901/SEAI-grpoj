import ezr
import random

def choose(header, rows, label):
    """
    Find the best row by using the acquire strategy from ezr.
    We label up to 50 rows, then return the index of the best one found.
    """
    # Create a table object to manage the data
    # We need to handle the fact that rows have '?' in goal columns
    # The acquire function in ezr expects a TBL and an oracle
    tbl = ezr.Tbl([header] + rows)
    
    # The acquire function returns the rows that were labeled.
    # We use the provided label function as the oracle.
    # We set the budget to 50 as requested.
    ezr.the.Stop = 50
    
    # We need to map the labeled rows back to their original indices.
    # Since acquire returns a list of rows, we find the best one
    # and then find its index in the original rows list.
    labeled_rows = ezr.acquire(tbl, cap=50, label=label)
    
    # Find the best row among the labeled ones
    # ydist is the metric for "closeness to ideal"
    best_row = min(labeled_rows, key=lambda r: ezr.ydist(tbl, r))
    
    # Return the index of this best row in the original rows list
    for i, row in enumerate(rows):
        if row == best_row:
            return i
    return 0

if __name__ == "__main__":
    filename = "pom3a.train.csv"
    rows_iter = ezr.csv(filename)
    header = next(rows_iter)
    all_rows = list(rows_iter)
    
    # The customer wants to trial 50 setups.
    # We define a simple oracle that returns the actual row (with goals)
    def oracle(row):
        # In this simulation, we know the full data.
        # We find the row in all_rows that matches the input features.
        for r in all_rows:
            # Match on input columns (all except last 3)
            if all(r[i] == row[i] for i in range(len(header) - 3)):
                return r
        return row

    # Run the selection
    best_idx = choose(header, all_rows, oracle)
    best_row = all_rows[best_idx]
    
    print("Optimization complete.")
    print(f"Best setup found at index: {best_idx}")
    print(f"Cost-: {best_row[-3]:.4f}, Score-: {best_row[-2]:.4f}, Idle-: {best_row[-1]:.4f}")
    print("These values represent the best configuration found within the 50-label budget.")
    
    exit(0)
