import csv
import pandas as pd
import numpy as np

def remember(header, rows):
    """
    Identifies rows that perform well across all goals.
    Goals ending in '-' are minimized, '+' are maximized.
    We use a simple Pareto-based approach: normalize goals and 
    find rows with the lowest aggregate distance from the ideal.
    """
    # Identify goal columns
    goal_indices = []
    goal_types = []
    for i, name in enumerate(header):
        if name.endswith('-'):
            goal_indices.append(i)
            goal_types.append('min')
        elif name.endswith('+'):
            goal_indices.append(i)
            goal_types.append('max')

    # Extract goal data
    data = []
    for row in rows:
        data.append([float(row[i]) for i in goal_indices])
    
    df_goals = pd.DataFrame(data)
    
    # Normalize goals to [0, 1]
    normalized = (df_goals - df_goals.min()) / (df_goals.max() - df_goals.min())
    
    # For 'min' goals, 0 is best. For 'max' goals, 1 is best.
    # We want to minimize distance to (0 for min, 1 for max)
    scores = []
    for i, col in enumerate(normalized.columns):
        if goal_types[i] == 'min':
            scores.append(normalized[col])
        else:
            scores.append(1 - normalized[col])
            
    # Aggregate score (sum of normalized distances)
    total_scores = pd.concat(scores, axis=1).sum(axis=1)
    
    # Return indices of the best 5% (or at least 1)
    num_to_return = max(1, int(len(rows) * 0.05))
    best_indices = total_scores.nsmallest(num_to_return).index.tolist()
    
    return best_indices

def guess(header, train, test):
    """
    Estimates the first goal column for test rows using K-Nearest Neighbors
    based on input columns (non-goal columns).
    """
    # Identify goal columns and input columns
    goal_indices = [i for i, name in enumerate(header) if name.endswith(('-', '+'))]
    input_indices = [i for i, name in enumerate(header) if not name.endswith(('-', '+'))]
    
    # Prepare training data
    X_train = [[float(row[i]) for i in input_indices] for row in train]
    y_train = [float(row[goal_indices[0]]) for row in train]
    
    # Prepare test data
    X_test = [[float(row[i]) for i in input_indices] for row in test]
    
    # Use KNN to predict the first goal

    model.fit(X_train, y_train)
    
    return model.predict(X_test).tolist()

def certify(xs, ys):
    """
    Returns True if the difference between two distributions is not statistically 
    significant (p > 0.05), False otherwise.
    """
    # Use Welch's t-test (equal_var=False) to compare means
    t_stat, p_value = stats.ttest_ind(xs, ys, equal_var=False)
    return p_value > 0.05

import csv
import pandas as pd
import numpy as np
from step01 import remember

def flow(header, row, seen):
    """
    Implements a simple threshold-based decision strategy.
    We accept a row if it is better than the median of the 'seen' rows
    based on the aggregate goal score.
    """
    if not seen:
        return True
    
    # Identify goal columns
    goal_indices = [i for i, name in enumerate(header) if name.endswith(('-', '+'))]
    goal_types = ['min' if header[i].endswith('-') else 'max' for i in goal_indices]
    
    def get_score(r):
        # Normalize goal values relative to the current 'seen' set
        # If seen is small, we use a simple heuristic
        vals = [float(r[i]) for i in goal_indices]
        scores = []
        for i, val in enumerate(vals):
            # Simple heuristic: if min, lower is better; if max, higher is better
            # We treat the current row's value as a score component
            scores.append(val if goal_types[i] == 'min' else -val)
        return sum(scores)

    current_score = get_score(row)
    seen_scores = [get_score(s) for s in seen]
    
    # Accept if better than the median of what we have seen so far
    return current_score <= np.median(seen_scores)

import csv

def justify(header, rows, row):
    """
    Returns (p, reason): p is the fraction of rows that 'row' is better than.
    'row' has goals as "?". We estimate the goals using KNN and compare
    the aggregate score against the distribution of all other rows.
    """
    # 1. Estimate the missing goals for the target row
    # We need to provide a list containing the single row to the guess function
    predicted_goals = guess(header, rows, [row])
    
    # 2. Calculate scores for all rows
    # Reuse logic from step01: normalize and aggregate
    goal_indices = [i for i, name in enumerate(header) if name.endswith(('-', '+'))]
    goal_types = ['min' if header[i].endswith('-') else 'max' for i in goal_indices]
    
    data = [[float(r[i]) for i in goal_indices] for r in rows]
    df_goals = pd.DataFrame(data)
    
    # Normalize
    normalized = (df_goals - df_goals.min()) / (df_goals.max() - df_goals.min())
    
    # Calculate aggregate scores for all rows
    scores = []
    for i, col in enumerate(normalized.columns):
        if goal_types[i] == 'min':
            scores.append(normalized[col])
        else:
            scores.append(1 - normalized[col])
    all_scores = pd.concat(scores, axis=1).sum(axis=1)
    
    # 3. Calculate score for the target row
    # We must normalize the predicted goal using the same min/max from the training set
    pred_row_vals = []
    # Note: guess() only predicts the first goal, so we assume others are 0 or 
    # use the mean for estimation if not provided. For simplicity, we use the 
    # predicted first goal and assume average performance for others.
    pred_row_vals.append(predicted_goals[0])
    for i in range(1, len(goal_indices)):
        pred_row_vals.append(df_goals.iloc[:, i].mean())
        
    pred_norm = []
    for i in range(len(goal_indices)):
        val = (pred_row_vals[i] - df_goals.iloc[:, i].min()) / (df_goals.iloc[:, i].max() - df_goals.iloc[:, i].min())
        pred_norm.append(val if goal_types[i] == 'min' else 1 - val)
    
    target_score = sum(pred_norm)
    
    # 4. Calculate fraction better
    better_count = (all_scores > target_score).sum()
    p = better_count / len(rows)
    
    reason = f"This setup is in the top {int((1-p)*100)}% of all possible configurations based on its estimated performance metrics."
    
    return p, reason
