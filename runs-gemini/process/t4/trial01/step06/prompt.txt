import csv
import pandas as pd
import numpy as np

def remember(header, rows):
    """
    Identifies rows that perform well across all goals.
    Goals ending in '-' are minimized, '+' are maximized.
    We use a simple Pareto-based approach: normalize goals and 
    find rows with the lowest aggregate distance from the ideal.
    """
    # Identify goal columns
    goal_indices = []
    goal_types = []
    for i, name in enumerate(header):
        if name.endswith('-'):
            goal_indices.append(i)
            goal_types.append('min')
        elif name.endswith('+'):
            goal_indices.append(i)
            goal_types.append('max')

    # Extract goal data
    data = []
    for row in rows:
        data.append([float(row[i]) for i in goal_indices])
    
    df_goals = pd.DataFrame(data)
    
    # Normalize goals to [0, 1]
    normalized = (df_goals - df_goals.min()) / (df_goals.max() - df_goals.min())
    
    # For 'min' goals, 0 is best. For 'max' goals, 1 is best.
    # We want to minimize distance to (0 for min, 1 for max)
    scores = []
    for i, col in enumerate(normalized.columns):
        if goal_types[i] == 'min':
            scores.append(normalized[col])
        else:
            scores.append(1 - normalized[col])
            
    # Aggregate score (sum of normalized distances)
    total_scores = pd.concat(scores, axis=1).sum(axis=1)
    
    # Return indices of the best 5% (or at least 1)
    num_to_return = max(1, int(len(rows) * 0.05))
    best_indices = total_scores.nsmallest(num_to_return).index.tolist()
    
    return best_indices

def certify(xs, ys):
    """
    Returns True if the difference between two distributions is not statistically 
    significant (p > 0.05), False otherwise.
    """
    # Use Welch's t-test (equal_var=False) to compare means

import csv
import pandas as pd
import numpy as np
from step01 import remember

def flow(header, row, seen):
    """
    Implements a simple threshold-based decision strategy.
    We accept a row if it is better than the median of the 'seen' rows
    based on the aggregate goal score.
    """
    if not seen:
        return True
    
    # Identify goal columns
    goal_indices = [i for i, name in enumerate(header) if name.endswith(('-', '+'))]
    goal_types = ['min' if header[i].endswith('-') else 'max' for i in goal_indices]
    
    def get_score(r):
        # Normalize goal values relative to the current 'seen' set
        # If seen is small, we use a simple heuristic
        vals = [float(r[i]) for i in goal_indices]
        scores = []
        for i, val in enumerate(vals):
            # Simple heuristic: if min, lower is better; if max, higher is better
            # We treat the current row's value as a score component
            scores.append(val if goal_types[i] == 'min' else -val)
        return sum(scores)

    current_score = get_score(row)
    seen_scores = [get_score(s) for s in seen]
    
    # Accept if better than the median of what we have seen so far
    return current_score <= np.median(seen_scores)

if __name__ == "__main__":

def blame(header, rows):
    """
    Identifies input columns that correlate most strongly with high 'Idle-' values.
    We use a Random Forest to rank feature importance for the 'Idle-' goal.
    """
    df = pd.DataFrame(rows, columns=header)
    
    # Identify input columns and the target goal
    goal_indices = [i for i, name in enumerate(header) if name.endswith(('-', '+'))]
    input_indices = [i for i, name in enumerate(header) if not name.endswith(('-', '+'))]
    
    input_cols = [header[i] for i in input_indices]
    target_col = 'Idle-'
    
    # Prepare data
    X = df[input_cols].astype(float)
    y = df[target_col].astype(float)
    
    # Use Random Forest to find feature importance
    model = RandomForestRegressor(n_estimators=50, random_state=42)
    model.fit(X, y)
    
    # Get top 3 features explaining the variance in 'Idle-'
    importances = model.feature_importances_
    indices = np.argsort(importances)[::-1]
    
    top_features = [input_cols[i] for i in indices[:3]]
    return top_features
