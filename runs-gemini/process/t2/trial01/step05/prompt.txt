import csv

def remember(header, rows):
    """
    Identifies the best rows by calculating a normalized score for each goal.
    Goals ending in '-' are minimized (1 - value), '+' are maximized (value).
    We look for rows that perform well across all goals simultaneously.
    """
    goals = []
    for i, name in enumerate(header):
        if name.endswith('-') or name.endswith('+'):
            goals.append((i, name.endswith('+')))

    # Calculate normalized scores for each goal
    # Since all inputs are 0-1, we can treat them as already normalized
    scores = []
    for idx, row in enumerate(rows):
        row_score = 0
        for col_idx, is_max in goals:
            val = float(row[col_idx])
            # If goal is to minimize (-), lower is better (1 - val)
            # If goal is to maximize (+), higher is better (val)
            row_score += val if is_max else (1.0 - val)
        
        # Average score across the 3 goals
        scores.append((row_score / len(goals), idx))

    # Sort by score descending (higher is better)
    scores.sort(key=lambda x: x[0], reverse=True)
    
    # Return top 5% of rows
    limit = max(1, len(rows) // 20)
    return [item[1] for item in scores[:limit]]

def certify(xs, ys):
    """
    Returns True if the distributions of xs and ys are statistically similar 
    (i.e., they don't differ significantly), False otherwise.
    Uses a simple mean comparison with a tolerance threshold (A12 effect size).
    """
    if not xs or not ys:
        return True
    
    # Calculate means
    mean_x = sum(xs) / len(xs)
    mean_y = sum(ys) / len(ys)
    
    # If the difference is less than 10% of the range, consider it noise
    diff = abs(mean_x - mean_y)
    return diff < 0.1

import csv
from step01 import remember

def flow(header, row, seen):
    """
    Implements a Secretary Problem-style strategy.
    We observe a sample (first 20% of data) to establish a baseline 'best' 
    score, then accept the first row that exceeds that baseline.
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith(('-', '+'))]
    is_max = [name.endswith('+') for name in header if name.endswith(('-', '+'))]

    def get_score(r):
        s = 0
        for i, idx in enumerate(goal_indices):
            val = float(r[idx])
            s += val if is_max[i] else (1.0 - val)
        return s / len(goal_indices)

    # If we haven't seen enough to establish a baseline, keep looking
    # Using 20% of the expected 16000 rows (3200) as a sample period
    sample_size = 3200
    if len(seen) < sample_size:
        return False
    
    # Calculate the best score seen so far in the sample
    best_baseline = max(get_score(r) for r in seen)
    
    # If the current row is better than the best seen in the sample, accept it
    return get_score(row) > best_baseline

if __name__ == "__main__":
    filename = "pom3a.train.csv"
    with open(filename, mode='r', encoding='utf-8') as f:
        reader = csv.reader(f)
        header = next(reader)
        rows = [row for row in reader]

    seen = []
    accepted = None
    
    for row in rows:
        if flow(header, row, seen):
            accepted = row
            break
        seen.append(row)
    
    if accepted:
        print("Decision: Approved the current proposal.")
        print(f"Row details: {accepted}")
    else:
        print("Decision: No suitable proposal found.")
