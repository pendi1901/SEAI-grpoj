import csv
from step01 import remember

def blame(header, rows):
    """
    Identifies input columns that correlate with high 'Idle-' values.
    We define 'worst' as rows in the bottom 10% of performance (high Idle-).
    We compare the average of input values in the worst rows vs the rest.
    """
    # Identify goal indices and input indices
    goal_indices = [i for i, name in enumerate(header) if name.endswith(('-', '+'))]
    input_indices = [i for i, name in enumerate(header) if not (name.endswith('-') or name.endswith('+'))]
    idle_idx = header.index("Idle-")

    # Sort rows by Idle- descending (worst first)
    sorted_rows = sorted(rows, key=lambda r: float(r[idle_idx]), reverse=True)
    
    # Define worst as the top 10% of the sorted list
    worst_rows = sorted_rows[:len(rows) // 10]
    rest_rows = sorted_rows[len(rows) // 10:]

    # Calculate average input values for worst vs rest
    def get_avg_inputs(subset):
        avgs = []
        for i in input_indices:
            vals = [float(r[i]) for r in subset]
            avgs.append(sum(vals) / len(vals))
        return avgs

    worst_avgs = get_avg_inputs(worst_rows)
    rest_avgs = get_avg_inputs(rest_rows)

    # Find inputs that differ most between worst and rest
    diffs = []
    for i, idx in enumerate(input_indices):
        diff = abs(worst_avgs[i] - rest_avgs[i])
        diffs.append((diff, header[idx]))
    
    # Sort by difference descending
    diffs.sort(key=lambda x: x[0], reverse=True)
    
    # Return the top 3 inputs that explain the bad idle time
    return [name for diff, name in diffs[:3]]

if __name__ == "__main__":
    filename = "pom3a.train.csv"
    with open(filename, mode='r', encoding='utf-8') as f:
        reader = csv.reader(f)
        header = next(reader)
        rows = [row for row in reader]

    explanations = blame(header, rows)
    
    print("Analysis: Why do our worst setups have high Idle- times?")
    print("The primary drivers for high idle time are:")
    for i, name in enumerate(explanations, 1):
        print(f"{i}. {name}")
    print("\nThese inputs show the largest deviation in setups where staff are waiting.")
