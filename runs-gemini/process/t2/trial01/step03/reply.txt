import csv
from step01 import remember

def certify(xs, ys):
    """
    Returns True if the distributions of xs and ys are statistically similar 
    (i.e., they don't differ significantly), False otherwise.
    Uses a simple mean comparison with a tolerance threshold (A12 effect size).
    """
    if not xs or not ys:
        return True
    
    # Calculate means
    mean_x = sum(xs) / len(xs)
    mean_y = sum(ys) / len(ys)
    
    # If the difference is less than 10% of the range, consider it noise
    diff = abs(mean_x - mean_y)
    return diff < 0.1

def get_data():
    filename = "pom3a.train.csv"
    with open(filename, mode='r', encoding='utf-8') as f:
        reader = csv.reader(f)
        header = next(reader)
        rows = [row for row in reader]
    return header, rows

if __name__ == "__main__":
    header, rows = get_data()
    
    # Identify column indices
    team_size_idx = header.index("Team Size")
    cost_idx = header.index("Cost-")
    
    # Split data into small-team and big-team based on median Team Size
    team_sizes = [float(row[team_size_idx]) for row in rows]
    median_team = sum(team_sizes) / len(team_sizes)
    
    small_team_costs = [float(row[cost_idx]) for row in rows if float(row[team_size_idx]) < median_team]
    big_team_costs = [float(row[cost_idx]) for row in rows if float(row[team_size_idx]) >= median_team]
    
    is_same = certify(small_team_costs, big_team_costs)
    
    print("Analysis: Comparing Cost- for small vs big team setups.")
    if is_same:
        print("Result: The difference is statistically insignificant. The observation was likely luck.")
    else:
        print("Result: There is a significant difference in costs. The observation is real.")
