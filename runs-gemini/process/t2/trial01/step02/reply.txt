import csv
from step01 import remember

def guess(header, train, test):
    """
    Estimates the first goal column for test rows by finding the 
    nearest neighbor in the training set based on input columns.
    """
    # Identify input columns (those not ending in - or +)
    input_indices = [i for i, name in enumerate(header) 
                     if not (name.endswith('-') or name.endswith('+'))]
    
    # Identify the first goal column index
    goal_idx = [i for i, name in enumerate(header) 
                if name.endswith('-') or name.endswith('+')][0]

    predictions = []
    for t_row in test:
        best_dist = float('inf')
        best_val = 0.0
        
        # Simple Euclidean distance in input space
        for train_row in train:
            dist = 0.0
            for i in input_indices:
                # Assuming numeric inputs are 0-1 as per prompt
                val_t = float(t_row[i])
                val_tr = float(train_row[i])
                dist += (val_t - val_tr) ** 2
            
            if dist < best_dist:
                best_dist = dist
                best_val = float(train_row[goal_idx])
        
        predictions.append(best_val)
    
    return predictions

if __name__ == "__main__":
    filename = "pom3a.train.csv"
    rows = []
    with open(filename, mode='r', encoding='utf-8') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            rows.append(row)

    # For demonstration, we use a subset of training data as 'test' 
    # by masking the goal columns.
    test_rows = [row[:] for row in rows[:5]]
    for row in test_rows:
        for i in range(len(header)):
            if header[i].endswith(('-', '+')):
                row[i] = "?"

    results = guess(header, rows, test_rows)
    
    print("Estimating Cost- for test rows:")
    for i, val in enumerate(results):
        print(f"Test Row {i}: Estimated Cost- = {val:.4f}")
