import ezr
import step01
import step05

def fix(header, row, levers):
    """
    Finds the smallest change to levers to improve a bad row.
    We identify which levers are most correlated with bad performance
    and nudge them toward the values seen in the 'best' rows.
    """
    # Load data to establish what 'good' looks like
    file_path = "pom3a.train.csv"
    rows_iter = ezr.csv(file_path)
    header_data = next(rows_iter)
    all_rows = list(rows_iter)
    tbl = ezr.Tbl([header_data] + all_rows)

    # Find the best rows to see what values they have for the levers
    scored = sorted([(ezr.ydist(tbl, r), r) for r in all_rows], key=lambda x: x[0])
    best_rows = [x[1] for x in scored[:len(all_rows)//10]]

    recommendations = {}
    for col_name, allowed in levers.items():
        idx = header.index(col_name)
        col = tbl.cols[idx]
        
        # Get the target value from the best rows
        target_vals = [r[idx] for r in best_rows if r[idx] != "?"]
        target = sum(target_vals) / len(target_vals) if isinstance(col, ezr.Num) else max(set(target_vals), key=target_vals.count)

        # Constrain target to allowed range/set
        if isinstance(col, ezr.Num):
            low, high = allowed
            recommendations[col_name] = max(low, min(high, target))
        else:
            recommendations[col_name] = target if target in allowed else allowed[0]
            
    return recommendations

    rows_iter = ezr.csv(file_path)
    header = next(header_data := list(rows_iter))

    # Identify a "bad" row (e.g., the worst one)
    tbl = ezr.Tbl([header] + all_rows)
    worst_row = max(all_rows, key=lambda r: ezr.ydist(tbl, r))
    
    # Define levers: Plan (0-1), Team Size (0-1)
    levers = {
        "Plan": (0.0, 1.0),
        "Team Size": (0.0, 1.0)
    }
    
    changes = fix(header, worst_row, levers)
    
    print("Rescue Plan for Over-Budget/Idle Project:")
    print(f"Current state: Plan={worst_row[header.index('Plan')]:.2f}, "
          f"Team Size={worst_row[header.index('Team Size')]:.2f}")
    print("Recommended adjustments:")
    for col, val in changes.items():
        print(f" - Set {col} to {val:.2f}")
    print("\nBy shifting these parameters toward the values found in high-performing "
          "configurations, we can reduce idle time and bring costs back under budget.")
