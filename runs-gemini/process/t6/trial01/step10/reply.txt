import ezr
import random
import step01
import step07

def race(header, rows, label):
    """
    Compares three strategies: random, acquire (from ezr), and a greedy 
    local search. Each gets a budget of 50 labels.
    """
    budget = 50
    
    def strategy_random(tbl, rows_masked, label_func):
        # Randomly sample 50 rows
        indices = random.sample(range(len(rows_masked)), min(budget, len(rows_masked)))
        for i in indices:
            label_func(i)
        return min(indices, key=lambda i: ezr.ydist(tbl, rows[i]))

    def strategy_acquire(tbl, rows_masked, label_func):
        # Uses ezr.acquire logic
        def oracle_wrapper(row):
            idx = rows.index(row)
            return label_func(idx)
        
        labeled = ezr.acquire(tbl, cap=budget, label=oracle_wrapper)
        best_row = min(labeled, key=lambda r: ezr.ydist(tbl, r))
        return rows.index(best_row)

    def strategy_greedy(tbl, rows_masked, label_func):
        # Start with one random, then always pick the nearest neighbor to the best
        curr_idx = random.choice(range(len(rows_masked)))
        labeled = {curr_idx: label_func(curr_idx)}
        
        for _ in range(budget - 1):
            # Find unlabelled row closest to current best
            best_so_far = min(labeled.keys(), key=lambda i: ezr.ydist(tbl, rows[i]))
            
            def dist_to_best(i):
                if i in labeled: return 1e30
                return ezr.xdist(tbl, rows[i], rows[best_so_far])
            
            next_idx = min(range(len(rows_masked)), key=dist_to_best)
            labeled[next_idx] = label_func(next_idx)
            
        return min(labeled.keys(), key=lambda i: ezr.ydist(tbl, rows[i]))

    # Setup
    tbl = ezr.Tbl([header] + rows)
    results = {}
    strategies = {
        "random": strategy_random,
        "acquire": strategy_acquire,
        "greedy": strategy_greedy
    }
    
    # Run race
    for name, func in strategies.items():
        # Reset label tracking for each strategy
        labeled_cache = {}
        def label_proxy(i):
            if i not in labeled_cache:
                labeled_cache[i] = label(i)
            return labeled_cache[i]
        
        best_idx = func(tbl, rows, label_proxy)
        results[name] = (best_idx, ezr.ydist(tbl, rows[best_idx]))

    winner = min(results, key=lambda n: results[n][1])
    return {
        "winner": winner,
        "row": results[winner][0],
        "tried": list(strategies.keys())
    }

if __name__ == "__main__":
    file_path = "pom3a.train.csv"
    rows_iter = ezr.csv(file_path)
    header = next(rows_iter)
    all_rows = list(rows_iter)
    
    # Mask goals
    masked_rows = [[row[i] if header[i][-1] not in "+-" else "?" for i in range(len(row))] for row in all_rows]
    
    def label_func(i):
        return all_rows[i]
    
    result = race(header, masked_rows, label_func)
    
    print(f"Race Results:")
    print(f"Winner: {result['winner']}")
    print(f"Best row index found: {result['row']}")
    print(f"Strategies compared: {', '.join(result['tried'])}")
    
    exit(0)
