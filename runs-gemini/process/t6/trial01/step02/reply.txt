import ezr
import step01

def guess(header, train, test):
    """
    Estimates the first goal column for each row in test.
    Uses the tree-based ranker from ezr to find the leaf mean.
    """
    # Create table from training data
    tbl = ezr.Tbl([header] + train)
    
    # Identify the first goal column index
    goal_col = None
    for i, name in enumerate(header):
        if name[-1] in "+-":
            goal_col = i
            break
            
    # Build a tree on the training data to predict goal values
    # We need a custom leaf mean for the specific goal column
    node = ezr.tree(tbl, tbl.rows)
    
    def get_leaf_mean(row):
        leaf = ezr.leaf(node, row)
        # Calculate mean of the specific goal column for rows in this leaf
        vals = [r[goal_col] for r in leaf.rows if r[goal_col] != "?"]
        return sum(vals) / len(vals) if vals else 0.0

    return [get_leaf_mean(row) for row in test]

if __name__ == "__main__":
    # Load data
    file_path = "pom3a.train.csv"
    rows_iter = ezr.csv(file_path)
    header = next(rows_iter)
    all_rows = list(rows_iter)
    
    # Split data: 80% train, 20% test
    n = int(len(all_rows) * 0.8)
    train = all_rows[:n]
    test = all_rows[n:]
    
    # Mask goals in test set
    masked_test = []
    for row in test:
        new_row = [row[i] if header[i][-1] not in "+-" else "?" for i in range(len(row))]
        masked_test.append(new_row)
        
    # Get estimates
    estimates = guess(header, train, masked_test)
    
    # Identify goal names
    goal_names = [name for name in header if name[-1] in "+-"]
    
    print(f"Loaded {len(all_rows)} records.")
    print(f"Estimated '{goal_names[0]}' for {len(estimates)} test records.")
    print(f"Sample estimates: {', '.join(f'{e:.3f}' for e in estimates[:5])}")
    
    exit(0)
