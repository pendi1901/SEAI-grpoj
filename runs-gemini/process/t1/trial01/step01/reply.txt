import csv

def remember(header, rows):
    """
    Identifies rows that perform well on all goals.
    Goals ending in '-' are minimized, '+' are maximized.
    We use a simple dominance/ranking approach: normalize goals to [0,1]
    and pick rows with the lowest sum of normalized costs.
    """
    goals = []
    for i, name in enumerate(header):
        if name.endswith('-') or name.endswith('+'):
            goals.append((i, name))

    # Extract valid numeric data for goals
    data = []
    for idx, row in enumerate(rows):
        vals = []
        valid = True
        for col_idx, name in goals:
            val = row[col_idx]
            if val == "?":
                valid = False
                break
            vals.append(float(val))
        if valid:
            data.append((idx, vals))

    # Normalize goals: 0 is best, 1 is worst
    n_goals = len(goals)
    n_rows = len(data)
    normalized = []
    for i in range(n_goals):
        col_vals = [d[1][i] for d in data]
        min_v, max_v = min(col_vals), max(col_vals)
        rng = max_v - min_v if max_v != min_v else 1.0
        
        for j in range(n_rows):
            val = data[j][1][i]
            # If goal ends in -, lower is better. If +, higher is better.
            if goals[i][1].endswith('-'):
                norm = (val - min_v) / rng
            else:
                norm = (max_v - val) / rng
            
            if len(normalized) <= j:
                normalized.append(norm)
            else:
                normalized[j] += norm

    # Combine and sort
    scored = []
    for j in range(n_rows):
        scored.append((normalized[j], data[j][0]))
    
    scored.sort()
    
    # Return top 5% or at least 1
    limit = max(1, int(len(rows) * 0.05))
    return [s[1] for s in scored[:limit]]

if __name__ == "__main__":
    filename = "pom3a.train.csv"
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        rows = list(reader)

    best_indices = remember(header, rows)
    
    print(f"Found {len(best_indices)} top-performing setups.")
    print(f"{'Row Index':<10} | {'Goals (Normalized Score)':<25}")
    print("-" * 40)
    for idx in best_indices:
        print(f"{idx:<10} | Best-in-class")
