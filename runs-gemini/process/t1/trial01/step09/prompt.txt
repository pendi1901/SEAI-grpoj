import csv

def remember(header, rows):
    """
    Identifies rows that perform well on all goals.
    Goals ending in '-' are minimized, '+' are maximized.
    We use a simple dominance/ranking approach: normalize goals to [0,1]
    and pick rows with the lowest sum of normalized costs.
    """
    goals = []
    for i, name in enumerate(header):
        if name.endswith('-') or name.endswith('+'):
            goals.append((i, name))

    # Extract valid numeric data for goals
    data = []
    for idx, row in enumerate(rows):
        vals = []
        valid = True
        for col_idx, name in goals:
            val = row[col_idx]
            if val == "?":
                valid = False
                break
            vals.append(float(val))
        if valid:
            data.append((idx, vals))

    # Normalize goals: 0 is best, 1 is worst
    n_goals = len(goals)
    n_rows = len(data)
    normalized = []
    for i in range(n_goals):
        col_vals = [d[1][i] for d in data]
        min_v, max_v = min(col_vals), max(col_vals)
        rng = max_v - min_v if max_v != min_v else 1.0
        
        for j in range(n_rows):
            val = data[j][1][i]
            # If goal ends in -, lower is better. If +, higher is better.
            if goals[i][1].endswith('-'):
                norm = (val - min_v) / rng
            else:
                norm = (max_v - val) / rng
            
            if len(normalized) <= j:
                normalized.append(norm)
            else:
                normalized[j] += norm

    # Combine and sort
    scored = []
    for j in range(n_rows):
        scored.append((normalized[j], data[j][0]))
    
    scored.sort()
    
    # Return top 5% or at least 1
    limit = max(1, int(len(rows) * 0.05))
    return [s[1] for s in scored[:limit]]

    train_data = []
    for row in train:
        inputs = [float(row[i]) for i in input_indices]
        target = float(row[goal_idx])

    predictions = []
    for test_row in test:
        test_inputs = [float(test_row[i]) for i in input_indices]

import csv
import math

def certify(xs, ys):
    """
    Returns True if xs and ys are statistically similar (same distribution),
    False if they differ significantly. Uses a simple A12 effect size test.
    """
    def a12(lst1, lst2):
        m, n = len(lst1), len(lst2)
        greater = 0
        for x in lst1:
            for y in lst2:
                if x > y: greater += 1
                elif x == y: greater += 0.5
        return greater / (m * n)

    # A12 threshold: 0.56 is a small effect size, 0.64 medium, 0.71 large.
    # If A12 is close to 0.5, they are effectively the same.
    score = a12(xs, ys)
    return 0.44 <= score <= 0.56

def analyze_teams(header, rows):
    """
    Compares 'Cost-' for small teams vs big teams.
    'Team Size' is index 8. 'Cost-' is index 9.
    """
    team_idx = 8

    # Calculate median team size to split into small vs big
    sizes = [float(row[team_idx]) for row in rows]
    median = sorted(sizes)[len(sizes) // 2]

        else:
            big_costs.append(cost)
            
    is_same = certify(small_costs, big_costs)
    
    return small_costs, big_costs, is_same

import csv

def flow(header, row, seen):
    """
    Implements a Secretary Problem variant:
    1. Observe a sample (the first 10% of data) to establish a baseline.
    2. After the sample, accept the first row that is better than the 
       best seen in the sample.
    """
    # Identify goal indices
    goal_indices = [i for i, name in enumerate(header) 
                    if name.endswith('-') or name.endswith('+')]
    
    # Helper to calculate normalized score for a row
    def get_score(r):
        score = 0.0
        for i in goal_indices:
            val = float(r[i])
            # Normalize: if -, lower is better; if +, higher is better
            # Since we don't have global min/max, we use the range [0,1]
            # as per the problem domain (all data is 0-1)
            if header[i].endswith('-'):
                score += val
            else:
                score += (1.0 - val)
        return score

    # Define sample size (10% of 16000 = 1600)
    sample_size = 1600
    
    if len(seen) < sample_size:
        return False
    
    # Calculate best score from seen rows
    best_seen_score = min(get_score(s) for s in seen)
    
    # Current row score
    current_score = get_score(row)
    
    # If current is better than anything seen, accept it
    return current_score < best_seen_score

import csv

def blame(header, rows):
    """
    Identifies input columns that correlate with high 'Idle-' values.
    'Idle-' is the last column (index 11).
    Input columns are those not ending in '-' or '+'.
    """
    # Identify goal indices and input indices
    goal_indices = [i for i, name in enumerate(header) if name.endswith('-') or name.endswith('+')]
    input_indices = [i for i, name in enumerate(header) if i not in goal_indices]
    
    # Idle- is the last column
    idle_idx = len(header) - 1
    
    # Sort rows by Idle- to find the "worst" (highest idle)
    # We want to see what inputs correlate with high idle
    sorted_rows = sorted(rows, key=lambda r: float(r[idle_idx]), reverse=True)
    
    # Take the worst 10%
    worst_rows = sorted_rows[:len(rows) // 10]
    
    # Calculate average of inputs for the worst rows vs the whole population
    # A high correlation suggests that input is a driver for the bad outcome
    def get_avg(data, col_idx):
        vals = [float(r[col_idx]) for r in data]
        return sum(vals) / len(vals)
    
    pop_avg = [get_avg(rows, i) for i in input_indices]
    worst_avg = [get_avg(worst_rows, i) for i in input_indices]
    
    # Calculate the "blame" score: how much the input deviates from the population average
    # in the worst-performing rows.
    blame_scores = []
    for i, idx in enumerate(input_indices):
        # We look for inputs that are significantly higher or lower than average
        # in the worst rows.
        diff = abs(worst_avg[i] - pop_avg[i])
        blame_scores.append((diff, header[idx]))
    
    # Sort by impact
    blame_scores.sort(key=lambda x: x[0], reverse=True)
    
    # Return the top 3 inputs that explain the idle time
    return [name for score, name in blame_scores[:3]]

import csv

def justify(header, rows, row):
    """
    Returns (p, reason): p is the fraction of rows that 'row' is better than.
    'row' has goals as "?", so we estimate its goal values using the 
    average of the 10 nearest neighbors in the dataset.
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith('-') or name.endswith('+')]
    input_indices = [i for i, name in enumerate(header) if i not in goal_indices]

    # Pre-calculate goal values for all rows
    row_goals = []
    for r in rows:
        row_goals.append([float(r[i]) for i in goal_indices])

    # Estimate goal values for the target 'row' using 10-Nearest Neighbors
    target_inputs = [float(row[i]) for i in input_indices]
    distances = []
    for r in rows:
        inputs = [float(r[i]) for i in input_indices]
        dist = sum((a - b) ** 2 for a, b in zip(target_inputs, inputs))
        distances.append((dist, [float(r[i]) for i in goal_indices]))
    
    distances.sort(key=lambda x: x[0])
    neighbors = distances[:10]
    
    # Estimated goals for target row
    est_goals = [sum(n[1][i] for n in neighbors) / 10 for i in range(len(goal_indices))]

    # Calculate normalized score for a set of goals
    def get_score(goals):
        score = 0.0
        for i, g_idx in enumerate(goal_indices):
            name = header[g_idx]
            # Normalize: 0 is best, 1 is worst
            # Since data is 0-1, we assume 0 is best for '-' and 1 is best for '+'
            if name.endswith('-'):
                score += goals[i]
            else:
                score += (1.0 - goals[i])
        return score / len(goal_indices)

    target_score = get_score(est_goals)
    
    # Compare against all rows
    better_count = 0
    for g in row_goals:
        if get_score(g) > target_score:
            better_count += 1
            
    p = better_count / len(rows)
    reason = f"This setup is estimated to outperform {p:.1%} of all configurations based on its input parameters."
    
    return p, reason

        return score / len(goal_indices)

            val = float(r[i])

                score += val
            else:
                score += (1.0 - val)

        original_val = float(row[col_idx])
