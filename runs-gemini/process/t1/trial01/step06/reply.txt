import csv

def justify(header, rows, row):
    """
    Returns (p, reason): p is the fraction of rows that 'row' is better than.
    'row' has goals as "?", so we estimate its goal values using the 
    average of the 10 nearest neighbors in the dataset.
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith('-') or name.endswith('+')]
    input_indices = [i for i, name in enumerate(header) if i not in goal_indices]

    # Pre-calculate goal values for all rows
    row_goals = []
    for r in rows:
        row_goals.append([float(r[i]) for i in goal_indices])

    # Estimate goal values for the target 'row' using 10-Nearest Neighbors
    target_inputs = [float(row[i]) for i in input_indices]
    distances = []
    for r in rows:
        inputs = [float(r[i]) for i in input_indices]
        dist = sum((a - b) ** 2 for a, b in zip(target_inputs, inputs))
        distances.append((dist, [float(r[i]) for i in goal_indices]))
    
    distances.sort(key=lambda x: x[0])
    neighbors = distances[:10]
    
    # Estimated goals for target row
    est_goals = [sum(n[1][i] for n in neighbors) / 10 for i in range(len(goal_indices))]

    # Calculate normalized score for a set of goals
    def get_score(goals):
        score = 0.0
        for i, g_idx in enumerate(goal_indices):
            name = header[g_idx]
            # Normalize: 0 is best, 1 is worst
            # Since data is 0-1, we assume 0 is best for '-' and 1 is best for '+'
            if name.endswith('-'):
                score += goals[i]
            else:
                score += (1.0 - goals[i])
        return score / len(goal_indices)

    target_score = get_score(est_goals)
    
    # Compare against all rows
    better_count = 0
    for g in row_goals:
        if get_score(g) > target_score:
            better_count += 1
            
    p = better_count / len(rows)
    reason = f"This setup is estimated to outperform {p:.1%} of all configurations based on its input parameters."
    
    return p, reason

if __name__ == "__main__":
    filename = "pom3a.train.csv"
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        rows = [row for row in reader]

    # Pick a sample row to justify (e.g., the last one)
    sample_row = rows[-1][:]
    # Mask goals
    for i in range(len(sample_row)):
        if header[i].endswith('-') or header[i].endswith('+'):
            sample_row[i] = "?"

    p, reason = justify(header, rows, sample_row)
    print(f"Rank: {p:.4f}")
    print(f"Justification: {reason}")
