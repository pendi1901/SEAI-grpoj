import csv
import random

def choose(header, rows, label):
    """
    Finds the best row by sampling 50 rows, labeling them, 
    and returning the index of the best one.
    """
    goal_indices = [i for i, name in enumerate(header) 
                    if name.endswith('-') or name.endswith('+')]
    
    def get_score(goals):
        score = 0.0
        for i, g_idx in enumerate(goal_indices):
            name = header[g_idx]
            # Normalize: 0 is best, 1 is worst.
            # For '-', lower is better. For '+', higher is better.
            if name.endswith('-'):
                score += goals[i]
            else:
                score += (1.0 - goals[i])
        return score / len(goal_indices)

    # We can afford 50 trials.
    # We pick 50 random indices to sample the space.
    indices = random.sample(range(len(rows)), 50)
    
    best_idx = -1
    best_score = float('inf')
    
    for i in indices:
        goals = label(i)
        score = get_score(goals)
        if score < best_score:
            best_score = score
            best_idx = i
            
    return best_idx

if __name__ == "__main__":
    filename = "pom3a.train.csv"
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        rows = list(reader)
        # Store true values for the label function
        true_data = []
        for row in rows:
            true_data.append([float(row[i]) for i, name in enumerate(header) 
                              if name.endswith('-') or name.endswith('+')])

    def label(i):
        return true_data[i]

    best_row_idx = choose(header, rows, label)
    
    print(f"Analysis complete.")
    print(f"After trialing 50 configurations, the best setup found is at row index: {best_row_idx}")
    print(f"This row was selected based on the lowest normalized goal score.")
