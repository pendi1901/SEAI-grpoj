import csv

def remember(header, rows):
    """
    Identifies rows that perform well on all goals.
    Goals ending in '-' are minimized, '+' are maximized.
    We use a simple dominance/ranking approach: normalize goals to [0,1]
    and pick rows with the lowest sum of normalized costs.
    """
    goals = []
    for i, name in enumerate(header):
        if name.endswith('-') or name.endswith('+'):
            goals.append((i, name))

    # Extract valid numeric data for goals
    data = []
    for idx, row in enumerate(rows):
        vals = []
        valid = True
        for col_idx, name in goals:
            val = row[col_idx]
            if val == "?":
                valid = False
                break
            vals.append(float(val))
        if valid:
            data.append((idx, vals))

    # Normalize goals: 0 is best, 1 is worst
    n_goals = len(goals)
    n_rows = len(data)
    normalized = []
    for i in range(n_goals):
        col_vals = [d[1][i] for d in data]
        min_v, max_v = min(col_vals), max(col_vals)
        rng = max_v - min_v if max_v != min_v else 1.0
        
        for j in range(n_rows):
            val = data[j][1][i]
            # If goal ends in -, lower is better. If +, higher is better.
            if goals[i][1].endswith('-'):
                norm = (val - min_v) / rng
            else:
                norm = (max_v - val) / rng
            
            if len(normalized) <= j:
                normalized.append(norm)
            else:
                normalized[j] += norm

    # Combine and sort
    scored = []
    for j in range(n_rows):
        scored.append((normalized[j], data[j][0]))
    
    scored.sort()
    
    # Return top 5% or at least 1
    limit = max(1, int(len(rows) * 0.05))
    return [s[1] for s in scored[:limit]]

import csv

def guess(header, train, test):
    """
    Estimates the first goal column (Cost-) for test rows using a 
    k-Nearest Neighbors approach based on input columns.
    """
    # Identify input columns (those not ending in + or -)
    input_indices = [i for i, name in enumerate(header) 
                     if not (name.endswith('-') or name.endswith('+'))]
    
    # Identify the first goal column index
    goal_idx = next(i for i, name in enumerate(header) 
                    if name.endswith('-') or name.endswith('+'))
    
    # Pre-process training data: inputs and the target goal
    train_data = []
    for row in train:
        inputs = [float(row[i]) for i in input_indices]
        target = float(row[goal_idx])
        train_data.append((inputs, target))
    
    predictions = []
    for test_row in test:
        test_inputs = [float(test_row[i]) for i in input_indices]
        
        # Find nearest neighbor (Euclidean distance)
        best_dist = float('inf')
        best_val = 0.0
        
        for train_inputs, target in train_data:
            dist = sum((a - b) ** 2 for a, b in zip(test_inputs, train_inputs))
            if dist < best_dist:
                best_dist = dist
                best_val = target
        
        predictions.append(best_val)
        
    return predictions

import csv
import math

def certify(xs, ys):
    """
    Returns True if xs and ys are statistically similar (same distribution),
    False if they differ significantly. Uses a simple A12 effect size test.
    """
    def a12(lst1, lst2):
        m, n = len(lst1), len(lst2)
        greater = 0
        for x in lst1:
            for y in lst2:
                if x > y: greater += 1
                elif x == y: greater += 0.5
        return greater / (m * n)

    # A12 threshold: 0.56 is a small effect size, 0.64 medium, 0.71 large.
    # If A12 is close to 0.5, they are effectively the same.
    score = a12(xs, ys)
    return 0.44 <= score <= 0.56

def analyze_teams(header, rows):
    """
    Compares 'Cost-' for small teams vs big teams.
    'Team Size' is index 8. 'Cost-' is index 9.
    """
    team_idx = 8
    cost_idx = 9
    
    # Calculate median team size to split into small vs big
    sizes = [float(row[team_idx]) for row in rows]
    median = sorted(sizes)[len(sizes) // 2]
    
    small_costs = []
    big_costs = []
    
    for row in rows:
        cost = float(row[cost_idx])
        if float(row[team_idx]) <= median:
            small_costs.append(cost)
        else:
            big_costs.append(cost)
            
    is_same = certify(small_costs, big_costs)
    
    return small_costs, big_costs, is_same
