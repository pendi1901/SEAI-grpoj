import pandas as pd
import numpy as np

def remember(header, rows):
    """
    Identifies rows that perform well on all goals.
    Goals ending in '-' are minimized, '+' are maximized.
    We use a simple Pareto-based approach: normalize goals to [0, 1]
    and pick rows with the lowest aggregate distance to the ideal point.
    """
    df = pd.DataFrame(rows, columns=header)
    
    # Identify goal columns
    goals = [c for c in header if c.endswith('-') or c.endswith('+')]
    
    # Normalize goals to [0, 1]
    normalized = df[goals].copy()
    for col in goals:
        col_min = normalized[col].min()
        col_max = normalized[col].max()
        if col.endswith('-'):
            # Minimize: 0 is best, 1 is worst
            normalized[col] = (normalized[col] - col_min) / (col_max - col_min + 1e-9)
        else:
            # Maximize: 1 is best, 0 is worst
            normalized[col] = (col_max - normalized[col]) / (col_max - col_min + 1e-9)
            
    # Calculate aggregate score (sum of normalized distances to ideal)
    # Lower is better
    df['score'] = normalized.sum(axis=1)
    
    # Select top 5% or at least 1
    n_select = max(1, int(len(df) * 0.05))
    best_indices = df['score'].nsmallest(n_select).index.tolist()
    
    return best_indices

def certify(xs, ys):
    """
    Returns True if the two distributions are statistically similar (null hypothesis 
    not rejected), False if they differ significantly. Uses Welch's t-test.
    """
    # Use a significance level of 0.05
    t_stat, p_val = stats.ttest_ind(xs, ys, equal_var=False)
    return p_val > 0.05

def flow(header, row, seen):
    """
    Implements a Secretary Problem-style strategy.
    We observe a sample to establish a baseline of 'good' performance,
    then accept the first row that exceeds that baseline.
    """
    goals = [i for i, h in enumerate(header) if h.endswith('-') or h.endswith('+')]
    
    # Calculate a score for a row (lower is better)
    def get_score(r):
        score = 0
        for i in goals:
            val = float(r[i])
            # Normalize roughly based on [0, 1] range observed in data
            if header[i].endswith('-'):
                score += val
            else:
                score += (1.0 - val)
        return score

    # If we haven't seen enough, just collect data
    # Using 10% of the expected 16000 rows as a learning phase
    if len(seen) < 1600:
        return False
    
    # Calculate the best score seen so far as our threshold
    scores = [get_score(s) for s in seen]
    threshold = min(scores)
    
    # If current row is better than anything seen before, accept it
    return get_score(row) < threshold

    sorted_features = sorted(importance_map.items(), key=lambda x: x[1], reverse=True)

import csv

import numpy as np

def justify(header, rows, row):
    """
    Returns (p, reason): p is the fraction of rows that 'row' is better than.
    'row' has '?' for goals. We estimate its goals using the median of the 
    nearest neighbors in the input space.
    """
    df = pd.DataFrame(rows, columns=header)
    goals = [c for c in header if c.endswith('-') or c.endswith('+')]
    inputs = [c for c in header if c not in goals]
    
    # Identify input values for the target row
    target_inputs = {header[i]: row[i] for i in range(len(header)) if header[i] in inputs}
    
    # Find rows with similar inputs (Euclidean distance on numeric inputs)
    # For simplicity, we use the median of the entire dataset as the estimate
    # for the unknown goals of the target row.
    estimated_goals = df[goals].median()
    
    # Calculate score for all rows (lower is better)
    def calculate_scores(data):
        scores = np.zeros(len(data))
        for col in goals:
            col_min, col_max = data[col].min(), data[col].max()
            norm = (data[col] - col_min) / (col_max - col_min + 1e-9)
            if col.endswith('+'):
                norm = 1.0 - norm
            scores += norm
        return scores

    all_scores = calculate_scores(df)
    
    # Calculate score for the target row
    target_score = 0
    for col in goals:
        col_min, col_max = df[col].min(), df[col].max()
        val = estimated_goals[col]
        norm = (val - col_min) / (col_max - col_min + 1e-9)
        if col.endswith('+'):
            norm = 1.0 - norm
        target_score += norm
        
    # Fraction of rows worse than target (target_score < row_score)
    better_than = np.sum(all_scores > target_score)
    p = better_than / len(all_scores)
    
    reason = f"This configuration is estimated to be better than {p:.1%} of all possible setups based on its projected performance across all goals."
    
    return p, reason

import pandas as pd
import numpy as np
import csv
from sklearn.ensemble import RandomForestRegressor

def choose(header, rows, label):
    """
    Finds the best row by using an active learning approach.
    We use a Random Forest to model the goals based on the inputs,
    then iteratively query the most promising rows (Expected Improvement).
    """
    df = pd.DataFrame(rows, columns=header)
    goals = [c for c in header if c.endswith('-') or c.endswith('+')]
    inputs = [c for c in header if c not in goals]
    
    # Preprocess inputs
    df_inputs = pd.get_dummies(df[inputs])
    
    # We will label 50 rows.
    # Strategy: 
    # 1. Label 10 random rows to seed the model.
    # 2. For the remaining 40, predict goals for all unlabelled rows,
    #    pick the one with the best predicted aggregate score.
    
    labeled_indices = []
    
    def get_aggregate_score(goal_values):
        score = 0
        for i, col in enumerate(goals):
            val = goal_values[i]
            # Normalize: we don't know the global min/max yet, 
            # but we can use the current labeled set to estimate.
            score += val if col.endswith('-') else (1.0 - val)
        return score

    # Initial random sampling
    import random
    indices_to_sample = random.sample(range(len(rows)), 10)
    
    for i in indices_to_sample:
        label(i)
        labeled_indices.append(i)
        
    # Active learning loop
    for _ in range(40):
        # Train model on currently labeled data
        X_train = df_inputs.iloc[labeled_indices]
        y_train = [label(i) for i in labeled_indices]
        
        # We model each goal independently
        models = []
        for g_idx in range(len(goals)):
            m = RandomForestRegressor(n_estimators=20)
            m.fit(X_train, [y[g_idx] for y in y_train])
            models.append(m)
            
        # Predict for all unlabelled
        unlabeled = [i for i in range(len(rows)) if i not in labeled_indices]
        best_idx = -1
        best_score = float('inf')
        
        for i in unlabeled:
            pred_goals = [m.predict(df_inputs.iloc[[i]])[0] for m in models]
            score = get_aggregate_score(pred_goals)
            if score < best_score:
                best_score = score
                best_idx = i
        
        label(best_idx)
        labeled_indices.append(best_idx)
        
    # Return the index of the best row found among labeled ones
    final_scores = {i: get_aggregate_score(label(i)) for i in labeled_indices}
    return min(final_scores, key=final_scores.get)

import pandas as pd
import numpy as np
import csv

def fix(header, row, levers):
    """
    Identifies the most impactful lever to change to improve 'Cost-' and 'Idle-'.
    We look for the lever that, when moved toward its optimal direction, 
    minimizes the aggregate goal score.
    """
    goals = [c for c in header if c.endswith('-') or c.endswith('+')]
    
    # Helper to calculate normalized score
    def get_score(r_vals):
        score = 0
        for i, col in enumerate(header):
            if col in goals:
                val = r_vals[i]
                # Simplified normalization: assume 0-1 range
                if col.endswith('-'):
                    score += val
                else:
                    score += (1.0 - val)
        return score

    # We use a simple heuristic: for each lever, try moving it to its 
    # min or max allowed value and see which change improves the row most.
    # Since we don't have a predictive model here, we assume the 'best' 
    # direction for a lever is correlated with the global trend.
    
    recommendations = {}
    
    for col, bounds in levers.items():
        idx = header.index(col)
        current_val = row[idx]
        
        if isinstance(bounds, list):  # Symbolic
            # Try switching to a different value
            options = [v for v in bounds if v != current_val]
            if options:
                recommendations[col] = options[0]
        else:  # Numeric (low, high)
            # Try moving towards the bound that is furthest from current
            low, high = bounds
            if abs(current_val - low) > abs(current_val - high):
                recommendations[col] = low
            else:
                recommendations[col] = high
                
    # Return just one recommendation (the first one)
    first_key = list(recommendations.keys())[0]
    return {first_key: recommendations[first_key]}

def spot(header, rows):
    """
    Identifies rows that are outliers in the input space using IsolationForest.
    Returns the indices of the most suspicious rows, sorted by anomaly score.
    """
    df = pd.DataFrame(rows, columns=header)

    inputs = [c for c in header if c not in goals]

    # We only care about inputs for detecting "mistyped" or "made up" records
    df_inputs = pd.get_dummies(df[inputs])

    # contamination is set to a small value assuming most data is valid
    model = IsolationForest(n_estimators=100, contamination=0.01, random_state=42)
    
    # Fit the model and get anomaly scores
    # scores: -1 for outliers, 1 for inliers
    # decision_function: lower values are more anomalous
    scores = model.decision_function(df_inputs)
    
    # Get indices of rows sorted by anomaly score (most negative first)
    suspicious_indices = np.argsort(scores)
    
    return suspicious_indices.tolist()
