import ezr
import random

def spot(header, rows):
    """
    Identify rows that are 'suspicious' (outliers in the input space).
    We define suspicion as having a high xdist (distance from the centroid 
    of the input space). Rows that are far from the 'typical' configuration 
    are flagged as potential failed runs.
    """
    # Create a table to use ezr's distance functions
    tbl = ezr.Tbl([header] + rows)
    
    # Calculate the centroid of the input space (mids)
    # ezr.mids(tbl) returns a dictionary of column medians/modes
    centroid = ezr.mids(tbl)
    
    # Calculate xdist for every row relative to the centroid
    # xdist measures how far a row's inputs are from the typical inputs
    scores = []
    for i, row in enumerate(rows):
        dist = ezr.xdist(tbl, row, centroid)
        scores.append((dist, i))
    
    # Sort by distance (descending) to put the most suspicious rows first
    scores.sort(key=lambda x: x[0], reverse=True)
    
    # Return the indices of the rows
    return [idx for dist, idx in scores]

if __name__ == "__main__":
    filename = "Health-ClosedIssues0000.train.csv"
    all_data = list(ezr.csv(filename))
    header = all_data[0]
    rows = all_data[1:]
    
    # Identify suspicious rows
    suspicious_indices = spot(header, rows)
    
    print(f"Analysis complete on {len(rows)} records.")
    print("Identifying potentially failed runs (outliers in input space).")
    print("Indices of the 10 most suspicious rows:")
    print(suspicious_indices[:10])
    
    # Provide a bit more context for the top 3
    print("\nDetails of the top 3 most suspicious rows:")
    for i in range(3):
        idx = suspicious_indices[i]
        print(f"Row {idx}: {rows[idx]}")
    
    exit(0)
