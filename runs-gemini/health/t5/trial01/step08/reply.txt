import ezr
import random

def fix(header, row, levers):
    """
    Find the smallest change to settings (levers) that improves the row's 
    performance (ydist) without violating constraints.
    """
    # Load the full dataset to build a model of the space
    filename = "Health-ClosedIssues0000.train.csv"
    all_data = list(ezr.csv(filename))
    header_data = all_data[0]
    rows = all_data[1:]
    tbl = ezr.Tbl(all_data)
    
    # Identify the current performance
    # We use the median of the column as a neutral estimate for missing goals
    current_row = list(row)
    for i, val in enumerate(current_row):
        if val == "?":
            current_row[i] = ezr.mid(tbl.cols[i])
    
    best_ydist = ezr.ydist(tbl, current_row)
    best_change = {}
    
    # Try small perturbations for each lever
    for col_name, allowed in levers.items():
        idx = header.index(col_name)
        col = tbl.cols[idx]
        
        if isinstance(col, ezr.Num):
            # For numeric, try moving towards the center of the distribution
            low, high = allowed
            current_val = row[idx]
            # Try a small step (10% of range)
            step = (high - low) * 0.1
            for direction in [-1, 1]:
                new_val = max(low, min(high, current_val + (direction * step)))
                if new_val != current_val:
                    test_row = list(current_row)
                    test_row[idx] = new_val
                    new_ydist = ezr.ydist(tbl, test_row)
                    if new_ydist < best_ydist:
                        best_ydist = new_ydist
                        best_change = {col_name: new_val}
        else:
            # For symbolic, try other allowed values
            for val in allowed:
                if val != row[idx]:
                    test_row = list(current_row)
                    test_row[idx] = val
                    new_ydist = ezr.ydist(tbl, test_row)
                    if new_ydist < best_ydist:
                        best_ydist = new_ydist
                        best_change = {col_name: val}
                        
    return best_change

if __name__ == "__main__":
    filename = "Health-ClosedIssues0000.train.csv"
    all_data = list(ezr.csv(filename))
    header = all_data[0]
    rows = all_data[1:]
    
    # Define levers based on column types
    levers = {
        "N_estimators": (10, 200),
        "criterion": ["squared_error", "absolute_error", "friedman_mse", "poisson"],
        "Min_sample_leaves": (1, 20),
        "Min_impurity_decrease": (0.0, 10.0),
        "Max_depth": (1, 20)
    }
    
    # Pick a "bad" row (high ydist) to fix
    tbl = ezr.Tbl(all_data)
    sorted_rows = sorted(rows, key=lambda r: ezr.ydist(tbl, r), reverse=True)
    bad_row = sorted_rows[0]
    
    # Mask goals
    masked_row = [bad_row[i] if not header[i].endswith(('-', '+')) else "?" 
                  for i in range(len(bad_row))]
    
    recommendation = fix(header, masked_row, levers)
    
    print("Optimization Analysis:")
    print(f"Bad configuration identified: {masked_row}")
    if recommendation:
        print(f"Recommended change to improve performance: {recommendation}")
    else:
        print("No single lever change found to improve performance.")
    
    exit(0)
