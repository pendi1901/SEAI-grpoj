import csv

def remember(header, rows):
    """
    Identifies the best rows based on multi-objective optimization.
    Goals ending in '-' are minimized, '+' are maximized.
    We normalize numeric values to [0, 1] and use a simple distance metric
    to the ideal point (0 for '-', 1 for '+').
    """
    goals = []
    for i, name in enumerate(header):
        if name.endswith('-'):
            goals.append((i, 'min'))
        elif name.endswith('+'):
            goals.append((i, 'max'))

    # Collect numeric data for normalization
    data = []
    for row in rows:
        clean_row = []
        for val in row:
            try:
                clean_row.append(float(val))
            except (ValueError, TypeError):
                clean_row.append(None)
        data.append(clean_row)

    # Find min/max for each goal column to normalize
    ranges = {}
    for idx, mode in goals:
        vals = [r[idx] for r in data if r[idx] is not None]
        if vals:
            ranges[idx] = (min(vals), max(vals))

    scores = []
    for i, row in enumerate(data):
        dist = 0
        count = 0
        for idx, mode in goals:
            val = row[idx]
            if val is not None and idx in ranges:
                low, high = ranges[idx]
                span = high - low if high > low else 1
                norm = (val - low) / span
                # If minimizing, ideal is 0. If maximizing, ideal is 1.
                dist += (norm if mode == 'min' else (1 - norm)) ** 2
                count += 1
        if count > 0:
            scores.append((dist / count, i))

    # Sort by distance (ascending) and return top 5% (or at least 1)
    scores.sort()
    limit = max(1, len(rows) // 20)
    return [item[1] for item in scores[:limit]]

import csv

def guess(header, train, test):
    """
    Estimates the first goal column (MRE-) for test rows.
    Uses a k-Nearest Neighbors approach based on Euclidean distance 
    of numeric input columns.
    """
    # Identify numeric input columns (not goals, not symbolic)
    # Symbolic columns start with lowercase, goals end with - or +
    input_indices = []
    for i, name in enumerate(header):
        if name[0].isupper() and not (name.endswith('-') or name.endswith('+')):
            input_indices.append(i)

    # Normalize training data for distance calculation
    train_inputs = []
    train_targets = [float(row[0]) for row in train]
    
    # Calculate ranges for normalization
    mins = []
    maxs = []
    for i in input_indices:
        vals = [float(row[i]) for row in train]
        mins.append(min(vals))
        maxs.append(max(vals) - min(vals) if max(vals) > min(vals) else 1.0)
        train_inputs.append([float(row[i]) for row in train])

    # Transpose train_inputs for easier row access
    train_rows = []
    for r in range(len(train)):
        train_rows.append([train_inputs[c][r] for c in range(len(input_indices))])

    predictions = []
    for test_row in test:
        test_vals = []
        for i, idx in enumerate(input_indices):
            val = (float(test_row[idx]) - mins[i]) / maxs[i]
            test_vals.append(val)
        
        # Find nearest neighbors (k=5)
        distances = []
        for i, row in enumerate(train_rows):
            norm_row = [(row[j] - mins[j]) / maxs[j] for j in range(len(input_indices))]
            dist = sum((norm_row[j] - test_vals[j])**2 for j in range(len(input_indices)))
            distances.append((dist, train_targets[i]))
        
        distances.sort()
        # Average of top 5 neighbors
        k = 5
        avg = sum(d[1] for d in distances[:k]) / k
        predictions.append(avg)
        
    return predictions
