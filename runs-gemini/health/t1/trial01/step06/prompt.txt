import csv

def remember(header, rows):
    """
    Identifies the best rows based on multi-objective optimization.
    Goals ending in '-' are minimized, '+' are maximized.
    We normalize numeric values to [0, 1] and use a simple distance metric
    to the ideal point (0 for '-', 1 for '+').
    """
    goals = []
    for i, name in enumerate(header):
        if name.endswith('-'):
            goals.append((i, 'min'))
        elif name.endswith('+'):
            goals.append((i, 'max'))

    # Collect numeric data for normalization
    data = []
    for row in rows:
        clean_row = []
        for val in row:
            try:
                clean_row.append(float(val))
            except (ValueError, TypeError):
                clean_row.append(None)
        data.append(clean_row)

    # Find min/max for each goal column to normalize
    ranges = {}
    for idx, mode in goals:
        vals = [r[idx] for r in data if r[idx] is not None]
        if vals:
            ranges[idx] = (min(vals), max(vals))

    scores = []
    for i, row in enumerate(data):
        dist = 0
        count = 0
        for idx, mode in goals:
            val = row[idx]
            if val is not None and idx in ranges:
                low, high = ranges[idx]
                span = high - low if high > low else 1
                norm = (val - low) / span
                # If minimizing, ideal is 0. If maximizing, ideal is 1.
                dist += (norm if mode == 'min' else (1 - norm)) ** 2
                count += 1
        if count > 0:
            scores.append((dist / count, i))

    # Sort by distance (ascending) and return top 5% (or at least 1)
    scores.sort()
    limit = max(1, len(rows) // 20)
    return [item[1] for item in scores[:limit]]

import csv

def guess(header, train, test):
    """
    Estimates the first goal column (MRE-) for test rows.
    Uses a k-Nearest Neighbors approach based on Euclidean distance 
    of numeric input columns.
    """
    # Identify numeric input columns (not goals, not symbolic)
    # Symbolic columns start with lowercase, goals end with - or +
    input_indices = []
    for i, name in enumerate(header):
        if name[0].isupper() and not (name.endswith('-') or name.endswith('+')):
            input_indices.append(i)

    # Normalize training data for distance calculation
    train_inputs = []
    train_targets = [float(row[0]) for row in train]
    
    # Calculate ranges for normalization
    mins = []
    maxs = []
    for i in input_indices:
        vals = [float(row[i]) for row in train]
        mins.append(min(vals))
        maxs.append(max(vals) - min(vals) if max(vals) > min(vals) else 1.0)
        train_inputs.append([float(row[i]) for row in train])

    # Transpose train_inputs for easier row access
    train_rows = []
    for r in range(len(train)):
        train_rows.append([train_inputs[c][r] for c in range(len(input_indices))])

    predictions = []
    for test_row in test:
        test_vals = []
        for i, idx in enumerate(input_indices):
            val = (float(test_row[idx]) - mins[i]) / maxs[i]
            test_vals.append(val)
        
        # Find nearest neighbors (k=5)
        distances = []
        for i, row in enumerate(train_rows):
            norm_row = [(row[j] - mins[j]) / maxs[j] for j in range(len(input_indices))]
            dist = sum((norm_row[j] - test_vals[j])**2 for j in range(len(input_indices)))
            distances.append((dist, train_targets[i]))
        
        distances.sort()
        # Average of top 5 neighbors
        k = 5
        avg = sum(d[1] for d in distances[:k]) / k
        predictions.append(avg)
        
    return predictions

import csv

def certify(xs, ys):
    """
    Returns True if the distributions of xs and ys are statistically 
    indistinguishable (noise), False if they differ significantly.
    Uses a simple A12 effect size test.
    """
    if not xs or not ys:
        return True
    
    # A12 test: probability that a random value from xs is greater than ys
    greater = 0
    equal = 0
    for x in xs:
        for y in ys:
            if x > y:
                greater += 1
            elif x == y:
                equal += 0.5
    
    a12 = (greater + equal) / (len(xs) * len(ys))
    
    # Threshold for "significant" difference (standard A12 threshold is 0.56)
    # If A12 is close to 0.5, they are likely the same (noise).
    return abs(a12 - 0.5) < 0.06

import csv

def flow(header, row, seen):
    """
    Decides whether to stop at the current row based on a simple 
    thresholding strategy. We define 'good enough' as being in the 
    top 5% of the expected distribution of quality.
    """
    goals = []
    for i, name in enumerate(header):
        if name.endswith('-') or name.endswith('+'):
            goals.append((i, name.endswith('-')))

    def get_score(r):
        # Calculate a normalized score for a row
        # Since we don't have the full distribution yet, we use the 
        # current 'seen' data to estimate the relative quality.
        if not seen:
            return 0.0
        
        score = 0.0
        for idx, is_min in goals:
            val = float(r[idx])
            vals = [float(s[idx]) for s in seen]
            low, high = min(vals), max(vals)
            span = (high - low) if high > low else 1.0
            norm = (val - low) / span
            score += (norm if is_min else (1.0 - norm))
        return score / len(goals)

    if not seen:
        return False

    # If the current row is better than 95% of what we've seen, stop.
    current_score = get_score(row)
    seen_scores = [get_score(s) for s in seen]
    
    # Threshold: if current is better than 95% of seen, accept
    threshold = sorted(seen_scores)[int(len(seen_scores) * 0.95)]
    return current_score <= threshold

import csv

def blame(header, rows):
    """
    Identifies input columns that correlate most strongly with poor performance.
    'Bad' rows are defined as those in the bottom 25% of the quality distribution.
    We compare the distribution of input values in 'bad' rows vs 'good' rows.
    """
    goals = []
    inputs = []
    for i, name in enumerate(header):
        if name.endswith('-') or name.endswith('+'):
            goals.append((i, name.endswith('-')))
        else:
            inputs.append((i, name))

    # Calculate quality score for each row
    def get_score(row):
        score = 0.0
        for idx, is_min in goals:
            val = float(row[idx])
            # We use a simple relative score based on the whole dataset
            # to rank rows from best to worst.
            score += val if is_min else -val
        return score

    scored_rows = sorted([(get_score(r), r) for r in rows])
    
    # Define bad rows as the worst 25%
    n = len(scored_rows)
    bad_rows = [r for _, r in scored_rows[int(0.75 * n):]]
    good_rows = [r for _, r in scored_rows[:int(0.25 * n)]]

    # For each input, calculate the difference in mean between bad and good
    # For numeric, we use normalized difference. For symbolic, we use frequency.
    impacts = []
    for idx, name in inputs:
        try:
            # Numeric
            bad_vals = [float(r[idx]) for r in bad_rows]
            good_vals = [float(r[idx]) for r in good_rows]
            diff = abs((sum(bad_vals) / len(bad_vals)) - (sum(good_vals) / len(good_vals)))
            impacts.append((diff, name))
        except ValueError:
            # Symbolic: compare mode frequency
            bad_freq = {}
            for r in bad_rows: bad_freq[r[idx]] = bad_freq.get(r[idx], 0) + 1
            good_freq = {}
            for r in good_rows: good_freq[r[idx]] = good_freq.get(r[idx], 0) + 1
            
            # Simple divergence: how much more common is the bad mode in bad rows
            bad_mode = max(bad_freq, key=bad_freq.get)
            diff = (bad_freq.get(bad_mode, 0) / len(bad_rows)) - (good_freq.get(bad_mode, 0) / len(good_rows))
            impacts.append((diff, name))

    # Return the top 3 columns that contribute most to the difference
    impacts.sort(key=lambda x: x[0], reverse=True)
    return [name for _, name in impacts[:3]]
