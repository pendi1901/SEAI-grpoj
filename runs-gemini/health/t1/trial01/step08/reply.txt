import csv

def fix(header, row, levers):
    """
    Finds the smallest change to a bad row's settings that improves its
    multi-objective score without increasing the 'N_estimators' (speed).
    """
    goals = []
    inputs = []
    for i, name in enumerate(header):
        if name.endswith('-') or name.endswith('+'):
            goals.append((i, name.endswith('-')))
        elif name in levers:
            inputs.append((i, name))

    def get_score(r):
        score = 0.0
        for idx, is_min in goals:
            val = float(r[idx])
            score += val if is_min else -val
        return score

    best_change = {}
    best_score = get_score(row)
    
    # Try changing one lever at a time
    for idx, name in inputs:
        allowed = levers[name]
        original_val = row[idx]
        
        # Generate candidates
        candidates = []
        if isinstance(allowed, tuple):  # Numeric (low, high)
            low, high = allowed
            # Try a few steps within range
            step = (high - low) / 5
            for i in range(1, 6):
                candidates.append(low + i * step)
        else:  # Symbolic
            candidates = [v for v in allowed if v != original_val]
            
        for val in candidates:
            # Constraint: Do not increase N_estimators (index 3)
            if name == "N_estimators":
                if float(val) > float(original_val):
                    continue
            
            new_row = list(row)
            new_row[idx] = val
            new_score = get_score(new_row)
            
            if new_score < best_score:
                best_score = new_score
                best_change = {name: val}
                
    return best_change

if __name__ == "__main__":
    filename = "Health-ClosedIssues0000.train.csv"
    rows = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            rows.append(row)

    # Define levers based on column types
    levers = {
        "N_estimators": (10, 200),
        "criterion": ["squared_error", "absolute_error", "friedman_mse", "poisson"],
        "Min_sample_leaves": (1, 20),
        "Min_impurity_decrease": (0, 10),
        "Max_depth": (1, 20)
    }

    # Pick a "bad" row (e.g., one with high MRE-)
    # MRE- is index 0
    bad_row = max(rows, key=lambda r: float(r[0]))
    
    recommendation = fix(header, bad_row, levers)
    
    print("Analysis complete.")
    if recommendation:
        print("To improve performance without increasing N_estimators, we recommend:")
        for col, val in recommendation.items():
            print(f"- Change '{col}' to {val}")
    else:
        print("No single-lever change found that improves performance under constraints.")
