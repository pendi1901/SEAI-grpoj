import csv

def spot(header, rows):
    """
    Identifies rows that are 'outliers' in the input space.
    Rows with input values that are statistically extreme (e.g., > 3 standard 
    deviations from the mean) are flagged as potentially failed runs.
    """
    input_indices = [i for i, name in enumerate(header) 
                     if name[0].isupper() and not (name.endswith('-') or name.endswith('+'))]
    
    # Collect numeric input data
    data = []
    for row in rows:
        clean_row = []
        for i in input_indices:
            try:
                clean_row.append(float(row[i]))
            except (ValueError, TypeError):
                clean_row.append(None)
        data.append(clean_row)

    # Calculate mean and std dev for each input column
    stats = []
    for i in range(len(input_indices)):
        vals = [r[i] for r in data if r[i] is not None]
        if not vals:
            stats.append((0, 1))
            continue
        mean = sum(vals) / len(vals)
        var = sum((x - mean) ** 2 for x in vals) / len(vals)
        stats.append((mean, var**0.5 if var > 0 else 1.0))

    # Score rows based on Z-score of their inputs
    # A high Z-score indicates an outlier input value
    suspects = []
    for idx, row in enumerate(data):
        z_scores = []
        for i, val in enumerate(row):
            if val is not None:
                mean, std = stats[i]
                z_scores.append(abs((val - mean) / std))
        
        if z_scores:
            # Use the maximum Z-score found in the row as the suspicion level
            suspects.append((max(z_scores), idx))

    # Sort by suspicion level descending
    suspects.sort(key=lambda x: x[0], reverse=True)
    
    # Return top 1% of rows as most suspicious
    limit = max(1, len(rows) // 100)
    return [idx for score, idx in suspects[:limit]]

if __name__ == "__main__":
    filename = "Health-ClosedIssues0000.train.csv"
    rows = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            rows.append(row)
    
    suspect_indices = spot(header, rows)
    
    print(f"Analysis complete. Identified {len(suspect_indices)} suspicious rows.")
    print("These rows contain input values that deviate significantly from the norm:")
    for idx in suspect_indices[:5]:
        print(f"- Row index {idx}: {rows[idx]}")
