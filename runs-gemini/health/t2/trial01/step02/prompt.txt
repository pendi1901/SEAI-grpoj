import csv

def remember(header, rows):
    """
    Identifies rows that are 'unusually trustworthy' by calculating a normalized
    score for each row based on its goals.
    """
    goals = []
    for i, name in enumerate(header):
        if name.endswith(('-', '+')):
            goals.append((i, name[-1]))

    # Normalize values to [0, 1] to compare different scales
    mins = [float('inf')] * len(header)
    maxs = [float('-inf')] * len(header)
    
    for row in rows:
        for i, val in enumerate(row):
            if isinstance(val, (int, float)):
                if val < mins[i]: mins[i] = val
                if val > maxs[i]: maxs[i] = val

    scores = []
    for idx, row in enumerate(rows):
        score = 0
        for col_idx, direction in goals:
            val = row[col_idx]
            if not isinstance(val, (int, float)):
                continue
            
            # Normalize: range 0 to 1
            rng = maxs[col_idx] - mins[col_idx]
            norm = (val - mins[col_idx]) / rng if rng > 0 else 0
            
            # If goal is '-', lower is better (1 - norm). If '+', higher is better (norm).
            score += (1 - norm) if direction == '-' else norm
        
        scores.append((score, idx))

    # Sort by score descending and return top 5% (or at least 1)
    scores.sort(key=lambda x: x[0], reverse=True)
    limit = max(1, len(rows) // 20)
    return [item[1] for item in scores[:limit]]
