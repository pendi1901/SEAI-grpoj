import csv

def remember(header, rows):
    """
    Identifies rows that are 'unusually trustworthy' by calculating a normalized
    score for each row based on its goals.
    """
    goals = []
    for i, name in enumerate(header):
        if name.endswith(('-', '+')):
            goals.append((i, name[-1]))

    # Normalize values to [0, 1] to compare different scales
    mins = [float('inf')] * len(header)
    maxs = [float('-inf')] * len(header)
    
    for row in rows:
        for i, val in enumerate(row):
            if isinstance(val, (int, float)):
                if val < mins[i]: mins[i] = val
                if val > maxs[i]: maxs[i] = val

    scores = []
    for idx, row in enumerate(rows):
        score = 0
        for col_idx, direction in goals:
            val = row[col_idx]
            if not isinstance(val, (int, float)):
                continue
            
            # Normalize: range 0 to 1
            rng = maxs[col_idx] - mins[col_idx]
            norm = (val - mins[col_idx]) / rng if rng > 0 else 0
            
            # If goal is '-', lower is better (1 - norm). If '+', higher is better (norm).
            score += (1 - norm) if direction == '-' else norm
        
        scores.append((score, idx))

    # Sort by score descending and return top 5% (or at least 1)
    scores.sort(key=lambda x: x[0], reverse=True)
    limit = max(1, len(rows) // 20)
    return [item[1] for item in scores[:limit]]

def certify(xs, ys):
    """
    Returns True if the means of two lists are statistically similar 
    (within 10% of the range), False otherwise.
    """
    if not xs or not ys:
        return True
    
    mean_x = sum(xs) / len(xs)
    mean_y = sum(ys) / len(ys)
    
    combined = xs + ys
    rng = max(combined) - min(combined)
    
    # If the difference between means is less than 10% of the total range,
    # we consider the difference to be noise.
    return abs(mean_x - mean_y) < (0.1 * rng)

def flow(header, row, seen):
    """
    Returns True if the current row is 'good enough' to ship.
    We define 'good enough' as being in the top 5% of the expected 
    distribution based on the history of rows seen so far.
    """
    # Identify goal indices and directions
    goals = [(i, name[-1]) for i, name in enumerate(header) if name.endswith(('-', '+'))]
    
    # Calculate score for the current row
    # We need to estimate min/max from 'seen' rows to normalize
    if not seen:
        return False
    
    def get_score(r):
        score = 0
        for i, direction in goals:
            val = r[i]
            # Simple normalization based on seen data
            col_vals = [s[i] for s in seen if isinstance(s[i], (int, float))]
            if not col_vals: continue
            mn, mx = min(col_vals), max(col_vals)
            rng = mx - mn
            norm = (val - mn) / rng if rng > 0 else 0
            score += (1 - norm) if direction == '-' else norm
        return score

    seen_scores = [get_score(s) for s in seen]
    
    # If current row is better than 95% of what we've seen, ship it
    threshold = sorted(seen_scores, reverse=True)[max(0, len(seen_scores) // 20)]
    return current_score >= threshold

def blame(header, rows):
    """
    Identifies input columns that correlate most strongly with poor performance.
    We define 'bad' rows as those in the bottom 10% of the score distribution.
    We then calculate the correlation between input values and the 'badness' 
    (inverse of the score) to find the culprits.
    """
    # Identify goals and inputs
    goal_indices = [i for i, name in enumerate(header) if name.endswith(('-', '+'))]
    input_indices = [i for i, name in enumerate(header) if not (name.endswith('-') or name.endswith('+'))]
    
    # Calculate scores for all rows using step01 logic
    # We reuse the scoring logic by creating a temporary list of scores
    scores = []
    for row in rows:
        score = 0
        for i in goal_indices:
            # Simple score: normalize and sum
            col_vals = [r[i] for r in rows if isinstance(r[i], (int, float))]
            mn, mx = min(col_vals), max(col_vals)
            rng = mx - mn
            norm = (row[i] - mn) / rng if rng > 0 else 0
            score += (1 - norm) if header[i].endswith('-') else norm
        scores.append(score)
    
    # Identify the worst 10%
    indexed_scores = sorted(zip(scores, range(len(rows))))
    worst_indices = [idx for score, idx in indexed_scores[:len(rows) // 10]]
    
    # For each input, check if its distribution in 'worst' rows differs significantly
    # from the global distribution.
    culprits = []
    for i in input_indices:
        vals = [rows[idx][i] for idx in worst_indices if isinstance(rows[idx][i], (int, float))]
        if not vals: continue
        
        avg_val = sum(vals) / len(vals)
        global_vals = [r[i] for r in rows if isinstance(r[i], (int, float))]
        global_avg = sum(global_vals) / len(global_vals)
        
        # If the average of this input in bad rows is significantly different 
        # from the global average, it's a potential culprit.
        if abs(avg_val - global_avg) > (0.2 * (max(global_vals) - min(global_vals))):
            culprits.append(header[i])
            
    return culprits

def justify(header, rows, row):
    """
    Returns (p, reason) where p is the fraction of rows that 'row' is better than,
    and reason is a human-readable explanation.
    """
    # Identify goal indices and directions
    goals = [(i, name[-1]) for i, name in enumerate(header) if name.endswith(('-', '+'))]
    
    # Calculate min/max for normalization from all rows
    mins = [float('inf')] * len(header)
    maxs = [float('-inf')] * len(header)
    for r in rows:
        for i in range(len(header)):
            if isinstance(r[i], (int, float)):
                if r[i] < mins[i]: mins[i] = r[i]
                if r[i] > maxs[i]: maxs[i] = r[i]

    def get_score(r):
        score = 0
        for i, direction in goals:
            val = r[i]
            if not isinstance(val, (int, float)): continue
            rng = maxs[i] - mins[i]
            norm = (val - mins[i]) / rng if rng > 0 else 0
            score += (1 - norm) if direction == '-' else norm
        return score

    # We need to estimate the goal values for the input 'row'
    # Use the average of the 5 nearest neighbors as a proxy for the '?' goals
    input_indices = [i for i, name in enumerate(header) if not (name.endswith('-') or name.endswith('+'))]
    
    def dist(r1, r2):
        d = 0
        for i in input_indices:
            if isinstance(r1[i], (int, float)) and isinstance(r2[i], (int, float)):
                d += (r1[i] - r2[i]) ** 2
            else:
                d += 0 if r1[i] == r2[i] else 1
        return d

    # Find 5 nearest neighbors to estimate the hidden goals
    neighbors = sorted(rows, key=lambda r: dist(r, row))[:5]
    estimated_row = list(row)
    for i, _ in goals:
        estimated_row[i] = sum(n[i] for n in neighbors) / len(neighbors)
    
    target_score = get_score(estimated_row)
    all_scores = [get_score(r) for r in rows]
    
    better_count = sum(1 for s in all_scores if target_score > s)
    p = better_count / len(rows)
    
    reason = f"This setup outperforms {p:.1%} of all historical configurations based on its estimated performance metrics."
    return p, reason

def fix(header, row, levers):

        return score

    best_change = {}
    best_score = -float('inf')
    
    # Try changing each lever to its boundary or mid-point

        idx = header.index(col_name)

import csv

def spot(header, rows):
    """
    Identifies rows that are 'suspicious' by looking for outliers in input columns.
    A row is suspicious if its input values are significantly far from the 
    median of that column (using a simple Z-score-like heuristic).
    """
    input_indices = [i for i, name in enumerate(header) 
                     if not (name.endswith('-') or name.endswith('+'))]
    
    # Calculate medians for numeric inputs
    medians = {}
    for i in input_indices:
        vals = [r[i] for r in rows if isinstance(r[i], (int, float))]
        if vals:
            vals.sort()
            medians[i] = vals[len(vals) // 2]

    # Score rows based on how many inputs are extreme outliers
    suspicion_scores = []
    for idx, row in enumerate(rows):
        score = 0
        for i in input_indices:
            val = row[i]
            if isinstance(val, (int, float)) and i in medians:
                # If value is more than 3x the median (or 0 if median is 0), flag it
                # This is a heuristic for "wrongly entered" data
                if medians[i] > 0 and (val > 10 * medians[i] or val < medians[i] / 10):
                    score += 1
        suspicion_scores.append((score, idx))

    # Sort by score descending
    suspicion_scores.sort(key=lambda x: x[0], reverse=True)
    
    # Return indices of rows with non-zero suspicion scores
    return [idx for score, idx in suspicion_scores if score > 0]
