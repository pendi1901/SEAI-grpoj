import csv

def remember(header, rows):
    """
    Identifies rows that are 'unusually trustworthy' by calculating a normalized
    score for each row based on its goals.
    """
    goals = []
    for i, name in enumerate(header):
        if name.endswith(('-', '+')):
            goals.append((i, name[-1]))

    # Normalize values to [0, 1] to compare different scales
    mins = [float('inf')] * len(header)
    maxs = [float('-inf')] * len(header)
    
    for row in rows:
        for i, val in enumerate(row):
            if isinstance(val, (int, float)):
                if val < mins[i]: mins[i] = val
                if val > maxs[i]: maxs[i] = val

    scores = []
    for idx, row in enumerate(rows):
        score = 0
        for col_idx, direction in goals:
            val = row[col_idx]
            if not isinstance(val, (int, float)):
                continue
            
            # Normalize: range 0 to 1
            rng = maxs[col_idx] - mins[col_idx]
            norm = (val - mins[col_idx]) / rng if rng > 0 else 0
            
            # If goal is '-', lower is better (1 - norm). If '+', higher is better (norm).
            score += (1 - norm) if direction == '-' else norm
        
        scores.append((score, idx))

    # Sort by score descending and return top 5% (or at least 1)
    scores.sort(key=lambda x: x[0], reverse=True)
    limit = max(1, len(rows) // 20)
    return [item[1] for item in scores[:limit]]

def guess(header, train, test):
    """
    Estimates the value of the first goal column for each test row by 
    finding the nearest neighbor in the training set based on input columns.
    """
    # Identify input columns (those not ending in - or +)
    input_indices = [i for i, name in enumerate(header) if not (name.endswith('-') or name.endswith('+'))]
    
    # Identify the first goal column index
    goal_idx = -1
    for i, name in enumerate(header):
        if name.endswith(('-', '+')):
            goal_idx = i
            break
            
    predictions = []
    for t_row in test:
        best_dist = float('inf')
        best_val = 0.0
        
        # Simple Euclidean distance on input columns
        for train_row in train:
            dist = 0
            for i in input_indices:
                val_t = t_row[i]
                val_tr = train_row[i]
                
                if isinstance(val_t, (int, float)) and isinstance(val_tr, (int, float)):
                    dist += (val_t - val_tr) ** 2
                else:
                    dist += 0 if val_t == val_tr else 1
            
            if dist < best_dist:
                best_dist = dist
                best_val = train_row[goal_idx]
        
        predictions.append(float(best_val))
        
    return predictions

import csv
import step01

def certify(xs, ys):
    """
    Returns True if the means of two lists are statistically similar 
    (within 10% of the range), False otherwise.
    """
    if not xs or not ys:
        return True
    
    mean_x = sum(xs) / len(xs)
    mean_y = sum(ys) / len(ys)
    
    combined = xs + ys
    rng = max(combined) - min(combined)
    
    # If the difference between means is less than 10% of the total range,
    # we consider the difference to be noise.
    return abs(mean_x - mean_y) < (0.1 * rng)

import csv
import step01

def flow(header, row, seen):
    """
    Returns True if the current row is 'good enough' to ship.
    We define 'good enough' as being in the top 5% of the expected 
    distribution based on the history of rows seen so far.
    """
    # Identify goal indices and directions
    goals = [(i, name[-1]) for i, name in enumerate(header) if name.endswith(('-', '+'))]
    
    # Calculate score for the current row
    # We need to estimate min/max from 'seen' rows to normalize
    if not seen:
        return False
    
    def get_score(r):
        score = 0
        for i, direction in goals:
            val = r[i]
            # Simple normalization based on seen data
            col_vals = [s[i] for s in seen if isinstance(s[i], (int, float))]
            if not col_vals: continue
            mn, mx = min(col_vals), max(col_vals)
            rng = mx - mn
            norm = (val - mn) / rng if rng > 0 else 0
            score += (1 - norm) if direction == '-' else norm
        return score

    current_score = get_score(row)
    seen_scores = [get_score(s) for s in seen]
    
    # If current row is better than 95% of what we've seen, ship it
    threshold = sorted(seen_scores, reverse=True)[max(0, len(seen_scores) // 20)]
    return current_score >= threshold
