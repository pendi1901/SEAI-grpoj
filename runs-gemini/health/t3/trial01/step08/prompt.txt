import csv
import pandas as pd
import numpy as np

def remember(header, rows):
    """
    Identifies the best rows based on multi-objective optimization.
    Goals ending in '-' are minimized, '+' are maximized.
    We normalize numeric columns to [0, 1] and calculate a score.
    """
    df = pd.DataFrame(rows, columns=header)
    
    # Identify goal columns
    min_cols = [c for c in header if c.endswith('-')]
    max_cols = [c for c in header if c.endswith('+')]
    
    # Convert numeric columns to float, handle potential errors
    for col in min_cols + max_cols:
        df[col] = pd.to_numeric(df[col], errors='coerce')
    
    # Normalize goals: (val - min) / (max - min)
    # For '-', we want low values, so we invert: (max - val) / (max - min)
    scores = pd.DataFrame(index=df.index)
    for col in min_cols:
        c_min, c_max = df[col].min(), df[col].max()
        scores[col] = (c_max - df[col]) / (c_max - c_min + 1e-9)
    for col in max_cols:
        c_min, c_max = df[col].min(), df[col].max()
        scores[col] = (df[col] - c_min) / (c_max - c_min + 1e-9)
        
    # Aggregate score (average of normalized goals)
    df['score'] = scores.mean(axis=1)
    
    # Return top 5% (or at least 1)
    n_top = max(1, int(len(df) * 0.05))
    top_indices = df['score'].nlargest(n_top).index.tolist()
    
    return sorted(top_indices)

def guess(header, train, test):
    """
    Predicts the first goal column for test rows using a Random Forest Regressor.
    """
    df_train = pd.DataFrame(train, columns=header)
    df_test = pd.DataFrame(test, columns=header)
    
    # Identify the target (first goal column)
    target_col = header[0]
    
    # Identify input columns (all columns except goals)
    # Goals end in '-' or '+'
    input_cols = [c for c in header if not (c.endswith('-') or c.endswith('+'))]
    
    # Prepare data
    X_train = df_train[input_cols].copy()
    y_train = df_train[target_col].astype(float)
    X_test = df_test[input_cols].copy()
    
    # Encode symbolic columns
    for col in input_cols:
        if X_train[col].dtype == 'object':
            le = LabelEncoder()
            # Combine to ensure all categories are seen
            combined = pd.concat([X_train[col], X_test[col]], axis=0).astype(str)
            le.fit(combined)
            X_train[col] = le.transform(X_train[col].astype(str))
            X_test[col] = le.transform(X_test[col].astype(str))
            
    # Train model
    model = RandomForestRegressor(n_estimators=100, random_state=42)
    model.fit(X_train, y_train)
    
    # Predict
    predictions = model.predict(X_test)
    return predictions.tolist()

import csv
import pandas as pd
import numpy as np
from scipy import stats

def certify(xs, ys):
    """
    Returns True if the difference between two distributions is likely noise,
    False if they are statistically significantly different.
    Uses Welch's t-test.
    """
    if len(xs) < 2 or len(ys) < 2:
        return True
    
    t_stat, p_value = stats.ttest_ind(xs, ys, equal_var=False)
    # If p_value > 0.05, we fail to reject the null hypothesis (they are the same)
    return p_value > 0.05

def analyze_setup_size(df):
    """
    Determines if 'big' setups (high N_estimators) perform better than 'small' ones.
    """
    # Identify goal columns
    min_cols = [c for c in df.columns if c.endswith('-')]
    max_cols = [c for c in df.columns if c.endswith('+')]
    
    # Normalize goals to a single performance score [0, 1]
    scores = pd.DataFrame(index=df.index)
    for col in min_cols:
        c_min, c_max = df[col].min(), df[col].max()
        scores[col] = (c_max - df[col]) / (c_max - c_min + 1e-9)
    for col in max_cols:
        c_min, c_max = df[col].min(), df[col].max()
        scores[col] = (df[col] - c_min) / (c_max - c_min + 1e-9)
    
    df['score'] = scores.mean(axis=1)
    
    # Split by N_estimators (median split)
    median_n = df['N_estimators'].median()
    small = df[df['N_estimators'] <= median_n]['score']
    big = df[df['N_estimators'] > median_n]['score']
    
    is_same = certify(small.tolist(), big.tolist())
    
    return is_same, small.mean(), big.mean()

import csv

def flow(header, row, seen):
    """
    Decides whether to stop at the current row based on a simple threshold.
    We define 'good enough' as being in the top 5% of the potential range 
    of the aggregate score based on historical data or current observations.
    """
    # Identify goal columns
    min_indices = [i for i, h in enumerate(header) if h.endswith('-')]
    max_indices = [i for i, h in enumerate(header) if h.endswith('+')]
    
    def get_score(r):
        # Calculate normalized score for a row
        # Since we don't have global min/max, we use the current row's values
        # relative to the 'seen' data distribution.
        if not seen:
            return 0.5
        
        scores = []
        for i in min_indices:
            vals = [s[i] for s in seen if isinstance(s[i], (int, float))]
            if not vals: continue
            c_min, c_max = min(vals), max(vals)
            val = r[i] if isinstance(r[i], (int, float)) else c_max
            scores.append((c_max - val) / (c_max - c_min + 1e-9) if c_max > c_min else 0)
            
        for i in max_indices:
            vals = [s[i] for s in seen if isinstance(s[i], (int, float))]
            if not vals: continue
            c_min, c_max = min(vals), max(vals)
            val = r[i] if isinstance(r[i], (int, float)) else c_min
            scores.append((val - c_min) / (c_max - c_min + 1e-9) if c_max > c_min else 0)
            
        return sum(scores) / len(scores) if scores else 0.5

    # If we have seen enough, stop if the current row is in the top 5%
    if len(seen) > 50:
        current_score = get_score(row)
        seen_scores = [get_score(s) for s in seen]
        threshold = sorted(seen_scores, reverse=True)[int(len(seen_scores) * 0.05)]
        if current_score >= threshold:
            return True
            
    return False

def blame(header, rows):
    """
    Identifies input columns that contribute most to poor performance.
    We define 'bad' rows as those in the bottom 25% of the score distribution.
    We then use feature importance from a Random Forest to see which inputs
    are most predictive of these low scores.
    """
    df = pd.DataFrame(rows, columns=header)
    
    # Identify goal columns
    min_cols = [c for c in header if c.endswith('-')]
    max_cols = [c for c in header if c.endswith('+')]
    input_cols = [c for c in header if not (c.endswith('-') or c.endswith('+'))]
    
    # Normalize goals to a single performance score [0, 1]
    scores = pd.DataFrame(index=df.index)
    for col in min_cols:
        c_min, c_max = df[col].min(), df[col].max()
        scores[col] = (c_max - df[col]) / (c_max - c_min + 1e-9)
    for col in max_cols:
        c_min, c_max = df[col].min(), df[col].max()
        scores[col] = (df[col] - c_min) / (c_max - c_min + 1e-9)
    
    df['score'] = scores.mean(axis=1)
    
    # Prepare features
    X = df[input_cols].copy()
    for col in input_cols:
        if X[col].dtype == 'object':
            le = LabelEncoder()
            X[col] = le.fit_transform(X[col].astype(str))
    
    # Train a model to predict the score
    # High importance features are those that drive the score (good or bad)
    rf = RandomForestRegressor(n_estimators=50, random_state=42)
    rf.fit(X, df['score'])
    
    # Get feature importances
    importances = rf.feature_importances_
    indices = np.argsort(importances)[::-1]
    
    # Return the top 3 most influential columns
    return [input_cols[i] for i in indices[:3]]

import csv

def justify(header, rows, row):
    """
    Returns (p, reason) where p is the fraction of rows that 'row' is better than,
    and reason is a human-readable explanation.
    """
    df = pd.DataFrame(rows, columns=header)
    min_cols = [c for c in header if c.endswith('-')]
    max_cols = [c for c in header if c.endswith('+')]
    
    # Calculate scores for all rows
    scores = pd.DataFrame(index=df.index)
    for col in min_cols:
        c_min, c_max = df[col].min(), df[col].max()
        scores[col] = (c_max - df[col]) / (c_max - c_min + 1e-9)
    for col in max_cols:
        c_min, c_max = df[col].min(), df[col].max()
        scores[col] = (df[col] - c_min) / (c_max - c_min + 1e-9)
    
    df['score'] = scores.mean(axis=1)
    
    # To justify the specific 'row', we need its values. 
    # Since the input 'row' has '?' for goals, we assume the user wants to 
    # evaluate the performance of the setup defined by the input columns.
    # We find the row in the dataset that matches the input columns of the provided 'row'.
    input_cols = [c for c in header if not (c.endswith('-') or c.endswith('+'))]
    
    # Match row by input columns
    match = df
    for col in input_cols:
        val = row[header.index(col)]
        match = match[match[col] == val]
    
    if match.empty:
        return (0.0, "The provided setup configuration could not be found in the historical data.")
    
    target_score = match['score'].mean()
    better_than_count = (df['score'] < target_score).sum()
    p = better_than_count / len(df)
    
    reason = f"This setup achieves a performance score better than {p:.1%} of all historical configurations."
    return (p, reason)

import csv
import pandas as pd
import numpy as np
from sklearn.ensemble import RandomForestRegressor
from sklearn.preprocessing import LabelEncoder

def choose(header, rows, label):
    """
    Finds the best setup by sampling 50 rows, using a Random Forest to 
    predict performance, and selecting the best predicted configuration.
    """
    n_rows = len(rows)
    input_cols = [c for c in header if not (c.endswith('-') or c.endswith('+'))]
    goal_cols = [c for c in header if c.endswith('-') or c.endswith('+')]
    
    # 1. Sample 50 indices to label
    # We pick a mix of random and spread out indices
    indices_to_label = np.linspace(0, n_rows - 1, 50, dtype=int)
    
    train_X = []
    train_y = []
    
    # 2. Label the sampled rows
    for i in indices_to_label:
        goals = label(i)
        # Normalize goals to a single score [0, 1]
        # We need to estimate global min/max for normalization
        # Since we don't have them, we use the sampled goals as a proxy
        train_X.append([rows[i][header.index(c)] for c in input_cols])
        train_y.append(goals)
        
    train_y = np.array(train_y)
    
    # Normalize goals for the score
    norm_y = np.zeros_like(train_y, dtype=float)
    for j, col in enumerate(goal_cols):
        c_min, c_max = train_y[:, j].min(), train_y[:, j].max()
        if c_max == c_min:
            norm_y[:, j] = 0.5
        elif col.endswith('-'):
            norm_y[:, j] = (c_max - train_y[:, j]) / (c_max - c_min)
        else:
            norm_y[:, j] = (train_y[:, j] - c_min) / (c_max - c_min)
    
    scores = norm_y.mean(axis=1)
    
    # 3. Train a model to predict the score
    df_X = pd.DataFrame(train_X, columns=input_cols)
    for col in input_cols:
        if df_X[col].dtype == 'object':
            le = LabelEncoder()
            df_X[col] = le.fit_transform(df_X[col].astype(str))
            
    model = RandomForestRegressor(n_estimators=50, random_state=42)
    model.fit(df_X, scores)
    
    # 4. Predict scores for all rows
    all_X = pd.DataFrame(rows, columns=header)[input_cols]
    for col in input_cols:
        if all_X[col].dtype == 'object':
            le = LabelEncoder()
            all_X[col] = le.fit_transform(all_X[col].astype(str))
            
    all_scores = model.predict(all_X)
    
    # 5. Return index of the best predicted row
    return int(np.argmax(all_scores))
