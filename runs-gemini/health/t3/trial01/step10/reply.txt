import csv
import pandas as pd
import numpy as np
from sklearn.ensemble import RandomForestRegressor
from sklearn.preprocessing import LabelEncoder

def get_score(row, header, min_cols, max_cols, min_vals, max_vals):
    scores = []
    for col in min_cols:
        idx = header.index(col)
        val = row[idx]
        c_min, c_max = min_vals[col], max_vals[col]
        scores.append((c_max - val) / (c_max - c_min + 1e-9))
    for col in max_cols:
        idx = header.index(col)
        val = row[idx]
        c_min, c_max = min_vals[col], max_vals[col]
        scores.append((val - c_min) / (c_max - c_min + 1e-9))
    return sum(scores) / len(scores)

def race(header, rows, label):
    input_cols = [c for c in header if not (c.endswith('-') or c.endswith('+'))]
    min_cols = [c for c in header if c.endswith('-')]
    max_cols = [c for c in header if c.endswith('+')]
    
    # Pre-calculate global min/max for normalization using a small sample or full data
    # Since we only have 50 labels, we use the first 50 as a proxy for normalization
    sampled_data = [label(i) for i in range(50)]
    df_goals = pd.DataFrame(sampled_data, columns=min_cols + max_cols)
    min_vals = df_goals.min().to_dict()
    max_vals = df_goals.max().to_dict()

    def evaluate(strategy_name, indices):
        best_score = -1.0
        best_row_idx = -1
        for i in indices:
            goals = label(i)
            row = rows[i][:]
            for j, h in enumerate(header):
                if h.endswith('-') or h.endswith('+'):
                    row[j] = goals[min_cols.index(h) if h in min_cols else max_cols.index(h) - len(min_cols)]
            score = get_score(row, header, min_cols, max_cols, min_vals, max_vals)
            if score > best_score:
                best_score = score
                best_row_idx = i
        return best_score, best_row_idx

    # Strategy 1: Random
    idx_random = np.random.choice(len(rows), 50, replace=False)
    score_rand, row_rand = evaluate("random", idx_random)

    # Strategy 2: First 50 (Baseline)
    idx_first = list(range(50))
    score_first, row_first = evaluate("first", idx_first)

    # Strategy 3: Every 100th (Spread)
    idx_spread = [i for i in range(0, len(rows), len(rows)//50)][:50]
    score_spread, row_spread = evaluate("spread", idx_spread)

    results = {
        "random": score_rand,
        "first": score_first,
        "spread": score_spread
    }
    
    best_name = max(results, key=results.get)
    best_row = row_rand if best_name == "random" else (row_first if best_name == "first" else row_spread)
    
    return {"winner": best_name, "row": best_row, "tried": ["random", "first", "spread"]}

if __name__ == "__main__":
    filename = 'Health-ClosedIssues0000.train.csv'
    data = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = [float(val) if val.replace('.','',1).isdigit() or (val.startswith('-') and val[1:].replace('.','',1).isdigit()) else val for val in row]
            data.append(processed_row)
            
    true_data = [row[:] for row in data]
    masked_data = [[("?" if (header[i].endswith('-') or header[i].endswith('+')) else row[i]) for i in range(len(row))] for row in data]
    
    def label(i):
        return [true_data[i][j] for j, h in enumerate(header) if h.endswith('-') or h.endswith('+')]
        
    result = race(header, masked_data, label)
    print(f"The winning strategy is '{result['winner']}', which found the best setup at row index {result['row']}.")
    print(f"Strategies compared: {', '.join(result['tried'])}.")
