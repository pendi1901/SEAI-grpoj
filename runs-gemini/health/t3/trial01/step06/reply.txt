```python
import csv
import pandas as pd

def justify(header, rows, row):
    """
    Returns (p, reason) where p is the fraction of rows that 'row' is better than,
    and reason is a human-readable explanation.
    """
    df = pd.DataFrame(rows, columns=header)
    min_cols = [c for c in header if c.endswith('-')]
    max_cols = [c for c in header if c.endswith('+')]
    
    # Calculate scores for all rows
    scores = pd.DataFrame(index=df.index)
    for col in min_cols:
        c_min, c_max = df[col].min(), df[col].max()
        scores[col] = (c_max - df[col]) / (c_max - c_min + 1e-9)
    for col in max_cols:
        c_min, c_max = df[col].min(), df[col].max()
        scores[col] = (df[col] - c_min) / (c_max - c_min + 1e-9)
    
    df['score'] = scores.mean(axis=1)
    
    # To justify the specific 'row', we need its values. 
    # Since the input 'row' has '?' for goals, we assume the user wants to 
    # evaluate the performance of the setup defined by the input columns.
    # We find the row in the dataset that matches the input columns of the provided 'row'.
    input_cols = [c for c in header if not (c.endswith('-') or c.endswith('+'))]
    
    # Match row by input columns
    match = df
    for col in input_cols:
        val = row[header.index(col)]
        match = match[match[col] == val]
    
    if match.empty:
        return (0.0, "The provided setup configuration could not be found in the historical data.")
    
    target_score = match['score'].mean()
    better_than_count = (df['score'] < target_score).sum()
    p = better_than_count / len(df)
    
    reason = f"This setup achieves a performance score better than {p:.1%} of all historical configurations."
    return (p, reason)

if __name__ == "__main__":
    filename = 'Health-ClosedIssues0000.train.csv'
    data = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = []
            for val in row:
                try:
                    processed_row.append(float(val))
                except ValueError:
                    processed_row.append(val)
            data.append(processed_row)
    
    # Example: Justify the first row in the dataset
    test_row = data[0][:]
    for i, col in enumerate(header):
        if col.endswith('-') or col.endswith('+'):
            test_row[i] = "?"
            
    p, reason = justify(header, data, test_row)
    print(f"Performance percentile: {p:.4f}")
    print(f"Justification: {reason}")
```
