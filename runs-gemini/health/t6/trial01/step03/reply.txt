import ezr
import step01

def certify(xs, ys):
    """
    Returns True if xs and ys are statistically indistinguishable,
    False if they differ significantly.
    """
    return ezr.same(xs, ys)

if __name__ == "__main__":
    # Load data
    filename = "Health-ClosedIssues0000.train.csv"
    data = list(ezr.csv(filename))
    header = data[0]
    rows = data[1:]
    
    # Identify the "big" vs "small" setups.
    # Based on the CSV, 'N_estimators' is a numeric column (starts with upper case).
    # We define "big" as having N_estimators > median, "small" otherwise.
    n_est_idx = header.index("N_estimators")
    n_ests = [r[n_est_idx] for r in rows]
    median_n = sorted(n_ests)[len(n_ests) // 2]
    
    # Create a table to calculate ydist (trustworthiness)
    tbl = ezr.Tbl(data)
    
    big_setups = []
    small_setups = []
    
    for row in rows:
        yd = ezr.ydist(tbl, row)
        if row[n_est_idx] > median_n:
            big_setups.append(yd)
        else:
            small_setups.append(yd)
            
    # Certify if the difference in accuracy (ydist) is real
    is_same = certify(big_setups, small_setups)
    
    print(f"Analysis of {len(rows)} records.")
    print(f"Big setups (N_estimators > {median_n}): {len(big_setups)} records.")
    print(f"Small setups (N_estimators <= {median_n}): {len(small_setups)} records.")
    
    if is_same:
        print("\nConclusion: The difference in accuracy between big and small setups is likely noise.")
    else:
        print("\nConclusion: There is a statistically significant difference in accuracy between big and small setups.")
