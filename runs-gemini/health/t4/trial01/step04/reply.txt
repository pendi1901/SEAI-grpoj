```python
import csv
import pandas as pd
import numpy as np
from step01 import remember

def flow(header, row, seen):
    """
    Decides whether to accept the current row based on whether it is 
    better than all previously seen rows.
    """
    if not seen:
        return True
    
    # Combine current row with seen rows to normalize and compare
    all_rows = seen + [row]
    df = pd.DataFrame(all_rows, columns=header)
    
    # Identify goal columns
    min_cols = [c for c in header if c.endswith('-')]
    max_cols = [c for c in header if c.endswith('+')]
    
    # Normalize goals
    scores = pd.DataFrame(index=df.index)
    for col in min_cols:
        col_min, col_max = df[col].min(), df[col].max()
        scores[col] = (col_max - df[col]) / (col_max - col_min + 1e-9)
    for col in max_cols:
        col_min, col_max = df[col].min(), df[col].max()
        scores[col] = (df[col] - col_min) / (col_max - col_min + 1e-9)
        
    df['total_score'] = scores.sum(axis=1)
    
    # If the current row (last index) is the best seen so far, accept it
    return df['total_score'].idxmax() == len(df) - 1

if __name__ == "__main__":
    filename = "Health-ClosedIssues0000.train.csv"
    data = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = [float(val) if val.replace('.', '', 1).isdigit() or 
                             (val.startswith('-') and val[1:].replace('.', '', 1).isdigit()) 
                             else val for val in row]
            data.append(processed_row)
    
    seen = []
    accepted = None
    for i, row in enumerate(data):
        if flow(header, row, seen):
            accepted = (i, row)
            break
        seen.append(row)
    
    if accepted:
        print(f"Stopped at row {accepted[0]}: Found a setup good enough to ship.")
        print(f"Row data: {accepted[1]}")
    else:
        print("No suitable setup found.")
```
