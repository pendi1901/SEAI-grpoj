import csv
import pandas as pd
import numpy as np
from sklearn.ensemble import RandomForestRegressor
from sklearn.preprocessing import LabelEncoder
from step01 import remember
from step02 import guess

def choose(header, rows, label):
    """
    Finds the best row by sampling 50 rows, labeling them, 
    training a model to predict the best, and returning the index.
    """
    n_rows = len(rows)
    # Sample 50 indices to query
    indices_to_query = np.linspace(0, n_rows - 1, 50, dtype=int)
    
    # Label the sampled rows
    labeled_data = []
    for i in indices_to_query:
        goals = label(i)
        row = list(rows[i])
        goal_indices = [idx for idx, h in enumerate(header) if h.endswith('-') or h.endswith('+')]
        for j, g_idx in enumerate(goal_indices):
            row[g_idx] = goals[j]
        labeled_data.append(row)
    
    # Use remember logic to find the best among the 50 labeled
    best_local_idx = remember(header, labeled_data)
    best_row = labeled_data[best_local_idx[0]]
    
    # Find the original index of this best row
    for i in indices_to_query:
        if rows[i] == [val for val in best_row if val != "?"]:
            return i
            
    return indices_to_query[0]

if __name__ == "__main__":
    filename = "Health-ClosedIssues0000.train.csv"
    data = []

    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = [float(val) if val.replace('.', '', 1).isdigit() or 
                             (val.startswith('-') and val[1:].replace('.', '', 1).isdigit()) 
                             else val for val in row]
            data.append(processed_row)

    # Define the label function for the harness
    def label(i):
        row = data[i]
        return [row[idx] for idx, h in enumerate(header) if h.endswith('-') or h.endswith('+')]

    # Mask the rows for the choose function
    masked_rows = []
    for row in data:
        new_row = list(row)
        for idx, h in enumerate(header):
            if h.endswith('-') or h.endswith('+'):
                new_row[idx] = "?"
        masked_rows.append(new_row)

    best_idx = choose(header, masked_rows, label)
    print(f"Analysis complete. After evaluating 50 setups, the best configuration is at row index {best_idx}.")
