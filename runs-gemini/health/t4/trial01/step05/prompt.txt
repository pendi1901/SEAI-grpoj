import csv
import pandas as pd
import numpy as np

def remember(header, rows):
    """
    Identifies the best rows based on multi-objective optimization.
    Goals ending in '-' are minimized, '+' are maximized.
    We normalize numeric columns and use a simple weighted sum (or Pareto-like approach)
    to find the most 'trustworthy' setups.
    """
    df = pd.DataFrame(rows, columns=header)
    
    # Identify goal columns
    min_cols = [c for c in header if c.endswith('-')]
    max_cols = [c for c in header if c.endswith('+')]
    
    # Convert numeric columns to float, handle potential errors
    for col in min_cols + max_cols:
        df[col] = pd.to_numeric(df[col], errors='coerce')
    
    # Normalize goals to [0, 1] range
    # For min_cols: lower is better (1 - normalized)
    # For max_cols: higher is better (normalized)
    scores = pd.DataFrame(index=df.index)
    
    for col in min_cols:
        col_min, col_max = df[col].min(), df[col].max()
        scores[col] = (col_max - df[col]) / (col_max - col_min + 1e-9)
        
    for col in max_cols:
        col_min, col_max = df[col].min(), df[col].max()
        scores[col] = (df[col] - col_min) / (col_max - col_min + 1e-9)
        
    # Aggregate scores
    df['total_score'] = scores.sum(axis=1)
    
    # Select top 5% (or at least 1)
    n_select = max(1, int(len(df) * 0.05))
    top_indices = df['total_score'].nlargest(n_select).index.tolist()
    
    return top_indices

def certify(xs, ys):
    """
    Returns True if the difference between two distributions is likely noise,
    False if they are statistically significantly different.
    Uses Welch's t-test.
    """
    t_stat, p_value = stats.ttest_ind(xs, ys, equal_var=False)
    # If p_value > 0.05, we fail to reject the null hypothesis (it's likely noise)
    return p_value > 0.05

import csv
import pandas as pd
import numpy as np
from step01 import remember

def flow(header, row, seen):
    """
    Decides whether to accept the current row based on whether it is 
    better than all previously seen rows.
    """
    if not seen:
        return True
    
    # Combine current row with seen rows to normalize and compare
    all_rows = seen + [row]
    df = pd.DataFrame(all_rows, columns=header)
    
    # Identify goal columns
    min_cols = [c for c in header if c.endswith('-')]
    max_cols = [c for c in header if c.endswith('+')]
    
    # Normalize goals
    scores = pd.DataFrame(index=df.index)
    for col in min_cols:
        col_min, col_max = df[col].min(), df[col].max()
        scores[col] = (col_max - df[col]) / (col_max - col_min + 1e-9)
    for col in max_cols:
        col_min, col_max = df[col].min(), df[col].max()
        scores[col] = (df[col] - col_min) / (col_max - col_min + 1e-9)
        
    df['total_score'] = scores.sum(axis=1)
    
    # If the current row (last index) is the best seen so far, accept it
    return df['total_score'].idxmax() == len(df) - 1
