def remember(header, rows):
    goal_cols = ['Cost-', 'Score-', 'Idle-']

    n = len(rows)
    if n == 0:
        return []

    # Normalise each goal column to [0,1] where 0 is best (lowest value).
    min_vals = {}
    max_vals = {}
    for col in goal_cols:
        vals = [float(row[header.index(col)]) for row in rows]
        mn, mx = min(vals), max(vals)
        if mx == mn:
            min_vals[col], max_vals[col] = 0.0, 1.0
        else:
            min_vals[col], max_vals[col] = 0.0, 1.0

    # Combined normalised score per row (lower is better).
    scores = []
    for row in rows:
        s = sum(
            (float(row[header.index(col)]) - min_vals[col]) / (max_vals[col] - min_vals[col])
            for col in goal_cols
        )
        scores.append(s)

    # Return positions of the best ~5% rows, at least one.
    k = max(1, int(n * 0.05))
    ranked = sorted(range(n), key=lambda i: scores[i])
    return list(ranked[:k])

def guess(header, train, test):
    goal_cols = ['Cost-', 'Score-', 'Idle-']

    def is_goal(col):
        return col.endswith('-') and col != '-'

    n_train = len(train)
    if n_train == 0:
        return [float('nan')] * len(test)

    # Stats for non-goal columns from training data.
    col_means = {}
    col_stds = {}
    for col in header:
        if not is_goal(col):
            vals = [float(row[header.index(col)]) for row in train]
            m = sum(vals) / len(vals)
            var = sum((v - m) ** 2 for v in vals) / len(vals)
            s = var ** 0.5
            col_means[col] = m
            col_stds[col] = s

    # Normalize Cost- from training data so 0 = best (lowest).
    cost_vals_train = [float(row[header.index('Cost-')]) for row in train]
    mn_cost, mx_cost = min(cost_vals_train), max(cost_vals_train)

    def norm_cost(v):
        if mx_cost == mn_cost:
            return 0.0
        return (v - mn_cost) / (mx_cost - mn_cost)

    cost_norm_train = [norm_cost(v) for v in cost_vals_train]

    # Correlations between non-goal columns and normalized Cost-.
    col_corrs = {}
    for col in header:
        if not is_goal(col):
            vals = [float(row[header.index(col)]) for row in train]
            n = len(vals)
            mx_v = sum(vals) / n
            mx_c = sum(cost_norm_train) / n
            num = sum((vals[i] - mx_v) * (cost_norm_train[i] - mx_c) for i in range(n))
            den_x = sum((v - mx_v) ** 2 for v in vals) ** 0.5
            den_y = sum((c - mx_c) ** 2 for c in cost_norm_train) ** 0.5
            col_corrs[col] = num / (den_x * den_y) if den_x != 0 and den_y != 0 else 0.0

    # Predict Cost- for each test row using the learned correlations.
    def predict(test_row):
        pred_norm = sum(
            col_corrs.get(col, 0.0) * (
                (float(test_row[header.index(col)]) - col_means[col]) / col_stds[col] if col_stds[col] != 0 else 0.0
            )
            for col in header if not is_goal(col)
        )
        return max(0.0, min(1.0, pred_norm))

    results = []
    for test_row in test:
        norm_pred = predict(test_row)
        cost_pred = mn_cost + norm_pred * (mx_cost - mn_cost) if mx_cost != mn_cost else 0.0
        results.append(cost_pred)

    return results

def certify(xs, ys):
    """Return True if xs and ys agree apart from noise, False if they really differ."""
    mx = max(len(xs), len(ys))
    if mx == 0:
        return True
    mean_x = sum(xs) / len(xs)
    mean_y = sum(ys) / len(ys)
    diff = abs(mean_x - mean_y)
    # Use the larger sample's std as a noise proxy.
    n1, n2 = len(xs), len(ys)
    if n1 > 0:
        var_x = sum((v - mean_x) ** 2 for v in xs) / (n1 - 1) if n1 > 1 else 0.0
        s_x = var_x ** 0.5
    else:
        s_x = 0.0
    if n2 > 0:
        var_y = sum((v - mean_y) ** 2 for v in ys) / (n2 - 1) if n2 > 1 else 0.0
        s_y = var_y ** 0.5
    else:
        s_y = 0.0
    noise = ((s_x ** 2 + s_y ** 2) ** 0.5) * (n1 ** -0.5 + n2 ** -0.5)
    return diff < 3.0 * noise

def blame(header, rows):
    goal_cols = ['Cost-', 'Score-', 'Idle-']

    def is_goal(col):
        return col.endswith('-') and col != '-'

    n = len(rows)
    if n == 0:
        return []

    # Normalise each goal column to [0,1] where 0 is best (lowest value).
    min_vals = {}
    max_vals = {}
    for col in goal_cols:
        vals = [float(row[header.index(col)]) for row in rows]
        mn, mx = min(vals), max(vals)
        if mx == mn:
            min_vals[col], max_vals[col] = 0.0, 1.0
        else:
            min_vals[col], max_vals[col] = mn, mx

    # Combined normalised badness per row (higher is worse).
    badnesses = []
    for row in rows:
        b = sum(
            (float(row[header.index(col)]) - min_vals[col]) / (max_vals[col] - min_vals[col])
            for col in goal_cols
        )
        badnesses.append(b)

    # For each non-goal column, compute Pearson correlation with overall badness.
    col_corrs = {}
    mean_badness = sum(badnesses) / n
    var_badness = sum((b - mean_badness) ** 2 for b in badnesses) ** 0.5

    for col in header:
        if not is_goal(col):
            vals = [float(row[header.index(col)]) for row in rows]
            mx_v = sum(vals) / n
            var_x = sum((v - mx_v) ** 2 for v in vals) ** 0.5
            col_corrs[col] = (sum(
                (vals[i] - mx_v) * (badnesses[i] - mean_badness) for i in range(n)
            ) / (var_x * var_badness)) if var_x != 0 and var_badness != 0 else 0.0

    # Rank by absolute correlation — strongest predictors of bad outcomes first.
    ranked = sorted(col_corrs.items(), key=lambda x: -abs(x[1]))
    return [col for col, _ in ranked[:5]]

def justify(header, rows, row):
    goal_cols = ['Cost-', 'Score-', 'Idle-']

    def is_goal(col):
        return col.endswith('-') and col != '-'

    n_rows = len(rows)
    if n_rows == 0:
        return (0.0, "No reference data available.")

    # Stats for non-goal columns from training rows.
    col_means = {}
    col_stds = {}
    for col in header:
        if not is_goal(col):
            vals = [float(r[header.index(col)]) for r in rows]
            m = sum(vals) / len(vals)
            var = sum((v - m) ** 2 for v in vals) / max(len(vals) - 1, 1)
            col_means[col] = m
            col_stds[col] = var ** 0.5

    # Correlations between non-goal columns and each goal column.
    goal_corrs = {}
    for gcol in goal_cols:
        if not is_goal(gcol):
            continue
        gvals = [float(r[header.index(gcol)]) for r in rows]
        n_g = len(gvals)
        mx_g = sum(gvals) / n_g
        var_g = sum((v - mx_g) ** 2 for v in gvals) ** 0.5

        goal_corrs[gcol] = {}
        for col in header:
            if not is_goal(col):
                cvals = [float(r[header.index(col)]) for r in rows]
                n_c = len(cvals)
                mx_c = sum(cvals) / n_c
                var_c = sum((v - mx_c) ** 2 for v in cvals) ** 0.5
                goal_corrs[gcol][col] = (sum(
                    (cvals[i] - mx_c) * (gvals[i] - mx_g) for i in range(n_g)
                ) / (var_c * var_g)) if var_c != 0 and var_g != 0 else 0.0

    # Predict goal values for the test row using correlations.
    def predict_goal(gcol):
        pred = sum(
            goal_corrs[gcol].get(col, 0.0) * (
                (float(row[header.index(col)]) - col_means[col]) / col_stds[col] if col_stds[col] != 0 else 0.0
            )
            for col in header if not is_goal(col)
        )
        return pred

    # Compare test row against all rows on goal columns (lower is better).
    better_count = 0
    for r in rows:
        all_better = True
        for gcol in goal_cols:
            if is_goal(gcol):
                actual_val = float(r[header.index(gcol)])
                pred_val = predict_goal(gcol)
                if pred_val > actual_val:  # test row worse (higher value)
                    all_better = False
                    break
        if all_better:
            better_count += 1

    p = better_count / n_rows
    reason = f"This setup would beat {p*100:.1f}% of reference setups."

    return (p, reason)
