import csv
import random
from step01 import remember

def race(header, rows, label):
    """
    Compares the customer's current strategy (biggest teams) against
    a random search and a heuristic search (top-performing setups).
    """
    budget = 50
    labeled = {}

    def get_score(row_idx):
        if row_idx not in labeled:
            if len(labeled) >= budget:
                raise Exception("Label budget exceeded")
            labeled[row_idx] = label(row_idx)
        
        vals = labeled[row_idx]
        score = 0
        for i, name in enumerate(header):
            if name.endswith('-') or name.endswith('+'):
                # Find index in the goal list returned by label
                goal_indices = [j for j, n in enumerate(header) if n.endswith('-') or n.endswith('+')]
                val = vals[goal_indices.index(i)]
                score += val if name.endswith('-') else (1.0 - val)
        return score

    # Strategy 1: Random
    def strategy_random():
        best_idx = -1
        best_score = float('inf')
        for _ in range(16):
            idx = random.randint(0, len(rows) - 1)
            score = get_score(idx)
            if score < best_score:
                best_score, best_idx = score, idx
        return best_idx, best_score

    # Strategy 2: Customer's Current (Biggest Teams)
    def strategy_biggest():
        team_idx = header.index('Team Size')
        # Sort by team size descending, pick top 16
        sorted_indices = sorted(range(len(rows)), key=lambda i: rows[i][team_idx], reverse=True)
        best_idx = -1
        best_score = float('inf')
        for idx in sorted_indices[:16]:
            score = get_score(idx)
            if score < best_score:
                best_score, best_idx = score, idx
        return best_idx, best_score

    # Strategy 3: Heuristic (Top 5% of historical data)
    def strategy_heuristic():
        # We use the remember logic from step01 to find candidates
        # Since we don't have labels for all, we use the first 100 as a proxy
        candidates = remember(header, rows[:100])
        best_idx = -1
        best_score = float('inf')
        for idx in candidates[:16]:
            score = get_score(idx)
            if score < best_score:
                best_score, best_idx = score, idx
        return best_idx, best_score

    results = {
        "random": strategy_random(),
        "biggest_teams": strategy_biggest(),
        "heuristic": strategy_heuristic()
    }

    winner = min(results, key=lambda k: results[k][1])
    return {
        "winner": winner,
        "row": results[winner][0],
        "tried": list(results.keys())
    }

if __name__ == "__main__":
    filename = "pom3a.n1000.train.csv"
    try:
        with open(filename, mode='r', encoding='utf-8') as f:
            reader = csv.reader(f)
            header = next(reader)
            rows = []
            true_goals = []
            for row in reader:
                processed = [float(val) if val.replace('.', '', 1).isdigit() or (val.startswith('-') and val[1:].replace('.', '', 1).isdigit()) else val for val in row]
                rows.append(processed)
                true_goals.append([float(row[i]) for i, name in enumerate(header) if name.endswith('-') or name.endswith('+')])
        
        masked_rows = [["?" for _ in row] for row in rows]
        
        def label(i):
            return true_goals[i]

        result = race(header, masked_rows, label)
        print("Customer Inquiry: Is your way of choosing better than our 'biggest teams' strategy?")
        print(f"Answer: The winning strategy was '{result['winner']}'.")
        if result['winner'] == "biggest_teams":
            print("It appears your current strategy of maximizing team size is effective.")
        else:
            print(f"Our heuristic search outperformed your current approach, finding a better setup at index {result['row']}.")
            
    except Exception as e:
        print(f"Error: {e}")
        exit(1)
