import csv
from step01 import remember

def flow(header, row, seen):
    """
    Decides whether to accept a row based on the 'Secretary Problem' logic:
    Observe a sample (37% of total expected data) to establish a baseline,
    then accept the first row that is better than the best seen so far.
    """
    # Identify goal indices
    goals = [(i, name.endswith('-')) for i, name in enumerate(header) 
             if name.endswith('-') or name.endswith('+')]

    def get_score(r):
        total = 0
        for col_idx, is_min in goals:
            val = float(r[col_idx])
            total += val if is_min else (1.0 - val)
        return total

    # If no rows seen yet, we cannot accept based on comparison
    if not seen:
        return False

    # 37% threshold for the secretary problem
    # We estimate total size based on the provided data context (800 rows)
    # or use the current seen count to define the sample phase.
    sample_size = max(1, int(len(seen) * 0.37))
    
    # Calculate best score seen so far
    best_score = min(get_score(s) for s in seen)
    
    if len(seen) < sample_size:
        return False
    
    current_score = get_score(row)
    
    # Accept if better than anything seen in the sample phase
    return current_score < best_score

if __name__ == "__main__":
    filename = "pom3a.n1000.train.csv"
    try:
        with open(filename, mode='r', encoding='utf-8') as f:
            reader = csv.reader(f)
            header = next(reader)
            rows = []
            for row in reader:
                rows.append([float(val) for val in row])
        
        accepted = []
        seen = []
        for row in rows:
            if flow(header, row, seen):
                accepted.append(row)
            seen.append(row)
            
        print(f"Processed {len(rows)} proposals.")
        print(f"Accepted {len(accepted)} proposals based on the optimal stopping strategy.")
        print("Strategy: Observe 37% of incoming data, then accept the first improvement.")
            
    except FileNotFoundError:
        print(f"Error: {filename} not found.")
        exit(1)
