def flow(header, row, seen):
    goal_cols = ['Cost-', 'Score-', 'Idle-']

    def is_goal(col):
        return col.endswith('-') and col != '-'

    n_seen = len(seen)
    if n_seen == 0:
        # First row — approve it to get started.
        return True

    # Learn stats from seen rows for non-goal columns.
    col_means = {}
    col_stds = {}
    for col in header:
        if not is_goal(col):
            vals = [float(r[header.index(col)]) for r in seen]
            m = sum(vals) / len(vals)
            var = sum((v - m) ** 2 for v in vals) / max(len(vals) - 1, 1)
            col_means[col] = m
            col_stds[col] = var ** 0.5

    # Normalise Cost- from seen rows (lower is better).
    cost_vals_seen = [float(r[header.index('Cost-')]) for r in seen]
    mn_cost, mx_cost = min(cost_vals_seen), max(cost_vals_seen)

    def norm_cost(v):
        if mx_cost == mn_cost:
            return 0.0
        return (v - mn_cost) / (mx_cost - mn_cost)

    cost_norm_seen = [norm_cost(v) for v in cost_vals_seen]

    # Correlations between non-goal columns and normalised Cost-.
    col_corrs = {}
    for col in header:
        if not is_goal(col):
            vals = [float(r[header.index(col)]) for r in seen]
            n = len(vals)
            mx_v = sum(vals) / n
            mx_c = sum(cost_norm_seen) / n
            num = sum((vals[i] - mx_v) * (cost_norm_seen[i] - mx_c) for i in range(n))
            den_x = sum((v - mx_v) ** 2 for v in vals) ** 0.5
            den_y = sum((c - mx_c) ** 2 for c in cost_norm_seen) ** 0.5
            col_corrs[col] = num / (den_x * den_y) if den_x != 0 and den_y != 0 else 0.0

    # Predict Cost- for the new row.
    def predict(row):
        pred_norm = sum(
            col_corrs.get(col, 0.0) * (
                (float(row[header.index(col)]) - col_means[col]) / col_stds[col] if col_stds[col] != 0 else 0.0
            )
            for col in header if not is_goal(col)
        )
        return max(0.0, min(1.0, pred_norm))

    norm_pred = predict(row)
    cost_pred = mn_cost + norm_pred * (mx_cost - mn_cost) if mx_cost != mn_cost else 0.0

    # Threshold: approve if predicted Cost- is in the top ~5% of seen rows.
    k = max(1, int(n_seen * 0.05))
    sorted_costs = sorted(cost_vals_seen)
    threshold = sorted_costs[-k]

    return cost_pred <= threshold


if __name__ == "__main__":
    import csv

    with open('pom3a.train.csv', 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        rows = [list(map(float, row)) for row in reader]

    # Simulate sequential arrival and show how many we'd approve.
    approved = []
    seen = []
    for i, row in enumerate(rows):
        if flow(header, row, seen):
            approved.append(i)
            break
        seen.append(row)

    print(f"Approved row {approved[0]} out of {len(rows)} total rows.")
