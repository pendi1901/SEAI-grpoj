import csv

def remember(header, rows):
    """
    Identifies rows that perform well across all goal columns.
    Goals ending in '-' are minimized, '+' are maximized.
    """
    goals = []
    for i, name in enumerate(header):
        if name.endswith('-') or name.endswith('+'):
            goals.append((i, name.endswith('-')))

    # Calculate a score for each row based on the normalized rank of its goals.
    # Since all goals are numeric and in [0, 1], we can use the values directly.
    # For minimization (-), lower is better. For maximization (+), higher is better.
    # We want to minimize (value) for '-' and minimize (1 - value) for '+'.
    
    scored_rows = []
    for idx, row in enumerate(rows):
        total_penalty = 0
        for col_idx, is_min in goals:
            val = float(row[col_idx])
            if is_min:
                total_penalty += val
            else:
                total_penalty += (1.0 - val)
        scored_rows.append((total_penalty, idx))
    
    # Sort by total penalty (ascending)
    scored_rows.sort()
    
    # Return top 5% of rows (or at least 1)
    limit = max(1, len(rows) // 20)
    return [item[1] for item in scored_rows[:limit]]

def certify(xs, ys):
    """
    Returns True if xs and ys are statistically similar (same distribution),
    False if they differ significantly. Uses a simple A12 effect size check.
    """
    def a12(lst1, lst2):
        m, n = len(lst1), len(lst2)
        lst1 = sorted(lst1)
        lst2 = sorted(lst2)
        
        # Count how many times items in lst1 are greater than items in lst2
        greater = 0
        j = 0
        for i in range(m):
            while j < n and lst2[j] < lst1[i]:
                j += 1
            greater += j
            
        # Count ties
        ties = 0
        for i in range(m):
            for j in range(n):
                if lst1[i] == lst2[j]:
                    ties += 1
        
        return (greater + 0.5 * ties) / (m * n)

    # A12 threshold: 0.56 is a small effect size, 0.64 medium, 0.71 large.
    # If A12 is close to 0.5, they are likely the same.
    score = a12(xs, ys)
    return 0.44 <= score <= 0.56

def analyze_teams(header, rows):

    team_idx = header.index('Team Size')

def flow(header, row, seen):
    """
    Decides whether to accept a row based on the 'Secretary Problem' logic:
    Observe a sample (37% of total expected data) to establish a baseline,
    then accept the first row that is better than the best seen so far.
    """
    # Identify goal indices
    goals = [(i, name.endswith('-')) for i, name in enumerate(header) 
             if name.endswith('-') or name.endswith('+')]

    def get_score(r):
        total = 0
        for col_idx, is_min in goals:
            val = float(r[col_idx])
            total += val if is_min else (1.0 - val)
        return total

    # If no rows seen yet, we cannot accept based on comparison
    if not seen:
        return False

    # 37% threshold for the secretary problem
    # We estimate total size based on the provided data context (800 rows)
    # or use the current seen count to define the sample phase.
    sample_size = max(1, int(len(seen) * 0.37))
    
    # Calculate best score seen so far
    best_score = min(get_score(s) for s in seen)
    
    if len(seen) < sample_size:
        return False
    
    current_score = get_score(row)
    
    # Accept if better than anything seen in the sample phase
    return current_score < best_score

def blame(header, rows):

    We define 'worst' as the bottom 10% of rows based on the 'Idle-' goal.

                     if not (name.endswith('-') or name.endswith('+'))]
    
    # Sort rows by Idle- (descending, so worst is at the top)
    sorted_rows = sorted(rows, key=lambda r: r[idle_idx], reverse=True)
    
    # Take the worst 10%
    limit = max(1, len(rows) // 10)
    worst_rows = sorted_rows[:limit]
    
    # Calculate average of each input column for the worst rows
    # and compare to the global average to find the most "deviant" inputs.
    def get_avg(data, idx):
        return sum(r[idx] for r in data) / len(data)
    
    global_avgs = [get_avg(rows, i) for i in input_indices]
    worst_avgs = [get_avg(worst_rows, i) for i in input_indices]
    
    # Find inputs where the worst rows deviate most from the global average
    # (Assuming higher values in inputs might drive higher idle time)
    diffs = []
    for i, col_idx in enumerate(input_indices):
        diff = worst_avgs[i] - global_avgs[i]
        diffs.append((diff, header[col_idx]))
    
    # Return the top 2 inputs that contribute most to the high idle time
    diffs.sort(key=lambda x: x[0], reverse=True)
    return [name for diff, name in diffs[:2]]

def justify(header, rows, row):
    """
    Estimates the goal values for the input row using the average of the 
    top 5% of training rows, then calculates the percentile rank.
    """
    # 1. Estimate goal values
    best_indices = remember(header, rows)
    goal_indices = [i for i, name in enumerate(header) 
                    if name.endswith('-') or name.endswith('+')]
    
    estimated_row = list(row)
    for g_idx in goal_indices:
        avg_val = sum(rows[i][g_idx] for i in best_indices) / len(best_indices)
        estimated_row[g_idx] = avg_val
    
    # 2. Calculate percentile rank
    target_score = get_score(header, estimated_row)
    better_count = 0
    for r in rows:

            better_count += 1
            
    p = better_count / len(rows)
    reason = f"This setup is better than {p:.1%} of all known configurations based on its projected performance metrics."
    
    return p, reason

def choose(header, rows, label):
    """
    Selects the best row out of 50 trials.
    Strategy: 
    1. Sample 50 rows randomly to get a baseline of the data distribution.
    2. Label these 50 rows to see their true goal values.
    3. Return the index of the row that performed best.
    """
    # We are allowed 50 labels.
    # We pick 50 indices spread across the dataset to get a representative sample.
    n = len(rows)
    step = max(1, n // 50)
    indices_to_label = [i for i in range(0, n, step)][:50]
    
    best_idx = -1
    best_score = float('inf')
    
    # Identify goal indices and directions
    goals = [(i, name.endswith('-')) for i, name in enumerate(header) 
             if name.endswith('-') or name.endswith('+')]

    for idx in indices_to_label:
        # Get true values from the oracle
        true_goals = label(idx)
        
        # Calculate score for this row

            else:
                total_penalty += (1.0 - val)

            best_score = total_penalty
            best_idx = idx
            
    return best_idx

import csv

def spot(header, rows):
    """
    Identifies rows that are outliers in the input space.
    We calculate the Mahalanobis-like distance (using standard deviation)
    for each row's input columns. Rows with high deviation are suspicious.
    """
    input_indices = [i for i, name in enumerate(header) 
                     if not (name.endswith('-') or name.endswith('+'))]
    
    # Calculate means and standard deviations for each input column
    stats = []
    for i in input_indices:
        col_vals = [row[i] for row in rows if isinstance(row[i], (int, float))]
        mean = sum(col_vals) / len(col_vals)
        std = (sum((x - mean)**2 for x in col_vals) / len(col_vals))**0.5
        stats.append((i, mean, std if std > 0 else 1.0))
    
    # Score each row by its total deviation from the mean
    suspicion_scores = []
    for idx, row in enumerate(rows):
        score = 0
        for col_idx, mean, std in stats:
            val = row[col_idx]
            if isinstance(val, (int, float)):
                score += abs(val - mean) / std
        suspicion_scores.append((score, idx))
    
    # Sort by suspicion score descending
    suspicion_scores.sort(key=lambda x: x[0], reverse=True)
    
    # Return the top 10 most suspicious row indices
    return [item[1] for item in suspicion_scores[:10]]
