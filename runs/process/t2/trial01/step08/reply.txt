import csv


def fix(header, row, levers):
    goal_cols = ['Cost-', 'Score-', 'Idle-']

    def is_goal(col):
        return col.endswith('-') and col != '-'

    # Load training data for correlations.
    with open('pom3a.train.csv', 'r') as f:
        reader = csv.reader(f)
        train_header = next(reader)
        train_rows = [list(map(float, r)) for r in reader]

    n_train = len(train_rows)
    if n_train == 0:
        return {}

    # Stats and correlations from training data.
    col_means = {}
    col_stds = {}
    for col in header:
        if not is_goal(col):
            vals = [float(r[header.index(col)]) for r in train_rows]
            m = sum(vals) / len(vals)
            var = sum((v - m) ** 2 for v in vals) / max(len(vals) - 1, 1)
            col_means[col] = m
            col_stds[col] = var ** 0.5

    goal_corrs = {}
    for gcol in goal_cols:
        if not is_goal(gcol):
            continue
        gvals = [float(r[header.index(gcol)]) for r in train_rows]
        n_g = len(gvals)
        mx_g = sum(gvals) / n_g
        var_g = sum((v - mx_g) ** 2 for v in gvals) ** 0.5

        goal_corrs[gcol] = {}
        for col in header:
            if not is_goal(col):
                cvals = [float(r[header.index(col)]) for r in train_rows]
                n_c = len(cvals)
                mx_c = sum(cvals) / n_c
                var_c = sum((v - mx_c) ** 2 for v in cvals) ** 0.5
                goal_corrs[gcol][col] = (sum(
                    (cvals[i] - mx_c) * (gvals[i] - mx_g) for i in range(n_g)
                ) / (var_c * var_g)) if var_c != 0 and var_g != 0 else 0.0

    # Predict goal values given a row's non-goal column values.
    def predict_goals(row):
        preds = {}
        for gcol in goal_cols:
            pred = sum(
                goal_corrs[gcol].get(col, 0.0) * (
                    (float(row[header.index(col)]) - col_means[col]) / col_stds[col] if col_stds[col] != 0 else 0.0
                )
                for col in header if not is_goal(col)
            )
            preds[gcol] = pred
        return preds

    # Current goal values of the bad row (lower is better).
    current_goals = {g: float(row[header.index(g)]) for g in goal_cols}

    # For each lever, try extreme values and score improvement.
    best_change = {}
    best_score = float('inf')

    for col, allowed_vals in levers.items():
        if is_goal(col):
            continue

        current_val = row[header.index(col)]

        # Try candidate values: extremes of numeric range or each symbolic value.
        candidates = []
        if isinstance(allowed_vals, tuple) and len(allowed_vals) == 2:
            candidates.append(allowed_vals[0])
            candidates.append(allowed_vals[1])
        elif isinstance(allowed_vals, list):
            candidates.extend(allowed_vals[:5])

        for new_val in candidates:
            test_row = list(row)
            idx = header.index(col)
            if isinstance(new_val, str):
                test_row[idx] = '?'
            else:
                test_row[idx] = float(new_val)

            preds = predict_goals(test_row)
            score = sum(
                max(0.0, current_goals[gcol] - preds[gcol]) for gcol in goal_cols
            )
            change_mag = abs(float(new_val) - current_val) if isinstance(current_val, float) else 1.0

            # Prefer more improvement per unit of change.
            ratio = score / max(change_mag, 1e-9)
            if ratio > best_score:
                best_score = ratio
                best_change[col] = new_val

    return best_change

    import csv

    with open('pom3a.train.csv', 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        rows = [list(map(float, r)) for r in reader]

    # Find a bad row (high Cost- and Idle-, low Score-).
    goal_cols = ['Cost-', 'Score-', 'Idle-']

    def is_goal(col):
        return col.endswith('-') and col != '-'

    best_idx = 0
    worst_score = float('inf')
    for i, row in enumerate(rows):
        score = sum(float(row[header.index(g)]) for g in goal_cols)
        if score < worst_score:
            worst_score = score
            best_idx = i

    bad_row = rows[best_idx]
    print(f"Bad setup (row {best_idx}):")
    for col in header:
        print(f"  {col}: {bad_row[header.index(col)]:.4f}" if is_goal(col) else f"  {col}")

    # Determine levers from training data range.
    lever_candidates = {}
    for col in header:
        if not is_goal(col):
            vals = [float(r[header.index(col)]) for r in rows]
            mn, mx = min(vals), max(vals)
            if isinstance(mn, float) and isinstance(mx, float):
                lever_candidates[col] = (mn, mx)
            else:
                lever_candidates[col] = list(set(str(v) for v in vals))

    result = fix(header, bad_row, lever_candidates)
    print(f"\nRecommended fixes:")
    for col, val in result.items():
        print(f"  {col}: {val}")
