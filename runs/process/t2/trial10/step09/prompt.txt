import csv

def remember(header, rows):
    """
    Identifies rows that perform well across all goal columns.
    Goals ending in '-' are minimized, '+' are maximized.
    """
    goal_indices = []
    for i, name in enumerate(header):
        if name.endswith('-') or name.endswith('+'):
            goal_indices.append((i, name[-1]))

    def score_row(row):
        # We want to minimize all goals. 
        # For '-', we want low values. For '+', we want high values (so we negate).
        scores = []
        for idx, op in goal_indices:
            val = float(row[idx])
            scores.append(val if op == '-' else -val)
        return scores

    # Calculate scores for all rows
    scored_rows = []
    for i, row in enumerate(rows):
        scored_rows.append((score_row(row), i))

    # Sort by the sum of normalized goals (simple Pareto-like approach)
    # Since all goals are in [0, 1], we can sum them directly
    scored_rows.sort(key=lambda x: sum(x[0]))

    # Return top 5% (or at least 1)
    limit = max(1, len(rows) // 20)
    return [item[1] for item in scored_rows[:limit]]

import csv

import math

def certify(xs, ys):
    """
    Returns True if xs and ys are statistically similar (null hypothesis),
    False if they differ significantly. Uses a simple t-test approach.
    """
    def mean(data):
        return sum(data) / len(data)

    def variance(data):
        m = mean(data)
        return sum((x - m) ** 2 for x in data) / (len(data) - 1)

    n1, n2 = len(xs), len(ys)
    m1, m2 = mean(xs), mean(ys)
    v1, v2 = variance(xs), variance(ys)

    # Standard error of the difference
    se = math.sqrt((v1 / n1) + (v2 / n2))
    if se == 0:
        return True
    
    t_stat = abs(m1 - m2) / se
    # For a 95% confidence level, critical t-value is approx 1.96
    return t_stat < 1.96

def analyze_teams(header, rows):
    # Find indices for Team Size and Cost-

    is_same = certify(small_teams, big_teams)

import csv

def flow(header, row, seen):
    """
    Decides whether to accept a row based on whether it is better than 
    the best seen so far. We use the logic from step01 to define 'better'.
    """
    # If it's the first row, we have no baseline, so we accept it.
    if not seen:
        return True

    # Identify goal indices and directions
    goal_indices = []
    for i, name in enumerate(header):
        if name.endswith('-') or name.endswith('+'):
            goal_indices.append((i, name[-1]))

    def get_score(r):
        # Lower is better: minimize '-' and negate '+'
        return sum(float(r[idx]) if op == '-' else -float(r[idx]) for idx, op in goal_indices)

    # Calculate score for current row
    current_score = get_score(row)
    
    # Calculate scores for all previously seen rows
    seen_scores = [get_score(r) for r in seen]
    
    # Accept if the current row is better than the best seen so far
    return current_score < min(seen_scores)

import csv
import step01

def blame(header, rows):
    """
    Identifies input columns that correlate with high 'Idle-' values.
    We define 'worst' as the top 10% of rows with the highest 'Idle-' values.
    We then compare the average of inputs in these worst rows against the 
    average of the rest of the population.
    """
    idle_idx = header.index("Idle-")
    input_indices = [i for i, name in enumerate(header) 
                     if not (name.endswith('-') or name.endswith('+'))]
    
    # Sort rows by Idle- descending to find the worst
    sorted_rows = sorted(rows, key=lambda r: float(r[idle_idx]), reverse=True)
    limit = max(1, len(rows) // 10)
    worst = sorted_rows[:limit]
    others = sorted_rows[limit:]
    
    # Calculate difference in means for each input
    diffs = []
    for i in input_indices:
        w_vals = [float(r[i]) for r in worst]
        o_vals = [float(r[i]) for r in others]
        w_avg = sum(w_vals) / len(w_vals)
        o_avg = sum(o_vals) / len(o_vals)
        diffs.append((abs(w_avg - o_avg), header[i]))
    
    # Sort by magnitude of difference and return the top 2
    diffs.sort(key=lambda x: x[0], reverse=True)
    return [name for _, name in diffs[:2]]

    # Create a full row representation for scoring

import csv

def fix(header, row, levers):
    """
    Finds the smallest change to levers that moves the row towards the 
    best-performing setups found in the training data.
    """
    # Identify goal indices and directions
    goal_indices = [(i, name[-1]) for i, name in enumerate(header) 
                    if name.endswith('-') or name.endswith('+')]
    
    # Calculate the average of the best 5% of rows to use as a target
    # We use step01.remember to find the best rows
    filename = "pom3a.n1000.train.csv"
    with open(filename, mode='r', newline='') as f:
        reader = csv.reader(f)
        header_file = next(reader)
        rows = [r for r in reader]
    
    best_indices = step01.remember(header_file, rows)
    
    # Calculate target values for each lever
    recommendations = {}
    for col_name, allowed in levers.items():
        col_idx = header.index(col_name)
        
        # If numeric, move towards the average of the best rows
        if isinstance(allowed, tuple):
            low, high = allowed
            best_vals = [float(rows[i][col_idx]) for i in best_indices]
            target = sum(best_vals) / len(best_vals)
            # Clamp to allowed range
            recommendations[col_name] = max(low, min(high, target))
        else:
            # If symbolic, pick the most frequent value in the best rows
            best_vals = [rows[i][col_idx] for i in best_indices]
            recommendations[col_name] = max(set(best_vals), key=best_vals.count)
            
    return recommendations
