import csv

def justify(header, rows, row):
    """
    Compares the provided row (with goals as '?') against all known rows.
    Estimates the goals for the row based on its inputs by finding the 
    average of the 5 nearest neighbors in the training set.
    """
    # Identify indices
    goal_indices = [(i, name[-1]) for i, name in enumerate(header) 
                    if name.endswith('-') or name.endswith('+')]
    input_indices = [i for i, name in enumerate(header) 
                     if not (name.endswith('-') or name.endswith('+'))]

    # Estimate goals for the row using 5-Nearest Neighbors
    def dist(r1, r2):
        d = 0
        for i in input_indices:
            d += (float(r1[i]) - float(r2[i]))**2
        return d**0.5

    distances = []
    for i, r in enumerate(rows):
        distances.append((dist(r, row), i))
    distances.sort(key=lambda x: x[0])
    
    # Average the goals of the 5 nearest neighbors
    k = 5
    estimated_goals = []
    for idx, _ in goal_indices:
        vals = [float(rows[i][idx]) for _, i in distances[:k]]
        estimated_goals.append(sum(vals) / k)

    # Create a full row representation for scoring
    full_row = list(row)
    for i, (idx, _) in enumerate(goal_indices):
        full_row[idx] = estimated_goals[i]

    def get_score(r):
        return sum(float(r[idx]) if op == '-' else -float(r[idx]) 
                   for idx, op in goal_indices)

    target_score = get_score(full_row)
    all_scores = [get_score(r) for r in rows]
    
    # Fraction of rows that are worse than the target row
    # A row is better if its score is lower
    better_than_count = sum(1 for s in all_scores if s > target_score)
    p = better_than_count / len(rows)
    
    reason = f"This setup is better than {p:.2f} of all known configurations based on its estimated performance goals."
    return p, reason

if __name__ == "__main__":
    filename = "pom3a.n1000.train.csv"
    try:
        with open(filename, mode='r', newline='') as f:
            reader = csv.reader(f)
            header = next(reader)
            rows = [row for row in reader]
            
            # Example: Compare a specific row (e.g., row 10)
            sample_row = list(rows[10])
            for i, name in enumerate(header):
                if name.endswith('-') or name.endswith('+'):
                    sample_row[i] = "?"
            
            p, reason = justify(header, rows, sample_row)
            print(f"Justification: {reason}")
            
    except FileNotFoundError:
        print(f"Error: {filename} not found.")
        exit(1)
