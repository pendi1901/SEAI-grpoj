import csv

def remember(header, rows):
    """
    Identifies rows that perform well across all goal columns.
    Goals ending in '-' are minimized, '+' are maximized.
    """
    goal_indices = []
    for i, name in enumerate(header):
        if name.endswith('-') or name.endswith('+'):
            goal_indices.append((i, name[-1]))

    def score_row(row):
        # We want to minimize all goals. 
        # For '-', we want low values. For '+', we want high values (so we negate).
        scores = []
        for idx, op in goal_indices:
            val = float(row[idx])
            scores.append(val if op == '-' else -val)
        return scores

    # Calculate scores for all rows
    scored_rows = []
    for i, row in enumerate(rows):
        scored_rows.append((score_row(row), i))

    # Sort by the sum of normalized goals (simple Pareto-like approach)
    # Since all goals are in [0, 1], we can sum them directly
    scored_rows.sort(key=lambda x: sum(x[0]))

    # Return top 5% (or at least 1)
    limit = max(1, len(rows) // 20)
    return [item[1] for item in scored_rows[:limit]]

def certify(xs, ys):
    """
    Returns True if xs and ys are statistically similar (null hypothesis),
    False if they differ significantly. Uses a simple t-test approach.
    """
    def mean(data):
        return sum(data) / len(data)

    def variance(data):
        m = mean(data)
        return sum((x - m) ** 2 for x in data) / (len(data) - 1)

    n1, n2 = len(xs), len(ys)
    m1, m2 = mean(xs), mean(ys)
    v1, v2 = variance(xs), variance(ys)

    # Standard error of the difference
    se = math.sqrt((v1 / n1) + (v2 / n2))
    if se == 0:
        return True
    
    t_stat = abs(m1 - m2) / se
    # For a 95% confidence level, critical t-value is approx 1.96
    return t_stat < 1.96

def flow(header, row, seen):
    """
    Decides whether to accept a row based on whether it is better than 
    the best seen so far. We use the logic from step01 to define 'better'.
    """
    # If it's the first row, we have no baseline, so we accept it.
    if not seen:
        return True

    # Identify goal indices and directions
    goal_indices = []
    for i, name in enumerate(header):
        if name.endswith('-') or name.endswith('+'):
            goal_indices.append((i, name[-1]))

    def get_score(r):
        # Lower is better: minimize '-' and negate '+'
        return sum(float(r[idx]) if op == '-' else -float(r[idx]) for idx, op in goal_indices)

    # Calculate score for current row
    current_score = get_score(row)
    
    # Calculate scores for all previously seen rows
    seen_scores = [get_score(r) for r in seen]
    
    # Accept if the current row is better than the best seen so far

def blame(header, rows):
    """
    Identifies input columns that correlate with high 'Idle-' values.
    We define 'worst' as the top 10% of rows with the highest 'Idle-' values.
    We then compare the average of inputs in these worst rows against the 
    average of the rest of the population.
    """
    idle_idx = header.index("Idle-")
    input_indices = [i for i, name in enumerate(header) 
                     if not (name.endswith('-') or name.endswith('+'))]
    
    # Sort rows by Idle- descending to find the worst
    sorted_rows = sorted(rows, key=lambda r: float(r[idle_idx]), reverse=True)
    limit = max(1, len(rows) // 10)
    worst = sorted_rows[:limit]
    others = sorted_rows[limit:]
    
    # Calculate difference in means for each input
    diffs = []
    for i in input_indices:
        w_vals = [float(r[i]) for r in worst]
        o_vals = [float(r[i]) for r in others]
        w_avg = sum(w_vals) / len(w_vals)
        o_avg = sum(o_vals) / len(o_vals)
        diffs.append((abs(w_avg - o_avg), header[i]))
    
    # Sort by magnitude of difference and return the top 2
    diffs.sort(key=lambda x: x[0], reverse=True)
    return [name for _, name in diffs[:2]]

import csv
import math

def spot(header, rows):
    """
    Identifies rows that contain outliers. We use the Interquartile Range (IQR)
    method on numeric columns to detect values that are statistically 
    unlikely (outside 1.5 * IQR).
    """
    numeric_indices = [i for i, name in enumerate(header) 
                       if not (name.endswith('-') or name.endswith('+'))]
    
    # Calculate IQR for each numeric column
    outlier_scores = [0] * len(rows)
    
    for i in numeric_indices:
        vals = sorted([float(row[i]) for row in rows])
        q1 = vals[len(vals) // 4]
        q3 = vals[3 * len(vals) // 4]
        iqr = q3 - q1
        low_bound = q1 - 1.5 * iqr
        high_bound = q3 + 1.5 * iqr
        
        # Increment suspicion score for rows outside bounds
        for r_idx, row in enumerate(rows):
            val = float(row[i])
            if val < low_bound or val > high_bound:
                outlier_scores[r_idx] += 1
                
    # Create list of (index, score) and sort by score descending
    indexed_scores = list(enumerate(outlier_scores))
    indexed_scores.sort(key=lambda x: x[1], reverse=True)
    
    # Return indices of rows with at least one outlier, most suspicious first
    return [idx for idx, score in indexed_scores if score > 0]

def find_similar(header, rows, target_row, k=5):
    """
    Finds the k rows most similar to the target_row using Euclidean distance
    on input columns.
    """
    input_indices = [i for i, name in enumerate(header) 
                     if not (name.endswith('-') or name.endswith('+'))]
    
    def dist(r1, r2):
        d = 0
        for i in input_indices:
            # Assuming numeric inputs for this dataset
            d += (float(r1[i]) - float(r2[i]))**2
        return d**0.5

    # Calculate distances to all rows
    distances = []
    for i, row in enumerate(rows):
        distances.append((dist(row, target_row), i))
    
    # Return indices of k nearest neighbors
    distances.sort(key=lambda x: x[0])
    return [idx for _, idx in distances[:k]]

def guess(header, train, test):
    """
    Estimates the first goal column for test rows.
    The customer asks about a scenario where requirements change twice as fast
    halfway through the data. We model this by splitting the training data
    into two halves and weighting the second half (the 'faster' change period)
    more heavily when calculating the average for the estimate.
    """
    goal_idx = -1
    for i, name in enumerate(header):
        if name.endswith('-') or name.endswith('+'):
            goal_idx = i
            break
    
    # Split training data into two halves
    mid = len(train) // 2
    first_half = train[:mid]
    second_half = train[mid:]
    
    # Identify best rows in each half using step01
    best_first = step01.remember(header, first_half)
    best_second = step01.remember(header, second_half)
    
    # Calculate weighted average: second half (faster changes) counts for 2x
    vals_first = [float(first_half[i][goal_idx]) for i in best_first]
    vals_second = [float(second_half[i][goal_idx]) for i in best_second]
    
    avg_first = sum(vals_first) / len(vals_first) if vals_first else 0
    avg_second = sum(vals_second) / len(vals_second) if vals_second else 0
    
    # Weighted estimate
    estimate = (avg_first + (2 * avg_second)) / 3
    
    return [estimate for _ in test]

import csv
import step01

def flow(header, row, seen):
    """
    Warns the customer if a new row is significantly worse than the 
    historical baseline. We define 'bad' as being in the bottom 25% 
    of performance compared to all previously seen rows.
    """
    if not seen:
        return True

    goal_indices = [(i, name[-1]) for i, name in enumerate(header) 
                    if name.endswith('-') or name.endswith('+')]

    def get_score(r):
        # Lower is better: minimize '-' and negate '+'
        return sum(float(r[idx]) if op == '-' else -float(r[idx]) 
                   for idx, op in goal_indices)

    current_score = get_score(row)
    seen_scores = [get_score(r) for r in seen]
    
    # Calculate the 75th percentile of seen scores
    # If the current score is worse (higher) than the 75th percentile, it's 'bad'
    seen_scores.sort()
    threshold_idx = int(len(seen_scores) * 0.75)
    threshold = seen_scores[threshold_idx]
    
    # Return True if the row is 'good' (<= threshold), False if 'bad'
    return current_score <= threshold

if __name__ == "__main__":
    filename = "pom3a.n1000.train.csv"
    try:
        with open(filename, mode='r', newline='') as f:
            reader = csv.reader(f)
            header = next(reader)
            rows = [row for row in reader]
            
            seen = []
            warnings = 0
            for i, row in enumerate(rows):
                if not flow(header, row, seen):
                    warnings += 1
                    if warnings <= 5:
                        print(f"Warning: Row {i} is performing poorly compared to history.")
                seen.append(row)
            
            print(f"Monitoring complete. Total programs processed: {len(rows)}.")
            print(f"Total performance warnings issued: {warnings}.")
            print("Logic: A program is flagged if its performance score falls into the bottom 25% of history.")
            
    except FileNotFoundError:
        print(f"Error: {filename} not found.")
        exit(1)

def blame(header, rows):
    """
    Identifies the two input columns that most strongly correlate with 
    poor performance (high Cost-).
    """
    # We reuse the logic from step05, which identifies the top 2 drivers
    # of high 'Idle-' values. We adapt it here to target 'Cost-'.

    # which is a standard correlation analysis of worst-performing rows.
    
    cost_idx = header.index("Cost-")
    input_indices = [i for i, name in enumerate(header) 
                     if not (name.endswith('-') or name.endswith('+'))]
    
    # Sort rows by Cost- descending to find the worst
    sorted_rows = sorted(rows, key=lambda r: float(r[cost_idx]), reverse=True)
    limit = max(1, len(rows) // 10)
    worst = sorted_rows[:limit]
    others = sorted_rows[limit:]
    
    # Calculate difference in means for each input
    diffs = []
    for i in input_indices:
        w_vals = [float(r[i]) for r in worst]
        o_vals = [float(r[i]) for r in others]

    # Sort by magnitude of difference and return the top 2
    diffs.sort(key=lambda x: x[0], reverse=True)
    return [name for _, name in diffs[:2]]
