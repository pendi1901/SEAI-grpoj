import csv
import math
from step01 import remember

def find_similar(header, rows, target_row):
    """
    Finds the 5 most similar rows to the target_row using Euclidean distance
    on numeric input columns.
    """
    input_indices = [i for i, name in enumerate(header) 
                     if not (name.endswith('-') or name.endswith('+'))]
    
    def dist(r1, r2):
        d = 0
        for i in input_indices:
            v1 = float(r1[i])
            v2 = float(r2[i])
            d += (v1 - v2) ** 2
        return math.sqrt(d)

    distances = []
    for i, row in enumerate(rows):
        distances.append((i, dist(row, target_row)))
    
    # Sort by distance ascending
    distances.sort(key=lambda x: x[1])
    return distances[:5]

def get_goals(header, row):
    return {header[i]: row[i] for i, name in enumerate(header) 
            if name.endswith('-') or name.endswith('+')}

if __name__ == "__main__":
    filename = "pom3a.n1000.train.csv"
    try:
        with open(filename, 'r') as f:
            reader = csv.reader(f)
            header = next(reader)
            rows = [row for row in reader]
            
        # Use the last row as the "proposed" setup
        target = rows[-1]
        # Mask the goals for the target
        target_masked = [("?" if (header[i].endswith('-') or header[i].endswith('+')) 
                          else val) for i, val in enumerate(target)]
        
        similar = find_similar(header, rows[:-1], target_masked)
        
        print("Proposed setup is similar to these past configurations:")
        for idx, d in similar:
            goals = get_goals(header, rows[idx])
            print(f"Row {idx} (dist {d:.4f}): {goals}")
            
    except FileNotFoundError:
        print(f"Error: {filename} not found.")
        exit(1)
