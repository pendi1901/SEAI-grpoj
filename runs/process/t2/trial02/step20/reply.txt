import csv
import random

def race(header, rows, label):
    """
    Compares the customer's current strategy (biggest teams) against 
    random sampling and a heuristic search.
    """
    budget = 50
    labeled = {}

    def get_true_goals(i):
        if i not in labeled:
            if len(labeled) >= budget:
                raise Exception("Budget exceeded")
            labeled[i] = label(i)
        return labeled[i]

    def get_score(i):
        row_goals = get_true_goals(i)
        s = 0
        for idx, name in enumerate(header):
            if name.endswith('-') or name.endswith('+'):
                val = float(row_goals[idx])
                s += val if name.endswith('-') else -val
        return s

    # Strategy 1: Random
    def strategy_random():
        idx = random.randint(0, len(rows) - 1)
        return idx, get_score(idx)

    # Strategy 2: Biggest Teams (Customer's current approach)
    def strategy_biggest_teams():
        team_idx = header.index('Team Size')
        # Find row with max Team Size
        idx = max(range(len(rows)), key=lambda i: float(rows[i][team_idx]))
        return idx, get_score(idx)

    # Strategy 3: Heuristic (Sample 10, pick best)
    def strategy_heuristic():
        indices = random.sample(range(len(rows)), 10)
        best_idx = indices[0]
        best_s = get_score(best_idx)
        for i in indices[1:]:
            s = get_score(i)
            if s < best_s:
                best_s = s
                best_idx = i
        return best_idx, best_s

    results = {}
    for name, func in [("random", strategy_random), 
                       ("biggest_teams", strategy_biggest_teams), 
                       ("heuristic", strategy_heuristic)]:
        idx, score = func()
        results[name] = (idx, score)

    winner = min(results, key=lambda n: results[n][1])
    return {
        "winner": winner,
        "row": results[winner][0],
        "tried": ["random", "biggest_teams", "heuristic"]
    }

if __name__ == "__main__":
    filename = "pom3a.n1000.train.csv"
    try:
        with open(filename, 'r') as f:
            reader = csv.reader(f)
            header = next(reader)
            rows = [row for row in reader]

        def label(i):
            return rows[i]

        outcome = race(header, rows, label)
        
        print("Comparison of project selection strategies:")
        print(f"The winning strategy is: {outcome['winner']}")
        print(f"The best configuration found was at row index: {outcome['row']}")
        
        if outcome['winner'] == "biggest_teams":
            print("Your current strategy of using the biggest teams is optimal.")
        else:
            print(f"Your current strategy ('biggest_teams') was outperformed by '{outcome['winner']}'.")
            print("We recommend adopting a more data-driven search approach.")
            
    except Exception as e:
        print(f"Error: {e}")
        exit(1)
