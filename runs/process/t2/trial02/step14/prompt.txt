import csv

def remember(header, rows):
    """
    Identifies rows that perform well across all goal columns.
    Goals ending in '-' are minimized, '+' are maximized.
    """
    goal_indices = []
    for i, name in enumerate(header):
        if name.endswith('-') or name.endswith('+'):
            goal_indices.append((i, name[-1]))

    def score_row(row):
        # We want to minimize all goals. 
        # For '-', lower is better. For '+', higher is better (so we negate it).
        scores = []
        for idx, op in goal_indices:
            val = float(row[idx])
            scores.append(val if op == '-' else -val)
        return scores

    # Calculate scores for all rows
    indexed_scores = []
    for i, row in enumerate(rows):
        indexed_scores.append((i, score_row(row)))

    # A row is "unusually well" if it dominates or performs well on all goals.
    # We sort by the sum of normalized scores or simply lexicographical comparison.
    # Here we sort by the sum of the goal values (since all are normalized to [0,1]).
    indexed_scores.sort(key=lambda x: sum(x[1]))

    # Return top 5% (or at least 1)
    limit = max(1, len(rows) // 20)
    return [item[0] for item in indexed_scores[:limit]]

def certify(xs, ys):
    """
    Returns True if xs and ys are statistically similar (same distribution),
    False if they differ significantly. Uses a simple bootstrap/permutation 
    logic or a basic difference of means check.
    """
    if not xs or not ys:
        return True
    
    # Calculate means
    mu_x = sum(xs) / len(xs)
    mu_y = sum(ys) / len(ys)
    
    # Calculate standard deviation
    def stdev(data, mean):
        return (sum((x - mean) ** 2 for x in data) / len(data)) ** 0.5
    
    sd_x = stdev(xs, mu_x)
    sd_y = stdev(ys, mu_y)
    
    # If the difference in means is less than the combined standard deviation,
    # we consider them "the same apart from noise".
    combined_sd = (sd_x + sd_y) / 2
    return abs(mu_x - mu_y) < combined_sd

def analyze_team_size_impact(header, rows):

import csv

def flow(header, row, seen):
    """
    Decides whether to accept a row based on whether it is better than 
    the current best seen so far.
    """
    # Identify goal indices
    goal_indices = [(i, name[-1]) for i, name in enumerate(header) 
                    if name.endswith('-') or name.endswith('+')]
    
    def get_score(r):
        # Calculate score: lower is better for '-', higher is better for '+'
        # We negate '+' values so that minimizing the sum always works.
        s = 0
        for idx, op in goal_indices:
            val = float(r[idx])
            s += val if op == '-' else -val
        return s

    # If this is the first row, accept it
    if not seen:
        return True
    
    # Calculate score of current row
    current_score = get_score(row)
    
    # Calculate scores of all previously seen rows
    seen_scores = [get_score(r) for r in seen]
    
    # Accept if current row is better than the best seen so far
    # (i.e., lower score than the minimum score seen)
    return current_score < min(seen_scores)

def blame(header, rows):
    """
    Identifies input columns that correlate with high 'Idle-' values.
    We define 'worst' as the bottom 10% of rows based on 'Idle-'.
    """
    idle_idx = header.index('Idle-')
    
    # Sort rows by Idle- (descending, so worst are at the top)
    sorted_rows = sorted(rows, key=lambda r: float(r[idle_idx]), reverse=True)
    worst_rows = sorted_rows[:len(rows) // 10]
    
    # Calculate average of each input column for the worst rows
    # and compare to the global average of the whole dataset.
    input_indices = [i for i, name in enumerate(header) 
                     if not (name.endswith('-') or name.endswith('+'))]
    
    def get_avg(data, idx):
        vals = [float(r[idx]) for r in data]
        return sum(vals) / len(vals)
    
    # We look for inputs where the worst rows have significantly higher values
    # than the global average (assuming higher input values drive higher Idle-)
    diffs = []
    for i in input_indices:
        worst_avg = get_avg(worst_rows, i)
        global_avg = get_avg(rows, i)
        diffs.append((header[i], worst_avg - global_avg))
    
    # Sort by the largest positive difference
    diffs.sort(key=lambda x: x[1], reverse=True)
    
    # Return the top 2 inputs that correlate most with high idle time
    return [d[0] for d in diffs[:2]]

import csv

def fix(header, row, levers):
    """
    Finds the best change to the provided levers to improve the row's score.
    """
    goal_indices = [(i, name[-1]) for i, name in enumerate(header) 
                    if name.endswith('-') or name.endswith('+')]
    
    def get_score(r):
        s = 0
        for idx, op in goal_indices:
            val = float(r[idx])
            s += val if op == '-' else -val
        return s

    best_change = {}
    best_score = get_score(row)
    
    for col_name, allowed in levers.items():
        if col_name not in header:
            continue
        idx = header.index(col_name)
        
        # Generate candidates based on lever type
        if isinstance(allowed, (list, tuple)) and len(allowed) == 2 and isinstance(allowed[0], (int, float)):
            low, high = allowed
            candidates = [low, (low + high) / 2, high]
        else:
            candidates = allowed
            
        for val in candidates:
            new_row = list(row)
            new_row[idx] = val
            new_score = get_score(new_row)
            
            if new_score < best_score:
                best_score = new_score
                best_change = {col_name: val}
                
    # Rule: Change at least one lever. If no improvement, pick the first lever's first option.
    if not best_change and levers:
        first_col = list(levers.keys())[0]
        allowed = levers[first_col]
        val = allowed[0] if isinstance(allowed, (list, tuple)) else allowed[0]
        best_change = {first_col: val}
        
    return best_change

import csv

def spot(header, rows):
    """
    Identifies rows that are outliers. We calculate the mean and standard 
    deviation for each numeric input column and flag rows that have values 
    more than 3 standard deviations away from the mean.
    """
    input_indices = [i for i, name in enumerate(header) 
                     if not (name.endswith('-') or name.endswith('+'))]
    
    # Calculate stats for each input column
    stats = {}
    for i in input_indices:
        vals = [float(row[i]) for row in rows]
        mu = sum(vals) / len(vals)
        sd = (sum((x - mu) ** 2 for x in vals) / len(vals)) ** 0.5
        stats[i] = (mu, sd)
    
    # Score rows based on how many standard deviations they are from the mean
    # A higher score means more suspicious.
    suspicion_scores = []
    for row_idx, row in enumerate(rows):
        score = 0
        for i in input_indices:
            mu, sd = stats[i]
            if sd > 0:
                val = float(row[i])
                dist = abs(val - mu) / sd
                if dist > 3:
                    score += dist
        if score > 0:
            suspicion_scores.append((row_idx, score))
            
    # Sort by suspicion score descending
    suspicion_scores.sort(key=lambda x: x[1], reverse=True)
    return [item[0] for item in suspicion_scores]

import csv

from step01 import remember

def race(header, rows, label):
    """
    Compares three strategies: random, greedy, and heuristic.
    Total budget of 50 labels shared.
    """
    budget = 50
    labeled = {}

    def get_true_goals(i):
        if i not in labeled:
            if len(labeled) >= budget:
                raise Exception("Budget exceeded")
            labeled[i] = label(i)
        return labeled[i]

    def get_score(i):
        row_goals = get_true_goals(i)
        s = 0
        goal_names = [name for name in header if name.endswith('-') or name.endswith('+')]
        for name in goal_names:
            idx = header.index(name)
            val = float(row_goals[idx])
            s += val if name.endswith('-') else -val
        return s

    # Strategy 1: Random
    def strategy_random():
        idx = random.randint(0, len(rows) - 1)
        return idx, get_score(idx)

    # Strategy 2: Greedy (uses remember logic on first half of data)
    def strategy_greedy():
        best_indices = remember(header, rows[:len(rows)//2])
        idx = best_indices[0]
        return idx, get_score(idx)

    # Strategy 3: Heuristic (sample 10 random and pick best)
    def strategy_heuristic():
        indices = random.sample(range(len(rows)), 10)
        best_idx = indices[0]
        best_s = get_score(best_idx)
        for i in indices[1:]:
            s = get_score(i)
            if s < best_s:
                best_s = s
                best_idx = i
        return best_idx, best_s

    results = {}
    for name, func in [("random", strategy_random), 
                       ("greedy", strategy_greedy), 
                       ("heuristic", strategy_heuristic)]:
        idx, score = func()
        results[name] = (idx, score)

    winner = min(results, key=lambda n: results[n][1])
    return {
        "winner": winner,
        "row": results[winner][0],
        "tried": ["random", "greedy", "heuristic"]
    }

if __name__ == "__main__":

def find_similar(header, rows, target_row):
    """
    Finds the 5 most similar rows to the target_row using Euclidean distance
    on numeric input columns.
    """
    input_indices = [i for i, name in enumerate(header) 
                     if not (name.endswith('-') or name.endswith('+'))]
    
    def dist(r1, r2):
        d = 0
        for i in input_indices:
            v1 = float(r1[i])
            v2 = float(r2[i])
            d += (v1 - v2) ** 2
        return math.sqrt(d)

    for i, row in enumerate(rows):
        distances.append((i, dist(row, target_row)))
    
    # Sort by distance ascending
    distances.sort(key=lambda x: x[1])
    return distances[:5]
