import csv

def remember(header, rows):
    """
    Identifies rows that perform well across all goal columns.
    Goals ending in '-' are minimized, '+' are maximized.
    """
    goal_indices = []
    for i, name in enumerate(header):
        if name.endswith('-') or name.endswith('+'):
            goal_indices.append((i, name[-1]))

    def score_row(row):
        # We want to minimize all goals. 
        # For '-', lower is better. For '+', higher is better (so we negate it).
        scores = []
        for idx, op in goal_indices:
            val = float(row[idx])
            scores.append(val if op == '-' else -val)
        return scores

    # Calculate scores for all rows
    indexed_scores = []
    for i, row in enumerate(rows):
        indexed_scores.append((i, score_row(row)))

    # A row is "unusually well" if it dominates or performs well on all goals.
    # We sort by the sum of normalized scores or simply lexicographical comparison.
    # Here we sort by the sum of the goal values (since all are normalized to [0,1]).
    indexed_scores.sort(key=lambda x: sum(x[1]))

    # Return top 5% (or at least 1)
    limit = max(1, len(rows) // 20)
    return [item[0] for item in indexed_scores[:limit]]

def guess(header, train, test):
    """
    Estimates the first goal column for test rows by finding the 
    average of the first goal column from the top 5% of training rows.
    """
    # Find the index of the first goal column
    goal_idx = -1
    for i, name in enumerate(header):
        if name.endswith('-') or name.endswith('+'):
            goal_idx = i
            break
    
    # Get indices of top-performing rows from step01
    best_indices = remember(header, train)
    
    # Calculate the average of the first goal column for these best rows
    total_val = sum(float(train[i][goal_idx]) for i in best_indices)
    avg_val = total_val / len(best_indices)
    
    # Return the estimate for each test row
    return [avg_val for _ in test]

import csv

def certify(xs, ys):
    """
    Returns True if xs and ys are statistically similar (same distribution),
    False if they differ significantly. Uses a simple bootstrap/permutation 
    logic or a basic difference of means check.
    """
    if not xs or not ys:
        return True
    
    # Calculate means
    mu_x = sum(xs) / len(xs)
    mu_y = sum(ys) / len(ys)
    
    # Calculate standard deviation
    def stdev(data, mean):
        return (sum((x - mean) ** 2 for x in data) / len(data)) ** 0.5
    
    sd_x = stdev(xs, mu_x)
    sd_y = stdev(ys, mu_y)
    
    # If the difference in means is less than the combined standard deviation,
    # we consider them "the same apart from noise".
    combined_sd = (sd_x + sd_y) / 2
    return abs(mu_x - mu_y) < combined_sd

def analyze_team_size_impact(header, rows):
    # Find indices
    team_size_idx = header.index('Team Size')
    cost_idx = header.index('Cost-')
    
    # Split into small (<= 0.5) and big (> 0.5) teams
    small_team_costs = []

        team_size = float(row[team_size_idx])
        cost = float(row[cost_idx])
        if team_size <= 0.5:

        else:
            big_team_costs.append(cost)
            
    is_same = certify(small_team_costs, big_team_costs)

import csv
from step01 import remember

def flow(header, row, seen):
    """
    Decides whether to accept a row based on whether it is better than 
    the current best seen so far.
    """
    # Identify goal indices
    goal_indices = [(i, name[-1]) for i, name in enumerate(header) 
                    if name.endswith('-') or name.endswith('+')]
    
    def get_score(r):
        # Calculate score: lower is better for '-', higher is better for '+'
        # We negate '+' values so that minimizing the sum always works.
        s = 0
        for idx, op in goal_indices:
            val = float(r[idx])
            s += val if op == '-' else -val
        return s

    # If this is the first row, accept it
    if not seen:
        return True
    
    # Calculate score of current row
    current_score = get_score(row)
    
    # Calculate scores of all previously seen rows
    seen_scores = [get_score(r) for r in seen]
    
    # Accept if current row is better than the best seen so far
    # (i.e., lower score than the minimum score seen)
    return current_score < min(seen_scores)

def blame(header, rows):
    """
    Identifies input columns that correlate with high 'Idle-' values.
    We define 'worst' as the bottom 10% of rows based on 'Idle-'.
    """
    idle_idx = header.index('Idle-')
    
    # Sort rows by Idle- (descending, so worst are at the top)
    sorted_rows = sorted(rows, key=lambda r: float(r[idle_idx]), reverse=True)
    worst_rows = sorted_rows[:len(rows) // 10]
    
    # Calculate average of each input column for the worst rows
    # and compare to the global average of the whole dataset.
    input_indices = [i for i, name in enumerate(header) 
                     if not (name.endswith('-') or name.endswith('+'))]
    
    def get_avg(data, idx):
        vals = [float(r[idx]) for r in data]
        return sum(vals) / len(vals)
    
    # We look for inputs where the worst rows have significantly higher values
    # than the global average (assuming higher input values drive higher Idle-)
    diffs = []
    for i in input_indices:
        worst_avg = get_avg(worst_rows, i)
        global_avg = get_avg(rows, i)
        diffs.append((header[i], worst_avg - global_avg))
    
    # Sort by the largest positive difference
    diffs.sort(key=lambda x: x[1], reverse=True)
    
    # Return the top 2 inputs that correlate most with high idle time
    return [d[0] for d in diffs[:2]]

def choose(header, rows, label):
    """
    Selects the best row by sampling 50 rows, labeling them, 
    and returning the index of the one with the best performance.
    """
    # Identify goal indices
    goal_indices = [(i, name[-1]) for i, name in enumerate(header) 
                    if name.endswith('-') or name.endswith('+')]

    def get_score(row_goals):
        # row_goals is a list of values corresponding to goal_indices
        s = 0
        for i, (idx, op) in enumerate(goal_indices):
            val = float(row_goals[i])
            s += val if op == '-' else -val
        return s

    # We have 50 labels. We sample 50 rows evenly across the dataset.
    # If the dataset is smaller than 50, we label all of them.
    n = len(rows)
    indices_to_label = []
    if n <= 50:
        indices_to_label = list(range(n))
    else:
        step = n // 50
        indices_to_label = [i * step for i in range(50)]

    best_idx = -1
    best_score = float('inf')

    for i in indices_to_label:
        # label(i) returns the true goal values for rows[i]
        true_goals = label(i)
        score = get_score(true_goals)
        
        if score < best_score:
            best_score = score
            best_idx = i
            
    return best_idx
