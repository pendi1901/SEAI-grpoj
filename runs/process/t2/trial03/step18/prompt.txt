import csv

def remember(header, rows):
    """
    Identifies rows that perform well across all goal columns.
    Goals ending in '-' are minimized, '+' are maximized.
    """
    goal_indices = []
    for i, name in enumerate(header):
        if name.endswith('-') or name.endswith('+'):
            goal_indices.append((i, name[-1]))

    def score_row(row):
        # Calculate a normalized score for each goal.
        # Since we want to find rows that are "unusually well", 
        # we look for rows that minimize the sum of normalized goal values.
        # For '-', we want low values. For '+', we want high values (so we use 1-val).
        total = 0.0
        for idx, op in goal_indices:
            val = float(row[idx])
            if op == '-':
                total += val
            else:
                total += (1.0 - val)
        return total

    # Score all rows
    scored = []
    for i, row in enumerate(rows):
        scored.append((score_row(row), i))
    
    # Sort by score ascending (lower is better)
    scored.sort()
    
    # Return top 5% (or at least 1)
    limit = max(1, len(rows) // 20)
    return [item[1] for item in scored[:limit]]

import csv
import math

def certify(xs, ys):
    """
    Returns True if xs and ys are statistically similar (same distribution),
    False if they differ significantly. Uses a simple A12 effect size check.
    """
    def a12(lst1, lst2):
        m, n = len(lst1), len(lst2)
        count = 0
        for x in lst1:
            for y in lst2:
                if x > y: count += 1
                elif x == y: count += 0.5
        return count / (m * n)

    # If A12 is between 0.44 and 0.56, they are effectively the same
    score = a12(xs, ys)
    return 0.44 <= score <= 0.56

def analyze_teams(header, rows):
    # Find indices for 'Team Size' and 'Cost-'

    return is_same, small_costs, big_costs

def flow(header, row, seen):
    """
    Decides whether to approve a row based on whether it is better than
    any row seen so far. Uses the scoring logic from step01.
    """
    # Identify goal indices and directions
    goal_indices = []
    for i, name in enumerate(header):
        if name.endswith('-') or name.endswith('+'):
            goal_indices.append((i, name[-1]))

    def score_row(r):
        total = 0.0
        for idx, op in goal_indices:
            val = float(r[idx])
            if op == '-':
                total += val
            else:
                total += (1.0 - val)
        return total

    current_score = score_row(row)
    
    # If this is the first row, we have no baseline, so we keep it
    if not seen:
        return True
    
    # Compare against the best score seen so far
    best_seen_score = min(score_row(r) for r in seen)
    
    # Approve if it's better (lower score) than anything seen before
    return current_score < best_seen_score

    # Sort rows by 'Idle-' descending to find the "worst" ones
    sorted_rows = sorted(rows, key=lambda r: r[idle_idx], reverse=True)
    
    # Take the worst 20%
    worst_rows = sorted_rows[:max(1, len(rows) // 5)]
    
    # Calculate correlation: average value of input in worst rows vs average in all rows
    # We look for inputs that are significantly higher or lower in the worst rows
    def get_avg(data, idx):
        return sum(r[idx] for r in data) / len(data)
    
    overall_avgs = [get_avg(rows, i) for i in input_indices]

    # Return the top 3 inputs that explain the bad idle time
    diffs.sort(key=lambda x: x[0], reverse=True)
    return [name for diff, name in diffs[:3]]

def justify(header, rows, row):

    Estimates the goals for the given row and calculates its percentile rank
    against the provided dataset.
    """

    # We need to pass the full dataset as training data to guess the goals

    # Use step02's logic to estimate the goals.

    input_indices = [i for i, name in enumerate(header) if i not in goal_indices]
    
    estimated_row = list(row)
    for g_idx in goal_indices:
        # Find nearest neighbor in rows based on input_indices
        best_dist = float('inf')
        best_val = 0.0
        for r in rows:
            dist = sum((float(row[i]) - float(r[i]))**2 for i in input_indices)
            if dist < best_dist:
                best_dist = dist
                best_val = r[g_idx]

        for idx in goal_indices:

    all_scores = [get_score(r) for r in rows]
    
    # 3. Calculate percentile: fraction of rows that are worse (have higher scores)

    p = better_than_count / len(rows)

import csv

def fix(header, row, levers):
    """
    Finds the best change to a 'bad' row by testing adjustments 
    to the provided levers to minimize the goal score.
    """
    goal_indices = [i for i, name in enumerate(header) 
                    if name.endswith('-') or name.endswith('+')]

    def score_row(r):
        total = 0.0
        for idx in goal_indices:
            val = float(r[idx])
            if header[idx].endswith('-'):
                total += val
            else:
                total += (1.0 - val)
        return total

    best_change = {}
    best_score = score_row(row)
    found_improvement = False

    # Try adjusting each lever
    for col_name, allowed in levers.items():
        if col_name not in header:
            continue
        idx = header.index(col_name)
        original_val = row[idx]
        
        # For numeric levers, test 10 points in the range
        if isinstance(allowed, tuple) or isinstance(allowed, list):
            low, high = allowed[0], allowed[1]
            for i in range(11):
                test_val = low + (high - low) * (i / 10.0)
                row[idx] = test_val
                current_score = score_row(row)
                if current_score < best_score:
                    best_score = current_score
                    best_change = {col_name: test_val}
                    found_improvement = True
        
        # Reset row
        row[idx] = original_val
        
    return best_change

import csv
import math

def spot(header, rows):
    """
    Identifies rows that are 'outliers' based on the Mahalanobis-like distance
    of their input values from the centroid of all input values.
    """
    input_indices = [i for i, name in enumerate(header) 
                     if not (name.endswith('-') or name.endswith('+'))]
    
    # Extract only input columns
    inputs = [[row[i] for i in input_indices] for row in rows]
    
    # Calculate centroid (mean of each input column)
    num_rows = len(inputs)
    num_cols = len(input_indices)
    means = [sum(col) / num_rows for col in zip(*inputs)]
    
    # Calculate standard deviation for each input column
    stdevs = []
    for j in range(num_cols):
        variance = sum((inputs[i][j] - means[j])**2 for i in range(num_rows)) / num_rows
        stdevs.append(math.sqrt(variance) or 1.0) # Avoid division by zero
        
    # Calculate distance for each row (Z-score sum of squares)
    scores = []
    for i, row_inputs in enumerate(inputs):
        dist = 0.0
        for j in range(num_cols):
            z_score = (row_inputs[j] - means[j]) / stdevs[j]
            dist += z_score**2
        scores.append((dist, i))
        
    # Sort by distance descending (most suspicious first)
    scores.sort(key=lambda x: x[0], reverse=True)
    
    # Return indices of the top 5% most suspicious rows
    limit = max(1, num_rows // 20)
    return [item[1] for item in scores[:limit]]

import csv
import random
import step01
import step07

def race(header, rows, label):
    """
    Compares three strategies: random, greedy (best of sample), 
    and a local search variant.
    """
    goal_indices = [i for i, name in enumerate(header) 
                    if name.endswith('-') or name.endswith('+')]

    def get_score(row):
        total = 0.0
        for idx in goal_indices:
            val = float(row[idx])
            if header[idx].endswith('-'):
                total += val
            else:
                total += (1.0 - val)
        return total

    budget = 50
    labeled = {}

    def get_labeled_row(i):
        if i not in labeled:
            if len(labeled) >= budget:
                raise Exception("Budget exceeded")
            goals = label(i)
            row = list(rows[i])
            for g_idx, val in zip(goal_indices, goals):
                row[g_idx] = val
            labeled[i] = row
        return labeled[i]

    # Strategy 1: Random
    def strategy_random():
        idx = random.choice(range(len(rows)))
        return idx, get_score(get_labeled_row(idx))

    # Strategy 2: Greedy (Sample 20)
    def strategy_greedy():
        samples = random.sample(range(len(rows)), 20)
        best_idx = -1
        best_s = float('inf')
        for i in samples:
            s = get_score(get_labeled_row(i))
            if s < best_s:
                best_s = s
                best_idx = i
        return best_idx, best_s

    # Strategy 3: Hill-Climb (Sample 10, then 10 neighbors)
    def strategy_hill():
        curr = random.choice(range(len(rows)))
        best_s = get_score(get_labeled_row(curr))

        simulated_t_row = list(t_row)
        if dyn_idx != -1 and simulated_t_row[dyn_idx] != "?":

                best_val = r_row[goal_idx]

    return predictions

import csv
import step04

def flow(header, row, seen):
    """
    Warns the customer by returning True if the current row is 'bad' 
    (i.e., significantly worse than the baseline of previous rows).
    
    We define 'bad' as having a score worse than the average score of 
    all previously seen rows.
    """
    goal_indices = []
    for i, name in enumerate(header):
        if name.endswith('-') or name.endswith('+'):
            goal_indices.append((i, name[-1]))

    def score_row(r):
        total = 0.0
        for idx, op in goal_indices:
            val = float(r[idx])
            if op == '-':
                total += val
            else:
                total += (1.0 - val)
        return total

    if not seen:
        return False

    # Calculate average score of all previously seen rows
    avg_score = sum(score_row(r) for r in seen) / len(seen)
    current_score = score_row(row)

    # If the current row is worse (higher score) than the average, 
    # we return True to flag it as "going bad".
    return current_score > avg_score

import csv
import step05

def blame(header, rows):
    """
    Identifies the input columns that matter most to 'Cost-' by comparing
    the worst rows to the overall population.
    
    We reuse the logic from step05, but specifically target 'Cost-' as the
    goal to minimize.
    """
    # Identify goal indices and input indices
    goal_indices = {name: i for i, name in enumerate(header) if name.endswith('-') or name.endswith('+')}
    input_indices = [i for i, name in enumerate(header) if not (name.endswith('-') or name.endswith('+'))]
    
    cost_idx = goal_indices['Cost-']
    
    # Sort rows by 'Cost-' descending to find the "worst" (most expensive) ones
    sorted_rows = sorted(rows, key=lambda r: r[cost_idx], reverse=True)
    
    # Take the worst 20%
    worst_rows = sorted_rows[:max(1, len(rows) // 5)]
    
    # Calculate average value of input in worst rows vs average in all rows
    def get_avg(data, idx):
        return sum(r[idx] for r in data) / len(data)
    
    overall_avgs = [get_avg(rows, i) for i in input_indices]
    worst_avgs = [get_avg(worst_rows, i) for i in input_indices]
    
    # Calculate absolute difference to find which inputs deviate most
    diffs = []
    for i, idx in enumerate(input_indices):
        diff = abs(worst_avgs[i] - overall_avgs[i])
        diffs.append((diff, header[idx]))
    
    # Return the top 2 inputs that explain the high cost
    diffs.sort(key=lambda x: x[0], reverse=True)
    return [name for diff, name in diffs[:2]]
