import csv

def remember(header, rows):
    """
    Identifies rows that perform well across all goal columns.
    Goals ending in '-' are minimized, '+' are maximized.
    """
    goal_indices = []
    for i, name in enumerate(header):
        if name.endswith('-') or name.endswith('+'):
            goal_indices.append((i, name[-1]))

    def score_row(row):
        # Calculate a normalized score for each goal.
        # Since we want to find rows that are "unusually well", 
        # we look for rows that minimize the sum of normalized goal values.
        # For '-', we want low values. For '+', we want high values (so we use 1-val).
        total = 0.0
        for idx, op in goal_indices:
            val = float(row[idx])
            if op == '-':
                total += val
            else:
                total += (1.0 - val)
        return total

    # Score all rows
    scored = []
    for i, row in enumerate(rows):
        scored.append((score_row(row), i))
    
    # Sort by score ascending (lower is better)
    scored.sort()
    
    # Return top 5% (or at least 1)
    limit = max(1, len(rows) // 20)
    return [item[1] for item in scored[:limit]]

import csv
import math

def certify(xs, ys):
    """
    Returns True if xs and ys are statistically similar (same distribution),
    False if they differ significantly. Uses a simple A12 effect size check.
    """
    def a12(lst1, lst2):
        m, n = len(lst1), len(lst2)
        count = 0
        for x in lst1:
            for y in lst2:
                if x > y: count += 1
                elif x == y: count += 0.5
        return count / (m * n)

    # If A12 is between 0.44 and 0.56, they are effectively the same
    score = a12(xs, ys)
    return 0.44 <= score <= 0.56

def analyze_teams(header, rows):
    # Find indices for 'Team Size' and 'Cost-'
    team_idx = header.index('Team Size')
    cost_idx = header.index('Cost-')
    
    # Split into small and big teams based on median team size
    team_sizes = [row[team_idx] for row in rows]
    median = sum(team_sizes) / len(team_sizes)
    
    small_costs = [row[cost_idx] for row in rows if row[team_idx] < median]
    big_costs = [row[cost_idx] for row in rows if row[team_idx] >= median]
    
    is_same = certify(small_costs, big_costs)
    
    return is_same, small_costs, big_costs

import csv
import step01

def flow(header, row, seen):
    """
    Decides whether to approve a row based on whether it is better than
    any row seen so far. Uses the scoring logic from step01.
    """
    # Identify goal indices and directions
    goal_indices = []
    for i, name in enumerate(header):
        if name.endswith('-') or name.endswith('+'):
            goal_indices.append((i, name[-1]))

    def score_row(r):
        total = 0.0
        for idx, op in goal_indices:
            val = float(r[idx])
            if op == '-':
                total += val
            else:
                total += (1.0 - val)
        return total

    current_score = score_row(row)
    
    # If this is the first row, we have no baseline, so we keep it
    if not seen:
        return True
    
    # Compare against the best score seen so far
    best_seen_score = min(score_row(r) for r in seen)
    
    # Approve if it's better (lower score) than anything seen before
    return current_score < best_seen_score

def blame(header, rows):
    """
    Identifies input columns that correlate with high 'Idle-' values.
    'Idle-' is a goal to minimize, so high values are 'bad'.
    """
    # Identify goal indices and input indices
    goal_indices = {name: i for i, name in enumerate(header) if name.endswith('-') or name.endswith('+')}
    input_indices = [i for i, name in enumerate(header) if not (name.endswith('-') or name.endswith('+'))]
    
    idle_idx = goal_indices['Idle-']
    
    # Sort rows by 'Idle-' descending to find the "worst" ones
    sorted_rows = sorted(rows, key=lambda r: r[idle_idx], reverse=True)
    
    # Take the worst 20%
    worst_rows = sorted_rows[:max(1, len(rows) // 5)]
    
    # Calculate correlation: average value of input in worst rows vs average in all rows
    # We look for inputs that are significantly higher or lower in the worst rows
    def get_avg(data, idx):
        return sum(r[idx] for r in data) / len(data)
    
    overall_avgs = [get_avg(rows, i) for i in input_indices]
    worst_avgs = [get_avg(worst_rows, i) for i in input_indices]
    
    # Calculate absolute difference to find which inputs deviate most
    diffs = []
    for i, idx in enumerate(input_indices):
        diff = abs(worst_avgs[i] - overall_avgs[i])
        diffs.append((diff, header[idx]))
    
    # Return the top 3 inputs that explain the bad idle time
    diffs.sort(key=lambda x: x[0], reverse=True)
    return [name for diff, name in diffs[:3]]

def justify(header, rows, row):
    """
    Estimates the goals for the given row and calculates its percentile rank
    against the provided dataset.
    """
    # 1. Estimate the missing goal values in 'row' using the nearest neighbor approach
    # We need to pass the full dataset as training data to guess the goals
    goal_indices = [i for i, name in enumerate(header) if name.endswith('-') or name.endswith('+')]
    
    # Use step02's logic to estimate the goals. 
    # Since step02.guess only estimates the first goal, we extend the logic 
    # to estimate all goal columns for the specific row.
    input_indices = [i for i, name in enumerate(header) if i not in goal_indices]
    
    estimated_row = list(row)
    for g_idx in goal_indices:
        # Find nearest neighbor in rows based on input_indices
        best_dist = float('inf')
        best_val = 0.0
        for r in rows:
            dist = sum((float(row[i]) - float(r[i]))**2 for i in input_indices)
            if dist < best_dist:
                best_dist = dist
                best_val = r[g_idx]
        estimated_row[g_idx] = best_val

    # 2. Score the estimated row and all rows in the dataset
    def get_score(r):
        total = 0.0
        for idx in goal_indices:
            val = float(r[idx])
            if header[idx].endswith('-'):
                total += val
            else:
                total += (1.0 - val)
        return total

    target_score = get_score(estimated_row)
    all_scores = [get_score(r) for r in rows]
    
    # 3. Calculate percentile: fraction of rows that are worse (have higher scores)
    # A row is better if its score is lower.
    better_than_count = sum(1 for s in all_scores if s > target_score)
    p = better_than_count / len(rows)
