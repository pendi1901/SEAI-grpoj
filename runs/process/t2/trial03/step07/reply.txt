import csv
import random
import step01

def choose(header, rows, label):
    """
    Finds the best row by sampling 50 rows, labeling them,
    and returning the index of the one with the best score.
    """
    # Identify goal indices
    goal_indices = [i for i, name in enumerate(header) 
                    if name.endswith('-') or name.endswith('+')]
    
    # We can label at most 50 rows.
    # We pick 50 random indices to sample the search space.
    indices_to_sample = random.sample(range(len(rows)), min(50, len(rows)))
    
    best_idx = -1
    best_score = float('inf')
    
    for i in indices_to_sample:
        # Get true goal values
        true_goals = label(i)
        
        # Create a temporary row with true goals to score it
        temp_row = list(rows[i])
        for g_idx, val in zip(goal_indices, true_goals):
            temp_row[g_idx] = val
            
        # Use step01 logic to score
        score = 0.0
        for idx in goal_indices:
            val = float(temp_row[idx])
            if header[idx].endswith('-'):
                score += val
            else:
                score += (1.0 - val)
        
        if score < best_score:
            best_score = score
            best_idx = i
            
    return best_idx

if __name__ == "__main__":
    filename = "pom3a.n1000.train.csv"
    try:
        with open(filename, mode='r', newline='') as f:
            reader = csv.reader(f)
            header = next(reader)
            rows = []
            true_data = []
            for row in reader:
                # Store full row for labeling
                full_row = [float(val) if val.replace('.','',1).isdigit() or '-' in val else val for val in row]
                true_data.append(full_row)
                
                # Masked row for the function
                masked = [val if not (header[i].endswith('-') or header[i].endswith('+')) else "?" 
                          for i, val in enumerate(full_row)]
                rows.append(masked)
        
        def label(i):
            # Return only the goal values for the requested row
            return [true_data[i][idx] for idx, name in enumerate(header) 
                    if name.endswith('-') or name.endswith('+')]
        
        best_index = choose(header, rows, label)
        
        print(f"Search complete. The best row found is at index {best_index}.")
        print("This row represents the optimal configuration among the 50 sampled setups.")
        
    except FileNotFoundError:
        print(f"Error: {filename} not found.")
