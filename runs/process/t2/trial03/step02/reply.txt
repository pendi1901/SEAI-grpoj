import csv
import step01

def guess(header, train, test):
    """
    Estimates the first goal column for each test row by finding the 
    nearest neighbor in the training set based on input columns.
    """
    # Identify input columns (those that are not goals)
    input_indices = [i for i, name in enumerate(header) 
                     if not (name.endswith('-') or name.endswith('+'))]
    
    # Identify the first goal column index
    goal_idx = next(i for i, name in enumerate(header) 
                    if name.endswith('-') or name.endswith('+'))

    predictions = []
    for t_row in test:
        # Find the row in train that is closest to t_row based on inputs
        best_dist = float('inf')
        best_val = 0.0
        
        for r_row in train:
            dist = 0.0
            for i in input_indices:
                # Simple Euclidean distance for numeric inputs
                dist += (float(t_row[i]) - float(r_row[i]))**2
            
            if dist < best_dist:
                best_dist = dist
                best_val = r_row[goal_idx]
        
        predictions.append(float(best_val))
        
    return predictions

if __name__ == "__main__":
    filename = "pom3a.n1000.train.csv"
    try:
        with open(filename, mode='r', newline='') as f:
            reader = csv.reader(f)
            header = next(reader)
            rows = []
            for row in reader:
                processed_row = [float(val) if val.replace('.','',1).isdigit() or '-' in val else val for val in row]
                rows.append(processed_row)
        
        # Split data: use 80% for training, 20% for testing
        split = int(len(rows) * 0.8)
        train_data = rows[:split]
        test_data = rows[split:]
        
        # Prepare test rows by masking the goals
        masked_test = []
        for row in test_data:
            new_row = [val if not (header[i].endswith('-') or header[i].endswith('+')) else "?" for i, val in enumerate(row)]
            masked_test.append(new_row)
            
        results = guess(header, train_data, masked_test)
        
        print("Estimation complete.")
        print(f"Estimated cost for the first 5 test setups: {[round(x, 4) for x in results[:5]]}")
        
    except FileNotFoundError:
        print(f"Error: {filename} not found.")
        exit(1)
