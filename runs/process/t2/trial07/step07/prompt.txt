import csv

def remember(header, rows):
    """
    Identifies rows that perform well across all goal columns.
    Goals ending in '-' are minimized, '+' are maximized.
    """
    goal_indices = []
    for i, name in enumerate(header):
        if name.endswith('-') or name.endswith('+'):
            goal_indices.append((i, name[-1]))

    def score_row(row):
        # Calculate a normalized score for each goal
        # Since all data is numeric and normalized [0, 1], 
        # we can simply sum the values (inverted for '-')
        total = 0.0
        for idx, op in goal_indices:
            val = float(row[idx])
            if op == '-':
                total += val  # Lower is better
            else:
                total += (1.0 - val)  # Higher is better
        return total

    # Score all rows
    scored = []
    for i, row in enumerate(rows):
        scored.append((score_row(row), i))

    # Sort by score (ascending, as we want lower total penalty)
    scored.sort()

    # Return top 5% (or at least 1)
    limit = max(1, len(rows) // 20)
    return [item[1] for item in scored[:limit]]

import csv
import math

def certify(xs, ys):
    """
    Returns True if xs and ys are statistically similar (same population),
    False if they differ significantly. Uses a simple t-test approach.
    """
    n1, n2 = len(xs), len(ys)
    if n1 < 2 or n2 < 2:
        return True
    
    mu1, mu2 = sum(xs) / n1, sum(ys) / n2
    var1 = sum((x - mu1) ** 2 for x in xs) / (n1 - 1)
    var2 = sum((y - mu2) ** 2 for y in ys) / (n2 - 1)
    
    # Pooled standard deviation
    pooled_std = math.sqrt(((n1 - 1) * var1 + (n2 - 1) * var2) / (n1 + n2 - 2))
    
    if pooled_std == 0:
        return mu1 == mu2
    
    t_stat = (mu1 - mu2) / (pooled_std * math.sqrt(1/n1 + 1/n2))
    # For a 95% confidence level, critical t-value is approx 1.96
    return abs(t_stat) < 1.96

import csv

def blame(header, rows):
    """
    Identifies input columns that correlate with high 'Idle-' values.
    We compare the top 25% worst rows (by 'Idle-') against the rest.
    """
    idle_idx = header.index('Idle-')
    input_indices = [i for i, name in enumerate(header) 
                     if not (name.endswith('-') or name.endswith('+'))]
    
    # Sort rows by 'Idle-' descending to find the worst
    sorted_rows = sorted(rows, key=lambda r: r[idle_idx], reverse=True)
    worst = sorted_rows[:len(rows) // 4]
    rest = sorted_rows[len(rows) // 4:]
    
    def get_avg(data, idx):
        vals = [r[idx] for r in data if isinstance(r[idx], (int, float))]
        return sum(vals) / len(vals) if vals else 0

    # Find inputs where the average in 'worst' is significantly higher than in 'rest'
    # We look for the largest delta in normalized space
    deltas = []
    for i in input_indices:
        avg_worst = get_avg(worst, i)
        avg_rest = get_avg(rest, i)
        deltas.append((avg_worst - avg_rest, header[i]))
    
    # Return the top 2 inputs that contribute most to the 'Idle-' increase
    deltas.sort(reverse=True)
    return [name for delta, name in deltas[:2]]

import csv
import step01

def estimate_goals(header, rows, row):
    """
    Estimates the goal values for a row with '?' by using the average 
    of the best-performing rows from the dataset.
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith(('-', '+'))]
    best_indices = step01.remember(header, rows)
    
    estimated_row = list(row)
    for idx in goal_indices:
        avg_val = sum(rows[i][idx] for i in best_indices) / len(best_indices)
        estimated_row[idx] = avg_val
    return estimated_row

def justify(header, rows, row):
    """
    Returns the percentile rank of the row and a justification string.
    """
    # 1. Estimate the missing goal values
    est_row = estimate_goals(header, rows, row)
    
    # 2. Score the estimated row
    goal_indices = []
    for i, name in enumerate(header):
        if name.endswith('-') or name.endswith('+'):
            goal_indices.append((i, name[-1]))

    def score(r):
        total = 0.0
        for idx, op in goal_indices:
            val = float(r[idx])
            total += val if op == '-' else (1.0 - val)
        return total

    est_score = score(est_row)
    
    # 3. Compare against all rows
    better_count = 0
    for r in rows:
        if score(r) < est_score:
            better_count += 1
            
    p = better_count / len(rows)
    reason = f"This setup is better than {int(p * 100)}% of all known configurations based on its projected performance."
    
    return p, reason
