import csv

def remember(header, rows):
    """
    Identifies rows that perform well across all goal columns.
    Goals ending in '-' are minimized, '+' are maximized.
    """
    goal_indices = []
    for i, name in enumerate(header):
        if name.endswith('-') or name.endswith('+'):
            goal_indices.append((i, name[-1]))

    def score_row(row):
        # Calculate a normalized score for each goal
        # Since all data is numeric and normalized [0, 1], 
        # we can simply sum the values (inverted for '-')
        total = 0.0
        for idx, op in goal_indices:
            val = float(row[idx])
            if op == '-':
                total += val  # Lower is better
            else:
                total += (1.0 - val)  # Higher is better
        return total

    # Score all rows
    scored = []
    for i, row in enumerate(rows):
        scored.append((score_row(row), i))

    # Sort by score (ascending, as we want lower total penalty)
    scored.sort()

    # Return top 5% (or at least 1)
    limit = max(1, len(rows) // 20)
    return [item[1] for item in scored[:limit]]

import csv
import math

def certify(xs, ys):
    """
    Returns True if xs and ys are statistically similar (same population),
    False if they differ significantly. Uses a simple t-test approach.
    """
    n1, n2 = len(xs), len(ys)
    if n1 < 2 or n2 < 2:
        return True
    
    mu1, mu2 = sum(xs) / n1, sum(ys) / n2
    var1 = sum((x - mu1) ** 2 for x in xs) / (n1 - 1)
    var2 = sum((y - mu2) ** 2 for y in ys) / (n2 - 1)
    
    # Pooled standard deviation
    pooled_std = math.sqrt(((n1 - 1) * var1 + (n2 - 1) * var2) / (n1 + n2 - 2))
    
    if pooled_std == 0:
        return mu1 == mu2
    
    t_stat = (mu1 - mu2) / (pooled_std * math.sqrt(1/n1 + 1/n2))
    # For a 95% confidence level, critical t-value is approx 1.96
    return abs(t_stat) < 1.96

    sorted_rows = sorted(rows, key=lambda r: r[idle_idx], reverse=True)
    worst = sorted_rows[:len(rows) // 4]
    rest = sorted_rows[len(rows) // 4:]
    
    def get_avg(data, idx):
        vals = [r[idx] for r in data if isinstance(r[idx], (int, float))]
        return sum(vals) / len(vals) if vals else 0

    # We look for the largest delta in normalized space
    deltas = []

    deltas.sort(reverse=True)

import csv
import math

def spot(header, rows):
    """
    Identifies suspicious rows by calculating the Mahalanobis-like distance
    of each row from the centroid of the dataset. Rows with high distance
    are considered outliers (potentially mistyped).
    """
    # Only consider input columns (not goals)
    input_indices = [i for i, name in enumerate(header) 
                     if not (name.endswith('-') or name.endswith('+'))]
    
    # Calculate mean and standard deviation for each input column
    n = len(rows)
    means = []
    stds = []
    for i in input_indices:
        vals = [row[i] for row in rows]
        mu = sum(vals) / n
        sigma = math.sqrt(sum((x - mu)**2 for x in vals) / n)
        means.append(mu)
        stds.append(sigma if sigma > 0 else 1.0)
    
    # Calculate distance for each row
    distances = []
    for idx, row in enumerate(rows):
        dist = 0
        for j, col_idx in enumerate(input_indices):
            z_score = (row[col_idx] - means[j]) / stds[j]
            dist += z_score**2
        distances.append((math.sqrt(dist), idx))
    
    # Sort by distance descending (most suspicious first)
    distances.sort(key=lambda x: x[0], reverse=True)
    
    return [idx for dist, idx in distances]

def race(header, rows, label):
    """
    Compares three strategies: random, greedy (first 50), and 
    a simple hill-climber, sharing a 50-label budget.
    """
    budget = 50
    labeled = {}
    goal_indices = [i for i, name in enumerate(header) if name.endswith(('-', '+'))]

    def get_score(row):
        score = 0.0
        for idx, name in enumerate(header):
            if name.endswith('-'):
                score += row[idx]
            elif name.endswith('+'):
                score += (1.0 - row[idx])
        return score

    def get_true_row(i):
        if i not in labeled:
            if len(labeled) >= budget:
                return None
            true_vals = label(i)
            for j, g_idx in enumerate(goal_indices):
                rows[i][g_idx] = true_vals[j]
            labeled[i] = rows[i]
        return labeled[i]

    # Strategy 1: Random
    def strategy_random():
        best_idx = -1
        best_score = float('inf')
        for _ in range(16): # 16 trials
            i = random.randint(0, len(rows) - 1)
            row = get_true_row(i)
            if row:
                s = get_score(row)
                if s < best_score:
                    best_score, best_idx = s, i
        return best_idx, best_score

    # Strategy 2: First-N (Greedy start)
    def strategy_first():
        best_idx = -1
        best_score = float('inf')
        for i in range(17): # 17 trials
            row = get_true_row(i)
            if row:
                s = get_score(row)
                if s < best_score:
                    best_score, best_idx = s, i
        return best_idx, best_score

    # Strategy 3: Hill-Climber (Local search)
    def strategy_hill():
        best_idx = -1
        best_score = float('inf')
        curr = random.randint(0, len(rows) - 1)
        for _ in range(17): # 17 trials
            row = get_true_row(curr)
            if not row: break
            s = get_score(row)
            if s < best_score:
                best_score, best_idx = s, curr
            # Move to a random neighbor
            curr = (curr + random.randint(1, 10)) % len(rows)
        return best_idx, best_score

    results = {
        "random": strategy_random(),
        "first": strategy_first(),
        "hill": strategy_hill()
    }

    winner = min(results, key=lambda k: results[k][1])
    return {"winner": winner, "row": results[winner][0], "tried": list(results.keys())}
