import csv
import math

def certify(xs, ys):
    """
    Returns True if the distributions of xs and ys are statistically 
    indistinguishable (same apart from noise), False otherwise.
    Uses a simple A12 effect size test (Vargha-Delaney).
    """
    def a12(lst1, lst2):
        m, n = len(lst1), len(lst2)
        lst1 = sorted(lst1)
        lst2 = sorted(lst2)
        r1 = 0
        for x in lst1:
            for y in lst2:
                if x > y: r1 += 1
                elif x == y: r1 += 0.5
        return abs(0.5 - (r1 / (m * n))) > 0.147

    return not a12(xs, ys)

def get_data():
    filename = "pom3a.n1000.train.csv"
    with open(filename, mode='r', encoding='utf-8') as f:
        reader = csv.reader(f)
        header = next(reader)
        rows = [[float(x) for x in row] for row in reader]
    return header, rows

if __name__ == "__main__":
    header, rows = get_data()
    
    # Identify Team Size index (input) and Cost- index (goal)
    team_idx = header.index('Team Size')
    cost_idx = header.index('Cost-')
    
    # Split into small teams (bottom 50%) and big teams (top 50%)
    # Based on the range [0, 1] for Team Size
    small_teams = [r[cost_idx] for r in rows if r[team_idx] < 0.5]
    big_teams = [r[cost_idx] for r in rows if r[team_idx] >= 0.5]
    
    is_same = certify(small_teams, big_teams)
    
    print("Customer Inquiry: Are small-team setups cheaper than big-team ones?")
    print(f"Small team average cost: {sum(small_teams)/len(small_teams):.4f}")
    print(f"Big team average cost: {sum(big_teams)/len(big_teams):.4f}")
    
    if not is_same:
        print("Conclusion: The difference is statistically significant. It is not just luck.")
    else:
        print("Conclusion: The difference is not statistically significant. It appears to be luck.")
