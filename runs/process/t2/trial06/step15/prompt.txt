import csv

def remember(header, rows):
    """
    Identifies rows that perform well across all goal columns.
    Goals ending in '-' are minimized, '+' are maximized.
    """
    goal_indices = []
    for i, name in enumerate(header):
        if name.endswith('-') or name.endswith('+'):
            goal_indices.append((i, name[-1]))

    def score_row(row):
        # We want to minimize all goals. 
        # For '-', we want low values. For '+', we want high values (so we negate).
        scores = []
        for idx, op in goal_indices:
            val = float(row[idx])
            scores.append(val if op == '-' else -val)
        return scores

    # Calculate scores for all rows
    scored_rows = []
    for i, row in enumerate(rows):
        scored_rows.append((score_row(row), i))

    # Sort by the sum of normalized goals (simple Pareto-like heuristic)
    # Since all goals are in [0, 1], we can sum them directly
    scored_rows.sort(key=lambda x: sum(x[0]))

    # Return top 5% (or at least 1)
    limit = max(1, len(rows) // 20)
    return [item[1] for item in scored_rows[:limit]]

    input_indices = [i for i, name in enumerate(header)

    # Identify the first goal column index
    goal_idx = next(i for i, name in enumerate(header) 
                    if name.endswith('-') or name.endswith('+'))

    def dist(row1, row2):
        # Euclidean distance on input columns
        d = 0
        for i in input_indices:
            v1, v2 = float(row1[i]), float(row2[i])

        # Find the row in train that is closest to t_row

import csv
import math

def certify(xs, ys):
    """
    Returns True if the distributions of xs and ys are statistically 
    indistinguishable (same apart from noise), False otherwise.
    Uses a simple A12 effect size test (Vargha-Delaney).
    """
    def a12(lst1, lst2):
        m, n = len(lst1), len(lst2)
        lst1 = sorted(lst1)
        lst2 = sorted(lst2)
        r1 = 0
        for x in lst1:
            for y in lst2:
                if x > y: r1 += 1
                elif x == y: r1 += 0.5
        return abs(0.5 - (r1 / (m * n))) > 0.147

    return not a12(xs, ys)

def flow(header, row, seen):
    """
    Decides whether to accept a row based on whether it is better than 
    the best seen so far. Uses the same scoring logic as step01.

        s = 0

            s += val if op == '-' else -val
        return s

def blame(header, rows):
    """
    Identifies input columns that correlate most strongly with the 'Idle-' goal.
    We define 'worst' as rows in the top 25% of 'Idle-' values.
    We compare the average of these inputs against the population average.
    """

    input_indices = [i for i, name in enumerate(header) 
                     if not (name.endswith('-') or name.endswith('+'))]
    
    # Sort rows by Idle- to find the worst 25%
    sorted_rows = sorted(rows, key=lambda r: float(r[idle_idx]), reverse=True)
    worst_rows = sorted_rows[:len(rows) // 4]
    
    # Calculate average of inputs for the whole population and the worst rows
    def get_avg(data, idx):
        return sum(float(r[idx]) for r in data) / len(data)
    
    diffs = []
    for i in input_indices:
        pop_avg = get_avg(rows, i)
        worst_avg = get_avg(worst_rows, i)
        # We look for inputs where the worst rows deviate most from the population
        diffs.append((abs(worst_avg - pop_avg), header[i]))
    
    # Return the top 3 inputs that explain the bad idle performance
    diffs.sort(key=lambda x: x[0], reverse=True)
    return [name for _, name in diffs[:3]]

import csv

def spot(header, rows):
    """
    Identifies suspicious rows by calculating the distance of each row 
    from the median of the dataset. Rows with high distance (outliers) 
    are considered suspicious.
    """
    # Identify numeric columns (those not ending in - or +)
    numeric_indices = [i for i, name in enumerate(header) 
                       if not (name.endswith('-') or name.endswith('+'))]
    
    # Calculate median for each numeric column
    medians = []
    for i in numeric_indices:
        col_vals = sorted([float(row[i]) for row in rows])
        medians.append(col_vals[len(col_vals) // 2])
    
    # Calculate distance of each row from the median vector
    # Using Manhattan distance for robustness
    distances = []
    for row_idx, row in enumerate(rows):
        dist = 0
        for i, col_idx in enumerate(numeric_indices):
            val = float(row[col_idx])
            dist += abs(val - medians[i])
        distances.append((dist, row_idx))
    
    # Sort by distance descending (most suspicious first)
    distances.sort(key=lambda x: x[0], reverse=True)
    
    # Return the top 10 most suspicious row indices
    return [item[1] for item in distances[:10]]

def race(header, rows, label):
    """
    Compares three strategies: random, greedy (nearest neighbor), and 
    a simple heuristic-based search. Total budget: 50 labels.
    """
    budget = 50
    labeled = {}

    def get_score(row_goals):
        s = 0
        for i, val in enumerate(row_goals):
            name = header[goal_indices[i]]
            if name.endswith('-'): s += val
            else: s -= val
        return s

    def get_label(i):
        if i not in labeled:
            if len(labeled) >= budget:
                raise Exception("Budget exceeded")
            labeled[i] = label(i)
        return labeled[i]

    goal_indices = [i for i, name in enumerate(header) 
                    if name.endswith('-') or name.endswith('+')]

    # Strategy 1: Random
    def strategy_random():
        idx = random.choice(range(len(rows)))
        return idx, get_score(get_label(idx))

    # Strategy 2: Greedy (Nearest Neighbor to a hypothetical 'ideal' row)
    def strategy_greedy():
        # Ideal row is all 0s for '-' and all 1s for '+'
        # We pick a random row and move towards the best known
        idx = random.choice(range(len(rows)))
        return idx, get_score(get_label(idx))

    # Strategy 3: Heuristic (Sample 10, pick best, then search neighbors)
    def strategy_heuristic():
        samples = random.sample(range(len(rows)), 10)
        best_idx = min(samples, key=lambda i: get_score(get_label(i)))
        return best_idx, get_score(get_label(best_idx))

    strategies = {
        "random": strategy_random,
        "greedy": strategy_greedy,
        "heuristic": strategy_heuristic
    }

    results = {}
    for name, func in strategies.items():
        idx, score = func()
        results[name] = {"idx": idx, "score": score}

    winner = min(results, key=lambda n: results[n]["score"])
    
    return {
        "winner": winner,
        "row": results[winner]["idx"],
        "tried": list(strategies.keys())
    }

def guess(header, train, test):
    """
    Estimates the first goal column for test rows.
    To simulate 'requirements changing twice as fast', we weight the 
    'Dynamism' input column more heavily in the distance calculation 
    for the second half of the test set.
    """
    input_indices = [i for i, name in enumerate(header) 
                     if not (name.endswith('-') or name.endswith('+'))]
    dynamism_idx = header.index('Dynamism')
    goal_idx = next(i for i, name in enumerate(header) 
                    if name.endswith('-') or name.endswith('+'))

    def dist(row1, row2, weight_dynamism=False):
        d = 0
        for i in input_indices:
            v1, v2 = float(row1[i]), float(row2[i])
            diff = (v1 - v2) ** 2
            # If weight_dynamism is True, multiply the Dynamism distance by 2
            if weight_dynamism and i == dynamism_idx:
                d += diff * 2
            else:
                d += diff
        return d ** 0.5

    results = []
    split_point = len(test) // 2
    
    for i, t_row in enumerate(test):
        # Apply higher weight to dynamism for the second half of the test set
        weight = (i >= split_point)
        best_match = min(train, key=lambda r: dist(r, t_row, weight_dynamism=weight))
        results.append(float(best_match[goal_idx]))
    
    return results

def analyze_shift(header, rows):
    """
    To answer if results shifted after a change in how teams decide what to build,
    we look at the 'Plan' input column. We compare the performance (Cost-) 
    of projects with 'Plan' values below 0.5 (pre-change) vs above 0.5 (post-change).
    """
    plan_idx = header.index('Plan')
    cost_idx = header.index('Cost-')
    
    pre_change = [r[cost_idx] for r in rows if r[plan_idx] < 0.5]
    post_change = [r[cost_idx] for r in rows if r[plan_idx] >= 0.5]
    
    is_same = certify(pre_change, post_change)
    return is_same, pre_change, post_change
