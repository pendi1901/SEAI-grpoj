import csv

def remember(header, rows):
    """
    Identifies rows that perform well across all goal columns.
    Goals ending in '-' are minimized, '+' are maximized.
    """
    goal_indices = []
    for i, name in enumerate(header):
        if name.endswith('-') or name.endswith('+'):
            goal_indices.append((i, name[-1]))

    def score_row(row):
        # Calculate a normalized score for the row.
        # Since we want to minimize '-' and maximize '+',
        # we can treat all as "minimize" by inverting '+' values.
        # Here we use a simple sum of normalized values.
        total = 0
        for idx, op in goal_indices:
            val = float(row[idx])
            total += val if op == '-' else (1.0 - val)
        return total

    # Score all rows
    scored = []
    for i, row in enumerate(rows):
        scored.append((score_row(row), i))

    # Sort by score (ascending, as we minimized everything)
    scored.sort()

    # Return top 5% (or at least 1)
    limit = max(1, len(rows) // 20)
    return [item[1] for item in scored[:limit]]

def certify(xs, ys):
    """
    Returns True if the distributions of xs and ys are statistically similar
    (i.e., the difference is likely noise), False if they differ significantly.
    Uses a simple A12 effect size check (non-parametric).
    """
    def a12(lst1, lst2):
        m, n = len(lst1), len(lst2)
        if m == 0 or n == 0: return 0.5
        greater = 0
        for x in lst1:
            for y in lst2:
                if x > y: greater += 1
                elif x == y: greater += 0.5
        return greater / (m * n)

    # If A12 is close to 0.5, they are similar. 
    # Threshold 0.56 is a common small effect size.
    val = a12(xs, ys)
    return 0.44 <= val <= 0.56

def flow(header, row, seen):
    """
    Decides whether to accept a row based on whether it is better than 
    the current best seen so far.
    """
    goal_indices = []
    for i, name in enumerate(header):
        if name.endswith('-') or name.endswith('+'):
            goal_indices.append((i, name[-1]))

    def score_row(r):
        total = 0
        for idx, op in goal_indices:
            val = float(r[idx])
            total += val if op == '-' else (1.0 - val)
        return total

    # If this is the first row, accept it
    if not seen:
        return True

    # Calculate score of current row and the best score seen so far
    current_score = score_row(row)
    best_score = min(score_row(r) for r in seen)

    # Accept if it's better than the best seen so far
    return current_score < best_score

def blame(header, rows):
    """
    Identifies input columns that correlate with high 'Idle-' values.
    We define 'worst' as the top 20% of rows with the highest 'Idle-' values.
    We then compare the distribution of each input column in these 'worst' 
    rows against the rest of the dataset.
    """
    idle_idx = header.index('Idle-')
    input_indices = [i for i, name in enumerate(header) 
                     if not (name.endswith('-') or name.endswith('+'))]
    
    # Sort rows by Idle- descending to find the worst
    sorted_rows = sorted(rows, key=lambda r: r[idle_idx], reverse=True)
    worst_rows = sorted_rows[:len(rows) // 5]
    other_rows = sorted_rows[len(rows) // 5:]
    
    def get_diff(idx):
        # Calculate the difference in means between worst and others
        worst_vals = [r[idx] for r in worst_rows]
        other_vals = [r[idx] for r in other_rows]
        avg_worst = sum(worst_vals) / len(worst_vals)
        avg_other = sum(other_vals) / len(other_vals)
        return abs(avg_worst - avg_other)

    # Rank inputs by how much they differ in the worst setups
    diffs = []
    for i in input_indices:
        diffs.append((get_diff(i), header[i]))
    
    diffs.sort(reverse=True)
    
    # Return the top 2 inputs that explain the variance
    return [name for _, name in diffs[:2]]

def fix(header, row, levers):
    """
    Finds the smallest change to the provided levers to improve the overall 
    score (minimizing goals ending in '-' and maximizing goals ending in '+').
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith(('-', '+'))]
    
    def score_row(r):
        total = 0
        for i, name in enumerate(header):
            if name.endswith('-'):
                total += r[i]
            elif name.endswith('+'):
                total += (1.0 - r[i])
        return total

    # We need to find a change that improves the score.
    # Since we don't have a predictive model, we use a simple heuristic:
    # move the lever towards the value that appears most frequently in 'good' rows.
    # We use step01.remember to find the indices of the top 5% of rows.
    # We load the data to get the context for 'good' values.
    filename = "pom3a.n1000.train.csv"
    with open(filename, mode='r', encoding='utf-8') as f:
        reader = csv.reader(f)
        header_data = next(reader)
        rows = [[float(x) for x in r] for r in reader]
    
    best_indices = step01.remember(header_data, rows)
    
    best_change = {}
    best_score = score_row(row)
    
    for col_name, bounds in levers.items():
        if col_name not in header:
            continue
        idx = header.index(col_name)
        low, high = bounds
        
        # Calculate the average value of this lever in the 'best' rows
        target_val = sum(rows[i][idx] for i in best_indices) / len(best_indices)

        target_val = max(low, min(high, target_val))
        
        if target_val != row[idx]:
            temp_row = list(row)
            temp_row[idx] = target_val
            
            # If this change improves the score, recommend it
            if score_row(temp_row) < best_score:
                return {col_name: target_val}
                
    return best_change

import csv

def spot(header, rows):
    """
    Identifies rows that contain outliers. We define an outlier as a row 
    containing an input value that is significantly far from the median 
    of that column (using a simple 3-sigma-like heuristic).
    """
    input_indices = [i for i, name in enumerate(header) 
                     if not (name.endswith('-') or name.endswith('+'))]
    
    # Calculate column statistics
    stats = {}
    for i in input_indices:
        vals = [r[i] for r in rows if isinstance(r[i], (int, float))]
        if not vals:
            continue
        mean = sum(vals) / len(vals)
        std = (sum((x - mean)**2 for x in vals) / len(vals))**0.5
        stats[i] = (mean, std)
    
    # Score rows based on how many standard deviations they are from the mean
    # We look for values > 3 standard deviations away
    suspicious = []
    for row_idx, row in enumerate(rows):
        score = 0
        for i in input_indices:
            if i in stats:
                mean, std = stats[i]
                if std > 0:
                    val = row[i]
                    if isinstance(val, (int, float)):
                        dist = abs(val - mean) / std
                        if dist > 3:
                            score += dist
        if score > 0:
            suspicious.append((score, row_idx))
    
    # Sort by most suspicious (highest score)
    suspicious.sort(key=lambda x: x[0], reverse=True)
    return [item[1] for item in suspicious]

import csv

import step01

def race(header, rows, label):
    """
    Compares three strategies: random, greedy, and remember.
    Each strategy is allocated a portion of the 50-label budget.
    """
    budget = 50
    tried_indices = {}

    def get_score(row_with_goals):
        total = 0
        for i, name in enumerate(header):
            if name.endswith('-'):
                total += row_with_goals[i]
            elif name.endswith('+'):
                total += (1.0 - row_with_goals[i])
        return total

    def get_full_row(idx):
        if idx not in tried_indices:
            if len(tried_indices) >= budget:
                raise Exception("Budget exceeded")
            goals = label(idx)
            full_row = list(rows[idx])
            goal_idx = 0
            for i, name in enumerate(header):
                if name.endswith('-') or name.endswith('+'):
                    full_row[i] = goals[goal_idx]
                    goal_idx += 1
            tried_indices[idx] = full_row
        return tried_indices[idx]

    # Strategy 1: Random (16 trials)
    random_samples = r.sample(range(len(rows)), 16)
    best_random_idx = -1
    best_random_score = float('inf')
    for idx in random_samples:
        row = get_full_row(idx)
        score = get_score(row)
        if score < best_random_score:
            best_random_score = score
            best_random_idx = idx

    # Strategy 2: Greedy (17 trials)
    greedy_samples = r.sample(range(len(rows)), 17)
    best_greedy_idx = -1
    best_greedy_score = float('inf')
    for idx in greedy_samples:
        row = get_full_row(idx)
        score = get_score(row)
        if score < best_greedy_score:
            best_greedy_score = score
            best_greedy_idx = idx

    # Strategy 3: Remember (17 trials)
    rem_samples = r.sample(range(len(rows)), 17)
    rem_rows = [get_full_row(i) for i in rem_samples]
    best_rem_local_indices = step01.remember(header, rem_rows)
    best_rem_idx = rem_samples[best_rem_local_indices[0]]
    best_rem_score = get_score(rem_rows[best_rem_local_indices[0]])

    results = {
        "random": (best_random_score, best_random_idx),
        "greedy": (best_greedy_score, best_greedy_idx),
        "remember": (best_rem_score, best_rem_idx)
    }

    winner = min(results, key=lambda k: results[k][0])
    
    return {
        "winner": winner,
        "row": results[winner][1],
        "tried": ["random", "greedy", "remember"]
    }

def guess(header, train, test):
    """
    Estimates the first goal column for each test row.
    The customer asks about a scenario where requirements change twice as fast
    halfway through. We simulate this by splitting the training data into two
    halves and observing the drift in the first goal column.
    """
    goal_idx = -1
    for i, name in enumerate(header):
        if name.endswith('-') or name.endswith('+'):
            goal_idx = i
            break
    
    # Split training data into two halves
    mid = len(train) // 2
    first_half = train[:mid]
    second_half = train[mid:]
    
    # Calculate average of first goal in both halves
    avg1 = sum(r[goal_idx] for r in first_half) / len(first_half)
    avg2 = sum(r[goal_idx] for r in second_half) / len(second_half)
    
    # The "change" factor: how much the goal drifted
    drift = avg2 - avg1
    
    # If requirements change twice as fast, we project the drift forward
    # We estimate the goal for test rows by taking the latest average + 2 * drift
    estimate = avg2 + (2 * drift)
    
    return [estimate for _ in test]

import csv
import step04

def flow(header, row, seen):
    """
    Warns the customer by returning True if the current row is significantly 
    worse than the historical average of 'seen' rows.
    
    We define 'going bad' as a row whose performance score is worse than 
    the mean score of all previously seen rows by at least one standard deviation.
    """
    if not seen:
        return False

    goal_indices = [(i, name[-1]) for i, name in enumerate(header) 
                    if name.endswith(('-', '+'))]

    def score_row(r):
        total = 0
        for idx, op in goal_indices:
            val = float(r[idx])
            total += val if op == '-' else (1.0 - val)
        return total

    # Calculate scores of all seen rows
    seen_scores = [score_row(r) for r in seen]
    
    # Calculate mean and standard deviation of seen scores
    mean = sum(seen_scores) / len(seen_scores)
    std = (sum((s - mean)**2 for s in seen_scores) / len(seen_scores))**0.5
    
    # Current row score
    current_score = score_row(row)
    
    # If current score is worse (higher) than mean + 1 std dev, warn (return True)
    if std > 0 and current_score > (mean + std):
        return True
    
    return False

def blame(header, rows):
    """
    Identifies input columns that correlate with high 'Cost-' values.
    We define 'worst' as the top 20% of rows with the highest 'Cost-' values.
    We then compare the distribution of each input column in these 'worst' 
    rows against the rest of the dataset.
    """
    cost_idx = header.index('Cost-')
    input_indices = [i for i, name in enumerate(header) 
                     if not (name.endswith('-') or name.endswith('+'))]
    
    # Sort rows by Cost- descending to find the worst
    sorted_rows = sorted(rows, key=lambda r: r[cost_idx], reverse=True)
    worst_rows = sorted_rows[:len(rows) // 5]
    other_rows = sorted_rows[len(rows) // 5:]
    
    def get_diff(idx):
        # Calculate the difference in means between worst and others
        worst_vals = [r[idx] for r in worst_rows]
        other_vals = [r[idx] for r in other_rows]
        avg_worst = sum(worst_vals) / len(worst_vals)
        avg_other = sum(other_vals) / len(other_vals)
        return abs(avg_worst - avg_other)

    # Rank inputs by how much they differ in the worst setups
    diffs = []
    for i in input_indices:
        diffs.append((get_diff(i), header[i]))
    
    diffs.sort(reverse=True)
    
    # Return the top 2 inputs that explain the variance
    return [name for _, name in diffs[:2]]

                    if name.endswith(('-', '+'))]

        total = 0
        for idx, op in goal_indices:
            val = float(r[idx])
            total += val if op == '-' else (1.0 - val)

    best_indices = step01.remember(header, rows)
    for idx, op in goal_indices:
        if row[idx] == "?":
            row[idx] = sum(rows[i][idx] for i in best_indices) / len(best_indices)

    # Calculate scores
    target_score = score_row(row)
    all_scores = [score_row(r) for r in rows]
    
    # Calculate percentile
    better_count = sum(1 for s in all_scores if s > target_score)

    else:
        reason = f"This setup is better than {p:.2f} of all configurations, but your preferred setup B is slightly more optimal for your specific goals."
        
    return p, reason

def choose(header, rows, label):
    """
    Finds the best setup for a large, high-criticality project.
    We filter for rows where 'Size' and 'Criticality' are high,
    then sample from those to find the best within the 50-label budget.
    """
    # Identify indices for filtering
    size_idx = header.index('Size')
    crit_idx = header.index('Criticality')
    
    # Filter for "large" and "high criticality" (above median)
    sizes = [r[size_idx] for r in rows]
    crits = [r[crit_idx] for r in rows]
    med_size = sum(sizes) / len(sizes)
    med_crit = sum(crits) / len(crits)
    
    candidates = [i for i, r in enumerate(rows) 
                  if r[size_idx] >= med_size and r[crit_idx] >= med_crit]
    
    # If not enough candidates, fall back to all rows
    if not candidates:
        candidates = list(range(len(rows)))
        
    # Sample up to 50 candidates
    import random
    sample_size = min(len(candidates), 50)
    to_test = random.sample(candidates, sample_size)
    
    best_idx = -1
    best_score = float('inf')
    
    def get_score(row_with_goals):
        total = 0
        for i, name in enumerate(header):
            if name.endswith('-'):
                total += row_with_goals[i]
            elif name.endswith('+'):
                total += (1.0 - row_with_goals[i])
        return total

    for idx in to_test:
        goals = label(idx)
        temp_row = list(rows[idx])
        goal_ptr = 0
        for i, name in enumerate(header):
            if name.endswith('-') or name.endswith('+'):
                temp_row[i] = goals[goal_ptr]
                goal_ptr += 1
        
        score = get_score(temp_row)
        if score < best_score:
            best_score = score
            best_idx = idx
            
    return best_idx
