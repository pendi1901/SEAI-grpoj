import csv

def spot(header, rows):
    """
    Identifies rows that contain outliers. We define an outlier as a row 
    containing an input value that is significantly far from the median 
    of that column (using a simple 3-sigma-like heuristic).
    """
    input_indices = [i for i, name in enumerate(header) 
                     if not (name.endswith('-') or name.endswith('+'))]
    
    # Calculate column statistics
    stats = {}
    for i in input_indices:
        vals = [r[i] for r in rows if isinstance(r[i], (int, float))]
        if not vals:
            continue
        mean = sum(vals) / len(vals)
        std = (sum((x - mean)**2 for x in vals) / len(vals))**0.5
        stats[i] = (mean, std)
    
    # Score rows based on how many standard deviations they are from the mean
    # We look for values > 3 standard deviations away
    suspicious = []
    for row_idx, row in enumerate(rows):
        score = 0
        for i in input_indices:
            if i in stats:
                mean, std = stats[i]
                if std > 0:
                    val = row[i]
                    if isinstance(val, (int, float)):
                        dist = abs(val - mean) / std
                        if dist > 3:
                            score += dist
        if score > 0:
            suspicious.append((score, row_idx))
    
    # Sort by most suspicious (highest score)
    suspicious.sort(key=lambda x: x[0], reverse=True)
    return [item[1] for item in suspicious]

if __name__ == "__main__":
    filename = "pom3a.n1000.train.csv"
    try:
        with open(filename, mode='r', encoding='utf-8') as f:
            reader = csv.reader(f)
            header = next(reader)
            rows = []
            for row in reader:
                processed_row = []
                for val in row:
                    try:
                        processed_row.append(float(val))
                    except ValueError:
                        processed_row.append(val)
                rows.append(processed_row)
        
        outliers = spot(header, rows)
        
        print("Customer Inquiry: Which records look mistyped or made up?")
        if outliers:
            print(f"Answer: Found {len(outliers)} suspicious records.")
            print(f"The most suspicious row index is {outliers[0]}.")
            print("These rows contain input values that deviate significantly from the norm.")
        else:
            print("Answer: No obviously mistyped records were found.")
            
    except FileNotFoundError:
        print(f"Error: {filename} not found.")
        exit(1)
