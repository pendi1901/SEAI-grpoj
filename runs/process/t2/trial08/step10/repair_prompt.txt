import csv

def remember(header, rows):
    """
    Identifies rows that perform well across all goal columns.
    Goals ending in '-' are minimized, '+' are maximized.
    """
    goal_indices = []
    for i, name in enumerate(header):
        if name.endswith('-') or name.endswith('+'):
            goal_indices.append((i, name[-1]))

    def score_row(row):
        # Calculate a normalized score for the row.
        # Since we want to minimize '-' and maximize '+',
        # we can treat all as "minimize" by inverting '+' values.
        # Here we use a simple sum of normalized values.
        total = 0
        for idx, op in goal_indices:
            val = float(row[idx])
            total += val if op == '-' else (1.0 - val)
        return total

    # Score all rows
    scored = []
    for i, row in enumerate(rows):
        scored.append((score_row(row), i))

    # Sort by score (ascending, as we minimized everything)
    scored.sort()

    # Return top 5% (or at least 1)
    limit = max(1, len(rows) // 20)
    return [item[1] for item in scored[:limit]]

def certify(xs, ys):
    """
    Returns True if the distributions of xs and ys are statistically similar
    (i.e., the difference is likely noise), False if they differ significantly.
    Uses a simple A12 effect size check (non-parametric).
    """
    def a12(lst1, lst2):
        m, n = len(lst1), len(lst2)
        if m == 0 or n == 0: return 0.5
        greater = 0
        for x in lst1:
            for y in lst2:
                if x > y: greater += 1
                elif x == y: greater += 0.5
        return greater / (m * n)

    # If A12 is close to 0.5, they are similar. 
    # Threshold 0.56 is a common small effect size.
    val = a12(xs, ys)
    return 0.44 <= val <= 0.56

def flow(header, row, seen):
    """
    Decides whether to accept a row based on whether it is better than 
    the current best seen so far.
    """
    goal_indices = []
    for i, name in enumerate(header):
        if name.endswith('-') or name.endswith('+'):
            goal_indices.append((i, name[-1]))

    def score_row(r):
        total = 0
        for idx, op in goal_indices:
            val = float(r[idx])
            total += val if op == '-' else (1.0 - val)
        return total

    # If this is the first row, accept it
    if not seen:
        return True

    # Calculate score of current row and the best score seen so far
    current_score = score_row(row)
    best_score = min(score_row(r) for r in seen)

    # Accept if it's better than the best seen so far
    return current_score < best_score

def blame(header, rows):
    """
    Identifies input columns that correlate with high 'Idle-' values.
    We define 'worst' as the top 20% of rows with the highest 'Idle-' values.
    We then compare the distribution of each input column in these 'worst' 
    rows against the rest of the dataset.
    """
    idle_idx = header.index('Idle-')
    input_indices = [i for i, name in enumerate(header) 
                     if not (name.endswith('-') or name.endswith('+'))]
    
    # Sort rows by Idle- descending to find the worst
    sorted_rows = sorted(rows, key=lambda r: r[idle_idx], reverse=True)
    worst_rows = sorted_rows[:len(rows) // 5]
    other_rows = sorted_rows[len(rows) // 5:]
    
    def get_diff(idx):
        # Calculate the difference in means between worst and others
        worst_vals = [r[idx] for r in worst_rows]
        other_vals = [r[idx] for r in other_rows]
        avg_worst = sum(worst_vals) / len(worst_vals)
        avg_other = sum(other_vals) / len(other_vals)
        return abs(avg_worst - avg_other)

    # Rank inputs by how much they differ in the worst setups
    diffs = []
    for i in input_indices:
        diffs.append((get_diff(i), header[i]))
    
    diffs.sort(reverse=True)
    
    # Return the top 2 inputs that explain the variance
    return [name for _, name in diffs[:2]]

import csv

def spot(header, rows):
    """
    Identifies rows that contain outliers. We define an outlier as a row 
    containing an input value that is significantly far from the median 
    of that column (using a simple 3-sigma-like heuristic).
    """
    input_indices = [i for i, name in enumerate(header) 
                     if not (name.endswith('-') or name.endswith('+'))]
    
    # Calculate column statistics
    stats = {}
    for i in input_indices:
        vals = [r[i] for r in rows if isinstance(r[i], (int, float))]
        if not vals:
            continue
        mean = sum(vals) / len(vals)
        std = (sum((x - mean)**2 for x in vals) / len(vals))**0.5
        stats[i] = (mean, std)
    
    # Score rows based on how many standard deviations they are from the mean
    # We look for values > 3 standard deviations away
    suspicious = []
    for row_idx, row in enumerate(rows):
        score = 0
        for i in input_indices:
            if i in stats:
                mean, std = stats[i]
                if std > 0:
                    val = row[i]
                    if isinstance(val, (int, float)):
                        dist = abs(val - mean) / std
                        if dist > 3:
                            score += dist
        if score > 0:
            suspicious.append((score, row_idx))
    
    # Sort by most suspicious (highest score)
    suspicious.sort(key=lambda x: x[0], reverse=True)
    return [item[1] for item in suspicious]
