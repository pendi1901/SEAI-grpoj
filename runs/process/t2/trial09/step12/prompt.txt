import csv

def remember(header, rows):
    """
    Identifies rows that perform well across all goal columns.
    Goals ending in '-' are minimized, '+' are maximized.
    """
    goal_indices = []
    for i, name in enumerate(header):
        if name.endswith('-') or name.endswith('+'):
            goal_indices.append((i, name[-1]))

    def score_row(row):
        # Calculate a normalized score for the row.
        # Since we want to minimize '-' and maximize '+', 
        # we can treat all goals as "minimize" by negating '+' values.
        # Here we use a simple sum of normalized values.
        total = 0
        for idx, op in goal_indices:
            val = float(row[idx])
            total += val if op == '-' else (1.0 - val)
        return total

    # Score all rows
    scored = []
    for i, row in enumerate(rows):
        scored.append((score_row(row), i))
    
    # Sort by score (ascending, as we minimized everything)
    scored.sort()
    
    # Return top 5% (or at least 1)
    limit = max(1, len(rows) // 20)
    return [item[1] for item in scored[:limit]]

import csv
import math

def certify(xs, ys):
    """
    Returns True if the distributions of xs and ys are statistically 
    indistinguishable (same apart from noise), False otherwise.
    Uses a simple A12 effect size check and a basic variance comparison.
    """
    def a12(lst1, lst2):
        m = len(lst1)
        n = len(lst2)
        count = 0
        for x in lst1:
            for y in lst2:
                if x > y: count += 1
                elif x == y: count += 0.5
        return count / (m * n)

    # If effect size is small (close to 0.5), they are likely the same
    # A12 threshold of 0.56 is a common small effect size
    effect = a12(xs, ys)
    return 0.44 <= effect <= 0.56

def blame(header, rows):
    """
    Identifies input columns that correlate with the 'Idle-' goal.
    We define 'worst' as the top 20% of rows with the highest 'Idle-' values.
    We then compare the average of inputs in these 'worst' rows against the 
    average of the rest of the population.
    """
    idle_idx = header.index('Idle-')
    input_indices = [i for i, name in enumerate(header) 
                     if not (name.endswith('-') or name.endswith('+'))]
    
    # Sort rows by Idle- to find the worst ones
    sorted_rows = sorted(rows, key=lambda r: r[idle_idx], reverse=True)
    worst_rows = sorted_rows[:len(rows) // 5]
    rest_rows = sorted_rows[len(rows) // 5:]
    
    # Calculate average difference for each input
    diffs = []
    for i in input_indices:
        avg_worst = sum(r[i] for r in worst_rows) / len(worst_rows)
        avg_rest = sum(r[i] for r in rest_rows) / len(rest_rows)
        # We look for inputs where the worst rows have significantly higher values
        diffs.append((abs(avg_worst - avg_rest), header[i]))
    
    # Return the top 2 inputs that explain the difference
    diffs.sort(key=lambda x: x[0], reverse=True)
    return [name for _, name in diffs[:2]]

import csv
import math

def spot(header, rows):
    """
    Identifies rows that are outliers. We calculate the Mahalanobis-like 
    distance for each row based on input columns. Rows with high 
    deviations from the mean are considered suspicious.
    """
    input_indices = [i for i, name in enumerate(header) 
                     if not (name.endswith('-') or name.endswith('+'))]
    
    # Calculate means and standard deviations for input columns
    means = []
    stds = []
    for i in input_indices:
        vals = [r[i] for r in rows if isinstance(r[i], (int, float))]
        mu = sum(vals) / len(vals)
        sigma = math.sqrt(sum((x - mu)**2 for x in vals) / len(vals))
        means.append(mu)
        stds.append(sigma if sigma > 0 else 1.0)
    
    # Calculate a "suspicion score" for each row based on Z-score distance
    scores = []
    for row_idx, row in enumerate(rows):
        dist = 0
        for j, col_idx in enumerate(input_indices):
            val = row[col_idx]
            if isinstance(val, (int, float)):
                z = (val - means[j]) / stds[j]
                dist += z**2
        scores.append((dist, row_idx))
    
    # Sort by distance descending (most suspicious first)
    scores.sort(key=lambda x: x[0], reverse=True)
    
    # Return the indices of the top 5% most suspicious rows
    limit = max(1, len(rows) // 20)
    return [item[1] for item in scores[:limit]]

        "first": (best_first, score_goals(label(best_first)))
    }
    
    winner = min(results, key=lambda k: results[k][1])

    }

def find_similar(header, rows, target_row, k=5):
    """
    Finds the k most similar rows to the target_row using Euclidean distance
    on input columns.
    """
    input_indices = [i for i, name in enumerate(header) 
                     if not (name.endswith('-') or name.endswith('+'))]
    
    def dist(r1, r2):
        d = 0
        for i in input_indices:
            d += (r1[i] - r2[i])**2
        return d**0.5

    # Calculate distances to all rows
    distances = []
    for i, row in enumerate(rows):
        # Skip if row is the target itself
        if row == target_row:
            continue
        distances.append((dist(target_row, row), i))
    
    distances.sort()
    return [idx for _, idx in distances[:k]]

    """
