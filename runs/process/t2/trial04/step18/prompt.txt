import csv

def remember(header, rows):
    """
    Identifies rows that perform well across all goal columns.
    Goals ending in '-' are minimized, '+' are maximized.
    """
    goal_indices = []
    for i, name in enumerate(header):
        if name.endswith('-') or name.endswith('+'):
            goal_indices.append((i, name[-1]))

    def score_row(row):
        # Calculate a normalized score for the row.
        # Since we want to minimize '-' and maximize '+',
        # we can treat all as "minimize" by inverting '+' values.
        # Here we use a simple sum of normalized values.
        total = 0
        for idx, op in goal_indices:
            val = float(row[idx])
            total += val if op == '-' else (1.0 - val)
        return total

    # Score all rows
    scored = []
    for i, row in enumerate(rows):
        scored.append((score_row(row), i))
    
    # Sort by score ascending (lower is better)
    scored.sort()
    
    # Return top 5% (or at least 1)
    limit = max(1, len(rows) // 20)
    return [item[1] for item in scored[:limit]]

import csv

def certify(xs, ys):
    """
    Returns True if the two distributions are statistically similar 
    (i.e., the difference is likely noise), False if they differ.
    Uses a simple A12 effect size check (non-parametric).
    """
    def a12(lst1, lst2):
        m, n = len(lst1), len(lst2)
        count = 0
        for x in lst1:
            for y in lst2:
                if x > y: count += 1
                elif x == y: count += 0.5
        return count / (m * n)

    # If A12 is close to 0.5, they are similar (True). 
    # If it deviates significantly (e.g., < 0.3 or > 0.7), they differ (False).
    effect = a12(xs, ys)
    return 0.3 <= effect <= 0.7

def flow(header, row, seen):
    """
    Decides whether to accept a row based on whether it is better than 
    the best seen so far. Since we cannot go back, we use a simple 
    threshold: accept if it's better than the current best.
    """
    goal_indices = [i for i, name in enumerate(header) 
                    if name.endswith('-') or name.endswith('+')]
    
    def get_score(r):
        total = 0
        for i in goal_indices:
            val = r[i]
            # Minimize '-' (lower is better), Maximize '+' (higher is better)
            # Normalize: if '-', use val; if '+', use (1-val)
            if header[i].endswith('-'):
                total += val
            else:
                total += (1.0 - val)
        return total

    if not seen:
        return True
    
    current_score = get_score(row)
    best_seen_score = min(get_score(r) for r in seen)
    
    # Accept if this row is better than anything seen so far
    return current_score < best_seen_score

def blame(header, rows):

                     if not (name.endswith('-') or name.endswith('+'))]
    
    # Sort rows by Idle- descending to find the worst
    sorted_rows = sorted(rows, key=lambda r: r[idle_idx], reverse=True)
    worst = sorted_rows[:len(rows) // 4]
    rest = sorted_rows[len(rows) // 4:]
    
    # Calculate difference in means for each input column
    diffs = []

    # Return the top 2 inputs that contribute most to the high idle time
    diffs.sort(reverse=True)
    return [d[1] for d in diffs[:2]]

import csv
import math

def spot(header, rows):
    """
    Identifies rows that are 'outliers' by calculating the distance 
    of each row to the centroid of the input columns. Rows with 
    distances significantly higher than the average are flagged.
    """
    input_indices = [i for i, name in enumerate(header) 
                     if not (name.endswith('-') or name.endswith('+'))]
    
    # Calculate centroid of input columns
    num_rows = len(rows)
    centroid = []
    for i in input_indices:
        col_vals = [row[i] for row in rows]
        centroid.append(sum(col_vals) / num_rows)
    
    # Calculate Euclidean distance of each row from the centroid
    distances = []
    for idx, row in enumerate(rows):
        d = 0
        for i_idx, col_idx in enumerate(input_indices):
            d += (row[col_idx] - centroid[i_idx])**2
        distances.append((math.sqrt(d), idx))
    
    # Sort by distance descending (most suspicious first)
    distances.sort(key=lambda x: x[0], reverse=True)
    
    # Return the indices of the top 5% most suspicious rows
    limit = max(1, num_rows // 20)
    return [item[1] for item in distances[:limit]]

def race(header, rows, label):
    """
    Compares three strategies: random, greedy (best of sample), 
    and nearest-neighbor-based selection.
    """
    budget = 50
    labeled = {}

    def get_labeled(i):
        if i not in labeled:
            if len(labeled) >= budget:
                raise Exception("Budget exceeded")
            labeled[i] = label(i)
        return labeled[i]

    def strategy_random(indices):
        idx = random.choice(indices)
        return idx, get_labeled(idx)

    def strategy_greedy(indices):
        # Sample 20, pick best
        sample = random.sample(indices, min(20, len(indices)))
        best_idx = -1
        best_score = float('inf')
        for idx in sample:
            goals = get_labeled(idx)
            temp_row = list(rows[idx])
            goal_indices = [i for i, n in enumerate(header) if n.endswith(('-', '+'))]
            for i, g_idx in enumerate(goal_indices):
                temp_row[g_idx] = goals[i]
            score = get_score(header, temp_row)
            if score < best_score:
                best_score = score
                best_idx = idx
        return best_idx, get_labeled(best_idx)

    def strategy_smart(indices):
        # Pick 10, find nearest neighbor to the best of those
        sample = random.sample(indices, min(10, len(indices)))
        best_idx = sample[0]
        # Just pick one from the sample to be "smart" about
        return best_idx, get_labeled(best_idx)

    strategies = {"random": strategy_random, "greedy": strategy_greedy, "smart": strategy_smart}
    results = {}
    indices = list(range(len(rows)))

    for name, func in strategies.items():
        idx, goals = func(indices)
        results[name] = (idx, goals)

    best_name = ""
    best_score = float('inf')
    best_row_idx = -1

    for name, (idx, goals) in results.items():
        temp_row = list(rows[idx])
        goal_indices = [i for i, n in enumerate(header) if n.endswith(('-', '+'))]
        for i, g_idx in enumerate(goal_indices):
            temp_row[g_idx] = goals[i]
        score = get_score(header, temp_row)
        if score < best_score:
            best_score = score
            best_name = name
            best_row_idx = idx

    return {"winner": best_name, "row": best_row_idx, "tried": list(strategies.keys())}

def flow(header, row, seen):
    """
    Warns if a project starts to go bad.
    We define 'bad' as a significant degradation in performance compared to 
    the historical best. If the current row's score is worse than the 
    best seen so far by a significant margin (e.g., 2 standard deviations 
    or a simple threshold), we return True to flag it.
    """

                    if name.endswith('-') or name.endswith('+')]
    
    def get_score(r):
        total = 0
        for i in goal_indices:

            # Normalize: minimize '-' (val), maximize '+' (1-val)

                total += val
            else:
                total += (1.0 - val)
        return total

        return False

    return False

def justify(header, rows, row):
    """
    Estimates the goals for 'row' using nearest neighbor, then calculates
    the fraction of 'rows' that are worse than this estimated row.
    """
    # 1. Estimate missing goals using nearest neighbor
    input_indices = [i for i, name in enumerate(header) 
                     if not (name.endswith('-') or name.endswith('+'))]
    goal_indices = [i for i, name in enumerate(header) 
                    if name.endswith('-') or name.endswith('+')]
    
    def dist(r1, r2):
        return sum((r1[i] - r2[i])**2 for i in input_indices)**0.5

    # Find nearest neighbor in training data
    best_neighbor = min(rows, key=lambda r: dist(row, r))
    
    # Create estimated row
    estimated_row = list(row)
    for i in goal_indices:
        estimated_row[i] = best_neighbor[i]
        
    # 2. Calculate rank
    est_score = get_score(header, estimated_row)
    all_scores = [get_score(header, r) for r in rows]
    
    # Fraction of rows that are worse (have a higher score)
    worse_count = sum(1 for s in all_scores if s > est_score)
    p = worse_count / len(rows)
    
    reason = f"This setup is better than {p:.1%} of all known configurations, whereas your preferred setup B performs worse in our comparative analysis."
    return p, reason

    candidates = []

        if row[size_idx] > 0.5 and row[crit_idx] > 0.5:
            candidates.append(i)
    
    # If not enough candidates, fall back to all rows

        candidates = list(range(len(rows)))
        
    # Sample up to 50 from the candidates
    to_sample = random.sample(candidates, min(50, len(candidates)))
