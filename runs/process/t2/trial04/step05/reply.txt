import csv
import step01

def blame(header, rows):
    """
    Identifies input columns that correlate with high 'Idle-' values.
    We define 'worst' as the top 25% of rows with the highest 'Idle-' values.
    We then compare the average of these inputs against the rest of the data.
    """
    idle_idx = header.index('Idle-')
    input_indices = [i for i, name in enumerate(header) 
                     if not (name.endswith('-') or name.endswith('+'))]
    
    # Sort rows by Idle- descending to find the worst
    sorted_rows = sorted(rows, key=lambda r: r[idle_idx], reverse=True)
    worst = sorted_rows[:len(rows) // 4]
    rest = sorted_rows[len(rows) // 4:]
    
    # Calculate difference in means for each input column
    diffs = []
    for i in input_indices:
        avg_worst = sum(r[i] for r in worst) / len(worst)
        avg_rest = sum(r[i] for r in rest) / len(rest)
        # We look for inputs that are significantly higher in the worst rows
        diffs.append((avg_worst - avg_rest, header[i]))
    
    # Return the top 2 inputs that contribute most to the high idle time
    diffs.sort(reverse=True)
    return [d[1] for d in diffs[:2]]

if __name__ == "__main__":
    filename = "pom3a.n1000.train.csv"
    try:
        with open(filename, mode='r', encoding='utf-8') as f:
            reader = csv.reader(f)
            header = next(reader)
            rows = [[float(val) for val in row] for row in reader]
        
        reasons = blame(header, rows)
        print("The worst setups leave people waiting primarily due to:")
        for reason in reasons:
            print(f"- {reason}")
            
    except FileNotFoundError:
        print(f"Error: {filename} not found.")
        exit(1)
