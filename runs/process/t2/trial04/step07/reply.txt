import csv
import random
import step01

def choose(header, rows, label):
    """
    Finds the best row by sampling 50 rows, labeling them,
    and returning the index of the one with the best score.
    """
    # Identify goal indices
    goal_indices = [i for i, name in enumerate(header) 
                    if name.endswith('-') or name.endswith('+')]
    
    # We can label at most 50 rows.
    # We pick 50 random indices to sample.
    indices_to_sample = random.sample(range(len(rows)), min(50, len(rows)))
    
    best_idx = -1
    best_score = float('inf')
    
    for idx in indices_to_sample:
        # Get true goal values
        true_goals = label(idx)
        
        # Construct a temporary row with true goals to score it
        temp_row = list(rows[idx])
        for i, g_idx in enumerate(goal_indices):
            temp_row[g_idx] = true_goals[i]
            
        # Score using step01 logic
        score = 0
        for g_idx in goal_indices:
            val = temp_row[g_idx]
            # Minimize '-' (lower is better), Maximize '+' (higher is better)
            if header[g_idx].endswith('-'):
                score += val
            else:
                score += (1.0 - val)
        
        if score < best_score:
            best_score = score
            best_idx = idx
            
    return best_idx

if __name__ == "__main__":
    filename = "pom3a.n1000.train.csv"
    try:
        with open(filename, mode='r', encoding='utf-8') as f:
            reader = csv.reader(f)
            header = next(reader)
            full_data = [[float(val) for val in row] for row in reader]
            
        # Prepare rows with goals masked as "?"
        goal_indices = [i for i, name in enumerate(header) 
                        if name.endswith('-') or name.endswith('+')]
        
        masked_rows = []
        for row in full_data:
            new_row = list(row)
            for i in goal_indices:
                new_row[i] = "?"
            masked_rows.append(new_row)
            
        # Define the label function for the harness
        def label(i):
            return [full_data[i][g] for g in goal_indices]
            
        best_idx = choose(header, masked_rows, label)
        
        print(f"The best setup found after 50 trials is at row index: {best_idx}")
        print("Goal values for this setup:")
        goals = {header[i]: full_data[best_idx][i] for i in goal_indices}
        for name, val in goals.items():
            print(f"  {name}: {val:.4f}")
            
    except FileNotFoundError:
        print(f"Error: {filename} not found.")
        exit(1)
