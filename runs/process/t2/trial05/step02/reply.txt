import csv
import step01

def guess(header, train, test):
    """
    Estimates the first goal column for each test row by finding the 
    nearest neighbor in the training set based on input columns.
    """
    # Identify input columns (those not ending in - or +)
    input_indices = [i for i, name in enumerate(header) 
                     if not (name.endswith('-') or name.endswith('+'))]
    
    # Identify the first goal column index
    goal_idx = [i for i, name in enumerate(header) 
                if name.endswith('-') or name.endswith('+')][0]

    predictions = []
    for t_row in test:
        best_dist = float('inf')
        best_val = 0.0
        
        # Simple Euclidean distance on input columns
        for train_row in train:
            dist = 0
            for i in input_indices:
                # Assuming normalized [0,1] data as per prompt
                dist += (float(t_row[i]) - float(train_row[i]))**2
            
            if dist < best_dist:
                best_dist = dist
                best_val = float(train_row[goal_idx])
        
        predictions.append(best_val)
    
    return predictions

if __name__ == "__main__":
    filename = "pom3a.n1000.train.csv"
    try:
        with open(filename, mode='r', encoding='utf-8') as f:
            reader = csv.reader(f)
            header = next(reader)
            rows = []
            for row in reader:
                rows.append([float(val) for val in row])
        
        # Split: use first 800 as train, next 100 as test (simulated)
        train = rows[:800]
        test_data = rows[800:900]
        
        # Mask goals in test data
        test_masked = []
        for row in test_data:
            new_row = [row[i] if not (header[i].endswith('-') or header[i].endswith('+')) else "?" 
                       for i in range(len(row))]
            test_masked.append(new_row)
            
        results = guess(header, train, test_masked)
        
        print("Estimates for the first goal column (Cost-):")
        for i, val in enumerate(results[:5]):
            print(f"Test Row {i}: {val:.4f}")
        print("...")
        
    except FileNotFoundError:
        print(f"Error: {filename} not found.")
        exit(1)
