import csv
import step01
import step02

def justify(header, rows, row):
    """
    Estimates the goals for the given row and calculates its percentile rank
    against the provided dataset.
    """
    # 1. Estimate the missing goals in 'row'
    # We use the logic from step02 to predict the goals.
    # Since step02 only predicts the first goal, we generalize:
    input_indices = [i for i, name in enumerate(header) 
                     if not (name.endswith('-') or name.endswith('+'))]
    goal_indices = [i for i, name in enumerate(header) 
                    if name.endswith('-') or name.endswith('+')]
    
    # Simple KNN estimation for all goal columns
    estimated_goals = []
    for g_idx in goal_indices:
        best_dist = float('inf')
        best_val = 0.0
        for train_row in rows:
            dist = sum((float(row[i]) - float(train_row[i]))**2 for i in input_indices)
            if dist < best_dist:
                best_dist = dist
                best_val = float(train_row[g_idx])
        estimated_goals.append(best_val)
    
    # 2. Calculate score for the estimated row
    def get_score(r_vals):
        total = 0
        for i, g_idx in enumerate(goal_indices):
            val = r_vals[i]
            total += (1.0 - val) if header[g_idx].endswith('-') else val
        return total

    row_score = get_score(estimated_goals)
    
    # 3. Compare against all rows
    all_scores = [get_score([r[g] for g in goal_indices]) for r in rows]
    better_than = sum(1 for s in all_scores if row_score > s)
    p = better_than / len(rows)
    
    reason = f"This setup is better than {p:.1%} of all known configurations based on its predicted performance across all objectives."
    return p, reason

if __name__ == "__main__":
    filename = "pom3a.n1000.train.csv"
    try:
        with open(filename, mode='r', encoding='utf-8') as f:
            reader = csv.reader(f)
            header = next(reader)
            rows = []
            for row in reader:
                rows.append([float(val) for val in row])
        
        # Pick a sample row and mask its goals to test the justify function
        sample_idx = 10
        test_row = list(rows[sample_idx])
        for i in range(len(test_row)):
            if header[i].endswith('-') or header[i].endswith('+'):
                test_row[i] = "?"
        
        p, reason = justify(header, rows, test_row)
        print(f"Rank: {p:.4f}")
        print(f"Justification: {reason}")
        
    except FileNotFoundError:
        print(f"Error: {filename} not found.")
        exit(1)
