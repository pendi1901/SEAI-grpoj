import csv

def remember(header, rows):
    """
    Identifies rows that perform well across all goal columns.
    Goals ending in '-' are minimized, '+' are maximized.
    """
    goals = []
    for i, name in enumerate(header):
        if name.endswith('-') or name.endswith('+'):
            goals.append((i, name.endswith('-')))

    # Calculate a score for each row based on the normalized rank of its goals.
    # Since all goals are numeric and in [0, 1], we can simply sum the 
    # normalized values (minimizing '-' is equivalent to maximizing 1 - value).
    scores = []
    for idx, row in enumerate(rows):
        total_score = 0
        for col_idx, is_min in goals:
            val = float(row[col_idx])
            total_score += (1.0 - val) if is_min else val
        scores.append((total_score, idx))

    # Sort by total score descending (best performers first)
    scores.sort(key=lambda x: x[0], reverse=True)

    # Return top 5% (or at least 1)
    limit = max(1, len(rows) // 20)
    return [item[1] for item in scores[:limit]]

import csv

def certify(xs, ys):
    """
    Returns True if the two distributions are statistically similar 
    (using a simple A12 effect size check), False otherwise.
    """
    def a12(lst1, lst2):
        m, n = len(lst1), len(lst2)
        lst1 = sorted(lst1)
        lst2 = sorted(lst2)
        
        # Calculate rank sum
        r1 = 0
        for x in lst1:
            for y in lst2:
                if x > y: r1 += 1
                elif x == y: r1 += 0.5
        
        return abs(r1 / (m * n) - 0.5) > 0.147  # Threshold for "small" effect size

    return not a12(xs, ys)

def flow(header, row, seen):
    """
    Decides whether to accept a row based on the 'Secretary Problem' logic
    adapted for multi-objective optimization. We observe the first 20% of 
    rows to establish a baseline, then accept the first row that outperforms 
    the best seen so far.
    """
    # Identify goal indices
    goal_indices = [i for i, name in enumerate(header) 
                    if name.endswith('-') or name.endswith('+')]
    
    def get_score(r):
        total = 0
        for i in goal_indices:
            val = float(r[i])
            # Minimize '-' (1-val), Maximize '+' (val)
            total += (1.0 - val) if header[i].endswith('-') else val
        return total

    # If it's the first row, we can't compare yet
    if not seen:
        return False

    # Calculate baseline: best score seen so far
    scores = [get_score(s) for s in seen]
    best_so_far = max(scores)
    
    # Strategy: 
    # 1. Observe at least 20% of the data (exploration phase)
    # 2. Accept if current row is better than anything seen before
    # 3. Accept the very last row if nothing else was chosen
    
    total_expected = 800
    if len(seen) < (0.2 * total_expected):
        return False
    
    if get_score(row) > best_so_far:
        return True
        
    if len(seen) >= (total_expected - 1):
        return True
        
    return False

def blame(header, rows):
    """
    Identifies input columns that correlate most strongly with the 'Idle-' goal.
    We define 'worst' as rows where 'Idle-' is in the top 25% (highest values).
    We then compare the average value of each input column in these 'worst' rows
    against the average of the rest of the data.
    """
    idle_idx = header.index('Idle-')
    input_indices = [i for i, name in enumerate(header) 
                     if not (name.endswith('-') or name.endswith('+'))]
    
    # Sort rows by Idle- descending to find the worst 25%
    sorted_rows = sorted(rows, key=lambda r: r[idle_idx], reverse=True)
    worst_rows = sorted_rows[:len(rows) // 4]
    
    # Calculate average of inputs for worst rows
    def get_avg(data, col_idx):
        vals = [r[col_idx] for r in data]
        return sum(vals) / len(vals)
    
    # Find inputs that are significantly higher or lower in worst rows
    # We look for the largest absolute difference in normalized space
    diffs = []
    for i in input_indices:
        avg_worst = get_avg(worst_rows, i)
        avg_all = get_avg(rows, i)
        diffs.append((abs(avg_worst - avg_all), header[i]))
    
    # Return the top 3 inputs that explain the variance
    diffs.sort(key=lambda x: x[0], reverse=True)
    return [name for _, name in diffs[:3]]

import csv
import math

def spot(header, rows):
    """
    Identifies rows that are 'outliers' by calculating the distance of each row
    to its nearest neighbor in the input space. Rows with unusually large 
    distances to their nearest neighbor are flagged as suspicious.
    """
    input_indices = [i for i, name in enumerate(header) 
                     if not (name.endswith('-') or name.endswith('+'))]
    
    # Calculate distance to nearest neighbor for each row
    distances = []
    for i, row1 in enumerate(rows):
        min_dist = float('inf')
        for j, row2 in enumerate(rows):
            if i == j:
                continue
            # Euclidean distance in input space
            dist = sum((row1[k] - row2[k])**2 for k in input_indices)
            if dist < min_dist:
                min_dist = dist
        distances.append((min_dist, i))
    
    # Sort by distance descending (most isolated/suspicious first)
    distances.sort(key=lambda x: x[0], reverse=True)
    
    # Return the indices of the top 5% most suspicious rows
    limit = max(1, len(rows) // 20)
    return [item[1] for item in distances[:limit]]

import csv

def race(header, rows, label):
    """
    Compares three strategies: random, greedy (based on step07), 
    and a 'first-best' approach.
    """
    goal_indices = [i for i, name in enumerate(header) 
                    if name.endswith('-') or name.endswith('+')]
    
    def get_score(row_vals):
        total = 0
        for i, g_idx in enumerate(goal_indices):
            val = float(row_vals[i])
            total += (1.0 - val) if header[g_idx].endswith('-') else val
        return total

    # Shared budget of 50 labels
    budget = 50
    labeled = {}

    def get_label(idx):
        if idx not in labeled:
            if len(labeled) >= budget:
                raise Exception("Budget exceeded")
            labeled[idx] = label(idx)
        return labeled[idx]

    # Strategy 1: Random
    def strategy_random():
        indices = random.sample(range(len(rows)), 16)
        best_idx = -1
        best_score = -1.0
        for idx in indices:
            score = get_score(get_label(idx))
            if score > best_score:
                best_score, best_idx = score, idx
        return best_idx, best_score

    # Strategy 2: Greedy (from step07 logic)
    def strategy_greedy():
        indices = [int(i * (len(rows) - 1) / 16) for i in range(17)]
        best_idx = -1
        best_score = -1.0
        for idx in indices:
            score = get_score(get_label(idx))
            if score > best_score:
                best_score, best_idx = score, idx
        return best_idx, best_score

    # Strategy 3: First-Best (Sample first 17)
    def strategy_first():
        best_idx = -1
        best_score = -1.0
        for idx in range(17):
            score = get_score(get_label(idx))
            if score > best_score:
                best_score, best_idx = score, idx
        return best_idx, best_score

    strategies = {
        "random": strategy_random,
        "greedy": strategy_greedy,
        "first": strategy_first
    }

    results = {}
    for name, func in strategies.items():
        idx, score = func()
        results[name] = (idx, score)

    winner = max(results, key=lambda k: results[k][1])
    
    return {
        "winner": winner,
        "row": results[winner][0],

def find_similar(header, rows, target_row, k=5):
    """
    Finds the k most similar rows to the target_row based on input columns.
    """
    input_indices = [i for i, name in enumerate(header) 
                     if not (name.endswith('-') or name.endswith('+'))]
    
    scored_rows = []
    for i, row in enumerate(rows):
        dist = sum((float(target_row[j]) - float(row[j]))**2 for j in input_indices)
        scored_rows.append((dist, i))
    
    scored_rows.sort(key=lambda x: x[0])
    return scored_rows[:k]
