import csv
import pandas as pd
import numpy as np

def remember(header, rows):
    """
    Identifies rows that perform well across all goal columns.
    Goals ending in '-' are minimized, '+' are maximized.
    """
    df = pd.DataFrame(rows, columns=header)
    
    # Identify goal columns
    goals = [c for c in header if c.endswith('-') or c.endswith('+')]
    
    # Normalize goals to a [0, 1] scale where 1 is always "best"
    # For '-', best is 0 (so we use 1 - normalized_value)
    # For '+', best is 1 (so we use normalized_value)
    scores = pd.DataFrame(index=df.index)
    for col in goals:
        vals = df[col].astype(float)
        norm = (vals - vals.min()) / (vals.max() - vals.min() + 1e-9)
        if col.endswith('-'):
            scores[col] = 1 - norm
        else:
            scores[col] = norm
            
    # Calculate aggregate score (mean of normalized goals)
    df['aggregate'] = scores.mean(axis=1)
    
    # Select top 5% (or at least 1)
    n_select = max(1, len(df) // 20)
    best_indices = df['aggregate'].nlargest(n_select).index.tolist()
    
    return sorted(best_indices)

import csv
import pandas as pd
from scipy import stats

def certify(xs, ys):
    """
    Returns True if the difference between two distributions is not statistically 
    significant (i.e., they are the same apart from noise), False otherwise.
    Uses Welch's t-test.
    """
    # If the p-value is greater than 0.05, we fail to reject the null hypothesis
    # (meaning the difference is likely due to noise/luck).
    t_stat, p_value = stats.ttest_ind(xs, ys, equal_var=False)
    return p_value > 0.05

def analyze_team_size_impact(header, rows):
    """
    Compares Cost- for small-team setups vs big-team setups.
    """
    df = pd.DataFrame(rows, columns=header)
    
    # Define small vs big based on the median of 'Team Size'
    median_team_size = df['Team Size'].median()
    
    small_teams = df[df['Team Size'] <= median_team_size]['Cost-']
    big_teams = df[df['Team Size'] > median_team_size]['Cost-']
    
    is_same = certify(small_teams.tolist(), big_teams.tolist())
    
    return is_same, small_teams.mean(), big_teams.mean()

import csv
import pandas as pd

def flow(header, row, seen):
    """
    Decides whether to accept a row based on whether it is better than 
    the best row seen so far. Uses a simple threshold strategy:
    accept if the row is in the top 5% of the current distribution.
    """
    if not seen:
        return True

    # Identify goal columns
    goals = [i for i, h in enumerate(header) if h.endswith('-') or h.endswith('+')]
    
    def get_score(r):
        # Calculate a simple normalized score for the row
        # We don't have the full distribution, so we compare against 'seen'
        # For this online task, we evaluate if the row is better than the current best
        score = 0
        for i in goals:
            val = float(r[i])
            # Simple heuristic: minimize '-' (lower is better), maximize '+' (higher is better)
            if header[i].endswith('-'):
                score -= val
            else:
                score += val
        return score

    current_score = get_score(row)
    seen_scores = [get_score(s) for s in seen]
    
    # If this row is better than the best seen so far, accept it
    if current_score >= max(seen_scores):
        return True
    
    # Otherwise, accept with a small probability (exploration)
    return False

def justify(header, rows, row):
    """
    Estimates the performance of a given row and returns its percentile rank
    relative to the provided dataset.
    """
    df = pd.DataFrame(rows, columns=header)
    goals = [c for c in header if c.endswith('-') or c.endswith('+')]
    inputs = [c for c in header if c not in goals]

    # 1. Estimate the goals for the input row using KNN
    X_train = df[inputs]
    y_train = df[goals]
    model = KNeighborsRegressor(n_neighbors=10)
    model.fit(X_train, y_train)
    
    # Extract inputs from the test row (ignoring '?')
    test_inputs = [row[i] for i, h in enumerate(header) if h in inputs]
    pred_goals = model.predict([test_inputs])[0]
    
    # 2. Calculate aggregate score for all rows
    # Normalize goals: 1 is best, 0 is worst
    def get_aggregate(data_goals):
        scores = []
        for i, col in enumerate(goals):
            vals = df[col]
            norm = (data_goals[i] - vals.min()) / (vals.max() - vals.min() + 1e-9)
            scores.append(1 - norm if col.endswith('-') else norm)
        return sum(scores) / len(scores)

    # Calculate scores for training set
    train_scores = []
    for _, r in df.iterrows():
        train_scores.append(get_aggregate(r[goals].values))
    
    # Calculate score for the test row
    test_score = get_aggregate(pred_goals)
    
    # 3. Calculate percentile (fraction of rows that this row is better than)
    better_than = [s for s in train_scores if test_score > s]
    p = len(better_than) / len(train_scores)
    
    reason = f"This setup is estimated to outperform {p:.1%} of all possible configurations based on its predicted goal metrics."
    
    return p, reason

import csv
import math

def spot(header, rows):
    """
    Identifies rows that are outliers using the Isolation Forest concept 
    (or a simplified distance-based anomaly detection).
    Returns the indices of rows that are most suspicious (furthest from the median).
    """
    # Identify numeric input columns (exclude goals)
    input_indices = [i for i, h in enumerate(header) 
                     if h[0].isupper() and not (h.endswith('-') or h.endswith('+'))]
    
    # Calculate median for each input column
    medians = []
    for i in input_indices:
        vals = sorted([row[i] for row in rows if row[i] != "?"])
        medians.append(vals[len(vals) // 2])
    
    # Calculate a "suspicion score" for each row based on Euclidean distance to median
    scores = []
    for idx, row in enumerate(rows):
        dist = 0
        count = 0
        for i, col_idx in enumerate(input_indices):
            val = row[col_idx]
            if val != "?":
                dist += (val - medians[i])**2
                count += 1
        
        # Normalize by number of valid inputs
        score = math.sqrt(dist / count) if count > 0 else 0
        scores.append((score, idx))
    
    # Sort by score descending (most suspicious first)
    scores.sort(key=lambda x: x[0], reverse=True)
    
    return [idx for score, idx in scores if score > 0]

import csv
import random

def race(header, rows, label):
    """
    Compares three strategies: random, greedy (hill climbing), and 
    a simple heuristic-based search.
    """
    goal_indices = [i for i, h in enumerate(header) if h.endswith('-') or h.endswith('+')]
    
    def get_score(row_vals):
        score = 0
        for i, g_idx in enumerate(goal_indices):
            val = row_vals[i]
            if header[g_idx].endswith('-'):
                score -= val
            else:
                score += val
        return score

    budget = 50
    labeled = {}

    def get_label(idx):
        if idx not in labeled:
            if len(labeled) >= budget:
                return None
            labeled[idx] = label(idx)
        return labeled[idx]

    # Strategy 1: Random
    def strategy_random():
        best_idx = -1
        best_score = -float('inf')
        for _ in range(16): # 16 trials
            idx = random.randint(0, len(rows) - 1)
            vals = get_label(idx)
            if vals is None: break
            score = get_score(vals)
            if score > best_score:
                best_score, best_idx = score, idx
        return best_idx, best_score

    # Strategy 2: Greedy (Hill Climbing)
    def strategy_greedy():
        curr = random.randint(0, len(rows) - 1)
        best_idx, best_score = curr, get_score(get_label(curr))
        for _ in range(16): # 16 trials
            neighbor = random.randint(0, len(rows) - 1)
            vals = get_label(neighbor)
            if vals is None: break
            score = get_score(vals)
            if score > best_score:

def remember(header, rows):

        for i, g_idx in enumerate(goal_indices):
            val = row[g_idx]
            low, high = goals_data[i]
            # Normalize to [0, 1]
            norm = (val - low) / (high - low + 1e-9)
            # If '-', lower is better (1-norm). If '+', higher is better (norm).
            if header[g_idx].endswith('-'):
                score += (1 - norm)
            else:
                score += norm
        return score / len(goal_indices)

    # Score all rows
    scored_rows = []
    for idx, row in enumerate(rows):
        scored_rows.append((get_score(row), idx))
    
    # Sort by score descending
    scored_rows.sort(key=lambda x: x[0], reverse=True)
    
    # Return top 5%
    n_select = max(1, len(rows) // 20)
    return [idx for score, idx in scored_rows[:n_select]]

import csv
from scipy import stats

def certify(xs, ys):
    """
    Returns True if the difference between two distributions is not statistically 
    significant (i.e., they are the same apart from noise), False otherwise.
    Uses Welch's t-test.
    """
    if len(xs) < 2 or len(ys) < 2:
        return True
    t_stat, p_value = stats.ttest_ind(xs, ys, equal_var=False)
    return p_value > 0.05

def analyze_shift(header, rows):
    """
    Compares the performance of the first half of the data vs the second half
    to determine if the 'spring' change resulted in a statistically significant shift.
    """
    goal_indices = [i for i, h in enumerate(header) if h.endswith('-') or h.endswith('+')]
    
    # Calculate aggregate score for each row
    # Normalize goals: 1 is best, 0 is worst
    def get_aggregate(row):
        scores = []
        for i in goal_indices:
            col_vals = [r[i] for r in rows]
            min_v, max_v = min(col_vals), max(col_vals)
            norm = (row[i] - min_v) / (max_v - min_v + 1e-9)
            scores.append(1 - norm if header[i].endswith('-') else norm)
        return sum(scores) / len(scores)

    scores = [get_aggregate(row) for row in rows]
    
    # Split into two halves (before and after the change)
    mid = len(scores) // 2
    before = scores[:mid]
    after = scores[mid:]
    
    is_same = certify(before, after)
    return is_same, sum(before)/len(before), sum(after)/len(after)

import csv

def flow(header, row, seen):
    """
    Detects if the current row is significantly worse than the historical trend.
    Returns True if the row is 'bad' (an outlier on the negative side), 
    triggering a warning.
    """
    if not seen:
        return False

    goal_indices = [i for i, h in enumerate(header) if h.endswith('-') or h.endswith('+')]
    
    def get_score(r):
        score = 0
        for i in goal_indices:
            val = float(r[i])
            # Normalize: for '-', lower is better; for '+', higher is better.
            # Since we don't have global min/max, we use the 'seen' data.
            col_vals = [float(s[i]) for s in seen]
            min_v, max_v = min(col_vals), max(col_vals)
            norm = (val - min_v) / (max_v - min_v + 1e-9)
            score += (1 - norm) if header[i].endswith('-') else norm
        return score / len(goal_indices)

    # Calculate scores for all seen rows
    seen_scores = [get_score(s) for s in seen]
    current_score = get_score(row)
    
    # If the current score is significantly lower than the average of seen rows
    # (e.g., more than 2 standard deviations below the mean), flag it as "bad".
    if len(seen_scores) > 10:
        mean = sum(seen_scores) / len(seen_scores)
        var = sum((s - mean)**2 for s in seen_scores) / len(seen_scores)
        std = var**0.5
        if current_score < (mean - 2 * std):
            return True
            
    return False

def blame(header, rows):
    """
    Identifies the input columns that best explain why the worst rows are bad.
    We define 'bad' rows as those with the lowest aggregate performance.
    We then use feature importance to see which inputs drive these outcomes.
    """
    df = pd.DataFrame(rows, columns=header)
    
    # Identify goal columns and input columns
    goals = [c for c in header if c.endswith('-') or c.endswith('+')]
    inputs = [c for c in header if c not in goals]
    
    # Normalize goals to calculate an aggregate "badness" score
    # For '-', lower is better (so higher is worse). For '+', higher is better (so lower is worse).
    scores = pd.DataFrame(index=df.index)
    for col in goals:
        vals = df[col].astype(float)
        norm = (vals - vals.min()) / (vals.max() - vals.min() + 1e-9)
        if col.endswith('-'):
            scores[col] = norm  # High norm = bad
        else:
            scores[col] = 1 - norm  # High norm = bad
            
    # Aggregate badness score
    df['badness'] = scores.mean(axis=1)
    
    # Train a Random Forest to predict 'badness' from inputs
    X = df[inputs]
    y = df['badness']
    
    model = RandomForestRegressor(n_estimators=100, random_state=42)
    model.fit(X, y)
    
    # Get feature importances
    importances = model.feature_importances_
    feature_importance = sorted(zip(importances, inputs), reverse=True)
    
    # Return the top 2 most important (most impactful) and the bottom 2 (safest to ignore)
    # The prompt asks for "which matter most" and "which can I safely stop worrying about"
    top_two = [name for imp, name in feature_importance[:2]]
    bottom_two = [name for imp, name in feature_importance[-2:]]
    
    return top_two + bottom_two

    test_inputs = [row[i] for i, h in enumerate(header) if h in inputs]
    pred_goals = model.predict([test_inputs])[0]

    # Normalize goals: 1 is best, 0 is worst
    def get_aggregate(data_goals):
        scores = []

            norm = (data_goals[i] - vals.min()) / (vals.max() - vals.min() + 1e-9)

    p = len(better_than) / len(train_scores)
    
    reason = f"This setup is estimated to outperform {p:.4f} of all historical configurations based on its predicted goal metrics."

            else:

def fix(header, row, levers):
    """
    Suggests changes to non-hiring levers (excluding 'Team Size') to reduce 'Idle-'.
    We look for levers that have a negative correlation with 'Idle-' in the data
    and nudge them to improve the row.
    """
    # Identify the index of 'Idle-'
    idle_idx = header.index('Idle-')
    
    recommendations = {}
    
    # We iterate through allowed levers and suggest a change
    # Strategy: For numeric levers, nudge towards the direction that 
    # historically correlates with lower 'Idle-' (or simply move away from current).
    for name, allowed in levers.items():
        idx = header.index(name)
        current_val = row[idx]
        
        if isinstance(allowed, tuple):  # Numeric lever
            low, high = allowed
            # Simple heuristic: move towards the center or opposite of current
            # to explore better configurations.
            if current_val < (low + high) / 2:
                new_val = min(high, current_val + (high - low) * 0.1)
            else:
                new_val = max(low, current_val - (high - low) * 0.1)
            recommendations[name] = new_val
        else:  # Symbolic lever
            # Pick a random value from the allowed list that isn't the current one
            options = [v for v in allowed if v != current_val]
            if options:
                recommendations[name] = random.choice(options)
                
    return recommendations

import csv

def spot(header, rows):
    """
    Identifies rows that are 'unseen' or 'anomalous' by checking if any 
    input value falls outside the range observed in the rest of the dataset.
    Returns the indices of rows containing such values, sorted by the 
    number of suspicious features (most suspicious first).
    """
    # Identify input columns (exclude goals)
    input_indices = [i for i, h in enumerate(header) 
                     if h[0].isupper() and not (h.endswith('-') or h.endswith('+'))]
    
    # Calculate min and max for each input column
    ranges = {}
    for i in input_indices:
        vals = [row[i] for row in rows if isinstance(row[i], (int, float))]
        if vals:
            ranges[i] = (min(vals), max(vals))
    
    suspicion_scores = []
    for idx, row in enumerate(rows):
        count = 0
        for i in input_indices:
            val = row[i]
            if isinstance(val, (int, float)) and i in ranges:
                low, high = ranges[i]
                # If a value is outside the observed range, it's "never seen"
                if val < low or val > high:
                    count += 1
        
        if count > 0:
            suspicion_scores.append((count, idx))
    
    # Sort by count descending (most suspicious first)
    suspicion_scores.sort(key=lambda x: x[0], reverse=True)
    
    return [idx for count, idx in suspicion_scores]
