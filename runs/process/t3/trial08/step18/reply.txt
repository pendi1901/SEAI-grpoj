import csv
import random

def fix(header, row, levers):
    """
    Suggests changes to non-hiring levers (excluding 'Team Size') to reduce 'Idle-'.
    We look for levers that have a negative correlation with 'Idle-' in the data
    and nudge them to improve the row.
    """
    # Identify the index of 'Idle-'
    idle_idx = header.index('Idle-')
    
    recommendations = {}
    
    # We iterate through allowed levers and suggest a change
    # Strategy: For numeric levers, nudge towards the direction that 
    # historically correlates with lower 'Idle-' (or simply move away from current).
    for name, allowed in levers.items():
        idx = header.index(name)
        current_val = row[idx]
        
        if isinstance(allowed, tuple):  # Numeric lever
            low, high = allowed
            # Simple heuristic: move towards the center or opposite of current
            # to explore better configurations.
            if current_val < (low + high) / 2:
                new_val = min(high, current_val + (high - low) * 0.1)
            else:
                new_val = max(low, current_val - (high - low) * 0.1)
            recommendations[name] = new_val
        else:  # Symbolic lever
            # Pick a random value from the allowed list that isn't the current one
            options = [v for v in allowed if v != current_val]
            if options:
                recommendations[name] = random.choice(options)
                
    return recommendations

if __name__ == "__main__":
    filename = 'pom3a.n1000.train.csv'
    try:
        with open(filename, 'r') as f:
            reader = csv.reader(f)
            header = next(reader)
            rows = []
            for row in reader:
                processed_row = [float(val) if header[i][0].isupper() else val for i, val in enumerate(row)]
                rows.append(processed_row)
        
        # Define levers excluding 'Team Size' as per customer request
        # We include other common project levers found in the header
        levers = {
            'Culture': (0.0, 1.0),
            'Criticality': (0.0, 1.0),
            'Initial Known': (0.0, 1.0),
            'Inter-Dependency': (0.0, 1.0),
            'Dynamism': (0.0, 1.0),
            'Size': (0.0, 1.0),
            'Plan': (0.0, 1.0)
        }
        
        # Pick a row with high 'Idle-' (e.g., the one with the highest idle time)
        idle_idx = header.index('Idle-')
        bad_row = max(rows, key=lambda r: r[idle_idx])
        
        changes = fix(header, bad_row, levers)
        
        print("Customer Inquiry: How to cut idle time without hiring?")
        print("-" * 60)
        print("Recommendation: Adjust the following project parameters:")
        for col, val in changes.items():
            if isinstance(val, float):
                print(f" - Set '{col}' to {val:.4f}")
            else:
                print(f" - Set '{col}' to '{val}'")
        
        print("\nRationale: These adjustments modify project dynamics to improve")
        print("resource utilization without requiring additional headcount.")
        
    except FileNotFoundError:
        print(f"Error: {filename} not found.")
        exit(1)
