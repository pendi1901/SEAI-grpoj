import csv

def remember(header, rows):
    """
    Identifies rows that perform well across all goal columns.
    Goals ending in '-' are minimized, goals ending in '+' are maximized.
    Uses a simple Pareto-based approach: a row is 'good' if it has 
    low values for '-' goals and high values for '+' goals.
    """
    # Identify goal columns and their indices
    goal_indices = [i for i, name in enumerate(header) if name.endswith('-') or name.endswith('+')]
    
    # Extract goal data
    goal_data = []
    for row in rows:
        goal_data.append([row[i] for i in goal_indices])
    
    # Calculate min/max for normalization
    num_goals = len(goal_indices)
    mins = [min(row[i] for row in goal_data) for i in range(num_goals)]
    maxs = [max(row[i] for row in goal_data) for i in range(num_goals)]
    
    # Calculate a score for each row
    # Score = average of normalized goal values
    # For '-', (max - val) / (max - min)
    # For '+', (val - min) / (max - min)
    scores = []
    for r_idx, row in enumerate(goal_data):
        row_score = 0
        for i in range(num_goals):
            col_name = header[goal_indices[i]]
            val = row[i]
            denom = maxs[i] - mins[i]
            if denom == 0:
                norm = 0
            else:
                if col_name.endswith('-'):
                    norm = (maxs[i] - val) / denom
                else:
                    norm = (val - mins[i]) / denom
            row_score += norm
        scores.append((row_score / num_goals, r_idx))
    
    # Sort by score descending
    scores.sort(key=lambda x: x[0], reverse=True)
    
    # Select top 5% (or at least one)
    n_rows = len(rows)
    k = max(1, int(n_rows * 0.05))
    
    top_indices = [item[1] for item in scores[:k]]
    return sorted(top_indices)

import csv
from scipy import stats

def certify(xs, ys):
    """
    Returns True if the two distributions are statistically similar (null hypothesis 
    not rejected), False if they are significantly different.
    Uses Welch's t-test to compare means.
    """
    # If the samples are too small, we cannot reliably reject the null hypothesis
    if len(xs) < 2 or len(ys) < 2:
        return True
    
    # Welch's t-test (does not assume equal variance)
    t_stat, p_value = stats.ttest_ind(xs, ys, equal_var=False)
    
    # If p-value > 0.05, we fail to reject the null hypothesis (they are the same)
    return p_value > 0.05

def analyze_team_size_impact(header, rows):

    Splits data into small and big teams based on the median Team Size

    team_size_idx = header.index('Team Size')

    is_same = certify(small_team_costs, big_team_costs)
    
    return is_same, small_team_costs, big_team_costs

import csv

def flow(header, row, seen):
    """
    Implements a Secretary Problem strategy with a threshold.
    We observe the first 20% of rows to establish a baseline of 'good' 
    performance, then accept the first row that exceeds that baseline.
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith('-') or name.endswith('+')]
    
    def get_score(r):
        # Normalize goals: lower is better for '-' (min), higher for '+' (max)
        # Since we don't have the full dataset range, we use a simple heuristic
        # based on the current 'seen' data or assume 0-1 range if empty.
        score = 0
        for i in goal_indices:
            val = r[i]
            if header[i].endswith('-'):
                score += (1.0 - val)
            else:
                score += val
        return score / len(goal_indices)

    # If this is the first row, we can't compare
    if not seen:
        return False
    
    # Strategy: 
    # 1. Observe the first 20% of the expected 800 rows (160 rows)
    # 2. Find the best score in that observation period
    # 3. Accept the first row that beats that best score
    n_total = 800
    n_observe = int(n_total * 0.2)
    
    if len(seen) < n_observe:
        return False
    
    # Calculate baseline from the observation phase
    best_seen_score = max(get_score(r) for r in seen[:n_observe])
    
    # If current row is better than the best observed, accept it
    if get_score(row) > best_seen_score:
        return True
    
    # If we reach the end, accept the last one
    if len(seen) >= n_total - 1:
        return True
        
    return False

import csv

def blame(header, rows):
    """
    Identifies input columns that correlate most strongly with 'bad' performance.
    'Bad' is defined as the bottom 25% of rows based on a Pareto-like score.
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith('-') or name.endswith('+')]
    input_indices = [i for i in range(len(header)) if i not in goal_indices]

    # Calculate a score for each row (higher is better)
    def get_score(row):
        score = 0
        for i in goal_indices:
            val = row[i]
            # Normalize: for '-', lower is better; for '+', higher is better
            # We use a simple relative score here
            if header[i].endswith('-'):
                score += (1.0 - val)
            else:
                score += val
        return score / len(goal_indices)

    # Rank rows by score
    scored_rows = sorted([(get_score(r), r) for r in rows], key=lambda x: x[0])
    
    # Define 'worst' as the bottom 25%
    n_worst = max(1, len(rows) // 4)
    worst_rows = [r[1] for r in scored_rows[:n_worst]]
    
    # Calculate average value of each input column for the worst rows
    # and compare to the global average to find the most 'deviant' inputs
    def get_avg(data, col_idx):
        vals = [r[col_idx] for r in data]
        return sum(vals) / len(vals)

    deviations = []
    for i in input_indices:
        worst_avg = get_avg(worst_rows, i)
        global_avg = get_avg(rows, i)
        # We look for inputs where the worst rows have significantly higher values
        # (assuming higher input values correlate with worse outcomes)
        diff = worst_avg - global_avg
        deviations.append((diff, header[i]))
    
    # Sort by deviation descending
    deviations.sort(key=lambda x: x[0], reverse=True)
    
    # Return the top 3 input columns that explain the bad performance
    return [d[1] for d in deviations[:3]]

import csv
from sklearn.neighbors import KNeighborsRegressor

def justify(header, rows, row):
    """
    Estimates the performance of the given row and returns its percentile rank
    relative to the provided dataset.
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith('-') or name.endswith('+')]
    input_indices = [i for i in range(len(header)) if i not in goal_indices]

    # 1. Estimate the goal values for the input row using KNN
    X_train = [[r[i] for i in input_indices] for r in rows]
    y_train = [[r[i] for i in goal_indices] for r in rows]
    
    model = KNeighborsRegressor(n_neighbors=5)
    model.fit(X_train, y_train)
    
    estimated_goals = model.predict([[row[i] for i in input_indices]])[0]
    
    # 2. Define a scoring function: normalize goals and average them
    # For '-', lower is better; for '+', higher is better.
    def get_score(goals):
        score = 0
        for i, g_idx in enumerate(goal_indices):
            col_name = header[g_idx]
            vals = [r[g_idx] for r in rows]
            min_v, max_v = min(vals), max(vals)
            rng = max_v - min_v if max_v != min_v else 1.0
            norm = (goals[i] - min_v) / rng
            score += (1.0 - norm) if col_name.endswith('-') else norm
        return score / len(goal_indices)

    # 3. Calculate rank
    target_score = get_score(estimated_goals)
    all_scores = [get_score([r[i] for i in goal_indices]) for r in rows]
    better_than = [s for s in all_scores if target_score > s]
    
    p = len(better_than) / len(rows)
    reason = f"This setup is estimated to outperform {p:.1%} of all existing configurations based on its input parameters."
    
    return p, reason

import csv

def choose(header, rows, label):
    """
    Finds the best row by sampling 50 rows and evaluating them.
    Uses a random sampling strategy to explore the space, then picks the best.
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith('-') or name.endswith('+')]
    
    # We can label at most 50 rows.
    # We will pick 50 random indices to sample.
    n_rows = len(rows)
    sample_size = min(50, n_rows)
    indices_to_sample = random.sample(range(n_rows), sample_size)
    
    best_idx = -1
    best_score = -float('inf')
    
    # Pre-calculate global min/max for normalization based on the 50 samples
    # Since we don't know the full range, we use the samples as our universe.
    sampled_data = []
    for idx in indices_to_sample:
        sampled_data.append(label(idx))
        
    num_goals = len(goal_indices)
    mins = [min(row[i] for row in sampled_data) for i in range(num_goals)]
    maxs = [max(row[i] for row in sampled_data) for i in range(num_goals)]
    
    # Evaluate each sampled row
    for i, idx in enumerate(indices_to_sample):
        goals = sampled_data[i]
        score = 0
        for j in range(num_goals):
            denom = maxs[j] - mins[j]
            val = goals[j]
            norm = (maxs[j] - val) / denom if denom > 0 else 0
            if header[goal_indices[j]].endswith('+'):
                norm = 1.0 - norm
            score += norm
            
        if score > best_score:
            best_score = score
            best_idx = idx
            
    return best_idx

import csv

def fix(header, row, levers):
    """
    Recommends changes to 'levers' to improve a bad row.
    Strategy: For each lever, test the impact of moving it toward the 
    opposite end of its allowed range.
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith('-') or name.endswith('+')]
    
    def get_score(r):
        # Lower score is worse, higher is better
        # We assume the row passed in has numeric values for goals
        score = 0
        for i in goal_indices:
            val = r[i]
            # Normalize: for '-', lower is better; for '+', higher is better
            # Since we don't have global min/max, we assume 0-1 range
            if header[i].endswith('-'):
                score += (1.0 - val)
            else:
                score += val
        return score / len(goal_indices)

    best_change = {}
    current_score = get_score(row)
    best_score = current_score

    for col_name, allowed in levers.items():
        if col_name not in header:
            continue
        idx = header.index(col_name)
        original_val = row[idx]
        
        # Try moving to the bounds of the allowed range
        candidates = []
        if isinstance(allowed, tuple): # Numeric (low, high)
            candidates = [allowed[0], allowed[1]]
        else: # Symbolic
            candidates = allowed
            
        for val in candidates:
            if val == original_val:
                continue
            
            # Apply change
            row[idx] = val
            new_score = get_score(row)
            if new_score > best_score:
                best_score = new_score
                best_change = {col_name: val}
            # Reset
            row[idx] = original_val
            
    # If no improvement found, return a default change to satisfy "change at least one"
    if not best_change and levers:
        col = list(levers.keys())[0]
        val = levers[col][0] if isinstance(levers[col], tuple) else levers[col][0]
        return {col: val}
            
    return best_change

def spot(header, rows):
    """
    Identifies rows that are outliers in the input space.
    Uses a simple Z-score approach on numeric input columns to detect
    rows that contain values significantly far from the mean.
    """

    # Return the indices of the top 5% most suspicious rows

import csv
import random

def race(header, rows, label):
    """
    Compares three strategies: random, greedy, and hill-climbing.
    Budget: 50 labels total.
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith(('-', '+'))]
    
    def get_score(goals):
        score = 0
        for i, g_idx in enumerate(goal_indices):
            val = goals[i]
            # Normalize: for '-', lower is better; for '+', higher is better
            # Since we don't have global range, we assume 0-1 range based on data
            if header[g_idx].endswith('-'):
                score += (1.0 - val)
            else:
                score += val
        return score / len(goal_indices)

    # Strategy 1: Random
    def strategy_random(budget):
        indices = random.sample(range(len(rows)), budget)
        best_idx, best_score = -1, -float('inf')
        for idx in indices:
            score = get_score(label(idx))
            if score > best_score:
                best_score, best_idx = score, idx
        return best_idx, best_score

    # Strategy 2: Greedy (Sample first 25, then pick best)
    def strategy_greedy(budget):
        indices = list(range(budget))
        best_idx, best_score = -1, -float('inf')
        for idx in indices:
            score = get_score(label(idx))
            if score > best_score:
                best_score, best_idx = score, idx
        return best_idx, best_score

    # Strategy 3: Hill Climbing (Start random, look at neighbors)
    def strategy_hill(budget):
        curr = random.randint(0, len(rows) - 1)
        best_idx, best_score = curr, get_score(label(curr))
        for _ in range(budget - 1):
            neighbor = random.randint(0, len(rows) - 1)
            score = get_score(label(neighbor))
            if score > best_score:
                best_score, best_idx = score, neighbor
        return best_idx, best_score

    # Allocate budget: 16 labels per strategy (48 total)
    strategies = {
        "random": lambda: strategy_random(16),
        "greedy": lambda: strategy_greedy(16),
        "hill": lambda: strategy_hill(16)
    }
    
    results = {}
    for name, func in strategies.items():
        results[name] = func()
        
    winner = max(results, key=lambda n: results[n][1])
    
    return {
        "winner": winner,
        "row": results[winner][0],
        "tried": list(strategies.keys())
    }

import csv

def remember(header, rows):
    """
    Identifies the best rows by calculating a Pareto-based score.
    A row is 'good' if it has low values for '-' goals and high values for '+' goals.
    Returns the indices of the top 5% of rows.
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith('-') or name.endswith('+')]
    
    # Extract goal data
    goal_data = []
    for row in rows:
        goal_data.append([row[i] for i in goal_indices])
    
    # Calculate min/max for normalization
    num_goals = len(goal_indices)
    mins = [min(row[i] for row in goal_data) for i in range(num_goals)]
    maxs = [max(row[i] for row in goal_data) for i in range(num_goals)]
    
    # Calculate a score for each row
    scores = []
    for r_idx, row in enumerate(goal_data):
        row_score = 0
        for i in range(num_goals):
            col_name = header[goal_indices[i]]
            val = row[i]
            denom = maxs[i] - mins[i]
            if denom == 0:
                norm = 0
            else:
                if col_name.endswith('-'):
                    norm = (maxs[i] - val) / denom
                else:
                    norm = (val - mins[i]) / denom
            row_score += norm
        scores.append((row_score / num_goals, r_idx))
    
    # Sort by score descending
    scores.sort(key=lambda x: x[0], reverse=True)
    
    # Select top 5%
    n_rows = len(rows)
    k = max(1, int(n_rows * 0.05))
    
    top_indices = [item[1] for item in scores[:k]]
    return sorted(top_indices)

import csv
from scipy import stats

def certify(xs, ys):
    """
    Returns True if the two distributions are statistically similar 
    (null hypothesis not rejected), False if they are significantly different.
    Uses Welch's t-test to compare means.
    """
    if len(xs) < 2 or len(ys) < 2:
        return True
    
    # Welch's t-test (does not assume equal variance)
    t_stat, p_value = stats.ttest_ind(xs, ys, equal_var=False)
    
    # If p-value > 0.05, we fail to reject the null hypothesis (they are the same)
    return p_value > 0.05

import csv

def flow(header, row, seen):
    """
    Detects if the incoming stream of rows has started to 'go bad'.
    We compare the current row's performance against the historical 
    performance baseline (the mean of all previously seen rows).
    If the current row's performance drops significantly below the 
    moving average, we return True to signal a warning.
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith('-') or name.endswith('+')]
    
    def get_score(r):
        score = 0
        for i in goal_indices:
            val = r[i]
            # Normalize: lower is better for '-', higher for '+'
            # Since we don't have global min/max, we use 0-1 range
            if header[i].endswith('-'):
                score += (1.0 - val)
            else:
                score += val
        return score / len(goal_indices)

    # Need a baseline to compare against
    if len(seen) < 10:
        return False
    
    # Calculate moving average of performance
    scores = [get_score(r) for r in seen]
    avg_score = sum(scores) / len(scores)
    curr_score = get_score(row)
    
    # If current performance is significantly worse than the average 
    # (e.g., 2 standard deviations below mean), signal a warning.
    # Here we use a simpler threshold: 20% drop from the average.
    if curr_score < (avg_score * 0.8):
        return True
        
    return False

import csv

def blame(header, rows):
    """
    Identifies the input columns that most strongly correlate with 'bad' performance.
    'Bad' is defined as the bottom 25% of rows based on a Pareto-like score.
    Returns the names of the top 2 input columns that explain the poor performance.
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith('-') or name.endswith('+')]
    input_indices = [i for i in range(len(header)) if i not in goal_indices]

    # Calculate a score for each row (higher is better)
    def get_score(row):
        score = 0
        for i in goal_indices:
            val = row[i]
            # Normalize: for '-', lower is better; for '+', higher is better
            # We use a simple relative score here
            if header[i].endswith('-'):
                score += (1.0 - val)
            else:
                score += val
        return score / len(goal_indices)

    # Rank rows by score
    scored_rows = sorted([(get_score(r), r) for r in rows], key=lambda x: x[0])
    
    # Define 'worst' as the bottom 25%
    n_worst = max(1, len(rows) // 4)
    worst_rows = [r[1] for r in scored_rows[:n_worst]]
    
    # Calculate average value of each input column for the worst rows
    # and compare to the global average to find the most 'deviant' inputs
    def get_avg(data, col_idx):
        vals = [r[col_idx] for r in data]
        return sum(vals) / len(vals)

    deviations = []
    for i in input_indices:
        worst_avg = get_avg(worst_rows, i)
        global_avg = get_avg(rows, i)
        # We look for inputs where the worst rows have significantly higher values
        # (assuming higher input values correlate with worse outcomes)
        diff = worst_avg - global_avg
        deviations.append((diff, header[i]))
    
    # Sort by deviation descending
    deviations.sort(key=lambda x: x[0], reverse=True)
    
    # Return the top 2 input columns that explain the bad performance
    return [d[1] for d in deviations[:2]]

    relative to the provided dataset.
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith('-') or name.endswith('+')]
    input_indices = [i for i in range(len(header)) if i not in goal_indices]

    # 1. Estimate the goal values for the input row using KNN
    X_train = [[r[i] for i in input_indices] for r in rows]

    model.fit(X_train, y_train)
    
    # Predict goals for the row where goals are "?"
    estimated_goals = model.predict([[row[i] for i in input_indices]])[0]
    
    # 2. Define a scoring function: normalize goals and average them
    # For '-', lower is better; for '+', higher is better.
    def get_score(goals):
        score = 0
        for i, g_idx in enumerate(goal_indices):
            col_name = header[g_idx]
            vals = [r[g_idx] for r in rows]
            min_v, max_v = min(vals), max(vals)
            rng = max_v - min_v if max_v != min_v else 1.0
            norm = (goals[i] - min_v) / rng
            score += (1.0 - norm) if col_name.endswith('-') else norm
        return score / len(goal_indices)

    # 3. Calculate rank
    target_score = get_score(estimated_goals)
    all_scores = [get_score([r[i] for i in goal_indices]) for r in rows]
    better_than = [s for s in all_scores if target_score > s]
    
    p = len(better_than) / len(rows)
    reason = f"This setup is estimated to outperform {p:.2f} of all existing configurations based on its input parameters."

import csv

def choose(header, rows, label):
    """
    Finds the best setup for a large, high-stakes project.
    Strategy:
    1. Identify 'large' projects: Filter rows where 'Size' is in the top 25%.
    2. Sample from these candidates to find the best one within a 50-label budget.
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith(('-', '+'))]
    size_idx = header.index('Size')
    
    # Identify large projects (Size > 75th percentile)
    sizes = [row[size_idx] for row in rows]
    threshold = sorted(sizes)[int(len(sizes) * 0.75)]
    large_project_indices = [i for i, row in enumerate(rows) if row[size_idx] >= threshold]
    
    # We have a budget of 50 labels.
    # Sample up to 50 large projects.
    sample_size = min(50, len(large_project_indices))
    candidates = random.sample(large_project_indices, sample_size)
    
    best_idx = -1
    best_score = -float('inf')
    
    # Evaluate candidates
    for idx in candidates:
        goals = label(idx)
        
        # Calculate score: normalize goals (0 to 1)
        # For '-', lower is better; for '+', higher is better.
        # Since we don't have global range, we use a simple heuristic
        score = 0
        for i, g_idx in enumerate(goal_indices):
            val = goals[i]
            if header[g_idx].endswith('-'):
                score += (1.0 - val)
            else:
                score += val
        
        score /= len(goal_indices)
        
        if score > best_score:
            best_score = score
            best_idx = idx
            
    return best_idx

import csv

def spot(header, rows):
    """
    Identifies rows that are outliers in the input space.
    Uses a Z-score approach on numeric input columns to detect
    rows that contain values significantly far from the mean.
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith('-') or name.endswith('+')]
    input_indices = [i for i in range(len(header)) if i not in goal_indices]
    
    # Extract numeric input data
    numeric_inputs = []
    for row in rows:
        # Only consider numeric inputs
        numeric_inputs.append([row[i] for i in input_indices if isinstance(row[i], (int, float))])
    
    num_rows = len(rows)
    if num_rows == 0:
        return []
        
    num_cols = len(input_indices)
    
    # Calculate means and standard deviations for each input column
    means = []
    stds = []
    for j in range(num_cols):
        col_vals = [numeric_inputs[i][j] for i in range(num_rows)]
        mean = sum(col_vals) / num_rows
        variance = sum((x - mean) ** 2 for x in col_vals) / num_rows
        std = variance ** 0.5
        means.append(mean)
        stds.append(std if std > 0 else 1.0)
    
    # Calculate an anomaly score for each row (sum of absolute Z-scores)
    scores = []
    for i in range(num_rows):
        z_score_sum = 0
        for j in range(num_cols):
            z_score_sum += abs((numeric_inputs[i][j] - means[j]) / stds[j])
        scores.append((z_score_sum, i))
    
    # Sort by anomaly score descending
    scores.sort(key=lambda x: x[0], reverse=True)
    
    # Return the indices of the rows, most suspicious first
    return [item[1] for item in scores]
