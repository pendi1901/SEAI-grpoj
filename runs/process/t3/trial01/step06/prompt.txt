import numpy as np


def remember(header, rows):
    goal_cols = [c for c in header if c.endswith('-')]

    n = len(rows)
    data = {}
    for col in goal_cols:
        vals = []
        for row in rows:
            v = row[header.index(col)]
            if v == '?':
                vals.append(np.nan)
            else:
                vals.append(float(v))
        data[col] = np.array(vals)

    min_vals = {}
    ranges = {}
    for col in goal_cols:
        arr = data[col]
        valid = arr[~np.isnan(arr)]
        if len(valid) == 0:
            continue
        mn = float(valid.min())
        mx = float(valid.max())
        min_vals[col] = mn
        ranges[col] = (mx - mn) if mx > mn else 1.0

    scores = []
    for i in range(n):
        s = 0.0
        for col in goal_cols:
            v = data[col][i]
            if np.isnan(v):
                continue
            s += (v - min_vals[col]) / ranges[col]
        scores.append(s)

    threshold_idx = int(0.05 * n)
    sorted_indices = np.argsort(scores)[:threshold_idx]
    return sorted_indices.tolist()

import numpy as np
from sklearn.linear_model import LinearRegression


def guess(header, train, test):
    goal_cols = [c for c in header if c.endswith('-')]
    first_goal = goal_cols[0]  # Cost-

    input_cols = [c for c in header if not c.endswith('-')]

    def parse_row(row):
        vals = []
        for col in input_cols:
            v = row[header.index(col)]
            try:
                vals.append(float(v))
            except (ValueError, TypeError):
                vals.append(np.nan)
        return np.array(vals)

    X_train = np.array([parse_row(r) for r in train])
    y_train = []
    for row in train:
        v = row[header.index(first_goal)]
        if v == '?':
            y_train.append(np.nan)
        else:
            y_train.append(float(v))

    X_test = np.array([parse_row(r) for r in test])
    y_test = []
    for row in test:
        v = row[header.index(first_goal)]
        if v == '?':
            y_test.append(np.nan)
        else:
            y_test.append(float(v))

    valid_mask = ~np.isnan(X_train).any(axis=1) & ~np.isnan(y_train)
    model = LinearRegression()
    model.fit(X_train[valid_mask], y_train[valid_mask])

    predictions = []
    for i in range(len(test)):
        x = X_test[i]
        if np.isnan(x).any():
            preds = model.predict(X_train[valid_mask].reshape(1, -1))
            predictions.append(float(preds[0]))
        else:
            predictions.append(float(model.predict([x])[0]))

    return predictions

import numpy as np


def certify(xs, ys):
    """Returns True if xs and ys are the same apart from noise."""
    arr_x = np.array(xs)
    arr_y = np.array(ys)
    
    mean_diff = abs(np.mean(arr_x) - np.mean(arr_y))
    std_x = float(np.std(arr_x))
    std_y = float(np.std(arr_y))
    combined_std = np.sqrt(std_x**2 + std_y**2) if (std_x > 0 or std_y > 0) else 1.0
    
    return mean_diff < 2 * combined_std

def flow(header, row, seen):
    goal_cols = [c for c in header if c.endswith('-')]

    def parse_goals(row_dict):
        vals = []
        for col in goal_cols:
            v = row_dict[header.index(col)]
            try:
                vals.append(float(v))
            except (ValueError, TypeError):
                return None
        return vals

    current = parse_goals(row)
    if current is None:
        return False

    if not seen:
        return True

    best = {}
    for s in seen:
        sg = parse_goals(s)
        if sg is None:
            continue
        for col, val in zip(goal_cols, sg):
            if col not in best or val < best[col]:
                best[col] = val

    for i, col in enumerate(goal_cols):
        if current[i] > best[col]:
            return False

    return True

import numpy as np


def blame(header, rows):
    goal_cols = [c for c in header if c.endswith('-')]
    
    def parse_row(row_dict):
        vals = {}
        for col in header:
            v = row_dict[header.index(col)]
            try:
                vals[col] = float(v)
            except (ValueError, TypeError):
                vals[col] = np.nan
        return vals
    
    parsed = [parse_row(r) for r in rows]
    
    idle_col = 'Idle-'
    idle_vals = []
    for p in parsed:
        v = p[idle_col]
        if not np.isnan(v):
            idle_vals.append(v)
    
    n = len(rows)
    worst_idx = int(0.1 * n)
    sorted_by_idle = np.argsort(idle_vals)[::-1][:worst_idx].tolist()
    
    input_cols = [c for c in header if not c.endswith('-')]
    
    correlations = []
    for col in input_cols:
        vals = []
        for i, p in enumerate(parsed):
            v = p[col]
            if np.isnan(v) or not np.isfinite(v):
                continue
            vals.append((v, idle_vals[i]))
        
        if len(vals) < 3:
            correlations.append(0.0)
            continue
        
        xs = [x for x, _ in vals]
        ys = [y for _, y in vals]
        
        mean_x = np.mean(xs)
        mean_y = np.mean(ys)
        std_x = float(np.std(xs))
        std_y = float(np.std(ys))
        
        if std_x < 1e-9 or std_y < 1e-9:
            correlations.append(0.0)
            continue
        
        corr = np.sum((np.array(xs) - mean_x) * (np.array(ys) - mean_y)) / (std_x * std_y)
        correlations.append(float(corr))
    
    ranked = sorted(range(len(correlations)), key=lambda i: correlations[i], reverse=True)
    
    result = []
    for rank in ranked[:5]:
        col = input_cols[rank]
        if abs(correlations[rank]) > 0.1:
            result.append(col)
    
    return result
