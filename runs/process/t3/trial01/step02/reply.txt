```python
import csv
from sklearn.neighbors import KNeighborsRegressor

def guess(header, train, test):
    """
    Estimates the value of the first goal column for each test row
    using K-Nearest Neighbors based on input columns.
    """
    # Identify goal columns and input columns
    goal_indices = [i for i, name in enumerate(header) if name.endswith('-') or name.endswith('+')]
    input_indices = [i for i in range(len(header)) if i not in goal_indices]
    
    # Target is the first goal column
    target_idx = goal_indices[0]
    
    # Prepare training data (inputs only)
    X_train = [[row[i] for i in input_indices] for row in train]
    y_train = [row[target_idx] for row in train]
    
    # Prepare test data (inputs only)
    X_test = [[row[i] for i in input_indices] for row in test]
    
    # Use KNN to predict the target value
    # k=5 is a standard heuristic for small datasets
    model = KNeighborsRegressor(n_neighbors=5)
    model.fit(X_train, y_train)
    
    return model.predict(X_test).tolist()

if __name__ == "__main__":
    filename = 'pom3a.n1000.train.csv'
    try:
        with open(filename, 'r') as f:
            reader = csv.reader(f)
            header = next(reader)
            rows = []
            for row in reader:
                rows.append([float(val) for val in row])
        
        # Split data: use 80% for training, 20% for testing
        split_idx = int(len(rows) * 0.8)
        train_data = rows[:split_idx]
        test_data = rows[split_idx:]
        
        # Mask the goals in test data to simulate the "?" requirement
        goal_indices = [i for i, name in enumerate(header) if name.endswith('-') or name.endswith('+')]
        masked_test = []
        for row in test_data:
            new_row = list(row)
            for idx in goal_indices:
                new_row[idx] = None # Representing "?"
            masked_test.append(new_row)
            
        predictions = guess(header, train_data, masked_test)
        
        # Print summary for the customer
        print("Customer Inquiry: Estimated Cost (First Goal Column)")
        print("-" * 50)
        for i in range(min(5, len(predictions))):
            print(f"Test Row {i+1}: Predicted Cost = {predictions[i]:.4f}")
        print("...")
        print(f"Total predictions made: {len(predictions)}")
        
    except Exception as e:
        print(f"Error: {e}")
        exit(1)
```
