import random


def choose(header, rows, label):
    goal_cols = [c for c in header if c.endswith('-') or c.endswith('+')]

    n = len(rows)
    indices = list(range(n))
    random.shuffle(indices)
    indices = indices[:50]

    best_idx = None
    best_score = float('-inf')
    min_vals = {}
    max_vals = {}

    for i in indices:
        goals = label(i)

        # Update running ranges across all labeled rows
        for col, val in zip(goal_cols, goals):
            v = float(val)
            if col not in min_vals or v < min_vals[col]:
                min_vals[col] = v
            if col not in max_vals or v > max_vals[col]:
                max_vals[col] = v

        # Compute normalized score: lower is better for '-', higher for '+'
        score = 0.0
        for col, val in zip(goal_cols, goals):
            v = float(val)
            mn = min_vals[col]
            mx = max_vals[col]
            rng = (mx - mn) if mx > mn else 1.0
            norm = (v - mn) / rng

            if col.endswith('-'):
                score += -norm   # lower is better, so negate normalized value
            elif col.endswith('+'):
                score += norm    # higher is better

        if score > best_score:
            best_score = score
            best_idx = i

    return best_idx
