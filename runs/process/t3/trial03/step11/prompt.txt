import csv
import pandas as pd
import numpy as np

def remember(header, rows):
    """
    Identifies rows that perform well across all goal columns.
    Goals ending in '-' are minimized, '+' are maximized.
    """
    df = pd.DataFrame(rows, columns=header)
    
    # Identify goal columns
    goals = [c for c in header if c.endswith('-') or c.endswith('+')]
    
    # Normalize goals to [0, 1] range to treat them equally
    # For '-', lower is better (1 - normalized_val)
    # For '+', higher is better (normalized_val)
    scores = []
    for col in goals:
        vals = df[col].astype(float)
        norm = (vals - vals.min()) / (vals.max() - vals.min())
        if col.endswith('-'):
            scores.append(1 - norm)
        else:
            scores.append(norm)
            
    # Aggregate scores (mean of normalized goals)
    # Higher aggregate score means better performance
    df['aggregate'] = np.mean(scores, axis=0)
    
    # Select top 5% (or at least 1)
    n_rows = len(rows)
    k = max(1, n_rows // 20)
    
    top_indices = df['aggregate'].nlargest(k).index.tolist()
    return top_indices

def certify(xs, ys):
    """
    Returns True if the difference between two distributions is statistically 
    insignificant (likely noise), False if they are significantly different.
    Uses Welch's t-test for independent samples with unequal variances.
    """
    # Use a standard significance level (alpha = 0.05)
    # If p-value > 0.05, we fail to reject the null hypothesis (they are the same)
    t_stat, p_value = stats.ttest_ind(xs, ys, equal_var=False)
    return p_value > 0.05

def analyze_team_size_impact(header, rows):
    """
    Splits data into small and big team setups and compares their costs.
    """
    df = pd.DataFrame(rows, columns=header)
    
    # Define 'small' vs 'big' based on the median of 'Team Size'
    median_team_size = df['Team Size'].median()
    
    small_teams = df[df['Team Size'] <= median_team_size]['Cost-']
    big_teams = df[df['Team Size'] > median_team_size]['Cost-']
    
    is_same = certify(small_teams.tolist(), big_teams.tolist())
    
    return small_teams.mean(), big_teams.mean(), is_same

import csv
import numpy as np

def flow(header, row, seen):
    """
    Decides whether to accept a row based on whether it is better than 
    the best seen so far. Uses a simple threshold strategy: accept if 
    the current row is in the top 5% of the observed distribution.
    """
    # Identify goal columns
    goal_indices = [i for i, name in enumerate(header) if name.endswith('-') or name.endswith('+')]
    
    def get_score(r):
        # Normalize goals: 1 - val for '-', val for '+' (assuming [0,1] range)
        scores = []
        for i in goal_indices:
            val = r[i]
            if header[i].endswith('-'):
                scores.append(1.0 - val)
            else:
                scores.append(val)
        return sum(scores) / len(scores)

    # If this is the first row, accept it
    if not seen:
        return True
    
    # Calculate score for current row
    current_score = get_score(row)
    
    # Calculate scores for all seen rows
    seen_scores = [get_score(r) for r in seen]
    
    # Threshold: Accept if current row is better than 95% of what we've seen
    # or if it's the best we've seen so far.
    threshold = np.percentile(seen_scores, 95)
    
    return current_score >= threshold

def justify(header, rows, row):
    """
    Estimates the goals for the given row using KNN, then calculates its 
    percentile rank against the provided dataset.
    """
    df = pd.DataFrame(rows, columns=header)
    goals = [c for c in header if c.endswith('-') or c.endswith('+')]
    inputs = [c for c in header if c not in goals]

    # 1. Estimate goals for the input row
    X_train = df[inputs]
    y_train = df[goals]
    model = KNeighborsRegressor(n_neighbors=5)
    model.fit(X_train, y_train)
    
    # Extract inputs from the provided row (excluding the "?" goal placeholders)
    row_inputs = [row[i] for i, col in enumerate(header) if col in inputs]
    estimated_goals = model.predict([row_inputs])[0]

    # 2. Calculate aggregate scores for all rows (including estimated one)
    def get_aggregate(r_goals):
        scores = []
        for i, col in enumerate(goals):
            vals = df[col]
            norm = (r_goals[i] - vals.min()) / (vals.max() - vals.min())
            scores.append(1 - norm if col.endswith('-') else norm)
        return np.mean(scores)

    # Calculate score for the estimated row
    row_score = get_aggregate(estimated_goals)
    
    # Calculate scores for all existing rows
    all_scores = []
    for _, r in df.iterrows():
        all_scores.append(get_aggregate(r[goals].tolist()))
    
    # 3. Calculate percentile rank (fraction of rows better than this one)
    # A row is better if its score is higher
    better_than = [s for s in all_scores if s < row_score]
    p = len(better_than) / len(all_scores)
    
    reason = f"This setup is estimated to be better than {p:.1%} of all known configurations based on its input parameters."
    
    return p, reason

import csv
import numpy as np
import pandas as pd
from sklearn.ensemble import RandomForestRegressor

def choose(header, rows, label):
    """
    Finds the best row by using an active learning approach:
    1. Use initial random samples to train a surrogate model.
    2. Use the model to predict goals for all rows.
    3. Use Expected Improvement (or similar) to pick the next best candidate.
    """
    goal_cols = [c for c in header if c.endswith('-') or c.endswith('+')]
    input_cols = [c for c in header if c not in goal_cols]
    
    df = pd.DataFrame(rows, columns=header)
    
    # Track which rows we have labeled
    labeled_indices = []
    
    # 1. Initial exploration: Sample 10 random rows to build a baseline
    np.random.seed(42)
    initial_indices = np.random.choice(len(rows), 10, replace=False).tolist()
    
    # 2. Iterative refinement: 40 more queries
    # We use a Random Forest to predict the aggregate score of rows
    for _ in range(40):
        # Update labels
        for idx in initial_indices:
            if idx not in labeled_indices:
                true_goals = label(idx)
                for i, col in enumerate(goal_cols):
                    rows[idx][header.index(col)] = true_goals[i]
                labeled_indices.append(idx)
        
        # Train model on currently labeled data
        df_labeled = pd.DataFrame([r for i, r in enumerate(rows) if i in labeled_indices], columns=header)
        
        # Calculate aggregate score for labeled data
        def get_agg(row_data):
            scores = []
            for col in goal_cols:
                val = row_data[col]
                # Normalize based on current labeled range
                col_min, col_max = df_labeled[col].min(), df_labeled[col].max()
                norm = (val - col_min) / (col_max - col_min + 1e-9)
                scores.append(1 - norm if col.endswith('-') else norm)
            return np.mean(scores)
        
        y = df_labeled.apply(get_agg, axis=1)
        X = df_labeled[input_cols]
        
        model = RandomForestRegressor(n_estimators=50)
        model.fit(X, y)
        
        # Predict for all unlabeled rows
        unlabeled_indices = [i for i in range(len(rows)) if i not in labeled_indices]
        if not unlabeled_indices:
            break
            
        X_unlabeled = df[input_cols].iloc[unlabeled_indices]
        preds = model.predict(X_unlabeled)
        
        # Pick the best predicted row
        best_unlabeled = unlabeled_indices[np.argmax(preds)]
        initial_indices = [best_unlabeled]
        
    # Final evaluation: return index of the best labeled row
    best_idx = labeled_indices[0]
    best_score = -1
    for idx in labeled_indices:
        s = get_agg(pd.Series({c: rows[idx][header.index(c)] for c in goal_cols}))
        if s > best_score:
            best_score = s
            best_idx = idx
            
    return best_idx

def fix(header, row, levers):
    """
    Recommends changes to a bad row to improve its goals.
    Uses a Random Forest to identify which lever change yields the best 
    predicted improvement in the aggregate goal score.
    """
    # Load data to build a surrogate model
    filename = 'pom3a.n1000.train.csv'
    df = pd.read_csv(filename)
    
    goals = [c for c in header if c.endswith('-') or c.endswith('+')]
    inputs = [c for c in header if c not in goals]
    
    # Normalize goals for aggregate score
    def get_agg(r):
        scores = []
        for col in goals:
            vals = df[col]
            norm = (r[col] - vals.min()) / (vals.max() - vals.min() + 1e-9)
            scores.append(1 - norm if col.endswith('-') else norm)
        return np.mean(scores)

    # Train model to predict aggregate score
    df['aggregate'] = df.apply(get_agg, axis=1)
    model = RandomForestRegressor(n_estimators=50, random_state=42)
    model.fit(df[inputs], df['aggregate'])
    
    best_change = {}
    best_score = -1
    
    # Evaluate each lever
    for col, bounds in levers.items():
        if col not in header: continue
        
        # Test a few values within the range
        test_vals = []
        if isinstance(bounds, list): # Symbolic
            test_vals = bounds
        else: # Numeric (low, high)
            test_vals = [bounds[0], (bounds[0] + bounds[1]) / 2, bounds[1]]
            
        for val in test_vals:
            temp_row = dict(zip(header, row))
            temp_row[col] = val
            
            # Predict score
            input_vec = pd.DataFrame([temp_row])[inputs]
            pred_score = model.predict(input_vec)[0]
            
            if pred_score > best_score:
                best_score = pred_score
                best_change = {col: val}
                
    return best_change

import csv

from sklearn.ensemble import IsolationForest

def spot(header, rows):
    """
    Identifies rows that are outliers (potentially mistyped or made up)
    using the Isolation Forest algorithm.
    Returns the indices of the rows, sorted by suspicion (most suspicious first).
    """
    df = pd.DataFrame(rows, columns=header)
    
    # Identify numeric columns for anomaly detection
    # We exclude goals as they are outcomes, not inputs that could be "mistyped"
    # in the context of project configuration.
    goals = [c for c in header if c.endswith('-') or c.endswith('+')]
    inputs = [c for c in header if c not in goals]
    
    # Prepare numeric input data
    X = df[inputs].select_dtypes(include=[np.number])
    
    # IsolationForest returns -1 for outliers and 1 for inliers.
    # The decision_function returns the anomaly score (lower is more anomalous).
    clf = IsolationForest(contamination=0.05, random_state=42)
    clf.fit(X)
    
    # Get anomaly scores (lower = more suspicious)
    scores = clf.decision_function(X)
    
    # Get indices of rows sorted by score (ascending, so lowest scores first)
    suspicious_indices = np.argsort(scores)
    
    return suspicious_indices.tolist()

def get_score(header, row_goals, df_full):
    """Calculates normalized aggregate score for a row."""
    goals = [c for c in header if c.endswith('-') or c.endswith('+')]
    scores = []
    for i, col in enumerate(goals):
        vals = df_full[col].astype(float)
        norm = (row_goals[i] - vals.min()) / (vals.max() - vals.min() + 1e-9)
        scores.append(1 - norm if col.endswith('-') else norm)
    return np.mean(scores)

def race(header, rows, label):
    """
    Compares three strategies: random, greedy, and uncertainty-based.
    Total budget: 50 labels.
    """
    goal_cols = [c for c in header if c.endswith('-') or c.endswith('+')]
    input_cols = [c for c in header if c not in goal_cols]
    
    # Ensure numeric types for processing
    df_full = pd.DataFrame(rows, columns=header)
    for col in header:
        if col not in goal_cols:
            df_full[col] = pd.to_numeric(df_full[col], errors='ignore')
    
    labeled = {}
    budget = 50
    
    def run_strategy(name, n_trials):
        nonlocal budget
        strat_labeled = {}
        
        for _ in range(n_trials):
            if budget <= 0: break
            
            # Pick index
            if name == "random":
                idx = np.random.choice(len(rows))
            else:
                # Train on all currently labeled data
                df_l = pd.DataFrame([rows[i] for i in labeled], columns=header)
                for i, col in enumerate(goal_cols):
                    df_l[col] = [labeled[idx][i] for idx in labeled]
                
                X = df_l[input_cols]
                y = df_l[goal_cols[0]]
                model = RandomForestRegressor(n_estimators=20).fit(X, y)
                
                if name == "greedy":
                    preds = model.predict(df_full[input_cols])
                    idx = np.argmax(preds)
                else: # uncertainty
                    preds = [tree.predict(df_full[input_cols]) for tree in model.estimators_]
                    idx = np.argmax(np.std(preds, axis=0))
            
            if idx not in labeled:
                labeled[idx] = label(idx)
                strat_labeled[idx] = labeled[idx]
                budget -= 1
        
        # Evaluate this strategy's performance
        best_idx, best_score = -1, -1
        for idx, g in strat_labeled.items():
            s = get_score(header, g, df_full)
            if s > best_score:
                best_score, best_idx = s, idx
        return best_idx, best_score

    s1_idx, s1_score = run_strategy("random", 16)
    s2_idx, s2_score = run_strategy("greedy", 17)
    s3_idx, s3_score = run_strategy("uncertainty", 17)
    
    results = {"random": s1_score, "greedy": s2_score, "uncertainty": s3_score}
    winner = max(results, key=results.get)
    winner_idx = {"random": s1_idx, "greedy": s2_idx, "uncertainty": s3_idx}[winner]
    
    return {"winner": winner, "row": winner_idx, "tried": ["random", "greedy", "uncertainty"]}
