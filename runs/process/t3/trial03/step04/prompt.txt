import csv
import pandas as pd
import numpy as np

def remember(header, rows):
    """
    Identifies rows that perform well across all goal columns.
    Goals ending in '-' are minimized, '+' are maximized.
    """
    df = pd.DataFrame(rows, columns=header)
    
    # Identify goal columns
    goals = [c for c in header if c.endswith('-') or c.endswith('+')]
    
    # Normalize goals to [0, 1] range to treat them equally
    # For '-', lower is better (1 - normalized_val)
    # For '+', higher is better (normalized_val)
    scores = []
    for col in goals:
        vals = df[col].astype(float)
        norm = (vals - vals.min()) / (vals.max() - vals.min())
        if col.endswith('-'):
            scores.append(1 - norm)
        else:
            scores.append(norm)
            
    # Aggregate scores (mean of normalized goals)
    # Higher aggregate score means better performance
    df['aggregate'] = np.mean(scores, axis=0)
    
    # Select top 5% (or at least 1)
    n_rows = len(rows)
    k = max(1, n_rows // 20)
    
    top_indices = df['aggregate'].nlargest(k).index.tolist()
    return top_indices

import csv

from sklearn.neighbors import KNeighborsRegressor

def guess(header, train, test):
    """
    Estimates the first goal column for each row in test using K-Nearest Neighbors.
    """
    df_train = pd.DataFrame(train, columns=header)
    df_test = pd.DataFrame(test, columns=header)
    
    # Identify goal columns
    goals = [c for c in header if c.endswith('-') or c.endswith('+')]
    first_goal = goals[0]
    
    # Inputs are all columns that are not goals
    inputs = [c for c in header if c not in goals]
    
    # Prepare data
    X_train = df_train[inputs]
    y_train = df_train[first_goal]
    X_test = df_test[inputs]
    
    # Use KNN to predict the first goal

    model = KNeighborsRegressor(n_neighbors=5)

import csv
import pandas as pd
import numpy as np
from scipy import stats

def certify(xs, ys):
    """
    Returns True if the difference between two distributions is statistically 
    insignificant (likely noise), False if they are significantly different.
    Uses Welch's t-test for independent samples with unequal variances.
    """
    # Use a standard significance level (alpha = 0.05)
    # If p-value > 0.05, we fail to reject the null hypothesis (they are the same)
    t_stat, p_value = stats.ttest_ind(xs, ys, equal_var=False)
    return p_value > 0.05

def analyze_team_size_impact(header, rows):
    """
    Splits data into small and big team setups and compares their costs.
    """
    df = pd.DataFrame(rows, columns=header)
    
    # Define 'small' vs 'big' based on the median of 'Team Size'
    median_team_size = df['Team Size'].median()
    
    small_teams = df[df['Team Size'] <= median_team_size]['Cost-']
    big_teams = df[df['Team Size'] > median_team_size]['Cost-']
    
    is_same = certify(small_teams.tolist(), big_teams.tolist())
    
    return small_teams.mean(), big_teams.mean(), is_same

                processed_row = [float(val) if val.replace('.', '', 1).isdigit() or

            print("The team size does not have a statistically significant impact on cost.")
        else:
            print("\nConclusion: The difference is statistically significant.")
            print("Small-team setups are genuinely cheaper than big-team ones.")
