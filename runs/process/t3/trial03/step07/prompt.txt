import csv
import pandas as pd
import numpy as np

def remember(header, rows):
    """
    Identifies rows that perform well across all goal columns.
    Goals ending in '-' are minimized, '+' are maximized.
    """
    df = pd.DataFrame(rows, columns=header)
    
    # Identify goal columns
    goals = [c for c in header if c.endswith('-') or c.endswith('+')]
    
    # Normalize goals to [0, 1] range to treat them equally
    # For '-', lower is better (1 - normalized_val)
    # For '+', higher is better (normalized_val)
    scores = []
    for col in goals:
        vals = df[col].astype(float)
        norm = (vals - vals.min()) / (vals.max() - vals.min())
        if col.endswith('-'):
            scores.append(1 - norm)
        else:
            scores.append(norm)
            
    # Aggregate scores (mean of normalized goals)
    # Higher aggregate score means better performance
    df['aggregate'] = np.mean(scores, axis=0)
    
    # Select top 5% (or at least 1)
    n_rows = len(rows)
    k = max(1, n_rows // 20)
    
    top_indices = df['aggregate'].nlargest(k).index.tolist()
    return top_indices

import csv
import pandas as pd

from scipy import stats

def certify(xs, ys):
    """
    Returns True if the difference between two distributions is statistically 
    insignificant (likely noise), False if they are significantly different.
    Uses Welch's t-test for independent samples with unequal variances.
    """
    # Use a standard significance level (alpha = 0.05)
    # If p-value > 0.05, we fail to reject the null hypothesis (they are the same)
    t_stat, p_value = stats.ttest_ind(xs, ys, equal_var=False)
    return p_value > 0.05

def analyze_team_size_impact(header, rows):
    """
    Splits data into small and big team setups and compares their costs.
    """
    df = pd.DataFrame(rows, columns=header)
    
    # Define 'small' vs 'big' based on the median of 'Team Size'
    median_team_size = df['Team Size'].median()
    
    small_teams = df[df['Team Size'] <= median_team_size]['Cost-']
    big_teams = df[df['Team Size'] > median_team_size]['Cost-']
    
    is_same = certify(small_teams.tolist(), big_teams.tolist())
    
    return small_teams.mean(), big_teams.mean(), is_same

import csv
import numpy as np

def flow(header, row, seen):
    """
    Decides whether to accept a row based on whether it is better than 
    the best seen so far. Uses a simple threshold strategy: accept if 
    the current row is in the top 5% of the observed distribution.
    """
    # Identify goal columns
    goal_indices = [i for i, name in enumerate(header) if name.endswith('-') or name.endswith('+')]
    
    def get_score(r):
        # Normalize goals: 1 - val for '-', val for '+' (assuming [0,1] range)
        scores = []
        for i in goal_indices:
            val = r[i]
            if header[i].endswith('-'):
                scores.append(1.0 - val)
            else:
                scores.append(val)
        return sum(scores) / len(scores)

    # If this is the first row, accept it
    if not seen:
        return True
    
    # Calculate score for current row
    current_score = get_score(row)
    
    # Calculate scores for all seen rows
    seen_scores = [get_score(r) for r in seen]
    
    # Threshold: Accept if current row is better than 95% of what we've seen
    # or if it's the best we've seen so far.
    threshold = np.percentile(seen_scores, 95)
    
    return current_score >= threshold

import csv
import pandas as pd
import numpy as np
from sklearn.neighbors import KNeighborsRegressor

def justify(header, rows, row):
    """
    Estimates the goals for the given row using KNN, then calculates its 
    percentile rank against the provided dataset.
    """
    df = pd.DataFrame(rows, columns=header)
    goals = [c for c in header if c.endswith('-') or c.endswith('+')]
    inputs = [c for c in header if c not in goals]

    # 1. Estimate goals for the input row
    X_train = df[inputs]
    y_train = df[goals]
    model = KNeighborsRegressor(n_neighbors=5)
    model.fit(X_train, y_train)
    
    # Extract inputs from the provided row (excluding the "?" goal placeholders)
    row_inputs = [row[i] for i, col in enumerate(header) if col in inputs]
    estimated_goals = model.predict([row_inputs])[0]

    # 2. Calculate aggregate scores for all rows (including estimated one)
    def get_aggregate(r_goals):
        scores = []
        for i, col in enumerate(goals):
            vals = df[col]
            norm = (r_goals[i] - vals.min()) / (vals.max() - vals.min())
            scores.append(1 - norm if col.endswith('-') else norm)
        return np.mean(scores)

    # Calculate score for the estimated row
    row_score = get_aggregate(estimated_goals)
    
    # Calculate scores for all existing rows
    all_scores = []
    for _, r in df.iterrows():
        all_scores.append(get_aggregate(r[goals].tolist()))
    
    # 3. Calculate percentile rank (fraction of rows better than this one)
    # A row is better if its score is higher
    better_than = [s for s in all_scores if s < row_score]
    p = len(better_than) / len(all_scores)
    
    reason = f"This setup is estimated to be better than {p:.1%} of all known configurations based on its input parameters."
    
    return p, reason
