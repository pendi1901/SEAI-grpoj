import csv
import pandas as pd
import numpy as np

def remember(header, rows):
    """
    Identifies rows that perform well across all goal columns.
    Goals ending in '-' are minimized, '+' are maximized.
    """
    df = pd.DataFrame(rows, columns=header)
    
    # Identify goal columns
    goals = [c for c in header if c.endswith('-') or c.endswith('+')]
    
    # Normalize goals to [0, 1] range to treat them equally
    # For '-', lower is better (1 - normalized_val)
    # For '+', higher is better (normalized_val)
    normalized_goals = pd.DataFrame(index=df.index)
    for goal in goals:
        col = df[goal].astype(float)
        norm = (col - col.min()) / (col.max() - col.min())
        if goal.endswith('-'):
            normalized_goals[goal] = 1 - norm
        else:
            normalized_goals[goal] = norm
            
    # Calculate a score: the product of normalized goals (geometric mean)
    # This favors rows that are good across all objectives simultaneously
    scores = normalized_goals.prod(axis=1)
    
    # Select top 5% (or at least 1)
    n_select = max(1, int(len(rows) * 0.05))
    best_indices = scores.nlargest(n_select).index.tolist()
    
    return sorted(best_indices)

import csv
import pandas as pd
from scipy import stats

def certify(xs, ys):
    """
    Returns True if the distributions of xs and ys are statistically 
    indistinguishable (same apart from noise), False if they differ.
    Uses Welch's t-test for independent samples.
    """
    # If samples are too small, we cannot reject the null hypothesis
    if len(xs) < 2 or len(ys) < 2:
        return True
    
    # Perform Welch's t-test (does not assume equal variance)
    t_stat, p_value = stats.ttest_ind(xs, ys, equal_var=False)
    
    # If p-value > 0.05, we fail to reject the null hypothesis (they are the same)
    return p_value > 0.05

def analyze_team_size_impact(header, rows):
    """
    Splits data into small and big team setups and compares their costs.
    """
    df = pd.DataFrame(rows, columns=header)
    
    # Identify 'Team Size' and 'Cost-'
    # Team Size is an input, Cost- is a goal
    median_team_size = df['Team Size'].median()
    
    small_teams = df[df['Team Size'] <= median_team_size]['Cost-']
    big_teams = df[df['Team Size'] > median_team_size]['Cost-']
    
    is_same = certify(small_teams.tolist(), big_teams.tolist())
    
    return is_same, small_teams.mean(), big_teams.mean()

import csv

def flow(header, row, seen):
    """
    Implements a Secretary Problem strategy with a threshold.
    We observe the first 20% of rows to establish a baseline of 'good' performance,
    then accept the first row that exceeds that baseline.
    """
    goal_indices = [i for i, h in enumerate(header) if h.endswith('-') or h.endswith('+')]
    
    def get_score(r):
        # Normalize goals: 1 - (val - min)/(max - min) for '-'
        # (val - min)/(max - min) for '+'
        # Since we don't have global min/max, we use the 'seen' data as a proxy.
        scores = []
        for idx in goal_indices:
            col_vals = [s[idx] for s in seen + [r]]
            min_v, max_v = min(col_vals), max(col_vals)
            rng = max_v - min_v if max_v != min_v else 1
            norm = (r[idx] - min_v) / rng
            scores.append(1 - norm if header[idx].endswith('-') else norm)
        return sum(scores) / len(scores)

    # If it's the first row, we can't evaluate it yet
    if not seen:
        return False
    
    # Strategy: Observe first 20% to set a benchmark
    n = len(seen) + 1
    limit = max(1, int(800 * 0.2))
    
    if n <= limit:
        return False
    
    # Calculate benchmark from seen rows
    benchmarks = [get_score(s) for s in seen]
    best_seen = max(benchmarks)
    
    # Accept if current row is better than anything seen so far
    return get_score(row) > best_seen

if __name__ == "__main__":

import csv

from sklearn.ensemble import RandomForestRegressor

def blame(header, rows):
    """
    Identifies input columns that correlate most strongly with the 'Idle-' goal
    in the worst-performing rows.
    """
    df = pd.DataFrame(rows, columns=header)
    
    # Identify goal columns and input columns
    goals = [c for c in header if c.endswith('-') or c.endswith('+')]
    inputs = [c for c in header if c not in goals]
    
    # Define "worst" rows as those with high 'Idle-' (since it is a minimize goal)
    # We look at the top 25% of rows where 'Idle-' is highest
    worst_rows = df.nlargest(int(len(df) * 0.25), 'Idle-')
    
    # Use a Random Forest to find feature importance regarding 'Idle-'
    # This helps identify which input variables drive the 'Idle-' value
    X = worst_rows[inputs]
    y = worst_rows['Idle-']
    
    # Convert symbolic columns to numeric codes for the model
    X_encoded = pd.get_dummies(X)
    
    model = RandomForestRegressor(n_estimators=50, random_state=42)
    model.fit(X_encoded, y)
    
    # Get feature importances
    importances = dict(zip(X_encoded.columns, model.feature_importances_))
    
    # Aggregate importance back to original input column names
    final_importances = {col: 0.0 for col in inputs}
    for feat, imp in importances.items():
        # Handle one-hot encoded column names
        base_col = feat.split('_')[0]
        if base_col in final_importances:
            final_importances[base_col] += imp
            
    # Return inputs sorted by importance
    sorted_inputs = sorted(final_importances, key=final_importances.get, reverse=True)
    return sorted_inputs
