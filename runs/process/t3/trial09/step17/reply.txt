```python
import csv
import random

def get_score(header, row_goals):
    """Calculates a normalized score for a row's goals."""
    score = 0
    for i, val in enumerate(row_goals):
        if header[i].endswith('-'):
            score += (1 - val)
        else:
            score += val
    return score / len(row_goals)

def choose(header, rows, label):
    """
    Finds the best setup for a large, high-criticality project.
    We filter for rows where 'Size' and 'Criticality' are high,
    then sample from those to find the best within a 50-label budget.
    """
    # Identify indices for filtering
    size_idx = header.index('Size')
    crit_idx = header.index('Criticality')
    goal_indices = [i for i, h in enumerate(header) if h.endswith('-') or h.endswith('+')]
    goal_headers = [header[i] for i in goal_indices]

    # Filter rows that match the "large and high criticality" criteria
    # We look for values in the top 50% of the range [0, 1]
    candidates = [
        i for i, row in enumerate(rows) 
        if row[size_idx] > 0.5 and row[crit_idx] > 0.5
    ]
    
    # If no candidates found, fallback to all rows
    if not candidates:
        candidates = list(range(len(rows)))
        
    # Sample up to 50 candidates
    sample_size = min(50, len(candidates))
    to_label = random.sample(candidates, sample_size)
    
    best_idx = -1
    best_score = -1.0
    
    for idx in to_label:
        goals = label(idx)
        score = get_score(goal_headers, goals)
        if score > best_score:
            best_score = score
            best_idx = idx
            
    return best_idx

if __name__ == "__main__":
    filename = 'pom3a.n1000.train.csv'
    try:
        with open(filename, 'r') as f:
            reader = csv.reader(f)
            header = next(reader)
            rows = []
            for row in reader:
                rows.append([float(val) if i < len(header) - 3 or (i >= len(header) - 3 and val.replace('.','',1).isdigit()) else val for i, val in enumerate(row)])
        
        # Keep a copy of true goals for the label function
        true_goals = [[row[j] for j in range(len(header)) if header[j].endswith(('-', '+'))] for row in rows]
        
        # Mask the rows
        for row in rows:
            for i, h in enumerate(header):
                if h.endswith('-') or h.endswith('+'):
                    row[i] = "?"
        
        def label(i):
            return true_goals[i]
        
        best_idx = choose(header, rows, label)
        
        print("Customer Inquiry: Best setup for a large, high-criticality project?")
        print("-" * 60)
        print(f"Analysis complete. The best performing setup found is at index: {best_idx}")
        print("This setup was selected by filtering for high-scale, high-risk parameters")
        print("and evaluating performance within a limited trial budget.")
        
    except FileNotFoundError:
        print(f"Error: {filename} not found.")
        exit(1)
```
