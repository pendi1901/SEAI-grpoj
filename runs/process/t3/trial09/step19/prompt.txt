import csv
import pandas as pd
import numpy as np

def remember(header, rows):
    """
    Identifies rows that perform well across all goal columns.
    Goals ending in '-' are minimized, '+' are maximized.
    """
    df = pd.DataFrame(rows, columns=header)
    
    # Identify goal columns
    goals = [c for c in header if c.endswith('-') or c.endswith('+')]
    
    # Normalize goals to [0, 1] range to treat them equally
    # For '-', lower is better (1 - normalized_val)
    # For '+', higher is better (normalized_val)
    normalized_goals = pd.DataFrame(index=df.index)
    for goal in goals:
        col = df[goal].astype(float)
        norm = (col - col.min()) / (col.max() - col.min())
        if goal.endswith('-'):
            normalized_goals[goal] = 1 - norm
        else:
            normalized_goals[goal] = norm
            
    # Calculate a score: the product of normalized goals (geometric mean)
    # This favors rows that are good across all objectives simultaneously
    scores = normalized_goals.prod(axis=1)
    
    # Select top 5% (or at least 1)
    n_select = max(1, int(len(rows) * 0.05))
    best_indices = scores.nlargest(n_select).index.tolist()
    
    return sorted(best_indices)

import csv
import pandas as pd
from scipy import stats

def certify(xs, ys):
    """
    Returns True if the distributions of xs and ys are statistically 
    indistinguishable (same apart from noise), False if they differ.
    Uses Welch's t-test for independent samples.
    """
    # If samples are too small, we cannot reject the null hypothesis
    if len(xs) < 2 or len(ys) < 2:
        return True
    
    # Perform Welch's t-test (does not assume equal variance)
    t_stat, p_value = stats.ttest_ind(xs, ys, equal_var=False)
    
    # If p-value > 0.05, we fail to reject the null hypothesis (they are the same)
    return p_value > 0.05

def analyze_team_size_impact(header, rows):
    """
    Splits data into small and big team setups and compares their costs.
    """
    df = pd.DataFrame(rows, columns=header)
    
    # Identify 'Team Size' and 'Cost-'
    # Team Size is an input, Cost- is a goal
    median_team_size = df['Team Size'].median()
    
    small_teams = df[df['Team Size'] <= median_team_size]['Cost-']
    big_teams = df[df['Team Size'] > median_team_size]['Cost-']
    
    is_same = certify(small_teams.tolist(), big_teams.tolist())
    
    return is_same, small_teams.mean(), big_teams.mean()

import csv

def flow(header, row, seen):
    """
    Implements a Secretary Problem strategy with a threshold.
    We observe the first 20% of rows to establish a baseline of 'good' performance,
    then accept the first row that exceeds that baseline.
    """
    goal_indices = [i for i, h in enumerate(header) if h.endswith('-') or h.endswith('+')]
    
    def get_score(r):
        # Normalize goals: 1 - (val - min)/(max - min) for '-'
        # (val - min)/(max - min) for '+'
        # Since we don't have global min/max, we use the 'seen' data as a proxy.
        scores = []
        for idx in goal_indices:
            col_vals = [s[idx] for s in seen + [r]]
            min_v, max_v = min(col_vals), max(col_vals)
            rng = max_v - min_v if max_v != min_v else 1
            norm = (r[idx] - min_v) / rng
            scores.append(1 - norm if header[idx].endswith('-') else norm)
        return sum(scores) / len(scores)

    # If it's the first row, we can't evaluate it yet
    if not seen:
        return False
    
    # Strategy: Observe first 20% to set a benchmark
    n = len(seen) + 1
    limit = max(1, int(800 * 0.2))
    
    if n <= limit:
        return False
    
    # Calculate benchmark from seen rows
    benchmarks = [get_score(s) for s in seen]
    best_seen = max(benchmarks)
    
    # Accept if current row is better than anything seen so far
    return get_score(row) > best_seen

def blame(header, rows):
    """
    Identifies input columns that correlate most strongly with the 'Idle-' goal
    in the worst-performing rows.
    """
    df = pd.DataFrame(rows, columns=header)
    
    # Identify goal columns and input columns
    goals = [c for c in header if c.endswith('-') or c.endswith('+')]
    inputs = [c for c in header if c not in goals]
    
    # Define "worst" rows as those with high 'Idle-' (since it is a minimize goal)
    # We look at the top 25% of rows where 'Idle-' is highest
    worst_rows = df.nlargest(int(len(df) * 0.25), 'Idle-')
    
    # Use a Random Forest to find feature importance regarding 'Idle-'
    # This helps identify which input variables drive the 'Idle-' value
    X = worst_rows[inputs]
    y = worst_rows['Idle-']
    
    # Convert symbolic columns to numeric codes for the model
    X_encoded = pd.get_dummies(X)
    
    model = RandomForestRegressor(n_estimators=50, random_state=42)
    model.fit(X_encoded, y)
    
    # Get feature importances
    importances = dict(zip(X_encoded.columns, model.feature_importances_))
    
    # Aggregate importance back to original input column names
    final_importances = {col: 0.0 for col in inputs}
    for feat, imp in importances.items():
        # Handle one-hot encoded column names
        base_col = feat.split('_')[0]
        if base_col in final_importances:
            final_importances[base_col] += imp
            
    # Return inputs sorted by importance
    sorted_inputs = sorted(final_importances, key=final_importances.get, reverse=True)
    return sorted_inputs

def justify(header, rows, row):
    """
    Estimates the performance of a given row and returns its percentile rank
    relative to the provided dataset.
    """
    df = pd.DataFrame(rows, columns=header)
    goals = [c for c in header if c.endswith('-') or c.endswith('+')]
    inputs = [c for c in header if c not in goals]

    # 1. Estimate the missing goal values for the target row
    X_train = df[inputs]
    y_train = df[goals]
    
    model = KNeighborsRegressor(n_neighbors=5)
    model.fit(X_train, y_train)
    
    input_vals = [row[i] for i, h in enumerate(header) if h in inputs]
    estimated_goals = model.predict([input_vals])[0]
    
    # 2. Calculate scores for all rows
    # Normalize goals: 1 - norm for '-', norm for '+'
    norm_df = pd.DataFrame(index=df.index)
    for goal in goals:
        col = df[goal]
        norm = (col - col.min()) / (col.max() - col.min() + 1e-9)
        norm_df[goal] = 1 - norm if goal.endswith('-') else norm
    
    all_scores = norm_df.mean(axis=1).tolist()
    
    # Calculate score for the estimated row
    target_norm = []
    for i, goal in enumerate(goals):
        col = df[goal]
        val = estimated_goals[i]
        norm = (val - col.min()) / (col.max() - col.min() + 1e-9)
        target_norm.append(1 - norm if goal.endswith('-') else norm)
    
    target_score = sum(target_norm) / len(target_norm)
    
    # 3. Calculate percentile
    better_than = [s for s in all_scores if target_score > s]
    p = len(better_than) / len(all_scores)
    
    reason = f"This setup is estimated to outperform {p:.2f} of all configurations based on its projected goal metrics."
    return p, reason

def choose(header, rows, label):
    """
    Finds the best row by sampling and labeling.
    We use a simple strategy: sample 50 rows, label them, 
    and return the index of the best one.
    """
    goal_indices = [i for i, h in enumerate(header) if h.endswith('-') or h.endswith('+')]
    
    # We can label at most 50 rows.
    # We will pick 50 random indices to sample.
    indices = list(range(len(rows)))
    sample_size = min(50, len(indices))
    chosen_indices = random.sample(indices, sample_size)
    
    best_idx = -1
    best_score = -float('inf')
    
    for idx in chosen_indices:
        # Get true values from the oracle
        goals = label(idx)
        
        # Calculate score: 
        # Normalize goals to [0, 1] relative to the sample set
        # Since we don't know the global min/max, we use the sample's min/max
        # as a proxy for the objective function.
        
        # For this specific task, we can just compute a simple weighted sum
        # where '-' goals are inverted and '+' are kept as is.
        # We'll assume values are roughly in [0, 1] based on the data description.
        current_score = 0
        for i, val in enumerate(goals):
            col_name = header[goal_indices[i]]
            if col_name.endswith('-'):
                current_score -= val
            else:
                current_score += val
        
        if current_score > best_score:
            best_score = current_score
            best_idx = idx
            
    return best_idx

import csv
import random

def fix(header, row, levers):
    """
    Recommends changes to a 'bad' row to improve 'Cost-' and 'Idle-'.
    We perform a local search by perturbing the levers to find a 
    configuration that minimizes the goal values.
    """
    # Identify goal indices
    goal_indices = [i for i, h in enumerate(header) if h.endswith('-') or h.endswith('+')]
    
    # Identify lever indices
    lever_indices = {h: i for i, h in enumerate(header) if h in levers}
    
    best_change = {}
    best_score = float('inf')
    
    # Try random perturbations within the allowed lever ranges
    for _ in range(100):
        candidate_row = list(row)
        current_change = {}
        
        for col, bounds in levers.items():
            idx = lever_indices[col]
            if isinstance(bounds, (list, tuple)) and not isinstance(bounds, str):
                # Numeric: pick a value in range
                new_val = random.uniform(bounds[0], bounds[1])
            else:
                # Symbolic: pick from list
                new_val = random.choice(bounds)
            
            candidate_row[idx] = new_val
            current_change[col] = new_val
            
        # Simple heuristic: minimize the sum of goals (Cost- and Idle-)
        # We assume lower is better for both.
        score = sum(candidate_row[i] for i in goal_indices)
        
        if score < best_score:
            best_score = score
            best_change = current_change
            
    return best_change

import csv

from sklearn.ensemble import IsolationForest

def spot(header, rows):
    """
    Identifies rows that are outliers (potentially mistyped or made up)
    using the Isolation Forest algorithm.
    """
    df = pd.DataFrame(rows, columns=header)
    
    # We only consider numeric columns for anomaly detection
    numeric_cols = [c for c in header if c in df.select_dtypes(include=['number']).columns]
    data = df[numeric_cols]
    
    # Isolation Forest is effective at detecting anomalies in high-dimensional data
    # contamination='auto' lets the model decide the proportion of outliers
    model = IsolationForest(n_estimators=100, contamination=0.05, random_state=42)
    
    # Fit the model and get anomaly scores
    # -1 indicates an outlier, 1 indicates an inlier
    model.fit(data)
    scores = model.decision_function(data)
    
    # Get indices of rows sorted by suspicion (lowest score = most suspicious)
    suspicious_indices = sorted(range(len(scores)), key=lambda i: scores[i])
    
    return suspicious_indices

def get_score(header, row_goals):

def race(header, rows, label):
    """
    Compares three strategies: random, greedy, and hill-climbing.
    Budget: 50 labels total.
    """
    goal_indices = [i for i, h in enumerate(header) if h.endswith('-') or h.endswith('+')]
    goal_headers = [header[i] for i in goal_indices]
    
    labeled = {}
    def get_labeled(i):
        if i not in labeled:
            if len(labeled) >= 50:
                raise Exception("Budget exceeded")
            labeled[i] = label(i)
        return labeled[i]

    # Strategy 1: Random (16 samples)
    def strategy_random():
        indices = random.sample(range(len(rows)), 16)
        best_idx = -1
        best_s = -1
        for idx in indices:
            s = get_score(goal_headers, get_labeled(idx))
            if s > best_s:
                best_s, best_idx = s, idx
        return best_idx, best_s

    # Strategy 2: Greedy (First 16 samples)
    def strategy_greedy():
        best_idx = -1
        best_s = -1
        for idx in range(16):
            s = get_score(goal_headers, get_labeled(idx))
            if s > best_s:
                best_s, best_idx = s, idx
        return best_idx, best_s

    # Strategy 3: Hill Climbing (18 samples)
    def strategy_hill():
        curr = random.randint(0, len(rows) - 1)
        best_idx, best_s = curr, get_score(goal_headers, get_labeled(curr))
        for _ in range(17):
            neighbors = random.sample(range(len(rows)), 2)
            for n in neighbors:
                s = get_score(goal_headers, get_labeled(n))
                if s > best_s:
                    best_s, best_idx = s, n
        return best_idx, best_s

    strategies = {"random": strategy_random, "greedy": strategy_greedy, "hill": strategy_hill}
    results = {}
    for name, func in strategies.items():
        idx, score = func()
        results[name] = {"idx": idx, "score": score}

    winner = max(results, key=lambda n: results[n]["score"])
    return {"winner": winner, "row": results[winner]["idx"], "tried": list(strategies.keys())}

    n_select = max(1, int(len(rows) * 0.05))
    best_indices = scores.nlargest(n_select).index.tolist()

import csv
from scipy import stats

def certify(xs, ys):
    """
    Returns True if the distributions of xs and ys are statistically 
    indistinguishable (same apart from noise), False if they differ.
    Uses Welch's t-test for independent samples.
    """
    if len(xs) < 2 or len(ys) < 2:
        return True
    
    # Welch's t-test does not assume equal variance
    t_stat, p_value = stats.ttest_ind(xs, ys, equal_var=False)
    
    # If p-value > 0.05, we fail to reject the null hypothesis (they are the same)
    return p_value > 0.05

def analyze_shift(header, rows):
    """
    Splits data into two halves (before/after spring change) and compares 
    the aggregate performance of the goal columns.
    """
    goal_indices = [i for i, h in enumerate(header) if h.endswith('-') or h.endswith('+')]
    
    # Calculate a composite score for each row
    # Normalize goals: 1 - (val - min)/(max - min) for '-'
    # (val - min)/(max - min) for '+'
    scores = []
    for row in rows:

        for idx in goal_indices:
            col_vals = [r[idx] for r in rows]
            min_v, max_v = min(col_vals), max(col_vals)
            rng = max_v - min_v if max_v != min_v else 1
            norm = (row[idx] - min_v) / rng
            row_score += (1 - norm) if header[idx].endswith('-') else norm
        scores.append(row_score / len(goal_indices))
    
    # Split into two halves (assuming chronological order)
    mid = len(scores) // 2
    before = scores[:mid]
    after = scores[mid:]
    
    is_same = certify(before, after)
    return is_same, sum(before)/len(before), sum(after)/len(after)

import csv

def flow(header, row, seen):
    """
    Detects if a project's performance is 'going bad'.
    We monitor the trend of the goals. If the current row's performance
    is significantly worse than the moving average of the 'seen' rows,
    we flag it as 'going bad' (returning True).
    """
    goal_indices = [i for i, h in enumerate(header) if h.endswith('-') or h.endswith('+')]
    
    def get_score(r):
        # Normalize goals: 1 - (val - min)/(max - min) for '-'
        # (val - min)/(max - min) for '+'
        # Use seen data as a proxy for global min/max
        all_data = seen + [r]
        scores = []
        for idx in goal_indices:
            col_vals = [s[idx] for s in all_data]
            min_v, max_v = min(col_vals), max(col_vals)
            rng = max_v - min_v if max_v != min_v else 1
            norm = (r[idx] - min_v) / rng
            scores.append(1 - norm if header[idx].endswith('-') else norm)
        return sum(scores) / len(scores)

    if len(seen) < 10:
        return False
    
    # Calculate moving average of the last 10 rows
    recent_scores = [get_score(s) for s in seen[-10:]]
    moving_avg = sum(recent_scores) / len(recent_scores)
    
    # Current score
    current_score = get_score(row)
    
    # If current performance drops significantly below the moving average
    # (e.g., more than 2 standard deviations or a fixed threshold), flag it.
    # Here we use a 20% drop as a heuristic for "going bad".
    if current_score < (moving_avg * 0.8):
        return True
        
    return False
