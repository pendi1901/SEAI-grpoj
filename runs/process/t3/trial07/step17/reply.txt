import csv
import random

def choose(header, rows, label):
    """
    Finds the best setup for a large, high-criticality project by sampling 
    the space and using a simple heuristic to identify the best configuration.
    """
    goal_indices = [i for i, h in enumerate(header) if h.endswith(('-', '+'))]
    
    # We need to find rows that are "large" and "critical".
    # Based on the header, 'Size' and 'Criticality' are the relevant inputs.
    # We prioritize rows where these are high.
    size_idx = header.index('Size')
    crit_idx = header.index('Criticality')
    
    # Score rows by their suitability for a "large, high-criticality" project
    # (Higher size and criticality is better for this specific request)
    def suitability(row):
        return row[size_idx] + row[crit_idx]
    
    # Sort all rows by suitability and pick the top 50 candidates to label
    indexed_rows = list(enumerate(rows))
    indexed_rows.sort(key=lambda x: suitability(x[1]), reverse=True)
    candidates = indexed_rows[:50]
    
    best_idx = -1
    best_score = -float('inf')
    
    # Label the candidates and find the one with the best aggregate goal performance
    labeled = []
    for i, row in candidates:
        goals = label(i)
        labeled.append((goals, i))
        
    # Normalize goals for the 50 labeled rows
    all_goals = [l[0] for l in labeled]
    mins = [min(g[i] for g in all_goals) for i in range(len(goal_indices))]
    maxs = [max(g[i] for g in all_goals) for i in range(len(goal_indices))]
    
    for goals, i in labeled:
        scores = []
        for j, g_idx in enumerate(goal_indices):
            val = goals[j]
            norm = (val - mins[j]) / (maxs[j] - mins[j] + 1e-9)
            # '-' minimize, '+' maximize
            if header[g_idx].endswith('-'):
                scores.append(1 - norm)
            else:
                scores.append(norm)
        
        agg = sum(scores) / len(scores)
        if agg > best_score:
            best_score = agg
            best_idx = i
            
    return best_idx

if __name__ == "__main__":
    filename = 'pom3a.n1000.train.csv'
    try:
        with open(filename, 'r') as f:
            reader = csv.reader(f)
            header = next(reader)
            rows = []
            true_goals = []
            for row in reader:
                processed = []
                goals = []
                for i, val in enumerate(header):
                    if val.endswith(('-', '+')):
                        goals.append(float(row[i]))
                        processed.append("?")
                    else:
                        processed.append(float(row[i]))
                rows.append(processed)
                true_goals.append(goals)
        
        def label(i):
            return true_goals[i]
        
        best_idx = choose(header, rows, label)
        
        print("Customer Inquiry: Best setup for a large, high-criticality project.")
        print(f"Analysis: After evaluating 50 high-potential candidates, we identified the optimal configuration.")
        print(f"The recommended row index is: {best_idx}")
        print("This setup balances project size and criticality while maintaining optimal performance goals.")
        exit(0)
        
    except FileNotFoundError:
        print(f"Error: {filename} not found.")
        exit(1)
