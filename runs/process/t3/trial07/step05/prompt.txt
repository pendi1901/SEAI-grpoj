import csv
import pandas as pd
import numpy as np

def remember(header, rows):
    """
    Identifies rows that perform well across all goal columns.
    Goals ending in '-' are minimized, '+' are maximized.
    """
    df = pd.DataFrame(rows, columns=header)
    
    # Identify goal columns
    goals = [col for col in header if col.endswith(('-', '+'))]
    
    # Normalize goals to [0, 1] range for comparison
    # For '-', lower is better (1 - normalized value)
    # For '+', higher is better (normalized value)
    scores = pd.DataFrame(index=df.index)
    for goal in goals:
        col_data = df[goal].astype(float)
        normalized = (col_data - col_data.min()) / (col_data.max() - col_data.min())
        if goal.endswith('-'):
            scores[goal] = 1 - normalized
        else:
            scores[goal] = normalized
            
    # Calculate aggregate score (mean of normalized goals)
    df['aggregate'] = scores.mean(axis=1)
    
    # Select top 5% (or at least 1)
    n_rows = len(df)
    k = max(1, int(n_rows * 0.05))
    
    best_indices = df['aggregate'].nlargest(k).index.tolist()
    return sorted(best_indices)

import csv
import pandas as pd
from scipy import stats

def certify(xs, ys):
    """
    Returns True if the difference between two distributions is statistically 
    insignificant (p > 0.05), False otherwise.
    """
    # Use Welch's t-test (equal_var=False) to compare two independent samples
    t_stat, p_value = stats.ttest_ind(xs, ys, equal_var=False)
    return p_value > 0.05

def analyze_team_size_impact(header, rows):
    df = pd.DataFrame(rows, columns=header)
    
    # Identify the 'Team Size' column and the 'Cost-' goal
    # Team Size is numeric, Cost- is the goal to minimize
    # We split into small vs big based on the median of Team Size
    median_team = df['Team Size'].median()
    
    small_team_costs = df[df['Team Size'] <= median_team]['Cost-']
    big_team_costs = df[df['Team Size'] > median_team]['Cost-']
    
    # Check if they are statistically different
    is_same = certify(small_team_costs.tolist(), big_team_costs.tolist())
    
    return is_same, small_team_costs.mean(), big_team_costs.mean()

import csv

def flow(header, row, seen):
    """
    Implements a Secretary Problem-style strategy.
    We observe a sample to establish a baseline of 'good' performance,
    then accept the first row that exceeds that baseline.
    """
    # Identify goal columns
    goal_indices = [i for i, h in enumerate(header) if h.endswith(('-', '+'))]
    
    def get_score(r):
        # Normalize goals: 1 - (val - min)/(max - min) for '-'
        # and (val - min)/(max - min) for '+'
        # Since we don't have global min/max, we use the current 'seen' data
        # to estimate relative quality.
        scores = []
        for i in goal_indices:
            col = [s[i] for s in seen + [r]]
            min_v, max_v = min(col), max(col)
            val = r[i]
            norm = (val - min_v) / (max_v - min_v + 1e-9)
            scores.append(1 - norm if header[i].endswith('-') else norm)
        return sum(scores) / len(scores)

    # Strategy: Observe 20% of the data to set a threshold, then pick the first
    # row that is better than the best seen so far.
    if len(seen) < 160:  # 20% of 800
        return False
    
    current_score = get_score(row)
    best_seen_score = max([get_score(s) for s in seen])
    
    return current_score > best_seen_score

if __name__ == "__main__":
    filename = 'pom3a.n1000.train.csv'

        else:
