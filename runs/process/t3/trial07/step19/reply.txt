import csv
from sklearn.ensemble import IsolationForest

def spot(header, rows):
    """
    Identifies rows that are outliers in the input space using IsolationForest.
    Returns the indices of the rows, sorted by suspicion (most suspicious first).
    """
    # Identify input columns (those not ending in - or +)
    input_indices = [i for i, h in enumerate(header) if not (h.endswith('-') or h.endswith('+'))]
    
    # Extract input data for anomaly detection
    # We filter out rows that might contain "?" if the customer is testing with hidden data
    X = []
    valid_indices = []
    for idx, row in enumerate(rows):
        # Only use numeric inputs for IsolationForest
        # If a row contains '?', we skip it or treat it as a missing value
        # For this specific task, we assume the input rows are fully populated
        try:
            X.append([float(row[i]) for i in input_indices])
            valid_indices.append(idx)
        except (ValueError, TypeError):
            continue
    
    if not X:
        return []

    # IsolationForest assigns -1 to outliers and 1 to inliers.
    # The decision_function returns the anomaly score (lower is more anomalous).
    clf = IsolationForest(contamination=0.05, random_state=42)
    clf.fit(X)
    
    # Get anomaly scores (lower = more suspicious)
    scores = clf.decision_function(X)
    
    # Pair scores with original indices
    indexed_scores = list(zip(valid_indices, scores))
    # Sort by score ascending (most suspicious first)
    indexed_scores.sort(key=lambda x: x[1])
    
    return [idx for idx, score in indexed_scores]

if __name__ == "__main__":
    filename = 'pom3a.n1000.train.csv'
    try:
        with open(filename, 'r') as f:
            reader = csv.reader(f)
            header = next(reader)
            rows = []
            for row in reader:
                processed_row = []
                for val in row:
                    try:
                        processed_row.append(float(val))
                    except ValueError:
                        processed_row.append(val)
                rows.append(processed_row)
        
        suspicious_indices = spot(header, rows)
        
        print("Customer Inquiry: A new team lead proposes a setup unlike anything we have run.")
        print("Analysis: I have identified configurations that deviate significantly from our historical data.")
        print(f"I found {len(suspicious_indices)} setups that look unusual.")
        print("The 5 most suspicious row indices are:")
        for i in range(min(5, len(suspicious_indices))):
            print(f"  Row {suspicious_indices[i]}")
        
        print("\nThese rows represent setups that are statistically unlike anything we have seen before.")
        exit(0)
        
    except FileNotFoundError:
        print(f"Error: {filename} not found.")
        exit(1)
