import csv
import pandas as pd
import numpy as np

def remember(header, rows):
    """
    Identifies rows that perform well across all goal columns.
    Goals ending in '-' are minimized, '+' are maximized.
    """
    df = pd.DataFrame(rows, columns=header)
    
    # Identify goal columns
    goals = [col for col in header if col.endswith(('-', '+'))]
    
    # Normalize goals to [0, 1] range for comparison
    # For '-', lower is better (1 - normalized value)
    # For '+', higher is better (normalized value)
    scores = pd.DataFrame(index=df.index)
    for goal in goals:
        col_data = df[goal].astype(float)
        normalized = (col_data - col_data.min()) / (col_data.max() - col_data.min())
        if goal.endswith('-'):
            scores[goal] = 1 - normalized
        else:
            scores[goal] = normalized
            
    # Calculate aggregate score (mean of normalized goals)
    df['aggregate'] = scores.mean(axis=1)
    
    # Select top 5% (or at least 1)
    n_rows = len(df)
    k = max(1, int(n_rows * 0.05))
    
    best_indices = df['aggregate'].nlargest(k).index.tolist()
    return sorted(best_indices)

def certify(xs, ys):
    """
    Returns True if the difference between two distributions is statistically 
    insignificant (p > 0.05), False otherwise.
    """
    # Use Welch's t-test (equal_var=False) to compare two independent samples
    t_stat, p_value = stats.ttest_ind(xs, ys, equal_var=False)
    return p_value > 0.05

    small_team_costs = df[df['Team Size'] <= median_team]['Cost-']

import csv

def flow(header, row, seen):
    """
    Implements a Secretary Problem-style strategy.
    We observe a sample to establish a baseline of 'good' performance,
    then accept the first row that exceeds that baseline.
    """
    # Identify goal columns
    goal_indices = [i for i, h in enumerate(header) if h.endswith(('-', '+'))]
    
    def get_score(r):
        # Normalize goals: 1 - (val - min)/(max - min) for '-'
        # and (val - min)/(max - min) for '+'
        # Since we don't have global min/max, we use the current 'seen' data
        # to estimate relative quality.
        scores = []
        for i in goal_indices:
            col = [s[i] for s in seen + [r]]
            min_v, max_v = min(col), max(col)
            val = r[i]
            norm = (val - min_v) / (max_v - min_v + 1e-9)
            scores.append(1 - norm if header[i].endswith('-') else norm)
        return sum(scores) / len(scores)

    # Strategy: Observe 20% of the data to set a threshold, then pick the first
    # row that is better than the best seen so far.
    if len(seen) < 160:  # 20% of 800
        return False
    
    current_score = get_score(row)
    best_seen_score = max([get_score(s) for s in seen])
    
    return current_score > best_seen_score

def justify(header, rows, row):
    """
    Estimates the performance of a given row and returns its percentile rank
    among the existing rows, along with a justification.
    """
    df = pd.DataFrame(rows, columns=header)
    goal_cols = [col for col in header if col.endswith(('-', '+'))]
    input_cols = [col for col in header if col not in goal_cols]

    # 1. Estimate the goals for the target row using KNN
    X_train = df[input_cols]
    y_train = df[goal_cols]
    
    model = KNeighborsRegressor(n_neighbors=5)
    model.fit(X_train, y_train)
    
    # Extract inputs from the target row (ignoring "?")
    target_inputs = [row[i] for i, col in enumerate(header) if col in input_cols]
    estimated_goals = model.predict([target_inputs])[0]
    
    # 2. Calculate aggregate score for all rows
    # Normalize: 1 - (val - min)/(max - min) for '-', (val - min)/(max - min) for '+'
    def get_aggregate(data_row):
        scores = []
        for i, col in enumerate(header):
            if col in goal_cols:
                col_data = df[col]
                norm = (data_row[i] - col_data.min()) / (col_data.max() - col_data.min() + 1e-9)
                scores.append(1 - norm if col.endswith('-') else norm)
        return sum(scores) / len(scores)

    all_scores = [get_aggregate(r) for r in rows]
    
    # 3. Calculate target row score
    target_score = 0
    for i, col in enumerate(header):
        if col in goal_cols:
            col_data = df[col]
            norm = (estimated_goals[goal_cols.index(col)] - col_data.min()) / (col_data.max() - col_data.min() + 1e-9)
            target_score += (1 - norm if col.endswith('-') else norm)
    target_score /= len(goal_cols)
    
    # 4. Calculate percentile
    better_than = [s for s in all_scores if s < target_score]
    p = len(better_than) / len(all_scores)
    
    reason = f"This setup is estimated to perform better than {p:.1%} of all known configurations based on its input parameters."
    return p, reason

import csv

def choose(header, rows, label):
    """
    Finds the best row by sampling 50 setups.
    We use a simple random sampling strategy to explore the space,
    labeling 50 rows and returning the index of the one with the best aggregate score.
    """
    goal_indices = [i for i, h in enumerate(header) if h.endswith(('-', '+'))]
    
    # Pre-calculate min/max for normalization based on the first 50 rows
    # or just use a heuristic since we only have 50 labels.
    # We will label 50 random rows.
    indices = list(range(len(rows)))
    random.shuffle(indices)
    sample_indices = indices[:50]
    
    best_idx = -1
    best_score = -float('inf')
    
    # Store labeled data to normalize
    labeled_data = []
    for i in sample_indices:
        goals = label(i)
        labeled_data.append((i, goals))
        
    # Normalize goals
    # Transpose goals to get min/max per goal
    all_goals = [d[1] for d in labeled_data]
    mins = [min(g[i] for g in all_goals) for i in range(len(goal_indices))]
    maxs = [max(g[i] for g in all_goals) for i in range(len(goal_indices))]
    
    for i, goals in labeled_data:
        scores = []
        for j, g_idx in enumerate(goal_indices):
            val = goals[j]
            norm = (val - mins[j]) / (maxs[j] - mins[j] + 1e-9)
            # '-' minimize, '+' maximize
            if header[g_idx].endswith('-'):
                scores.append(1 - norm)
            else:
                scores.append(norm)
        
        agg = sum(scores) / len(scores)
        if agg > best_score:
            best_score = agg
            best_idx = i
            
    return best_idx

import csv
import random

def fix(header, row, levers):
    """
    Recommends a change to a 'bad' row to improve its goals.
    We perform a local search by perturbing the levers and selecting 
    the change that results in the best estimated improvement.
    """
    # Identify goal indices
    goal_indices = [i for i, h in enumerate(header) if h.endswith(('-', '+'))]
    
    def get_score(r):
        # Simple heuristic: sum of normalized goals
        # Since we don't have the full dataset context here, we assume
        # lower is better for '-' and higher for '+'
        scores = []
        for i in goal_indices:
            val = r[i]
            # Normalize based on typical ranges for pom3a (0 to 1)
            if header[i].endswith('-'):
                scores.append(1.0 - val)
            else:
                scores.append(val)
        return sum(scores) / len(scores)

    best_change = {}
    best_score = get_score(row)
    
    # Try changing each lever
    for col, allowed in levers.items():
        idx = header.index(col)
        original_val = row[idx]
        
        # Generate a candidate value
        if isinstance(allowed, tuple):  # Numeric (low, high)
            # Try moving towards the middle or a random point in range
            candidate = random.uniform(allowed[0], allowed[1])
        else:  # Symbolic
            candidate = random.choice(allowed)
            
        # Evaluate
        new_row = list(row)
        new_row[idx] = candidate
        new_score = get_score(new_row)
        
        if new_score > best_score:
            best_score = new_score
            best_change = {col: candidate}
            
    # If no improvement found, force a change to one lever
    if not best_change:
        col = random.choice(list(levers.keys()))
        allowed = levers[col]
        val = random.uniform(allowed[0], allowed[1]) if isinstance(allowed, tuple) else random.choice(allowed)
        best_change = {col: val}
        
    return best_change

import csv
from sklearn.ensemble import IsolationForest

def spot(header, rows):
    """
    Identifies rows that are outliers in the input space using IsolationForest.
    Returns the indices of the rows, sorted by suspicion (most suspicious first).
    """
    # Identify input columns (those not ending in - or +)
    input_indices = [i for i, h in enumerate(header) if not (h.endswith('-') or h.endswith('+'))]
    
    # Extract input data for anomaly detection
    X = []
    for row in rows:
        # Only use numeric inputs for IsolationForest
        X.append([row[i] for i in input_indices])
    
    # IsolationForest assigns -1 to outliers and 1 to inliers.
    # The decision_function returns the anomaly score (lower is more anomalous).
    clf = IsolationForest(contamination=0.05, random_state=42)
    clf.fit(X)
    
    # Get anomaly scores (lower = more suspicious)
    scores = clf.decision_function(X)
    
    # Get indices of rows, sorted by score ascending (most suspicious first)
    indexed_scores = list(enumerate(scores))
    indexed_scores.sort(key=lambda x: x[1])
    
    return [idx for idx, score in indexed_scores]

import csv
from scipy import stats

def certify(xs, ys):
    """
    Returns True if the difference between two distributions is statistically 
    insignificant (p > 0.05), meaning the results are effectively the same.
    Returns False if they are statistically different.
    """
    # Use Welch's t-test (equal_var=False) to compare two independent samples
    t_stat, p_value = stats.ttest_ind(xs, ys, equal_var=False)
    return p_value > 0.05

def analyze_shift(header, rows):
    """
    Splits the data into two halves (before and after the 'spring' change)
    and compares the aggregate performance of the goals.
    """
    goal_indices = [i for i, h in enumerate(header) if h.endswith(('-', '+'))]
    
    # Calculate aggregate score for each row
    # Normalize goals: 1 - (val - min)/(max - min) for '-', (val - min)/(max - min) for '+'
    all_vals = [[row[i] for row in rows] for i in goal_indices]
    mins = [min(col) for col in all_vals]
    maxs = [max(col) for col in all_vals]
    
    def get_score(row):
        scores = []
        for j, idx in enumerate(goal_indices):
            norm = (row[idx] - mins[j]) / (maxs[j] - mins[j] + 1e-9)
            scores.append(1 - norm if header[idx].endswith('-') else norm)
        return sum(scores) / len(scores)
    
    scores = [get_score(row) for row in rows]
    
    # Split into two halves (before and after the change)
    mid = len(scores) // 2
    before = scores[:mid]
    after = scores[mid:]
    
    is_same = certify(before, after)
    return is_same, sum(before)/len(before), sum(after)/len(after)

def flow(header, row, seen):
    """
    Warns the customer if a new row is significantly worse than the 
    historical baseline. We define 'bad' as being in the bottom 10% 
    of performance compared to all previously seen data.
    """

    def get_score(r):
        # Calculate normalized score for a row based on seen data
        scores = []
        for i in goal_indices:
            col_vals = [s[i] for s in seen] + [r[i]]
            min_v, max_v = min(col_vals), max(col_vals)
            norm = (r[i] - min_v) / (max_v - min_v + 1e-9)
            scores.append(1 - norm if header[i].endswith('-') else norm)
        return sum(scores) / len(scores)

    # Need a baseline before we can warn
    if len(seen) < 50:
        return False
    
    # Calculate scores for all seen rows
    seen_scores = [get_score(s) for s in seen]
    current_score = get_score(row)
    
    # Threshold: if current row is worse than 90% of history, warn (return True)
    threshold = sorted(seen_scores)[int(len(seen_scores) * 0.1)]
    
    return current_score < threshold

def blame(header, rows):
    """
    Identifies the input columns that best explain why the worst rows are bad.
    We define 'bad' as having a high aggregate cost/score/idle penalty.
    We use Random Forest feature importance to rank which inputs drive these bad outcomes.
    """
    df = pd.DataFrame(rows, columns=header)

    goal_cols = [col for col in header if col.endswith(('-', '+'))]
    input_cols = [col for col in header if col not in goal_cols]

    # For '-', higher is worse. For '+', lower is worse.
    scores = pd.DataFrame(index=df.index)
    for goal in goal_cols:
        col_data = df[goal].astype(float)
        norm = (col_data - col_data.min()) / (col_data.max() - col_data.min() + 1e-9)
        if goal.endswith('-'):
            scores[goal] = norm  # Higher is worse
        else:
            scores[goal] = 1 - norm  # Lower is worse
            
    df['badness'] = scores.mean(axis=1)
    
    # Train a Random Forest to predict 'badness' from inputs
    X = df[input_cols]
    y = df['badness']
    
    model = RandomForestRegressor(n_estimators=50, random_state=42)
    model.fit(X, y)
    
    # Get feature importances
    importances = model.feature_importances_
    features_with_importance = sorted(
        zip(input_cols, importances), 
        key=lambda x: x[1], 
        reverse=True
    )
    
    # Return the top 2 most influential input columns
    return [feat for feat, score in features_with_importance[:2]]

    target_inputs = [row[i] for i, col in enumerate(header) if col in input_cols]
    estimated_goals = model.predict([target_inputs])[0]

    # Normalize: 1 - (val - min)/(max - min) for '-', (val - min)/(max - min) for '+'

        for i, col in enumerate(header):

                col_data = df[col]
                norm = (data_row[i] - col_data.min()) / (col_data.max() - col_data.min() + 1e-9)

    target_score = 0
    for i, col in enumerate(header):
        if col in goal_cols:
