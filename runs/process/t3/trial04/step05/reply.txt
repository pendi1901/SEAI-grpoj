import csv

def blame(header, rows):
    """
    Identifies input columns that correlate most strongly with the 'Idle-' goal
    in the worst-performing setups.
    """
    # Identify goal columns and input columns
    goal_indices = [i for i, name in enumerate(header) if name.endswith(('-', '+'))]
    input_indices = [i for i, name in enumerate(header) if not name.endswith(('-', '+'))]
    
    # Identify the 'Idle-' column index
    idle_idx = -1
    for i in goal_indices:
        if header[i] == 'Idle-':
            idle_idx = i
            break
            
    if idle_idx == -1:
        return []

    # Calculate a composite score for each row to define "worst"
    # A row is bad if its goals are high (for -) or low (for +)
    def get_badness(row):
        badness = 0
        for i in goal_indices:
            val = row[i]
            # Since we want to find the worst, we look for high values in '-' goals
            # and low values in '+' goals.
            if header[i].endswith('-'):
                badness += val
            else:
                badness += (1.0 - val)
        return badness

    # Sort rows by badness and take the worst 20%
    sorted_rows = sorted(rows, key=get_badness, reverse=True)
    worst_rows = sorted_rows[:len(rows) // 5]
    
    # For each input column, calculate the correlation with 'Idle-'
    # We use a simple difference in means: (mean in worst) - (mean in all)
    correlations = []
    all_idle_mean = sum(r[idle_idx] for r in rows) / len(rows)
    worst_idle_mean = sum(r[idle_idx] for r in worst_rows) / len(worst_rows)
    
    for i in input_indices:
        # Calculate mean of this input in the worst rows
        input_mean_worst = sum(r[i] for r in worst_rows) / len(worst_rows)
        input_mean_all = sum(r[i] for r in rows) / len(rows)
        
        # If the input is high when Idle is high, it's a candidate for blame
        # We use the product of differences to see if they move together
        score = (input_mean_worst - input_mean_all) * (worst_idle_mean - all_idle_mean)
        correlations.append((header[i], score))
    
    # Return the top 3 inputs that correlate with high Idle-
    correlations.sort(key=lambda x: x[1], reverse=True)
    return [name for name, score in correlations[:3]]

if __name__ == "__main__":
    filename = 'pom3a.n1000.train.csv'
    try:
        with open(filename, 'r') as f:
            reader = csv.reader(f)
            header = next(reader)
            rows = []
            for row in reader:
                rows.append([float(val) for val in row])
        
        blame_list = blame(header, rows)
        
        print("Customer Inquiry Analysis: Why do worst setups have high Idle time?")
        print("The following input factors are most strongly associated with high idle times:")
        for factor in blame_list:
            print(f"- {factor}")
        print("\nThese factors contribute to resource under-utilization in your worst-performing project setups.")
        
    except Exception as e:
        print(f"An error occurred: {e}")
        exit(1)
