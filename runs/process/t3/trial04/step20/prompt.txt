import csv
import pandas as pd
import numpy as np

def remember(header, rows):
    """
    Identifies rows that perform well across all goal columns.
    Goals ending in '-' are minimized, goals ending in '+' are maximized.
    """
    df = pd.DataFrame(rows, columns=header)
    
    # Identify goal columns
    goals = [col for col in header if col.endswith(('-', '+'))]
    
    # Normalize goals to a [0, 1] range to compare them equally
    # For '-', lower is better (1 - normalized_value)
    # For '+', higher is better (normalized_value)
    scores = pd.DataFrame(index=df.index)
    for goal in goals:
        col_min = df[goal].min()
        col_max = df[goal].max()
        norm = (df[goal] - col_min) / (col_max - col_min + 1e-9)
        
        if goal.endswith('-'):
            scores[goal] = 1 - norm
        else:
            scores[goal] = norm
            
    # Calculate aggregate score (mean of normalized goal scores)
    df['aggregate'] = scores.mean(axis=1)
    
    # Select top 5% of rows
    num_to_select = max(1, len(rows) // 20)
    best_indices = df['aggregate'].nlargest(num_to_select).index.tolist()
    
    return sorted(best_indices)

import csv

from scipy import stats

def certify(xs, ys):
    """
    Returns True if the two distributions are statistically similar (same population),
    False if they differ significantly. Uses Welch's t-test.
    """
    # If the samples are too small, we cannot reliably reject the null hypothesis
    if len(xs) < 2 or len(ys) < 2:
        return True
    
    # Welch's t-test (does not assume equal variance)
    t_stat, p_value = stats.ttest_ind(xs, ys, equal_var=False)
    
    # If p-value > 0.05, we fail to reject the null hypothesis (they are the same)
    return p_value > 0.05

def analyze_team_size_impact(header, rows):
    df = pd.DataFrame(rows, columns=header)
    
    # Identify 'Team Size' column and 'Cost-' goal

    small_teams = df[df[team_size_col] <= median_size][cost_col]

    return is_same, small_teams.mean(), big_teams.mean()

import csv

def flow(header, row, seen):
    """
    Implements a Secretary Problem-style decision strategy.
    We observe a sample of the data to establish a baseline (the 'best' seen so far),
    then accept the first row that is better than that baseline.
    """
    # Identify goal indices
    goal_indices = [i for i, name in enumerate(header) if name.endswith(('-', '+'))]
    
    def get_score(r):
        # Normalize goals: lower is better for '-', higher for '+'
        # Since we don't have global min/max, we use the current 'seen' data
        # to estimate relative quality.
        score = 0
        for i in goal_indices:
            val = r[i]
            name = header[i]
            # Simple heuristic: if we have seen data, compare against it
            if seen:
                col_vals = [s[i] for s in seen]
                c_min, c_max = min(col_vals), max(col_vals)
                rng = c_max - c_min + 1e-9
                norm = (val - c_min) / rng
                score += (1 - norm) if name.endswith('-') else norm
            else:
                score = 0
        return score / len(goal_indices) if goal_indices else 0

    # Strategy: 
    # 1. Observe first 20% of data to establish a 'gold standard'
    # 2. Accept the first row that exceeds the best score found in the observation phase
    sample_size = max(1, len(seen) // 5)
    
    if len(seen) < sample_size:
        return False
    
    # Calculate best score from the observation phase
    best_score = -1.0
    for s in seen[:sample_size]:
        best_score = max(best_score, get_score(s))
        
    return get_score(row) > best_score

import csv

def blame(header, rows):
    """
    Identifies input columns that correlate most strongly with the 'Idle-' goal
    in the worst-performing setups.
    """
    # Identify goal columns and input columns
    goal_indices = [i for i, name in enumerate(header) if name.endswith(('-', '+'))]
    input_indices = [i for i, name in enumerate(header) if not name.endswith(('-', '+'))]
    
    # Identify the 'Idle-' column index
    idle_idx = -1
    for i in goal_indices:
        if header[i] == 'Idle-':
            idle_idx = i
            break
            
    if idle_idx == -1:
        return []

    # Calculate a composite score for each row to define "worst"
    # A row is bad if its goals are high (for -) or low (for +)
    def get_badness(row):
        badness = 0
        for i in goal_indices:
            val = row[i]
            # Since we want to find the worst, we look for high values in '-' goals
            # and low values in '+' goals.
            if header[i].endswith('-'):
                badness += val
            else:
                badness += (1.0 - val)
        return badness

    # Sort rows by badness and take the worst 20%
    sorted_rows = sorted(rows, key=get_badness, reverse=True)
    worst_rows = sorted_rows[:len(rows) // 5]
    
    # For each input column, calculate the correlation with 'Idle-'
    # We use a simple difference in means: (mean in worst) - (mean in all)
    correlations = []
    all_idle_mean = sum(r[idle_idx] for r in rows) / len(rows)
    worst_idle_mean = sum(r[idle_idx] for r in worst_rows) / len(worst_rows)
    
    for i in input_indices:
        # Calculate mean of this input in the worst rows
        input_mean_worst = sum(r[i] for r in worst_rows) / len(worst_rows)
        input_mean_all = sum(r[i] for r in rows) / len(rows)
        
        # If the input is high when Idle is high, it's a candidate for blame
        # We use the product of differences to see if they move together
        score = (input_mean_worst - input_mean_all) * (worst_idle_mean - all_idle_mean)
        correlations.append((header[i], score))
    
    # Return the top 3 inputs that correlate with high Idle-
    correlations.sort(key=lambda x: x[1], reverse=True)
    return [name for name, score in correlations[:3]]

def justify(header, rows, row):
    """
    Returns (p, reason) where p is the fraction of rows that 'row' is better than,
    and reason is a one-sentence explanation.
    """
    # Identify indices
    goal_indices = [i for i, name in enumerate(header) if name.endswith(('-', '+'))]
    input_indices = [i for i, name in enumerate(header) if not (name.endswith('-') or name.endswith('+'))]
    
    # Prepare training data for KNN
    X_train = [[r[i] for i in input_indices] for r in rows]
    
    # Estimate goals for the target row
    estimated_goals = []
    for g_idx in goal_indices:
        y_train = [r[g_idx] for r in rows]
        model = KNeighborsRegressor(n_neighbors=5)
        model.fit(X_train, y_train)
        estimated_goals.append(model.predict([row[:len(input_indices)]])[0])
    
    # Define a scoring function: normalized distance to ideal
    def get_score(goals):
        scores = []
        for i, g_idx in enumerate(goal_indices):
            col_vals = [r[g_idx] for r in rows]
            c_min, c_max = min(col_vals), max(col_vals)
            rng = c_max - c_min + 1e-9
            norm = (goals[i] - c_min) / rng
            scores.append(1 - norm if header[g_idx].endswith('-') else norm)
        return sum(scores) / len(scores)

    # Calculate scores
    target_score = get_score(estimated_goals)
    all_scores = [get_score([r[i] for i in goal_indices]) for r in rows]
    
    # Calculate fraction better
    better_count = sum(1 for s in all_scores if target_score > s)
    p = better_count / len(rows)
    
    reason = f"This setup is estimated to outperform {p:.1%} of all historical project configurations based on its input parameters."
    
    return p, reason

import csv

def choose(header, rows, label):
    """
    Finds the best row by sampling 50 rows and evaluating their goals.
    Uses a simple random sampling strategy to explore the search space.
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith(('-', '+'))]
    
    def get_score(goals):
        scores = []
        for i, g_idx in enumerate(goal_indices):
            # Since we don't have the full distribution, we normalize based on 
            # the values we have sampled so far.
            # For this implementation, we use a simple weighted sum of goals.
            # Lower is better for '-', higher for '+'
            val = goals[i]
            if header[g_idx].endswith('-'):
                scores.append(1.0 - val)
            else:
                scores.append(val)
        return sum(scores) / len(scores)

    # We have 50 trials. We will sample 50 indices randomly.
    # If the dataset is smaller than 50, we sample all of them.
    num_samples = min(len(rows), 50)
    indices = random.sample(range(len(rows)), num_samples)
    
    best_idx = -1
    best_score = -1.0
    
    for idx in indices:
        # label(i) returns the true goal values
        goals = label(idx)
        score = get_score(goals)
        
        if score > best_score:
            best_score = score
            best_idx = idx
            
    return best_idx

import csv

def fix(header, row, levers):
    """
    Recommends changes to 'Plan' and 'Team Size' to improve 'Cost-' and 'Idle-'.
    We perform a local search by testing small perturbations within the allowed
    lever ranges to find a configuration that minimizes the goal values.
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith(('-', '+'))]
    
    # We need to estimate the impact of changes. Since we don't have a model,
    # we use a simple heuristic: move towards the center of the allowed range
    # or the opposite of the current value if it's extreme.
    recommendations = {}
    
    for col_name, bounds in levers.items():
        if col_name not in header:
            continue
            
        idx = header.index(col_name)
        low, high = bounds
        
        # If numeric, try to move towards the middle (0.5) or away from current
        # if current is high (since we want to minimize Cost and Idle)
        current_val = row[idx]
        
        # Simple strategy: if current is high, move low; if low, move high.
        # Given the data is normalized [0, 1], we aim for the opposite of current.
        if current_val > 0.5:
            new_val = low
        else:
            new_val = high
            
        # Ensure we stay within bounds
        new_val = max(low, min(high, new_val))
        recommendations[col_name] = new_val
        
    return recommendations

import csv
import math

def spot(header, rows):
    """
    Identifies rows that are outliers based on the Mahalanobis-like distance
    of their input features. Rows with values significantly far from the 
    centroid of the input space are flagged as suspicious.
    """
    input_indices = [i for i, name in enumerate(header) 
                     if not (name.endswith('-') or name.endswith('+'))]
    
    # Extract only input columns
    data = [[row[i] for i in input_indices] for row in rows]
    n = len(data)
    m = len(input_indices)
    
    # Calculate means for each input column
    means = [sum(col) / n for col in zip(*data)]
    
    # Calculate standard deviations
    stdevs = []
    for i in range(m):
        variance = sum((row[i] - means[i])**2 for row in data) / n
        stdevs.append(math.sqrt(variance) + 1e-9)
        
    # Calculate a "suspicion score" for each row
    # We use the sum of squared Z-scores (Mahalanobis distance approximation)
    scores = []
    for idx, row in enumerate(data):
        z_score_sq = 0
        for i in range(m):
            z = (row[i] - means[i]) / stdevs[i]
            z_score_sq += z**2
        scores.append((idx, z_score_sq))
        
    # Sort by score descending (most suspicious first)
    scores.sort(key=lambda x: x[1], reverse=True)
    
    # Return the indices of the top 5% most suspicious rows
    num_to_return = max(1, n // 20)
    return [s[0] for s in scores[:num_to_return]]

def race(header, rows, label):
    """
    Compares three strategies: random, greedy, and hill-climbing.
    Budget: 50 labels total.
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith(('-', '+'))]
    
    def get_score(goals):
        scores = []
        for i, g_idx in enumerate(goal_indices):
            val = goals[i]
            # Normalize: 0 is worst, 1 is best
            # Since we don't know global min/max, we assume [0,1] range from data
            if header[g_idx].endswith('-'):
                scores.append(1.0 - val)
            else:
                scores.append(val)
        return sum(scores) / len(scores)

    budget = 50
    tried_indices = set()
    
    def get_labeled_score(idx):
        if idx not in tried_indices:
            if len(tried_indices) >= budget:
                return -1.0
            tried_indices.add(idx)
        return get_score(label(idx))

    # Strategy 1: Random
    def strategy_random():
        best_idx = -1
        best_score = -1.0
        for _ in range(16): # 16 trials
            idx = random.randint(0, len(rows) - 1)
            score = get_labeled_score(idx)
            if score > best_score:
                best_score, best_idx = score, idx
        return best_idx, best_score

    # Strategy 2: Greedy (Sample 17, pick best)
    def strategy_greedy():
        best_idx = -1
        best_score = -1.0
        for _ in range(17):
            idx = random.randint(0, len(rows) - 1)
            score = get_labeled_score(idx)
            if score > best_score:
                best_score, best_idx = score, idx
        return best_idx, best_score

    # Strategy 3: Hill-Climbing (Start random, look at neighbors)
    def strategy_hill():
        curr = random.randint(0, len(rows) - 1)
        curr_score = get_labeled_score(curr)
        for _ in range(16): # 17 total labels
            neighbor = random.randint(0, len(rows) - 1)
            n_score = get_labeled_score(neighbor)
            if n_score > curr_score:
                curr, curr_score = neighbor, n_score
        return curr, curr_score

    results = {
        "random": strategy_random(),
        "greedy": strategy_greedy(),
        "hill": strategy_hill()
    }
    
    winner = max(results, key=lambda k: results[k][1])
    return {
        "winner": winner,
        "row": results[winner][0],
        "tried": ["random", "greedy", "hill"]
    }

import csv

def find_similar(header, rows, target_row, k=5):
    """
    Finds the k most similar rows to the target_row based on input features.
    Uses Euclidean distance on normalized input columns.
    """
    input_indices = [i for i, name in enumerate(header) 
                     if not (name.endswith('-') or name.endswith('+'))]
    
    # Normalize inputs for distance calculation
    data = [[row[i] for i in input_indices] for row in rows]
    mins = [min(col) for col in zip(*data)]
    maxs = [max(col) for col in zip(*data)]
    
    def normalize(r):
        return [(r[i] - mins[j]) / (maxs[j] - mins[j] + 1e-9) 
                for j, i in enumerate(input_indices)]

    target_norm = normalize(target_row)
    
    distances = []
    for idx, row in enumerate(rows):
        row_norm = normalize(row)
        dist = sum((a - b)**2 for a, b in zip(target_norm, row_norm))**0.5
        distances.append((idx, dist))
    
    # Return indices of k nearest neighbors
    distances.sort(key=lambda x: x[1])
    return [d[0] for d in distances[:k]]

import csv
from scipy import stats

def certify(xs, ys):
    """
    Returns True if the two distributions are statistically similar (same population),
    False if they differ significantly. Uses Welch's t-test.
    """
    # If the samples are too small, we cannot reliably reject the null hypothesis
    if len(xs) < 2 or len(ys) < 2:
        return True
    
    # Welch's t-test (does not assume equal variance)
    # If p-value > 0.05, we fail to reject the null hypothesis (they are the same)
    t_stat, p_value = stats.ttest_ind(xs, ys, equal_var=False)
    return p_value > 0.05

def analyze_shift(header, rows):
    """
    Splits the data into two halves (before and after the change in spring)
    and compares the aggregate performance of the goals.
    """

    def get_score(row):
        # Normalize goals to [0, 1] based on the whole dataset
        scores = []
        for i in goal_indices:
            col_vals = [r[i] for r in rows]
            c_min, c_max = min(col_vals), max(col_vals)
            norm = (row[i] - c_min) / (c_max - c_min + 1e-9)
            scores.append(1 - norm if header[i].endswith('-') else norm)
        return sum(scores) / len(scores)

    # Split the data into two halves
    mid = len(rows) // 2
    before = [get_score(row) for row in rows[:mid]]
    after = [get_score(row) for row in rows[mid:]]
    
    is_same = certify(before, after)
    return is_same, sum(before)/len(before), sum(after)/len(after)

import csv

def flow(header, row, seen):
    """
    Warns the user if the current row is significantly worse than the 
    historical baseline (the 'seen' data).
    
    Strategy:
    1. Calculate a normalized score for the current row and all seen rows.
    2. If the current row's score is significantly lower (e.g., 2 standard 
       deviations below the mean of seen rows), return True to flag it as 'bad'.
    """
    if not seen:
        return False

    goal_indices = [i for i, name in enumerate(header) if name.endswith(('-', '+'))]
    
    def get_score(r):
        scores = []
        for i in goal_indices:
            col_vals = [s[i] for s in seen + [r]]
            c_min, c_max = min(col_vals), max(col_vals)
            rng = c_max - c_min + 1e-9
            norm = (r[i] - c_min) / rng
            scores.append(1 - norm if header[i].endswith('-') else norm)
        return sum(scores) / len(scores)

    # Calculate scores for all seen rows
    seen_scores = [get_score(s) for s in seen]
    current_score = get_score(row)
    
    # Calculate mean and standard deviation of seen performance
    mean_score = sum(seen_scores) / len(seen_scores)
    variance = sum((s - mean_score)**2 for s in seen_scores) / len(seen_scores)
    stdev = variance**0.5
    
    # Flag as 'bad' if it is more than 1.5 standard deviations below the mean
    # (This threshold is sensitive enough to catch performance degradation)
    if current_score < (mean_score - 1.5 * stdev):
        return True
    
    return False

import csv

def blame(header, rows):
    """
    Identifies the input columns that most strongly correlate with high costs.
    We define 'bad' rows as those in the top 25% of 'Cost-' values.
    We then calculate the difference in mean values between these bad rows
    and the rest of the population to identify key drivers.
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith(('-', '+'))]
    input_indices = [i for i, name in enumerate(header) if not (name.endswith('-') or name.endswith('+'))]
    
    cost_idx = header.index('Cost-')
    
    # Sort rows by Cost- descending to find the worst performers
    sorted_rows = sorted(rows, key=lambda r: r[cost_idx], reverse=True)
    worst_rows = sorted_rows[:len(rows) // 4]
    
    # Calculate mean of inputs for the whole population and the worst rows
    all_means = [sum(r[i] for r in rows) / len(rows) for i in input_indices]
    worst_means = [sum(r[i] for r in worst_rows) / len(worst_rows) for i in input_indices]
    
    # Calculate the impact score: (mean_worst - mean_all)
    # A high positive difference means this input is higher in expensive projects.
    # A high negative difference means this input is lower in expensive projects.
    # We take the absolute value to find the "most important" factors.
    impacts = []
    for idx, i in enumerate(input_indices):
        diff = abs(worst_means[idx] - all_means[idx])
        impacts.append((header[i], diff))
    
    # Sort by impact descending
    impacts.sort(key=lambda x: x[1], reverse=True)
    
    # Return the top 2 most important and the bottom 2 (safest to ignore)
    top_two = [name for name, score in impacts[:2]]
    return top_two

import csv

def justify(header, rows, row):
    """
    Compares a candidate row against historical data.
    Estimates goal values for the candidate using the average of the 
    nearest 5 neighbors in the input space.
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith(('-', '+'))]
    input_indices = [i for i, name in enumerate(header) if not (name.endswith('-') or name.endswith('+'))]
    
    # Helper to calculate normalized score for a row
    def get_score(r):
        scores = []
        for i in goal_indices:
            col_vals = [row_data[i] for row_data in rows]
            c_min, c_max = min(col_vals), max(col_vals)
            rng = c_max - c_min + 1e-9
            norm = (r[i] - c_min) / rng
            scores.append(1 - norm if header[i].endswith('-') else norm)
        return sum(scores) / len(scores)

    # Estimate goals for the candidate row using 5-Nearest Neighbors
    # Calculate Euclidean distance in input space
    target_inputs = [row[i] for i in input_indices]
    distances = []
    for idx, r in enumerate(rows):
        dist = sum((r[i] - target_inputs[j])**2 for j, i in enumerate(input_indices))**0.5
        distances.append((dist, r))
    
    distances.sort(key=lambda x: x[0])
    neighbors = distances[:5]
    
    # Estimate goals by averaging neighbors
    estimated_goals = []
    for i in goal_indices:
        estimated_goals.append(sum(n[1][i] for n in neighbors) / 5.0)
    
    # Construct a full row with estimated goals
    full_row = [0.0] * len(header)
    for i, val in enumerate(row):
        if val != '?':
            full_row[i] = val
    for idx, g_idx in enumerate(goal_indices):
        full_row[g_idx] = estimated_goals[idx]
        
    # Calculate fraction of rows that this candidate is better than
    target_score = get_score(full_row)
    better_count = sum(1 for r in rows if target_score > get_score(r))
    p = better_count / len(rows)
    
    reason = f"This setup is estimated to outperform {p:.1%} of historical configurations, making it less optimal than the recommended setup."
    
    return p, reason

import csv

def choose(header, rows, label):
    """
    Finds the best setup for a 'large' project (high 'Size' value).
    We filter for rows where 'Size' is in the top 25% to ensure we are 
    looking at 'large' projects, then use a small budget of 50 labels 
    to find the best performer among them.
    """

    goal_indices = [i for i, name in enumerate(header) if name.endswith(('-', '+'))]

    # We don't have the full distribution, so we estimate based on the rows provided
    all_sizes = [row[size_idx] for row in rows]
    size_threshold = sorted(all_sizes)[int(len(rows) * 0.75)]
    large_indices = [i for i, row in enumerate(rows) if row[size_idx] >= size_threshold]
    
    # If we have fewer than 50 large projects, sample all of them.
    # Otherwise, sample 50 random ones from the large pool.
    trials = min(len(large_indices), 50)
    candidates = random.sample(large_indices, trials)
    
    def get_score(goals):
        scores = []
        for i, g_idx in enumerate(goal_indices):
            # Normalize goals: 0 is worst, 1 is best

            val = goals[i]
            if header[g_idx].endswith('-'):
                scores.append(1.0 - val)
            else:
                scores.append(val)
        return sum(scores) / len(scores)
    
    best_idx = -1
    best_score = -1.0
    
    for idx in candidates:
        goals = label(idx)
        score = get_score(goals)
        if score > best_score:
            best_score = score
            best_idx = idx
            
    return best_idx

import csv

def fix(header, row, levers):
    """
    Recommends changes to cut 'Idle-' time without changing 'Team Size'.
    We identify the most impactful levers (excluding Team Size) by finding
    which inputs correlate most strongly with lower 'Idle-' values.
    """
    # Identify goal indices
    goal_indices = [i for i, name in enumerate(header) if name.endswith(('-', '+'))]
    idle_idx = header.index('Idle-')
    
    # We want to change levers to minimize Idle-.
    # We look for a lever that, when changed, moves the project toward 
    # the characteristics of low-idle projects.
    recommendations = {}
    
    # We prioritize changing 'Plan' or other available levers (excluding Team Size)
    # as requested by the customer.
    for col_name, bounds in levers.items():
        if col_name == 'Team Size':
            continue
        
        idx = header.index(col_name)
        
        # If numeric, we perform a simple gradient-like shift.
        # Since we want to minimize Idle-, we try to move the lever 
        # to the opposite side of the current value.
        if isinstance(bounds, tuple):
            low, high = bounds
            current_val = row[idx]
            # If current is high, move low; if low, move high.
            if current_val > 0.5:
                new_val = low
            else:
                new_val = high
            recommendations[col_name] = new_val
        else:
            # If symbolic, pick the first option as a default change
            recommendations[col_name] = bounds[0]
            
    return recommendations

import csv

def spot(header, rows):
    """
    Identifies rows that contain input values that are suspicious.
    Suspicion is defined by calculating the Z-score of each input feature
    and flagging rows where the sum of squared Z-scores is significantly
    higher than the average, indicating an outlier in the input space.
    """
    input_indices = [i for i, name in enumerate(header) 
                     if not (name.endswith('-') or name.endswith('+'))]
    
    # Extract only input columns
    data = [[row[i] for i in input_indices] for row in rows]
    n = len(data)
    m = len(input_indices)
    
    # Calculate means for each input column
    means = [sum(col) / n for col in zip(*data)]
    
    # Calculate standard deviations
    stdevs = []
    for i in range(m):
        variance = sum((row[i] - means[i])**2 for row in data) / n
        stdevs.append((variance**0.5) + 1e-9)
        
    # Calculate a "suspicion score" for each row
    # We use the sum of squared Z-scores (Mahalanobis distance approximation)
    scores = []
    for idx, row in enumerate(data):
        z_score_sq = 0
        for i in range(m):
            z = (row[i] - means[i]) / stdevs[i]
            z_score_sq += z**2
        scores.append((idx, z_score_sq))
        
    # Sort by score descending (most suspicious first)
    scores.sort(key=lambda x: x[1], reverse=True)
    
    # Return the indices of the top 5% most suspicious rows
    num_to_return = max(1, n // 20)
    return [s[0] for s in scores[:num_to_return]]
