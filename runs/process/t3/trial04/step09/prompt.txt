def remember(header, rows):
    """
    Identifies rows that perform well across all goal columns.
    Goals ending in '-' are minimized, goals ending in '+' are maximized.
    """
    df = pd.DataFrame(rows, columns=header)
    
    # Identify goal columns
    goals = [col for col in header if col.endswith(('-', '+'))]
    
    # Normalize goals to a [0, 1] range to compare them equally
    # For '-', lower is better (1 - normalized_value)
    # For '+', higher is better (normalized_value)
    scores = pd.DataFrame(index=df.index)
    for goal in goals:
        col_min = df[goal].min()
        col_max = df[goal].max()
        norm = (df[goal] - col_min) / (col_max - col_min + 1e-9)
        
        if goal.endswith('-'):
            scores[goal] = 1 - norm
        else:
            scores[goal] = norm
            
    # Calculate aggregate score (mean of normalized goal scores)
    df['aggregate'] = scores.mean(axis=1)
    
    # Select top 5% of rows
    num_to_select = max(1, len(rows) // 20)
    best_indices = df['aggregate'].nlargest(num_to_select).index.tolist()
    
    return sorted(best_indices)

import csv
import pandas as pd
from scipy import stats

def certify(xs, ys):
    """
    Returns True if the two distributions are statistically similar (same population),
    False if they differ significantly. Uses Welch's t-test.
    """
    # If the samples are too small, we cannot reliably reject the null hypothesis
    if len(xs) < 2 or len(ys) < 2:
        return True
    
    # Welch's t-test (does not assume equal variance)
    t_stat, p_value = stats.ttest_ind(xs, ys, equal_var=False)
    
    # If p-value > 0.05, we fail to reject the null hypothesis (they are the same)
    return p_value > 0.05

def analyze_team_size_impact(header, rows):
    df = pd.DataFrame(rows, columns=header)

import csv

def flow(header, row, seen):
    """
    Implements a Secretary Problem-style decision strategy.
    We observe a sample of the data to establish a baseline (the 'best' seen so far),
    then accept the first row that is better than that baseline.
    """
    # Identify goal indices
    goal_indices = [i for i, name in enumerate(header) if name.endswith(('-', '+'))]
    
    def get_score(r):
        # Normalize goals: lower is better for '-', higher for '+'
        # Since we don't have global min/max, we use the current 'seen' data
        # to estimate relative quality.
        score = 0
        for i in goal_indices:
            val = r[i]
            name = header[i]
            # Simple heuristic: if we have seen data, compare against it
            if seen:
                col_vals = [s[i] for s in seen]
                c_min, c_max = min(col_vals), max(col_vals)
                rng = c_max - c_min + 1e-9
                norm = (val - c_min) / rng
                score += (1 - norm) if name.endswith('-') else norm
            else:
                score = 0
        return score / len(goal_indices) if goal_indices else 0

    # Strategy: 
    # 1. Observe first 20% of data to establish a 'gold standard'
    # 2. Accept the first row that exceeds the best score found in the observation phase
    sample_size = max(1, len(seen) // 5)
    
    if len(seen) < sample_size:
        return False
    
    # Calculate best score from the observation phase
    best_score = -1.0
    for s in seen[:sample_size]:
        best_score = max(best_score, get_score(s))
        
    return get_score(row) > best_score

import csv

def blame(header, rows):
    """
    Identifies input columns that correlate most strongly with the 'Idle-' goal
    in the worst-performing setups.
    """
    # Identify goal columns and input columns
    goal_indices = [i for i, name in enumerate(header) if name.endswith(('-', '+'))]
    input_indices = [i for i, name in enumerate(header) if not name.endswith(('-', '+'))]
    
    # Identify the 'Idle-' column index
    idle_idx = -1
    for i in goal_indices:
        if header[i] == 'Idle-':
            idle_idx = i
            break
            
    if idle_idx == -1:
        return []

    # Calculate a composite score for each row to define "worst"
    # A row is bad if its goals are high (for -) or low (for +)
    def get_badness(row):
        badness = 0
        for i in goal_indices:
            val = row[i]
            # Since we want to find the worst, we look for high values in '-' goals
            # and low values in '+' goals.
            if header[i].endswith('-'):
                badness += val
            else:
                badness += (1.0 - val)
        return badness

    # Sort rows by badness and take the worst 20%
    sorted_rows = sorted(rows, key=get_badness, reverse=True)
    worst_rows = sorted_rows[:len(rows) // 5]
    
    # For each input column, calculate the correlation with 'Idle-'
    # We use a simple difference in means: (mean in worst) - (mean in all)
    correlations = []
    all_idle_mean = sum(r[idle_idx] for r in rows) / len(rows)
    worst_idle_mean = sum(r[idle_idx] for r in worst_rows) / len(worst_rows)
    
    for i in input_indices:
        # Calculate mean of this input in the worst rows
        input_mean_worst = sum(r[i] for r in worst_rows) / len(worst_rows)
        input_mean_all = sum(r[i] for r in rows) / len(rows)
        
        # If the input is high when Idle is high, it's a candidate for blame
        # We use the product of differences to see if they move together
        score = (input_mean_worst - input_mean_all) * (worst_idle_mean - all_idle_mean)
        correlations.append((header[i], score))
    
    # Return the top 3 inputs that correlate with high Idle-
    correlations.sort(key=lambda x: x[1], reverse=True)
    return [name for name, score in correlations[:3]]

import csv

def justify(header, rows, row):
    """
    Returns (p, reason) where p is the fraction of rows that 'row' is better than,
    and reason is a one-sentence explanation.
    """
    # Identify indices
    goal_indices = [i for i, name in enumerate(header) if name.endswith(('-', '+'))]
    input_indices = [i for i, name in enumerate(header) if not (name.endswith('-') or name.endswith('+'))]
    
    # Prepare training data for KNN
    X_train = [[r[i] for i in input_indices] for r in rows]
    
    # Estimate goals for the target row
    estimated_goals = []
    for g_idx in goal_indices:
        y_train = [r[g_idx] for r in rows]
        model = KNeighborsRegressor(n_neighbors=5)
        model.fit(X_train, y_train)
        estimated_goals.append(model.predict([row[:len(input_indices)]])[0])
    
    # Define a scoring function: normalized distance to ideal
    def get_score(goals):
        scores = []
        for i, g_idx in enumerate(goal_indices):
            col_vals = [r[g_idx] for r in rows]
            c_min, c_max = min(col_vals), max(col_vals)
            rng = c_max - c_min + 1e-9
            norm = (goals[i] - c_min) / rng
            scores.append(1 - norm if header[g_idx].endswith('-') else norm)
        return sum(scores) / len(scores)

    # Calculate scores
    target_score = get_score(estimated_goals)
    all_scores = [get_score([r[i] for i in goal_indices]) for r in rows]
    
    # Calculate fraction better
    better_count = sum(1 for s in all_scores if target_score > s)
    p = better_count / len(rows)
    
    reason = f"This setup is estimated to outperform {p:.1%} of all historical project configurations based on its input parameters."
    
    return p, reason

import csv

def choose(header, rows, label):
    """
    Finds the best row by sampling 50 rows and evaluating their goals.
    Uses a simple random sampling strategy to explore the search space.
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith(('-', '+'))]
    
    def get_score(goals):
        scores = []
        for i, g_idx in enumerate(goal_indices):
            # Since we don't have the full distribution, we normalize based on 
            # the values we have sampled so far.
            # For this implementation, we use a simple weighted sum of goals.
            # Lower is better for '-', higher for '+'
            val = goals[i]
            if header[g_idx].endswith('-'):
                scores.append(1.0 - val)
            else:
                scores.append(val)
        return sum(scores) / len(scores)

    # We have 50 trials. We will sample 50 indices randomly.
    # If the dataset is smaller than 50, we sample all of them.
    num_samples = min(len(rows), 50)
    indices = random.sample(range(len(rows)), num_samples)
    
    best_idx = -1
    best_score = -1.0
    
    for idx in indices:
        # label(i) returns the true goal values
        goals = label(idx)
        score = get_score(goals)
        
        if score > best_score:
            best_score = score
            best_idx = idx
            
    return best_idx

import csv

def fix(header, row, levers):
    """
    Recommends changes to 'Plan' and 'Team Size' to improve 'Cost-' and 'Idle-'.
    We perform a local search by testing small perturbations within the allowed
    lever ranges to find a configuration that minimizes the goal values.
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith(('-', '+'))]
    
    # We need to estimate the impact of changes. Since we don't have a model,
    # we use a simple heuristic: move towards the center of the allowed range
    # or the opposite of the current value if it's extreme.
    recommendations = {}
    
    for col_name, bounds in levers.items():
        if col_name not in header:
            continue
            
        idx = header.index(col_name)
        low, high = bounds
        
        # If numeric, try to move towards the middle (0.5) or away from current
        # if current is high (since we want to minimize Cost and Idle)
        current_val = row[idx]
        
        # Simple strategy: if current is high, move low; if low, move high.
        # Given the data is normalized [0, 1], we aim for the opposite of current.
        if current_val > 0.5:
            new_val = low
        else:
            new_val = high
            
        # Ensure we stay within bounds
        new_val = max(low, min(high, new_val))
        recommendations[col_name] = new_val
        
    return recommendations
