import csv

def spot(header, rows):
    """
    Identifies rows that contain outliers. We use the Interquartile Range (IQR)
    method on each input column to detect values that are statistically 
    unlikely (outside 1.5 * IQR). Rows with more outliers are considered 
    more suspicious.
    """
    input_indices = [i for i, h in enumerate(header) 
                     if not (h.endswith('-') or h.endswith('+'))]
    
    # Collect data for each input column
    col_data = {i: [] for i in input_indices}
    for row in rows:
        for i in input_indices:
            col_data[i].append(row[i])
            
    # Calculate IQR bounds for each input column
    bounds = {}
    for i in input_indices:
        vals = sorted(col_data[i])
        q1 = vals[int(len(vals) * 0.25)]
        q3 = vals[int(len(vals) * 0.75)]
        iqr = q3 - q1
        bounds[i] = (q1 - 1.5 * iqr, q3 + 1.5 * iqr)
        
    # Score rows based on how many values fall outside the bounds
    suspicion_scores = []
    for idx, row in enumerate(rows):
        outlier_count = 0
        for i in input_indices:
            val = row[i]
            low, high = bounds[i]
            if val < low or val > high:
                outlier_count += 1
        suspicion_scores.append((outlier_count, idx))
        
    # Sort by count descending, then by index
    suspicion_scores.sort(key=lambda x: (-x[0], x[1]))
    
    # Return only rows that have at least one outlier
    return [idx for count, idx in suspicion_scores if count > 0]

if __name__ == "__main__":
    filename = 'pom3a.n1000.train.csv'
    try:
        with open(filename, 'r') as f:
            reader = csv.reader(f)
            header = next(reader)
            rows = []
            for row in reader:
                processed_row = []
                for val in row:
                    try:
                        processed_row.append(float(val))
                    except ValueError:
                        processed_row.append(val)
                rows.append(processed_row)
        
        suspicious_indices = spot(header, rows)
        
        print("Customer Report: Data Quality Analysis")
        if not suspicious_indices:
            print("No suspicious records found.")
        else:
            print(f"Identified {len(suspicious_indices)} potentially mistyped records.")
            print("The most suspicious row indices (ordered by severity) are:")
            print(suspicious_indices[:10])
            print("These rows contain input values that fall significantly outside the expected statistical range.")
            
    except FileNotFoundError:
        print(f"Error: {filename} not found.")
        exit(1)
