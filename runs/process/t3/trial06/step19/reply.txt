import csv

def spot(header, rows):
    """
    Identifies rows that are 'unlike anything we have run'.
    We use a distance-based approach: for each row, calculate the distance
    to its nearest neighbor in the input space. Rows with a large distance
    to their nearest neighbor are 'unseen' or 'unlike' previous setups.
    """
    input_indices = [i for i, h in enumerate(header) 
                     if not (h.endswith('-') or h.endswith('+'))]
    
    # Extract input data for distance calculation
    inputs = [[row[i] for i in input_indices] for row in rows]
    
    # Calculate distance to nearest neighbor for each row
    # Using Euclidean distance
    distances = []
    for i in range(len(inputs)):
        min_dist = float('inf')
        for j in range(len(inputs)):
            if i == j:
                continue
            dist = sum((inputs[i][k] - inputs[j][k])**2 for k in range(len(input_indices)))**0.5
            if dist < min_dist:
                min_dist = dist
        distances.append((min_dist, i))
    
    # Sort by distance descending (most distant/unseen first)
    distances.sort(key=lambda x: x[0], reverse=True)
    
    # Return indices of the most suspicious (unseen) rows
    return [idx for dist, idx in distances]

if __name__ == "__main__":
    filename = 'pom3a.n1000.train.csv'
    try:
        with open(filename, 'r') as f:
            reader = csv.reader(f)
            header = next(reader)
            rows = []
            for row in reader:
                processed_row = []
                for val in row:
                    try:
                        processed_row.append(float(val))
                    except ValueError:
                        processed_row.append(val)
                rows.append(processed_row)
        
        suspicious_indices = spot(header, rows)
        
        print("Customer Report: Novelty Analysis")
        print("We have identified setups that are significantly different from your historical data.")
        print("The following row indices represent configurations you have never seen before:")
        print(suspicious_indices[:10])
        print("These setups are outliers in your input space and may carry higher risk.")
            
    except FileNotFoundError:
        print(f"Error: {filename} not found.")
        exit(1)
