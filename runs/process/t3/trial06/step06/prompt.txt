import csv
import pandas as pd
import numpy as np

def remember(header, rows):
    """
    Identifies rows that perform well across all goal columns.
    Goals ending in '-' are minimized, '+' are maximized.
    """
    df = pd.DataFrame(rows, columns=header)
    
    # Identify goal columns
    goals = [c for c in header if c.endswith('-') or c.endswith('+')]
    
    # Normalize goals to [0, 1] range to treat them equally
    # For '-', lower is better (1 - normalized_val)
    # For '+', higher is better (normalized_val)
    scores = []
    for col in goals:
        vals = df[col].astype(float)
        norm = (vals - vals.min()) / (vals.max() - vals.min())
        if col.endswith('-'):
            scores.append(1 - norm)
        else:
            scores.append(norm)
            
    # Combine scores: average of normalized goal performance
    combined = np.mean(scores, axis=0)
    
    # Select top 5% (or at least 1)
    n_select = max(1, len(rows) // 20)
    best_indices = np.argsort(combined)[-n_select:].tolist()
    
    return sorted(best_indices)

import csv
import pandas as pd
import numpy as np
from scipy import stats

def certify(xs, ys):
    """
    Returns True if the difference between two distributions is statistically 
    insignificant (likely noise), False if they are significantly different.
    Uses Welch's t-test for independent samples with unequal variances.
    """
    # If the samples are too small, we cannot reliably reject the null hypothesis
    if len(xs) < 2 or len(ys) < 2:
        return True
    
    # Perform Welch's t-test
    t_stat, p_value = stats.ttest_ind(xs, ys, equal_var=False)
    
    # If p-value > 0.05, we fail to reject the null hypothesis (they are the same)
    return p_value > 0.05

import csv
import numpy as np

def flow(header, row, seen):
    """
    Decides whether to accept a row based on the Secretary Problem logic
    adapted for multi-objective optimization. We observe a sample to 
    establish a baseline, then accept the first row that outperforms 
    the best seen so far.
    """
    goal_indices = [i for i, h in enumerate(header) if h.endswith('-') or h.endswith('+')]
    
    def get_score(r):
        # Normalize goals: 0 to 1, where 1 is best
        scores = []
        for i in goal_indices:
            val = r[i]
            # Since we don't have global min/max easily, we use relative 
            # performance against the 'seen' set
            col_vals = [s[i] for s in seen]
            if not col_vals:
                return 0
            
            min_v, max_v = min(col_vals), max(col_vals)
            rng = max_v - min_v if max_v != min_v else 1
            norm = (val - min_v) / rng
            
            if header[i].endswith('-'):
                scores.append(1 - norm)
            else:
                scores.append(norm)
        return np.mean(scores)

    # If it's the first row, we can't evaluate it yet
    if not seen:
        return False
    
    # Secretary problem: explore 20% of the data to set a threshold
    # If we haven't seen enough, keep looking
    if len(seen) < 160:
        return False
    
    # Calculate threshold: best score seen so far
    best_score = max(get_score(s) for s in seen)
    current_score = get_score(row)
    
    # Accept if it's better than anything seen before
    return current_score > best_score

if __name__ == "__main__":

        print("Strategy: Observed first 20% to establish baseline, then accepted improvements.")

        print(f"Error: {filename} not found.")

import csv

def blame(header, rows):
    """
    Identifies which input columns correlate most strongly with the 'Idle-' goal.
    'Idle-' is a goal to minimize, so we look for inputs that have a high 
    correlation with high 'Idle-' values.
    """
    # Identify goal and input columns
    goal_idx = header.index('Idle-')
    input_indices = [i for i, h in enumerate(header) if not (h.endswith('-') or h.endswith('+'))]
    
    # Calculate the correlation between each input and the 'Idle-' goal
    # We use a simple Pearson-like correlation: cov(X, Y) / (std(X) * std(Y))
    # Since we need to identify the "worst" (highest Idle-), we look for 
    # inputs that increase as Idle- increases.
    
    idle_vals = [row[goal_idx] for row in rows]
    mean_idle = sum(idle_vals) / len(idle_vals)
    
    correlations = {}
    for i in input_indices:
        input_vals = [row[i] for row in rows]
        mean_input = sum(input_vals) / len(input_vals)
        
        # Covariance
        num = sum((input_vals[j] - mean_input) * (idle_vals[j] - mean_idle) for j in range(len(rows)))
        # Variance
        den_x = sum((input_vals[j] - mean_input) ** 2 for j in range(len(rows)))
        den_y = sum((idle_vals[j] - mean_idle) ** 2 for j in range(len(rows)))
        
        if den_x > 0 and den_y > 0:
            correlations[header[i]] = num / ((den_x * den_y) ** 0.5)
        else:
            correlations[header[i]] = 0
            
    # Sort by correlation descending
    sorted_inputs = sorted(correlations.items(), key=lambda x: x[1], reverse=True)
    
    # Return the top 2 inputs that contribute most to high idle time
    return [name for name, corr in sorted_inputs[:2]]
