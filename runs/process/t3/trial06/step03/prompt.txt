--- step01.py ---
import csv
import pandas as pd
import numpy as np

def remember(header, rows):
    """
    Identifies rows that perform well across all goal columns.
    Goals ending in '-' are minimized, '+' are maximized.
    """
    df = pd.DataFrame(rows, columns=header)
    
    # Identify goal columns
    goals = [c for c in header if c.endswith('-') or c.endswith('+')]
    
    # Normalize goals to [0, 1] range to treat them equally
    # For '-', lower is better (1 - normalized_val)
    # For '+', higher is better (normalized_val)
    scores = []
    for col in goals:
        vals = df[col].astype(float)
        norm = (vals - vals.min()) / (vals.max() - vals.min())
        if col.endswith('-'):
            scores.append(1 - norm)
        else:
            scores.append(norm)
            
    # Combine scores: average of normalized goal performance
    combined = np.mean(scores, axis=0)
    
    # Select top 5% (or at least 1)
    n_select = max(1, len(rows) // 20)
    best_indices = np.argsort(combined)[-n_select:].tolist()
    
    return sorted(best_indices)

import csv
import pandas as pd
import numpy as np
from sklearn.neighbors import KNeighborsRegressor

def guess(header, train, test):
    """
    Estimates the first goal column for test rows using K-Nearest Neighbors.
    """
    df_train = pd.DataFrame(train, columns=header)
    df_test = pd.DataFrame(test, columns=header)
    
    # Identify input columns (those not ending in - or +)
    inputs = [c for c in header if not (c.endswith('-') or c.endswith('+'))]
    
    # Identify the first goal column
    goals = [c for c in header if c.endswith('-') or c.endswith('+')]
    target = goals[0]
    
    # Prepare data
    X_train = df_train[inputs]
    y_train = df_train[target]
    X_test = df_test[inputs]
    
    # Use KNN to predict the target value
    # n_neighbors=5 is a robust default for this dataset size
    model = KNeighborsRegressor(n_neighbors=5)
    model.fit(X_train, y_train)
    
    predictions = model.predict(X_test)
    return predictions.tolist()

if __name__ == "__main__":
