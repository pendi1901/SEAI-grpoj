import csv
import pandas as pd
import numpy as np

def remember(header, rows):
    """
    Identifies rows that perform well across all goal columns.
    Goals ending in '-' are minimized, '+' are maximized.
    """
    df = pd.DataFrame(rows, columns=header)
    
    # Identify goal columns
    goals = [c for c in header if c.endswith('-') or c.endswith('+')]
    
    # Normalize goals to [0, 1] range to treat them equally
    # For '-', lower is better (1 - normalized_val)
    # For '+', higher is better (normalized_val)
    scores = []
    for col in goals:
        vals = df[col].astype(float)
        norm = (vals - vals.min()) / (vals.max() - vals.min())
        if col.endswith('-'):
            scores.append(1 - norm)
        else:
            scores.append(norm)
            
    # Combine scores: average of normalized goal performance
    combined = np.mean(scores, axis=0)
    
    # Select top 5% (or at least 1)
    n_select = max(1, len(rows) // 20)
    best_indices = np.argsort(combined)[-n_select:].tolist()
    
    return sorted(best_indices)

from scipy import stats

def certify(xs, ys):
    """
    Returns True if the difference between two distributions is statistically 
    insignificant (likely noise), False if they are significantly different.
    Uses Welch's t-test for independent samples with unequal variances.
    """
    # If the samples are too small, we cannot reliably reject the null hypothesis
    if len(xs) < 2 or len(ys) < 2:
        return True
    
    # Perform Welch's t-test
    t_stat, p_value = stats.ttest_ind(xs, ys, equal_var=False)
    
    # If p-value > 0.05, we fail to reject the null hypothesis (they are the same)
    return p_value > 0.05

import csv
import numpy as np

def flow(header, row, seen):
    """
    Decides whether to accept a row based on the Secretary Problem logic
    adapted for multi-objective optimization. We observe a sample to 
    establish a baseline, then accept the first row that outperforms 
    the best seen so far.
    """
    goal_indices = [i for i, h in enumerate(header) if h.endswith('-') or h.endswith('+')]
    
    def get_score(r):
        # Normalize goals: 0 to 1, where 1 is best
        scores = []
        for i in goal_indices:
            val = r[i]
            # Since we don't have global min/max easily, we use relative 
            # performance against the 'seen' set
            col_vals = [s[i] for s in seen]
            if not col_vals:
                return 0
            
            min_v, max_v = min(col_vals), max(col_vals)
            rng = max_v - min_v if max_v != min_v else 1
            norm = (val - min_v) / rng
            
            if header[i].endswith('-'):
                scores.append(1 - norm)
            else:
                scores.append(norm)
        return np.mean(scores)

    # If it's the first row, we can't evaluate it yet
    if not seen:
        return False
    
    # Secretary problem: explore 20% of the data to set a threshold
    # If we haven't seen enough, keep looking
    if len(seen) < 160:
        return False
    
    # Calculate threshold: best score seen so far
    best_score = max(get_score(s) for s in seen)
    current_score = get_score(row)
    
    # Accept if it's better than anything seen before
    return current_score > best_score

import csv

def blame(header, rows):
    """
    Identifies which input columns correlate most strongly with the 'Idle-' goal.
    'Idle-' is a goal to minimize, so we look for inputs that have a high 
    correlation with high 'Idle-' values.
    """
    # Identify goal and input columns
    goal_idx = header.index('Idle-')
    input_indices = [i for i, h in enumerate(header) if not (h.endswith('-') or h.endswith('+'))]
    
    # Calculate the correlation between each input and the 'Idle-' goal
    # We use a simple Pearson-like correlation: cov(X, Y) / (std(X) * std(Y))
    # Since we need to identify the "worst" (highest Idle-), we look for 
    # inputs that increase as Idle- increases.
    
    idle_vals = [row[goal_idx] for row in rows]
    mean_idle = sum(idle_vals) / len(idle_vals)
    
    correlations = {}
    for i in input_indices:
        input_vals = [row[i] for row in rows]
        mean_input = sum(input_vals) / len(input_vals)
        
        # Covariance
        num = sum((input_vals[j] - mean_input) * (idle_vals[j] - mean_idle) for j in range(len(rows)))
        # Variance
        den_x = sum((input_vals[j] - mean_input) ** 2 for j in range(len(rows)))
        den_y = sum((idle_vals[j] - mean_idle) ** 2 for j in range(len(rows)))
        
        if den_x > 0 and den_y > 0:
            correlations[header[i]] = num / ((den_x * den_y) ** 0.5)
        else:
            correlations[header[i]] = 0
            
    # Sort by correlation descending
    sorted_inputs = sorted(correlations.items(), key=lambda x: x[1], reverse=True)
    
    # Return the top 2 inputs that contribute most to high idle time
    return [name for name, corr in sorted_inputs[:2]]

def get_score(header, row, goal_indices, min_vals, max_vals):
    """Calculates a normalized score [0, 1] for a row based on goals."""
    scores = []
    for i in goal_indices:
        val = row[i]
        rng = max_vals[i] - min_vals[i]
        norm = (val - min_vals[i]) / rng if rng != 0 else 0
        if header[i].endswith('-'):
            scores.append(1 - norm)
        else:
            scores.append(norm)
    return np.mean(scores)

def justify(header, rows, row):
    """
    Estimates the goals for 'row' using the average of the 5 nearest neighbors,
    then ranks it against the existing 'rows'.
    """
    goal_indices = [i for i, h in enumerate(header) if h.endswith('-') or h.endswith('+')]
    input_indices = [i for i, h in enumerate(header) if not (h.endswith('-') or h.endswith('+'))]
    
    # Calculate min/max for normalization
    min_vals = [min(r[i] for r in rows) for i in range(len(header))]
    max_vals = [max(r[i] for r in rows) for i in range(len(header))]
    
    # Estimate goals for the input row using KNN (5 neighbors)
    inputs = [row[i] for i in input_indices]
    distances = []
    for r in rows:
        dist = sum((r[i] - inputs[idx])**2 for idx, i in enumerate(input_indices))**0.5
        distances.append((dist, r))
    
    distances.sort(key=lambda x: x[0])
    neighbors = distances[:5]
    
    estimated_row = list(row)
    for i in goal_indices:
        estimated_row[i] = sum(n[1][i] for n in neighbors) / len(neighbors)
    
    # Calculate score for estimated row
    my_score = get_score(header, estimated_row, goal_indices, min_vals, max_vals)
    
    # Compare against all rows
    all_scores = [get_score(header, r, goal_indices, min_vals, max_vals) for r in rows]
    better_count = sum(1 for s in all_scores if my_score > s)
    p = better_count / len(rows)
    
    reason = f"This setup is estimated to outperform {p:.1%} of all known project configurations based on its input parameters."
    return p, reason

import csv

def fix(header, row, levers):
    """
    Finds the smallest change to levers to improve the row's goals.
    We evaluate the impact of changing each lever to its bounds or 
    midpoints to find an improvement.
    """
    goal_indices = [i for i, h in enumerate(header) if h.endswith('-') or h.endswith('+')]
    
    def get_score(r):
        scores = []
        for i in goal_indices:
            val = r[i]
            # Goals ending in '-' are minimized (1-val), '+' are maximized (val).
            # Data is normalized [0,1].
            if header[i].endswith('-'):
                scores.append(1.0 - val)
            else:
                scores.append(val)
        return sum(scores) / len(scores)

    best_change = {}
    best_score = get_score(row)
    
    # Iterate through each lever and test potential values
    for col_name, allowed in levers.items():
        if col_name not in header:
            continue
        idx = header.index(col_name)
        
        candidates = []
        if isinstance(allowed, tuple): # Numeric (low, high)
            low, high = allowed
            # Test boundaries and midpoint
            candidates = [low, (low + high) / 2.0, high]
        else: # Symbolic
            candidates = allowed
            
        for val in candidates:
            new_row = list(row)
            new_row[idx] = val
            new_score = get_score(new_row)
            
            # We look for any improvement
            if new_score > best_score:
                best_score = new_score
                best_change = {col_name: val}
                
    return best_change

import csv

def spot(header, rows):
    """
    Identifies rows that contain outliers. We use the Interquartile Range (IQR)
    method on each input column to detect values that are statistically 
    unlikely (outside 1.5 * IQR). Rows with more outliers are considered 
    more suspicious.
    """
    input_indices = [i for i, h in enumerate(header) 
                     if not (h.endswith('-') or h.endswith('+'))]
    
    # Collect data for each input column
    col_data = {i: [] for i in input_indices}
    for row in rows:
        for i in input_indices:
            col_data[i].append(row[i])
            
    # Calculate IQR bounds for each input column
    bounds = {}
    for i in input_indices:
        vals = sorted(col_data[i])
        q1 = vals[int(len(vals) * 0.25)]
        q3 = vals[int(len(vals) * 0.75)]
        iqr = q3 - q1
        bounds[i] = (q1 - 1.5 * iqr, q3 + 1.5 * iqr)
        
    # Score rows based on how many values fall outside the bounds
    suspicion_scores = []
    for idx, row in enumerate(rows):
        outlier_count = 0
        for i in input_indices:
            val = row[i]
            low, high = bounds[i]
            if val < low or val > high:
                outlier_count += 1
        suspicion_scores.append((outlier_count, idx))
        
    # Sort by count descending, then by index
    suspicion_scores.sort(key=lambda x: (-x[0], x[1]))
    
    # Return only rows that have at least one outlier
    return [idx for count, idx in suspicion_scores if count > 0]

            best_g_idx = idx

    curr_idx = random.randint(0, len(rows) - 1)

    winner = max(results, key=lambda k: results[k][0])

import csv

def remember(header, rows):
    """
    Identifies the best rows by normalizing goal columns and calculating
    a weighted performance score. Returns the indices of the top 5% of rows.
    """
    goal_indices = [i for i, h in enumerate(header) if h.endswith('-') or h.endswith('+')]
    
    # Extract goal values for normalization
    goals_data = [[row[i] for i in goal_indices] for row in rows]
    
    # Calculate min and max for each goal to normalize to [0, 1]
    mins = [min(g[i] for g in goals_data) for i in range(len(goal_indices))]
    maxs = [max(g[i] for g in goals_data) for i in range(len(goal_indices))]
    
    def get_score(row):
        scores = []
        for idx, i in enumerate(goal_indices):
            val = row[i]
            rng = maxs[idx] - mins[idx]
            norm = (val - mins[idx]) / rng if rng != 0 else 0
            # '-' goals: lower is better (1 - norm), '+' goals: higher is better (norm)
            if header[i].endswith('-'):
                scores.append(1.0 - norm)
            else:
                scores.append(norm)
        return sum(scores) / len(scores)

    # Score all rows
    scored_rows = []
    for idx, row in enumerate(rows):
        scored_rows.append((get_score(row), idx))
    
    # Sort by score descending
    scored_rows.sort(key=lambda x: x[0], reverse=True)
    
    # Return top 5% (or at least 1)
    n_select = max(1, len(rows) // 20)
    return sorted([item[1] for item in scored_rows[:n_select]])

import csv
from scipy import stats

def certify(xs, ys):
    """
    Returns True if the difference between two distributions is statistically 
    insignificant (likely noise), False if they are significantly different.
    Uses Welch's t-test for independent samples with unequal variances.
    """
    if len(xs) < 2 or len(ys) < 2:
        return True
    
    # Welch's t-test
    t_stat, p_value = stats.ttest_ind(xs, ys, equal_var=False)
    
    # If p-value > 0.05, we fail to reject the null hypothesis (they are the same)
    return p_value > 0.05

def analyze_shift(header, rows):
    """
    Splits the data into two halves to compare performance before and after 
    the change in team decision-making.
    """
    goal_indices = [i for i, h in enumerate(header) if h.endswith('-') or h.endswith('+')]

    def get_score(row):
        scores = []
        for i in goal_indices:
            val = row[i]
            # Simple normalization based on min/max of the whole set
            col_vals = [r[i] for r in rows]
            min_v, max_v = min(col_vals), max(col_vals)
            rng = max_v - min_v if max_v != min_v else 1
            norm = (val - min_v) / rng
            if header[i].endswith('-'):
                scores.append(1.0 - norm)
            else:
                scores.append(norm)
        return sum(scores) / len(scores)

    mid = len(rows) // 2
    before = [get_score(rows[i]) for i in range(mid)]
    after = [get_score(rows[i]) for i in range(mid, len(rows))]
    
    is_same = certify(before, after)
    return is_same, sum(before)/len(before), sum(after)/len(after)

def flow(header, row, seen):
    """
    Detects if the incoming stream of data is 'going bad'.
    We compare the current row's performance against the moving average
    of the last 50 rows. If the current performance drops significantly
    below the recent trend, we flag it (return True).
    """
    if len(seen) < 50:
        return False
    
    # Calculate performance of the current row
    current_score = get_score(header, row)
    
    # Calculate moving average of the last 50 rows
    recent_scores = [get_score(header, r) for r in seen[-50:]]
    avg_recent = sum(recent_scores) / len(recent_scores)
    
    # If the current score is significantly lower than the recent average
    # (e.g., 2 standard deviations or a fixed threshold), it's 'going bad'.
    # Here we use a threshold of 0.15 drop from the moving average.
    if current_score < (avg_recent - 0.15):
        return True
    
    return False

def blame(header, rows):
    """
    Identifies input columns that correlate most strongly with poor performance.
    'Worst' rows are those with the lowest composite goal scores.
    We calculate the correlation between each input and the performance score.
    """
    input_indices = [i for i, h in enumerate(header) if not (h.endswith('-') or h.endswith('+'))]
    
    # Calculate scores for all rows
    scores = [get_score(header, row) for row in rows]
    
    # Calculate correlation between each input and the score
    # We want inputs that, when changed, correlate with the score.
    # We look for inputs that have a high absolute correlation with the score.
    correlations = {}
    mean_score = sum(scores) / len(scores)
    var_score = sum((s - mean_score)**2 for s in scores)
    
    for i in input_indices:
        vals = [row[i] for row in rows]
        mean_val = sum(vals) / len(vals)
        var_val = sum((v - mean_val)**2 for v in vals)
        
        if var_val == 0 or var_score == 0:
            correlations[header[i]] = 0
            continue
            
        cov = sum((vals[j] - mean_val) * (scores[j] - mean_score) for j in range(len(rows)))
        correlations[header[i]] = abs(cov / ((var_val * var_score) ** 0.5))
        
    # Sort by correlation descending
    sorted_inputs = sorted(correlations.items(), key=lambda x: x[1], reverse=True)
    
    # Return the top 2 as the ones that "matter most"
    return [name for name, corr in sorted_inputs[:2]]
