```python
import csv
import pandas as pd
import numpy as np
from scipy import stats

def certify(xs, ys):
    """
    Returns True if the two distributions are statistically similar (null hypothesis 
    not rejected), False if they are significantly different.
    Uses Welch's t-test for independent samples with unequal variances.
    """
    # Use a significance level of 0.05
    t_stat, p_value = stats.ttest_ind(xs, ys, equal_var=False)
    return p_value > 0.05

def analyze_team_size_impact(header, rows):
    """
    Splits data into small-team and big-team setups and compares their Cost- goals.
    """
    df = pd.DataFrame(rows, columns=header)
    
    # Define team size threshold (median)
    median_team_size = df['Team Size'].median()
    
    small_teams = df[df['Team Size'] <= median_team_size]['Cost-']
    big_teams = df[df['Team Size'] > median_team_size]['Cost-']
    
    is_same = certify(small_teams.tolist(), big_teams.tolist())
    
    return is_same, small_teams.mean(), big_teams.mean()

if __name__ == "__main__":
    filename = 'pom3a.n1000.train.csv'
    try:
        with open(filename, 'r') as f:
            reader = csv.reader(f)
            header = next(reader)
            rows = []
            for row in reader:
                processed_row = [float(val) if val.replace('.', '', 1).isdigit() or 
                                 (val.startswith('-') and val[1:].replace('.', '', 1).isdigit()) 
                                 else val for val in row]
                rows.append(processed_row)
        
        is_same, mean_small, mean_big = analyze_team_size_impact(header, rows)
        
        print("Customer Inquiry: Is the difference between small and big team costs real?")
        print(f"Small team average cost: {mean_small:.4f}")
        print(f"Big team average cost:   {mean_big:.4f}")
        
        if is_same:
            print("Answer: The difference is likely due to luck (not statistically significant).")
        else:
            print("Answer: The difference is statistically significant; small teams are genuinely cheaper.")
            
    except FileNotFoundError:
        print(f"Error: {filename} not found.")
        exit(1)
```
