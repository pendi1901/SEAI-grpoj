import csv
import pandas as pd
import numpy as np

def remember(header, rows):
    """
    Identifies rows that perform well across all goal columns.
    Goals ending in '-' are minimized, goals ending in '+' are maximized.
    We normalize the goals and find the Pareto front or simply 
    the rows with the lowest aggregate normalized cost.
    """
    df = pd.DataFrame(rows, columns=header)
    
    # Identify goal columns
    goals = [c for c in header if c.endswith('-') or c.endswith('+')]
    
    # Normalize goals to [0, 1] range
    normalized_goals = pd.DataFrame(index=df.index)
    for goal in goals:
        col = df[goal].astype(float)
        if goal.endswith('-'):
            # Minimize: lower is better (0 is best)
            normalized_goals[goal] = (col - col.min()) / (col.max() - col.min() + 1e-9)
        else:
            # Maximize: higher is better (0 is best after inversion)
            normalized_goals[goal] = (col.max() - col) / (col.max() - col.min() + 1e-9)
            
    # Calculate aggregate score (sum of normalized goals)
    # Lower aggregate score is better
    df['score'] = normalized_goals.sum(axis=1)
    
    # Select the top 5% (or at least one)
    n_rows = len(df)
    k = max(1, n_rows // 20)
    
    best_indices = df.nsmallest(k, 'score').index.tolist()
    return best_indices

def certify(xs, ys):
    """
    Returns True if the two distributions are statistically similar (null hypothesis 
    not rejected), False if they are significantly different.
    Uses Welch's t-test for independent samples with unequal variances.
    """
    # Use a significance level of 0.05
    t_stat, p_value = stats.ttest_ind(xs, ys, equal_var=False)
    return p_value > 0.05

def analyze_team_size_impact(header, rows):
    """
    Splits data into small-team and big-team setups and compares their Cost- goals.
    """
    df = pd.DataFrame(rows, columns=header)
    
    # Define team size threshold (median)
    median_team_size = df['Team Size'].median()

    big_teams = df[df['Team Size'] > median_team_size]['Cost-']
    
    is_same = certify(small_teams.tolist(), big_teams.tolist())

def flow(header, row, seen):
    """
    Decides whether to accept a row based on the Secretary Problem logic 
    combined with a Pareto-based quality threshold.
    """
    # Identify goal columns
    goals = [c for c in header if c.endswith('-') or c.endswith('+')]
    
    # If it's the first row, we don't have enough info to compare
    if not seen:
        return False
    
    # Calculate a simple score for the current row
    # Lower is better: normalize goals to [0, 1] relative to seen data
    df_seen = pd.DataFrame(seen, columns=header)
    
    def get_score(r):
        score = 0
        for g in goals:
            col = df_seen[g].astype(float)
            val = float(r[header.index(g)])
            norm = (val - col.min()) / (col.max() - col.min() + 1e-9)
            if g.endswith('+'):
                norm = 1 - norm
            score += norm
        return score

    # Strategy: Accept if the current row is better than the best seen so far
    # and we have passed the initial exploration phase (e.g., 20% of data)
    current_score = get_score(row)
    best_seen_score = min([get_score(s) for s in seen])
    
    # Exploration phase: 20% of the expected 800 rows (160 rows)
    if len(seen) < 160:
        return False
    
    # If current row is better than anything seen before, accept it
    return current_score < best_seen_score

def justify(header, rows, row):
    """
    Estimates the goal values for the provided row using KNN, then calculates
    the percentile rank of this row against the existing dataset.
    """
    df = pd.DataFrame(rows, columns=header)
    goals = [c for c in header if c.endswith('-') or c.endswith('+')]
    inputs = [c for c in header if c not in goals]
    
    # 1. Estimate goal values for the input row
    X_train = df[inputs]
    y_train = df[goals]
    knn = KNeighborsRegressor(n_neighbors=5)
    knn.fit(X_train, y_train)
    
    # Predict goals for the row (which has '?' in goal columns)
    row_inputs = [row[header.index(i)] for i in inputs]
    predicted_goals = knn.predict([row_inputs])[0]
    
    # 2. Calculate aggregate score for all rows
    # Normalize: lower is better for all (invert '+' goals)
    def get_scores(data_df):
        scores = pd.DataFrame(index=data_df.index)
        for i, goal in enumerate(goals):
            col = data_df[goal]
            norm = (col - col.min()) / (col.max() - col.min() + 1e-9)
            if goal.endswith('+'):
                norm = 1 - norm
            scores[goal] = norm
        return scores.sum(axis=1)

    all_scores = get_scores(df)
    
    # Calculate score for the target row
    row_norm_scores = []
    for i, goal in enumerate(goals):
        col = df[goal]
        val = predicted_goals[i]
        norm = (val - col.min()) / (col.max() - col.min() + 1e-9)
        if goal.endswith('+'):
            norm = 1 - norm
        row_norm_scores.append(norm)
    row_score = sum(row_norm_scores)
    
    # 3. Calculate percentile (fraction of rows that are worse than this one)
    # A row is better if its score is lower
    better_than_count = (all_scores > row_score).sum()
    p = better_than_count / len(df)
    
    reason = f"This configuration is estimated to outperform {p:.1%} of all possible setups based on its predicted performance metrics."
    
    return p, reason

import csv
import pandas as pd
import numpy as np
from sklearn.neighbors import KNeighborsRegressor

def choose(header, rows, label):
    """
    Selects the best row by using an active learning approach:
    1. Sample a small initial set to train a surrogate model.
    2. Use the model to predict performance on remaining rows.
    3. Query the most promising rows (Expected Improvement or best predicted).
    4. Return the index of the best row found after 50 labels.
    """
    goals = [c for c in header if c.endswith('-') or c.endswith('+')]
    inputs = [c for c in header if c not in goals]
    
    # Convert rows to DataFrame for easier handling
    df = pd.DataFrame(rows, columns=header)
    X = df[inputs].values
    
    # Keep track of labeled data
    labeled_indices = []
    
    def get_score(goal_values):
        # Normalize goals: lower is better for all
        # Since we don't have the full dataset distribution here, 
        # we use the values returned by label() to compute a relative score.
        # However, for simplicity, we treat the goal vector as a point to minimize.
        score = 0
        for i, g in enumerate(goals):
            val = goal_values[i]
            if g.endswith('+'):
                score -= val  # Maximize
            else:
                score += val  # Minimize
        return score

    # Initial sampling: 10 random rows to build a baseline
    np.random.seed(42)
    indices_to_sample = np.random.choice(len(rows), 10, replace=False).tolist()
    
    best_idx = -1
    best_score = float('inf')
    
    # Main loop: 50 labels allowed
    for _ in range(50):
        # Label the selected indices
        for idx in indices_to_sample:
            if idx not in labeled_indices:
                goal_vals = label(idx)
                score = get_score(goal_vals)
                if score < best_score:
                    best_score = score
                    best_idx = idx
                labeled_indices.append(idx)
        
        if len(labeled_indices) >= 50:
            break
            
        # Train model on labeled data
        X_train = [rows[i][:len(inputs)] for i in labeled_indices]
        y_train = [label(i) for i in labeled_indices]
        
        knn = KNeighborsRegressor(n_neighbors=3)
        knn.fit(X_train, y_train)
        
        # Predict on all rows
        preds = knn.predict(X)
        scores = [get_score(p) for p in preds]
        
        # Pick the next best candidate that hasn't been labeled
        remaining = [i for i in range(len(rows)) if i not in labeled_indices]
        if not remaining:
            break
        
        # Strategy: pick the one with the best predicted score
        next_idx = remaining[np.argmin([scores[i] for i in remaining])]
        indices_to_sample = [next_idx]
        
    return best_idx

def spot(header, rows):
    """
    Identifies rows that are outliers in the dataset using Isolation Forest.
    Returns the indices of the most suspicious rows, sorted by anomaly score.
    """
    df = pd.DataFrame(rows, columns=header)
    
    # Identify numeric columns (starting with uppercase)
    numeric_cols = [c for c in header if c[0].isupper()]
    data = df[numeric_cols].values
    
    # Isolation Forest is effective for detecting anomalies.
    # We must fit the model before calling decision_function.
    clf = IsolationForest(n_estimators=100, contamination=0.05, random_state=42)
    clf.fit(data)
    
    # decision_function returns the anomaly score (lower is more anomalous)
    scores = clf.decision_function(data)
    
    # Get indices sorted by score (ascending, so most anomalous first)
    suspicious_indices = np.argsort(scores)
    
    return suspicious_indices.tolist()

    """
    goals = [c for c in header if c.endswith('-') or c.endswith('+')]
    inputs = [c for c in header if c not in goals]
    
    # Helper to score a row
    def get_score(goal_vals):
        score = 0
        for i, g in enumerate(goals):
            val = goal_vals[i]
            if g.endswith('+'):
                score -= val
            else:
                score += val
        return score

    # Track labels
    labeled = {}
    def get_label(idx):
        if idx not in labeled:
            if len(labeled) >= 50:
                raise ValueError("Budget exceeded")
            labeled[idx] = label(idx)
        return labeled[idx]

    # Strategies
    def strategy_random(indices):
        return np.random.choice(indices)

    def strategy_greedy(indices, labeled_indices):
        X_train = [rows[i][:len(inputs)] for i in labeled_indices]
        y_train = [get_label(i) for i in labeled_indices]
        knn = KNeighborsRegressor(n_neighbors=3).fit(X_train, y_train)
        preds = knn.predict([rows[i][:len(inputs)] for i in indices])
        scores = [get_score(p) for p in preds]
        return indices[np.argmin(scores)]

    def strategy_uncertain(indices, labeled_indices):
        # Pick the one furthest from current labeled data (exploration)
        X_train = np.array([rows[i][:len(inputs)] for i in labeled_indices])
        X_pool = np.array([rows[i][:len(inputs)] for i in indices])
        dists = np.min(np.linalg.norm(X_pool[:, np.newaxis] - X_train, axis=2), axis=1)
        return indices[np.argmax(dists)]

    # Run race
    strategies = {"random": strategy_random, "greedy": strategy_greedy, "uncertain": strategy_uncertain}
    results = {name: {"best_idx": -1, "best_score": float('inf')} for name in strategies}
    
    all_indices = list(range(len(rows)))
    # Initial 3 labels (1 per strategy)
    for name in strategies:
        idx = np.random.choice(all_indices)
        results[name]["best_idx"] = idx
        results[name]["best_score"] = get_score(get_label(idx))

    # Remaining 47 labels
    for _ in range(47):
        for name in strategies:
            if len(labeled) >= 50: break
            pool = [i for i in all_indices if i not in labeled]
            if not pool: break
            
            if name == "random":
                idx = strategies[name](pool)
            else:
                idx = strategies[name](pool, list(labeled.keys()))
            
            score = get_score(get_label(idx))
            if score < results[name]["best_score"]:
                results[name]["best_score"] = score
                results[name]["best_idx"] = idx

    winner = min(results, key=lambda n: results[n]["best_score"])
    return {"winner": winner, "row": results[winner]["best_idx"], "tried": list(strategies.keys())}

def remember(header, rows):

    Identifies the 5% of rows that are most similar to the 'best' performing rows
    in the dataset. We define 'best' as those with the lowest aggregate normalized
    cost across all goal columns.
    """
    df = pd.DataFrame(rows, columns=header)
    
    # Identify goal columns
    goals = [c for c in header if c.endswith('-') or c.endswith('+')]
    inputs = [c for c in header if c not in goals]
    
    # 1. Calculate aggregate score to find the 'best' rows

    normalized_goals = pd.DataFrame(index=df.index)
    for goal in goals:
        col = df[goal].astype(float)
        if goal.endswith('-'):
            normalized_goals[goal] = (col - col.min()) / (col.max() - col.min() + 1e-9)
        else:
            normalized_goals[goal] = (col.max() - col) / (col.max() - col.min() + 1e-9)
            
    df['score'] = normalized_goals.sum(axis=1)
    
    # 2. Find the top 1% of rows (the "gold standard" examples)
    n_rows = len(df)
    k_best = max(1, n_rows // 100)
    best_indices = df.nsmallest(k_best, 'score').index.tolist()

    # We use numeric inputs for similarity
    X = df[inputs].select_dtypes(include=[np.number])
    nn = NearestNeighbors(n_neighbors=max(1, n_rows // 20))
    nn.fit(X)
    
    # Find neighbors of the best rows
    distances, indices = nn.kneighbors(X.iloc[best_indices])
    
    # Flatten and get unique indices, limited to 5% of total rows
    result_indices = list(set(indices.flatten()))
    limit = max(1, n_rows // 20)
    
    return sorted(result_indices)[:limit]

def certify(xs, ys):
    """
    Returns True if the two distributions are statistically similar (null hypothesis 
    not rejected), False if they are significantly different.
    Uses Welch's t-test for independent samples with unequal variances.
    """
    # Use a significance level of 0.05
    # If p > 0.05, we fail to reject the null hypothesis (they are the same)
    t_stat, p_value = stats.ttest_ind(xs, ys, equal_var=False)
    return p_value > 0.05

def analyze_shift(header, rows):
    """
    Splits the data into two halves (before and after the change in spring)
    and compares the performance (aggregate goal score) to see if results shifted.
    """
    df = pd.DataFrame(rows, columns=header)
    
    # Identify goal columns
    goals = [c for c in header if c.endswith('-') or c.endswith('+')]
    
    # Normalize goals to [0, 1] range to create a single performance metric
    normalized_goals = pd.DataFrame(index=df.index)
    for goal in goals:
        col = df[goal].astype(float)
        if goal.endswith('-'):
            normalized_goals[goal] = (col - col.min()) / (col.max() - col.min() + 1e-9)
        else:
            normalized_goals[goal] = (col.max() - col) / (col.max() - col.min() + 1e-9)
            
    df['score'] = normalized_goals.sum(axis=1)
    
    # Split data into two halves (assuming chronological order)
    mid = len(df) // 2
    before = df.iloc[:mid]['score'].tolist()
    after = df.iloc[mid:]['score'].tolist()
    
    is_same = certify(before, after)
    return is_same, np.mean(before), np.mean(after)

import csv

def flow(header, row, seen):
    """
    Warns the customer if a row is 'going bad'.
    A row is 'going bad' if its performance is significantly worse than the 
    average performance of the rows seen so far.
    """
    if not seen:
        return False

    goals = [c for c in header if c.endswith('-') or c.endswith('+')]
    
    def get_score(r):
        score = 0
        for g in goals:
            idx = header.index(g)
            val = float(r[idx])
            # Normalize relative to the range of seen data
            seen_vals = [float(s[idx]) for s in seen]
            min_v, max_v = min(seen_vals), max(seen_vals)
            norm = (val - min_v) / (max_v - min_v + 1e-9)
            if g.endswith('+'):
                norm = 1 - norm
            score += norm
        return score

    # Calculate current row score and average of seen scores
    current_score = get_score(row)
    seen_scores = [get_score(s) for s in seen]
    avg_seen_score = np.mean(seen_scores)
    std_seen_score = np.std(seen_scores) + 1e-9

    # If the current row is more than 2 standard deviations worse than the average,
    # we flag it as "going bad".
    if current_score > (avg_seen_score + 2 * std_seen_score):
        return True
    
    return False

    X_train = df[inputs]
    y_train = df[goals]

    row_inputs = [row[header.index(i)] for i in inputs]
    predicted_goals = knn.predict([row_inputs])[0]
    
    # 2. Calculate aggregate score for all rows
    # Normalize: lower is better for all (invert '+' goals)
    def get_scores(data_df):
        scores = pd.DataFrame(index=data_df.index)
        for goal in goals:
            col = data_df[goal]
            norm = (col - col.min()) / (col.max() - col.min() + 1e-9)
            if goal.endswith('+'):
                norm = 1 - norm
            scores[goal] = norm
        return scores.sum(axis=1)

    all_scores = get_scores(df)
    
    # 3. Calculate score for the target row
    row_norm_scores = []
    for i, goal in enumerate(goals):
        col = df[goal]
        val = predicted_goals[i]
        norm = (val - col.min()) / (col.max() - col.min() + 1e-9)
        if goal.endswith('+'):

        row_norm_scores.append(norm)

    # 4. Calculate percentile (fraction of rows that are worse than this one)

    better_than_count = (all_scores > row_score).sum()
    p = better_than_count / len(df)
    
    reason = f"This configuration is estimated to outperform {p:.2f} of all possible setups based on its predicted performance metrics."

import csv
import numpy as np
from sklearn.neighbors import KNeighborsRegressor

def choose(header, rows, label):
    """
    Finds the best setup for a large, high-criticality project by sampling 
    the search space using an active learning strategy with a budget of 50 labels.
    """
    goals = [c for c in header if c.endswith('-') or c.endswith('+')]
    inputs = [c for c in header if c not in goals]
    
    # Identify indices of rows that match "large" and "high criticality"
    # Large: Size > 0.5, Criticality > 0.5
    size_idx = header.index('Size')
    crit_idx = header.index('Criticality')
    candidates = [i for i, r in enumerate(rows) if r[size_idx] > 0.5 and r[crit_idx] > 0.5]
    
    # If no candidates match, fall back to all rows
    if not candidates:
        candidates = list(range(len(rows)))

    def get_score(goal_vals):
        # Normalize: lower is better for all (invert '+' goals)
        score = 0
        for i, g in enumerate(goals):
            val = goal_vals[i]
            if g.endswith('+'):
                score += (1.0 - val)
            else:
                score += val
        return score

    labeled = {}
    def get_label(idx):
        if idx not in labeled:
            labeled[idx] = label(idx)
        return labeled[idx]

    # Initial sampling: 5 random candidates to start
    np.random.seed(42)
    sample_size = min(5, len(candidates))
    to_label = np.random.choice(candidates, sample_size, replace=False).tolist()
    
    best_idx = -1
    best_score = float('inf')
    
    # Active learning loop
    for _ in range(50):
        if not to_label:
            break
            
        idx = to_label.pop(0)
        score = get_score(get_label(idx))
        if score < best_score:
            best_score = score
            best_idx = idx
            
        if len(labeled) >= 50:
            break
            
        # Train model to predict scores
        X_train = [rows[i][:len(inputs)] for i in labeled]
        y_train = [get_score(labeled[i]) for i in labeled]
        
        # Adjust n_neighbors based on available data
        n_neighbors = min(3, len(labeled))
        knn = KNeighborsRegressor(n_neighbors=n_neighbors).fit(X_train, y_train)
        
        # Predict on remaining candidates
        pool = [i for i in candidates if i not in labeled]
        if not pool:
            break
            
        preds = knn.predict([rows[i][:len(inputs)] for i in pool])
        # Pick the candidate with the lowest predicted score
        next_idx = pool[np.argmin(preds)]
        to_label.append(next_idx)
        
    return best_idx

def fix(header, row, levers):
    """
    Identifies the best change to non-hiring levers to minimize 'Idle-'.
    We use a Random Forest to predict 'Idle-' and perform a local search
    over the allowed lever ranges.
    """
    # Load data to train a surrogate model
    filename = 'pom3a.n1000.train.csv'
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header_data = next(reader)
        rows = []
        for r in reader:
            rows.append([float(val) if val.replace('.', '', 1).isdigit() or 
                         (val.startswith('-') and val[1:].replace('.', '', 1).isdigit()) 
                         else val for val in r])
    
    df = pd.DataFrame(rows, columns=header_data)
    goals = [c for c in header_data if c.endswith('-') or c.endswith('+')]
    inputs = [c for c in header_data if c not in goals]
    
    # Train model to predict Idle-
    X = df[inputs]
    y = df['Idle-']
    model = RandomForestRegressor(n_estimators=50, random_state=42)
    model.fit(X, y)
    
    best_change = {}
    best_idle = float('inf')
    
    # Search for the best lever adjustment
    for col, allowed in levers.items():
        if isinstance(allowed, tuple):
            low, high = allowed
            # Test 10 points in the range
            test_values = np.linspace(low, high, 10)
        else:
            test_values = allowed
            
        for val in test_values:
            temp_row = list(row)
            temp_row[header.index(col)] = val
            
            # Predict Idle- for this change
            pred_idle = model.predict([temp_row[:len(inputs)]])[0]
            
            if pred_idle < best_idle:
