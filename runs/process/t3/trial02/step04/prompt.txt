import csv
import pandas as pd
import numpy as np

def remember(header, rows):
    """
    Identifies rows that perform well across all goal columns.
    Goals ending in '-' are minimized, goals ending in '+' are maximized.
    We normalize the goals and find the Pareto front or simply 
    the rows with the lowest aggregate normalized cost.
    """
    df = pd.DataFrame(rows, columns=header)
    
    # Identify goal columns
    goals = [c for c in header if c.endswith('-') or c.endswith('+')]
    
    # Normalize goals to [0, 1] range
    normalized_goals = pd.DataFrame(index=df.index)
    for goal in goals:
        col = df[goal].astype(float)
        if goal.endswith('-'):
            # Minimize: lower is better (0 is best)
            normalized_goals[goal] = (col - col.min()) / (col.max() - col.min() + 1e-9)
        else:
            # Maximize: higher is better (0 is best after inversion)
            normalized_goals[goal] = (col.max() - col) / (col.max() - col.min() + 1e-9)
            
    # Calculate aggregate score (sum of normalized goals)
    # Lower aggregate score is better
    df['score'] = normalized_goals.sum(axis=1)
    
    # Select the top 5% (or at least one)
    n_rows = len(df)
    k = max(1, n_rows // 20)
    
    best_indices = df.nsmallest(k, 'score').index.tolist()
    return best_indices

import csv
import pandas as pd
import numpy as np
from scipy import stats

def certify(xs, ys):
    """
    Returns True if the two distributions are statistically similar (null hypothesis 
    not rejected), False if they are significantly different.
    Uses Welch's t-test for independent samples with unequal variances.
    """
    # Use a significance level of 0.05
    t_stat, p_value = stats.ttest_ind(xs, ys, equal_var=False)
    return p_value > 0.05

def analyze_team_size_impact(header, rows):
    """
    Splits data into small-team and big-team setups and compares their Cost- goals.
    """
    df = pd.DataFrame(rows, columns=header)
    
    # Define team size threshold (median)
    median_team_size = df['Team Size'].median()
    
    small_teams = df[df['Team Size'] <= median_team_size]['Cost-']
    big_teams = df[df['Team Size'] > median_team_size]['Cost-']
    
    is_same = certify(small_teams.tolist(), big_teams.tolist())
    
    return is_same, small_teams.mean(), big_teams.mean()
