import csv
import pandas as pd
import numpy as np

def remember(header, rows):
    """
    Identifies rows that perform well across all goal columns.
    Goals ending in '-' are minimized, goals ending in '+' are maximized.
    """
    df = pd.DataFrame(rows, columns=header)
    
    # Identify goal columns
    goals = [c for c in header if c.endswith('-') or c.endswith('+')]
    
    # Normalize goals to a [0, 1] scale where 1 is always "best"
    normalized_goals = pd.DataFrame(index=df.index)
    for goal in goals:
        col = df[goal].astype(float)
        if goal.endswith('-'):
            # Lower is better: (max - x) / (max - min)
            normalized_goals[goal] = (col.max() - col) / (col.max() - col.min())
        else:
            # Higher is better: (x - min) / (max - min)
            normalized_goals[goal] = (col - col.min()) / (col.max() - col.min())
            
    # Calculate a composite score (average of normalized goals)
    # We want rows that are consistently good across all goals
    df['composite'] = normalized_goals.mean(axis=1)
    
    # Select top 5% (or at least 1)
    n_rows = len(df)
    k = max(1, int(n_rows * 0.05))
    
    best_indices = df['composite'].nlargest(k).index.tolist()
    return sorted(best_indices)

import csv
import pandas as pd
from scipy import stats

def certify(xs, ys):
    """
    Returns True if the difference between two distributions is statistically 
    insignificant (likely noise), False if they are significantly different.
    Uses Welch's t-test.
    """
    if len(xs) < 2 or len(ys) < 2:
        return True
    
    t_stat, p_value = stats.ttest_ind(xs, ys, equal_var=False)
    
    # If p-value > 0.05, we fail to reject the null hypothesis (they are the same)
    return p_value > 0.05

def analyze_team_size_impact(header, rows):
    """
    Splits data into small and big teams and compares their 'Cost-' values.
    """
    df = pd.DataFrame(rows, columns=header)
    
    # Define small vs big based on the median of 'Team Size'
    median_team_size = df['Team Size'].median()
    
    small_teams = df[df['Team Size'] <= median_team_size]['Cost-']
    big_teams = df[df['Team Size'] > median_team_size]['Cost-']
    
    is_same = certify(small_teams.tolist(), big_teams.tolist())
    
    return is_same, small_teams.mean(), big_teams.mean()

import csv

def flow(header, row, seen):
    """
    Implements a Secretary Problem strategy:
    1. Observe a sample (the first 30% of the data).
    2. Establish the best performance seen in that sample.
    3. Accept the first subsequent row that is better than that benchmark.
    """
    # Identify goal columns and their optimization direction
    goal_indices = [i for i, h in enumerate(header) if h.endswith('-') or h.endswith('+')]
    
    def get_score(r):
        # Calculate a simple normalized score for the row
        # Lower is better for '-', Higher is better for '+'
        score = 0
        for i in goal_indices:
            val = r[i]
            if header[i].endswith('-'):
                score -= val
            else:
                score += val
        return score

    # If this is the first row, we can't compare
    if not seen:
        return False

    # Define sample size (30% of total expected data, but we don't know total)
    # We use a fixed threshold based on the problem context (800 rows)
    sample_size = 240 
    
    if len(seen) < sample_size:
        return False
    
    # Calculate the best score seen during the sampling phase
    sample_scores = [get_score(s) for s in seen[:sample_size]]
    best_sample_score = max(sample_scores)
    
    # Accept if current row is better than the best in the sample
    current_score = get_score(row)
    return current_score > best_sample_score

def blame(header, rows):
    """
    Identifies input columns that best explain high 'Idle-' values.
    We define 'worst' as rows with the highest 'Idle-' values.
    We use a Random Forest to find feature importance for 'Idle-'.
    """
    df = pd.DataFrame(rows, columns=header)
    
    # Identify input columns (not goals)
    input_cols = [c for c in header if not (c.endswith('-') or c.endswith('+'))]
    
    # Target is 'Idle-'
    target = 'Idle-'
    
    # Train a regressor to see which inputs correlate with high Idle-
    model = RandomForestRegressor(n_estimators=50, random_state=42)
    model.fit(df[input_cols], df[target])
    
    # Get feature importances
    importances = model.feature_importances_
    
    # Pair features with their importance scores
    feat_imp = sorted(zip(input_cols, importances), key=lambda x: x[1], reverse=True)
    
    # Return the names of the top 3 most influential input columns
    return [name for name, score in feat_imp[:3]]
