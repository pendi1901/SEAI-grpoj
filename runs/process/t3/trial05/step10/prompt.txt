import csv
import pandas as pd
import numpy as np

def remember(header, rows):
    """
    Identifies rows that perform well across all goal columns.
    Goals ending in '-' are minimized, goals ending in '+' are maximized.
    """
    df = pd.DataFrame(rows, columns=header)
    
    # Identify goal columns
    goals = [c for c in header if c.endswith('-') or c.endswith('+')]
    
    # Normalize goals to a [0, 1] scale where 1 is always "best"
    normalized_goals = pd.DataFrame(index=df.index)
    for goal in goals:
        col = df[goal].astype(float)
        if goal.endswith('-'):
            # Lower is better: (max - x) / (max - min)
            normalized_goals[goal] = (col.max() - col) / (col.max() - col.min())
        else:
            # Higher is better: (x - min) / (max - min)
            normalized_goals[goal] = (col - col.min()) / (col.max() - col.min())
            
    # Calculate a composite score (average of normalized goals)
    # We want rows that are consistently good across all goals
    df['composite'] = normalized_goals.mean(axis=1)
    
    # Select top 5% (or at least 1)
    n_rows = len(df)
    k = max(1, int(n_rows * 0.05))
    
    best_indices = df['composite'].nlargest(k).index.tolist()
    return sorted(best_indices)

import csv

from scipy import stats

def certify(xs, ys):
    """
    Returns True if the difference between two distributions is statistically 
    insignificant (likely noise), False if they are significantly different.
    Uses Welch's t-test.
    """
    if len(xs) < 2 or len(ys) < 2:
        return True
    
    t_stat, p_value = stats.ttest_ind(xs, ys, equal_var=False)
    
    # If p-value > 0.05, we fail to reject the null hypothesis (they are the same)
    return p_value > 0.05

def analyze_team_size_impact(header, rows):

import csv

def flow(header, row, seen):
    """
    Implements a Secretary Problem strategy:
    1. Observe a sample (the first 30% of the data).
    2. Establish the best performance seen in that sample.
    3. Accept the first subsequent row that is better than that benchmark.
    """
    # Identify goal columns and their optimization direction
    goal_indices = [i for i, h in enumerate(header) if h.endswith('-') or h.endswith('+')]
    
    def get_score(r):
        # Calculate a simple normalized score for the row
        # Lower is better for '-', Higher is better for '+'
        score = 0
        for i in goal_indices:
            val = r[i]
            if header[i].endswith('-'):
                score -= val
            else:
                score += val
        return score

    # If this is the first row, we can't compare
    if not seen:
        return False

    # Define sample size (30% of total expected data, but we don't know total)
    # We use a fixed threshold based on the problem context (800 rows)
    sample_size = 240 
    
    if len(seen) < sample_size:
        return False
    
    # Calculate the best score seen during the sampling phase
    sample_scores = [get_score(s) for s in seen[:sample_size]]
    best_sample_score = max(sample_scores)
    
    # Accept if current row is better than the best in the sample
    current_score = get_score(row)
    return current_score > best_sample_score

def justify(header, rows, row):
    """
    Estimates the performance of a given row and returns its percentile rank
    relative to the existing dataset.
    """
    df = pd.DataFrame(rows, columns=header)
    
    # Identify input and goal columns
    input_cols = [c for c in header if not (c.endswith('-') or c.endswith('+'))]
    goal_cols = [c for c in header if c.endswith('-') or c.endswith('+')]
    
    # Train models to estimate the missing goal values for the input row
    estimates = {}
    for goal in goal_cols:
        model = KNeighborsRegressor(n_neighbors=5)
        model.fit(df[input_cols], df[goal])
        estimates[goal] = model.predict([row[:len(input_cols)]])[0]
    
    # Calculate a composite score for the dataset
    # Normalize goals: 1 is best, 0 is worst
    scores = pd.DataFrame(index=df.index)
    for goal in goal_cols:
        col = df[goal]
        if goal.endswith('-'):
            scores[goal] = (col.max() - col) / (col.max() - col.min())
        else:
            scores[goal] = (col - col.min()) / (col.max() - col.min())
    
    df['composite'] = scores.mean(axis=1)
    
    # Calculate the composite score for the input row
    row_scores = []
    for goal in goal_cols:
        col = df[goal]
        val = estimates[goal]
        if goal.endswith('-'):
            row_scores.append((col.max() - val) / (col.max() - col.min()))
        else:
            row_scores.append((val - col.min()) / (col.max() - col.min()))
    
    row_composite = sum(row_scores) / len(row_scores)
    
    # Calculate percentile (fraction of rows that are worse than this one)
    p = (df['composite'] < row_composite).mean()

def choose(header, rows, label):
    """
    Finds the best row by sampling and using a surrogate model to guide selection.
    """
    goal_indices = [i for i, h in enumerate(header) if h.endswith('-') or h.endswith('+')]
    input_indices = [i for i, h in enumerate(header) if not (h.endswith('-') or h.endswith('+'))]
    
    # 1. Initial exploration: Label 20 random rows to build a training set
    labeled_indices = random.sample(range(len(rows)), 20)
    train_X = []
    train_y = []
    
    for i in labeled_indices:
        goals = label(i)
        train_X.append([rows[i][j] for j in input_indices])
        # Normalize goals: minimize '-' (negate), maximize '+' (keep)
        score = 0
        for idx, g_val in enumerate(goals):
            col_name = header[goal_indices[idx]]
            if col_name.endswith('-'):
                score -= g_val
            else:
                score += g_val
        train_y.append(score)
        
    # 2. Iterative improvement: Use remaining 30 labels to find the best
    for _ in range(30):
        model = RandomForestRegressor(n_estimators=20, random_state=42)
        model.fit(train_X, train_y)
        
        # Predict scores for all unlabeled rows
        best_idx = -1
        best_pred = -float('inf')
        
        for i in range(len(rows)):
            if i not in labeled_indices:
                pred = model.predict([[rows[i][j] for j in input_indices]])[0]
                if pred > best_pred:
                    best_pred = pred
                    best_idx = i
        
        # Label the best predicted row
        labeled_indices.append(best_idx)
        goals = label(best_idx)
        train_X.append([rows[best_idx][j] for j in input_indices])
        score = 0
        for idx, g_val in enumerate(goals):
            col_name = header[goal_indices[idx]]
            if col_name.endswith('-'):
                score -= g_val
            else:
                score += g_val
        train_y.append(score)
        
    # Return the index of the row with the highest observed score
    max_score = -float('inf')
    best_row_idx = -1
    for i, score in enumerate(train_y):
        if score > max_score:
            max_score = score
            best_row_idx = labeled_indices[i]
            
    return best_row_idx

    """
    Recommends changes to a 'bad' row to improve its performance.
    Uses a Random Forest model trained on the dataset to predict the impact
    of changing levers on the composite goal score.
    """

        else:
            scores[goal] = (col - col.min()) / (col.max() - col.min())

    # Train model
    model = RandomForestRegressor(n_estimators=50, random_state=42)
    model.fit(df[input_cols], df['composite'])
    
    # Find the best lever change
    best_change = {}
    best_score = -1
    
    # Try changing each lever to its bounds or midpoint

            # Predict score
            pred = model.predict([test_row[:len(input_cols)]])[0]
            if pred > best_score:
                best_score = pred
                best_change = {col: val}
                
    return best_change

def spot(header, rows):
    """
    Identifies rows that are outliers in the input space using Isolation Forest.
    Returns the indices of the rows, sorted by their anomaly score (most suspicious first).
    """
    df = pd.DataFrame(rows, columns=header)
    
    # Identify input columns (those not ending in - or +)
    input_cols = [c for c in header if not (c.endswith('-') or c.endswith('+'))]
    
    # Use Isolation Forest to detect anomalies
    # contamination='auto' lets the model decide the threshold
    model = IsolationForest(n_estimators=100, contamination=0.05, random_state=42)
    
    # Fit the model on input columns
    # decision_function returns the anomaly score (lower is more abnormal)
    scores = model.fit(df[input_cols]).decision_function(df[input_cols])
    
    # Get indices of rows sorted by anomaly score (ascending, so lowest scores first)
    # The lower the score, the more "suspicious" the row is.
    suspicious_indices = np.argsort(scores)
    
    # Filter to only include rows that were actually flagged as anomalies (score < 0)
    # and return them as a list of ints
    flagged = [int(i) for i in suspicious_indices if scores[i] < 0]
    
    return flagged
