import csv
import pandas as pd
import numpy as np

def remember(header, rows):
    """
    Identifies rows that perform well across all goal columns.
    Goals ending in '-' are minimized, '+' are maximized.
    """
    df = pd.DataFrame(rows, columns=header)
    
    # Identify goal columns
    goals = [col for col in header if col.endswith(('-', '+'))]
    
    # Normalize goals to [0, 1] range for comparison
    # For '-', lower is better (1 - normalized value)
    # For '+', higher is better (normalized value)
    scores = []
    for col in goals:
        col_min = df[col].min()
        col_max = df[col].max()
        norm = (df[col] - col_min) / (col_max - col_min + 1e-9)
        if col.endswith('-'):
            scores.append(1 - norm)
        else:
            scores.append(norm)
            
    # Aggregate score: product of normalized goals (geometric mean approach)
    # This ensures a row is only "unusually well" if it performs well on ALL goals
    combined_score = np.prod(scores, axis=0)
    
    # Select top 5% of rows
    num_to_select = max(1, len(rows) // 20)
    top_indices = np.argsort(combined_score)[-num_to_select:]
    
    return sorted(top_indices.tolist())

from scipy import stats

def certify(xs, ys):
    """
    Returns True if the difference between two distributions is statistically 
    insignificant (likely noise), False if they are significantly different.
    Uses Welch's t-test for independent samples.
    """
    # If samples are too small, we cannot reliably reject the null hypothesis
    if len(xs) < 2 or len(ys) < 2:
        return True
    
    # Perform Welch's t-test (does not assume equal variance)
    t_stat, p_value = stats.ttest_ind(xs, ys, equal_var=False)
    
    # If p-value > 0.05, we fail to reject the null hypothesis (they are the same)
    return p_value > 0.05

import csv
import numpy as np

def flow(header, row, seen):
    """
    Implements the Secretary Problem strategy:
    1. Observe a sample (the first 37% of the data).
    2. Establish the best performance seen in that sample.
    3. Accept the first subsequent row that is better than the sample best.
    """
    # Identify goal columns
    goal_indices = [i for i, col in enumerate(header) if col.endswith(('-', '+'))]
    
    def get_score(r):
        # Calculate a normalized score for a row
        # Since we don't have global min/max, we use the current 'seen' data
        # to estimate relative performance.
        scores = []
        for i in goal_indices:
            val = r[i]
            # Simple heuristic: if we don't have enough data, return 0
            # Otherwise, we treat lower '-' and higher '+' as better
            scores.append(val if header[i].endswith('+') else -val)
        return sum(scores)

    # If this is the first row, we can't decide yet
    if not seen:
        return False
    
    # Strategy: Use the first 37% of the total expected data (approx 300 rows)
    # as the 'rejection' phase to learn the baseline.
    sample_size = 300
    
    if len(seen) < sample_size:
        return False
    
    # Find the best score in the sample
    sample_scores = [get_score(r) for r in seen[:sample_size]]
    best_sample_score = max(sample_scores)
    
    # If current row is better than the best in the sample, accept it
    current_score = get_score(row)
    if current_score > best_sample_score:
        return True
    
    # If we reach the end, we must accept the last one
    return False

    X_train = df[input_cols]

    # Create a DataFrame for the single row to predict
    row_df = pd.DataFrame([row], columns=header)
    predicted_goals = model.predict(row_df[input_cols])[0]
    
    # 2. Calculate scores for all rows (normalized 0 to 1)
    # For '-', lower is better; for '+', higher is better.

        scores = []
        for i, col in enumerate(goal_cols):
            col_min = df[col].min()
            col_max = df[col].max()
            norm = (data_frame[col] - col_min) / (col_max - col_min + 1e-9)
            scores.append(1 - norm if col.endswith('-') else norm)
        return np.mean(scores, axis=1)

    all_scores = calculate_scores(df)
    
    # 3. Calculate score for the target row
    target_df = pd.DataFrame([predicted_goals], columns=goal_cols)
    # We need to normalize the target row using the same min/max as the training set
    norm_target = []
    for i, col in enumerate(goal_cols):

    target_score = np.mean(norm_target)
    
    # 4. Calculate percentile (fraction of rows that are worse)
    p = np.mean(all_scores < target_score)

def choose(header, rows, label):
    """
    Finds the best row by sampling and labeling.
    Uses a simple strategy:
    1. Label 50 random rows.
    2. Return the index of the best one found.
    """
    # Identify goal indices
    goal_indices = [i for i, col in enumerate(header) if col.endswith(('-', '+'))]
    
    # We can label at most 50 rows
    num_samples = min(len(rows), 50)
    indices_to_sample = random.sample(range(len(rows)), num_samples)
    
    best_idx = -1
    best_score = -float('inf')
    
    for idx in indices_to_sample:
        # Get true goal values
        goals = label(idx)
        
        # Calculate score: 
        # For '-', minimize (so negate). For '+', maximize (keep as is).
        # We use a simple sum of normalized goals if possible, 
        # but since we don't have global min/max, we use raw values 
        # assuming they are already in a comparable range [0, 1].
        current_score = 0
        for i, val in enumerate(goals):
            col_name = header[goal_indices[i]]
            if col_name.endswith('-'):
                current_score -= val
            else:
                current_score += val
        
        if current_score > best_score:
            best_score = current_score
            best_idx = idx
            
    return best_idx

import csv
import math

def spot(header, rows):
    """
    Identifies rows that contain outliers or 'mistyped' data.
    Uses the Interquartile Range (IQR) method to detect anomalies in numeric columns.
    Returns the indices of the most suspicious rows, sorted by the number of 
    outlier values found.
    """
    num_rows = len(rows)
    num_cols = len(header)
    
    # Identify numeric columns
    numeric_indices = [i for i, val in enumerate(rows[0]) if isinstance(val, (int, float))]
    
    # Calculate IQR for each numeric column
    outlier_counts = [0] * num_rows
    
    for col_idx in numeric_indices:
        col_data = [rows[i][col_idx] for i in range(num_rows) if rows[i][col_idx] != "?"]
        if not col_data:
            continue
            
        sorted_col = sorted(col_data)
        q1 = sorted_col[int(0.25 * len(sorted_col))]
        q3 = sorted_col[int(0.75 * len(sorted_col))]
        iqr = q3 - q1
        
        # Define bounds for outliers (1.5 * IQR rule)
        lower_bound = q1 - 1.5 * iqr
        upper_bound = q3 + 1.5 * iqr
        
        # Mark rows that fall outside these bounds
        for i in range(num_rows):
            val = rows[i][col_idx]
            if val != "?" and (val < lower_bound or val > upper_bound):
                outlier_counts[i] += 1
                
    # Get indices of rows with at least one outlier, sorted by count descending
    suspicious = []
    for i in range(num_rows):
        if outlier_counts[i] > 0:
            suspicious.append((i, outlier_counts[i]))
            
    suspicious.sort(key=lambda x: x[1], reverse=True)
    
    return [item[0] for item in suspicious]

        best_score = -float('inf')
        for idx in indices:
            score = get_score(get_label(idx))
            if score > best_score:
                best_score, best_idx = score, idx

        "greedy": strategy_greedy(),
        "hill": strategy_hill()
    }

    winner = max(results, key=lambda k: results[k][1])

import csv
import pandas as pd
import numpy as np
from sklearn.neighbors import NearestNeighbors

def remember(header, rows):
    """
    Identifies the best rows by calculating a normalized score for each row
    and returning the indices of the top 5% of performers.
    """
    df = pd.DataFrame(rows, columns=header)
    
    # Identify goal columns
    goal_cols = [col for col in header if col.endswith(('-', '+'))]
    
    # Normalize goals to [0, 1] range
    # For '-', lower is better (1 - normalized value)
    # For '+', higher is better (normalized value)
    scores = []
    for col in goal_cols:
        col_min = df[col].min()
        col_max = df[col].max()
        # Avoid division by zero
        rng = col_max - col_min
        if rng == 0:
            norm = 0
        else:
            norm = (df[col] - col_min) / rng
            
        if col.endswith('-'):
            scores.append(1 - norm)
        else:
            scores.append(norm)
            
    # Aggregate score: mean of normalized goals
    combined_score = np.mean(scores, axis=0)
    
    # Select top 5% of rows
    num_to_select = max(1, len(rows) // 20)
    top_indices = np.argsort(combined_score)[-num_to_select:]
    
    return sorted(top_indices.tolist())

def find_similar(header, rows, target_row, n=3):
    """
    Finds the n most similar rows to the target_row using Euclidean distance
    on input columns.
    """
    input_cols = [col for col in header if not (col.endswith('-') or col.endswith('+'))]
    df = pd.DataFrame(rows, columns=header)
    
    # Extract input data
    X = df[input_cols].values
    target_vals = np.array([target_row[i] for i, col in enumerate(header) if col in input_cols]).reshape(1, -1)
    
    # Find nearest neighbors
    nn = NearestNeighbors(n_neighbors=n)
    nn.fit(X)
    distances, indices = nn.kneighbors(target_vals)
    
    return indices[0]

import csv
from scipy import stats

def certify(xs, ys):
    """
    Returns True if the difference between two distributions is statistically 
    insignificant (likely noise), False if they are significantly different.
    Uses Welch's t-test for independent samples.
    """
    if len(xs) < 2 or len(ys) < 2:
        return True
    
    # Welch's t-test (does not assume equal variance)
    t_stat, p_value = stats.ttest_ind(xs, ys, equal_var=False)
    
    # If p-value > 0.05, we fail to reject the null hypothesis (they are the same)
    return p_value > 0.05

def analyze_shift(header, rows):
    """
    Compares the first half of the data (pre-change) with the second half (post-change)
    across all goal columns to see if the results shifted significantly.
    """
    goal_indices = [i for i, col in enumerate(header) if col.endswith(('-', '+'))]
    
    # Split data into two halves
    mid = len(rows) // 2
    first_half = rows[:mid]
    second_half = rows[mid:]
    
    results = {}
    for idx in goal_indices:
        col_name = header[idx]
        xs = [row[idx] for row in first_half]
        ys = [row[idx] for row in second_half]
        
        is_same = certify(xs, ys)
        results[col_name] = is_same
        
    return results

import csv

def flow(header, row, seen):
    """
    Warns the customer if a new row is significantly worse than the 
    historical baseline. We define 'going bad' as a row whose 
    aggregate goal performance is significantly below the 
    moving average of the 'seen' data.
    """
    goal_indices = [i for i, col in enumerate(header) if col.endswith(('-', '+'))]
    
    def get_score(r):
        # Normalize goals: '-' minimized (negated), '+' maximized (as is)
        score = 0
        for i in goal_indices:
            val = r[i]
            if header[i].endswith('-'):
                score -= val
            else:
                score += val
        return score

    # Need a baseline to determine what 'bad' is
    if len(seen) < 20:
        return False
    
    # Calculate moving average of performance
    scores = [get_score(r) for r in seen]
    avg_score = sum(scores) / len(scores)
    std_dev = (sum((s - avg_score)**2 for s in scores) / len(scores))**0.5
    
    # If the new row is more than 2 standard deviations below the mean,
    # it is performing significantly worse than the established trend.
    current_score = get_score(row)
    if current_score < (avg_score - 2 * std_dev):
        return True
        
    return False
