```python
import csv
import pandas as pd
import numpy as np

def remember(header, rows):
    """
    Identifies rows that perform well across all goal columns.
    Goals ending in '-' are minimized, '+' are maximized.
    """
    df = pd.DataFrame(rows, columns=header)
    
    # Identify goal columns
    goals = [col for col in header if col.endswith(('-', '+'))]
    
    # Normalize goals to [0, 1] range for comparison
    # For '-', lower is better (1 - normalized value)
    # For '+', higher is better (normalized value)
    scores = []
    for col in goals:
        col_min = df[col].min()
        col_max = df[col].max()
        norm = (df[col] - col_min) / (col_max - col_min + 1e-9)
        if col.endswith('-'):
            scores.append(1 - norm)
        else:
            scores.append(norm)
            
    # Aggregate score: product of normalized goals (geometric mean approach)
    # This ensures a row is only "unusually well" if it performs well on ALL goals
    combined_score = np.prod(scores, axis=0)
    
    # Select top 5% of rows
    num_to_select = max(1, len(rows) // 20)
    top_indices = np.argsort(combined_score)[-num_to_select:]
    
    return sorted(top_indices.tolist())

if __name__ == "__main__":
    filename = 'pom3a.n1000.train.csv'
    try:
        with open(filename, 'r') as f:
            reader = csv.reader(f)
            header = next(reader)
            rows = []
            for row in reader:
                # Convert numeric columns to floats
                processed_row = []
                for i, val in enumerate(row):
                    try:
                        processed_row.append(float(val))
                    except ValueError:
                        processed_row.append(val)
                rows.append(processed_row)
        
        best_indices = remember(header, rows)
        
        print(f"Found {len(best_indices)} top-performing setups.")
        print("Indices of best rows (0-based):")
        print(best_indices)
        
        # Display a summary of the first best row
        if best_indices:
            print("\nExample of a top-performing row:")
            print(rows[best_indices[0]])
            
    except FileNotFoundError:
        print(f"Error: {filename} not found.")
        exit(1)
```
