import csv
import pandas as pd
import numpy as np

def remember(header, rows):
    """
    Identifies rows that perform well across all goal columns.
    Goals ending in '-' are minimized, '+' are maximized.
    """
    df = pd.DataFrame(rows, columns=header)
    
    # Identify goal columns
    goals = [col for col in header if col.endswith(('-', '+'))]
    
    # Normalize goals to a [0, 1] scale where 1 is always "best"
    # For '-', lower is better (1 - normalized value)
    # For '+', higher is better (normalized value)
    scores = pd.DataFrame(index=df.index)
    for goal in goals:
        col_data = df[goal].astype(float)
        normalized = (col_data - col_data.min()) / (col_data.max() - col_data.min())
        if goal.endswith('-'):
            scores[goal] = 1 - normalized
        else:
            scores[goal] = normalized
            
    # Calculate aggregate score (mean of normalized goals)
    df['aggregate'] = scores.mean(axis=1)
    
    # Select top 5% (or at least 1)
    n_select = max(1, int(len(df) * 0.05))
    best_indices = df['aggregate'].nlargest(n_select).index.tolist()
    
    return sorted(best_indices)

import csv
import pandas as pd
import numpy as np
from scipy import stats

def certify(xs, ys):
    """
    Returns True if the difference between two distributions is statistically 
    insignificant (p > 0.05), False otherwise.
    """
    # Use Welch's t-test (equal_var=False) to compare means
    t_stat, p_value = stats.ttest_ind(xs, ys, equal_var=False)
    
    # If p > 0.05, we fail to reject the null hypothesis (they are the same)
    return p_value > 0.05

def analyze_teams(header, rows):
    df = pd.DataFrame(rows, columns=header)
    
    # Define "small" vs "big" based on the median of 'Team Size'
    median_team = df['Team Size'].median()
    small_teams = df[df['Team Size'] <= median_team]['Cost-']
    big_teams = df[df['Team Size'] > median_team]['Cost-']
    
    is_same = certify(small_teams.tolist(), big_teams.tolist())
    
    return is_same, small_teams.mean(), big_teams.mean()

if __name__ == "__main__":

import csv
import pandas as pd
import numpy as np
from step01 import remember

def flow(header, row, seen):
    """
    Decides whether to accept a row based on whether it is better than 
    the best seen so far. We use the aggregate scoring logic from step01.
    """
    if not seen:
        return True
    
    # Combine seen rows and current row to calculate relative performance
    all_data = seen + [row]
    df = pd.DataFrame(all_data, columns=header)
    
    # Identify goal columns
    goals = [col for col in header if col.endswith(('-', '+'))]
    
    # Normalize goals
    scores = pd.DataFrame(index=df.index)
    for goal in goals:
        col_data = df[goal].astype(float)
        # Handle case where all values are identical
        diff = col_data.max() - col_data.min()
        normalized = (col_data - col_data.min()) / diff if diff != 0 else 0
        
        if goal.endswith('-'):
            scores[goal] = 1 - normalized
        else:
            scores[goal] = normalized
            
    # Calculate aggregate scores
    df['aggregate'] = scores.mean(axis=1)
    
    # Accept if the current row is the best seen so far
    return df['aggregate'].iloc[-1] >= df['aggregate'].iloc[:-1].max()

def justify(header, rows, row):
    """
    Estimates goal values for the target row, calculates its rank relative to 
    all rows, and provides a justification string.
    """
    df = pd.DataFrame(rows, columns=header)
    input_cols = [col for col in header if not (col.endswith('-') or col.endswith('+'))]
    goal_cols = [col for col in header if col.endswith(('-', '+'))]
    
    # Estimate goals for the target row using KNN
    X_train = df[input_cols]
    y_train = df[goal_cols]
    model = KNeighborsRegressor(n_neighbors=5)
    model.fit(X_train, y_train)
    
    # Predict goals for the target row
    target_inputs = pd.DataFrame([row[:len(input_cols)]], columns=input_cols)
    predicted_goals = model.predict(target_inputs)[0]
    
    # Create a combined dataset to calculate aggregate scores
    all_data = df.copy()
    # Add the predicted row to the dataset
    predicted_row = list(target_inputs.iloc[0]) + list(predicted_goals)
    all_data.loc[len(all_data)] = predicted_row
    
    # Calculate aggregate scores
    scores = pd.DataFrame(index=all_data.index)
    for goal in goal_cols:
        col_data = all_data[goal].astype(float)
