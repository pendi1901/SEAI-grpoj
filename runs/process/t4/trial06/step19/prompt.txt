import csv
import pandas as pd
import numpy as np

def remember(header, rows):
    """
    Identifies rows that perform well across all goal columns.
    Goals ending in '-' are minimized, '+' are maximized.
    """
    df = pd.DataFrame(rows, columns=header)
    
    # Identify goal columns
    goals = [col for col in header if col.endswith(('-', '+'))]
    
    # Normalize goals to a [0, 1] scale where 1 is always "best"
    # For '-', lower is better (1 - normalized value)
    # For '+', higher is better (normalized value)
    scores = pd.DataFrame(index=df.index)
    for goal in goals:
        col_data = df[goal].astype(float)
        normalized = (col_data - col_data.min()) / (col_data.max() - col_data.min())
        if goal.endswith('-'):
            scores[goal] = 1 - normalized
        else:
            scores[goal] = normalized
            
    # Calculate aggregate score (mean of normalized goals)
    df['aggregate'] = scores.mean(axis=1)
    
    # Select top 5% (or at least 1)
    n_select = max(1, int(len(df) * 0.05))
    best_indices = df['aggregate'].nlargest(n_select).index.tolist()
    
    return sorted(best_indices)

def certify(xs, ys):
    """
    Returns True if the difference between two distributions is statistically 
    insignificant (p > 0.05), False otherwise.
    """
    # Use Welch's t-test (equal_var=False) to compare means
    t_stat, p_value = stats.ttest_ind(xs, ys, equal_var=False)
    
    # If p > 0.05, we fail to reject the null hypothesis (they are the same)
    return p_value > 0.05

def analyze_teams(header, rows):
    df = pd.DataFrame(rows, columns=header)
    
    # Define "small" vs "big" based on the median of 'Team Size'
    median_team = df['Team Size'].median()
    small_teams = df[df['Team Size'] <= median_team]['Cost-']
    big_teams = df[df['Team Size'] > median_team]['Cost-']
    
    is_same = certify(small_teams.tolist(), big_teams.tolist())
    
    return is_same, small_teams.mean(), big_teams.mean()

import csv
import pandas as pd

from step01 import remember

def flow(header, row, seen):
    """
    Decides whether to accept a row based on whether it is better than 
    the best seen so far. We use the aggregate scoring logic from step01.
    """
    if not seen:
        return True
    
    # Combine seen rows and current row to calculate relative performance
    all_data = seen + [row]
    df = pd.DataFrame(all_data, columns=header)
    
    # Identify goal columns
    goals = [col for col in header if col.endswith(('-', '+'))]
    
    # Normalize goals
    scores = pd.DataFrame(index=df.index)
    for goal in goals:
        col_data = df[goal].astype(float)
        # Handle case where all values are identical
        diff = col_data.max() - col_data.min()
        normalized = (col_data - col_data.min()) / diff if diff != 0 else 0
        
        if goal.endswith('-'):
            scores[goal] = 1 - normalized
        else:
            scores[goal] = normalized
            
    # Calculate aggregate scores
    df['aggregate'] = scores.mean(axis=1)
    
    # Accept if the current row is the best seen so far
    return df['aggregate'].iloc[-1] >= df['aggregate'].iloc[:-1].max()

import csv
import numpy as np
import pandas as pd
from sklearn.ensemble import RandomForestRegressor

def choose(header, rows, label):
    """
    Finds the best row by sampling 50 candidates using an acquisition function.
    """
    input_cols = [col for col in header if not (col.endswith('-') or col.endswith('+'))]
    goal_cols = [col for col in header if col.endswith(('-', '+'))]
    
    # Convert rows to DataFrame, ensuring goal columns are numeric (float)
    df = pd.DataFrame(rows, columns=header)
    for col in goal_cols:
        df[col] = pd.to_numeric(df[col], errors='coerce')
    
    # 1. Initial exploration: Sample 10 random rows to build a baseline
    indices = list(range(len(rows)))
    np.random.shuffle(indices)
    
    labeled_indices = []
    for i in indices[:10]:
        goals = label(i)
        for idx, col in enumerate(goal_cols):
            df.at[i, col] = float(goals[idx])
        labeled_indices.append(i)
        
    # 2. Active learning: Sample remaining 40 rows
    for _ in range(40):
        train_df = df.loc[labeled_indices]
        X = train_df[input_cols]
        
        # Calculate aggregate score for labeled data
        scores = pd.DataFrame(index=train_df.index)
        for goal in goal_cols:
            col_data = train_df[goal]
            diff = col_data.max() - col_data.min()
            norm = (col_data - col_data.min()) / diff if diff != 0 else 0
            scores[goal] = (1 - norm) if goal.endswith('-') else norm
        y = scores.mean(axis=1)
        
        model = RandomForestRegressor(n_estimators=20, random_state=42)
        model.fit(X, y)
        
        # Predict scores for all unlabeled rows
        unlabeled = [i for i in range(len(rows)) if i not in labeled_indices]
        X_unlabeled = df.loc[unlabeled, input_cols]
        preds = model.predict(X_unlabeled)
        
        # Pick the best predicted row
        best_unlabeled_idx = unlabeled[np.argmax(preds)]
        
        # Label it
        goals = label(best_unlabeled_idx)
        for idx, col in enumerate(goal_cols):
            df.at[best_unlabeled_idx, col] = float(goals[idx])
        labeled_indices.append(best_unlabeled_idx)
        
    # 3. Return the index of the best row found
    final_scores = pd.DataFrame(index=labeled_indices)
    for goal in goal_cols:
        col_data = df.loc[labeled_indices, goal]
        diff = col_data.max() - col_data.min()
        norm = (col_data - col_data.min()) / diff if diff != 0 else 0
        final_scores[goal] = (1 - norm) if goal.endswith('-') else norm
    
    best_idx = final_scores.mean(axis=1).idxmax()
    return int(best_idx)

def spot(header, rows):
    """
    Identifies rows that are outliers (potentially mistyped or made up)
    using an Isolation Forest algorithm.
    """
    df = pd.DataFrame(rows, columns=header)
    
    # Identify input columns (non-goal columns)
    input_cols = [col for col in header if not (col.endswith('-') or col.endswith('+'))]
    
    # Use Isolation Forest to detect anomalies
    # contamination is set to a small value assuming few errors
    model = IsolationForest(n_estimators=100, contamination=0.05, random_state=42)
    
    # Fit on input columns
    X = df[input_cols]
    model.fit(X)
    
    # decision_function returns the anomaly score (lower is more abnormal)
    scores = model.decision_function(X)
    
    # Get indices of rows sorted by anomaly score (most suspicious first)
    suspicious_indices = np.argsort(scores)
    
    return suspicious_indices.tolist()

import csv
import pandas as pd
import numpy as np
from sklearn.ensemble import RandomForestRegressor
from step01 import remember

def race(header, rows, label):
    """
    Compares three strategies: random, greedy, and active learning.
    Budget is 50 labels total.
    """
    input_cols = [col for col in header if not (col.endswith('-') or col.endswith('+'))]
    goal_cols = [col for col in header if col.endswith(('-', '+'))]
    
    df = pd.DataFrame(rows, columns=header)
    labeled_indices = {} # index: goals
    
    def get_score(row_idx):
        if row_idx not in labeled_indices:
            if len(labeled_indices) >= 50:
                raise ValueError("Budget exceeded")
            labeled_indices[row_idx] = label(row_idx)
        
        goals = labeled_indices[row_idx]
        scores = []
        for i, col in enumerate(goal_cols):
            val = float(goals[i])
            # Normalize relative to full dataset range if possible, 
            # but here we use simple min-max of current knowledge
            scores.append(1 - val if col.endswith('-') else val)
        return np.mean(scores)

    # 1. Random Strategy
    random_indices = np.random.choice(len(rows), 16, replace=False)
    for idx in random_indices:
        get_score(idx)
    
    # 2. Greedy Strategy (Pick rows with best inputs based on initial random)
    # Simple heuristic: sum of inputs
    greedy_indices = df[input_cols].sum(axis=1).nlargest(17).index.tolist()
    for idx in greedy_indices:
        get_score(idx)
        
    # 3. Active Learning Strategy
    while len(labeled_indices) < 50:
        train_idx = list(labeled_indices.keys())
        X = df.loc[train_idx, input_cols]
        y = [get_score(i) for i in train_idx]
        model = RandomForestRegressor(n_estimators=10).fit(X, y)
        
        unlabeled = [i for i in range(len(rows)) if i not in labeled_indices]
        preds = model.predict(df.loc[unlabeled, input_cols])
        best_unlabeled = unlabeled[np.argmax(preds)]
        get_score(best_unlabeled)

    # Evaluate
    results = {}
    for name, idxs in [("random", random_indices), ("greedy", greedy_indices), ("active", [i for i in labeled_indices if i not in random_indices and i not in greedy_indices])]:
        scores = {i: get_score(i) for i in idxs}
        best_i = max(scores, key=scores.get)
        results[name] = (best_i, scores[best_i])
        
    winner = max(results, key=lambda k: results[k][1])
    return {"winner": winner, "row": results[winner][0], "tried": ["random", "greedy", "active"]}

if __name__ == "__main__":
    filename = 'pom3a.n1000.train.csv'
    df = pd.read_csv(filename)

import csv
import pandas as pd
from step01 import remember

def flow(header, row, seen):
    """
    Returns True if the current row is 'bad' (a significant drop in performance)
    compared to the historical trend, acting as an early warning system.
    """
    if not seen:
        return False
    
    # Combine seen rows to establish a baseline
    df_seen = pd.DataFrame(seen, columns=header)
    
    # Identify goal columns
    goals = [col for col in header if col.endswith(('-', '+'))]
    
    # Calculate aggregate scores for seen data
    scores = pd.DataFrame(index=df_seen.index)
    for goal in goals:
        col_data = df_seen[goal].astype(float)
        diff = col_data.max() - col_data.min()
        normalized = (col_data - col_data.min()) / diff if diff != 0 else 0
        scores[goal] = (1 - normalized) if goal.endswith('-') else normalized
    
    df_seen['aggregate'] = scores.mean(axis=1)
    
    # Calculate the current row's aggregate score
    # We use the mean of the seen data to normalize the new row's goals
    current_agg = 0
    for goal in goals:
        val = float(row[header.index(goal)])
        col_data = df_seen[goal].astype(float)
        diff = col_data.max() - col_data.min()
        norm = (val - col_data.min()) / diff if diff != 0 else 0
        current_agg += (1 - norm) if goal.endswith('-') else norm
    current_agg /= len(goals)
    
    # If the new row is significantly worse than the average of the seen rows
    # (e.g., 2 standard deviations below the mean), warn the user.
    mean_agg = df_seen['aggregate'].mean()
    std_agg = df_seen['aggregate'].std()
    
    # Return True if performance has "gone bad" (dropped significantly)
    if std_agg > 0 and current_agg < (mean_agg - 2 * std_agg):
        return True
    return False

if __name__ == "__main__":

    try:

            rows = [[float(val) for val in row] for row in reader]
            
        seen = []
        warnings = 0
        for row in rows:
            if flow(header, row, seen):

def justify(header, rows, row):
    """
    Estimates goal values for the target row, calculates its rank relative to 
    all rows, and provides a justification string explaining why it is 
    better or worse than other configurations.
    """
    df = pd.DataFrame(rows, columns=header)
    input_cols = [col for col in header if not (col.endswith('-') or col.endswith('+'))]
    goal_cols = [col for col in header if col.endswith(('-', '+'))]
    
    # Estimate goals for the target row using KNN
    X_train = df[input_cols]
    y_train = df[goal_cols]
    model = KNeighborsRegressor(n_neighbors=5)
    model.fit(X_train, y_train)
    
    # Predict goals for the target row
    target_inputs = pd.DataFrame([row[:len(input_cols)]], columns=input_cols)
    predicted_goals = model.predict(target_inputs)[0]
    
    # Create a combined dataset to calculate aggregate scores
    all_data = df.copy()
    # Add the predicted row to the dataset
    predicted_row = list(target_inputs.iloc[0]) + list(predicted_goals)
    all_data.loc[len(all_data)] = predicted_row
    
    # Calculate aggregate scores
    scores = pd.DataFrame(index=all_data.index)
    for goal in goal_cols:
        col_data = all_data[goal].astype(float)
        diff = col_data.max() - col_data.min()
        normalized = (col_data - col_data.min()) / diff if diff != 0 else 0
        if goal.endswith('-'):
            scores[goal] = 1 - normalized
        else:
            scores[goal] = normalized
    
    all_data['aggregate'] = scores.mean(axis=1)
    
    # Calculate rank (fraction of rows worse than target)
    target_score = all_data['aggregate'].iloc[-1]
    others = all_data['aggregate'].iloc[:-1]
    p = (others < target_score).mean()
    
    reason = f"This setup is better than {p:.1%} of all configurations because its predicted performance metrics balance the goals more effectively than the alternative."
    
    return p, reason

import csv
import pandas as pd
import numpy as np
from sklearn.ensemble import RandomForestRegressor

def choose(header, rows, label):
    """
    Finds the best setup for a large, high-stakes project.
    We define 'large' as having a high 'Size' value.
    We use active learning to find the best configuration within 50 trials.
    """
    # Convert rows to a DataFrame, handling '?' as NaN for now
    df = pd.DataFrame(rows, columns=header)
    input_cols = [col for col in header if not (col.endswith('-') or col.endswith('+'))]
    goal_cols = [col for col in header if col.endswith(('-', '+'))]
    
    # Filter for 'large' projects (Size > median)
    size_median = df['Size'].median()
    large_indices = df[df['Size'] > size_median].index.tolist()
    
    # Store labeled data in a dictionary to avoid re-labeling
    labeled_data = {}
    
    def get_labeled_goals(i):
        if i not in labeled_data:
            labeled_data[i] = [float(x) for x in label(i)]
        return labeled_data[i]
    
    # 1. Initial exploration: Sample 10 random large projects
    sample_size = min(10, len(large_indices))
    indices_to_sample = np.random.choice(large_indices, sample_size, replace=False).tolist()
    
    for i in indices_to_sample:
        get_labeled_goals(i)
        
    # 2. Active learning: Sample remaining 40 rows from the large project pool
    for _ in range(40):
        if len(labeled_data) >= 50:
            break
            
        # Prepare training data from labeled rows
        train_indices = list(labeled_data.keys())
        X_train = df.loc[train_indices, input_cols]
        
        # Calculate aggregate score for training
        y_train = []
        for i in train_indices:
            goals = labeled_data[i]
            scores = []
            for idx, col in enumerate(goal_cols):
                val = goals[idx]
                # Normalize using global min/max if possible, or local
                scores.append((1 - val) if col.endswith('-') else val)
            y_train.append(np.mean(scores))
        
        model = RandomForestRegressor(n_estimators=20, random_state=42)
        model.fit(X_train, y_train)
        
        # Predict for unlabeled large projects
        unlabeled = [i for i in large_indices if i not in labeled_data]
        if not unlabeled:
            break
            
        X_unlabeled = df.loc[unlabeled, input_cols]
        preds = model.predict(X_unlabeled)
        
        best_unlabeled_idx = unlabeled[np.argmax(preds)]
        get_labeled_goals(best_unlabeled_idx)
        
    # 3. Return the index of the best row found among the labeled ones
    best_idx = -1
    best_score = -float('inf')
    
    for i in labeled_data:
        goals = labeled_data[i]
        scores = []
        for idx, col in enumerate(goal_cols):
            val = goals[idx]
            scores.append((1 - val) if col.endswith('-') else val)
        
        current_score = np.mean(scores)
        if current_score > best_score:
            best_score = current_score
            best_idx = i
            
    return int(best_idx)
