import csv
import pandas as pd
import numpy as np

def remember(header, rows):
    """
    Identifies rows that perform well across all goal columns.
    Goals ending in '-' are minimized, goals ending in '+' are maximized.
    """
    df = pd.DataFrame(rows, columns=header)
    
    # Identify goal columns
    goals = [col for col in header if col.endswith(('-', '+'))]
    
    # Normalize goals to [0, 1] range for comparison
    # For '-', lower is better (1 - normalized_value)
    # For '+', higher is better (normalized_value)
    scores = pd.DataFrame(index=df.index)
    for goal in goals:
        col_min = df[goal].min()
        col_max = df[goal].max()
        norm = (df[goal] - col_min) / (col_max - col_min + 1e-9)
        
        if goal.endswith('-'):
            scores[goal] = 1 - norm
        else:
            scores[goal] = norm
            
    # Calculate aggregate score (mean of normalized goals)
    df['aggregate'] = scores.mean(axis=1)
    
    # Select top 5% (or at least 1)
    n_select = max(1, int(len(df) * 0.05))
    top_indices = df['aggregate'].nlargest(n_select).index.tolist()
    
    return sorted(top_indices)

import csv
import pandas as pd
import numpy as np
from scipy.stats import ttest_ind

def certify(xs, ys):
    """
    Returns True if the difference between two groups is statistically 
    insignificant (p > 0.05), False if they are significantly different.
    """
    if len(xs) < 2 or len(ys) < 2:
        return True
    _, p_value = ttest_ind(xs, ys, equal_var=False)
    return p_value > 0.05

def analyze_team_size_impact(header, rows):
    df = pd.DataFrame(rows, columns=header)
    
    # Define small vs big teams based on the median of 'Team Size'
    median_team_size = df['Team Size'].median()
    small_teams = df[df['Team Size'] <= median_team_size]['Cost-']
    big_teams = df[df['Team Size'] > median_team_size]['Cost-']
    
    is_same = certify(small_teams.tolist(), big_teams.tolist())
    
    return is_same, small_teams.mean(), big_teams.mean()

if __name__ == "__main__":
    filename = 'pom3a.n1000.train.csv'
    try:
        with open(filename, 'r') as f:
            reader = csv.reader(f)
            header = next(reader)
            rows = []
            for row in reader:
                rows.append([float(val) for val in row])
        
        is_same, small_avg, big_avg = analyze_team_size_impact(header, rows)
        
        print("Customer Inquiry: Are small-team setups cheaper than big-team ones?")
        print(f"Average Cost (Small Teams): {small_avg:.4f}")
        print(f"Average Cost (Big Teams):   {big_avg:.4f}")
        
        if is_same:
            print("Conclusion: The difference is statistically insignificant. It was likely luck.")
        else:
            print("Conclusion: There is a statistically significant difference in costs.")
            
    except Exception as e:
        print(f"Error: {e}")
        exit(1)
