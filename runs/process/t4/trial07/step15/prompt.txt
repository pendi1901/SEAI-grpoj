import csv
import pandas as pd
import numpy as np

def remember(header, rows):
    """
    Identifies rows that perform well across all goal columns.
    Goals ending in '-' are minimized, goals ending in '+' are maximized.
    """
    df = pd.DataFrame(rows, columns=header)
    
    # Identify goal columns
    goals = [col for col in header if col.endswith(('-', '+'))]
    
    # Normalize goals to [0, 1] range for comparison
    # For '-', lower is better (1 - normalized_value)
    # For '+', higher is better (normalized_value)
    scores = pd.DataFrame(index=df.index)
    for goal in goals:
        col_min = df[goal].min()
        col_max = df[goal].max()
        norm = (df[goal] - col_min) / (col_max - col_min + 1e-9)
        
        if goal.endswith('-'):
            scores[goal] = 1 - norm
        else:
            scores[goal] = norm
            
    # Calculate aggregate score (mean of normalized goals)
    df['aggregate'] = scores.mean(axis=1)
    
    # Select top 5% (or at least 1)
    n_select = max(1, int(len(df) * 0.05))
    top_indices = df['aggregate'].nlargest(n_select).index.tolist()
    
    return sorted(top_indices)

def certify(xs, ys):
    """
    Returns True if the difference between two groups is statistically 
    insignificant (p > 0.05), False if they are significantly different.
    """
    if len(xs) < 2 or len(ys) < 2:
        return True
    _, p_value = ttest_ind(xs, ys, equal_var=False)
    return p_value > 0.05

def analyze_team_size_impact(header, rows):
    df = pd.DataFrame(rows, columns=header)
    
    # Define small vs big teams based on the median of 'Team Size'
    median_team_size = df['Team Size'].median()
    small_teams = df[df['Team Size'] <= median_team_size]['Cost-']
    big_teams = df[df['Team Size'] > median_team_size]['Cost-']
    
    is_same = certify(small_teams.tolist(), big_teams.tolist())
    
    return is_same, small_teams.mean(), big_teams.mean()

def flow(header, row, seen):
    """
    Decides whether to accept a row based on the 'Secretary Problem' logic 
    combined with the 'remember' heuristic. We observe the first 30% of 
    incoming rows to establish a baseline, then accept the first row that 
    is better than the best seen so far.
    """
    # Identify goal columns
    goals = [col for col in header if col.endswith(('-', '+'))]
    
    # Calculate aggregate score for a row
    def get_score(r):
        score = 0
        for i, col in enumerate(header):
            if col.endswith('-'):
                score -= r[i]
            elif col.endswith('+'):
                score += r[i]
        return score

    # If it's the first row, we can't compare, so keep looking
    if not seen:
        return False

    # Establish baseline: look at the first 30% of the expected 800 rows (approx 240)
    # If we haven't seen enough, just keep looking
    if len(seen) < 240:
        return False

    # Find the best score seen so far
    best_score = max(get_score(s) for s in seen)
    current_score = get_score(row)

    # If current row is better than anything seen, accept it
    if current_score > best_score:
        return True
    
    # If we are at the very end, accept it
    if len(seen) >= 799:
        return True

    return False

import csv

from sklearn.neighbors import KNeighborsRegressor
from step01 import remember

def justify(header, rows, row):
    """
    Estimates the performance of a given row and returns its percentile rank
    relative to the provided dataset.
    """
    df = pd.DataFrame(rows, columns=header)
    inputs = [col for col in header if not (col.endswith('-') or col.endswith('+'))]
    goals = [col for col in header if col.endswith(('-', '+'))]

    # Estimate missing goals for the input row
    predictions = {}
    for goal in goals:
        model = KNeighborsRegressor(n_neighbors=5)
        model.fit(df[inputs], df[goal])
        predictions[goal] = model.predict([row[:len(inputs)]])[0]

    # Calculate aggregate score for all rows
    def get_aggregate(data_df):
        scores = pd.DataFrame(index=data_df.index)
        for goal in goals:
            col_min, col_max = data_df[goal].min(), data_df[goal].max()
            norm = (data_df[goal] - col_min) / (col_max - col_min + 1e-9)
            scores[goal] = (1 - norm) if goal.endswith('-') else norm
        return scores.mean(axis=1)

    all_scores = get_aggregate(df)
    
    # Calculate score for the input row
    row_df = pd.DataFrame([list(row[:len(inputs)]) + [predictions[g] for g in goals]], columns=header)
    row_score = get_aggregate(row_df).iloc[0]

    # Calculate percentile (fraction of rows that are worse than this one)
    p = (all_scores < row_score).mean()
    
    reason = f"This setup is better than {p:.1%} of all known configurations based on its estimated performance across all goals."
    return p, reason

def choose(header, rows, label):
    """
    Finds the best row by querying 50 setups. Since we start with no labels,
    we sample 50 rows across the space to build an initial model, then 
    query the most promising ones.
    """
    inputs = [col for col in header if not (col.endswith('-') or col.endswith('+'))]
    goals = [col for col in header if col.endswith(('-', '+'))]
    
    # 1. Initial sampling: Query 50 rows spread across the dataset
    # to build a baseline model.
    n_rows = len(rows)
    indices_to_query = np.linspace(0, n_rows - 1, 50, dtype=int)
    
    known_data = []
    for idx in indices_to_query:
        true_goals = label(idx)
        row_data = rows[idx][:len(inputs)] + true_goals
        known_data.append(row_data)
        
    df_known = pd.DataFrame(known_data, columns=header)
    
    # 2. Calculate aggregate score for the 50 queried rows
    scores = pd.DataFrame(index=df_known.index)
    for goal in goals:
        g_min, g_max = df_known[goal].min(), df_known[goal].max()
        norm = (df_known[goal] - g_min) / (g_max - g_min + 1e-9)
        scores[goal] = 1 - norm if goal.endswith('-') else norm
    df_known['aggregate'] = scores.mean(axis=1)
    
    # 3. Train surrogate model
    model = RandomForestRegressor(n_estimators=50, random_state=42)
    model.fit(df_known[inputs], df_known['aggregate'])
    
    # 4. Predict for all rows and find the best one
    # We use the model to find the best candidate among all rows
    all_inputs = pd.DataFrame([r[:len(inputs)] for r in rows], columns=inputs)
    preds = model.predict(all_inputs)
    
    best_idx = int(np.argmax(preds))
    return best_idx

def fix(header, row, levers):
    """
    Finds the smallest change to levers that improves the row's aggregate score.
    Uses a Random Forest surrogate model to predict the impact of changes.
    """
    # Load data to train a surrogate model
    filename = 'pom3a.n1000.train.csv'
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header_csv = next(reader)
        rows = [[float(val) for val in r] for r in reader]
    
    df = pd.DataFrame(rows, columns=header_csv)
    inputs = [c for c in header_csv if not (c.endswith('-') or c.endswith('+'))]
    goals = [c for c in header_csv if c.endswith(('-', '+'))]
    
    # Calculate aggregate score for training
    scores = pd.DataFrame(index=df.index)
    for g in goals:
        norm = (df[g] - df[g].min()) / (df[g].max() - df[g].min() + 1e-9)
        scores[g] = 1 - norm if g.endswith('-') else norm
    df['aggregate'] = scores.mean(axis=1)
    
    # Train surrogate
    model = RandomForestRegressor(n_estimators=20, random_state=42)
    model.fit(df[inputs], df['aggregate'])
    
    # Find best lever change
    best_change = {}
    best_score = -1
    
    # Try small perturbations for each lever
    for col, bounds in levers.items():
        idx = header.index(col)
        original_val = row[idx]
        
        # Test a few points within the allowed range
        if isinstance(bounds, (tuple, list)):
            low, high = bounds
            test_vals = np.linspace(low, high, 5)
        else:
            test_vals = bounds
            
        for val in test_vals:
            if val == original_val: continue
            
            new_row = list(row)
            new_row[idx] = val
            
            # Predict score
            pred = model.predict([new_row[:len(inputs)]])[0]
            if pred > best_score:
                best_score = pred
                best_change = {col: val}
                
    return best_change

def spot(header, rows):
    """
    Identifies rows that are outliers (potentially mistyped or made up)
    using an Isolation Forest model.
    """
    df = pd.DataFrame(rows, columns=header)
    
    # Identify input columns (those not ending in - or +)
    inputs = [col for col in header if not (col.endswith(('-', '+')))]
    
    # Isolation Forest is effective for anomaly detection
    # contamination is set to 0.05 as a heuristic for "a few" rows
    model = IsolationForest(n_estimators=100, contamination=0.05, random_state=42)
    
    # Fit on input columns
    preds = model.fit_predict(df[inputs])
    scores = model.decision_function(df[inputs])
    
    # Rows with -1 are outliers. 
    # We sort by decision_function score (lower is more anomalous)
    outlier_indices = np.where(preds == -1)[0]
    sorted_outliers = sorted(outlier_indices, key=lambda i: scores[i])
    
    return sorted_outliers

import csv
import numpy as np
import pandas as pd
from sklearn.ensemble import RandomForestRegressor
from step01 import remember

def race(header, rows, label):
    """
    Compares three strategies: random, greedy (best predicted), and uncertainty (exploration).
    Budget: 50 labels total.
    """
    inputs = [c for c in header if not (c.endswith(('-', '+')))]
    goals = [c for c in header if c.endswith(('-', '+'))]
    
    budget = 50
    queried = {} # index: true_goals

    def get_score(row_goals):
        # Normalize and aggregate
        scores = []
        for i, g in enumerate(goals):
            # Use global min/max if possible, or local relative to current knowledge
            val = row_goals[i]
            if header[len(inputs) + i].endswith('-'):
                scores.append(1 - val)
            else:
                scores.append(val)
        return np.mean(scores)

    def get_label(idx):
        if idx not in queried:
            if len(queried) >= budget:
                raise Exception("Budget exceeded")
            queried[idx] = label(idx)
        return queried[idx]

    # Strategy 1: Random
    def strategy_random():
        idx = np.random.choice(len(rows))
        return idx, get_label(idx)

    # Strategy 2: Greedy (Surrogate model)
    def strategy_greedy():
        if len(queried) < 10: return strategy_random()
        df = pd.DataFrame([rows[i][:len(inputs)] + queried[i] for i in queried], columns=header)
        model = RandomForestRegressor().fit(df[inputs], [get_score(queried[i]) for i in queried])
        preds = model.predict([r[:len(inputs)] for r in rows])
        idx = int(np.argmax(preds))
        return idx, get_label(idx)

    # Strategy 3: Uncertainty (Pick row with highest variance in forest)
    def strategy_uncertain():
        if len(queried) < 10: return strategy_random()
        df = pd.DataFrame([rows[i][:len(inputs)] + queried[i] for i in queried], columns=header)
        model = RandomForestRegressor().fit(df[inputs], [get_score(queried[i]) for i in queried])
        # Variance of trees as uncertainty proxy
        preds = np.array([tree.predict([r[:len(inputs)] for r in rows]) for tree in model.estimators_])
        idx = int(np.argmax(np.var(preds, axis=0)))
        return idx, get_label(idx)

    strategies = {"random": strategy_random, "greedy": strategy_greedy, "uncertain": strategy_uncertain}
    results = {name: {"best_score": -1, "best_idx": -1} for name in strategies}

    # Run 50 trials total, split among strategies
    for i in range(budget):
        name = list(strategies.keys())[i % 3]
        idx, g = strategies[name]()
        score = get_score(g)
        if score > results[name]["best_score"]:
            results[name]["best_score"] = score
            results[name]["best_idx"] = idx

    winner = max(results, key=lambda n: results[n]["best_score"])
    return {"winner": winner, "row": results[winner]["best_idx"], "tried": list(strategies.keys())}

def analyze_shift(header, rows):
    """
    The customer asks if results shifted after a change in team decision-making.
    We assume the dataset is split into two halves: 'before' and 'after' the change.
    We compare the aggregate performance of both halves.
    """
    df = pd.DataFrame(rows, columns=header)
    
    # Identify goal columns
    goals = [col for col in header if col.endswith(('-', '+'))]
    
    # Normalize goals to [0, 1] range
    scores = pd.DataFrame(index=df.index)
    for goal in goals:
        col_min, col_max = df[goal].min(), df[goal].max()
        norm = (df[goal] - col_min) / (col_max - col_min + 1e-9)
        scores[goal] = 1 - norm if goal.endswith('-') else norm
    
    df['aggregate'] = scores.mean(axis=1)
    
    # Split data into two halves
    mid = len(df) // 2
    before = df.iloc[:mid]['aggregate'].tolist()
    after = df.iloc[mid:]['aggregate'].tolist()
    
    # Use the certify function from step03 to check for statistical difference
    is_same = certify(before, after)
    
    return is_same, sum(before)/len(before), sum(after)/len(after)

import csv

from step01 import remember

def get_score(header, row):
    """Calculates a normalized aggregate score for a row."""
    goals = [i for i, col in enumerate(header) if col.endswith(('-', '+'))]
    scores = []
    for i in goals:
        val = row[i]
        # Since we don't have global min/max for every arrival, 
        # we treat values as normalized [0,1] based on problem description.
        if header[i].endswith('-'):
            scores.append(1.0 - val)
        else:
            scores.append(val)
    return np.mean(scores) if scores else 0

def flow(header, row, seen):
    """
    Warns if a row is 'going bad'. We define 'going bad' as a significant 
    drop in performance compared to the moving average of the last 50 rows.
    """
    if len(seen) < 50:
        return False
    
    current_score = get_score(header, row)
    recent_scores = [get_score(header, s) for s in seen[-50:]]
    avg_recent = np.mean(recent_scores)
    std_recent = np.std(recent_scores)
    
    # If the current score is more than 2 standard deviations below the recent average,
    # we flag it as 'going bad'.
    if current_score < (avg_recent - 2 * (std_recent + 1e-9)):
        return True
    
    return False
