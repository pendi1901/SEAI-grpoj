import csv
import pandas as pd
import numpy as np

def remember(header, rows):
    """
    Identifies rows that perform well across all goal columns.
    Goals ending in '-' are minimized, goals ending in '+' are maximized.
    """
    df = pd.DataFrame(rows, columns=header)
    
    # Identify goal columns
    goals = [col for col in header if col.endswith(('-', '+'))]
    
    # Normalize goals to [0, 1] range for comparison
    # For '-', lower is better (1 - normalized_value)
    # For '+', higher is better (normalized_value)
    scores = pd.DataFrame(index=df.index)
    for goal in goals:
        col_min = df[goal].min()
        col_max = df[goal].max()
        norm = (df[goal] - col_min) / (col_max - col_min + 1e-9)
        
        if goal.endswith('-'):
            scores[goal] = 1 - norm
        else:
            scores[goal] = norm
            
    # Calculate aggregate score (mean of normalized goals)
    df['aggregate'] = scores.mean(axis=1)
    
    # Select top 5% (or at least 1)
    n_select = max(1, int(len(df) * 0.05))
    top_indices = df['aggregate'].nlargest(n_select).index.tolist()
    
    return sorted(top_indices)

import csv

from scipy.stats import ttest_ind

def certify(xs, ys):
    """
    Returns True if the difference between two groups is statistically 
    insignificant (p > 0.05), False if they are significantly different.
    """
    if len(xs) < 2 or len(ys) < 2:
        return True
    _, p_value = ttest_ind(xs, ys, equal_var=False)
    return p_value > 0.05

def analyze_team_size_impact(header, rows):
    df = pd.DataFrame(rows, columns=header)
    
    # Define small vs big teams based on the median of 'Team Size'
    median_team_size = df['Team Size'].median()
    small_teams = df[df['Team Size'] <= median_team_size]['Cost-']
    big_teams = df[df['Team Size'] > median_team_size]['Cost-']
    
    is_same = certify(small_teams.tolist(), big_teams.tolist())
    
    return is_same, small_teams.mean(), big_teams.mean()

import csv

def flow(header, row, seen):
    """
    Decides whether to accept a row based on the 'Secretary Problem' logic 
    combined with the 'remember' heuristic. We observe the first 30% of 
    incoming rows to establish a baseline, then accept the first row that 
    is better than the best seen so far.
    """
    # Identify goal columns
    goals = [col for col in header if col.endswith(('-', '+'))]
    
    # Calculate aggregate score for a row
    def get_score(r):
        score = 0
        for i, col in enumerate(header):
            if col.endswith('-'):
                score -= r[i]
            elif col.endswith('+'):
                score += r[i]
        return score

    # If it's the first row, we can't compare, so keep looking
    if not seen:
        return False

    # Establish baseline: look at the first 30% of the expected 800 rows (approx 240)
    # If we haven't seen enough, just keep looking
    if len(seen) < 240:
        return False

    # Find the best score seen so far
    best_score = max(get_score(s) for s in seen)
    current_score = get_score(row)

    # If current row is better than anything seen, accept it
    if current_score > best_score:
        return True
    
    # If we are at the very end, accept it
    if len(seen) >= 799:
        return True

    return False

def justify(header, rows, row):
    """
    Estimates the performance of a given row and returns its percentile rank
    relative to the provided dataset.
    """
    df = pd.DataFrame(rows, columns=header)
    inputs = [col for col in header if not (col.endswith('-') or col.endswith('+'))]
    goals = [col for col in header if col.endswith(('-', '+'))]

    # Estimate missing goals for the input row
    predictions = {}
    for goal in goals:
        model = KNeighborsRegressor(n_neighbors=5)
        model.fit(df[inputs], df[goal])
        predictions[goal] = model.predict([row[:len(inputs)]])[0]

    # Calculate aggregate score for all rows
    def get_aggregate(data_df):
        scores = pd.DataFrame(index=data_df.index)
        for goal in goals:
            col_min, col_max = data_df[goal].min(), data_df[goal].max()
            norm = (data_df[goal] - col_min) / (col_max - col_min + 1e-9)
            scores[goal] = (1 - norm) if goal.endswith('-') else norm
        return scores.mean(axis=1)

    all_scores = get_aggregate(df)
    
    # Calculate score for the input row
    row_df = pd.DataFrame([list(row[:len(inputs)]) + [predictions[g] for g in goals]], columns=header)
    row_score = get_aggregate(row_df).iloc[0]

    # Calculate percentile (fraction of rows that are worse than this one)
    p = (all_scores < row_score).mean()
    
    reason = f"This setup is better than {p:.1%} of all known configurations based on its estimated performance across all goals."
    return p, reason
