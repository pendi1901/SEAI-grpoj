import csv
import pandas as pd
import numpy as np

def remember(header, rows):
    """
    Identifies rows that perform well across all goal columns.
    Goals ending in '-' are minimized, goals ending in '+' are maximized.
    """
    df = pd.DataFrame(rows, columns=header)
    
    # Identify goal columns
    goals = [col for col in header if col.endswith(('-', '+'))]
    
    # Normalize goals to a 0-1 scale where 1 is "best"
    # For '-', lower is better (1 - normalized value)
    # For '+', higher is better (normalized value)
    scores = pd.DataFrame(index=df.index)
    for goal in goals:
        col_data = df[goal].astype(float)
        min_val, max_val = col_data.min(), col_data.max()
        normalized = (col_data - min_val) / (max_val - min_val + 1e-9)
        
        if goal.endswith('-'):
            scores[goal] = 1 - normalized
        else:
            scores[goal] = normalized
            
    # Calculate aggregate score (mean of normalized goals)
    df['aggregate_score'] = scores.mean(axis=1)
    
    # Select top 5% of rows
    num_to_select = max(1, len(rows) // 20)
    top_indices = df['aggregate_score'].nlargest(num_to_select).index.tolist()
    
    return sorted(top_indices)

import csv

from scipy import stats

def certify(xs, ys):
    """
    Returns True if the two distributions are statistically similar (p > 0.05),
    False if they are significantly different.
    """
    # Use Welch's t-test (does not assume equal variance)
    t_stat, p_value = stats.ttest_ind(xs, ys, equal_var=False)
    return p_value > 0.05

def analyze_teams(header, rows):
    """
    Compares 'Cost-' for small vs large teams.
    Small teams: Team Size < median. Large teams: Team Size >= median.
    """
    df = pd.DataFrame(rows, columns=header)
    median_team_size = df['Team Size'].median()
    
    small_teams = df[df['Team Size'] < median_team_size]['Cost-']
    large_teams = df[df['Team Size'] >= median_team_size]['Cost-']
    
    is_same = certify(small_teams.tolist(), large_teams.tolist())
    
    return small_teams.mean(), large_teams.mean(), is_same

import csv
import pandas as pd
from step01 import remember

def flow(header, row, seen):
    """
    Decides whether to accept a row based on the 'Secretary Problem' logic:
    1. Observe a sample (first 20%) to establish a baseline of 'good' performance.
    2. Accept the first subsequent row that is better than the best seen in the sample.
    """
    # If this is the first row, we can't compare yet
    if not seen:
        return False
    
    # Define sample size (20% of total expected, but we use seen length)
    # We need a baseline. If we haven't seen enough, keep looking.
    sample_size = max(5, len(seen) // 5)
    
    # Calculate aggregate scores for all seen rows
    all_rows = seen + [row]
    df = pd.DataFrame(all_rows, columns=header)
    
    goals = [col for col in header if col.endswith(('-', '+'))]
    scores = pd.DataFrame(index=df.index)
    for goal in goals:
        col_data = df[goal].astype(float)
        min_val, max_val = col_data.min(), col_data.max()
        normalized = (col_data - min_val) / (max_val - min_val + 1e-9)
        scores[goal] = 1 - normalized if goal.endswith('-') else normalized
            
    df['agg'] = scores.mean(axis=1)
    
    # If we are still in the sampling phase, don't accept
    if len(seen) < sample_size:
        return False
    
    # If the current row is better than the best seen so far in the sample, accept
    best_sample_score = df['agg'].iloc[:len(seen)].max()
    return df['agg'].iloc[-1] > best_sample_score

def justify(header, rows, row):
    """
    Estimates the performance of a row and returns its percentile rank
    and a justification string.
    """
    df = pd.DataFrame(rows, columns=header)
    inputs = [col for col in header if not (col.endswith('-') or col.endswith('+'))]
    goals = [col for col in header if col.endswith(('-', '+'))]
    
    # Estimate the goals for the provided row using KNN
    knn = KNeighborsRegressor(n_neighbors=5)
    knn.fit(df[inputs], df[goals])
    
    # Extract inputs from the provided row (which has '?' for goals)
    input_vals = [row[i] for i, col in enumerate(header) if col in inputs]
    estimated_goals = knn.predict([input_vals])[0]
    
    # Calculate aggregate score for all rows
    def get_agg(data):
        scores = pd.DataFrame(index=data.index)
        for i, goal in enumerate(goals):
            col_data = data[goal].astype(float)
            norm = (col_data - col_data.min()) / (col_data.max() - col_data.min() + 1e-9)
            scores[goal] = 1 - norm if goal.endswith('-') else norm
        return scores.mean(axis=1)

    all_scores = get_agg(df)
    
    # Calculate score for the estimated row
    # We normalize the estimated goal using the training data's min/max
    est_scores = []
    for i, goal in enumerate(goals):
        col_data = df[goal].astype(float)
        val = estimated_goals[i]
        norm = (val - col_data.min()) / (col_data.max() - col_data.min() + 1e-9)
        est_scores.append(1 - norm if goal.endswith('-') else norm)
    
    row_score = np.mean(est_scores)
    
    # Calculate percentile (fraction of rows worse than this one)
    p = (all_scores < row_score).mean()
    
    reason = f"This setup is estimated to outperform {p:.1%} of all known configurations based on its input parameters."
    return p, reason

import csv
import pandas as pd
import numpy as np
from sklearn.ensemble import RandomForestRegressor
from step01 import remember

def choose(header, rows, label):
    """
    Finds the best row by sampling 50 rows, training a surrogate model,
    and selecting the best predicted row.
    """
    # Identify indices
    goal_indices = [i for i, h in enumerate(header) if h.endswith(('-', '+'))]
    input_indices = [i for i, h in enumerate(header) if not (h.endswith('-') or h.endswith('+'))]
    
    # 1. Sample 50 rows to build a model
    # We pick indices spread across the dataset
    indices_to_label = np.linspace(0, len(rows) - 1, 50, dtype=int)
    
    train_inputs = []
    train_goals = []
    
    for i in indices_to_label:
        goals = label(i)
        train_goals.append(goals)
        train_inputs.append([rows[i][j] for j in input_indices])
        
    # 2. Train a surrogate model to predict goals from inputs
    model = RandomForestRegressor(n_estimators=50, random_state=42)
    model.fit(train_inputs, train_goals)
    
    # 3. Predict goals for all rows
    all_inputs = [[rows[i][j] for j in input_indices] for i in range(len(rows))]
    predictions = model.predict(all_inputs)
    
    # 4. Score predictions
    # Normalize goals based on the 50 samples
    pred_df = pd.DataFrame(predictions, columns=[header[i] for i in goal_indices])
    scores = pd.DataFrame(index=pred_df.index)
    
    for col in pred_df.columns:
        col_data = pred_df[col]
        norm = (col_data - col_data.min()) / (col_data.max() - col_data.min() + 1e-9)
        scores[col] = 1 - norm if col.endswith('-') else norm
        
    aggregate_scores = scores.mean(axis=1)
    
    # Return index of the best row
    return int(aggregate_scores.idxmax())
