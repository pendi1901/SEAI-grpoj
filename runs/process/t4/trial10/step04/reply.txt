```python
import csv
import pandas as pd
from step01 import remember

def flow(header, row, seen):
    """
    Decides whether to accept a row based on whether it is better than 
    the best seen so far. Uses the logic from step01 to evaluate quality.
    """
    if not seen:
        return True
    
    # Combine seen rows and current row to evaluate relative quality
    all_rows = seen + [row]
    
    # Get indices of the best rows using the logic from step01
    best_indices = remember(header, all_rows)
    
    # If the current row (last index) is in the top 5% of all seen so far, accept it
    return len(all_rows) - 1 in best_indices

if __name__ == "__main__":
    filename = 'pom3a.n1000.train.csv'
    try:
        with open(filename, 'r') as f:
            reader = csv.reader(f)
            header = next(reader)
            rows = [[float(val) for val in row] for row in reader]
        
        seen = []
        accepted_count = 0
        # Process rows one by one as per customer requirement
        for row in rows:
            if flow(header, row, seen):
                accepted_count += 1
            seen.append(row)
            
        print(f"Processed {len(rows)} proposals.")
        print(f"Total accepted: {accepted_count}")
        print("Decision strategy: Accepted rows that ranked in the top 5% of all proposals seen to date.")
        
    except FileNotFoundError:
        print(f"Error: {filename} not found.")
        exit(1)
```
