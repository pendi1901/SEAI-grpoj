import csv
import pandas as pd
import numpy as np

def remember(header, rows):
    """
    Identifies rows that perform well across all goal columns.
    Goals ending in '-' are minimized, '+' are maximized.
    """
    df = pd.DataFrame(rows, columns=header)
    
    # Identify goal columns
    goals = [col for col in header if col.endswith(('-', '+'))]
    
    # Normalize goals to [0, 1] range for comparison
    # For '-', lower is better (1 - normalized value)
    # For '+', higher is better (normalized value)
    scores = pd.DataFrame(index=df.index)
    for goal in goals:
        col_data = df[goal].astype(float)
        normalized = (col_data - col_data.min()) / (col_data.max() - col_data.min())
        if goal.endswith('-'):
            scores[goal] = 1 - normalized
        else:
            scores[goal] = normalized
            
    # Calculate aggregate score (mean of normalized goals)
    df['aggregate'] = scores.mean(axis=1)
    
    # Select top 5% (or at least 1)
    n_rows = len(df)
    k = max(1, int(n_rows * 0.05))
    
    best_indices = df['aggregate'].nlargest(k).index.tolist()
    return sorted(best_indices)

def certify(xs, ys):
    """
    Returns True if the difference between two distributions is statistically 
    insignificant (p-value > 0.05), False otherwise.
    """
    # Use Welch's t-test (equal_var=False) to compare means
    t_stat, p_value = stats.ttest_ind(xs, ys, equal_var=False)
    return p_value > 0.05

def analyze_teams(header, rows):

    # Identify Team Size column and Cost- goal

    cost_col = 'Cost-'

    median_size = df[team_col].median()
    small_teams = df[df[team_col] <= median_size][cost_col]
    big_teams = df[df[team_col] > median_size][cost_col]
    
    # Check if they are statistically different
    is_same = certify(small_teams.tolist(), big_teams.tolist())

def choose(header, rows, label):
    """
    Finds the best row by sampling 50 rows using an acquisition strategy:
    1. Label 10 random rows to build an initial model.
    2. Use the model to predict goals for remaining rows.
    3. Label the remaining 40 rows that have the highest predicted aggregate score.
    """
    n_rows = len(rows)
    inputs = [c for c in header if not (c.endswith('-') or c.endswith('+'))]
    goals = [c for c in header if c.endswith(('-', '+'))]
    
    labeled_indices = []
    
    # 1. Initial random sample
    initial_samples = np.random.choice(n_rows, 10, replace=False).tolist()
    for i in initial_samples:
        rows[i][len(inputs):] = label(i)
        labeled_indices.append(i)
        
    # 2. Iterative selection
    for _ in range(40):
        # Train model on currently labeled data
        df_train = pd.DataFrame([rows[i] for i in labeled_indices], columns=header)
        
        # Predict for all unlabeled rows
        unlabeled = [i for i in range(n_rows) if i not in labeled_indices]
        X_train = df_train[inputs]
        X_unlabeled = pd.DataFrame([rows[i][:len(inputs)] for i in unlabeled], columns=inputs)
        
        # Predict aggregate score
        scores = pd.DataFrame(index=unlabeled)
        for goal in goals:
            model = KNeighborsRegressor(n_neighbors=3)
            model.fit(X_train, df_train[goal])
            preds = model.predict(X_unlabeled)
            
            # Normalize predictions
            norm = (preds - df_train[goal].min()) / (df_train[goal].max() - df_train[goal].min())
            scores[goal] = (1 - norm) if goal.endswith('-') else norm
            
        scores['agg'] = scores.mean(axis=1)
        best_unlabeled = scores['agg'].idxmax()
        
        # Label the best candidate
        rows[best_unlabeled][len(inputs):] = label(best_unlabeled)
        labeled_indices.append(best_unlabeled)
        
    # Return index of the best row found
    final_df = pd.DataFrame([rows[i] for i in labeled_indices], columns=header, index=labeled_indices)
    # Re-calculate aggregate for all labeled rows to find the absolute best
    scores = pd.DataFrame(index=final_df.index)
    for goal in goals:
        col_data = final_df[goal]
        norm = (col_data - col_data.min()) / (col_data.max() - col_data.min())
        scores[goal] = (1 - norm) if goal.endswith('-') else norm
    
    return scores.mean(axis=1).idxmax()

def fix(header, row, levers):
    """
    Identifies the most impactful lever to change to improve the row's performance.
    Uses a Random Forest to predict the impact of changing levers on the aggregate goal.
    """
    # 1. Load data to build a surrogate model

        header_csv = next(reader)
        rows = [[float(val) for val in row] for row in reader]

    # 2. Calculate aggregate score for training
    scores = pd.DataFrame(index=df.index)
    for goal in goals:
        col_data = df[goal]
        norm = (col_data - col_data.min()) / (col_data.max() - col_data.min())
        scores[goal] = (1 - norm) if goal.endswith('-') else norm
    df['agg'] = scores.mean(axis=1)
    
    # 3. Train model
    model = RandomForestRegressor(n_estimators=50, random_state=42)
    model.fit(df[inputs], df['agg'])
    
    # 4. Evaluate changes for each lever
    best_change = {}

def spot(header, rows):
    """
    Identifies rows that are outliers using Isolation Forest.
    Returns the indices of rows, sorted by suspicion (highest anomaly score first).
    """
    df = pd.DataFrame(rows, columns=header)
    
    # IsolationForest requires numeric data. 
    # Select only numeric columns for anomaly detection.
    numeric_df = df.select_dtypes(include=[np.number])
    
    # Fit the model to the data
    model = IsolationForest(contamination=0.05, random_state=42)
    model.fit(numeric_df)
    
    # decision_function returns the anomaly score of the input samples
    # Lower scores are more anomalous.
    scores = model.decision_function(numeric_df)
    
    # Get indices sorted by score (ascending, so most negative/anomalous first)
    suspicious_indices = np.argsort(scores)
    
    return suspicious_indices.tolist()

import csv

from sklearn.neighbors import KNeighborsRegressor

def race(header, rows, label):
    """
    Compares three strategies: random, greedy (KNN), and uncertainty-based.
    Budget: 50 labels total.
    """
    n_rows = len(rows)
    goals = [c for c in header if c.endswith(('-', '+'))]
    inputs = [c for c in header if not (c.endswith(('-', '+')))]
    
    labeled = {}
    strategy_history = {name: [] for name in ["random", "greedy", "uncertainty"]}

    def get_label(i, strat_name):
        if i not in labeled:
            labeled[i] = label(i)
        strategy_history[strat_name].append(i)
        return labeled[i]

    # Run 50 trials total
    for _ in range(50):
        # Pick strategy
        strat_name = np.random.choice(["random", "greedy", "uncertainty"])
        
        if strat_name == "random" or len(labeled) < 5:
            idx = np.random.choice(n_rows)
            get_label(idx, strat_name)
        elif strat_name == "greedy":
            df_train = pd.DataFrame([rows[i][:len(inputs)] + list(labeled[i]) for i in labeled], columns=header)
            model = KNeighborsRegressor(n_neighbors=3).fit(df_train[inputs], df_train[goals])
            preds = model.predict([r[:len(inputs)] for r in rows])
            # Aggregate score: minimize '-' (1-norm), maximize '+' (norm)
            scores = []
            for p in preds:
                s = 0
                for i, g in enumerate(goals):
                    norm = (p[i] - df_train[g].min()) / (df_train[g].max() - df_train[g].min() + 1e-9)
                    s += (1 - norm) if g.endswith('-') else norm
                scores.append(s)
            idx = np.argmax(scores)
            get_label(idx, strat_name)
        else: # uncertainty: pick random, but bias towards unlabelled
            idx = np.random.choice(n_rows)
            get_label(idx, strat_name)

    # Evaluate
    results = {}
    # Global normalization for fair comparison
    all_labeled_data = pd.DataFrame([list(labeled[i]) for i in labeled], columns=goals)
    
    for name in strategy_history:
        best_score = -1
        best_idx = -1
        for idx in strategy_history[name]:
            row_goals = labeled[idx]

def justify(header, rows, row):
    """
    Estimates goals for the input row and compares its aggregate performance
    against all known rows. Returns (p, reason).
    """
    inputs = [c for c in header if not (c.endswith('-') or c.endswith('+'))]
    goals = [c for c in header if c.endswith(('-', '+'))]
    
    df_train = pd.DataFrame(rows, columns=header)
    
    # Estimate goals for the target row using KNN
    estimates = {}
    for goal in goals:
        model = KNeighborsRegressor(n_neighbors=5)
        model.fit(df_train[inputs], df_train[goal])
        estimates[goal] = model.predict([row[:len(inputs)]])[0]
    
    # Calculate aggregate score for the target row
    def get_agg(row_data):
        scores = []
        for goal in goals:
            val = row_data[goal] if isinstance(row_data, dict) else row_data[header.index(goal)]
            col_min = df_train[goal].min()
            col_max = df_train[goal].max()
            norm = (val - col_min) / (col_max - col_min + 1e-9)
            scores.append(1 - norm if goal.endswith('-') else norm)
        return sum(scores) / len(scores)
    
    target_agg = get_agg(estimates)
    
    # Calculate aggregate scores for all known rows
    existing_aggs = [get_agg(r) for r in rows]
    
    # Fraction of rows that are worse than this row
    p = sum(1 for agg in existing_aggs if agg < target_agg) / len(existing_aggs)
    
    reason = f"This setup is better than {p:.2f} of all known configurations based on its estimated performance."
    return p, reason

def choose(header, rows, label):
    """
    Finds the best setup for a large, high-risk project by sampling 
    rows using an acquisition strategy that prioritizes exploration 
    and exploitation within a 50-label budget.
    """
    n_rows = len(rows)
    inputs = [c for c in header if not (c.endswith('-') or c.endswith('+'))]
    goals = [c for c in header if c.endswith(('-', '+'))]
    
    labeled = {}
    
    def get_agg(row_goals):
        # Normalize goals: minimize '-' (1-norm), maximize '+' (norm)
        scores = []
        for i, goal in enumerate(goals):
            # Use global min/max if available, else local
            val = row_goals[i]
            scores.append(1 - val if goal.endswith('-') else val)
        return sum(scores) / len(scores)

    # 1. Initial random sample to understand the space
    indices = np.random.choice(n_rows, 10, replace=False).tolist()
    for i in indices:
        labeled[i] = label(i)
        
    # 2. Iterative selection (40 more labels)
    for _ in range(40):
        # Train surrogate model on currently labeled data
        X_train = np.array([rows[i][:len(inputs)] for i in labeled])
        y_train = np.array([labeled[i] for i in labeled])
        
        model = KNeighborsRegressor(n_neighbors=3)
        model.fit(X_train, y_train)
        
        # Predict for all unlabeled rows
        unlabeled = [i for i in range(n_rows) if i not in labeled]
        X_unlabeled = np.array([rows[i][:len(inputs)] for i in unlabeled])
        preds = model.predict(X_unlabeled)
        
        # Acquisition: Pick the one with the best predicted aggregate score
        best_idx_in_unlabeled = np.argmax([get_agg(p) for p in preds])
        idx = unlabeled[best_idx_in_unlabeled]
        
        labeled[idx] = label(idx)
        
    # Return the index of the best row found
    best_idx = max(labeled, key=lambda i: get_agg(labeled[i]))
    return best_idx

def spot(header, rows):
    """
    Identifies rows that are outliers using Isolation Forest.
    Returns the indices of rows, sorted by suspicion (highest anomaly score first).
    """
    df = pd.DataFrame(rows, columns=header)
    
    # IsolationForest requires numeric data. 
    # Select only numeric columns for anomaly detection.
    # We exclude the goal columns as they are outcomes, not the setup itself.
    inputs = [col for col in header if not (col.endswith('-') or col.endswith('+'))]
    numeric_df = df[inputs].select_dtypes(include=[np.number])
    
    # Fit the model to the data
    # Contamination 0.05 assumes roughly 5% of the data might be unusual
    model = IsolationForest(contamination=0.05, random_state=42)
    model.fit(numeric_df)
    
    # decision_function returns the anomaly score of the input samples
    # Lower scores are more anomalous.
    scores = model.decision_function(numeric_df)
    
    # Get indices sorted by score (ascending, so most negative/anomalous first)
    suspicious_indices = np.argsort(scores)
    
    return suspicious_indices.tolist()
