```python
import csv
import pandas as pd
from sklearn.ensemble import RandomForestRegressor
from step01 import remember

def blame(header, rows):
    """
    Identifies input columns that correlate most strongly with the 'Idle-' goal
    in the worst-performing rows.
    """
    df = pd.DataFrame(rows, columns=header)
    
    # Identify input columns (not goals)
    inputs = [col for col in header if not (col.endswith('-') or col.endswith('+'))]
    
    # Identify the 'Idle-' goal
    target = 'Idle-'
    
    # Define "worst" rows: those with the highest 'Idle-' values
    # We look at the bottom 25% of rows based on the 'Idle-' goal
    threshold = df[target].quantile(0.75)
    worst_rows = df[df[target] >= threshold]
    
    # Use a Random Forest to find feature importance for 'Idle-'
    # This identifies which inputs contribute most to the 'Idle-' value
    model = RandomForestRegressor(n_estimators=100, random_state=42)
    model.fit(worst_rows[inputs], worst_rows[target])
    
    # Get feature importances
    importances = model.feature_importances_
    
    # Pair inputs with their importance scores
    feat_imp = sorted(zip(inputs, importances), key=lambda x: x[1], reverse=True)
    
    # Return the top 2 input columns that explain the idle time
    return [feat_imp[0][0], feat_imp[1][0]]

if __name__ == "__main__":
    filename = 'pom3a.n1000.train.csv'
    try:
        with open(filename, 'r') as f:
            reader = csv.reader(f)
            header = next(reader)
            rows = [[float(val) for val in row] for row in reader]
        
        reasons = blame(header, rows)
        
        print("Analysis of why worst setups have high idle time:")
        print(f"The primary drivers for high idle time are: {', '.join(reasons)}.")
        print("These input factors show the strongest correlation with idle resources in our worst-performing setups.")
        
    except FileNotFoundError:
        print(f"Error: {filename} not found.")
        exit(1)
```
