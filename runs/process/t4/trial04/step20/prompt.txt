import csv
import pandas as pd
import numpy as np

def remember(header, rows):
    """
    Identifies rows that perform well across all goal columns.
    Goals ending in '-' are minimized, '+' are maximized.
    """
    df = pd.DataFrame(rows, columns=header)
    
    # Identify goal columns
    goals = [col for col in header if col.endswith(('-', '+'))]
    
    # Normalize goals to [0, 1] range for comparison
    # For '-', lower is better (1 - normalized_value)
    # For '+', higher is better (normalized_value)
    scores = pd.DataFrame(index=df.index)
    for goal in goals:
        col_min = df[goal].min()
        col_max = df[goal].max()
        norm = (df[goal] - col_min) / (col_max - col_min + 1e-9)
        
        if goal.endswith('-'):
            scores[goal] = 1 - norm
        else:
            scores[goal] = norm
            
    # Aggregate score: product of normalized goals (geometric mean approach)
    # This ensures a row is only "unusually well" if it performs well on ALL goals
    final_scores = scores.prod(axis=1)
    
    # Select top 5% (or at least 1)
    num_to_select = max(1, int(len(rows) * 0.05))
    best_indices = final_scores.nlargest(num_to_select).index.tolist()
    
    return sorted(best_indices)

def certify(xs, ys):
    """
    Returns True if the two distributions are statistically similar (null hypothesis 
    not rejected), False if they significantly differ. Uses Welch's t-test.
    """
    # Use a significance level of 0.05
    t_stat, p_value = stats.ttest_ind(xs, ys, equal_var=False)
    return p_value > 0.05

import csv
import pandas as pd
import numpy as np
from step01 import remember

def flow(header, row, seen):
    """
    Implements the Secretary Problem logic:
    1. Observe a sample (37% of total) to establish a baseline of 'good' performance.
    2. After the sample, accept the first row that is better than the best seen so far.
    """
    # Identify goal columns
    goals = [col for col in header if col.endswith(('-', '+'))]
    
    # If this is the first row, we can't compare yet
    if not seen:
        return False
    
    # Calculate a simple score for a row (normalized distance to ideal)
    # Since we don't have the full dataset min/max, we use the 'seen' data as context
    all_rows = seen + [row]
    df = pd.DataFrame(all_rows, columns=header)
    
    scores = pd.DataFrame(index=df.index)
    for goal in goals:
        col_min, col_max = df[goal].min(), df[goal].max()
        # Avoid division by zero
        rng = col_max - col_min if col_max != col_min else 1.0
        norm = (df[goal] - col_min) / rng
        scores[goal] = (1 - norm) if goal.endswith('-') else norm
    
    final_scores = scores.prod(axis=1)
    
    # Secretary Problem threshold: 1/e ≈ 0.37
    sample_size = int(len(all_rows) * 0.37)
    
    if len(seen) < sample_size:
        return False
    
    # Accept if current row is the best seen so far
    return final_scores.iloc[-1] >= final_scores.iloc[:-1].max()

import csv
import pandas as pd
import numpy as np
from sklearn.ensemble import RandomForestRegressor
from step01 import remember

def choose(header, rows, label):
    """
    Finds the best row by using a surrogate model (Random Forest) to predict 
    the performance of rows and querying the most promising ones.
    """
    goals = [col for col in header if col.endswith(('-', '+'))]
    inputs = [col for col in header if col not in goals]
    
    # 1. Initial sample: Label 20 rows to build an initial model
    # Convert indices to a list to allow appending
    indices = list(np.linspace(0, len(rows) - 1, 20, dtype=int))
    
    labeled_data = []
    for i in indices:
        true_goals = label(i)
        row_data = rows[i][:]
        for j, goal in enumerate(goals):
            row_data[header.index(goal)] = true_goals[j]
        labeled_data.append(row_data)
    
    # 2. Active Learning loop: 30 more queries
    for _ in range(30):
        df_train = pd.DataFrame(labeled_data, columns=header)
        
        # Train models for each goal
        models = {}
        for goal in goals:
            model = RandomForestRegressor(n_estimators=20, random_state=42)
            model.fit(df_train[inputs], df_train[goal])
            models[goal] = model
            
        # Predict on all unlabeled rows
        unlabeled_indices = [i for i in range(len(rows)) if i not in indices]
        df_unlabeled = pd.DataFrame([rows[i] for i in unlabeled_indices], columns=header)
        
        scores = pd.DataFrame(index=unlabeled_indices)
        for goal in goals:
            pred = models[goal].predict(df_unlabeled[inputs])
            # Normalize predictions to [0, 1] based on current model range
            rng = pred.max() - pred.min()
            norm = (pred - pred.min()) / (rng + 1e-9)
            scores[goal] = (1 - norm) if goal.endswith('-') else norm
            
        # Pick the row with the highest predicted aggregate score
        best_unlabeled_idx = scores.prod(axis=1).idxmax()
        
        # Label it
        true_goals = label(best_unlabeled_idx)
        row_data = rows[best_unlabeled_idx][:]
        for j, goal in enumerate(goals):
            row_data[header.index(goal)] = true_goals[j]
        labeled_data.append(row_data)
        indices.append(best_unlabeled_idx)
        
    # 3. Return the best row found among all labeled ones
    best_idx_in_labeled = remember(header, labeled_data)[0]
    
    return indices[best_idx_in_labeled]

import csv

def fix(header, row, levers):
    """
    Finds the smallest change to levers to improve Cost- and Idle- goals.
    """
    filename = 'pom3a.n1000.train.csv'
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header_csv = next(reader)
        rows = [[float(val) for val in r] for r in reader]
    
    df = pd.DataFrame(rows, columns=header_csv)
    goals = [c for c in header_csv if c.endswith(('-', '+'))]
    inputs = [c for c in header_csv if c not in goals]
    
    # Train models to predict goals
    models = {}
    for goal in goals:
        m = RandomForestRegressor(n_estimators=20, random_state=42)
        m.fit(df[inputs], df[goal])
        models[goal] = m
        
    def get_score(r_list):
        # r_list is a list of values in header order
        r_df = pd.DataFrame([r_list], columns=header)
        scores = [models[g].predict(r_df[inputs])[0] for g in goals]
        return sum(scores)

    best_change = {}
    best_score = get_score(row)
    
    for col, bounds in levers.items():
        idx = header.index(col)
        low, high = bounds
        # Test a few points in the range
        for val in [low, (low + high) / 2, high]:
            new_row = list(row)
            new_row[idx] = val
            new_score = get_score(new_row)
            
            if new_score < best_score:
                best_score = new_score
                best_change = {col: val}
                
    return best_change

from sklearn.ensemble import IsolationForest

def spot(header, rows):
    """
    Identifies rows that are outliers using Isolation Forest.
    Returns the indices of the rows, sorted by suspicion (most suspicious first).
    """
    df = pd.DataFrame(rows, columns=header)
    
    # IsolationForest works on numeric data. 
    # Select only numeric columns for anomaly detection.
    numeric_df = df.select_dtypes(include=[np.number])
    
    # Isolation Forest: contamination is the proportion of outliers in the data.
    # We set it to 0.05 (5%) as a reasonable heuristic for "mistyped" data.
    model = IsolationForest(contamination=0.05, random_state=42)
    
    # Fit the model to the data
    model.fit(numeric_df)
    
    # decision_function returns the anomaly score. 
    # Lower scores are more anomalous.
    scores = model.decision_function(numeric_df)
    
    # Get indices sorted by score (ascending, so lowest scores first)
    suspicious_indices = np.argsort(scores)
    
    return suspicious_indices.tolist()

import csv
import pandas as pd
import numpy as np
import random
from step01 import remember

def race(header, rows, label):
    """
    Compares three strategies: random, greedy, and a simple surrogate-based search.
    Total budget is 50 labels.
    """
    budget = 50
    labeled = {}

    def get_true_goals(i):
        if i not in labeled:
            if len(labeled) >= budget:
                raise Exception("Budget exceeded")
            labeled[i] = label(i)
        return labeled[i]

    def get_score(i):
        goals = get_true_goals(i)
        # Normalize goals: lower is better for '-', higher for '+'
        # We assume goals are in [0, 1] based on problem description
        score = 0
        goal_cols = [c for c in header if c.endswith(('-', '+'))]
        for j, col in enumerate(goal_cols):
            val = goals[j]
            score += (1 - val) if col.endswith('-') else val
        return score / len(goal_cols)

    # 1. Random Strategy: Pick 16 random rows
    def strategy_random():
        indices = random.sample(range(len(rows)), 16)
        for i in indices: get_true_goals(i)
        return max(indices, key=get_score)

    # 2. Greedy Strategy: Pick first 16 rows
    def strategy_greedy():
        indices = list(range(16))
        for i in indices: get_true_goals(i)
        return max(indices, key=get_score)

    # 3. Surrogate Strategy: Pick 18 random, then 16 best-predicted
    def strategy_surrogate():

def remember(header, rows):

    Finds the 5% of rows most similar to the 'best' performing rows 
    in the dataset, using Euclidean distance on input columns.
    """
    df = pd.DataFrame(rows, columns=header)
    goals = [col for col in header if col.endswith(('-', '+'))]
    inputs = [col for col in header if col not in goals]
    
    # Calculate performance score for all rows
    scores = pd.DataFrame(index=df.index)
    for goal in goals:
        col_min, col_max = df[goal].min(), df[goal].max()
        rng = col_max - col_min if col_max != col_min else 1.0
        norm = (df[goal] - col_min) / rng
        scores[goal] = (1 - norm) if goal.endswith('-') else norm
    
    # Identify the top 5% of rows based on performance
    num_to_select = max(1, int(len(rows) * 0.05))
    best_indices = scores.prod(axis=1).nlargest(num_to_select).index.tolist()
    
    return sorted(best_indices)

import csv
import pandas as pd
import numpy as np

def flow(header, row, seen):
    """
    Warns the customer if a new row is 'going bad'.
    A row is 'going bad' if its performance is significantly worse 
    than the historical average of the 'seen' rows.
    """
    if not seen:
        return False

    goals = [col for col in header if col.endswith(('-', '+'))]
    
    def calculate_score(r):
        # Normalize goals based on the 'seen' data context
        df_seen = pd.DataFrame(seen, columns=header)
        scores = []
        for goal in goals:
            val = r[header.index(goal)]
            # Use min/max from seen data to normalize
            c_min = df_seen[goal].min()
            c_max = df_seen[goal].max()
            rng = c_max - c_min if c_max != c_min else 1.0
            norm = (val - c_min) / rng
            scores.append((1 - norm) if goal.endswith('-') else norm)
        return sum(scores) / len(scores)

    # Calculate score for the new row and the average of seen rows
    current_score = calculate_score(row)
    seen_scores = [calculate_score(r) for r in seen]
    avg_seen_score = sum(seen_scores) / len(seen_scores)
    
    # If the new row is significantly worse than the average (e.g., 2 standard deviations)
    # we flag it as "going bad".
    if len(seen_scores) > 5:
        std_dev = np.std(seen_scores)
        if current_score < (avg_seen_score - 2 * std_dev):
            return True
            
    return False

import csv
import pandas as pd
import numpy as np
from sklearn.ensemble import RandomForestRegressor
from step01 import remember

def choose(header, rows, label):
    """
    Finds the best setup for a large, high-criticality project.
    We filter for rows where 'Size' and 'Criticality' are high,
    then use active learning to find the best among those.
    """
    # 1. Filter for large, high-criticality projects
    # We define "large" and "high-criticality" as being in the top 50%
    df = pd.DataFrame(rows, columns=header)
    size_thresh = df['Size'].median()
    crit_thresh = df['Criticality'].median()
    
    candidates = [i for i, row in enumerate(rows) 
                  if row[header.index('Size')] >= size_thresh 
                  and row[header.index('Criticality')] >= crit_thresh]
    
    # If no candidates, fall back to all rows
    if not candidates:
        candidates = list(range(len(rows)))
        
    # 2. Active Learning: Label up to 50 candidates
    goals = [col for col in header if col.endswith(('-', '+'))]
    inputs = [col for col in header if col not in goals]
    
    labeled_indices = []
    labeled_data = []
    
    # Initial sample: 20 from candidates
    sample_size = min(len(candidates), 20)
    indices_to_label = candidates[:sample_size]
    
    for i in indices_to_label:
        true_goals = label(i)
        row_data = rows[i][:]
        for j, goal in enumerate(goals):
            row_data[header.index(goal)] = true_goals[j]
        labeled_data.append(row_data)
        labeled_indices.append(i)
        
    # 3. Iterative search for remaining budget
    remaining_budget = 50 - len(labeled_indices)
    for _ in range(remaining_budget):
        if len(labeled_indices) >= len(candidates):
            break
            
        df_train = pd.DataFrame(labeled_data, columns=header)
        
        # Train models
        models = {}
        for goal in goals:
            m = RandomForestRegressor(n_estimators=20, random_state=42)
            m.fit(df_train[inputs], df_train[goal])
            models[goal] = m
            
        # Predict on remaining candidates
        unlabeled = [i for i in candidates if i not in labeled_indices]
        df_unlabeled = pd.DataFrame([rows[i] for i in unlabeled], columns=header)
        
        scores = pd.DataFrame(index=unlabeled)
        for goal in goals:
            pred = models[goal].predict(df_unlabeled[inputs])
            norm = (pred - pred.min()) / (pred.max() - pred.min() + 1e-9)
            scores[goal] = (1 - norm) if goal.endswith('-') else norm
            
        best_idx = scores.prod(axis=1).idxmax()
        
        # Label
        true_goals = label(best_idx)
        row_data = rows[best_idx][:]
        for j, goal in enumerate(goals):
            row_data[header.index(goal)] = true_goals[j]
        labeled_data.append(row_data)
        labeled_indices.append(best_idx)
        
    # 4. Return best from labeled
    best_in_labeled = remember(header, labeled_data)[0]
    return labeled_indices[best_in_labeled]

        if isinstance(bounds, tuple):

                new_score = get_idle_score(new_row)
                if new_score < best_score:
                    best_score = new_score

        elif isinstance(bounds, list):
            for val in bounds:
                new_row = list(row)
                new_row[idx] = val
                new_score = get_idle_score(new_row)
                if new_score < best_score:
                    best_score = new_score
                    best_change = {col: val}

import csv
import pandas as pd
import numpy as np
from sklearn.ensemble import IsolationForest

def spot(header, rows):
    """
    Identifies rows that are 'unlike anything we have run' (outliers)
    using Isolation Forest on the input columns.
    """
    df = pd.DataFrame(rows, columns=header)
    
    # Identify goal columns and input columns
    goals = [col for col in header if col.endswith(('-', '+'))]
    inputs = [col for col in header if col not in goals]
    
    # Isolation Forest works on numeric data. 
    # Select only numeric input columns for anomaly detection.
    numeric_inputs = df[inputs].select_dtypes(include=[np.number])
    
    # Isolation Forest: contamination is the proportion of outliers.
    # We use a small value to flag truly unusual setups.
    model = IsolationForest(contamination=0.02, random_state=42)
    
    # Fit the model to the input data
    model.fit(numeric_inputs)
    
    # decision_function returns the anomaly score. 
    # Lower scores are more anomalous (less like what we have seen).
    scores = model.decision_function(numeric_inputs)
    
    # Get indices sorted by score (ascending, so lowest scores/most suspicious first)
    suspicious_indices = np.argsort(scores)
    
    return suspicious_indices.tolist()
