```python
import csv
import pandas as pd
import numpy as np

def remember(header, rows):
    """
    Identifies rows that perform well across all goal columns.
    Goals ending in '-' are minimized, '+' are maximized.
    """
    df = pd.DataFrame(rows, columns=header)
    
    # Identify goal columns
    goals = [col for col in header if col.endswith(('-', '+'))]
    
    # Normalize goals to [0, 1] range for comparison
    # For '-', lower is better (1 - normalized_value)
    # For '+', higher is better (normalized_value)
    scores = pd.DataFrame(index=df.index)
    for goal in goals:
        col_min = df[goal].min()
        col_max = df[goal].max()
        norm = (df[goal] - col_min) / (col_max - col_min + 1e-9)
        
        if goal.endswith('-'):
            scores[goal] = 1 - norm
        else:
            scores[goal] = norm
            
    # Aggregate score: product of normalized goals (geometric mean approach)
    # This ensures a row is only "unusually well" if it performs well on ALL goals
    final_scores = scores.prod(axis=1)
    
    # Select top 5% (or at least 1)
    num_to_select = max(1, int(len(rows) * 0.05))
    best_indices = final_scores.nlargest(num_to_select).index.tolist()
    
    return sorted(best_indices)

if __name__ == "__main__":
    filename = 'pom3a.n1000.train.csv'
    try:
        with open(filename, 'r') as f:
            reader = csv.reader(f)
            header = next(reader)
            rows = []
            for row in reader:
                # Convert numeric columns to floats, keep symbolic as strings
                processed_row = []
                for i, val in enumerate(row):
                    try:
                        processed_row.append(float(val))
                    except ValueError:
                        processed_row.append(val)
                rows.append(processed_row)
        
        best_indices = remember(header, rows)
        
        print(f"Analysis complete. Found {len(best_indices)} top-performing setups.")
        print("Indices of the best rows (0-based):")
        print(best_indices)
        
        # Print a summary of the first best row found
        if best_indices:
            print("\nExample of a top-performing row:")
            print(dict(zip(header, rows[best_indices[0]])))
            
    except FileNotFoundError:
        print(f"Error: {filename} not found.")
        exit(1)
```
