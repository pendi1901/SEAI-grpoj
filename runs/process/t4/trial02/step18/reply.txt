```python
import csv
import pandas as pd
import numpy as np
from sklearn.ensemble import RandomForestRegressor
from step01 import remember

def fix(header, row, levers):
    """
    Finds the best change to non-hiring levers to reduce 'Idle-'.
    We use a Random Forest to predict 'Idle-' and minimize it.
    """
    filename = 'pom3a.n1000.train.csv'
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header_csv = next(reader)
        rows = []
        for r in reader:
            rows.append([float(val) for val in r])
    
    df = pd.DataFrame(rows, columns=header_csv)
    goals = [c for c in header if c.endswith('-') or c.endswith('+')]
    inputs = [c for c in header if c not in goals]
    
    # Train model to predict 'Idle-'
    rf = RandomForestRegressor(n_estimators=50, random_state=42)
    rf.fit(df[inputs], df['Idle-'])
    
    current_inputs = [row[header.index(i)] for i in inputs]
    best_idle = rf.predict([current_inputs])[0]
    best_change = {}
    
    # Evaluate changes for each allowed lever
    for col, bounds in levers.items():
        idx = inputs.index(col)
        # Test 5 points in the allowed range
        for val in np.linspace(bounds[0], bounds[1], 5):
            test_inputs = list(current_inputs)
            test_inputs[idx] = val
            pred_idle = rf.predict([test_inputs])[0]
            if pred_idle < best_idle:
                best_idle = pred_idle
                best_change = {col: val}
                
    return best_change

if __name__ == "__main__":
    filename = 'pom3a.n1000.train.csv'
    try:
        with open(filename, 'r') as f:
            reader = csv.reader(f)
            header = next(reader)
            rows = [[float(val) for val in row] for row in reader]
        
        # Define levers excluding 'Team Size' (hiring)
        levers = {
            'Culture': (0, 1),
            'Criticality': (0, 1),
            'Size': (0, 1),
            'Plan': (0, 1),
            'Initial Known': (0, 1)
        }
        
        # Pick a row with high 'Idle-'
        df = pd.DataFrame(rows, columns=header)
        bad_row = df.nlargest(1, 'Idle-').iloc[0].tolist()
        
        recommendation = fix(header, bad_row, levers)
        
        print("Customer Inquiry: I am not allowed to hire. What else can I change to cut idle time?")
        print(f"Recommendation: {recommendation}")
        print("Analysis: By adjusting these non-hiring parameters, the surrogate model predicts a reduction in 'Idle-'.")
        
    except FileNotFoundError:
        print(f"Error: {filename} not found.")
        exit(1)
```
