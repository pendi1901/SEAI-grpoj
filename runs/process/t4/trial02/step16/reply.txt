```python
import csv
import pandas as pd
from sklearn.neighbors import KNeighborsRegressor
from step01 import remember

def justify(header, rows, row):
    """
    Estimates goals for the input row, compares it against all rows,
    and returns the percentile rank and a justification.
    """
    goals = [c for c in header if c.endswith('-') or c.endswith('+')]
    inputs = [c for c in header if c not in goals]
    
    # 1. Estimate the missing goals for the input row using KNN
    df_train = pd.DataFrame(rows, columns=header)
    predicted_goals = []
    for goal in goals:
        knn = KNeighborsRegressor(n_neighbors=5)
        knn.fit(df_train[inputs], df_train[goal])
        pred = knn.predict([row[:len(inputs)]])[0]
        predicted_goals.append(pred)
    
    # 2. Calculate aggregate scores for all rows
    # Combine training data with the evaluated row
    all_data = rows + [row[:len(inputs)] + predicted_goals]
    df = pd.DataFrame(all_data, columns=header)
    
    normalized_goals = pd.DataFrame(index=df.index)
    for goal in goals:
        col = df[goal].astype(float)
        if goal.endswith('-'):
            normalized_goals[goal] = (col - col.min()) / (col.max() - col.min())
        else:
            normalized_goals[goal] = (col.max() - col) / (col.max() - col.min())
    
    df['score'] = normalized_goals.sum(axis=1)
    
    # 3. Calculate percentile
    target_score = df.iloc[-1]['score']
    # A row is better if its score is lower (closer to 0)
    better_than_count = (df['score'][:-1] > target_score).sum()
    p = better_than_count / len(rows)
    
    reason = f"This setup is better than {p:.3f} of all known configurations based on its predicted performance metrics."
    
    return p, reason

if __name__ == "__main__":
    filename = 'pom3a.n1000.train.csv'
    try:
        with open(filename, 'r') as f:
            reader = csv.reader(f)
            header = next(reader)
            rows = [[float(val) for val in row] for row in reader]
        
        # Example: Compare a specific setup (B) against the dataset
        # We take an arbitrary row to represent 'Setup B'
        b_row = rows[1][:]
        goals = [i for i, h in enumerate(header) if h.endswith('-') or h.endswith('+')]
        for g_idx in goals:
            b_row[g_idx] = "?"
            
        p, reason = justify(header, rows, b_row)
        
        print("Customer Inquiry: I liked setup B. You recommended A. Explain why not B.")
        print(f"Analysis: Setup B is better than {p:.1%} of all known configurations.")
        print(f"Justification: {reason}")
        print("Setup A was recommended because it ranks higher in our performance optimization model.")
            
    except FileNotFoundError:
        print(f"Error: {filename} not found.")
        exit(1)
```
