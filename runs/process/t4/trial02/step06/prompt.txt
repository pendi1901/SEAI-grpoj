import csv
import pandas as pd

def remember(header, rows):
    """
    Identifies rows that perform well across all goals.
    Goals ending in '-' are minimized; goals ending in '+' are maximized.
    We use a simple Pareto-like approach: normalize goals and find rows 
    with the lowest sum of normalized costs (or highest normalized gains).
    """
    df = pd.DataFrame(rows, columns=header)
    
    # Identify goal columns
    goals = [c for c in header if c.endswith('-') or c.endswith('+')]
    
    # Normalize goals to [0, 1] range
    normalized_goals = pd.DataFrame(index=df.index)
    for goal in goals:
        col = df[goal].astype(float)
        if goal.endswith('-'):
            # Minimize: lower is better (0 is best)
            normalized_goals[goal] = (col - col.min()) / (col.max() - col.min())
        else:
            # Maximize: higher is better (0 is best after inversion)
            normalized_goals[goal] = (col.max() - col) / (col.max() - col.min())
            
    # Calculate aggregate score (sum of normalized goal distances)
    # Lower aggregate score is better
    df['score'] = normalized_goals.sum(axis=1)
    
    # Select top 5% (or at least 1)
    n_select = max(1, len(df) // 20)
    best_indices = df.nsmallest(n_select, 'score').index.tolist()
    
    return best_indices

import csv
import pandas as pd
from scipy import stats

def certify(xs, ys):
    """
    Returns True if the difference between two distributions is statistically 
    insignificant (p > 0.05), False otherwise.
    """
    # Using Welch's t-test to compare means of two independent samples
    t_stat, p_value = stats.ttest_ind(xs, ys, equal_var=False)
    return p_value > 0.05

def analyze_team_size_impact(header, rows):
    """
    Compares 'Cost-' for small vs large teams.
    Small teams: Team Size < median. Large teams: Team Size >= median.
    """
    df = pd.DataFrame(rows, columns=header)
    
    median_team_size = df['Team Size'].median()
    
    small_teams = df[df['Team Size'] < median_team_size]['Cost-']
    large_teams = df[df['Team Size'] >= median_team_size]['Cost-']
    
    is_same = certify(small_teams.tolist(), large_teams.tolist())
    
    return is_same, small_teams.mean(), large_teams.mean()

import csv
import pandas as pd
from sklearn.ensemble import RandomForestRegressor
from step01 import remember

def blame(header, rows):
    """
    Identifies input columns that best explain the 'Idle-' goal for the worst setups.
    We define 'worst' as the bottom 10% of rows based on the aggregate score.
    We then use a Random Forest to find feature importances for 'Idle-'.
    """
    df = pd.DataFrame(rows, columns=header)
    
    # Identify goal columns and input columns
    goals = [c for c in header if c.endswith('-') or c.endswith('+')]
    inputs = [c for c in header if c not in goals]
    
    # Calculate aggregate score to find the worst rows
    # We reuse remember logic: lower score is better, so higher score is worse
    # We'll calculate the score manually to get the bottom 10%
    normalized_goals = pd.DataFrame(index=df.index)
    for goal in goals:
        col = df[goal].astype(float)
        if goal.endswith('-'):
            normalized_goals[goal] = (col - col.min()) / (col.max() - col.min())
        else:
            normalized_goals[goal] = (col.max() - col) / (col.max() - col.min())
    
    df['score'] = normalized_goals.sum(axis=1)
    
    # Define worst rows as the bottom 10%
    n_worst = max(1, len(df) // 10)
    worst_df = df.nlargest(n_worst, 'score')
    
    # Train a regressor to see what drives 'Idle-' in these worst setups
    rf = RandomForestRegressor(n_estimators=50, random_state=42)
    rf.fit(worst_df[inputs], worst_df['Idle-'])
    
    # Get feature importances
    importances = pd.Series(rf.feature_importances_, index=inputs)
    
    # Return the top 3 inputs that explain the variance in 'Idle-'
    top_features = importances.nlargest(3).index.tolist()
    
    return top_features
