import csv
import pandas as pd

def remember(header, rows):
    """
    Identifies rows that perform well across all goals.
    Goals ending in '-' are minimized; goals ending in '+' are maximized.
    We use a simple Pareto-like approach: normalize goals and find rows 
    with the lowest sum of normalized costs (or highest normalized gains).
    """
    df = pd.DataFrame(rows, columns=header)
    
    # Identify goal columns
    goals = [c for c in header if c.endswith('-') or c.endswith('+')]
    
    # Normalize goals to [0, 1] range
    normalized_goals = pd.DataFrame(index=df.index)
    for goal in goals:
        col = df[goal].astype(float)
        if goal.endswith('-'):
            # Minimize: lower is better (0 is best)
            normalized_goals[goal] = (col - col.min()) / (col.max() - col.min())
        else:
            # Maximize: higher is better (0 is best after inversion)
            normalized_goals[goal] = (col.max() - col) / (col.max() - col.min())
            
    # Calculate aggregate score (sum of normalized goal distances)
    # Lower aggregate score is better
    df['score'] = normalized_goals.sum(axis=1)
    
    # Select top 5% (or at least 1)
    n_select = max(1, len(df) // 20)
    best_indices = df.nsmallest(n_select, 'score').index.tolist()
    
    return best_indices

def certify(xs, ys):
    """
    Returns True if the difference between two distributions is statistically 
    insignificant (p > 0.05), False otherwise.
    """
    # Using Welch's t-test to compare means of two independent samples
    t_stat, p_value = stats.ttest_ind(xs, ys, equal_var=False)
    return p_value > 0.05

def analyze_team_size_impact(header, rows):
    """
    Compares 'Cost-' for small vs large teams.
    Small teams: Team Size < median. Large teams: Team Size >= median.
    """
    df = pd.DataFrame(rows, columns=header)
    
    median_team_size = df['Team Size'].median()
    
    small_teams = df[df['Team Size'] < median_team_size]['Cost-']
    large_teams = df[df['Team Size'] >= median_team_size]['Cost-']
    
    is_same = certify(small_teams.tolist(), large_teams.tolist())
    
    return is_same, small_teams.mean(), large_teams.mean()

def choose(header, rows, label):
    """
    Finds the best row by using an active learning approach:
    1. Use a surrogate model (Random Forest) to predict goals.
    2. Use Expected Improvement or simply pick the best predicted row.
    3. Label the best candidates until 50 are used.
    """
    goals = [c for c in header if c.endswith('-') or c.endswith('+')]
    inputs = [c for c in header if c not in goals]
    
    # Track which rows have been labeled
    labeled_indices = {}
    
    # Initial strategy: sample 10 rows randomly to seed the model
    initial_samples = np.random.choice(len(rows), min(10, len(rows)), replace=False)
    for idx in initial_samples:
        labeled_indices[idx] = label(idx)
        
    # Iteratively improve
    for _ in range(50 - len(labeled_indices)):
        # Train model on currently labeled data
        train_x = [rows[i][:-len(goals)] for i in labeled_indices]
        train_y = [labeled_indices[i] for i in labeled_indices]
        
        # Predict goals for all unlabeled rows
        unlabeled_indices = [i for i in range(len(rows)) if i not in labeled_indices]
        if not unlabeled_indices:
            break
            
        # Use RF to predict each goal
        preds = []
        for g_idx in range(len(goals)):
            rf = RandomForestRegressor(n_estimators=20)
            rf.fit(train_x, [y[g_idx] for y in train_y])
            preds.append(rf.predict([rows[i][:-len(goals)] for i in unlabeled_indices]))
        
        # Evaluate predicted performance
        # Create a temporary list of rows with predicted goals
        temp_rows = []
        for i, u_idx in enumerate(unlabeled_indices):
            row_vals = rows[u_idx][:-len(goals)] + [preds[g_idx][i] for g_idx in range(len(goals))]
            temp_rows.append(row_vals)
            
        # Use remember logic to find the best predicted row
        best_in_temp = remember(header, temp_rows)[0]
        best_idx = unlabeled_indices[best_in_temp]
        
        # Label it
        labeled_indices[best_idx] = label(best_idx)
        
    # Final selection: find the best among all labeled rows
    final_data = []
    indices = []
    for idx, vals in labeled_indices.items():
        final_data.append(rows[idx][:-len(goals)] + vals)
        indices.append(idx)
        
    best_local_idx = remember(header, final_data)[0]
    return indices[best_local_idx]

def fix(header, row, levers):
    """
    Finds the smallest change to levers that improves the row's performance.
    We use a Random Forest to predict the impact of changing levers on the goals.
    """
    # Load data to train a surrogate model
    filename = 'pom3a.n1000.train.csv'
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header_csv = next(reader)
        rows = [[float(val) for val in row] for row in reader]
    
    df = pd.DataFrame(rows, columns=header_csv)
    goals = [c for c in header if c.endswith('-') or c.endswith('+')]
    inputs = [c for c in header if c not in goals]
    
    # Train models for each goal
    models = {}
    for goal in goals:
        rf = RandomForestRegressor(n_estimators=20, random_state=42)
        rf.fit(df[inputs], df[goal])
        models[goal] = rf
        
    def predict_score(r):
        # Predict goals and calculate aggregate score (lower is better)
        scores = []
        for goal in goals:
            pred = models[goal].predict([r])[0]
            # Normalize based on global min/max
            col = df[goal]
            norm = (pred - col.min()) / (col.max() - col.min()) if goal.endswith('-') else (col.max() - pred) / (col.max() - col.min())
            scores.append(norm)
        return sum(scores)

    # Current row inputs
    current_inputs = [row[header.index(i)] for i in inputs]
    best_score = predict_score(current_inputs)
    best_change = {}
    
    # Try changing each lever
    for col, bounds in levers.items():
        idx = inputs.index(col)
        # Try a few points in the range
        for val in np.linspace(bounds[0], bounds[1], 5):
            test_inputs = list(current_inputs)
            test_inputs[idx] = val
            new_score = predict_score(test_inputs)
            if new_score < best_score:
                best_score = new_score
                best_change = {col: val}
                
    return best_change

import csv

from sklearn.ensemble import IsolationForest

def spot(header, rows):
    """
    Identifies rows that are outliers using Isolation Forest.
    Returns the indices of the rows, sorted by suspicion (highest anomaly score first).
    """
    df = pd.DataFrame(rows, columns=header)
    
    # Isolation Forest works on numeric data.
    df_numeric = df.copy()
    for col in df_numeric.columns:
        # If column is symbolic, convert to numeric codes
        if df_numeric[col].dtype == 'object':
            df_numeric[col] = pd.factorize(df_numeric[col])[0]
        else:
            # Replace "?" with mean for anomaly detection
            mask = df_numeric[col] == "?"
            if mask.any():
                mean_val = df_numeric.loc[~mask, col].mean()
                df_numeric.loc[mask, col] = mean_val
                df_numeric[col] = df_numeric[col].astype(float)

    # Fit Isolation Forest
    clf = IsolationForest(contamination=0.05, random_state=42)
    clf.fit(df_numeric)
    scores = clf.decision_function(df_numeric)
    
    # Lower scores are more anomalous. 
    # We return indices sorted by score ascending (most suspicious first).
    indices = sorted(range(len(scores)), key=lambda i: scores[i])
    
    return indices
