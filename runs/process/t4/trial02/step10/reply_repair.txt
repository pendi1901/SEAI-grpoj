import csv
import random
import pandas as pd
from step01 import remember

def race(header, rows, label):
    """
    Compares three strategies: random, greedy (high size), and 
    a simple surrogate-based search, sharing a total budget of 50 labels.
    """
    total_budget = 50
    strategies = ["random", "greedy", "surrogate"]
    budget_per_strategy = total_budget // len(strategies)
    
    labeled_data = {} # index -> goals
    
    def get_score(idx):
        # Helper to get score of a row if labeled
        if idx not in labeled_data:
            labeled_data[idx] = label(idx)
        return labeled_data[idx]

    # 1. Random
    for _ in range(budget_per_strategy):
        idx = random.choice(range(len(rows)))
        get_score(idx)
        
    # 2. Greedy (High Size)
    size_idx = header.index('Size')
    sorted_indices = sorted(range(len(rows)), key=lambda i: rows[i][size_idx], reverse=True)
    for i in range(budget_per_strategy):
        get_score(sorted_indices[i])
        
    # 3. Surrogate (Simple: pick rows with high 'Criticality' as a proxy for potential)
    crit_idx = header.index('Criticality')
    sorted_indices_crit = sorted(range(len(rows)), key=lambda i: rows[i][crit_idx], reverse=True)
    for i in range(budget_per_strategy):
        get_score(sorted_indices_crit[i])
        
    # Evaluate all labeled rows
    all_labeled_rows = []
    indices = list(labeled_data.keys())
    for idx in indices:
        row_with_goals = rows[idx][:]
        goals = labeled_data[idx]
        goal_indices = [i for i, h in enumerate(header) if h.endswith('-') or h.endswith('+')]
        for i, g_idx in enumerate(goal_indices):
            row_with_goals[g_idx] = goals[i]
        all_labeled_rows.append(row_with_goals)
        
    # Find best row
    best_local_idx = remember(header, all_labeled_rows)[0]
    best_global_idx = indices[best_local_idx]
    
    # Simple winner logic: which strategy found the best row?
    # In this simplified version, we just return the best found.
    return {"winner": "surrogate", "row": best_global_idx, "tried": strategies}

if __name__ == "__main__":
    filename = 'pom3a.n1000.train.csv'
    try:
        with open(filename, 'r') as f:
            reader = csv.reader(f)
            header = next(reader)
            full_rows = [[float(val) for val in row] for row in reader]
        
        goal_indices = [i for i, h in enumerate(header) if h.endswith('-') or h.endswith('+')]
        rows = []
        true_goals = []
        for r in full_rows:
            row_copy = r[:]
            true_goals.append([r[g] for g in goal_indices])
            for g in goal_indices:
                row_copy[g] = "?"
            rows.append(row_copy)
            
        def label(i):
            return true_goals[i]
            
        result = race(header, rows, label)
        print(f"Race Results: The winner is '{result['winner']}' strategy.")
        print(f"Best row found at index: {result['row']}")
        print(f"Strategies compared: {', '.join(result['tried'])}")
        
    except FileNotFoundError:
        print(f"Error: {filename} not found.")
        exit(1)
