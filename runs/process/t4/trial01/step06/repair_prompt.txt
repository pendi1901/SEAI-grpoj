import csv

def remember(header, rows):
    """
    Identifies rows that perform well across all goal columns.
    Goals ending in '-' are minimized, goals ending in '+' are maximized.
    Uses a simple Pareto-dominance heuristic: we look for rows that are 
    not dominated by others or have high aggregate normalized scores.
    """
    # Identify goal columns and their directions
    goal_indices = []
    goal_directions = []
    for i, name in enumerate(header):
        if name.endswith('-'):
            goal_indices.append(i)
            goal_directions.append(-1)  # Minimize
        elif name.endswith('+'):
            goal_indices.append(i)
            goal_directions.append(1)   # Maximize

    # Normalize data to [0, 1]
    n = len(rows)
    m = len(goal_indices)
    
    # Extract columns for normalization
    cols = [[float(row[idx]) for row in rows] for idx in goal_indices]
    mins = [min(c) for c in cols]
    maxs = [max(c) for c in cols]
    
    # Calculate normalized scores
    # For minimize (-1): (max - val) / (max - min)
    # For maximize (1): (val - min) / (max - min)
    normalized = []
    for r in range(n):
        row_scores = []
        for i in range(m):
            val = float(rows[r][goal_indices[i]])
            denom = maxs[i] - mins[i]
            if denom == 0:
                row_scores.append(0)
            else:
                norm = (val - mins[i]) / denom
                row_scores.append(1 - norm if goal_directions[i] == -1 else norm)
        normalized.append(row_scores)

    # Calculate aggregate score (min of normalized goals ensures balance)
    # We want rows that are good on all objectives simultaneously
    agg_scores = [min(row_scores) for row_scores in normalized]
    
    # Select top 5% of rows
    num_to_select = max(1, n // 20)
    
    # Get indices of top scores
    indexed_scores = list(enumerate(agg_scores))
    indexed_scores.sort(key=lambda x: x[1], reverse=True)
    
    best_indices = [x[0] for x in indexed_scores[:num_to_select]]
    return sorted(best_indices)

def guess(header, train, test):
    """
    Estimates the first goal column for test rows using K-Nearest Neighbors
    based on the input columns (non-goal columns).
    """
    # Identify goal indices
    goal_indices = [i for i, name in enumerate(header) if name.endswith(('-', '+'))]
    first_goal_idx = goal_indices[0]
    
    # Identify input columns (all columns that are not goals)
    input_indices = [i for i in range(len(header)) if i not in goal_indices]
    
    # Prepare training data
    X_train = [[row[i] for i in input_indices] for row in train]
    y_train = [row[first_goal_idx] for row in train]
    
    # Prepare test data
    X_test = [[row[i] for i in input_indices] for row in test]
    
    # Use KNN to estimate the goal value
    # k=5 is a standard heuristic for small datasets
    model = KNeighborsRegressor(n_neighbors=5)
    model.fit(X_train, y_train)
    
    return model.predict(X_test).tolist()

import csv
from scipy import stats

def certify(xs, ys):
    """
    Returns True if the two distributions are statistically similar (null hypothesis 
    not rejected), False if they differ significantly. Uses a t-test.
    """
    # A p-value > 0.05 suggests we cannot reject the null hypothesis (they are the same)
    t_stat, p_value = stats.ttest_ind(xs, ys, equal_var=False)
    return p_value > 0.05

def run_analysis():
    filename = 'pom3a.n1000.train.csv'
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        rows = []
        for row in reader:
            rows.append([float(x) for x in row])

    # Identify indices
    team_size_idx = header.index('Team Size')
    cost_idx = header.index('Cost-')

    # Define small vs big team (median split)
    team_sizes = [row[team_size_idx] for row in rows]
    median_size = sum(team_sizes) / len(team_sizes)

    small_team_costs = [row[cost_idx] for row in rows if row[team_size_idx] <= median_size]
    big_team_costs = [row[cost_idx] for row in rows if row[team_size_idx] > median_size]

    is_same = certify(small_team_costs, big_team_costs)

import csv
from step01 import remember

def blame(header, rows):
    """
    Identifies input columns that correlate most strongly with the 'Idle-' goal.
    'Idle-' is a goal to minimize, so 'worst' rows are those with high Idle- values.
    We compare the average of input features in the worst 25% of rows 
    against the average of the best 25% of rows.
    """
    goal_idx = header.index('Idle-')
    input_indices = [i for i, name in enumerate(header) if not (name.endswith('-') or name.endswith('+'))]
    
    # Sort rows by Idle- (ascending: best to worst)
    sorted_rows = sorted(rows, key=lambda r: r[goal_idx])
    
    n = len(sorted_rows)
    quarter = n // 4
    best_rows = sorted_rows[:quarter]
    worst_rows = sorted_rows[-quarter:]
    
    # Calculate the difference in means for each input column
    diffs = []
    for i in input_indices:
        best_avg = sum(r[i] for r in best_rows) / quarter
        worst_avg = sum(r[i] for r in worst_rows) / quarter
        # We want to know what makes Idle- high (worst). 
        # If worst_avg > best_avg, this feature contributes to idleness.
        diffs.append((header[i], worst_avg - best_avg))
    
    # Sort by the largest positive difference
    diffs.sort(key=lambda x: x[1], reverse=True)
    
    # Return the top 2 columns that explain the idleness
    return [d[0] for d in diffs[:2]]
