def blame(header, rows):
    goal_cols = [h for h in header if h.endswith('-')]

    # Per-row score: low is good for -cols, high is good for +cols
    scores = []
    for row in rows:
        s = 0.0
        for col in goal_cols:
            try:
                val = float(row[header.index(col)])
            except (ValueError, TypeError):
                continue
            if col.endswith('-'):
                s -= val
            else:
                s += val
        scores.append(s)

    # Worst and best rows by score
    n_worst = max(1, int(0.1 * len(rows)))
    worst_idx = sorted(range(len(scores)), key=lambda i: scores[i])[:n_worst]
    best_idx = sorted(range(len(scores)), key=lambda i: -scores[i])[:max(1, int(0.1 * len(rows)))]

    # For each input column, compare mean in worst vs best rows
    input_cols = [c for c in header if not c.endswith('-')]
    blame_data = []

    for col in input_cols:
        worst_vals = []
        best_vals = []
        for i in worst_idx:
            try:
                worst_vals.append(float(rows[i][header.index(col)]))
            except (ValueError, TypeError):
                pass
        for i in best_idx:
            try:
                best_vals.append(float(rows[i][header.index(col)]))
            except (ValueError, TypeError):
                pass

        if len(worst_vals) < 3 or len(best_vals) < 3:
            continue

        mean_w = sum(worst_vals) / len(worst_vals)
        mean_b = sum(best_vals) / len(best_vals)
        blame_data.append((abs(mean_w - mean_b), col))

    # Sort by magnitude of difference — biggest gap first
    blame_data.sort(key=lambda x: -x[0])

    return [col for _, col in blame_data[:3]]


if __name__ == "__main__":
    import pandas as pd

    df = pd.read_csv("pom3a.train.csv")
    header = list(df.columns)
    rows = []
    for _, row in df.iterrows():
        r = []
        for col in header:
            val = row[col]
            if pd.isna(val):
                r.append("?")
            else:
                r.append(float(val))
        rows.append(r)

    blamed = blame(header, rows)
    print(f"Blame: {', '.join(blamed)}")
