import csv

def remember(header, rows):
    """
    Identifies rows that perform well across all goal columns.
    Goals ending in '-' are minimized, goals ending in '+' are maximized.
    Uses a simple Pareto-dominance heuristic: we look for rows that are 
    not dominated by others or have high aggregate normalized scores.
    """
    # Identify goal columns and their directions
    goal_indices = []
    goal_directions = []
    for i, name in enumerate(header):
        if name.endswith('-'):
            goal_indices.append(i)
            goal_directions.append(-1)  # Minimize
        elif name.endswith('+'):
            goal_indices.append(i)
            goal_directions.append(1)   # Maximize

    # Normalize data to [0, 1]
    n = len(rows)
    m = len(goal_indices)
    
    # Extract columns for normalization
    cols = [[float(row[idx]) for row in rows] for idx in goal_indices]
    mins = [min(c) for c in cols]
    maxs = [max(c) for c in cols]
    
    # Calculate normalized scores
    # For minimize (-1): (max - val) / (max - min)
    # For maximize (1): (val - min) / (max - min)
    normalized = []
    for r in range(n):
        row_scores = []
        for i in range(m):
            val = float(rows[r][goal_indices[i]])
            denom = maxs[i] - mins[i]
            if denom == 0:
                row_scores.append(0)
            else:
                norm = (val - mins[i]) / denom
                row_scores.append(1 - norm if goal_directions[i] == -1 else norm)
        normalized.append(row_scores)

    # Calculate aggregate score (min of normalized goals ensures balance)
    # We want rows that are good on all objectives simultaneously
    agg_scores = [min(row_scores) for row_scores in normalized]
    
    # Select top 5% of rows
    num_to_select = max(1, n // 20)
    
    # Get indices of top scores
    indexed_scores = list(enumerate(agg_scores))
    indexed_scores.sort(key=lambda x: x[1], reverse=True)
    
    best_indices = [x[0] for x in indexed_scores[:num_to_select]]
    return sorted(best_indices)

import csv
from scipy import stats

def certify(xs, ys):
    """
    Returns True if the two distributions are statistically similar (null hypothesis 
    not rejected), False if they differ significantly. Uses a t-test.
    """
    # A p-value > 0.05 suggests we cannot reject the null hypothesis (they are the same)
    t_stat, p_value = stats.ttest_ind(xs, ys, equal_var=False)
    return p_value > 0.05

def blame(header, rows):
    """
    Identifies input columns that correlate most strongly with the 'Idle-' goal.
    'Idle-' is a goal to minimize, so 'worst' rows are those with high Idle- values.
    We compare the average of input features in the worst 25% of rows 
    against the average of the best 25% of rows.
    """
    goal_idx = header.index('Idle-')
    input_indices = [i for i, name in enumerate(header) if not (name.endswith('-') or name.endswith('+'))]
    
    # Sort rows by Idle- (ascending: best to worst)
    sorted_rows = sorted(rows, key=lambda r: r[goal_idx])
    
    n = len(sorted_rows)
    quarter = n // 4
    best_rows = sorted_rows[:quarter]
    worst_rows = sorted_rows[-quarter:]
    
    # Calculate the difference in means for each input column
    diffs = []
    for i in input_indices:
        best_avg = sum(r[i] for r in best_rows) / quarter
        worst_avg = sum(r[i] for r in worst_rows) / quarter
        # We want to know what makes Idle- high (worst). 
        # If worst_avg > best_avg, this feature contributes to idleness.
        diffs.append((header[i], worst_avg - best_avg))
    
    # Sort by the largest positive difference
    diffs.sort(key=lambda x: x[1], reverse=True)
    
    # Return the top 2 columns that explain the idleness
    return [d[0] for d in diffs[:2]]

def spot(header, rows):
    """
    Identifies rows that are outliers using IsolationForest.
    Returns the indices of the rows, sorted by suspicion (highest anomaly score first).
    """
    # Use only numeric columns for anomaly detection
    numeric_indices = [i for i, val in enumerate(rows[0]) if isinstance(val, (int, float))]
    X = [[row[i] for i in numeric_indices] for row in rows]
    
    # IsolationForest assigns lower scores to anomalies
    clf = IsolationForest(contamination=0.05, random_state=42)
    scores = clf.fit_predict(X)
    anomaly_scores = clf.decision_function(X)
    
    # Combine index with score, sort by score (lowest first = most suspicious)
    indexed_scores = list(enumerate(anomaly_scores))
    indexed_scores.sort(key=lambda x: x[1])
    
    # Return indices of rows identified as anomalies
    return [idx for idx, score in indexed_scores if score < 0]

import csv
import random
from step01 import remember
from step07 import choose

def race(header, rows, label):
    """
    Compares three strategies: random, greedy (top-k), and a simple heuristic.
    Budget is 50 labels total.
    """
    budget = 50
    state = {'labeled': {}}

    def get_label(i):
        if i not in state['labeled']:
            if len(state['labeled']) >= budget:
                raise Exception("Budget exceeded")
            state['labeled'][i] = label(i)
        return state['labeled'][i]

    def get_score(row_idx):
        # Normalize goals for comparison
        goals = get_label(row_idx)
        goal_indices = [i for i, name in enumerate(header) if name.endswith(('-', '+'))]
        goal_directions = [-1 if header[i].endswith('-') else 1 for i in goal_indices]
        
        # We need global min/max to normalize, but we only have 50 labels.
        # We use the 50 sampled rows as the population for normalization.
        sampled_indices = list(state['labeled'].keys())
        all_goals = [state['labeled'][idx] for idx in sampled_indices]
        
        scores = []
        for i, g_idx in enumerate(goal_indices):
            col_vals = [g[i] for g in all_goals]
            mn, mx = min(col_vals), max(col_vals)
            val = goals[i]
            norm = (val - mn) / (mx - mn) if mx > mn else 0
            scores.append(1 - norm if goal_directions[i] == -1 else norm)
        return min(scores)

    # Strategies
    def strategy_random():
        idx = random.choice(range(len(rows)))
        get_label(idx)
        return idx

    def strategy_greedy():
        # Sample 10, pick best, repeat
        best_idx = -1
        best_score = -1
        for _ in range(5):
            sample = random.sample(range(len(rows)), 10)
            for idx in sample:
                s = get_score(idx)
                if s > best_score:
                    best_score = s
                    best_idx = idx
        return best_idx

    def strategy_heuristic():
        # Pick rows with high 'Size' or 'Team Size' (common drivers)
        candidates = [i for i, r in enumerate(rows) if r[header.index('Size')] > 0.5]
        idx = random.choice(candidates) if candidates else random.choice(range(len(rows)))

def remember(header, rows):
    """
    Finds the most similar past configurations to the proposed setup (the last row).
    Returns the indices of the most similar rows.
    """
    # Identify input columns (non-goal columns)
    goal_indices = [i for i, name in enumerate(header) if name.endswith(('-', '+'))]
    input_indices = [i for i in range(len(header)) if i not in goal_indices]
    
    # The last row is the proposal
    proposal = rows[-1]
    past_rows = rows[:-1]
    
    # Prepare data for similarity search
    X = [[row[i] for i in input_indices] for row in past_rows]
    query = [[proposal[i] for i in input_indices]]
    
    # Find 5 most similar rows
    nn = NearestNeighbors(n_neighbors=5)
    nn.fit(X)
    distances, indices = nn.kneighbors(query)
    
    return indices[0].tolist()
