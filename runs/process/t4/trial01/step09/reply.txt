import csv
import numpy as np


def remember(header, rows):
    goal_cols = [h for h in header if h.endswith('-')]
    n = len(rows)

    def safe_float(val):
        try:
            return float(val)
        except (ValueError, TypeError):
            return None

    col_vals = {}
    for col in goal_cols:
        vals = []
        for row in rows:
            f = safe_float(row[header.index(col)])
            if f is not None:
                vals.append(f)
        col_vals[col] = np.array(vals, dtype=float)

    normalized = {}
    for col in goal_cols:
        arr = col_vals[col]
        mn, mx = float(arr.min()), float(arr.max())
        if mx == mn or len(arr) == 0:
            norm = np.zeros(n)
        else:
            norm = (arr - mn) / (mx - mn)
        normalized[col] = norm

    total_scores = []
    for i in range(n):
        s = 0.0
        for col in goal_cols:
            if normalized[col][i] is not None:
                s += float(normalized[col][i])
        total_scores.append(s)

    max_ret = max(1, int(0.05 * n))
    sorted_indices = np.argsort(total_scores)[:max_ret]
    return sorted_indices.tolist()


def spot(header, rows):
    """Return row indices whose input values look mistyped or made up."""
    goal_cols = [h for h in header if h.endswith('-')]

    def safe_float(val):
        try:
            return float(val)
        except (ValueError, TypeError):
            return None

    # Identify numeric columns: uppercase start, not a goal column
    input_cols = []
    for col in header:
        if col[0].isupper() and not col.endswith('-'):
            input_cols.append(col)

    n = len(rows)
    suspicious = [set() for _ in range(n)]  # sets of outlier columns per row

    for col in input_cols:
        idx = header.index(col)
        vals = []
        valid_rows = []
        for i, row in enumerate(rows):
            f = safe_float(row[idx])
            if f is not None:
                vals.append(f)
                valid_rows.append(i)

        if len(vals) < 3:
            continue

        arr = np.array(vals)
        mean_val = float(arr.mean())
        std_val = float(arr.std())

        # Flag rows where value is > 2.5 std devs from mean (outlier detection)
        for i, row in enumerate(rows):
            f = safe_float(row[idx])
            if f is not None:
                z = abs(f - mean_val) / max(std_val, 1e-9)
                if z > 2.5:
                    suspicious[i].add(col)

    # Score each row by number of outlier columns (more outliers = more suspicious)
    scores = [len(suspicious[i]) for i in range(n)]

    # Return indices sorted by suspicion, most suspicious first
    return sorted(range(n), key=lambda i: -scores[i])


if __name__ == "__main__":
    import pandas as pd

    header = list(df.columns)

    for _, row in df.iterrows():

                r.append("?")
            else:
                r.append(float(val))
        rows.append(r)

    suspicious = spot(header, rows)
    print(f"Suspicious rows (most to least): {suspicious[:10]}")
