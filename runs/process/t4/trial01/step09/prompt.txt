import csv
import numpy as np


def remember(header, rows):
    goal_cols = [h for h in header if h.endswith('-')]
    n = len(rows)

    def safe_float(val):
        try:
            return float(val)
        except (ValueError, TypeError):
            return None

    col_vals = {}
    for col in goal_cols:
        vals = []
        for row in rows:
            f = safe_float(row[header.index(col)])
            if f is not None:
                vals.append(f)
        col_vals[col] = np.array(vals, dtype=float)

    normalized = {}
    for col in goal_cols:
        arr = col_vals[col]
        mn, mx = float(arr.min()), float(arr.max())
        if mx == mn or len(arr) == 0:
            norm = np.zeros(n)
        else:
            norm = (arr - mn) / (mx - mn)
        normalized[col] = norm

    total_scores = []
    for i in range(n):
        s = 0.0
        for col in goal_cols:
            if normalized[col][i] is not None:
                s += float(normalized[col][i])
        total_scores.append(s)

    max_ret = max(1, int(0.05 * n))
    sorted_indices = np.argsort(total_scores)[:max_ret]
    return sorted_indices.tolist()

import csv
import numpy as np
from sklearn.linear_model import LinearRegression


def guess(header, train, test):
    goal_cols = [h for h in header if h.endswith('-')]
    first_goal = goal_cols[0]  # "Cost-"

    n_train = len(train)
    n_test = len(test)

    def safe_float(val):
        try:
            return float(val)
        except (ValueError, TypeError):
            return None

    col_vals = {}
    for col in goal_cols:
        vals = []
        for row in train:
            f = safe_float(row[header.index(col)])
            if f is not None:
                vals.append(f)
        col_vals[col] = np.array(vals, dtype=float)

    normalized = {}
    for col in goal_cols:
        arr = col_vals[col]
        mn, mx = float(arr.min()), float(arr.max())
        if mx == mn or len(arr) == 0:
            norm = np.zeros(n_train)
        else:
            norm = (arr - mn) / (mx - mn)
        normalized[col] = norm

    # Build feature matrix from train rows (all columns except goals)
    input_cols = [c for c in header if not c.endswith('-')]
    X_train = np.zeros((n_train, len(input_cols)))
    y_train = normalized[first_goal].copy()

    for i, row in enumerate(train):
        for j, col in enumerate(input_cols):
            f = safe_float(row[header.index(col)])
            if f is not None:
                X_train[i, j] = f

    model = LinearRegression().fit(X_train, y_train)

    # Predict on test rows
    predictions = []
    for row in test:
        x_test = np.zeros(len(input_cols))
        for j, col in enumerate(input_cols):
            f = safe_float(row[header.index(col)])
            if f is not None:
                x_test[j] = f
        pred = model.predict(x_test.reshape(1, -1))[0]

        # Denormalize back to original scale
        mn, mx = float(y_train.min()), float(y_train.max())
        if mx == mn or len(y_train) == 0:
            predictions.append(float(mn))
        else:
            predictions.append(pred * (mx - mn) + mn)

    return predictions

def certify(xs, ys):
    """Return True if xs and ys are essentially the same apart from noise."""
    if len(xs) == 0 or len(ys) == 0:
        return False
    mean_x = sum(xs) / len(xs)
    mean_y = sum(ys) / len(ys)
    diff = abs(mean_x - mean_y)
    std_x = (sum((v - mean_x) ** 2 for v in xs) / max(len(xs), 1)) ** 0.5
    std_y = (sum((v - mean_y) ** 2 for v in ys) / max(len(ys), 1)) ** 0.5
    pooled_std = ((std_x ** 2 + std_y ** 2) ** 0.5) if (std_x or std_y) else 1.0
    return diff < 1.5 * pooled_std

        if ts < 0.3:

    return score >= 0.0

def blame(header, rows):
    goal_cols = [h for h in header if h.endswith('-')]

    # Per-row score: low is good for -cols, high is good for +cols
    scores = []
    for row in rows:
        s = 0.0
        for col in goal_cols:
            try:
                val = float(row[header.index(col)])
            except (ValueError, TypeError):
                continue
            if col.endswith('-'):
                s -= val
            else:
                s += val
        scores.append(s)

    # Worst and best rows by score
    n_worst = max(1, int(0.1 * len(rows)))
    worst_idx = sorted(range(len(scores)), key=lambda i: scores[i])[:n_worst]
    best_idx = sorted(range(len(scores)), key=lambda i: -scores[i])[:max(1, int(0.1 * len(rows)))]

    # For each input column, compare mean in worst vs best rows
    input_cols = [c for c in header if not c.endswith('-')]
    blame_data = []

    for col in input_cols:
        worst_vals = []
        best_vals = []
        for i in worst_idx:
            try:
                worst_vals.append(float(rows[i][header.index(col)]))
            except (ValueError, TypeError):
                pass
        for i in best_idx:
            try:
                best_vals.append(float(rows[i][header.index(col)]))
            except (ValueError, TypeError):
                pass

        if len(worst_vals) < 3 or len(best_vals) < 3:
            continue

        mean_w = sum(worst_vals) / len(worst_vals)
        mean_b = sum(best_vals) / len(best_vals)
        blame_data.append((abs(mean_w - mean_b), col))

    # Sort by magnitude of difference — biggest gap first
    blame_data.sort(key=lambda x: -x[0])

    return [col for _, col in blame_data[:3]]

def justify(header, rows, row):
    goal_cols = [h for h in header if h.endswith('-')]

    def safe_float(val):
        try:
            return float(val)
        except (ValueError, TypeError):
            return None

    scores = []
    for r in rows:
        s = 0.0
        for col in goal_cols:
            f = safe_float(r[header.index(col)])
            if f is not None:
                if col.endswith('-'):
                    s -= f
                else:
                    s += f
        scores.append(s)

    # Score the query row (goals are "?") using its input features via a simple heuristic:
    # We approximate by scoring it as "average" since we can't know goals.
    q_score = 0.0
    for col in goal_cols:
        f = safe_float(row[header.index(col)])
        if f is not None:
            if col.endswith('-'):
                q_score -= f
            else:
                q_score += f

    # Fraction of rows that the query row beats (i.e., has a better score)
    n_better = sum(1 for s in scores if s < q_score)
    p = n_better / len(scores) if scores else 0.5

    return p, f"Better than {p:.0%} of the setups."

def choose(header, rows, label):
    """Pick the best setup by labelling up to 50 candidates."""
    # remember() picks top indices; with "?" goals all scores are equal so it
    # returns the first ~5% of row indices — a cheap sampling strategy.
    candidates = remember(header, rows)

    goal_cols = [h for h in header if h.endswith('-')]
    best_idx = None
    best_score = float('-inf')

    for idx in candidates:
        true_goals = label(idx)
        score = 0.0
        for col in goal_cols:
            val = true_goals[header.index(col)]
            try:
                f = float(val)
            except (ValueError, TypeError):
                continue
            if col.endswith('-'):
                score -= f
            else:
                score += f

        if score > best_score:
            best_score = score
            best_idx = idx

    return best_idx

def fix(header, row, levers):
    goal_cols = [h for h in header if h.endswith('-')]

    def safe_float(val):
        try:
            return float(val)
        except (ValueError, TypeError):
            return None

    # Score a complete row based on goals: low is good for -cols, high for +cols
    def score(row_dict):
        s = 0.0
        for col in goal_cols:
            f = safe_float(row_dict[header.index(col)])
            if f is not None:
                if col.endswith('-'):
                    s -= f
                else:
                    s += f
        return s

    # Try all combinations of lever values and pick the one with best score.
    # Since goals are "?" in a bad row, we use training data patterns via remember().
    # We evaluate each candidate by checking if its lever values match any known-good setup.

    best_combo = {}
    best_score = float('-inf')

    def evaluate(combo):
        nonlocal best_combo, best_score
        test_row = dict(row)
        for col, val in combo.items():
            test_row[header.index(col)] = val
        s = score(test_row)
        if s > best_score:
            best_score = s
            best_combo = dict(combo)

    # Start with original row (no changes yet) — baseline
    evaluate({})

    for lever_col, allowed in levers.items():
        new_combos = []
        for existing in [best_combo]:  # extend from current best only? No — try all.
            pass
        # Actually we need to enumerate ALL combinations across all levers.
        # Simplest: build cartesian product of allowed values for each lever column.

    # Build full cartesian product of allowed values
    combo_keys = list(levers.keys())
    combo_values = [levers[k] for k in combo_keys]

    def cartesian_product(values_list):
        if not values_list:
            return [[]]
        rest = cartesian_product(values_list[1:])
        return [[v] + r for v, r in zip(values_list[0], rest)]

    all_combos = cartesian_product(combo_values)

    for combo in all_combos:
        evaluate(dict(zip(combo_keys, combo)))

    # Apply best changes to row and return them
    result = {}
    for col, val in best_combo.items():
        result[col] = val

    return result
