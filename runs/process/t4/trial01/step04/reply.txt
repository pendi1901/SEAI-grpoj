import sys

from step01 import remember


def flow(header, row, seen):
    """Approve or reject a proposal as it arrives."""
    
    goal_cols = [h for h in header if h.endswith('-')]
    
    def safe_float(val):
        try:
            return float(val)
        except (ValueError, TypeError):
            return None
    
    parsed = []
    for col in header:
        val = row[col]
        if val == "?" or val is None:
            parsed.append(None)
        else:
            try:
                parsed.append(float(val))
            except (ValueError, TypeError):
                parsed.append(None)
    
    # Score based on goals: low for -cols, high for +cols
    score = 0.0
    for col in goal_cols:
        f = safe_float(row[col])
        if f is not None:
            score += (-1.0 if col.endswith('-') else 1.0) * (f - 0.5)
    
    # Bonus for small teams — step03 showed they're cheaper
    n_seen = len(seen)
    if n_seen > 0 and parsed[header.index("Team Size")] is not None:
        ts = parsed[header.index("Team Size")]
        if ts < 0.3:
            score += 0.1
    
    return score >= 0.0


if __name__ == "__main__":
    import pandas as pd

    df = pd.read_csv("pom3a.train.csv")
    header = list(df.columns)
    rows = []
    for _, row in df.iterrows():
        r = []
        for col in header:
            val = row[col]
            if pd.isna(val):
                r.append("?")
            else:
                r.append(float(val))
        rows.append(r)

    approved, rejected = 0, 0
    for i, row in enumerate(rows):
        if flow(header, row, rows[:i]):
            approved += 1
        else:
            rejected += 1

    print(f"Approved {approved}, rejected {rejected} out of {len(rows)} proposals.")
