import csv
import numpy as np


def remember(header, rows):
    goal_cols = [h for h in header if h.endswith('-')]
    n = len(rows)

    def safe_float(val):
        try:
            return float(val)
        except (ValueError, TypeError):
            return None

    col_vals = {}
    for col in goal_cols:
        vals = []
        for row in rows:
            f = safe_float(row[header.index(col)])
            if f is not None:
                vals.append(f)
        col_vals[col] = np.array(vals, dtype=float)

    normalized = {}
    for col in goal_cols:
        arr = col_vals[col]
        mn, mx = float(arr.min()), float(arr.max())
        if mx == mn or len(arr) == 0:
            norm = np.zeros(n)
        else:
            norm = (arr - mn) / (mx - mn)
        normalized[col] = norm

    total_scores = []
    for i in range(n):
        s = 0.0
        for col in goal_cols:
            if normalized[col][i] is not None:
                s += float(normalized[col][i])
        total_scores.append(s)

    max_ret = max(1, int(0.05 * n))
    sorted_indices = np.argsort(total_scores)[:max_ret]
    return sorted_indices.tolist()

import csv
import numpy as np
from sklearn.linear_model import LinearRegression


def guess(header, train, test):
    goal_cols = [h for h in header if h.endswith('-')]
    first_goal = goal_cols[0]  # "Cost-"

    n_train = len(train)
    n_test = len(test)

    def safe_float(val):
        try:
            return float(val)
        except (ValueError, TypeError):
            return None

    col_vals = {}
    for col in goal_cols:
        vals = []
        for row in train:
            f = safe_float(row[header.index(col)])
            if f is not None:
                vals.append(f)
        col_vals[col] = np.array(vals, dtype=float)

    normalized = {}
    for col in goal_cols:
        arr = col_vals[col]
        mn, mx = float(arr.min()), float(arr.max())
        if mx == mn or len(arr) == 0:
            norm = np.zeros(n_train)
        else:
            norm = (arr - mn) / (mx - mn)
        normalized[col] = norm

    # Build feature matrix from train rows (all columns except goals)
    input_cols = [c for c in header if not c.endswith('-')]
    X_train = np.zeros((n_train, len(input_cols)))
    y_train = normalized[first_goal].copy()

    for i, row in enumerate(train):
        for j, col in enumerate(input_cols):
            f = safe_float(row[header.index(col)])
            if f is not None:
                X_train[i, j] = f

    model = LinearRegression().fit(X_train, y_train)

    # Predict on test rows
    predictions = []
    for row in test:
        x_test = np.zeros(len(input_cols))
        for j, col in enumerate(input_cols):
            f = safe_float(row[header.index(col)])
            if f is not None:
                x_test[j] = f
        pred = model.predict(x_test.reshape(1, -1))[0]

        # Denormalize back to original scale
        mn, mx = float(y_train.min()), float(y_train.max())
        if mx == mn or len(y_train) == 0:
            predictions.append(float(mn))
        else:
            predictions.append(pred * (mx - mn) + mn)

    return predictions
