```python
import csv
import numpy as np
import pandas as pd
from step01 import remember as step01_remember

def flow(header, row, seen):
    """
    Decides whether to accept a row. A row is accepted if it is 
    statistically 'bad' compared to the history (i.e., it is an outlier 
    in terms of performance degradation).
    """
    if not seen:
        return False

    # Identify goal columns
    goal_cols = [col for col in header if col.endswith(('-', '+'))]
    
    # Calculate aggregate scores for history
    df_seen = pd.DataFrame(seen, columns=header)
    
    # Normalize goals
    def get_scores(data):
        scores = []
        for goal in goal_cols:
            col_min = df_seen[goal].min()
            col_max = df_seen[goal].max()
            norm = (data[goal] - col_min) / (col_max - col_min + 1e-9)
            if goal.endswith('-'):
                scores.append(1 - norm)
            else:
                scores.append(norm)
        return np.mean(scores, axis=0)

    seen_scores = get_scores(df_seen)
    
    # Score the current row
    df_row = pd.DataFrame([row], columns=header)
    row_score = get_scores(df_row)[0]
    
    # If the current row's performance is significantly worse than the 
    # historical average (e.g., 2 standard deviations below mean), warn the user.
    mean_score = np.mean(seen_scores)
    std_score = np.std(seen_scores)
    
    # Return True if the row is "going bad" (performance is significantly lower)
    return row_score < (mean_score - 2 * std_score)

if __name__ == "__main__":
    filename = 'pom3a.n1000.train.csv'
    try:
        with open(filename, 'r') as f:
            reader = csv.reader(f)
            header = next(reader)
            rows = [[float(val) for val in row] for row in reader]
        
        seen = []
        warnings = 0
        for row in rows:
            if flow(header, row, seen):
                warnings += 1
            seen.append(row)
            
        print("Customer Inquiry: Monitoring project performance.")
        print(f"Analysis: Monitored {len(rows)} incoming records.")
        print(f"Warnings triggered: {warnings}")
        print("Strategy: Flagged records that performed significantly worse than historical norms.")
        
    except Exception as e:
        print(f"Error: {e}")
        exit(1)
```
