import csv
import pandas as pd
import numpy as np

def remember(header, rows):
    """
    Identifies rows that perform well across all goal columns.
    Goals ending in '-' are minimized, '+' are maximized.
    """
    df = pd.DataFrame(rows, columns=header)
    
    # Identify goal columns
    goals = [col for col in header if col.endswith(('-', '+'))]
    
    # Normalize goals to [0, 1] range for comparison
    # For '-', lower is better (1 - normalized value)
    # For '+', higher is better (normalized value)
    normalized_scores = []
    for goal in goals:
        col_min = df[goal].min()
        col_max = df[goal].max()
        norm = (df[goal] - col_min) / (col_max - col_min + 1e-9)
        if goal.endswith('-'):
            normalized_scores.append(1 - norm)
        else:
            normalized_scores.append(norm)
            
    # Calculate aggregate score (average of normalized goals)
    # Higher aggregate score means better overall performance
    df['aggregate'] = np.mean(normalized_scores, axis=0)
    
    # Select top 5% of rows
    num_to_select = max(1, len(rows) // 20)
    top_indices = df['aggregate'].nlargest(num_to_select).index.tolist()
    
    return sorted(top_indices)

import csv

from scipy import stats

def certify(xs, ys):
    """
    Returns True if the difference between two distributions is statistically 
    insignificant (p > 0.05), False otherwise.
    """
    # Using Welch's t-test (equal_var=False) to compare two independent samples
    t_stat, p_value = stats.ttest_ind(xs, ys, equal_var=False)
    return p_value > 0.05

def analyze_team_size_impact(header, rows):
    """
    Compares 'Cost-' for small vs big teams.
    """
    df = pd.DataFrame(rows, columns=header)
    
    # Define small vs big based on median split of 'Team Size'
    median_team = df['Team Size'].median()
    small_teams = df[df['Team Size'] <= median_team]['Cost-']
    big_teams = df[df['Team Size'] > median_team]['Cost-']
    
    is_same = certify(small_teams.tolist(), big_teams.tolist())
    
    return is_same, small_teams.mean(), big_teams.mean()

    # Return the top 3 input columns that explain the variance in Idle-

def justify(header, rows, row):
    """
    Estimates the goal values for the provided row using KNN, 
    then compares it against all other rows to determine its percentile rank.
    """
    # 1. Estimate the goals for the input row
    # We need to provide a list of rows to guess()
    predictions = guess(header, rows, [row])
    
    # 2. Reconstruct the row with predicted goals
    goal_indices = [i for i, h in enumerate(header) if h.endswith(('-', '+'))]
    full_row = list(row)
    for i, idx in enumerate(goal_indices):
        full_row[idx] = predictions[0][i] if isinstance(predictions[0], list) else predictions[0]
    
    # 3. Calculate aggregate scores for all rows
    # We use the logic from step01 to rank performance
    df = pd.DataFrame(rows + [full_row], columns=header)
    goals = [col for col in header if col.endswith(('-', '+'))]
    
    normalized_scores = []
    for goal in goals:
        col_min = df[goal].min()
        col_max = df[goal].max()
        norm = (df[goal] - col_min) / (col_max - col_min + 1e-9)
        if goal.endswith('-'):
            normalized_scores.append(1 - norm)
        else:
            normalized_scores.append(norm)
            
    df['aggregate'] = np.mean(normalized_scores, axis=0)
    
    # 4. Determine percentile rank
    # p is the fraction of rows that this row is better than
    my_score = df['aggregate'].iloc[-1]
    others = df['aggregate'].iloc[:-1]
    p = (others < my_score).mean()
    
    reason = f"This setup is better than {p:.1%} of all known configurations based on its predicted performance across all goal metrics."
    
    return p, reason

import csv
import numpy as np
import pandas as pd
from sklearn.neighbors import KNeighborsRegressor
from step01 import remember
from step02 import guess

def choose(header, rows, label):
    """
    Finds the best row by sampling 50 rows, using KNN to estimate 
    performance for the rest, and selecting the best overall.
    """
    # Identify input columns
    input_cols = [col for col in header if not (col.endswith('-') or col.endswith('+'))]
    goal_cols = [col for col in header if col.endswith(('-', '+'))]
    
    # 1. Sample 50 rows to build a training set
    # We pick indices spread across the dataset
    indices = np.linspace(0, len(rows) - 1, 50, dtype=int)
    
    train_rows = []
    train_goals = []
    
    for idx in indices:
        true_goals = label(idx)
        train_rows.append(rows[idx])
        train_goals.append(true_goals)
        
    # 2. Train a regressor to predict all goals
    # We use a simple MultiOutput approach via KNN
    model = KNeighborsRegressor(n_neighbors=3)
    model.fit([r[:len(input_cols)] for r in train_rows], train_goals)
    
    # 3. Predict goals for all rows
    all_inputs = [r[:len(input_cols)] for r in rows]
    all_preds = model.predict(all_inputs)
    
    # 4. Rank rows based on predicted aggregate score
    df_preds = pd.DataFrame(all_preds, columns=goal_cols)
    
    normalized_scores = []
    for goal in goal_cols:
        col_min = df_preds[goal].min()
        col_max = df_preds[goal].max()
        norm = (df_preds[goal] - col_min) / (col_max - col_min + 1e-9)
        if goal.endswith('-'):
            normalized_scores.append(1 - norm)
        else:
            normalized_scores.append(norm)
            
    aggregate = np.mean(normalized_scores, axis=0)
    
    # Return index of the best row
    return int(np.argmax(aggregate))

                numeric_row = [float(val) for val in row]
                rows.append(numeric_row)
                true_data.append(numeric_row[-3:]) # Last 3 are goals

            return true_data[i]
