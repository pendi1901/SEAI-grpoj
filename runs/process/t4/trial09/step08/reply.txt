import csv
import pandas as pd
import numpy as np
from step01 import remember

def fix(header, row, levers):
    """
    Finds the smallest change to 'Plan' or 'Team Size' that improves 
    the aggregate score of a bad row.
    """
    df_row = pd.DataFrame([row], columns=header)
    goals = [col for col in header if col.endswith(('-', '+'))]
    
    # We need a baseline to compare against. 
    # Since we don't have the full dataset in the function signature,
    # we assume the goal is to move the row toward the median of the input space.
    # For simplicity, we perform a local search on the provided levers.
    
    best_change = {}
    best_val = -1.0
    
    # Try small perturbations on the levers
    for col, bounds in levers.items():
        idx = header.index(col)
        original_val = row[idx]
        
        # Try 5 steps within the bounds
        low, high = bounds
        for step in np.linspace(low, high, 5):
            new_row = list(row)
            new_row[idx] = step
            
            # Heuristic: minimize Cost- and Idle- (the goals)
            # We want to minimize the sum of normalized goals
            score = 0
            for g in goals:
                g_idx = header.index(g)
                val = new_row[g_idx]
                # If we don't know the true goal, we approximate improvement
                # by moving numeric inputs toward the center (0.5)
                score -= abs(new_row[idx] - 0.5)
            
            if score > best_val:
                best_val = score
                best_change = {col: step}
                
    return best_change

        cost_idx = header.index('Cost-')

        recommendation = fix(header, bad_row, levers)
