import csv
import pandas as pd
import numpy as np

def remember(header, rows):
    """
    Identifies rows that perform well across all goal columns.
    Goals ending in '-' are minimized, '+' are maximized.
    """
    df = pd.DataFrame(rows, columns=header)
    
    # Identify goal columns
    goals = [col for col in header if col.endswith(('-', '+'))]
    
    # Normalize goals to [0, 1] range for comparison
    # For '-', lower is better (1 - normalized value)
    # For '+', higher is better (normalized value)
    normalized_scores = []
    for goal in goals:
        col_min = df[goal].min()
        col_max = df[goal].max()
        norm = (df[goal] - col_min) / (col_max - col_min + 1e-9)
        if goal.endswith('-'):
            normalized_scores.append(1 - norm)
        else:
            normalized_scores.append(norm)
            
    # Calculate aggregate score (average of normalized goals)
    # Higher aggregate score means better overall performance
    df['aggregate'] = np.mean(normalized_scores, axis=0)
    
    # Select top 5% of rows
    num_to_select = max(1, len(rows) // 20)
    top_indices = df['aggregate'].nlargest(num_to_select).index.tolist()
    
    return sorted(top_indices)

def certify(xs, ys):
    """
    Returns True if the difference between two distributions is statistically 
    insignificant (p > 0.05), False otherwise.
    """
    # Using Welch's t-test (equal_var=False) to compare two independent samples
    t_stat, p_value = stats.ttest_ind(xs, ys, equal_var=False)
    return p_value > 0.05

def analyze_team_size_impact(header, rows):
    """
    Compares 'Cost-' for small vs big teams.
    """
    df = pd.DataFrame(rows, columns=header)
    
    # Define small vs big based on median split of 'Team Size'
    median_team = df['Team Size'].median()
    small_teams = df[df['Team Size'] <= median_team]['Cost-']
    big_teams = df[df['Team Size'] > median_team]['Cost-']
    
    is_same = certify(small_teams.tolist(), big_teams.tolist())
    
    return is_same, small_teams.mean(), big_teams.mean()

def justify(header, rows, row):

        full_row[idx] = predictions[0][i] if isinstance(predictions[0], list) else predictions[0]

    df = pd.DataFrame(rows + [full_row], columns=header)

def choose(header, rows, label):
    """
    Finds the best row by sampling 50 rows, using KNN to estimate 
    performance for the rest, and selecting the best overall.
    """
    # Identify input columns
    input_cols = [col for col in header if not (col.endswith('-') or col.endswith('+'))]
    goal_cols = [col for col in header if col.endswith(('-', '+'))]
    
    # 1. Sample 50 rows to build a training set
    # We pick indices spread across the dataset
    indices = np.linspace(0, len(rows) - 1, 50, dtype=int)
    
    train_rows = []
    train_goals = []
    
    for idx in indices:
        true_goals = label(idx)
        train_rows.append(rows[idx])
        train_goals.append(true_goals)
        
    # 2. Train a regressor to predict all goals
    # We use a simple MultiOutput approach via KNN
    model = KNeighborsRegressor(n_neighbors=3)
    model.fit([r[:len(input_cols)] for r in train_rows], train_goals)
    
    # 3. Predict goals for all rows
    all_inputs = [r[:len(input_cols)] for r in rows]
    all_preds = model.predict(all_inputs)
    
    # 4. Rank rows based on predicted aggregate score
    df_preds = pd.DataFrame(all_preds, columns=goal_cols)
    
    normalized_scores = []
    for goal in goal_cols:
        col_min = df_preds[goal].min()
        col_max = df_preds[goal].max()
        norm = (df_preds[goal] - col_min) / (col_max - col_min + 1e-9)
        if goal.endswith('-'):
            normalized_scores.append(1 - norm)
        else:
            normalized_scores.append(norm)
            
    aggregate = np.mean(normalized_scores, axis=0)
    
    # Return index of the best row
    return int(np.argmax(aggregate))

def spot(header, rows):
    """
    Identifies rows that are outliers using the Isolation Forest algorithm.
    Returns the indices of the rows, sorted by suspicion level (highest first).
    """
    df = pd.DataFrame(rows, columns=header)
    
    # IsolationForest works well for detecting anomalies in numeric data.
    # We use the contamination parameter to estimate the proportion of outliers.
    model = IsolationForest(contamination=0.05, random_state=42)
    
    # Fit the model to the data
    model.fit(df)
    
    # Get anomaly scores (lower scores are more anomalous)
    scores = model.decision_function(df)
    
    # Get indices sorted by score (ascending, so most negative/anomalous first)
    suspicious_indices = np.argsort(scores)
    
    return suspicious_indices.tolist()

import csv
import numpy as np
import pandas as pd
from sklearn.neighbors import KNeighborsRegressor
from step01 import remember
from step02 import guess

def race(header, rows, label):
    """
    Compares three strategies: random, knn-greedy, and knn-uncertainty.
    Budget: 50 labels total.
    """
    n_rows = len(rows)
    input_cols = [i for i, h in enumerate(header) if not (h.endswith('-') or h.endswith('+'))]
    goal_cols = [i for i, h in enumerate(header) if h.endswith(('-', '+'))]
    
    labeled_indices = {}
    
    def get_label(idx):
        if idx not in labeled_indices:
            if len(labeled_indices) >= 50:
                raise ValueError("Budget exceeded")
            labeled_indices[idx] = label(idx)
        return labeled_indices[idx]

    def get_aggregate(row_goals):
        # Normalize goals based on current knowledge
        goals_arr = np.array(list(labeled_indices.values()))
        norm = (row_goals - goals_arr.min(axis=0)) / (goals_arr.max(axis=0) - goals_arr.min(axis=0) + 1e-9)
        scores = []
        for i, col_idx in enumerate(goal_cols):
            if header[col_idx].endswith('-'):
                scores.append(1 - norm[i])
            else:
                scores.append(norm[i])
        return np.mean(scores)

    # Strategy 1: Random
    def strategy_random():
        indices = np.random.choice(n_rows, 16, replace=False)
        best_idx = -1
        best_score = -1
        for idx in indices:
            score = get_aggregate(get_label(idx))
            if score > best_score:
                best_score, best_idx = score, idx
        return best_idx

    # Strategy 2: KNN Greedy
    def strategy_knn_greedy():
        # Initial 10 samples
        for i in range(10): get_label(i * (n_rows // 10))
        # Train and predict
        X = [[rows[i][c] for c in input_cols] for i in labeled_indices]
        y = list(labeled_indices.values())
        model = KNeighborsRegressor(n_neighbors=3).fit(X, y)
        preds = model.predict([[rows[i][c] for c in input_cols] for i in range(n_rows)])
        # Pick best predicted
        return int(np.argmax([get_aggregate(p) for p in preds]))

    # Strategy 3: KNN Uncertainty (Exploration)
    def strategy_knn_explore():

def analyze_shift(header, rows):

    """
    # Identify goal columns
    goal_indices = [i for i, h in enumerate(header) if h.endswith(('-', '+'))]

import csv

def flow(header, row, seen):
    """
    Decides whether to accept a row. A row is accepted if it is 
    statistically 'bad' compared to the history (i.e., it is an outlier 
    in terms of performance degradation).
    """
    if not seen:
        return False

    # Identify goal columns
    goal_cols = [col for col in header if col.endswith(('-', '+'))]
    
    # Calculate aggregate scores for history
    df_seen = pd.DataFrame(seen, columns=header)
    
    # Normalize goals
    def get_scores(data):
        scores = []
        for goal in goal_cols:
            col_min = df_seen[goal].min()
            col_max = df_seen[goal].max()
            norm = (data[goal] - col_min) / (col_max - col_min + 1e-9)
            if goal.endswith('-'):
                scores.append(1 - norm)
            else:
                scores.append(norm)
        return np.mean(scores, axis=0)

    seen_scores = get_scores(df_seen)
    
    # Score the current row
    df_row = pd.DataFrame([row], columns=header)
    row_score = get_scores(df_row)[0]
    
    # If the current row's performance is significantly worse than the 
    # historical average (e.g., 2 standard deviations below mean), warn the user.
    mean_score = np.mean(seen_scores)
    std_score = np.std(seen_scores)
    
    # Return True if the row is "going bad" (performance is significantly lower)
    return row_score < (mean_score - 2 * std_score)
