import csv
import pandas as pd
import numpy as np

def remember(header, rows):
    """
    Identifies rows that perform well across all goal columns.
    Goals ending in '-' are minimized, '+' are maximized.
    """
    df = pd.DataFrame(rows, columns=header)
    
    # Identify goal columns
    goals = [c for c in header if c.endswith('-') or c.endswith('+')]
    
    # Normalize goals to [0, 1] range to treat them equally
    # For '-', lower is better (1 - normalized_val)
    # For '+', higher is better (normalized_val)
    normalized_goals = pd.DataFrame(index=df.index)
    for goal in goals:
        col = df[goal].astype(float)
        norm = (col - col.min()) / (col.max() - col.min())
        if goal.endswith('-'):
            normalized_goals[goal] = 1 - norm
        else:
            normalized_goals[goal] = norm
            
    # Calculate a score: the product of normalized goals (geometric mean)
    # This favors rows that are good across all objectives
    scores = normalized_goals.prod(axis=1)
    
    # Select top 5% (or at least 1)
    n_select = max(1, len(rows) // 20)
    best_indices = scores.nlargest(n_select).index.tolist()
    
    return sorted(best_indices)

def certify(xs, ys):
    """
    Returns True if the difference between two distributions is statistically 
    insignificant (p > 0.05), False if they are significantly different.
    """
    # Use Welch's t-test (equal_var=False) to compare means
    t_stat, p_value = stats.ttest_ind(xs, ys, equal_var=False)
    return p_value > 0.05

import csv
import pandas as pd

def flow(header, row, seen):
    """
    Decides whether to accept a row based on the 'Secretary Problem' logic.
    We observe a sample to establish a baseline of 'good' performance,
    then accept the first row that exceeds that baseline.
    """
    # If this is the first row, we cannot compare it yet
    if not seen:
        return False
    
    # Use the first 20% of the data as a calibration phase to define 'good'
    # We use the logic from step01 to identify what a 'good' row looks like
    all_rows = seen + [row]
    if len(seen) < len(all_rows) * 0.2:
        return False
    
    # Identify goal columns
    goals = [c for c in header if c.endswith('-') or c.endswith('+')]
    
    # Calculate the 'goodness' of the current row relative to seen rows
    df = pd.DataFrame(all_rows, columns=header)
    
    # Normalize goals
    norm_df = pd.DataFrame(index=df.index)
    for goal in goals:
        col = df[goal].astype(float)
        norm = (col - col.min()) / (col.max() - col.min())
        norm_df[goal] = (1 - norm) if goal.endswith('-') else norm
    
    scores = norm_df.prod(axis=1)
    
    # Threshold: Is the current row better than the median of what we've seen?
    current_score = scores.iloc[-1]
    median_score = scores.iloc[:-1].median()
    
    # Accept if it's better than the median of previous observations
    return current_score > median_score

def justify(header, rows, row):
    """
    Estimates the performance of a row and returns its percentile rank 
    and a justification string.
    """
    df = pd.DataFrame(rows, columns=header)
    inputs = [c for c in header if not (c.endswith('-') or c.endswith('+'))]
    goals = [c for c in header if c.endswith('-') or c.endswith('+')]
    
    # 1. Estimate the missing goals for the input row
    # We use KNN to predict each goal column independently
    predictions = {}
    for goal in goals:
        model = KNeighborsRegressor(n_neighbors=5)
        model.fit(df[inputs], df[goal])
        predictions[goal] = model.predict([row[:len(inputs)]])[0]
    
    # 2. Calculate "goodness" score for all rows
    # Normalize goals: 1 - (val - min)/(max - min) for '-', (val - min)/(max - min) for '+'
    norm_df = pd.DataFrame(index=df.index)
    for goal in goals:
        col = df[goal].astype(float)
        norm = (col - col.min()) / (col.max() - col.min())
        norm_df[goal] = (1 - norm) if goal.endswith('-') else norm
    scores = norm_df.prod(axis=1)
    
    # 3. Calculate "goodness" score for the input row
    norm_row = []
    for goal in goals:
        col = df[goal].astype(float)
        val = predictions[goal]
        norm = (val - col.min()) / (col.max() - col.min())
        norm_row.append((1 - norm) if goal.endswith('-') else norm)
    row_score = np.prod(norm_row)
    
    # 4. Calculate percentile (fraction of rows that row is better than)
    p = (scores < row_score).mean()
    
    reason = f"This setup is estimated to be better than {p:.1%} of all known configurations based on its predicted performance across all goal metrics."
    return p, reason

def choose(header, rows, label):
    """
    Selects the best row by using a small sample to train a regressor,
    predicting performance for all rows, and querying the top candidates.
    """
    # 1. Identify inputs and goals
    inputs = [c for c in header if not (c.endswith('-') or c.endswith('+'))]
    goals = [c for c in header if c.endswith('-') or c.endswith('+')]
    
    # 2. Sample 20 rows to build an initial model
    # We use the first 20 rows to get a baseline
    sample_indices = list(range(20))
    for i in sample_indices:
        true_goals = label(i)
        for j, goal in enumerate(goals):
            rows[i][header.index(goal)] = true_goals[j]
            
    # 3. Predict performance for all rows
    # We use the logic from step02 to estimate goals for the remaining rows
    # We treat the rows with known goals as training data
    train_rows = [rows[i] for i in sample_indices]
    
    # Predict for all rows
    # We use a simple KNN approach to estimate the "goodness" of all rows
    # based on the 20 labeled samples
    from sklearn.neighbors import KNeighborsRegressor
    
    predictions = []
    for i in range(len(rows)):
        if i in sample_indices:
            # Use actual values
            row_vals = [rows[i][header.index(g)] for g in goals]
        else:
            # Predict values
            row_vals = []
            for goal in goals:
                model = KNeighborsRegressor(n_neighbors=3)
                model.fit([r[:len(inputs)] for r in train_rows], [r[header.index(goal)] for r in train_rows])
                row_vals.append(model.predict([rows[i][:len(inputs)]])[0])
        predictions.append(row_vals)
        
    # 4. Score all rows and pick the best
    # Normalize and calculate geometric mean as per step01
    df_preds = pd.DataFrame(predictions, columns=goals)
    norm_df = pd.DataFrame(index=df_preds.index)
    for goal in goals:
        col = df_preds[goal]
        norm = (col - col.min()) / (col.max() - col.min())
        norm_df[goal] = (1 - norm) if goal.endswith('-') else norm
    
    scores = norm_df.prod(axis=1)
    
    # Get the index of the best row
    best_idx = scores.idxmax()
    
    # If we haven't labeled it yet, label it (we have 30 labels left)
    if best_idx not in sample_indices:
        label(best_idx)
        
    return int(best_idx)

import csv

def fix(header, row, levers):
    """
    Identifies the smallest change to 'Plan' or 'Team Size' to improve 
    the 'Cost-' and 'Idle-' goals for a bad row.
    """
    # Load data to train a surrogate model for local sensitivity analysis
    filename = 'pom3a.n1000.train.csv'
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header_csv = next(reader)
        rows = [[float(val) for val in row] for row in reader]
    
    df = pd.DataFrame(rows, columns=header_csv)
    inputs = [c for c in header if c in levers]
    goals = ['Cost-', 'Idle-']
    
    # Train a model to predict the goals we want to minimize
    model = RandomForestRegressor(n_estimators=20, random_state=42)
    model.fit(df[inputs], df[goals])
    
    best_change = {}
    min_score = float('inf')
    
    # Test small perturbations within allowed lever ranges
    # We try to find a change that minimizes the predicted Cost- and Idle-
    for col in inputs:
        low, high = levers[col]
        # Try 5 steps between low and high
        for val in np.linspace(low, high, 5):
            test_row = list(row)
            test_row[header.index(col)] = val
            
            # Predict impact
            pred = model.predict([ [test_row[header.index(c)] for c in inputs] ])[0]
            score = sum(pred) # Minimize sum of goals
            
            if score < min_score:
                min_score = score
                best_change = {col: val}
                
    return best_change

    Identifies rows that are outliers in the input space using IsolationForest.

    """
    df = pd.DataFrame(rows, columns=header)

    model = IsolationForest(contamination=0.05, random_state=42)

    preds = model.fit_predict(X)
    scores = model.decision_function(X)
    
    # The decision_function returns lower values for more anomalous points
    # We want the most suspicious (lowest scores) first
    suspicious_indices = np.argsort(scores)
    
    # Return only the indices where the model flagged an anomaly (-1)
    # or simply return all sorted by suspicion level
    return suspicious_indices.tolist()

import csv

from step01 import remember

def flow(header, row, seen):
    """
    Warns the customer if the performance of the incoming row is significantly 
    worse than the historical baseline (the 20th percentile of seen rows).
    """
    if not seen:
        return False
    
    # Identify goal columns
    goals = [c for c in header if c.endswith('-') or c.endswith('+')]
    
    # Convert seen rows and current row to DataFrame
    df = pd.DataFrame(seen + [row], columns=header)
    
    # Normalize goals
    norm_df = pd.DataFrame(index=df.index)
    for goal in goals:
        col = df[goal].astype(float)
        # Handle potential division by zero if all values are the same
        rng = col.max() - col.min()
        norm = (col - col.min()) / rng if rng > 0 else 0
        norm_df[goal] = (1 - norm) if goal.endswith('-') else norm
    
    scores = norm_df.prod(axis=1)
    
    # Baseline: 20th percentile of seen rows
    # If the current row's score is below this, it's "going bad"
    baseline = scores.iloc[:-1].quantile(0.2)
    current_score = scores.iloc[-1]
    
    # Return True if it's performing poorly (below the 20th percentile)
    return current_score < baseline

    worst_rows = df[df['Cost-'] >= worst_threshold]

    # within the context of the worst setups
    X = worst_rows[inputs]

    feat_imp = sorted(zip(inputs, importances), key=lambda x: x[1], reverse=True)
    
    # Return the top 2 features that explain the 'Cost-' variance

def get_score(row_goals, header):
    """Calculates geometric mean of normalized goals."""
    goals = [c for c in header if c.endswith('-') or c.endswith('+')]
    norm_goals = []
    for i, goal in enumerate(goals):
        val = row_goals[i]
        # Normalize assuming 0-1 range based on POM3a data properties
        if goal.endswith('-'):
            norm_goals.append(1.0 - val)
        else:
            norm_goals.append(val)
    return np.prod(norm_goals)

def choose(header, rows, label):

    We filter for rows where 'Size' and 'Criticality' are high,
    then use a small budget to sample and predict the best.

    size_idx = header.index('Size')
    crit_idx = header.index('Criticality')
    
    candidates = []
    for i, row in enumerate(rows):
        if row[size_idx] > 0.7 and row[crit_idx] > 0.7:
            candidates.append(i)

    budget = 50
    tried = {}

    sample_size = min(20, len(candidates))

        idx = candidates[i]
        tried[idx] = label(idx)
        
    # 3. Train model and predict
    inputs = [c for c in header if not (c.endswith('-') or c.endswith('+'))]
    goals = [c for c in header if c.endswith('-') or c.endswith('+')]

    # Predict for remaining candidates
