import ezr

def choose(header, rows, label):
    """
    Finds the best row by using the acquire strategy from ezr.
    We label up to 50 rows and return the index of the best one found.
    """
    # Create a table object. Note: rows currently have '?' in goal columns.
    # We need to track the original indices to return the correct one.
    indexed_rows = []
    for i, row in enumerate(rows):
        indexed_rows.append((i, row))
    
    # We use the acquire logic from ezr.
    # The acquire function takes a TBL and an ORACLE.
    # We need to wrap the label function to handle our indexed rows.
    tbl = ezr.Tbl([header] + rows)
    
    # The acquire function returns a list of rows. 
    # We need to map these back to their original indices.
    # We define a custom oracle that calls the provided label function
    # and keeps track of the original index.
    labeled_data = []
    
    def oracle_wrapper(row):
        # Find the original index by matching the row (assuming unique rows)
        # or by passing the index through the row if possible.
        # Since we can't change the row structure, we find the index.
        for i, r in enumerate(rows):
            if r == row:
                true_goals = label(i)
                new_row = row[:]
                # Update the goal columns
                goal_indices = [idx for idx, name in enumerate(header) 
                                if name.endswith("-") or name.endswith("+")]
                for g_idx, val in zip(goal_indices, true_goals):
                    new_row[g_idx] = val
                return new_row
        return row

    # Use ezr.acquire to select and label rows
    # It uses the budget defined in ezr.the.Stop (default 50)
    best_rows = ezr.acquire(tbl, cap=50, label=oracle_wrapper)
    
    # Find the best row among the labeled ones
    # A row is better when its goals are closer to their best (ydist is low)
    best_row = min(best_rows, key=lambda r: ezr.ydist(tbl, r))
    
    # Return the index of the best row
    for i, r in enumerate(rows):
        if r == best_row:
            return i
    return 0

if __name__ == "__main__":

        for i, name in enumerate(header):
            if name.endswith("-") or name.endswith("+"):

    best_idx = choose(header, masked_rows, label)
