import ezr
import step01
import step04

def flow(header, row, seen):
    """
    Warn the customer if a new row is 'bad'.
    We define 'bad' as having a ydist significantly worse than the 
    average ydist of the rows seen so far.
    """
    if not seen:
        return False
    
    # Create a table to calculate ydist
    tbl = ezr.Tbl([header] + seen + [row])
    
    # Calculate ydist for the new row and all seen rows
    new_ydist = ezr.ydist(tbl, row)
    seen_ydists = [ezr.ydist(tbl, r) for r in seen]
    
    # Calculate the average ydist of seen rows
    avg_seen_ydist = sum(seen_ydists) / len(seen_ydists)
    
    # If the new row is significantly worse than the average (e.g., 1.5x worse),
    # return True to flag it as "going bad".
    if new_ydist > (avg_seen_ydist * 1.5):
        return True
    
    return False

if __name__ == "__main__":
    filename = "pom3a.n1000.train.csv"
    data = list(ezr.csv(filename))
    header = data[0]
    rows = data[1:]
    
    seen = []
    alerts = []
    
    for i, row in enumerate(rows):
        if flow(header, row, seen):
            alerts.append((i, row))
        seen.append(row)
        
    print(f"Monitoring project performance: Processed {len(rows)} weekly updates.")
    print(f"Alerts triggered: {len(alerts)} instances where performance degraded significantly.")
    
    if alerts:
        print("\nFirst 3 alerts (index and row):")
        for i, (idx, row) in enumerate(alerts[:3]):
            print(f"Alert {i+1} at week {idx}: {row}")
    else:
        print("No significant performance degradation detected.")
    
    exit(0)
