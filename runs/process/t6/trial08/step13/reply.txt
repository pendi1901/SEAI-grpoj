```python
import ezr
import step03

def certify(xs, ys):
    """
    Returns True if the two distributions are statistically indistinguishable 
    (same apart from noise), False if they really differ.
    """
    # We reuse the logic from step03 which uses ezr.same (Cliff's Delta and KS test)
    return step03.certify(xs, ys)

if __name__ == "__main__":
    # Load data
    filename = "pom3a.n1000.train.csv"
    data = list(ezr.csv(filename))
    header = data[0]
    rows = data[1:]
    
    # The customer asks if results shifted after changing how teams decide what to build.
    # We simulate this by comparing the first half of the data (pre-change) 
    # to the second half (post-change).
    mid = len(rows) // 2
    pre_change = rows[:mid]
    post_change = rows[mid:]
    
    # We evaluate performance using the primary goal (Cost-)
    cost_idx = header.index("Cost-")
    
    pre_costs = [r[cost_idx] for r in pre_change]
    post_costs = [r[cost_idx] for r in post_change]
    
    # Perform certification
    is_same = certify(pre_costs, post_costs)
    
    print(f"Comparison of project results (Pre-change vs Post-change):")
    print(f"Mean Cost (Pre):  {sum(pre_costs)/len(pre_costs):.4f}")
    print(f"Mean Cost (Post): {sum(post_costs)/len(post_costs):.4f}")
    
    if is_same:
        print("Conclusion: The results are statistically the same; the change had no significant impact.")
    else:
        print("Conclusion: The results have shifted significantly; the change had a real impact.")
    
    exit(0)
```
