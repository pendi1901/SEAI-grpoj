import ezr
import step01

def choose(header, rows, label):
    """
    Find the best row by using the acquire function from ezr.
    We label up to 50 rows and return the index of the best one found.
    """
    # Create a table object. Note: rows currently have '?' in goal columns.
    # We need to label them to get real values.
    # The acquire function in ezr.py is designed to label rows and 
    # return a set of rows that are near the best.
    
    # We wrap the label function to keep track of indices
    labeled_data = []
    
    def oracle_wrapper(row):
        # Find the original index of this row
        idx = rows.index(row)
        real_row = label(idx)
        # Update the row in place with real values
        for i, val in enumerate(real_row):
            row[i] = val
        return row

    # Create a table with the initial rows (goals are '?')
    tbl = ezr.Tbl([header] + rows)
    
    # Use ezr.acquire to intelligently pick 50 rows to label
    # This will call our oracle_wrapper which calls the provided label function
    best_rows = ezr.acquire(tbl, cap=50, label=oracle_wrapper)
    
    # After labeling, find the best row among all labeled ones
    # We re-evaluate ydist using the now-populated goal values
    best_row = min(best_rows, key=lambda r: ezr.ydist(tbl, r))
    
    return rows.index(best_row)

if __name__ == "__main__":
    filename = "pom3a.n1000.train.csv"
    data = list(ezr.csv(filename))
    header = data[0]
    rows = data[1:]
    
    # Keep a copy of original rows to simulate the oracle
    full_rows = [list(r) for r in rows]
    
    # Mask the goals for the rows passed to choose
    masked_rows = []
    for r in rows:
        new_row = list(r)
        for i, name in enumerate(header):
            if name.endswith(('-', '+')):
                new_row[i] = "?"
        masked_rows.append(new_row)
        
    # Define the label function for the harness
    labeled_count = 0
    labeled_indices = set()
    
    def label_func(idx):
        global labeled_count
        if idx not in labeled_indices:
            labeled_count += 1
            labeled_indices.add(idx)
        if labeled_count > 50:
            raise Exception("Label budget exceeded!")
        return full_rows[idx]
    
    best_idx = choose(header, masked_rows, label_func)
    
    print(f"Best setup found at index: {best_idx}")
    print(f"Total setups labeled: {labeled_count}")
    print(f"Best row values: {full_rows[best_idx]}")
