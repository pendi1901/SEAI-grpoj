import ezr
import step08

def fix(header, row, levers):
    """
    Finds the best change to a row to minimize 'Idle-'.
    We reuse the logic from step08 but specifically target the 'Idle-' goal.
    """
    # Load data to establish the context of 'good'
    filename = "pom3a.n1000.train.csv"
    data = list(ezr.csv(filename))
    tbl = ezr.Tbl(data)
    
    # We want to minimize 'Idle-'. 
    # ydist already calculates how far a row is from the ideal (0 for minimize).
    # We look for the change that results in the lowest ydist.
    best_ydist = ezr.ydist(tbl, row)
    best_change = {}

    # Try changing each lever to find an improvement
    for col_name, allowed in levers.items():
        if col_name not in header:
            continue
        idx = header.index(col_name)
        
        # Generate test values
        test_values = []
        if isinstance(allowed, tuple): # Numeric (low, high)
            low, high = allowed
            # Try 10 points in the range
            for i in range(10):
                val = low + (high - low) * (i / 9)
                test_values.append(val)
        else: # Symbolic
            test_values = allowed
            
        for val in test_values:
            candidate = list(row)
            candidate[idx] = val
            
            # Evaluate
            d = ezr.ydist(tbl, candidate)
            if d < best_ydist:
                best_ydist = d
                best_change = {col_name: val}
                
    return best_change

if __name__ == "__main__":
    filename = "pom3a.n1000.train.csv"
    data = list(ezr.csv(filename))
    header = data[0]
    rows = data[1:]
    
    # Identify a row with high 'Idle-' (a "bad" row for this specific goal)
    idle_idx = header.index("Idle-")
    tbl = ezr.Tbl(data)
    # Sort by Idle- descending
    scored = sorted([(r[idle_idx], r) for r in rows], key=lambda x: x[0], reverse=True)
    bad_row = scored[0][1]
    
    # Levers excluding 'Team Size' since the customer said they cannot hire.
    levers = {
        "Plan": (0.0, 1.0),
        "Size": (0.0, 1.0),
        "Culture": (0.0, 1.0),
        "Criticality": (0.0, 1.0),
        "Dynamism": (0.0, 1.0)
    }
    
    recommendation = fix(header, bad_row, levers)
    
    print("Analysis of high-idle project setup:")
    if recommendation:
        print("Recommended changes to reduce idle time without hiring:")
        for col, val in recommendation.items():
            print(f" - Change '{col}' to {val:.4f}")
    else:
        print("No lever changes found to improve this setup.")
    
    exit(0)
