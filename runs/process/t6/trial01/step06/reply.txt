import ezr


def justify(header, rows, row):
    """Return (p, reason) where p is fraction of rows that `row` beats."""
    # Build table from complete training rows only
    tbl = ezr.Tbl(rows)

    # Train a tree on the full dataset
    node = ezr.tree(tbl, tbl.rows)

    def predict_ydist(r):
        """Walk row to its leaf and return mean ydist of that leaf."""
        leaf_node = ezr.leaf(node, r)
        return leaf_node.mu

    # Compute actual ydist for each training row
    train_ydists = [ezr.ydist(tbl, r) for r in rows]

    # Predict the query row's ydist (goal cells are "?", but tree only splits on x cols)
    pred = predict_ydist(row)

    # Fraction of training rows that this prediction beats
    p = sum(1 for y in train_ydists if pred < y) / max(1, len(train_ydists))

    reason = (f"Better than {p:.0%} of setups on the data.")
    return (p, reason)


if __name__ == "__main__":
    import os
    path = os.path.join(os.path.dirname(__file__) or ".", "pom3a.train.csv")
    with open(path, encoding="utf-8-sig") as f:
        lines = [l.strip() for l in f if l.strip()]
    header = lines[0].split(",")

    rows = []
    for line in lines[1:]:
        cells = line.split(",")
        row = []
        for c in cells:
            s = c.strip()
            try: row.append(float(s))
            except ValueError:
                if s.lower() == "true":  row.append(True)
                elif s.lower() == "false": row.append(False)
                else:                   row.append(s)
        rows.append(row)

    # Use the first complete row as a reference query (goals are real values there)
    ref_row = rows[0]
    p, reason = justify(header, rows, ref_row)
    print(f"{reason}")
