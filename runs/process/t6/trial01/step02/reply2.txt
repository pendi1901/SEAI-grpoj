import ezr
from math import exp, sqrt

def guess(header, train, test):
    """Estimate each goal column for a test row using tree-based prediction."""
    # Identify goal columns (end with '-' or '+') and their order
    goals = [(i, header[i].endswith('-')) for i in range(len(header))
              if header[i].endswith('-') or header[i].endswith('+')]

    # Build a table from training data using ezr's infrastructure
    tbl = ezr.Tbl(train)

    # Train a tree on the full training set (using all x columns, goals as y)
    node = ezr.tree(tbl, tbl.rows)

    def predict_row(row):
        """Walk row to its leaf and return mean ydist of that leaf."""
        leaf_node = ezr.leaf(node, row)
        return leaf_node.mu

    # For each test row, we need to estimate the FIRST goal column's value.
    # Since goals are hidden as "?" in test rows, we predict ydist first,
    # then map it back to an estimated goal value using the training distribution.
    
    # Compute per-goal-column statistics from training data
    goal_stats = []
    for gi, minimize in goals:
        vals = [r[gi] if isinstance(r[gi], (int, float)) else 0.0 
                for r in train]
        mean_val = sum(vals) / max(1, len(vals))
        var_val = sum((v - mean_val) ** 2 for v in vals) / max(1, len(vals))
        std_val = sqrt(max(0.001, var_val))
        goal_stats.append((mean_val, std_val, minimize))

    results = []
    for row in test:
        # Predict ydist for this test row
        pred_ydist = predict_row(row)
        
        # Map predicted ydist back to estimated first goal value
        if goals:
            gi, minimize = goals[0]
            mean_val, std_val, _ = goal_stats[0]
            
            # Use the training distribution to map ydist -> estimated value
            # We approximate by assuming a linear-ish relationship
            # between ydist and normalized goal values from training
            train_ydists = []
            for r in train:
                val = r[gi] if isinstance(r[gi], (int, float)) else 0.0
                z = (val - mean_val) / std_val
                normed = 1.0 / (1.0 + exp(-1.7 * z))
                ideal = 0.0 if minimize else 1.0
                train_ydists.append(abs(normed - ideal))
            
            # Fit a simple linear regression: ydist -> goal value
            n = len(train)
            sum_xy = sum(ty * (r[gi] if isinstance(r[gi], (int, float)) else 0.0) 
                         for ty, r in zip(train_ydists, train))
            sum_x2 = sum(ty ** 2 for ty in train_ydists)
            sum_y2 = sum((r[gi] if isinstance(r[gi], (int, float)) else 0.0) ** 2 
                         for r in train)
            
            slope = (n * sum_xy - n * mean_val * pred_ydist) / \
                    (n * sum_x2 - n * sum(train_ydists) ** 2 + 1e-32)
            intercept = mean_val - slope * pred_ydist
            
            est = intercept + slope * pred_ydist
        else:
            est = 0.0
        
        results.append(float(est))

    return results


if __name__ == "__main__":
    import os
    path = os.path.join(os.path.dirname(__file__) or ".", "pom3a.train.csv")
    with open(path, encoding="utf-8-sig") as f:
        lines = [l.strip() for l in f if l.strip()]
    header = lines[0].split(",")
    rows = []
    for line in lines[1:]:
        cells = line.split(",")
        row = []
        for c in cells:
            s = c.strip()
            try: row.append(float(s))
            except ValueError:
                if s.lower() == "true": row.append(True)
                elif s.lower() == "false": row.append(False)
                else: row.append(s)
        rows.append(row)

    best = guess(header, rows, [])  # train on all data; no test rows needed for demo
    print(f"Estimated goals for {len(best)} rows:")
    for i, v in enumerate(best):
        print(f"  Row {i}: first goal est. = {v:.4f}")
