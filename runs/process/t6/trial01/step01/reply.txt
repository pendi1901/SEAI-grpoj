import ezr
from math import exp, sqrt

def remember(header, rows):
    """Return positions (0-based) of rows unusually good on all goal columns."""
    n = len(rows)
    if n == 0: return []

    # Identify goal columns and their ideal targets
    goals = [(i, header[i].endswith('-')) for i in range(len(header))
              if header[i].endswith('-') or header[i].endswith('+')]
    if not goals: return [0]

    # Compute mean per goal column (skip '?' cells)
    goal_means = []
    for gi, _ in goals:
        vals = [r[gi] for r in rows if isinstance(r[gi], (int, float))]
        goal_means.append(sum(vals) / max(1, len(vals)))

    # Compute std per goal column
    goal_sds = []
    for gi, _ in goals:
        vals = [r[gi] for r in rows if isinstance(r[gi], (int, float))]
        m = sum(vals) / len(vals)
        var = sum((v - m) ** 2 for v in vals) / max(1, len(vals))
        goal_sds.append(max(0.001, sqrt(var)))

    # Score each row: total distance from ideal (low is good for '-', high for '+')
    scores = []
    for row in rows:
        s = 0.0
        for gi, minimize in goals:
            val = row[gi] if isinstance(row[gi], (int, float)) else goal_means[goals.index((gi, _))]
            z = (val - goal_means[goals.index((gi, _))]) / goal_sds[goals.index((gi, _))]
            normed = 1.0 / (1.0 + exp(-1.7 * z))
            ideal = 0.0 if minimize else 1.0
            s += abs(normed - ideal)
        scores.append(s)

    # Return rows in the bottom ~5% of total distance
    threshold = sorted(scores)[max(1, int(0.05 * n))]
    return [i for i, sc in enumerate(scores) if sc <= threshold]


if __name__ == "__main__":
    import os
    path = os.path.join(os.path.dirname(__file__) or ".", "pom3a.train.csv")
    with open(path, encoding="utf-8-sig") as f:
        lines = [l.strip() for l in f if l.strip()]
    header = lines[0].split(",")
    rows = []
    for line in lines[1:]:
        cells = line.split(",")
        row = []
        for c in cells:
            s = c.strip()
            try: row.append(float(s))
            except ValueError:
                if s.lower() == "true": row.append(True)
                elif s.lower() == "false": row.append(False)
                else: row.append(s)
        rows.append(row)

    best = remember(header, rows)
    print(f"Best {len(best)} of {len(rows)} rows:")
    for i in sorted(best):
        print(f"  Row {i}: Cost-={rows[i][2]:.4f} Score-={rows[i][9]:.4f} Idle-={rows[i][11]:.4f}")
