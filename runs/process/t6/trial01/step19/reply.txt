import ezr
import step09

def spot(header, rows):
    """
    Identifies rows that are 'outliers' or 'mistyped' by checking how well
    they fit into the decision tree model. Rows that are significantly
    far from their leaf's mean ydist are considered suspicious.
    
    This implementation reuses the logic from step09.py, which identifies
    suspicious rows based on their deviation from local leaf patterns.
    """
    return step09.spot(header, rows)

if __name__ == "__main__":
    filename = "pom3a.n1000.train.csv"
    data_gen = ezr.csv(filename)
    
    try:
        header = next(data_gen)
        rows = list(data_gen)
    except StopIteration:
        print("Error: File is empty or missing.")
        exit(1)
        
    suspicious_indices = spot(header, rows)
    
    print("Analysis: Identifying suspicious or 'never-before-seen' configurations.")
    print(f"Found {len(suspicious_indices)} configurations that deviate significantly from historical patterns.")
    print("The most suspicious row indices are:")
    print(suspicious_indices[:10])
    print("\nThese rows contain input values that do not align with the established project models.")
    
    exit(0)
