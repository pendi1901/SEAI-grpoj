import ezr
import step01

def flow(header, row, seen):
    """
    Decide whether to approve a row.
    We approve if the row is in the top 5% of the distribution 
    of the currently seen data.
    """
    if not seen:
        return False
    
    # Combine seen rows with the current row to evaluate its relative quality
    all_rows = seen + [row]
    tbl = ezr.Tbl([header] + all_rows)
    
    # Calculate ydist for all seen rows plus the current one
    # Lower ydist is better.
    scored = [(ezr.ydist(tbl, r), i) for i, r in enumerate(all_rows)]
    scored.sort(key=lambda x: x[0])
    
    # If the current row (last in the list) is in the top 5% of all seen so far, approve it.
    limit = max(1, len(all_rows) // 20)
    top_indices = [idx for _, idx in scored[:limit]]
    
    return (len(all_rows) - 1) in top_indices

if __name__ == "__main__":
    filename = "pom3a.n1000.train.csv"
    data = list(ezr.csv(filename))
    header = data[0]
    rows = data[1:]
    
    approved = []
    seen = []
    
    for row in rows:
        if flow(header, row, seen):
            approved.append(row)
        seen.append(row)
        
    print(f"Decision process complete.")
    print(f"Total rows processed: {len(rows)}")
    print(f"Total rows approved: {len(approved)}")
    
    if approved:
        goal_indices = [i for i, name in enumerate(header) if name.endswith(('-', '+'))]
        print(f"Average goals of approved rows:")
        for idx in goal_indices:
            avg = sum(r[idx] for r in approved) / len(approved)
            print(f"  {header[idx]}: {avg:.4f}")
