import ezr
import step01

def fix(header, row, levers):
    """
    Finds the smallest change to levers that moves a bad row toward the 
    best-performing region of the data.
    """
    # We want to move the row towards the 'best' region.
    # We define 'best' as the average of the top 5% of rows.
    data = list(ezr.csv("pom3a.n1000.train.csv"))
    rows = data[1:]
    tbl = ezr.Tbl(data)
    
    # Identify best rows
    best_indices = step01.remember(header, rows)
    best_rows = [rows[i] for i in best_indices]
    
    # Calculate the centroid of the best rows for the levers
    recommendations = {}
    for col_name, allowed in levers.items():
        idx = header.index(col_name)
        col = tbl.cols[idx]
        
        if isinstance(col, ezr.Num):
            # Target the mean of the best rows for this numeric lever
            target = sum(r[idx] for r in best_rows) / len(best_rows)
            # Clamp to allowed range
            low, high = allowed
            recommendations[col_name] = max(low, min(high, target))
        else:
            # For symbolic, pick the most frequent value in the best rows
            counts = {}
            for r in best_rows:
                val = r[idx]
                counts[val] = counts.get(val, 0) + 1
            best_val = max(counts, key=counts.get)
            if best_val in allowed:
                recommendations[col_name] = best_val
                
    return recommendations

if __name__ == "__main__":
    # Load data
    data = list(ezr.csv("pom3a.n1000.train.csv"))
    header = data[0]
    rows = data[1:]
    
    # Define levers: Plan and Team Size
    # Plan is numeric (0-1), Team Size is numeric (0-1)
    levers = {
        "Plan": (0.0, 1.0),
        "Team Size": (0.0, 1.0)
    }
    
    # Pick a "bad" row (e.g., one with high ydist)
    tbl = ezr.Tbl(data)
    scored = sorted([(ezr.ydist(tbl, r), r) for r in rows], key=lambda x: x[0])
    bad_row = scored[-1][1]
    
    # Get recommendation
    rec = fix(header, bad_row, levers)
    
    print("Rescue Plan for under-performing project:")
    for col, val in rec.items():
        print(f"  Adjust '{col}' to {val:.4f}")
    print("\nBy shifting these parameters toward the values seen in our top-performing "
          "projects, we can optimize resource allocation and reduce idle time.")
