import ezr
import step03

def certify(xs, ys):
    """
    Returns True if the distributions are statistically indistinguishable,
    False if they are significantly different.
    Uses the same/cliffs/ks tests from ezr.
    """
    return ezr.same(xs, ys)

def get_results_by_period(header, rows):
    """
    Splits rows into two periods (first half vs second half) 
    and extracts the ydist values for each.
    """
    tbl = ezr.Tbl([header] + rows)
    mid = len(rows) // 2
    
    period1 = [ezr.ydist(tbl, r) for r in rows[:mid]]
    period2 = [ezr.ydist(tbl, r) for r in rows[mid:]]
    
    return period1, period2

if __name__ == "__main__":
    file_path = "pom3a.n1000.train.csv"
    data = list(ezr.csv(file_path))
    header = data[0]
    rows = data[1:]
    
    # Get performance distributions for the two periods
    p1_results, p2_results = get_results_by_period(header, rows)
    
    # Perform the statistical test
    is_same = certify(p1_results, p2_results)
    
    print("Analysis: Did the team's decision-making shift in the second half?")
    print(f"Period 1 count: {len(p1_results)}, Period 2 count: {len(p2_results)}")
    print(f"Average ydist (Period 1): {sum(p1_results)/len(p1_results):.4f}")
    print(f"Average ydist (Period 2): {sum(p2_results)/len(p2_results):.4f}")
    
    if is_same:
        print("Result: The difference is statistically insignificant.")
        print("Conclusion: The results are effectively the same as before.")
    else:
        print("Result: The difference is statistically significant.")
        print("Conclusion: The results have shifted.")
    
    exit(0)
