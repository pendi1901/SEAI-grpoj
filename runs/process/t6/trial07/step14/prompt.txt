def say(x: Any, p: int = 2) -> str:
  "Floats get P decimals; dicts list their keys."
  if isinstance(x,float): x= f"{x:.{p}f}".rstrip("0").rstrip(".")
  if isinstance(x, dict):
    x= "{"+", ".join(f"{k}: {say(v,p)}" for k,v in x.items())+"}"
  return str(x)

class o(dict):
  "A dict you can poke with a dot."
  __getattr__, __setattr__ = dict.__getitem__, dict.__setitem__
  __repr__ = say

def atom(s: str) -> ATOM:
  "'22' -> 22. '3.1' -> 3.1. 'True' -> True. 'x' -> 'x'."
  for fn in (int, float):
    try: return fn(s)
    except ValueError: pass
  s = s.strip()
  return {"True": True, "False": False}.get(s, s)

def csv(file: str) -> Iterator[ROW]:
  "ROWS of FILE, each cell coerced."
  with open(os.path.expanduser(file),  # -sig drops any BOM
           encoding="utf-8-sig") as f:
    for line in f:
      if line.strip(): yield [atom(s) for s in line.split(",")]

def Col(txt: str = " ", at: int = 0) -> COL:
  "Factory that returns NUM or SYM."
  return Num(txt,at) if txt[0].isupper() else Sym(txt,at)

class Num:
  "Place to summarize stream of numbers."
  __slots__ = ("at", "txt", "n", "mu", "m2", "sd", "goal")
  def __init__(self, txt: str = " ", at: int = 0) -> None:
    self.at, self.txt, self.n = at, txt, 0
    self.mu = self.m2 = self.sd = 0
    self.goal = 0 if txt[-1] == "-" else 1
  __repr__ = lambda self: say({k: getattr(self,k)
                               for k in self.__slots__})

class Sym:
  "Place to summarize stream of Symbols.  The class is the tag."
  __slots__ = ("at", "txt", "n", "has")
  def __init__(self, txt: str = " ", at: int = 0) -> None:
    self.at, self.txt, self.n, self.has = at, txt, 0, {}
  __repr__ = lambda self: say({k: getattr(self,k)
                               for k in self.__slots__})

def add(col: COL, v: ATOM, inc: int = 1) -> None:
  "Show V to COL.  INC=-1 takes it away.  `?` changes nothing."
  if v != "?":
    col.n += inc
    if type(col) is Sym:
      col.has[v] = col.has.get(v, 0) + inc
    else:
      d = v - col.mu                      # Welford, for `div`
      col.mu += inc * d / max(1, col.n)
      col.m2  = max(0, col.m2 + inc * d * (v - col.mu))
      col.sd  = 0 if col.n < 2 else (col.m2/(col.n-1))**.5

def mid(col: COL) -> ATOM:
  "Middle: the mean, or the most common symbol; ? if none."
  return (max(col.has, key=col.has.get) if col.has else "?"
          ) if type(col) is Sym else col.mu

def mids(tbl: TBL) -> MIDS:
  "Every column's middle, cached until the next addCols."
  tbl.mids = tbl.mids or {at: mid(c) for at,c in tbl.cols.items()}
  return tbl.mids

def div(col: COL) -> float:
  "Spread: entropy, or standard deviation."
  return (-sum(v/col.n*log2(v/col.n) for v in col.has.values())
          if type(col) is Sym else col.sd)  # sd kept by `add`

def norm(col: COL, v: ATOM) -> ATOM:
  "To 0..1, by the logistic curve.  Symbols do not scale."
  if type(col) is Sym: return v
  z = max(-3, min(3, (v - col.mu) / (1e-32 + div(col))))
  return 1 / (1 + exp(-1.7 * z))

#-- tables -------------------------------------------------
def Tbl(src: Iterable[ROW]) -> TBL:
  "Header names the columns: X skip, +- goal, ! klass."
  src = iter(src)
  tbl = Cols(o(rows=[], cols={}, x=[], y=[], mids=None,
               names=next(src)))
  for row in src: addRow(tbl, row)
  return tbl

def Cols(tbl: TBL) -> TBL:
  "Create column roles."
  for at, s in enumerate(tbl.names):
    if s[-1] == "X": continue            # skip me entirely
    tbl.cols[at] = Col(s,at)
    (tbl.y if s[-1] in "+-!" else tbl.x).append(tbl.cols[at])
  return tbl

def addRow(tbl: TBL, row: ROW) -> ROW:
  "Keep ROW, and show it to my columns."
  tbl.rows += [row]
  return addCols(tbl, row)

def addCols(tbl: TBL, row: ROW, inc: int = 1) -> ROW:
  "Show ROW to my columns, without keeping it."
  tbl.mids = None                 # a new row moves the middles
  for at, col in tbl.cols.items(): add(col, row[at], inc)
  return row

def clone(tbl: TBL, rows: ROWS = None) -> TBL:
  "An empty copy of TBL, plus ROWS."
  return Tbl([tbl.names] + (rows or []))

def dist(vs: Iterable[QTY], n: int) -> float:
  "Root mean square of N gaps."
  return sqrt(sum(v*v for v in vs) / n)

def ydist(tbl: TBL, row: ROW) -> float:
  "How far ROW's goals are from the best they could be."
  return dist((abs(norm(c, row[c.at]) - c.goal)
                    for c in tbl.y), len(tbl.y))

def gap(col: COL, a: ATOM, b: ATOM) -> QTY:
  "Distance between two values of one column."
  if a == "?" or b == "?": return 1    # unknown = far
  return a!=b if type(col) is Sym else abs(norm(col,a)-norm(col,b))

def xdist(tbl: TBL, r1: ROW, r2: ROW | MIDS) -> float:
  "How far apart two rows are, over the x columns."
  return dist((gap(c, r1[c.at], r2[c.at])
                    for c in tbl.x), len(tbl.x))

def oracle(row: ROW) -> ROW:
  """Labeller. Rows here arrive with ys, so nothing to do.
  Real oracles fill in any missing y cells (and can close
  over a TBL to know which cells those are)."""
  return row

def acquire(tbl: TBL, cap: int | None = None,
            label: ORACLE = oracle) -> ROWS:
  "Label near the best rows, far from the rest."
  cap  = cap or the.Stop
  todo = random.sample(tbl.rows, len(tbl.rows))[:the.Few]
  both = clone(tbl, [label(todo.pop())
                     for _ in range(the.Start)])
  both.rows.sort(key=lambda r: ydist(both, r))
  n    = int(sqrt(the.Start))
  best,rest = clone(tbl,both.rows[:n]), clone(tbl,both.rows[n:])
  while todo and len(both.rows) < cap:
    cb, cr = mids(best), mids(rest)    # two centroids
    todo.sort(key=lambda z: xdist(tbl,z,cr) - xdist(tbl,z,cb))
    addRow(best, addRow(both, label(todo.pop())))
    best.rows.sort(key=lambda z: ydist(both, z))  # no y leak
    if len(best.rows) > sqrt(len(both.rows)): # best stays small
      addRow(rest, addCols(best, best.rows.pop(), -1))
  return both.rows

def grabs(tbl: TBL, cap: int | None = None,
          label: ORACLE = oracle) -> ROWS:
  "The straw man: same budget, taken at random."
  cap = cap or the.Stop
  return [label(r) for r in
          random.sample(tbl.rows, min(cap, len(tbl.rows)))]

# Calls core.  Nothing above this line calls down here.
# NODE = o(rows, at, v, go, kids, mu).
def cut(tbl: TBL, rows: ROWS) -> tuple[int,ATOM,GO]|None:
  "The (at, v, go) whose two sides have the tightest ys."
  ys = [ydist(tbl, r) for r in rows]
  out, least = None, 1e30
  for col in tbl.x:
    for v, go in candidates(col, rows):
      a, b = Num(), Num()   # one pass, so one `go` per row
      for r,y in zip(rows,ys): add(a if go(r) else b, y)
      if a.n >= the.Leaf and b.n >= the.Leaf:
        score = (div(a)*a.n + div(b)*b.n) / (a.n + b.n)
        if score < least: out, least = (col.at, v, go), score
  return out

def candidates(col: COL, rows: ROWS) -> Iterator[tuple[ATOM,GO]]:
  "What splits to try: one per symbol, or CUTS per num."
  at = col.at
  if type(col) is Sym:
    for v in sorted({r[at] for r in rows if r[at] != "?"}):
      yield v, lambda r,v=v,at=at: r[at] == v
  else:                # by rank, not value: THESE rows are dense
    xs = sorted(r[at] for r in rows if r[at] != "?")
    k  = max(1, len(xs)//the.Cuts)     # cuts, spaced by rank
    for v in xs[k-1:-1:k]:
      yield v, (lambda r,v=v,at=at: r[at] != "?" and r[at] <= v)

def tree(tbl: TBL, rows: ROWS) -> NODE:
  "Split while a split keeps both sides big enough."
  node = o(rows=rows, at=None, v=None, go=None, kids=[],
           mu=sum(ydist(tbl,r) for r in rows)/len(rows))
  if len(rows) > the.Leaf and (found := cut(tbl, rows)):
    at, v, go = found
    yes, no = [], []
    for r in rows: (yes if go(r) else no).append(r)
    node.at, node.v, node.go = at, v, go # cut kepts sides big
    node.kids = [tree(tbl, yes), tree(tbl, no)]
  return node

def leaf(node: NODE, row: ROW) -> NODE:
  "Walk ROW down to its leaf."
  while node.kids: node = node.kids[0 if node.go(row) else 1]
  return node

def leafs(node: NODE) -> list[NODE]:
  "Return leaves of this tree."
  return [x for k in node.kids for x in leafs(k)] or [node]

def show(tbl: TBL, node: NODE, pre: str | None = None,
         edge: str = "") -> None:
  "Tree on the left; ydist and n on the right."
  ls = sorted(leafs(node), key=lambda z: z.mu)
  def walk(z: NODE, pre: str | None, edge: str) -> None:
    m = "+" if z is ls[0] else "-" if z is ls[-1] else " "
    print(f"{m} {round(100*z.mu):>4} {len(z.rows):>5}   "
          f"{(pre or '')+edge}".rstrip())
    op = "=" if type(tbl.cols[z.at or 0]) is Sym else "<="
    no = "!=" if op == "=" else ">"
    for kid, oper in zip(z.kids, [op, no]):
      walk(kid, "" if pre is None else pre + "|  ",
           f"{tbl.names[z.at]} {oper} {say(z.v)}")
  print(" ydist     n")
  walk(node, pre, edge)

# Calls core, acquire, and one RANKER -- tree's, by default.
def ranker(lab: TBL) -> Callable[[ROW], float]:
  "Grow a tree on LAB; score any row by its leaf's mean ydist."
  node = tree(lab, lab.rows)
  return lambda r: leaf(node, r).mu

def wins(tbl: TBL) -> Callable[[ROW], float]:
  "100 at the best row, 0 at an average one."
  ys = sorted(ydist(tbl, r) for r in tbl.rows)
  lo, avg = ys[0], sum(ys)/len(ys)
  return lambda r: max(-100, min(100, 100*(1 - (ydist(tbl,r)-lo)
                                            / (avg-lo+1e-32))))

def holdout(tbl: TBL, pick: PICKER = acquire,
            rank: RANKER = ranker,
            label: ORACLE = oracle) -> ROW:
  "Train on half; of CHECK guesses on the rest, pick best."
  rows  = random.sample(tbl.rows, len(tbl.rows))
  n     = len(rows)//2
  tr    = clone(tbl, rows[:n][:the.Few])
  lab   = clone(tbl, pick(tr, the.Stop - the.Check, label))
  top   = sorted(rows[n:], key=rank(lab))
  return min(top[:the.Check], key=lambda r: ydist(lab, label(r)))

# A leaf: it calls core, and nothing here calls it.
def adds(vs: Iterable[ATOM], col: COL | None = None) -> COL:
  "Every item of VS into COL."
  col = Num() if col is None else col
  for v in vs: add(col, v)
  return col

def cohen(xs: NUMS, ys: NUMS, d: float = 0.35,
          eps: float = 0) -> bool:
  "Are the means within D pooled standard deviations?"
  a, b = adds(xs), adds(ys)
  pool = ((a.n-1)*div(a)**2 + (b.n-1)*div(b)**2)/(a.n+b.n-2)
  return abs(a.mu - b.mu) <= max(eps, d * sqrt(pool))

def cliffs(xs: NUMS, ys: NUMS, d: float = 0.197) -> bool:
  "Sorted in.  Is the rank imbalance small enough?"
  gt = lt = j = k = 0
  for x in xs:
    while j < len(ys) and ys[j] <  x: j += 1; k = j
    while k < len(ys) and ys[k] <= x: k += 1
    gt += j; lt += len(ys) - k
  return abs(gt - lt) / (len(xs) * len(ys)) <= d

def ks(xs: NUMS, ys: NUMS, a: float = 1.36) -> bool:
  "Sorted in.  95% Kolmogorov-Smirnov."
  n, m, i, j, d = len(xs), len(ys), 0, 0, 0
  while i < n and j < m:
    v = min(xs[i], ys[j])
    while i < n and xs[i] <= v: i += 1
    while j < m and ys[j] <= v: j += 1
    d = max(d, abs(i/n - j/m))
  return d <= a * sqrt((n + m) / (n * m))

def same(xs: NUMS, ys: NUMS, eps: float = 0) -> bool:
  """Indistinguishable.  Always cliffs and ks; EPS adds cohen, and
  doubles as its effect size (eps=0 skips that third test)."""
  xs, ys = sorted(xs), sorted(ys)
  return (cliffs(xs,ys) and ks(xs,ys)
          and (not eps or cohen(xs,ys,d=eps)))

import ezr
import random

def remember(header, rows):
    """
    Identify the top 5% of rows that perform best across all goals.
    Goals ending in '-' are minimized, '+' are maximized.
    """
    # Create a table object to use ezr's utility functions
    # We construct the input as [header] + rows
    data = [header] + rows
    tbl = ezr.Tbl(data)
    
    # Calculate ydist for every row.
    # ydist calculates how far a row's goals are from the ideal (0 for -, 1 for +).
    # Lower ydist is better.
    scored_rows = []
    for i, row in enumerate(rows):
        d = ezr.ydist(tbl, row)
        scored_rows.append((d, i))
    
    # Sort by distance (ascending, as lower is better)
    scored_rows.sort(key=lambda x: x[0])
    
    # Return the top 5% (or at least 1)
    limit = max(1, len(rows) // 20)
    return [item[1] for item in scored_rows[:limit]]

import ezr
import step01

def guess(header, train, test):
    """
    Estimates the first goal column for each test row.
    Uses the tree-based ranker from ezr to find the leaf mean of the first goal.
    """
    # Create a table for the training data
    train_tbl = ezr.Tbl([header] + train)
    
    # Identify the index of the first goal column
    # Goals are columns ending in '-' or '+'
    goal_indices = [i for i, name in enumerate(header) if name.endswith(('-', '+'))]
    first_goal_idx = goal_indices[0]
    
    # Build a tree on the training data
    # We need to predict the value of the first goal column
    # We can adapt the tree structure to store the mean of the first goal
    # instead of the mean of ydist.
    
    def get_goal_mean(node):
        vals = [r[first_goal_idx] for r in node.rows]
        return sum(vals) / len(vals)

    # Build a custom tree where each node stores the mean of the first goal
    def build_tree(tbl, rows):
        node = ezr.o(rows=rows, at=None, v=None, go=None, kids=[], 
                     mu=sum(r[first_goal_idx] for r in rows)/len(rows))
        if len(rows) > ezr.the.Leaf and (found := ezr.cut(tbl, rows)):
            at, v, go = found
            yes, no = [], []
            for r in rows: (yes if go(r) else no).append(r)
            node.at, node.v, node.go = at, v, go
            node.kids = [build_tree(tbl, yes), build_tree(tbl, no)]
        return node

    tree_root = build_tree(train_tbl, train)
    
    # For each test row, walk the tree to the leaf and return the leaf's mean
    results = []
    for row in test:
        node = ezr.leaf(tree_root, row)
        results.append(node.mu)
        
    return results

if __name__ == "__main__":
    # Load data
    file_path = "pom3a.n1000.train.csv"
    data = list(ezr.csv(file_path))
    header = data[0]
    rows = data[1:]
    
    # Split: 800 training, 200 testing

import ezr
import step01

def certify(xs, ys):
    """
    Returns True if the distributions are statistically indistinguishable,
    False if they are significantly different.

import ezr
import step01

def flow(header, row, seen):
    """
    Decide whether to approve a row on the spot.
    We approve if the row is in the top 5% of the expected distribution
    based on the data seen so far.
    """
    if not seen:
        return False
    
    # Create a table from seen rows to calculate ydist
    # We use the header to define the columns
    tbl = ezr.Tbl([header] + seen)
    
    # Calculate ydist for all seen rows
    seen_ydists = [ezr.ydist(tbl, r) for r in seen]
    
    # Calculate ydist for the current row
    # Note: row might have '?' for goals, but ydist handles that via norm
    current_ydist = ezr.ydist(tbl, row)
    
    # We define "top 5%" as being better than the 5th percentile of seen rows
    seen_ydists.sort()
    threshold_idx = max(0, len(seen_ydists) // 20)
    threshold = seen_ydists[threshold_idx]
    
    # Approve if current row is better (lower ydist) than the threshold
    return current_ydist <= threshold

import ezr
import step01

def blame(header, rows):
    """
    Identify input columns that best explain why the worst rows are bad.
    We define 'worst' as the bottom 10% of rows by ydist.
    We then use the tree-based split logic to find which input columns 
    most frequently appear in the branches leading to these worst rows.
    """
    tbl = ezr.Tbl([header] + rows)
    
    # Calculate ydist for all rows and identify the worst 10%
    scored = sorted([(ezr.ydist(tbl, r), r) for r in rows], key=lambda x: x[0], reverse=True)
    worst_rows = [x[1] for x in scored[:max(1, len(rows) // 10)]]
    
    # Build a tree on the whole dataset to see how it partitions the space
    root = ezr.tree(tbl, rows)
    
    # Count how often each input column is used in a split that leads to a 'worst' leaf
    # A 'worst' leaf is one where the mean ydist is high
    counts = {col.txt: 0 for col in tbl.x}
    
    def traverse(node):
        if not node.kids:
            # If this leaf is in the worst 10% of all leaves by ydist, count its ancestors
            return
        
        # Check if this node's split is relevant to the worst rows
        # We look for nodes that contain a high density of worst_rows
        for kid in node.kids:
            traverse(kid)
            
    # Simpler approach: find the leaves that contain the worst rows
    # and count the columns used in the path to those leaves
    def get_path_cols(node, row, path):
        if not node.kids:
            return path
        path.append(header[node.at])
        return get_path_cols(node.kids[0] if node.go(row) else node.kids[1], row, path)

    for row in worst_rows:
        for col_name in get_path_cols(root, row, []):
            counts[col_name] += 1
            
    # Return columns sorted by frequency of appearance in paths to worst rows
    sorted_cols = sorted(counts.items(), key=lambda x: x[1], reverse=True)
    return [name for name, count in sorted_cols if count > 0]

import ezr

def justify(header, rows, row):
    """
    Returns a pair (p, reason): p is the fraction of rows that 'row' is better than;
    reason is a sentence explaining the ranking.
    """
    # Create a table to handle normalization and distance calculations
    tbl = ezr.Tbl([header] + rows)
    
    # Estimate the goals for the input row using the tree-based approach from step02
    # We need to fill in the '?' goals to calculate ydist
    goal_indices = [i for i, name in enumerate(header) if name.endswith(('-', '+'))]
    
    # Build a tree to estimate the goal values
    # We use the tree to predict the ydist directly
    def get_ydist_mean(node):
        return sum(ezr.ydist(tbl, r) for r in node.rows) / len(node.rows)

    def build_tree(tbl, rows):
        node = ezr.o(rows=rows, at=None, v=None, go=None, kids=[], 
                     mu=sum(ezr.ydist(tbl, r) for r in rows)/len(rows))
        if len(rows) > ezr.the.Leaf and (found := ezr.cut(tbl, rows)):
            at, v, go = found
            yes, no = [], []
            for r in rows: (yes if go(r) else no).append(r)
            node.at, node.v, node.go = at, v, go
            node.kids = [build_tree(tbl, yes), build_tree(tbl, no)]
        return node

    tree_root = build_tree(tbl, rows)
    
    # Estimate the ydist for the target row
    estimated_ydist = ezr.leaf(tree_root, row).mu
    
    # Calculate how many rows are worse than this estimated ydist
    # A row is better if its ydist is lower.
    # We are better than a row if our ydist is lower than theirs.
    all_ydists = [ezr.ydist(tbl, r) for r in rows]
    better_than_count = sum(1 for d in all_ydists if d > estimated_ydist)
    p = better_than_count / len(rows)
    
    reason = f"This setup is estimated to be in the top {int((1-p)*100)}% of all possible configurations based on its input characteristics."
    
    return p, reason

import ezr

def choose(header, rows, label):
    """
    Find the best row using a budget of 50 labels.
    We use the acquire logic from ezr to sample the space effectively.
    The error in the previous version occurred because '?' strings were 
    passed to norm() during ydist calculation. We must ensure that 
    only fully labeled rows are used for distance calculations.
    """
    # Create a table structure to manage the data
    tbl = ezr.Tbl([header] + rows)
    
    # We need to track which rows have been labeled to avoid passing '?' to ydist
    labeled_data = []

    def oracle_wrapper(row):
        # Find the index of this row in the original list
        idx = rows.index(row)
        # Get the true values
        true_row = label(idx)
        # Update the row in place
        for i in range(len(row)):
            row[i] = true_row[i]
        # Keep track of fully labeled rows for distance calculations
        labeled_data.append(row)
        return row

    # Use the acquire function to pick and label up to 50 rows
    # We pass the oracle_wrapper which fills the row with real data
    ezr.acquire(tbl, cap=50, label=oracle_wrapper)
    
    # After labeling, find the best row among the labeled ones.
    # We use the table 'tbl' which now has the updated stats from the labeled rows.
    # A row is better if its ydist is lower.
    best_row = min(labeled_data, key=lambda r: ezr.ydist(tbl, r))
    
    # Return the index of the best row found
    return rows.index(best_row)

def fix(header, row, levers):
    """
    Find the best change to a bad row to improve its goals.
    We use the tree-based model to find a path to a better leaf,
    then suggest a change to a lever that moves the row toward that leaf.
    """

    file_path = "pom3a.n1000.train.csv"
    data = list(ezr.csv(file_path))
    header_data = data[0]
    rows = data[1:]

    # Build a tree to understand the decision space
    root = ezr.tree(tbl, rows)
    
    # Find the leaf the current row falls into
    node = ezr.leaf(root, row)
    
    # Find a better leaf (lower ydist)
    all_leaves = ezr.leafs(root)
    better_leaves = [l for l in all_leaves if l.mu < node.mu]
    
    if not better_leaves:
        return {}
    
    # Pick the best leaf
    target = min(better_leaves, key=lambda l: l.mu)
    
    # Find a lever that can be changed to move toward the target leaf
    # We look for a split in the path to the target leaf that involves a lever
    recommendations = {}
    
    # Simple heuristic: find a lever that is in the path to the target
    # and suggest a value that satisfies the split condition
    for col_name, allowed in levers.items():
        if col_name in header:

                recommendations[col_name] = low + (high - low) * 0.5
            else: # Symbolic
                recommendations[col_name] = random.choice(allowed)
            break # Only need one change
            
    return recommendations

import ezr

def spot(header, rows):
    """
    Identify rows that do not fit the data distribution.
    We use the tree-based model to identify rows that are outliers
    relative to their predicted ydist (performance).
    """
    tbl = ezr.Tbl([header] + rows)
    
    # Build a tree to model the expected performance (ydist) based on inputs
    root = ezr.tree(tbl, rows)
    
    # Calculate the residual: the difference between actual ydist and 
    # the mean ydist of the leaf the row falls into.
    # A large residual indicates the row does not fit the local pattern.
    residuals = []
    for i, row in enumerate(rows):
        actual = ezr.ydist(tbl, row)
        leaf_node = ezr.leaf(root, row)
        # Residual is the absolute difference from the leaf mean
        residual = abs(actual - leaf_node.mu)
        residuals.append((residual, i))
    
    # Sort by residual descending (most suspicious first)
    residuals.sort(key=lambda x: x[0], reverse=True)
    
    # Return the indices of the top 10 most suspicious rows
    return [item[1] for item in residuals[:10]]

import ezr
import random

def race(header, rows, label):
    """
    Compare three strategies: random, acquire, and hill-climbing.
    Budget is 50 labels total.
    """
    budget = 50
    labeled = {}

    def get_label(i):
        if i not in labeled:
            if len(labeled) >= budget:
                return None
            labeled[i] = label(i)
        return labeled[i]

    # Strategy 1: Random
    def strategy_random():
        indices = random.sample(range(len(rows)), 16)
        for i in indices: get_label(i)
        return min(indices, key=lambda i: ezr.ydist(ezr.Tbl([header] + [labeled[k] for k in labeled]), labeled[i]))

    # Strategy 2: Acquire (ezr's built-in)
    def strategy_acquire():
        # We use a temporary table to guide acquisition
        tbl = ezr.Tbl([header] + rows)
        # Acquire uses its own internal labeler, we wrap it to use our budget
        def oracle_wrapper(r):
            idx = rows.index(r)
            return get_label(idx)
        
        subset = ezr.acquire(tbl, cap=17, label=oracle_wrapper)
        return rows.index(min(subset, key=lambda r: ezr.ydist(tbl, r)))

    # Strategy 3: Hill Climbing
    def strategy_hill():
        curr = random.randint(0, len(rows) - 1)
        for _ in range(17):
            get_label(curr)
            neighbors = random.sample(range(len(rows)), 5)
            best_n = min(neighbors, key=lambda n: ezr.ydist(ezr.Tbl([header] + [labeled[k] for k in labeled]), labeled.get(n, rows[n])))
            if ezr.ydist(ezr.Tbl([header] + [labeled[k] for k in labeled]), labeled.get(best_n, rows[best_n])) < \
               ezr.ydist(ezr.Tbl([header] + [labeled[k] for k in labeled]), labeled.get(curr, rows[curr])):
                curr = best_n
        return curr

    results = {
        "random": strategy_random(),
        "acquire": strategy_acquire(),
        "hill": strategy_hill()
    }

    # Evaluate winners based on true ydist
    tbl = ezr.Tbl([header] + rows)
    winner = min(results, key=lambda name: ezr.ydist(tbl, rows[results[name]]))
    
    return {"winner": winner, "row": results[winner], "tried": list(results.keys())}

import ezr
import step02

def guess(header, train, test):
    """
    Estimates the first goal column for each test row.
    To simulate requirements changing twice as fast halfway through,
    we weight the training data: rows from the second half of the 
    training set are given more influence (higher weight) in the 
    tree-based estimation model.
    """
    # Identify the index of the first goal column
    goal_indices = [i for i, name in enumerate(header) if name.endswith(('-', '+'))]
    first_goal_idx = goal_indices[0]
    
    # Create a table for the training data
    train_tbl = ezr.Tbl([header] + train)
    
    # Build a custom tree where each node stores the weighted mean of the first goal.
    # We weight rows from the second half (index >= len(train)/2) twice as much.
    def get_weighted_mean(rows):
        total_val = 0
        total_weight = 0
        mid = len(train) // 2
        for r in rows:
            # Find original index to determine if it's in the second half
            # Since we don't have indices, we use the row's position in the original train list
            # We can map rows back to their original index by reference or by searching
            weight = 2 if train.index(r) >= mid else 1
            total_val += r[first_goal_idx] * weight
            total_weight += weight
        return total_val / total_weight if total_weight > 0 else 0

    def build_tree(tbl, rows):
        node = ezr.o(rows=rows, at=None, v=None, go=None, kids=[], 
                     mu=get_weighted_mean(rows))
        if len(rows) > ezr.the.Leaf and (found := ezr.cut(tbl, rows)):
            at, v, go = found
            yes, no = [], []
            for r in rows: (yes if go(r) else no).append(r)
            node.at, node.v, node.go = at, v, go
            node.kids = [build_tree(tbl, yes), build_tree(tbl, no)]
        return node

    tree_root = build_tree(train_tbl, train)
    
    # For each test row, walk the tree to the leaf and return the leaf's weighted mean
    results = []
    for row in test:
        node = ezr.leaf(tree_root, row)
        results.append(node.mu)
        
    return results

    """
