import ezr
import step01

def certify(xs, ys):
    """
    Returns True if the distributions are statistically indistinguishable,
    False if they are significantly different.
    Uses the same/cliffs/ks tests from ezr.
    """
    return ezr.same(xs, ys)

def get_team_costs(header, rows):
    """
    Splits rows into small-team (Team Size < 0.5) and big-team (Team Size >= 0.5)
    and extracts the 'Cost-' values for each.
    """
    # Find indices for 'Team Size' and 'Cost-'
    team_idx = header.index('Team Size')
    cost_idx = header.index('Cost-')
    
    small, big = [], []
    for row in rows:
        if row[team_idx] < 0.5:
            small.append(row[cost_idx])
        else:
            big.append(row[cost_idx])
    return small, big

if __name__ == "__main__":
    # Load data
    file_path = "pom3a.n1000.train.csv"
    data = list(ezr.csv(file_path))
    header = data[0]
    rows = data[1:]
    
    small_costs, big_costs = get_team_costs(header, rows)
    
    # Perform the statistical test
    is_same = certify(small_costs, big_costs)
    
    print("Analysis: Are small-team setups cheaper than big-team ones?")
    print(f"Small team count: {len(small_costs)}, Big team count: {len(big_costs)}")
    print(f"Average Cost (Small): {sum(small_costs)/len(small_costs):.4f}")
    print(f"Average Cost (Big):   {sum(big_costs)/len(big_costs):.4f}")
    
    if is_same:
        print("Result: The difference is statistically insignificant (it was likely luck).")
    else:
        print("Result: The difference is statistically significant (it is real).")
    
    exit(0)
