import ezr
import step01

def remember(header, rows):
    """
    Find the past rows most similar to the current setup.
    The customer asks for past setups most like a proposed one.
    We interpret this as finding the rows that are most similar in the 
    input space (x-columns) to the best performing rows found in step01.
    """
    # Create table to handle distance calculations
    tbl = ezr.Tbl([header] + rows)
    
    # Identify the best rows (using the logic from step01)
    best_indices = step01.remember(header, rows)
    best_rows = [rows[i] for i in best_indices]
    
    # For each row in the dataset, find its distance to the set of best rows
    # We use xdist to compare input characteristics
    def dist_to_best(row):
        return min(ezr.xdist(tbl, row, best_row) for best_row in best_rows)
    
    # Rank all rows by how similar they are to the best ones
    scored = []
    for i, row in enumerate(rows):
        scored.append((dist_to_best(row), i))
    
    scored.sort(key=lambda x: x[0])
    
    # Return the top 5% of rows that are most similar to the best ones
    limit = max(1, len(rows) // 20)
    return [item[1] for item in scored[:limit]]

if __name__ == "__main__":
    file_path = "pom3a.n1000.train.csv"
    data = list(ezr.csv(file_path))
    header = data[0]
    rows = data[1:]
    
    similar_indices = remember(header, rows)
    
    print("Analysis: Past setups most similar to the best-performing configurations.")
    print("-" * 60)
    print(f"{'Index':<10} | {'Similarity (xdist to best)'}")
    print("-" * 60)
    
    tbl = ezr.Tbl([header] + rows)
    best_indices = step01.remember(header, rows)
    best_rows = [rows[i] for i in best_indices]
    
    for idx in similar_indices:
        row = rows[idx]
        d = min(ezr.xdist(tbl, row, b) for b in best_rows)
        print(f"{idx:<10} | {d:.4f}")
        
    print("\nThese setups share the most similar input characteristics with our top performers.")
    exit(0)
