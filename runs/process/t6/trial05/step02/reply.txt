import ezr
import step01

def guess(header, train, test):
    """
    Estimates the first goal column for each test row.
    Uses the tree-based ranker from ezr to predict the outcome.
    """
    # Build a table from the training data
    tbl = ezr.Tbl([header] + train)
    
    # Grow a tree on the training data to learn the relationship
    # between inputs and the goal space.
    node = ezr.tree(tbl, tbl.rows)
    
    # For each test row, walk the tree to find the leaf,
    # then return the mean of the first goal column in that leaf.
    results = []
    for row in test:
        leaf_node = ezr.leaf(node, row)
        # Find the index of the first goal column (first column ending in - or +)
        goal_idx = -1
        for i, name in enumerate(header):
            if name.endswith("-") or name.endswith("+"):
                goal_idx = i
                break
        
        # Calculate the mean of the goal column for rows in this leaf
        vals = [r[goal_idx] for r in leaf_node.rows if r[goal_idx] != "?"]
        if vals:
            results.append(sum(vals) / len(vals))
        else:
            # Fallback to global mean if leaf is empty or data missing
            all_vals = [r[goal_idx] for r in tbl.rows]
            results.append(sum(all_vals) / len(all_vals))
            
    return results

if __name__ == "__main__":
    # Load data
    filename = "pom3a.n1000.train.csv"
    data = list(ezr.csv(filename))
    header = data[0]
    rows = data[1:]
    
    # Split data: 80% train, 20% test
    split_idx = int(len(rows) * 0.8)
    train = rows[:split_idx]
    test_full = rows[split_idx:]
    
    # Mask the goals in the test set to simulate the customer's request
    test = []
    for row in test_full:
        new_row = [
            "?" if (name.endswith("-") or name.endswith("+")) else val
            for name, val in zip(header, row)
        ]
        test.append(new_row)
    
    # Get predictions
    predictions = guess(header, train, test)
    
    # Print a summary for the customer
    print("Cost estimation complete.")
    print(f"Estimated cost for the first 5 test setups: {[round(p, 3) for p in predictions[:5]]}")
    
    exit(0)
