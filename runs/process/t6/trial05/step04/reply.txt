import ezr
import step01

def flow(header, row, seen):
    """
    Decide whether to accept a row based on whether it is better than
    the current best seen so far.
    """
    # If this is the first row, we have no baseline, so let it go.
    if not seen:
        return False
    
    # Create a table to use ydist for scoring
    # We use the header to define the goals
    tbl = ezr.Tbl([header] + seen)
    
    # Calculate the ydist of the current row
    # Note: ydist works on full rows. If the row has '?', it uses the 
    # column's mid value (via norm/mid logic in ezr).
    current_score = ezr.ydist(tbl, row)
    
    # Calculate scores for all seen rows
    seen_scores = [ezr.ydist(tbl, r) for r in seen]
    
    # If the current row is better (lower ydist) than the best seen so far,
    # we accept it.
    return current_score < min(seen_scores)

if __name__ == "__main__":
    # Load data
    filename = "pom3a.n1000.train.csv"
    data = list(ezr.csv(filename))
    header = data[0]
    rows = data[1:]
    
    # Simulate the arrival process
    seen = []
    accepted_count = 0
    
    for row in rows:
        if flow(header, row, seen):
            accepted_count += 1
        seen.append(row)
        
    print(f"Simulation complete.")
    print(f"Total rows processed: {len(rows)}")
    print(f"Total setups accepted: {accepted_count}")
    print("Decision logic: Accepted a setup if it outperformed all previously seen options.")
    
    exit(0)
