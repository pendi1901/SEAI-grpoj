#-- misc ---------------------------------------------------
def say(x: Any, p: int = 2) -> str:
  "Floats get P decimals; dicts list their keys."
  if isinstance(x,float): x= f"{x:.{p}f}".rstrip("0").rstrip(".")
  if isinstance(x, dict):
    x= "{"+", ".join(f"{k}: {say(v,p)}" for k,v in x.items())+"}"
  return str(x)

class o(dict):
  "A dict you can poke with a dot."
  __getattr__, __setattr__ = dict.__getitem__, dict.__setitem__
  __repr__ = say

def atom(s: str) -> ATOM:
  "'22' -> 22. '3.1' -> 3.1. 'True' -> True. 'x' -> 'x'."
  for fn in (int, float):
    try: return fn(s)
    except ValueError: pass
  s = s.strip()
  return {"True": True, "False": False}.get(s, s)

def csv(file: str) -> Iterator[ROW]:
  "ROWS of FILE, each cell coerced."
  with open(os.path.expanduser(file),  # -sig drops any BOM
           encoding="utf-8-sig") as f:
    for line in f:
      if line.strip(): yield [atom(s) for s in line.split(",")]

def Col(txt: str = " ", at: int = 0) -> COL:
  "Factory that returns NUM or SYM."
  return Num(txt,at) if txt[0].isupper() else Sym(txt,at)

class Num:
  "Place to summarize stream of numbers."
  __slots__ = ("at", "txt", "n", "mu", "m2", "sd", "goal")
  def __init__(self, txt: str = " ", at: int = 0) -> None:
    self.at, self.txt, self.n = at, txt, 0
    self.mu = self.m2 = self.sd = 0
    self.goal = 0 if txt[-1] == "-" else 1
  __repr__ = lambda self: say({k: getattr(self,k)
                               for k in self.__slots__})

class Sym:
  "Place to summarize stream of Symbols.  The class is the tag."
  __slots__ = ("at", "txt", "n", "has")
  def __init__(self, txt: str = " ", at: int = 0) -> None:
    self.at, self.txt, self.n, self.has = at, txt, 0, {}
  __repr__ = lambda self: say({k: getattr(self,k)
                               for k in self.__slots__})

def add(col: COL, v: ATOM, inc: int = 1) -> None:
  "Show V to COL.  INC=-1 takes it away.  `?` changes nothing."
  if v != "?":
    col.n += inc
    if type(col) is Sym:
      col.has[v] = col.has.get(v, 0) + inc
    else:
      d = v - col.mu                      # Welford, for `div`
      col.mu += inc * d / max(1, col.n)
      col.m2  = max(0, col.m2 + inc * d * (v - col.mu))
      col.sd  = 0 if col.n < 2 else (col.m2/(col.n-1))**.5

def mid(col: COL) -> ATOM:
  "Middle: the mean, or the most common symbol; ? if none."
  return (max(col.has, key=col.has.get) if col.has else "?"
          ) if type(col) is Sym else col.mu

def mids(tbl: TBL) -> MIDS:
  "Every column's middle, cached until the next addCols."
  tbl.mids = tbl.mids or {at: mid(c) for at,c in tbl.cols.items()}
  return tbl.mids

def div(col: COL) -> float:
  "Spread: entropy, or standard deviation."
  return (-sum(v/col.n*log2(v/col.n) for v in col.has.values())
          if type(col) is Sym else col.sd)  # sd kept by `add`

def norm(col: COL, v: ATOM) -> ATOM:
  "To 0..1, by the logistic curve.  Symbols do not scale."
  if type(col) is Sym: return v
  z = max(-3, min(3, (v - col.mu) / (1e-32 + div(col))))
  return 1 / (1 + exp(-1.7 * z))

#-- tables -------------------------------------------------
def Tbl(src: Iterable[ROW]) -> TBL:
  "Header names the columns: X skip, +- goal, ! klass."
  src = iter(src)
  tbl = Cols(o(rows=[], cols={}, x=[], y=[], mids=None,
               names=next(src)))
  for row in src: addRow(tbl, row)
  return tbl

def Cols(tbl: TBL) -> TBL:
  "Create column roles."
  for at, s in enumerate(tbl.names):
    if s[-1] == "X": continue            # skip me entirely
    tbl.cols[at] = Col(s,at)
    (tbl.y if s[-1] in "+-!" else tbl.x).append(tbl.cols[at])
  return tbl

def addRow(tbl: TBL, row: ROW) -> ROW:
  "Keep ROW, and show it to my columns."
  tbl.rows += [row]
  return addCols(tbl, row)

def addCols(tbl: TBL, row: ROW, inc: int = 1) -> ROW:
  "Show ROW to my columns, without keeping it."
  tbl.mids = None                 # a new row moves the middles
  for at, col in tbl.cols.items(): add(col, row[at], inc)
  return row

def clone(tbl: TBL, rows: ROWS = None) -> TBL:
  "An empty copy of TBL, plus ROWS."
  return Tbl([tbl.names] + (rows or []))

def dist(vs: Iterable[QTY], n: int) -> float:
  "Root mean square of N gaps."
  return sqrt(sum(v*v for v in vs) / n)

def ydist(tbl: TBL, row: ROW) -> float:
  "How far ROW's goals are from the best they could be."
  return dist((abs(norm(c, row[c.at]) - c.goal)
                    for c in tbl.y), len(tbl.y))

def gap(col: COL, a: ATOM, b: ATOM) -> QTY:
  "Distance between two values of one column."
  if a == "?" or b == "?": return 1    # unknown = far
  return a!=b if type(col) is Sym else abs(norm(col,a)-norm(col,b))

def xdist(tbl: TBL, r1: ROW, r2: ROW | MIDS) -> float:
  "How far apart two rows are, over the x columns."
  return dist((gap(c, r1[c.at], r2[c.at])
                    for c in tbl.x), len(tbl.x))

def oracle(row: ROW) -> ROW:
  """Labeller. Rows here arrive with ys, so nothing to do.
  Real oracles fill in any missing y cells (and can close
  over a TBL to know which cells those are)."""
  return row

def acquire(tbl: TBL, cap: int | None = None,
            label: ORACLE = oracle) -> ROWS:
  "Label near the best rows, far from the rest."
  cap  = cap or the.Stop
  todo = random.sample(tbl.rows, len(tbl.rows))[:the.Few]
  both = clone(tbl, [label(todo.pop())
                     for _ in range(the.Start)])
  both.rows.sort(key=lambda r: ydist(both, r))
  n    = int(sqrt(the.Start))
  best,rest = clone(tbl,both.rows[:n]), clone(tbl,both.rows[n:])
  while todo and len(both.rows) < cap:
    cb, cr = mids(best), mids(rest)    # two centroids
    todo.sort(key=lambda z: xdist(tbl,z,cr) - xdist(tbl,z,cb))
    addRow(best, addRow(both, label(todo.pop())))
    best.rows.sort(key=lambda z: ydist(both, z))  # no y leak
    if len(best.rows) > sqrt(len(both.rows)): # best stays small
      addRow(rest, addCols(best, best.rows.pop(), -1))
  return both.rows

def grabs(tbl: TBL, cap: int | None = None,
          label: ORACLE = oracle) -> ROWS:
  "The straw man: same budget, taken at random."
  cap = cap or the.Stop
  return [label(r) for r in
          random.sample(tbl.rows, min(cap, len(tbl.rows)))]

# Calls core.  Nothing above this line calls down here.
# NODE = o(rows, at, v, go, kids, mu).
def cut(tbl: TBL, rows: ROWS) -> tuple[int,ATOM,GO]|None:
  "The (at, v, go) whose two sides have the tightest ys."
  ys = [ydist(tbl, r) for r in rows]
  out, least = None, 1e30
  for col in tbl.x:
    for v, go in candidates(col, rows):
      a, b = Num(), Num()   # one pass, so one `go` per row
      for r,y in zip(rows,ys): add(a if go(r) else b, y)
      if a.n >= the.Leaf and b.n >= the.Leaf:
        score = (div(a)*a.n + div(b)*b.n) / (a.n + b.n)
        if score < least: out, least = (col.at, v, go), score
  return out

def candidates(col: COL, rows: ROWS) -> Iterator[tuple[ATOM,GO]]:
  "What splits to try: one per symbol, or CUTS per num."
  at = col.at
  if type(col) is Sym:
    for v in sorted({r[at] for r in rows if r[at] != "?"}):
      yield v, lambda r,v=v,at=at: r[at] == v
  else:                # by rank, not value: THESE rows are dense
    xs = sorted(r[at] for r in rows if r[at] != "?")
    k  = max(1, len(xs)//the.Cuts)     # cuts, spaced by rank
    for v in xs[k-1:-1:k]:
      yield v, (lambda r,v=v,at=at: r[at] != "?" and r[at] <= v)

def tree(tbl: TBL, rows: ROWS) -> NODE:
  "Split while a split keeps both sides big enough."
  node = o(rows=rows, at=None, v=None, go=None, kids=[],
           mu=sum(ydist(tbl,r) for r in rows)/len(rows))
  if len(rows) > the.Leaf and (found := cut(tbl, rows)):
    at, v, go = found
    yes, no = [], []
    for r in rows: (yes if go(r) else no).append(r)
    node.at, node.v, node.go = at, v, go # cut kepts sides big
    node.kids = [tree(tbl, yes), tree(tbl, no)]
  return node

def leaf(node: NODE, row: ROW) -> NODE:
  "Walk ROW down to its leaf."
  while node.kids: node = node.kids[0 if node.go(row) else 1]
  return node

def leafs(node: NODE) -> list[NODE]:
  "Return leaves of this tree."
  return [x for k in node.kids for x in leafs(k)] or [node]

def show(tbl: TBL, node: NODE, pre: str | None = None,
         edge: str = "") -> None:
  "Tree on the left; ydist and n on the right."
  ls = sorted(leafs(node), key=lambda z: z.mu)
  def walk(z: NODE, pre: str | None, edge: str) -> None:
    m = "+" if z is ls[0] else "-" if z is ls[-1] else " "
    print(f"{m} {round(100*z.mu):>4} {len(z.rows):>5}   "
          f"{(pre or '')+edge}".rstrip())
    op = "=" if type(tbl.cols[z.at or 0]) is Sym else "<="
    no = "!=" if op == "=" else ">"
    for kid, oper in zip(z.kids, [op, no]):
      walk(kid, "" if pre is None else pre + "|  ",
           f"{tbl.names[z.at]} {oper} {say(z.v)}")
  print(" ydist     n")
  walk(node, pre, edge)



#-- rig ----------------------------------------------------
# Calls core, acquire, and one RANKER -- tree's, by default.
def ranker(lab: TBL) -> Callable[[ROW], float]:
  "Grow a tree on LAB; score any row by its leaf's mean ydist."
  node = tree(lab, lab.rows)
  return lambda r: leaf(node, r).mu

def wins(tbl: TBL) -> Callable[[ROW], float]:
  "100 at the best row, 0 at an average one."
  ys = sorted(ydist(tbl, r) for r in tbl.rows)
  lo, avg = ys[0], sum(ys)/len(ys)
  return lambda r: max(-100, min(100, 100*(1 - (ydist(tbl,r)-lo)
                                            / (avg-lo+1e-32))))

def holdout(tbl: TBL, pick: PICKER = acquire,
            rank: RANKER = ranker,
            label: ORACLE = oracle) -> ROW:
  "Train on half; of CHECK guesses on the rest, pick best."
  rows  = random.sample(tbl.rows, len(tbl.rows))
  n     = len(rows)//2
  tr    = clone(tbl, rows[:n][:the.Few])
  lab   = clone(tbl, pick(tr, the.Stop - the.Check, label))
  top   = sorted(rows[n:], key=rank(lab))
  return min(top[:the.Check], key=lambda r: ydist(lab, label(r)))

# A leaf: it calls core, and nothing here calls it.
def adds(vs: Iterable[ATOM], col: COL | None = None) -> COL:
  "Every item of VS into COL."
  col = Num() if col is None else col
  for v in vs: add(col, v)
  return col

def cohen(xs: NUMS, ys: NUMS, d: float = 0.35,
          eps: float = 0) -> bool:
  "Are the means within D pooled standard deviations?"
  a, b = adds(xs), adds(ys)
  pool = ((a.n-1)*div(a)**2 + (b.n-1)*div(b)**2)/(a.n+b.n-2)
  return abs(a.mu - b.mu) <= max(eps, d * sqrt(pool))

def cliffs(xs: NUMS, ys: NUMS, d: float = 0.197) -> bool:
  "Sorted in.  Is the rank imbalance small enough?"
  gt = lt = j = k = 0
  for x in xs:
    while j < len(ys) and ys[j] <  x: j += 1; k = j
    while k < len(ys) and ys[k] <= x: k += 1
    gt += j; lt += len(ys) - k
  return abs(gt - lt) / (len(xs) * len(ys)) <= d

def ks(xs: NUMS, ys: NUMS, a: float = 1.36) -> bool:
  "Sorted in.  95% Kolmogorov-Smirnov."
  n, m, i, j, d = len(xs), len(ys), 0, 0, 0
  while i < n and j < m:
    v = min(xs[i], ys[j])
    while i < n and xs[i] <= v: i += 1
    while j < m and ys[j] <= v: j += 1
    d = max(d, abs(i/n - j/m))
  return d <= a * sqrt((n + m) / (n * m))

def same(xs: NUMS, ys: NUMS, eps: float = 0) -> bool:
  """Indistinguishable.  Always cliffs and ks; EPS adds cohen, and
  doubles as its effect size (eps=0 skips that third test)."""
  xs, ys = sorted(xs), sorted(ys)
  return (cliffs(xs,ys) and ks(xs,ys)
          and (not eps or cohen(xs,ys,d=eps)))

import ezr
import random

def remember(header, rows):
    """
    Identify the top 5% of rows that perform best across all goals.
    Goals ending in '-' are minimized, '+' are maximized.
    """
    # Create a table object to use ezr's utility functions
    # We construct the input as [header] + rows
    data = [header] + rows
    tbl = ezr.Tbl(data)
    
    # ydist calculates how far a row is from the ideal goal values
    # (0 for minimize, 1 for maximize). Lower ydist is better.
    scored_rows = []
    for i, row in enumerate(rows):
        d = ezr.ydist(tbl, row)
        scored_rows.append((d, i))
    
    # Sort by distance (ascending, as lower is better)
    scored_rows.sort(key=lambda x: x[0])
    
    # Return the top 5% of rows (800 rows * 0.05 = 40)
    limit = max(1, int(len(rows) * 0.05))
    return [item[1] for item in scored_rows[:limit]]

import ezr
import step01

def guess(header, train, test):
    """
    Estimates the first goal column for each test row.
    Uses the tree-based ranker from ezr to estimate the performance
    of the first goal column.
    """
    # Create a table from training data to learn the model
    data = [header] + train
    tbl = ezr.Tbl(data)
    
    # Grow a tree on the training data
    node = ezr.tree(tbl, tbl.rows)
    
    # Identify the index of the first goal column
    # The first goal column is the first column ending in '-' or '+'
    goal_idx = -1
    for i, name in enumerate(header):
        if name.endswith(('-', '+')):
            goal_idx = i
            break
            
    # For each test row, find the leaf node and return the mean of the 
    # first goal column at that leaf
    results = []
    for row in test:
        leaf = ezr.leaf(node, row)
        # Calculate the average of the goal column for rows in this leaf
        vals = [r[goal_idx] for r in leaf.rows if r[goal_idx] != "?"]
        if vals:
            results.append(sum(vals) / len(vals))
        else:
            # Fallback to global mean if leaf is empty or has no data
            results.append(tbl.cols[goal_idx].mu)
            
    return results

if __name__ == "__main__":
    # Load the data
    filename = "pom3a.n1000.train.csv"
    data_gen = ezr.csv(filename)
    
    try:
        header = next(data_gen)
        all_rows = list(data_gen)
    except StopIteration:
        exit(1)
        
    # Split data: 80% train, 20% test
    n = len(all_rows)
    split = int(n * 0.8)
    train = all_rows[:split]
    test = all_rows[split:]
    
    # Mask the goals in the test set to simulate the customer's request
    goal_indices = [i for i, name in enumerate(header) if name.endswith(('-', '+'))]
    masked_test = []
    for row in test:
        new_row = list(row)
        for idx in goal_indices:
            new_row[idx] = "?"
        masked_test.append(new_row)
        
    # Get estimates
    estimates = guess(header, train, masked_test)
    
    # Print summary
    print(f"Estimated first goal (Cost-) for {len(estimates)} test cases:")
    print(f"Average estimated cost: {sum(estimates)/len(estimates):.4f}")
    print("-" * 30)
    for i in range(min(5, len(estimates))):
        print(f"Test row {i}: Estimated Cost- = {estimates[i]:.4f}")
        
    exit(0)

import ezr
import step01

def certify(xs, ys):
    """
    Returns True if the two distributions are statistically indistinguishable,
    False if they are significantly different.
    """

import ezr

def blame(header, rows):
    """
    Identify input columns that best explain why the worst rows are bad.
    We define 'worst' as the bottom 25% of rows by ydist.
    We then compare the distribution of input values in these worst rows
    against the rest of the data to find which inputs correlate with high ydist.
    """
    tbl = ezr.Tbl([header] + rows)
    
    # Calculate ydist for all rows
    scored = [(ezr.ydist(tbl, r), r) for r in rows]
    scored.sort(key=lambda x: x[0], reverse=True)
    
    # Split into worst 25% and the rest
    n_worst = len(rows) // 4
    worst = [x[1] for x in scored[:n_worst]]
    rest = [x[1] for x in scored[n_worst:]]
    
    # For each input column, check if the distribution in 'worst'
    # is significantly different from 'rest' using ezr's statistical tests.
    # We look for columns where the values are 'different' (not same).
    significant_inputs = []
    for col in tbl.x:
        xs = [r[col.at] for r in worst if r[col.at] != "?"]
        ys = [r[col.at] for r in rest if r[col.at] != "?"]
        
        if len(xs) > 1 and len(ys) > 1:
            # If they are NOT the same, this column helps explain the difference
            if not ezr.same(xs, ys):
                significant_inputs.append(header[col.at])
                
    return significant_inputs

def justify(header, rows, row):
    """
    Returns (p, reason): p is the fraction of rows that 'row' is better than,
    and reason is a sentence explaining the performance.
    """
    # Create a table to calculate ydist
    tbl = ezr.Tbl([header] + rows)
    
    # Estimate the goals for the input row using the tree-based ranker
    # We use the training data (all rows) to build the model
    ranker = step02.guess(header, rows, [row])
    
    # Since step02.guess returns a list of estimates for each goal column,
    # we need to reconstruct the row with estimated values to calculate ydist.
    goal_indices = [i for i, name in enumerate(header) if name.endswith(('-', '+'))]
    
    # Create a copy and fill in the estimated goals
    estimated_row = list(row)
    # The ranker in step02 returns the mean of the first goal column.
    # For a full justification, we need to estimate all goals.
    # We'll use the leaf mean for each goal column.
    node = ezr.tree(tbl, tbl.rows)
    leaf = ezr.leaf(node, row)
    
    for i in goal_indices:
        vals = [r[i] for r in leaf.rows if r[i] != "?"]
        estimated_row[i] = sum(vals) / len(vals) if vals else tbl.cols[i].mu
        
    # Calculate ydist for the estimated row
    est_ydist = ezr.ydist(tbl, estimated_row)
    
    # Compare against all rows
    all_ydists = [ezr.ydist(tbl, r) for r in rows]
    better_count = sum(1 for d in all_ydists if d > est_ydist)
    p = better_count / len(rows)
    
    reason = (f"This setup is in the top {int((1-p)*100)}% of all possible configurations "
              f"because its predicted performance metrics are closer to the ideal "
              f"than {int(p*100)}% of the known data.")
    
    return p, reason

import ezr

def choose(header, rows, label):
    """
    Find the best row using a budget of 50 labels.
    We use ezr's acquire function to intelligently sample the space.
    """
    # The error occurred because ezr.ydist tries to normalize '?' values.
    # We must provide a TBL that has valid goal values for the logic to work.
    # Since we can only label 50 rows, we build a TBL using only the 
    # rows we have already labeled.
    
    # We need a way to track which rows are labeled.
    # We will use a dictionary to store the full rows once labeled.
    labeled_cache = {}

    def oracle_wrapper(row):
        # Find index by object identity or by searching
        idx = rows.index(row)
        if idx not in labeled_cache:
            full_row = label(idx)
            labeled_cache[idx] = full_row
            # Update the row in place
            for i in range(len(row)):
                row[i] = full_row[i]
        return row

    # Initialize with a small sample to build the initial TBL structure
    # We must ensure the TBL is initialized with at least some valid data
    # so that norm() doesn't fail on '?'
    ezr.the.Stop = 50
    ezr.the.Start = 4
    
    # Create a TBL using the rows. We must ensure the TBL knows these are goals.
    # We use the provided rows (which have '?' in goals).
    # To prevent the TypeError, we ensure we only call ydist on rows 
    # that have been fully labeled.
    data = [header] + rows
    tbl = ezr.Tbl(data)
    
    # The acquire function expects the TBL to have valid Y values.
    # We override the oracle to ensure the TBL is updated.
    def smart_label(row):
        full = oracle_wrapper(row)
        # Update the TBL's internal column stats with the new real values
        ezr.addCols(tbl, full)
        return full

    # Perform the search
    labeled_rows = ezr.acquire(tbl, cap=50, label=smart_label)
    
    # Find the best row among the labeled ones
    best_row = min(labeled_rows, key=lambda r: ezr.ydist(tbl, r))
    
    return rows.index(best_row)

import ezr

def spot(header, rows):
    """
    Identify rows that do not fit the distribution of the data.
    We use the distance from the centroid of the input (X) columns
    as a measure of 'suspicion'. Rows that are far from the 
    center of the input space are considered outliers.
    """
    # Create a table to handle column definitions and distance calculations
    data = [header] + rows
    tbl = ezr.Tbl(data)
    
    # Calculate the centroid of the input space (the 'mids' of X columns)
    # ezr.mids(tbl) returns a dict of {at: mid_value}
    centroid = ezr.mids(tbl)
    
    # Calculate the distance of each row from the centroid
    # Higher distance = more suspicious
    suspicions = []
    for i, row in enumerate(rows):
        d = ezr.xdist(tbl, row, centroid)
        suspicions.append((d, i))
    
    # Sort by distance descending (most suspicious first)
    suspicions.sort(key=lambda x: x[0], reverse=True)
    
    # Return the indices of the rows
    return [item[1] for item in suspicions]

import ezr
import random

def race(header, rows, label):
    """
    Compare three strategies: random, acquire (tree-based), and centroid-based.
    Budget is 50 labels total.
    """
    budget = 50
    labeled_cache = {}

    def oracle_wrapper(idx):
        if idx not in labeled_cache:
            if len(labeled_cache) >= budget:
                raise Exception("Budget exceeded")
            labeled_cache[idx] = label(idx)
        return labeled_cache[idx]

    # Create a base table for distance calculations
    # We use a copy of rows to ensure we don't mutate the original input
    tbl = ezr.Tbl([header] + [list(r) for r in rows])

    def get_ydist(idx):
        full_row = oracle_wrapper(idx)
        return ezr.ydist(tbl, full_row)

    # 1. Random Strategy
    def strategy_random():
        indices = random.sample(range(len(rows)), budget // 3)
        return min(indices, key=get_ydist)

    # 2. Acquire Strategy (using ezr's built-in acquire)
    def strategy_acquire():
        # Create a local table for the acquire function
        local_tbl = ezr.Tbl([header] + [list(r) for r in rows])
        
        def smart_label(row):
            # Find index of this row in the original list
            idx = rows.index(row)
            full = oracle_wrapper(idx)
            # Update the row in the local table
            for i in range(len(row)): row[i] = full[i]
            return row
            
        # acquire returns a list of rows, we find the best one
        labeled = ezr.acquire(local_tbl, cap=budget // 3, label=smart_label)
        best_row = min(labeled, key=lambda r: ezr.ydist(local_tbl, r))
        return rows.index(best_row)

    # 3. Centroid Strategy
    def strategy_centroid():
        centroid = ezr.mids(tbl)
        # Pick rows closest to the centroid
        candidates = sorted(range(len(rows)), key=lambda i: ezr.xdist(tbl, rows[i], centroid))
        return min(candidates[:budget // 3], key=get_ydist)

    results = {
        "random": strategy_random(),
        "acquire": strategy_acquire(),
        "centroid": strategy_centroid()
    }
    
    # Determine winner: lowest ydist
    winner = min(results, key=lambda name: get_ydist(results[name]))
    
    return {
        "winner": winner,
        "row": results[winner],
        "tried": list(results.keys())
    }

def justify(header, rows, row):
    """
    Compares a proposed 'row' against the dataset.
    Estimates goals for 'row' using the tree-based ranker,
    calculates its performance (ydist), and determines its rank
    relative to the existing rows.
    """
    # Create a table to handle normalization and distance calculations
    tbl = ezr.Tbl([header] + rows)
    
    # Estimate goals for the input row using the tree-based ranker
    # We use the training data (all rows) to build the model
    # We use the logic from step06 to estimate all goal columns
    goal_indices = [i for i, name in enumerate(header) if name.endswith(('-', '+'))]
    
    # Build a tree to find the leaf for the input row
    node = ezr.tree(tbl, tbl.rows)
    leaf = ezr.leaf(node, row)
    
    # Create a copy and fill in the estimated goals from the leaf mean
    estimated_row = list(row)
    for i in goal_indices:
        vals = [r[i] for r in leaf.rows if r[i] != "?"]
        estimated_row[i] = sum(vals) / len(vals) if vals else tbl.cols[i].mu
        
    # Calculate ydist for the estimated row
    est_ydist = ezr.ydist(tbl, estimated_row)
    
    # Compare against all rows
    all_ydists = [ezr.ydist(tbl, r) for r in rows]
    better_count = sum(1 for d in all_ydists if d > est_ydist)
    p = better_count / len(rows)
    
    reason = (f"This setup is better than {int(p*100)}% of known configurations "
              f"because its predicted performance metrics are closer to the ideal "
              f"than most other options in our database.")
    
    return p, reason

import ezr

def choose(header, rows, label):
    """
    Find the best setup for a large, high-criticality project.
    We use ezr's acquire function to intelligently sample the space
    within the 50-label budget.
    """
    # 1. Setup: Create a table for distance calculations.
    # We use a copy of rows to avoid mutating the original input.
    tbl = ezr.Tbl([header] + [list(r) for r in rows])
    
    # 2. Oracle: Map the acquire function's row-based label calls 
    # back to the index-based label function provided by the harness.
    # We cache results to ensure we don't exceed the 50-label budget.
    labeled_cache = {}

    def smart_label(row):
        # Find the index of the row in the original list
        # We search for the row by identity or content
        idx = -1
        for i, r in enumerate(rows):
            if r is row:
                idx = i
                break
        
        if idx not in labeled_cache:
            full_row = label(idx)
            labeled_cache[idx] = full_row
            # Update the row in place to reflect the new values
            for i in range(len(row)):
                row[i] = full_row[i]
        return row

    # 3. Acquire: Use ezr's built-in acquisition logic.
    # We set the budget to 50 as per the rules.
    ezr.the.Stop = 50
    labeled_rows = ezr.acquire(tbl, cap=50, label=smart_label)
    
    # 4. Find the best row among the labeled ones.
    # ydist calculates distance to the ideal goal values.
    best_row = min(labeled_rows, key=lambda r: ezr.ydist(tbl, r))
    
    # Return the index of the best row found.
    return rows.index(best_row)
