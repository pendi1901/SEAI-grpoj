```python
import ezr
import random

def fix(header, row, levers):
    """
    Finds the smallest change to levers that moves the row toward the 
    best possible performance (minimizing ydist).
    """
    # Create a table to use ydist
    # We need a dummy table to get the goal definitions
    data = [header, row]
    tbl = ezr.Tbl(data)
    
    best_change = {}
    best_ydist = ezr.ydist(tbl, row)
    
    # Try changing each lever to its bounds or midpoint
    for col_name, bounds in levers.items():
        col_idx = header.index(col_name)
        original_val = row[col_idx]
        
        # Test values: for numeric, try low and high bounds
        test_vals = bounds if isinstance(bounds, list) else [bounds[0], bounds[1]]
        
        for val in test_vals:
            if val == original_val:
                continue
                
            # Create a candidate row
            candidate = row[:]
            candidate[col_idx] = val
            
            # Evaluate
            d = ezr.ydist(tbl, candidate)
            if d < best_ydist:
                best_ydist = d
                best_change = {col_name: val}
                
    return best_change

if __name__ == "__main__":
    filename = "pom3a.n1000.train.csv"
    data_iter = ezr.csv(filename)
    header = next(data_iter)
    rows = list(data_iter)
    
    # Define levers based on the problem domain (numeric inputs)
    # Inputs are columns 0-8
    levers = {
        "Culture": (0, 1),
        "Criticality": (0, 1),
        "Criticality Modifier": (0, 1),
        "Initial Known": (0, 1),
        "Inter-Dependency": (0, 1),
        "Dynamism": (0, 1),
        "Size": (0, 1),
        "Plan": (0, 1),
        "Team Size": (0, 1)
    }
    
    # Pick a "bad" row (e.g., one with high ydist)
    tbl = ezr.Tbl([header] + rows)
    scored = sorted([(ezr.ydist(tbl, r), r) for r in rows], key=lambda x: x[0], reverse=True)
    bad_row = scored[0][1]
    
    recommendation = fix(header, bad_row, levers)
    
    print("Rescue Plan for Underperforming Configuration:")
    if recommendation:
        for col, val in recommendation.items():
            print(f"Change '{col}' to {val:.4f} to improve performance.")
    else:
        print("No simple lever change found to improve this configuration.")
    
    exit(0)
```
