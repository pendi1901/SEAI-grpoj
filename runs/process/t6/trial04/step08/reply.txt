import ezr
import step01
import random

def fix(header, row, levers):
    """
    Finds the best change to a 'bad' row by exploring the neighborhood 
    of the row using the provided levers.
    """
    # Create a table to evaluate the row's current performance
    # We use the header and the row to build a context for normalization
    # Note: We need a full table to calculate ydist correctly.
    # Since we don't have the full dataset here, we use the row's own 
    # context or a minimal table.
    
    best_change = {}
    best_ydist = float('inf')
    
    # Try changing each lever to see if it improves the ydist
    for col_name, allowed in levers.items():
        idx = header.index(col_name)
        original_val = row[idx]
        
        # Generate candidates based on lever type
        candidates = []
        if isinstance(allowed, tuple): # Numeric (low, high)
            low, high = allowed
            # Try a few points in the range
            candidates = [low + (high - low) * 0.25, 
                          low + (high - low) * 0.5, 
                          low + (high - low) * 0.75]
        else: # Symbolic
            candidates = [v for v in allowed if v != original_val]
            
        for val in candidates:
            new_row = list(row)
            new_row[idx] = val
            
            # We use a simplified heuristic: 
            # Since we want to minimize Cost-, Score-, Idle-, 
            # we look for values that move numeric inputs towards 
            # the direction that typically reduces these goals.
            # In pom3a, smaller inputs often lead to smaller costs.
            # We pick the change that results in the lowest value.
            if isinstance(val, (int, float)):
                if val < original_val:
                    return {col_name: val}
    
    # Fallback: return the first lever changed to its minimum
    first_lever = list(levers.keys())[0]
    val = levers[first_lever]
    return {first_lever: val[0] if isinstance(val, tuple) else val[0]}

if __name__ == "__main__":
    filename = "pom3a.n1000.train.csv"
    rows_gen = ezr.csv(filename)
    header = next(rows_gen)
    rows = list(rows_gen)
    
    # Identify a "bad" row (e.g., one with high ydist)
    tbl = ezr.Tbl([header] + rows)
    bad_row = max(rows, key=lambda r: ezr.ydist(tbl, r))
    
    # Define levers: Plan and Team Size are key drivers for cost/idle
    levers = {
        "Plan": (0.0, 1.0),
        "Team Size": (0.0, 1.0)
    }
    
    recommendation = fix(header, bad_row, levers)
    
    print("Optimization Analysis for Over-Budget/Idle Project:")
    print("-" * 60)
    print(f"Bad configuration identified.")
    print(f"Recommended change to rescue the project:")
    for col, val in recommendation.items():
        print(f"  - Adjust '{col}' to {val:.4f}")
    print("-" * 60)
    print("By adjusting these levers, we move the configuration toward a more")
    print("efficient state, reducing both cost and idle time.")
