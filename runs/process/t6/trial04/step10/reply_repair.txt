import ezr
import random

def race(header, rows, label):
    """
    Compare three strategies: random, acquire (from ezr), and a greedy search.
    Total budget is 50 labels shared across all strategies.
    """
    budget = 50
    labeled_indices = {} # map index to row

    def oracle_wrapper(i):
        if i not in labeled_indices:
            if len(labeled_indices) >= budget:
                return None
            labeled_indices[i] = label(i)
        return labeled_indices[i]

    # Build a table from the full dataset to allow ydist calculations
    # We use the provided rows (which have '?' in goals)
    tbl = ezr.Tbl([header] + rows)

    def strategy_random(budget_per):
        indices = random.sample(range(len(rows)), budget_per)
        return [oracle_wrapper(i) for i in indices if oracle_wrapper(i) is not None]

    def strategy_acquire(budget_per):
        # Use a subset of rows to simulate the acquire process
        ezr.the.Stop = budget_per
        ezr.the.Start = 4
        # We need to pass a function that labels rows by index
        def local_label(r):
            for i, row in enumerate(rows):
                if row is r: return oracle_wrapper(i)
            return r
        return ezr.acquire(tbl, cap=budget_per, label=local_label)

    def strategy_greedy(budget_per):
        # Start with a random row
        idx = random.randint(0, len(rows) - 1)
        results = [oracle_wrapper(idx)]
        for _ in range(budget_per - 1):
            # Pick a random sample and move to the best one
            candidates = random.sample(range(len(rows)), 10)
            best_idx = min(candidates, key=lambda i: ezr.ydist(tbl, rows[i]))
            results.append(oracle_wrapper(best_idx))
        return [r for r in results if r is not None]

    strategies = {
        "random": strategy_random,
        "acquire": strategy_acquire,
        "greedy": strategy_greedy
    }
    
    budget_per = budget // len(strategies)
    best_overall_ydist = float('inf')
    winner = None
    best_row_idx = -1

    for name, func in strategies.items():
        labeled = func(budget_per)
        for r in labeled:
            d = ezr.ydist(tbl, r)
            if d < best_overall_ydist:
                best_overall_ydist = d
                winner = name
                # Find index of this row
                for i, row in enumerate(rows):
                    if row is r:
                        best_row_idx = i
                        break

    return {"winner": winner, "row": best_row_idx, "tried": list(strategies.keys())}

if __name__ == "__main__":
    filename = "pom3a.n1000.train.csv"
    rows_gen = ezr.csv(filename)
    header = next(rows_gen)
    rows = list(rows_gen)
    
    goal_indices = [i for i, name in enumerate(header) if name.endswith(('-', '+'))]
    masked_rows = []
    for row in rows:
        new_row = list(row)
        for idx in goal_indices:
            new_row[idx] = "?"
        masked_rows.append(new_row)
    
    def label(i):
        return rows[i]
    
    result = race(header, masked_rows, label)
    
    print("Race Results:")
    print("-" * 30)
    print(f"Winner: {result['winner']}")
    print(f"Best Row Index: {result['row']}")
    print(f"Strategies Tried: {', '.join(result['tried'])}")
    print("-" * 30)
