```python
import csv

def guess(header, train, test):
    """
    Estimates the first goal column (Cost-) for test rows by finding the 
    nearest neighbor in the training set based on input columns.
    """
    # Identify goal columns and input columns
    goal_indices = [i for i, name in enumerate(header) if name.endswith(('-', '+'))]
    input_indices = [i for i, name in enumerate(header) if not name.endswith(('-', '+'))]
    
    first_goal_idx = goal_indices[0]
    
    def dist(row_a, row_b):
        d = 0
        for i in input_indices:
            val_a = row_a[i]
            val_b = row_b[i]
            if isinstance(val_a, (int, float)):
                d += (val_a - val_b) ** 2
            else:
                d += 0 if val_a == val_b else 1
        return d ** 0.5

    results = []
    for t_row in test:
        # Find the nearest neighbor in training data
        best_dist = float('inf')
        best_val = 0
        for train_row in train:
            d = dist(t_row, train_row)
            if d < best_dist:
                best_dist = d
                best_val = train_row[first_goal_idx]
        results.append(best_val)
    return results

if __name__ == "__main__":
    filename = 'pom3a.n1000.train.csv'
    try:
        with open(filename, 'r') as f:
            reader = csv.reader(f)
            header = next(reader)
            rows = []
            for row in reader:
                processed_row = []
                for i, val in enumerate(row):
                    try:
                        processed_row.append(float(val))
                    except ValueError:
                        processed_row.append(val)
                rows.append(processed_row)
        
        # Split data: use first 700 for training, next 100 for testing
        train_data = rows[:700]
        test_data = []
        for r in rows[700:]:
            new_row = list(r)
            # Mask goals in test data
            for i in range(len(header)):
                if header[i].endswith(('-', '+')):
                    new_row[i] = "?"
            test_data.append(new_row)
            
        predictions = guess(header, train_data, test_data)
        
        # Print summary for the customer
        avg_cost = sum(predictions) / len(predictions)
        print(f"Estimated average cost for the requested setups: {avg_cost:.4f}")
        print(f"Processed {len(test_data)} test scenarios.")
        
    except FileNotFoundError:
        print(f"Error: {filename} not found.")
        exit(1)
```
