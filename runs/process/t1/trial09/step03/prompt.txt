import csv

def remember(header, rows):
    """
    Identifies the best rows based on multi-objective optimization.
    Goals ending in '-' are minimized, '+' are maximized.
    We use the Pareto frontier concept: a row is 'unusually well' if 
    no other row is better in all goal dimensions.
    """
    goals = []
    for i, name in enumerate(header):
        if name.endswith('-'):
            goals.append((i, -1))  # Minimize
        elif name.endswith('+'):
            goals.append((i, 1))   # Maximize

    def is_better(row_a, row_b):
        # Returns True if row_a dominates row_b
        better_in_at_least_one = False
        for idx, direction in goals:
            val_a = row_a[idx]
            val_b = row_b[idx]
            if direction == -1:  # Minimize
                if val_a > val_b: return False
                if val_a < val_b: better_in_at_least_one = True
            else:  # Maximize
                if val_a < val_b: return False
                if val_a > val_b: better_in_at_least_one = True
        return better_in_at_least_one

    # Find Pareto frontier
    pareto = []
    for i, row in enumerate(rows):
        is_dominated = False
        for j, other in enumerate(rows):
            if i == j: continue
            if is_better(other, row):
                is_dominated = True
                break
        if not is_dominated:
            pareto.append(i)
            
    # Return at most 5% of rows
    limit = max(1, len(rows) // 20)
    return pareto[:limit]

import csv

def guess(header, train, test):
    """
    Estimates the first goal column (Cost-) for test rows by finding the 
    nearest neighbor in the training set based on input columns.
    """
    # Identify goal columns and input columns
    goal_indices = [i for i, name in enumerate(header) if name.endswith(('-', '+'))]
    input_indices = [i for i, name in enumerate(header) if not name.endswith(('-', '+'))]
    
    first_goal_idx = goal_indices[0]
    
    def dist(row_a, row_b):
        d = 0
        for i in input_indices:
            val_a = row_a[i]
            val_b = row_b[i]
            if isinstance(val_a, (int, float)):
                d += (val_a - val_b) ** 2
            else:
                d += 0 if val_a == val_b else 1
        return d ** 0.5

    results = []
    for t_row in test:
        # Find the nearest neighbor in training data
        best_dist = float('inf')
        best_val = 0
        for train_row in train:
            d = dist(t_row, train_row)
            if d < best_dist:
                best_dist = d
                best_val = train_row[first_goal_idx]
        results.append(best_val)
    return results
