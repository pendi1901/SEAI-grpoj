import csv

def remember(header, rows):
    """
    Identifies the best rows based on multi-objective optimization.
    Goals ending in '-' are minimized, '+' are maximized.
    We use the Pareto frontier concept: a row is 'unusually well' if 
    no other row is better in all goal dimensions.
    """
    goals = []
    for i, name in enumerate(header):
        if name.endswith('-'):
            goals.append((i, -1))  # Minimize
        elif name.endswith('+'):
            goals.append((i, 1))   # Maximize

    def is_better(row_a, row_b):
        # Returns True if row_a dominates row_b
        better_in_at_least_one = False
        for idx, direction in goals:
            val_a = row_a[idx]
            val_b = row_b[idx]
            if direction == -1:  # Minimize
                if val_a > val_b: return False
                if val_a < val_b: better_in_at_least_one = True
            else:  # Maximize
                if val_a < val_b: return False
                if val_a > val_b: better_in_at_least_one = True
        return better_in_at_least_one

    # Find Pareto frontier
    pareto = []
    for i, row in enumerate(rows):
        is_dominated = False
        for j, other in enumerate(rows):
            if i == j: continue
            if is_better(other, row):
                is_dominated = True
                break
        if not is_dominated:
            pareto.append(i)
            
    # Return at most 5% of rows
    limit = max(1, len(rows) // 20)
    return pareto[:limit]

    nearest neighbor in the training set based on input columns.
    """
    # Identify goal columns and input columns
    goal_indices = [i for i, name in enumerate(header) if name.endswith(('-', '+'))]
    input_indices = [i for i, name in enumerate(header) if not name.endswith(('-', '+'))]
    
    first_goal_idx = goal_indices[0]
    
    def dist(row_a, row_b):
        d = 0
        for i in input_indices:
            val_a = row_a[i]
            val_b = row_b[i]
            if isinstance(val_a, (int, float)):
                d += (val_a - val_b) ** 2
            else:
                d += 0 if val_a == val_b else 1
        return d ** 0.5

    results = []

        best_dist = float('inf')

import csv

def certify(xs, ys):
    """
    Returns True if xs and ys are statistically similar (same distribution),
    False if they differ significantly. Uses a simple A12 effect size check.
    """
    def a12(lst1, lst2):
        m, n = len(lst1), len(lst2)
        if m == 0 or n == 0: return 0.5
        greater = 0
        for x in lst1:
            for y in lst2:
                if x > y: greater += 1
                elif x == y: greater += 0.5
        return greater / (m * n)

    # A12 threshold: 0.56 is a small effect size, 0.64 medium, 0.71 large.
    # If A12 is close to 0.5, they are likely the same.
    score = a12(xs, ys)
    return 0.44 <= score <= 0.56

import csv

def flow(header, row, seen):
    """
    Decides whether to accept a row based on the Secretary Problem logic:
    1. Observe a sample (first 25% of data) to establish a baseline.
    2. Accept the first row that is better than the best seen in the sample.
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith(('-', '+'))]
    
    def is_better(row_a, row_b):
        # Returns True if row_a is better than row_b
        for idx in goal_indices:
            name = header[idx]
            val_a, val_b = row_a[idx], row_b[idx]
            if name.endswith('-'):
                if val_a > val_b: return False
            else:
                if val_a < val_b: return False
        return True

    # Use 25% of the total expected data as a training sample
    sample_size = 200 
    
    if len(seen) < sample_size:
        return False
    
    # Find the best in the sample
    best_in_sample = seen[0]
    for r in seen[1:sample_size]:
        if is_better(r, best_in_sample):
            best_in_sample = r
            
    # If current row is better than the best in sample, accept it
    return is_better(row, best_in_sample)

if __name__ == "__main__":
