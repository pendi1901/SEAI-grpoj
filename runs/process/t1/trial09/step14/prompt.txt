import csv

def remember(header, rows):
    """
    Identifies the best rows based on multi-objective optimization.
    Goals ending in '-' are minimized, '+' are maximized.
    We use the Pareto frontier concept: a row is 'unusually well' if 
    no other row is better in all goal dimensions.
    """
    goals = []
    for i, name in enumerate(header):
        if name.endswith('-'):
            goals.append((i, -1))  # Minimize
        elif name.endswith('+'):
            goals.append((i, 1))   # Maximize

    def is_better(row_a, row_b):
        # Returns True if row_a dominates row_b
        better_in_at_least_one = False
        for idx, direction in goals:
            val_a = row_a[idx]
            val_b = row_b[idx]
            if direction == -1:  # Minimize
                if val_a > val_b: return False
                if val_a < val_b: better_in_at_least_one = True
            else:  # Maximize
                if val_a < val_b: return False
                if val_a > val_b: better_in_at_least_one = True
        return better_in_at_least_one

    # Find Pareto frontier
    pareto = []
    for i, row in enumerate(rows):
        is_dominated = False
        for j, other in enumerate(rows):
            if i == j: continue
            if is_better(other, row):
                is_dominated = True
                break
        if not is_dominated:
            pareto.append(i)
            
    # Return at most 5% of rows
    limit = max(1, len(rows) // 20)
    return pareto[:limit]

import csv

def certify(xs, ys):
    """
    Returns True if xs and ys are statistically similar (same distribution),
    False if they differ significantly. Uses a simple A12 effect size check.
    """
    def a12(lst1, lst2):
        m, n = len(lst1), len(lst2)
        if m == 0 or n == 0: return 0.5
        greater = 0
        for x in lst1:
            for y in lst2:
                if x > y: greater += 1
                elif x == y: greater += 0.5
        return greater / (m * n)

    # A12 threshold: 0.56 is a small effect size, 0.64 medium, 0.71 large.
    # If A12 is close to 0.5, they are likely the same.
    score = a12(xs, ys)
    return 0.44 <= score <= 0.56

import csv

def flow(header, row, seen):
    """
    Decides whether to accept a row based on the Secretary Problem logic:
    1. Observe a sample (first 25% of data) to establish a baseline.
    2. Accept the first row that is better than the best seen in the sample.
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith(('-', '+'))]
    
    def is_better(row_a, row_b):
        # Returns True if row_a is better than row_b
        for idx in goal_indices:
            name = header[idx]
            val_a, val_b = row_a[idx], row_b[idx]
            if name.endswith('-'):
                if val_a > val_b: return False
            else:
                if val_a < val_b: return False
        return True

    # Use 25% of the total expected data as a training sample
    sample_size = 200 
    
    if len(seen) < sample_size:
        return False
    
    # Find the best in the sample
    best_in_sample = seen[0]
    for r in seen[1:sample_size]:
        if is_better(r, best_in_sample):
            best_in_sample = r
            
    # If current row is better than the best in sample, accept it
    return is_better(row, best_in_sample)

def blame(header, rows):
    """
    Identifies input columns that correlate most strongly with the 'Idle-' goal.
    We define 'worst' as rows in the top quartile of 'Idle-' values.
    We then calculate the correlation between each input and the 'Idle-' goal.
    """
    goal_idx = header.index('Idle-')
    input_indices = [i for i, name in enumerate(header) if not name.endswith(('-', '+'))]
    
    # Sort rows by Idle- to find the worst ones
    sorted_rows = sorted(rows, key=lambda r: r[goal_idx], reverse=True)
    worst_rows = sorted_rows[:len(rows) // 4]
    
    # Calculate simple correlation: mean of input in worst rows vs mean of input in all rows
    # A high difference indicates that the input is a driver for the bad outcome
    scores = []
    for i in input_indices:
        all_vals = [r[i] for r in rows if isinstance(r[i], (int, float))]
        worst_vals = [r[i] for r in worst_rows if isinstance(r[i], (int, float))]
        
        if not all_vals or not worst_vals:
            continue

        mean_worst = sum(worst_vals) / len(worst_vals)
        
        # We look for inputs that deviate significantly in the worst setups
        diff = abs(mean_worst - mean_all)
        scores.append((diff, header[i]))
    
    # Return the top 2 inputs that explain the 'Idle-' behavior
    scores.sort(key=lambda x: x[0], reverse=True)
    return [name for diff, name in scores[:2]]

import csv

def justify(header, rows, row):
    """
    Estimates the goals for the given row using a nearest-neighbor approach,
    then calculates the percentile rank of this row against all other rows.
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith(('-', '+'))]
    input_indices = [i for i, name in enumerate(header) if not name.endswith(('-', '+'))]

    def get_dist(r1, r2):
        d = 0
        for i in input_indices:
            d += (r1[i] - r2[i]) ** 2
        return d ** 0.5

    # Estimate goals for the input row using the nearest neighbor
    best_dist = float('inf')
    nearest_row = None
    for r in rows:
        d = get_dist(row, r)
        if d < best_dist:
            best_dist = d
            nearest_row = r
    
    estimated_goals = [nearest_row[i] for i in goal_indices]

    def is_better(goals_a, goals_b):
        # Returns True if goals_a is better than goals_b
        for i, idx in enumerate(goal_indices):
            name = header[idx]
            if name.endswith('-'):
                if goals_a[i] > goals_b[i]: return False
            else:
                if goals_a[i] < goals_b[i]: return False
        return True

    # Compare estimated goals against all rows
    better_count = 0
    for r in rows:
        actual_goals = [r[i] for i in goal_indices]
        if is_better(estimated_goals, actual_goals):
            better_count += 1
            
    p = better_count / len(rows)
    reason = f"This setup is estimated to outperform {p:.1%} of all configurations based on its input parameters."
    return p, reason

import csv

def spot(header, rows):
    """
    Identifies suspicious rows by calculating the Mahalanobis-like distance
    (z-score deviation) for each numeric input column. Rows with values 
    significantly outside the expected range (e.g., > 3 standard deviations) 
    are flagged as suspicious.
    """
    input_indices = [i for i, name in enumerate(header) if not name.endswith(('-', '+'))]
    
    # Calculate mean and standard deviation for each input column
    stats = {}
    for i in input_indices:
        vals = [r[i] for r in rows if isinstance(r[i], (int, float))]
        if not vals:
            continue
        mean = sum(vals) / len(vals)
        std = (sum((x - mean) ** 2 for x in vals) / len(vals)) ** 0.5
        stats[i] = (mean, std)

    # Score each row based on the maximum z-score of its inputs
    scores = []
    for row_idx, row in enumerate(rows):
        max_z = 0
        for i in input_indices:
            if i in stats:
                mean, std = stats[i]
                if std > 0:
                    z = abs(row[i] - mean) / std
                    if z > max_z:
                        max_z = z
        scores.append((max_z, row_idx))

    # Sort by suspicion score descending
    scores.sort(key=lambda x: x[0], reverse=True)
    
    # Return indices of rows with high deviation (e.g., z > 3)
    return [idx for score, idx in scores if score > 3.0]

import csv
import random

def race(header, rows, label):
    """
    Compares three strategies: 'random', 'greedy', and 'hill_climber'.
    Budget: 50 labels total.
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith(('-', '+'))]
    
    def get_score(goals):
        # Normalize goals: minimize '-' (val), maximize '+' (1-val)
        score = 0
        for i, val in enumerate(goals):
            name = header[goal_indices[i]]
            if name.endswith('-'):
                score += val
            else:
                score += (1 - val)
        return score

    labeled = {}
    def get_label(i):
        if i not in labeled:
            if len(labeled) >= 50:
                raise Exception("Budget exceeded")
            labeled[i] = label(i)
        return labeled[i]

    # Strategies
    def strategy_random():
        best_idx = -1
        best_score = float('inf')
        for _ in range(16): # 16 labels
            idx = random.randint(0, len(rows) - 1)
            score = get_score(get_label(idx))
            if score < best_score:
                best_score, best_idx = score, idx
        return best_idx

    def strategy_greedy():
        # Sample 17, pick best
        best_idx = -1
        best_score = float('inf')
        for _ in range(17):
            idx = random.randint(0, len(rows) - 1)
            score = get_score(get_label(idx))
            if score < best_score:
                best_score, best_idx = score, idx
        return best_idx

    def strategy_hill_climber():
        # Start random, then move to neighbors
        curr = random.randint(0, len(rows) - 1)
        curr_score = get_score(get_label(curr))
        for _ in range(16):
            neighbor = random.randint(0, len(rows) - 1)
            n_score = get_score(get_label(neighbor))
            if n_score < curr_score:

import csv

def remember(header, rows):
    """
    Identifies the best rows based on multi-objective optimization.
    A row is 'unusually well' if no other row is better in all goal dimensions.
    Returns the indices of the Pareto frontier, limited to 5% of the total rows.
    """
    goal_indices = []
    for i, name in enumerate(header):
        if name.endswith('-'):
            goal_indices.append((i, -1))  # Minimize
        elif name.endswith('+'):
            goal_indices.append((i, 1))   # Maximize

    def is_better(row_a, row_b):
        # Returns True if row_a dominates row_b
        better_in_at_least_one = False
        for idx, direction in goal_indices:
            val_a = row_a[idx]
            val_b = row_b[idx]
            if direction == -1:  # Minimize
                if val_a > val_b: return False
                if val_a < val_b: better_in_at_least_one = True
            else:  # Maximize
                if val_a < val_b: return False
                if val_a > val_b: better_in_at_least_one = True
        return better_in_at_least_one

    # Find Pareto frontier
    pareto = []
    for i, row in enumerate(rows):
        is_dominated = False
        for j, other in enumerate(rows):
            if i == j: continue
            if is_better(other, row):
                is_dominated = True
                break
        if not is_dominated:
            pareto.append(i)
            
    # Return at most 5% of rows
    limit = max(1, len(rows) // 20)
    return pareto[:limit]

import csv

def guess(header, train, test):
    """
    Estimates the first goal column (Cost-) for test rows.
    To simulate 'requirements changing twice as fast', we weight the 
    'Dynamism' input column more heavily when calculating distance,
    as dynamism represents the rate of change in requirements.
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith(('-', '+'))]
    input_indices = [i for i, name in enumerate(header) if not name.endswith(('-', '+'))]
    
    first_goal_idx = goal_indices[0]
    dynamism_idx = header.index('Dynamism')
    
    def dist(row_a, row_b):
        d = 0
        for i in input_indices:
            val_a = row_a[i]
            val_b = row_b[i]
            # Weight Dynamism by 2.0 to simulate the "twice as fast" requirement change
            weight = 2.0 if i == dynamism_idx else 1.0
            if isinstance(val_a, (int, float)):
                d += weight * ((val_a - val_b) ** 2)
            else:
                d += weight * (0 if val_a == val_b else 1)
        return d ** 0.5

    results = []
    for t_row in test:
        # Find the nearest neighbor in training data using the weighted distance
        best_dist = float('inf')
        best_val = 0
        for train_row in train:
            d = dist(t_row, train_row)
            if d < best_dist:
                best_dist = d
                best_val = train_row[first_goal_idx]
        results.append(best_val)
    return results

import csv

def certify(xs, ys):
    """
    Returns True if xs and ys are statistically similar (same distribution),
    False if they differ significantly. Uses the A12 effect size check.
    """
    m, n = len(xs), len(ys)
    if m == 0 or n == 0:
        return True
    
    greater = 0
    for x in xs:
        for y in ys:
            if x > y:
                greater += 1
            elif x == y:
                greater += 0.5
    
    a12 = greater / (m * n)
    # A12 threshold: 0.56 is a small effect size.
    # If A12 is between 0.44 and 0.56, the difference is negligible.
    return 0.44 <= a12 <= 0.56
