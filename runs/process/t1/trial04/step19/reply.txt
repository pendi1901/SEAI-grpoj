import csv

def spot(header, rows):
    """
    Identifies rows that are outliers in the input space.
    A row is considered suspicious if its input values are statistically 
    unlikely compared to the rest of the dataset. We use the Interquartile 
    Range (IQR) method to flag values outside the [Q1 - 1.5*IQR, Q3 + 1.5*IQR] 
    range. Rows with more outliers or more extreme deviations are ranked 
    as more suspicious.
    """
    input_indices = [i for i, name in enumerate(header) 
                     if not (name.endswith('-') or name.endswith('+'))]
    
    # Calculate bounds for each input column
    bounds = {}
    for i in input_indices:
        col_vals = sorted([row[i] for row in rows if isinstance(row[i], (int, float))])
        if not col_vals:
            continue
        q1 = col_vals[len(col_vals) // 4]
        q3 = col_vals[3 * len(col_vals) // 4]
        iqr = q3 - q1
        # Use a slightly wider range for robustness, but flag values outside 1.5*IQR
        bounds[i] = (q1 - 1.5 * iqr, q3 + 1.5 * iqr)

    # Score rows based on how many inputs are outside the "normal" range
    # and the magnitude of the deviation.
    suspicion_scores = []
    for idx, row in enumerate(rows):
        score = 0
        for i in input_indices:
            val = row[i]
            if isinstance(val, (int, float)) and i in bounds:
                low, high = bounds[i]
                if val < low or val > high:
                    # Increment score for each outlier found
                    score += 1
        
        if score > 0:
            suspicion_scores.append((score, idx))
    
    # Sort by score descending (most suspicious first)
    suspicion_scores.sort(key=lambda x: x[0], reverse=True)
    
    return [item[1] for item in suspicion_scores]

if __name__ == "__main__":
    filename = "pom3a.n1000.train.csv"
    try:
        with open(filename, mode='r', encoding='utf-8') as f:
            reader = csv.reader(f)
            header = next(reader)
            rows = []
            for row in reader:
                processed_row = []
                for val in row:
                    try:
                        processed_row.append(float(val))
                    except ValueError:
                        processed_row.append(val)
                rows.append(processed_row)
        
        suspicious_indices = spot(header, rows)
        
        print("Outlier Detection Analysis:")
        if not suspicious_indices:
            print("No suspicious records found. All setups appear within normal operational parameters.")
        else:
            print(f"Found {len(suspicious_indices)} setups that deviate from historical norms.")
            print("The most suspicious row indices are:", suspicious_indices[:10])
            print("These configurations contain input values that have not been observed in previous runs.")
            
    except FileNotFoundError:
        print(f"Error: {filename} not found.")
        exit(1)
