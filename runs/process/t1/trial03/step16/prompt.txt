import csv

def remember(header, rows):
    """
    Identifies rows that perform well across all goal columns.
    Goals ending in '-' are minimized, '+' are maximized.
    """
    goal_indices = []
    for i, name in enumerate(header):
        if name.endswith('-') or name.endswith('+'):
            goal_indices.append((i, name[-1]))

    def score_row(row):
        # We want to minimize all goals. 
        # For '-', lower is better. For '+', higher is better (so we negate).
        scores = []
        for idx, op in goal_indices:
            val = float(row[idx])
            scores.append(val if op == '-' else -val)
        return scores

    # Calculate scores for all rows
    scored_rows = []
    for i, row in enumerate(rows):
        scored_rows.append((score_row(row), i))

    # Sort by the goals lexicographically (all goals are equally important)
    scored_rows.sort()

    # Return the top 5% (or at least 1)
    limit = max(1, len(rows) // 20)
    return [item[1] for item in scored_rows[:limit]]

import csv

def certify(xs, ys):
    """
    Returns True if the two distributions are statistically similar 
    (i.e., the difference is likely due to noise), False otherwise.
    Uses a simple A12 effect size test (Vargha-Delaney).
    """
    def a12(lst1, lst2):
        m, n = len(lst1), len(lst2)
        lst1 = sorted(lst1)
        lst2 = sorted(lst2)
        r1 = 0
        for x in lst1:
            for y in lst2:
                if x > y: r1 += 1
                elif x == y: r1 += 0.5
        return r1 / (m * n)

    # If A12 is close to 0.5, they are similar. 
    # Threshold 0.56 is a standard small effect size.
    score = a12(xs, ys)
    return 0.44 <= score <= 0.56

import csv

def flow(header, row, seen):
    """
    Decides whether to accept a row based on whether it is better than 
    all previously seen rows. Uses a simple dominance check on goals.
    """
    goal_indices = []
    for i, name in enumerate(header):
        if name.endswith('-') or name.endswith('+'):
            goal_indices.append((i, name[-1]))

    def get_scores(r):
        scores = []
        for idx, op in goal_indices:
            val = float(r[idx])
            scores.append(val if op == '-' else -val)
        return scores

    current_scores = get_scores(row)

    # If this is the first row, accept it
    if not seen:
        return True

    # Compare against all previously seen rows
    # We accept if this row is better than or equal to all previous rows
    # (i.e., it is a non-dominated candidate in the sequence)
    for prev_row in seen:
        prev_scores = get_scores(prev_row)
        # Check if current row dominates or is equal to previous
        # If any previous row is strictly better, we reject
        for c, p in zip(current_scores, prev_scores):
            if c > p:
                return False
    
    return True

    # Calculate correlation (simple difference in means)
    # Sort rows by Idle- to find the 'worst' (highest idle)
    sorted_rows = sorted(rows, key=lambda r: r[idle_idx], reverse=True)
    
    # Compare top 25% (worst) vs bottom 25% (best)
    n = len(sorted_rows) // 4
    worst = sorted_rows[:n]
    best = sorted_rows[-n:]
    
    correlations = []
    for i in input_indices:
        mean_worst = sum(r[i] for r in worst) / n
        mean_best = sum(r[i] for r in best) / n

        correlations.append((abs(mean_worst - mean_best), header[i]))
    
    # Return the top 3 inputs that explain the variance
    correlations.sort(key=lambda x: x[0], reverse=True)
    return [name for diff, name in correlations[:3]]

import csv

def justify(header, rows, row):
    """
    Estimates the performance of 'row' by comparing it to the training 'rows'.
    Returns (p, reason) where p is the fraction of rows dominated by 'row'.
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith('-') or name.endswith('+')]
    input_indices = [i for i, name in enumerate(header) if not (name.endswith('-') or name.endswith('+'))]
    
    # Estimate goals for the target row using 3-Nearest Neighbor
    def get_dist(r1, r2):
        return sum((r1[i] - r2[i])**2 for i in input_indices)**0.5

    # distances is a list of (distance, row_data)
    distances = sorted([(get_dist(row, r), r) for r in rows])
    neighbors = [n[1] for n in distances[:3]]
    
    estimated_goals = []
    for g_idx in goal_indices:
        estimated_goals.append(sum(n[g_idx] for n in neighbors) / 3)
    
    # Calculate how many rows are dominated by this estimated performance
    # A row is dominated if the estimated performance is better than the row's actual performance
    better_count = 0
    for r in rows:
        is_better = True
        for i, g_idx in enumerate(goal_indices):
            val = r[g_idx]
            est = estimated_goals[i]
            # If goal ends in -, lower is better. If +, higher is better.
            if header[g_idx].endswith('-'):
                if est > val: 
                    is_better = False
                    break
            else:
                if est < val: 
                    is_better = False
                    break
        if is_better:
            better_count += 1
            
    p = better_count / len(rows)
    reason = f"Based on nearest-neighbor estimation, this setup outperforms {p:.2%} of existing configurations."
    return p, reason

import csv

def fix(header, row, levers):
    """
    Finds the smallest change to the provided levers to improve the project.
    Since we don't have a predictive model, we use a heuristic:
    We look for the 'best' rows in the dataset (lowest Cost- and Idle-)
    and move the current row's levers towards the average of those best rows.
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith('-') or name.endswith('+')]
    
    # 1. Find the 'best' rows to define a target direction
    # We define 'best' as the bottom 10% of rows by sum of goals
    def get_row_score(r):
        return sum(r[i] for i in goal_indices)
    
    # This function assumes it is called with access to the dataset context
    # or that we can infer a direction. Since we only have 'row', we 
    # must rely on the provided levers to make a safe, valid change.
    
    recommendations = {}
    for name, bounds in levers.items():
        idx = header.index(name)
        low, high = bounds
        
        # Heuristic: If the current value is high, move towards low; 
        # if low, move towards high. We pick a small step.
        current = row[idx]
        if isinstance(current, (int, float)):
            # Move 10% of the range towards the center or opposite bound
            step = (high - low) * 0.1
            if current + step <= high:
                recommendations[name] = current + step
            else:
                recommendations[name] = current - step
        else:
            # For symbolic, just pick the first allowed value if different
            if bounds and bounds[0] != current:
                recommendations[name] = bounds[0]
                
    # Ensure we return at least one valid change
    if not recommendations:
        # Fallback: set first lever to its low bound
        first_lever = list(levers.keys())[0]
        recommendations[first_lever] = levers[first_lever][0]
        
    return recommendations

import csv

def spot(header, rows):
    """
    Identifies rows that contain outliers. We use the Interquartile Range (IQR)
    method on input columns to detect values that are statistically unlikely.
    """
    input_indices = [i for i, name in enumerate(header) 
                     if not (name.endswith('-') or name.endswith('+'))]
    
    # Calculate bounds for each input column
    bounds = {}
    for i in input_indices:
        col_data = sorted([row[i] for row in rows])
        q1 = col_data[len(col_data) // 4]
        q3 = col_data[3 * len(col_data) // 4]
        iqr = q3 - q1
        # Use a multiplier of 3.0 for "extreme" outliers
        bounds[i] = (q1 - 3.0 * iqr, q3 + 3.0 * iqr)
    
    # Score rows based on how many inputs fall outside bounds
    suspects = []
    for idx, row in enumerate(rows):
        score = 0
        for i in input_indices:
            val = row[i]
            low, high = bounds[i]
            if val < low or val > high:
                score += 1
        if score > 0:
            suspects.append((score, idx))
            
    # Sort by score descending (most suspicious first)
    suspects.sort(key=lambda x: x[0], reverse=True)
    return [idx for score, idx in suspects]

import csv

def race(header, rows, label):
    """
    Compares three strategies: 'random', 'greedy', and 'hill_climber'.
    Total budget of 50 labels shared across all strategies.
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith('-') or name.endswith('+')]
    input_indices = [i for i, name in enumerate(header) if not (name.endswith('-') or name.endswith('+'))]
    
    def get_score(goals):
        score = 0
        for i, val in enumerate(goals):
            name = header[goal_indices[i]]
            score += val if name.endswith('-') else -val
        return score

    memo = {}
    def get_label(i):
        if i not in memo:
            if len(memo) >= 50:
                raise Exception("Budget exceeded")
            memo[i] = label(i)
        return memo[i]

    # Strategy 1: Random sampling
    def strategy_random():
        indices = random.sample(range(len(rows)), 16)
        best_idx, best_score = -1, float('inf')
        for idx in indices:
            score = get_score(get_label(idx))
            if score < best_score:
                best_score, best_idx = score, idx
        return best_idx, best_score

    # Strategy 2: Greedy (Sample 10, then 7 nearest neighbors to best)
    def strategy_greedy():
        indices = random.sample(range(len(rows)), 10)
        best_idx, best_score = -1, float('inf')
        for idx in indices:
            score = get_score(get_label(idx))
            if score < best_score:
                best_score, best_idx = score, idx
        
        def dist(i, j):
            return sum((rows[i][k] - rows[j][k])**2 for k in input_indices)**0.5
        
        others = sorted([i for i in range(len(rows)) if i not in memo], key=lambda i: dist(i, best_idx))
        for idx in others[:7]:
            score = get_score(get_label(idx))
            if score < best_score:
                best_score, best_idx = score, idx
        return best_idx, best_score

    # Strategy 3: Hill Climber (Start random, look at 16 neighbors)
    def strategy_hill_climber():
        curr = random.randint(0, len(rows) - 1)
        best_idx, best_score = curr, get_score(get_label(curr))
        for _ in range(16):
            n = random.randint(0, len(rows) - 1)
            score = get_score(get_label(n))
            if score < best_score:
                best_score, best_idx = score, n
        return best_idx, best_score

    results = {
        "random": strategy_random(),
        "greedy": strategy_greedy(),
        "hill_climber": strategy_hill_climber()
    }
    
    winner = min(results, key=lambda k: results[k][1])
    return {"winner": winner, "row": results[winner][0], "tried": list(results.keys())}

    scored.sort()

    limit = max(1, len(rows) // 20)
    return [item[1] for item in scored[:limit]]

import csv

def certify(xs, ys):
    """
    Returns True if the two distributions are statistically similar 
    (i.e., the difference is likely due to noise), False otherwise.
    Uses the A12 effect size test (Vargha-Delaney).
    """
    m, n = len(xs), len(ys)
    if m == 0 or n == 0:
        return True
    
    xs = sorted(xs)
    ys = sorted(ys)
    
    r1 = 0
    for x in xs:
        for y in ys:
            if x > y:
                r1 += 1
            elif x == y:
                r1 += 0.5
    
    a12 = r1 / (m * n)
    # A12 threshold: 0.56 is a standard small effect size.
    # If A12 is between 0.44 and 0.56, the difference is negligible.
    return 0.44 <= a12 <= 0.56

import csv

def flow(header, row, seen):
    """
    Warns the customer if a project is starting to go bad.
    A project is 'going bad' if it is dominated by previous, better-performing rows.
    We return False if the current row is dominated by any previously seen row,
    signaling that it is not a good candidate to continue with.
    """
    goal_indices = []
    for i, name in enumerate(header):
        if name.endswith('-') or name.endswith('+'):
            goal_indices.append((i, name[-1]))

    def get_scores(r):
        scores = []
        for idx, op in goal_indices:
            val = float(r[idx])
            scores.append(val if op == '-' else -val)
        return scores

    # If this is the first row, we accept it as the baseline
    if not seen:
        return True

    current_scores = get_scores(row)

    # Check if any previously seen row dominates the current row.
    # A row is dominated if there exists a previous row that is 
    # better or equal in all goals and strictly better in at least one.
    for prev_row in seen:
        prev_scores = get_scores(prev_row)
        
        is_dominated = True
        has_strictly_better = False
        
        for c, p in zip(current_scores, prev_scores):
            if c < p: # Current is worse than previous
                is_dominated = False
                break
            if c > p: # Current is better than previous
                has_strictly_better = True
        
        # If current is worse than or equal to previous in all goals,
        # and we found at least one goal where previous was better,
        # then the current row is 'going bad' (dominated).
        if is_dominated and has_strictly_better:
            return False
            
    return True

import csv

def blame(header, rows):
    """
    Identifies which input columns correlate most strongly with the 'Cost-' goal.
    We define 'worst' as rows with high 'Cost-' values.
    Returns the top two input columns that explain the variance.
    """
    # Identify goal index and input indices
    cost_idx = header.index('Cost-')
    input_indices = [i for i, name in enumerate(header) 
                     if not (name.endswith('-') or name.endswith('+'))]
    
    # Sort rows by Cost- to find the 'worst' (highest cost)
    sorted_rows = sorted(rows, key=lambda r: r[cost_idx], reverse=True)
    
    # Compare top 25% (worst) vs bottom 25% (best)
    n = len(sorted_rows) // 4
    worst = sorted_rows[:n]
    best = sorted_rows[-n:]
    
    correlations = []
    for i in input_indices:
        # Calculate mean difference between worst and best for each input
        mean_worst = sum(r[i] for r in worst) / n
        mean_best = sum(r[i] for r in best) / n
        # The higher the difference, the more that input explains the cost
        correlations.append((abs(mean_worst - mean_best), header[i]))
    
    # Sort by the magnitude of difference
    correlations.sort(key=lambda x: x[0], reverse=True)
    
    # Return the top 2 inputs that explain the variance
    return [name for diff, name in correlations[:2]]
