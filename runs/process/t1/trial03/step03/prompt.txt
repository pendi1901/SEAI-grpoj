import csv

def remember(header, rows):
    """
    Identifies rows that perform well across all goal columns.
    Goals ending in '-' are minimized, '+' are maximized.
    """
    goal_indices = []
    for i, name in enumerate(header):
        if name.endswith('-') or name.endswith('+'):
            goal_indices.append((i, name[-1]))

    def score_row(row):
        # We want to minimize all goals. 
        # For '-', lower is better. For '+', higher is better (so we negate).
        scores = []
        for idx, op in goal_indices:
            val = float(row[idx])
            scores.append(val if op == '-' else -val)
        return scores

    # Calculate scores for all rows
    scored_rows = []
    for i, row in enumerate(rows):
        scored_rows.append((score_row(row), i))

    # Sort by the goals lexicographically (all goals are equally important)
    scored_rows.sort()

    # Return the top 5% (or at least 1)
    limit = max(1, len(rows) // 20)
    return [item[1] for item in scored_rows[:limit]]

import csv

def guess(header, train, test):
    """
    Estimates the value of the first goal column for each test row
    by finding the nearest neighbor in the training data based on 
    Euclidean distance of input columns.
    """
    # Identify input columns (those that are not goals)
    input_indices = [i for i, name in enumerate(header) 
                     if not (name.endswith('-') or name.endswith('+'))]
    
    # Identify the first goal column index
    goal_idx = next(i for i, name in enumerate(header) 
                    if name.endswith('-') or name.endswith('+'))

    predictions = []
    for t_row in test:
        best_dist = float('inf')
        best_val = 0.0
        
        for train_row in train:
            dist = 0
            for i in input_indices:
                # Assuming numeric inputs as per problem description
                dist += (float(t_row[i]) - float(train_row[i]))**2
            
            if dist < best_dist:
                best_dist = dist
                best_val = float(train_row[goal_idx])
        
        predictions.append(best_val)
        
    return predictions
