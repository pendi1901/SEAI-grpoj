import csv

def remember(header, rows):
    """
    Identifies rows that perform well across all goal columns.
    Goals ending in '-' are minimized, '+' are maximized.
    """
    goal_indices = []
    for i, name in enumerate(header):
        if name.endswith('-') or name.endswith('+'):
            goal_indices.append((i, name[-1]))

    def score_row(row):
        # Calculate a normalized score for each goal
        # Since all data is numeric and normalized [0, 1], 
        # we can simply sum the values (minimizing '-' and maximizing '+')
        total = 0
        for idx, op in goal_indices:
            val = float(row[idx])
            if op == '-':
                total += val
            else:
                total += (1.0 - val)
        return total

    # Score all rows
    scored = []
    for i, row in enumerate(rows):
        scored.append((score_row(row), i))

    # Sort by score (ascending, as we want lower values for '-' and higher for '+')
    scored.sort()

    # Return top 5% (or at least 1)
    limit = max(1, len(rows) // 20)
    return [item[1] for item in scored[:limit]]

import csv

def guess(header, train, test):
    """
    Estimates the first goal column (Cost-) for test rows by finding the 
    nearest neighbor in the training data based on input columns.
    """
    # Identify input columns (those not ending in - or +)
    input_indices = [i for i, name in enumerate(header) 
                     if not (name.endswith('-') or name.endswith('+'))]
    
    # Identify the index of the first goal column
    goal_idx = next(i for i, name in enumerate(header) 
                    if name.endswith('-') or name.endswith('+'))

    def dist(row1, row2):
        """Euclidean distance between input columns."""
        d = 0
        for i in input_indices:
            v1, v2 = float(row1[i]), float(row2[i])
            d += (v1 - v2) ** 2
        return d ** 0.5

    results = []
    for t_row in test:
        # Find the row in train that is closest to the current test row
        best_dist = float('inf')
        best_val = 0.0
        for train_row in train:
            d = dist(t_row, train_row)
            if d < best_dist:
                best_dist = d
                best_val = float(train_row[goal_idx])
        results.append(best_val)
    
    return results
