def remember(header, rows):
    """
    Identifies rows that perform well across all goal columns.
    Goals ending in '-' are minimized, '+' are maximized.
    """
    goal_indices = []
    for i, name in enumerate(header):
        if name.endswith('-') or name.endswith('+'):
            goal_indices.append((i, name[-1]))

    def score_row(row):
        # Calculate a normalized score for each goal
        # Since all data is numeric and normalized [0, 1], 
        # we can simply sum the values (minimizing '-' and maximizing '+')
        total = 0
        for idx, op in goal_indices:
            val = float(row[idx])
            if op == '-':
                total += val
            else:
                total += (1.0 - val)
        return total

    # Score all rows
    scored = []
    for i, row in enumerate(rows):
        scored.append((score_row(row), i))

    # Sort by score (ascending, as we want lower values for '-' and higher for '+')
    scored.sort()

    # Return top 5% (or at least 1)
    limit = max(1, len(rows) // 20)
    return [item[1] for item in scored[:limit]]

import csv

def certify(xs, ys):
    """
    Returns True if xs and ys are statistically similar (same distribution),
    False if they differ significantly. Uses a simple A12 effect size check
    and a basic variance check to determine if the difference is 'real'.
    """
    if not xs or not ys:
        return True

    # A12 effect size: probability that a random value from xs > random value from ys
    def a12(lst1, lst2):
        more = 0
        same = 0
        for x in lst1:
            for y in lst2:
                if x > y: more += 1
                elif x == y: same += 1
        return (more + 0.5 * same) / (len(lst1) * len(lst2))

    # If A12 is close to 0.5, they are likely the same.
    # Threshold 0.56 is a common small effect size threshold.
    effect = a12(xs, ys)
    return 0.44 <= effect <= 0.56

import csv

def flow(header, row, seen):
    """
    Decides whether to accept a row based on the 'Secretary Problem' logic.
    We observe the first 20% of rows to establish a baseline (the best seen so far),
    then accept the first row that is better than that baseline.
    """
    # Identify goal indices
    goal_indices = []
    for i, name in enumerate(header):
        if name.endswith('-') or name.endswith('+'):
            goal_indices.append((i, name[-1]))

    def score_row(r):
        total = 0
        for idx, op in goal_indices:
            val = float(r[idx])
            if op == '-':
                total += val
            else:
                total += (1.0 - val)
        return total

    # If this is the first row, we must let it go to start building a baseline
    if not seen:
        return False

    # Calculate the best score seen so far
    best_seen_score = min(score_row(r) for r in seen)
    
    # Strategy: 
    # 1. Observe for the first 20% of the total expected population (approx 160 rows)
    # 2. After that, accept if the current row is better than the best seen so far.
    # 3. If we reach the last few rows, accept the best available.
    
    current_score = score_row(row)
    
    # If we are in the exploration phase (first 20%), keep looking
    if len(seen) < 160:
        return False
    
    # If we are at the very end, accept if it's better than the best seen
    # Otherwise, accept if it beats the baseline
    if current_score < best_seen_score:
        return True
        
    return False

def blame(header, rows):
    """
    Identifies input columns that correlate most strongly with the 'Idle-' goal.
    We define 'worst' as rows with high 'Idle-' values.
    """
    # Identify goal indices and input indices
    goal_indices = {name: i for i, name in enumerate(header) if name.endswith('-') or name.endswith('+')}
    input_indices = [i for i, name in enumerate(header) if not (name.endswith('-') or name.endswith('+'))]
    
    idle_idx = goal_indices['Idle-']
    
    # Sort rows by 'Idle-' to find the worst ones (highest idle time)
    # We want to see which input features are high/low in these worst rows
    sorted_rows = sorted(rows, key=lambda r: float(r[idle_idx]), reverse=True)
    
    # Take the worst 20%
    worst_rows = sorted_rows[:max(1, len(rows) // 5)]
    
    # Calculate correlation: simple difference between mean of worst rows and global mean
    # A high positive difference suggests the input feature drives the idle time
    correlations = []
    for i in input_indices:
        col_vals = [float(r[i]) for r in rows]
        mean_all = sum(col_vals) / len(col_vals)
        
        worst_vals = [float(r[i]) for r in worst_rows]
        mean_worst = sum(worst_vals) / len(worst_vals)
        
        # We look for features that are significantly higher in the worst rows
        diff = mean_worst - mean_all
        correlations.append((diff, header[i]))
    
    # Sort by correlation strength
    correlations.sort(key=lambda x: x[0], reverse=True)
    
    # Return the top 3 input columns that correlate with high idle time
    return [name for diff, name in correlations[:3]]

import csv

def justify(header, rows, row):
    """
    Estimates the performance of a given row by finding the nearest neighbor
    in the training data, then calculates its percentile rank against all rows.
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith('-') or name.endswith('+')]
    input_indices = [i for i, name in enumerate(header) if not (name.endswith('-') or name.endswith('+'))]
    
    def get_score(r):
        total = 0
        for i in goal_indices:
            val = float(r[i])
            if header[i].endswith('-'):
                total += val
            else:
                total += (1.0 - val)
        return total

    # Estimate goals for the input row using 1-nearest neighbor
    def dist(r1, r2):
        d = 0
        for i in input_indices:
            d += (float(r1[i]) - float(r2[i])) ** 2
        return d ** 0.5

    best_dist = float('inf')
    best_row = None
    for r in rows:
        d = dist(row, r)
        if d < best_dist:
            best_dist = d
            best_row = r
            
    # Use the estimated score based on the nearest neighbor's goal values
    estimated_score = get_score(best_row)
    
    # Calculate percentile rank: fraction of rows better than this one
    all_scores = [get_score(r) for r in rows]
    better_count = sum(1 for s in all_scores if s < estimated_score)
    p = better_count / len(rows)
    
    reason = f"This setup is estimated to be better than {int(p * 100)}% of all known configurations based on its similarity to high-performing historical data."
    return p, reason

import csv

def spot(header, rows):
    """
    Identifies rows that are 'outliers' based on the Mahalanobis-like distance
    or simply by checking if input values are outside the expected range [0, 1].
    Since the data is normalized [0, 1], any value outside this range is suspicious.
    Additionally, we calculate the distance of each row from the centroid of the data
    to find rows that are statistically unusual.
    """
    input_indices = [i for i, name in enumerate(header) 
                     if not (name.endswith('-') or name.endswith('+'))]
    
    suspicious = []
    
    # 1. Check for values outside [0, 1] (Data entry errors)
    for i, row in enumerate(rows):
        for idx in input_indices:
            val = float(row[idx])
            if val < 0 or val > 1:
                suspicious.append((1.0, i)) # High priority
                break
    
    # 2. Check for statistical outliers (rows far from the mean)
    # Calculate column means
    means = []
    for idx in input_indices:
        vals = [float(r[idx]) for r in rows]
        means.append(sum(vals) / len(vals))
    
    # Calculate distance from centroid
    distances = []
    for i, row in enumerate(rows):
        d = 0
        for j, idx in enumerate(input_indices):
            d += (float(row[idx]) - means[j]) ** 2
        distances.append((d**0.5, i))
    
    # Sort by distance descending (furthest are most suspicious)
    distances.sort(key=lambda x: x[0], reverse=True)
    
    # Add top 5% of statistical outliers to suspicious list if not already there
    limit = len(rows) // 20
    for d, i in distances[:limit]:
        suspicious.append((d, i))
        
    # Sort by suspicion score (descending) and return unique indices
    suspicious.sort(key=lambda x: x[0], reverse=True)
    
    seen = set()
    result = []
    for _, i in suspicious:
        if i not in seen:
            result.append(i)
            seen.add(i)
            
    return result
