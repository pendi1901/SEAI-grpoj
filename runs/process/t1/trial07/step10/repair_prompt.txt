def remember(header, rows):

    Identifies rows that perform well across all goal columns.
    Goals ending in '-' are minimized, '+' are maximized.
    """
    goal_indices = []
    for i, name in enumerate(header):
        if name.endswith('-') or name.endswith('+'):
            goal_indices.append((i, name[-1]))

    def score_row(row):
        # Calculate a normalized score for the row.
        # Since we want to be "unusually well" on all, we look for 
        # rows that are consistently low for '-' and high for '+'.
        # We use a simple sum of normalized values.
        total = 0
        for idx, op in goal_indices:
            val = float(row[idx])
            if op == '-':
                total += val
            else:
                total += (1.0 - val)
        return total

    # Score all rows
    scored = []
    for i, row in enumerate(rows):
        scored.append((score_row(row), i))
    
    # Sort by score (ascending, as we want lower values for '-' and 
    # higher values for '+' which we inverted to 1-val)
    scored.sort()
    
    # Return top 5% (or at least 1)
    limit = max(1, len(rows) // 20)
    return [item[1] for item in scored[:limit]]

import csv
import math

def certify(xs, ys):
    """
    Returns True if xs and ys are statistically similar (same distribution),
    False if they are significantly different. Uses a simple A12 effect size test.
    """
    def a12(lst1, lst2):
        m, n = len(lst1), len(lst2)
        greater = 0
        for x in lst1:
            for y in lst2:
                if x > y: greater += 1
                elif x == y: greater += 0.5
        return greater / (m * n)

    # A12 threshold: 0.56 is a small effect size, 0.64 medium, 0.71 large.
    # If A12 is close to 0.5, they are the same.
    score = a12(xs, ys)
    return abs(0.5 - score) < 0.147

import csv

def flow(header, row, seen):
    """
    Decides whether to accept a row based on whether it is better than 
    any row seen so far. We use a simple dominance/score metric.
    """
    goal_indices = [i for i, name in enumerate(header) 
                    if name.endswith('-') or name.endswith('+')]

    def get_score(r):
        # Lower is better for '-', higher is better for '+'
        score = 0.0
        for i in goal_indices:
            val = r[i]
            if header[i].endswith('-'):
                score += val
            else:
                score += (1.0 - val)
        return score

    current_score = get_score(row)
    
    # If this is the first row, accept it
    if not seen:
        return True
    
    # Compare against the best seen so far
    best_seen_score = min(get_score(r) for r in seen)
    
    # Accept if it's better than everything seen so far
    return current_score < best_seen_score

def blame(header, rows):
    """
    Identifies input columns that correlate most strongly with high 'Idle-' values.
    'Idle-' is a goal to minimize, so high values represent the 'worst' setups.
    """
    # Identify goal indices and input indices
    goal_indices = [i for i, name in enumerate(header) if name.endswith('-') or name.endswith('+')]
    input_indices = [i for i, name in enumerate(header) if i not in goal_indices]
    
    idle_idx = header.index('Idle-')
    
    # Sort rows by 'Idle-' to find the worst ones (highest values)
    sorted_rows = sorted(rows, key=lambda r: r[idle_idx], reverse=True)
    
    # Take the worst 20%
    worst_rows = sorted_rows[:max(1, len(rows) // 5)]
    
    # Calculate correlation (simple difference in means)
    # We look for inputs that are significantly higher or lower in the worst rows
    # compared to the average of the whole dataset.
    all_means = [sum(r[i] for r in rows) / len(rows) for i in input_indices]
    worst_means = [sum(r[i] for r in worst_rows) / len(worst_rows) for i in input_indices]
    
    # Calculate absolute delta to see which inputs deviate most in bad setups
    deltas = []
    for i, idx in enumerate(input_indices):
        delta = abs(worst_means[i] - all_means[i])
        deltas.append((delta, header[idx]))
    
    # Sort by delta descending and return top 3
    deltas.sort(key=lambda x: x[0], reverse=True)
    return [name for delta, name in deltas[:3]]

def justify(header, rows, row):

    Estimates the performance of a given row by comparing it to the 
    training data and returns its percentile rank and a justification.
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith('-') or name.endswith('+')]
    input_indices = [i for i, name in enumerate(header) if i not in goal_indices]

    def get_score(r):
        score = 0.0
        for i in goal_indices:
            val = r[i]
            if header[i].endswith('-'):
                score += val
            else:
                score += (1.0 - val)
        return score

    # Estimate goal values for the input row using 1-nearest neighbor
    best_dist = float('inf')
    best_row = None
    for train_row in rows:
        dist = sum((row[i] - train_row[i])**2 for i in input_indices)
        if dist < best_dist:
            best_dist = dist
            best_row = train_row
    
    # Create a copy of the row with estimated goals
    estimated_row = list(row)
    for i in goal_indices:
        estimated_row[i] = best_row[i]
    
    # Calculate percentile rank
    my_score = get_score(estimated_row)
    better_count = sum(1 for r in rows if get_score(r) > my_score)
    p = better_count / len(rows)
    
    reason = f"This setup is better than {p:.1%} of all known configurations based on its input characteristics."
    return p, reason

def fix(header, row, levers):

    """

    def get_score(r):
        score = 0.0
        for i in goal_indices:
            val = r[i]
            if header[i].endswith('-'):
                score += val
            else:
                score += (1.0 - val)
        return score

    best_change = {}
    best_score = get_score(row)
    
    # Try small adjustments for each lever
    for col_name, bounds in levers.items():
        if col_name not in header:
            continue
        idx = header.index(col_name)
        low, high = bounds
        
        # Test a few points in the range
        for step in [0.05, 0.1, 0.2]:

                    s = get_score(temp_row)
                    if s < best_score:
                        best_score = s
                        best_change = {col_name: new_val}
                    
    return best_change

import csv

def spot(header, rows):
    """
    Identifies rows that are outliers in the input space.
    We use the Mahalanobis-like distance (or simple Euclidean distance to the centroid)
    to find rows that deviate significantly from the rest of the data.
    """
    # Identify input columns (those not ending in - or +)
    input_indices = [i for i, name in enumerate(header) 
                     if not (name.endswith('-') or name.endswith('+'))]
    
    # Calculate the centroid of the input space
    num_rows = len(rows)
    if num_rows == 0:
        return []
    
    centroid = []
    for i in input_indices:
        col_vals = [row[i] for row in rows]
        centroid.append(sum(col_vals) / num_rows)
    
    # Calculate distance of each row from the centroid
    # Rows with high distance are considered "mistyped" or outliers
    distances = []
    for idx, row in enumerate(rows):
        dist_sq = 0.0
        for i, col_idx in enumerate(input_indices):
            dist_sq += (row[col_idx] - centroid[i]) ** 2
        distances.append((dist_sq**0.5, idx))
    
    # Sort by distance descending (most suspicious first)
    distances.sort(key=lambda x: x[0], reverse=True)
    
    # Return the indices of the top 5% most suspicious rows
    limit = max(1, num_rows // 20)
    return [item[1] for item in distances[:limit]]
