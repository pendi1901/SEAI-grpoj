import csv

def remember(header, rows):
    """
    Identifies rows that perform well across all goal columns.
    Goals ending in '-' are minimized, '+' are maximized.
    """
    goal_indices = []
    for i, name in enumerate(header):
        if name.endswith('-') or name.endswith('+'):
            goal_indices.append((i, name[-1]))

    def score_row(row):
        # Calculate a normalized score for the row.
        # Since we want to be "unusually well" on all, we look for 
        # rows that are consistently low for '-' and high for '+'.
        # We use a simple sum of normalized values.
        total = 0
        for idx, op in goal_indices:
            val = float(row[idx])
            if op == '-':
                total += val
            else:
                total += (1.0 - val)
        return total

    # Score all rows
    scored = []
    for i, row in enumerate(rows):
        scored.append((score_row(row), i))
    
    # Sort by score (ascending, as we want lower values for '-' and 
    # higher values for '+' which we inverted to 1-val)
    scored.sort()
    
    # Return top 5% (or at least 1)
    limit = max(1, len(rows) // 20)
    return [item[1] for item in scored[:limit]]

import csv
import math

def certify(xs, ys):
    """
    Returns True if xs and ys are statistically similar (same distribution),
    False if they are significantly different. Uses a simple A12 effect size test.
    """
    def a12(lst1, lst2):
        m, n = len(lst1), len(lst2)
        greater = 0
        for x in lst1:
            for y in lst2:
                if x > y: greater += 1
                elif x == y: greater += 0.5
        return greater / (m * n)

    # A12 threshold: 0.56 is a small effect size, 0.64 medium, 0.71 large.
    # If A12 is close to 0.5, they are the same.
    score = a12(xs, ys)
    return abs(0.5 - score) < 0.147

import csv

def flow(header, row, seen):
    """
    Decides whether to accept a row based on whether it is better than 
    any row seen so far. We use a simple dominance/score metric.
    """
    goal_indices = [i for i, name in enumerate(header) 
                    if name.endswith('-') or name.endswith('+')]

    def get_score(r):
        # Lower is better for '-', higher is better for '+'
        score = 0.0
        for i in goal_indices:
            val = r[i]
            if header[i].endswith('-'):
                score += val
            else:
                score += (1.0 - val)
        return score

    current_score = get_score(row)
    
    # If this is the first row, accept it
    if not seen:
        return True
    
    # Compare against the best seen so far
    best_seen_score = min(get_score(r) for r in seen)
    
    # Accept if it's better than everything seen so far
    return current_score < best_seen_score

def blame(header, rows):
    """
    Identifies input columns that correlate most strongly with high 'Idle-' values.
    'Idle-' is a goal to minimize, so high values represent the 'worst' setups.
    """
    # Identify goal indices and input indices
    goal_indices = [i for i, name in enumerate(header) if name.endswith('-') or name.endswith('+')]
    input_indices = [i for i, name in enumerate(header) if i not in goal_indices]
    
    idle_idx = header.index('Idle-')
    
    # Sort rows by 'Idle-' to find the worst ones (highest values)
    sorted_rows = sorted(rows, key=lambda r: r[idle_idx], reverse=True)
    
    # Take the worst 20%
    worst_rows = sorted_rows[:max(1, len(rows) // 5)]
    
    # Calculate correlation (simple difference in means)
    # We look for inputs that are significantly higher or lower in the worst rows
    # compared to the average of the whole dataset.
    all_means = [sum(r[i] for r in rows) / len(rows) for i in input_indices]
    worst_means = [sum(r[i] for r in worst_rows) / len(worst_rows) for i in input_indices]
    
    # Calculate absolute delta to see which inputs deviate most in bad setups
    deltas = []
    for i, idx in enumerate(input_indices):
        delta = abs(worst_means[i] - all_means[i])
        deltas.append((delta, header[idx]))
    
    # Sort by delta descending and return top 3
    deltas.sort(key=lambda x: x[0], reverse=True)
    return [name for delta, name in deltas[:3]]

def justify(header, rows, row):
    """
    Estimates the performance of a given row by comparing it to the 
    training data and returns its percentile rank and a justification.
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith('-') or name.endswith('+')]
    input_indices = [i for i, name in enumerate(header) if i not in goal_indices]

    def get_score(r):
        score = 0.0
        for i in goal_indices:
            val = r[i]
            if header[i].endswith('-'):
                score += val
            else:
                score += (1.0 - val)
        return score

    # Estimate goal values for the input row using 1-nearest neighbor
    best_dist = float('inf')
    best_row = None
    for train_row in rows:
        dist = sum((row[i] - train_row[i])**2 for i in input_indices)
        if dist < best_dist:
            best_dist = dist
            best_row = train_row
    
    # Create a copy of the row with estimated goals
    estimated_row = list(row)
    for i in goal_indices:
        estimated_row[i] = best_row[i]
    
    # Calculate percentile rank
    my_score = get_score(estimated_row)
    better_count = sum(1 for r in rows if get_score(r) > my_score)
    p = better_count / len(rows)
    
    reason = f"This setup is better than {p:.1%} of all known configurations based on its input characteristics."
    return p, reason

import csv

def fix(header, row, levers):
    """
    Finds the best change to 'Plan' and 'Team Size' to minimize 'Cost-' and 'Idle-'.
    We perform a local search by testing small perturbations within the allowed levers.
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith('-') or name.endswith('+')]
    
    def get_score(r):
        score = 0.0
        for i in goal_indices:
            val = r[i]
            if header[i].endswith('-'):
                score += val
            else:
                score += (1.0 - val)
        return score

    best_change = {}
    best_score = get_score(row)
    
    # Try small adjustments for each lever
    for col_name, bounds in levers.items():
        if col_name not in header:
            continue
        idx = header.index(col_name)
        low, high = bounds
        
        # Test a few points in the range
        for step in [0.05, 0.1, 0.2]:
            for direction in [-1, 1]:
                new_val = row[idx] + (step * direction)
                new_val = max(low, min(high, new_val))
                
                if new_val != row[idx]:
                    temp_row = list(row)
                    temp_row[idx] = new_val
                    s = get_score(temp_row)
                    if s < best_score:
                        best_score = s
                        best_change = {col_name: new_val}
                    
    return best_change

import csv

def spot(header, rows):
    """
    Identifies rows that are outliers in the input space.
    We use the Mahalanobis-like distance (or simple Euclidean distance to the centroid)
    to find rows that deviate significantly from the rest of the data.
    """
    # Identify input columns (those not ending in - or +)
    input_indices = [i for i, name in enumerate(header) 
                     if not (name.endswith('-') or name.endswith('+'))]
    
    # Calculate the centroid of the input space
    num_rows = len(rows)
    if num_rows == 0:
        return []
    
    centroid = []
    for i in input_indices:
        col_vals = [row[i] for row in rows]
        centroid.append(sum(col_vals) / num_rows)
    
    # Calculate distance of each row from the centroid
    # Rows with high distance are considered "mistyped" or outliers
    distances = []
    for idx, row in enumerate(rows):
        dist_sq = 0.0
        for i, col_idx in enumerate(input_indices):
            dist_sq += (row[col_idx] - centroid[i]) ** 2
        distances.append((dist_sq**0.5, idx))
    
    # Sort by distance descending (most suspicious first)
    distances.sort(key=lambda x: x[0], reverse=True)
    
    # Return the indices of the top 5% most suspicious rows
    limit = max(1, num_rows // 20)
    return [item[1] for item in distances[:limit]]

import csv
import random

def race(header, rows, label):
    """
    Compares three strategies: 'random', 'greedy', and 'hillclimbing'.
    Total budget of 50 labels shared across all strategies.
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith('-') or name.endswith('+')]
    
    def get_score(row_goals):
        score = 0.0
        for i, val in enumerate(row_goals):
            name = header[goal_indices[i]]
            if name.endswith('-'):
                score += val
            else:
                score += (1.0 - val)
        return score

    labeled = {}
    def get_labeled_score(i):
        if i not in labeled:
            labeled[i] = get_score(label(i))
        return labeled[i]

    # Budget: 50 labels total. 3 strategies = 16 labels each (48 total).
    budget_per = 16
    results = {}

    # 1. Random Strategy
    best_random_idx = -1
    best_random_score = float('inf')
    for _ in range(budget_per):
        idx = random.randrange(len(rows))
        score = get_labeled_score(idx)
        if score < best_random_score:
            best_random_score = score
            best_random_idx = idx
    results["random"] = (best_random_score, best_random_idx)

    # 2. Greedy Strategy (Sample 16, pick best)
    sample = random.sample(range(len(rows)), budget_per)
    best_greedy_idx = min(sample, key=get_labeled_score)
    results["greedy"] = (get_labeled_score(best_greedy_idx), best_greedy_idx)

    # 3. Hillclimbing Strategy (Start at 1, then 15 steps)
    curr = random.randrange(len(rows))
    for _ in range(budget_per - 1):
        # Look at 5 random neighbors
        neighbors = random.sample(range(len(rows)), 5)
        best_n = min(neighbors, key=get_labeled_score)
        if get_labeled_score(best_n) < get_labeled_score(curr):
            curr = best_n
    results["hillclimbing"] = (get_labeled_score(curr), curr)

    # Determine winner
    winner = min(results, key=lambda k: results[k][0])
    
    return {
        "winner": winner,
        "row": results[winner][1],
        "tried": ["random", "greedy", "hillclimbing"]
    }

import csv

def certify(xs, ys):
    """
    Returns True if xs and ys are statistically similar (same distribution),
    False if they are significantly different. Uses a simple A12 effect size test.
    """
    def a12(lst1, lst2):
        m, n = len(lst1), len(lst2)
        if m == 0 or n == 0: return 0.5
        greater = 0
        for x in lst1:
            for y in lst2:
                if x > y: greater += 1
                elif x == y: greater += 0.5
        return greater / (m * n)

    # A12 threshold: 0.56 is a small effect size, 0.64 medium, 0.71 large.
    # If A12 is close to 0.5, they are the same.
    score = a12(xs, ys)
    return abs(0.5 - score) < 0.147

import csv

def flow(header, row, seen):
    """
    Warns the user if a new row is significantly worse than the historical trend.
    Returns True if the row is 'bad' (a warning), False otherwise.
    
    A row is considered 'bad' if its goal performance is significantly worse 
    than the average performance of the rows seen so far.
    """
    goal_indices = [i for i, name in enumerate(header) 
                    if name.endswith('-') or name.endswith('+')]
    
    def get_score(r):
        score = 0.0
        for i in goal_indices:
            val = r[i]
            # Normalize: '-' minimized, '+' maximized
            if header[i].endswith('-'):
                score += val
            else:
                score += (1.0 - val)
        return score

    if not seen:
        return False

    # Calculate the average score of all previously seen rows
    seen_scores = [get_score(r) for r in seen]
    avg_seen_score = sum(seen_scores) / len(seen_scores)
    
    # Calculate standard deviation to define a "bad" threshold
    variance = sum((s - avg_seen_score) ** 2 for s in seen_scores) / len(seen_scores)
    std_dev = variance ** 0.5
    
    # If the current row's score is more than 1 standard deviation worse 
    # than the average, we warn the user.
    current_score = get_score(row)
    
    # Threshold: if score is significantly higher than average (worse)
    if current_score > (avg_seen_score + std_dev):
        return True
    
    return False

def blame(header, rows):
    """
    Identifies the most important input columns by comparing the 
    distribution of inputs in the 'worst' rows versus the 'best' rows.
    
    Returns the names of the two input columns that show the largest 
    difference in mean values between the worst and best quartiles.
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith('-') or name.endswith('+')]
    input_indices = [i for i, name in enumerate(header) if i not in goal_indices]
    
    # Identify the goal column to minimize (Cost-)
    cost_idx = header.index('Cost-')
    
    # Score rows to find best and worst
    def get_score(row):
        return row[cost_idx]
    
    sorted_rows = sorted(rows, key=get_score)
    
    # Use quartiles to compare extremes
    q = len(sorted_rows) // 4
    best_rows = sorted_rows[:q]
    worst_rows = sorted_rows[-q:]
    
    # Find inputs with the largest difference in means between worst and best
    diffs = []
    for i in input_indices:
        best_mean = sum(r[i] for r in best_rows) / len(best_rows)
        worst_mean = sum(r[i] for r in worst_rows) / len(worst_rows)
        diffs.append((abs(worst_mean - best_mean), header[i]))
    
    # Sort by difference descending
    diffs.sort(key=lambda x: x[0], reverse=True)
    
    # Return the top 2 most impactful inputs
    return [name for diff, name in diffs[:2]]

def justify(header, rows, row):
    """
    Returns (p, reason) where p is the fraction of rows that 'row' is better than,
    and reason explains why 'row' is not as good as the best known configuration.
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith('-') or name.endswith('+')]
    input_indices = [i for i, name in enumerate(header) if i not in goal_indices]

    def get_score(r):
        score = 0.0
        for i in goal_indices:
            val = r[i]
            # Normalize: '-' minimized, '+' maximized
            if header[i].endswith('-'):
                score += val
            else:
                score += (1.0 - val)
        return score

    # 1. Estimate the goals for the provided 'row' using 1-nearest neighbor
    best_dist = float('inf')
    best_match = None
    for train_row in rows:
        dist = sum((row[i] - train_row[i])**2 for i in input_indices)
        if dist < best_dist:
            best_dist = dist
            best_match = train_row
    
    # Create a copy of the row with estimated goals
    estimated_row = list(row)
    for i in goal_indices:
        estimated_row[i] = best_match[i]
    
    # 2. Calculate percentile rank
    my_score = get_score(estimated_row)
    better_count = sum(1 for r in rows if get_score(r) < my_score)
    p = better_count / len(rows)
    
    # 3. Find the best row in the dataset to compare against
    best_overall = min(rows, key=get_score)
    best_score = get_score(best_overall)
    
    # 4. Construct the reason
    # Identify which goal is the biggest contributor to the score difference
    diffs = []
    for i in goal_indices:
        diff = abs(estimated_row[i] - best_overall[i])
        diffs.append((diff, header[i]))
    diffs.sort(key=lambda x: x[0], reverse=True)
    
    reason = (f"Your setup is better than {p:.4f} of configurations, but it is "
              f"outperformed by the best setup primarily due to differences in {diffs[0][1]}.")
    
    return p, reason

import csv

def spot(header, rows):
    """
    Identifies rows that are outliers in the input space.
    We calculate the centroid of the input columns and return the indices 
    of rows sorted by their Euclidean distance from that centroid.
    """
    # Identify input columns (those not ending in - or +)
    input_indices = [i for i, name in enumerate(header) 
                     if not (name.endswith('-') or name.endswith('+'))]
    
    num_rows = len(rows)
    if num_rows == 0:
        return []
    
    # Calculate the centroid of the input space
    centroid = []
    for i in input_indices:
        col_vals = [row[i] for row in rows if isinstance(row[i], (int, float))]
        if col_vals:
            centroid.append(sum(col_vals) / len(col_vals))
        else:
            centroid.append(0.0)
    
    # Calculate distance of each row from the centroid
    # Rows with high distance are considered "unseen" or outliers
    distances = []
    for idx, row in enumerate(rows):
        dist_sq = 0.0
        count = 0
        for i, col_idx in enumerate(input_indices):
            if isinstance(row[col_idx], (int, float)):
                dist_sq += (row[col_idx] - centroid[i]) ** 2
                count += 1
        
        # Normalize distance by number of valid inputs
        if count > 0:
            distances.append(((dist_sq / count)**0.5, idx))
        else:
            distances.append((0.0, idx))
    
    # Sort by distance descending (most suspicious/unseen first)
    distances.sort(key=lambda x: x[0], reverse=True)
    
    return [item[1] for item in distances]
