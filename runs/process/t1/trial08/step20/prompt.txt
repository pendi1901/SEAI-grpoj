import csv

def remember(header, rows):
    """
    Identifies rows that perform well across all goal columns.
    Goals ending in '-' are minimized, '+' are maximized.
    """
    goal_indices = []
    for i, name in enumerate(header):
        if name.endswith('-') or name.endswith('+'):
            goal_indices.append((i, name[-1]))

    def score_row(row):
        # Calculate a normalized score for the row.
        # Since we want to minimize '-' and maximize '+', 
        # we can treat all as "minimize" by negating '+' values.
        # We use the raw values since they are already normalized [0, 1].
        total = 0
        for idx, sign in goal_indices:
            val = float(row[idx])
            total += val if sign == '-' else (1.0 - val)
        return total

    # Score all rows
    scored = []
    for i, row in enumerate(rows):
        scored.append((score_row(row), i))

    # Sort by score (ascending, as we minimized everything)
    scored.sort()

    # Return top 1% (or at least 1, max 5%)
    limit = max(1, len(rows) // 100)
    return [item[1] for item in scored[:limit]]

import csv

def certify(xs, ys):
    """
    Returns True if the distributions of xs and ys are statistically 
    indistinguishable (the same apart from noise), False otherwise.
    Uses the A12 effect size test (Vargha-Delaney).
    """
    def a12(lst1, lst2):
        m, n = len(lst1), len(lst2)
        if m == 0 or n == 0: return 0.5
        
        # Count how many times items in lst1 are greater than items in lst2
        greater = 0
        for x in lst1:
            for y in lst2:
                if x > y: greater += 1
                elif x == y: greater += 0.5
        
        return greater / (m * n)

    # A12 threshold: 0.56 is a small effect size, 0.64 medium, 0.71 large.
    # If A12 is close to 0.5, they are effectively the same.
    score = a12(xs, ys)
    return 0.44 <= score <= 0.56

import csv

def flow(header, row, seen):
    """
    Implements a Secretary Problem strategy.
    We observe a sample (the first 37% of the data) to establish a baseline 
    of the 'best' seen so far, then accept the first row that beats that baseline.
    """
    goal_indices = []
    for i, name in enumerate(header):
        if name.endswith('-') or name.endswith('+'):
            goal_indices.append((i, name[-1]))

    def score(r):
        # Lower is better: minimize '-' and maximize '+' (by negating)
        s = 0
        for idx, sign in goal_indices:
            val = float(r[idx])
            s += val if sign == '-' else (1.0 - val)
        return s

    # Strategy: 1/e (~37%) exploration phase
    n = len(seen) + 1
    limit = int(len(seen) * 0.37) + 1
    
    # Calculate best score seen during exploration
    best_seen = float('inf')
    for i in range(min(len(seen), limit)):
        best_seen = min(best_seen, score(seen[i]))
    
    # If we are past exploration, accept if better than best_seen
    if len(seen) >= limit:
        return score(row) < best_seen
    
    return False

    # vs (mean of input in best 10% of Idle-)
    sorted_by_idle = sorted(worst_rows, key=lambda r: r[idle_idx])
    bottom_idle = sorted_by_idle[:len(worst_rows) // 2]
    top_idle = sorted_by_idle[len(worst_rows) // 2:]

def justify(header, rows, row):
    """
    Estimates the goals of 'row' using a 1-NN approach, then calculates
    the percentile rank of that row against the provided 'rows'.

    def get_score(r):
        s = 0
        for i in goal_indices:
            val = float(r[i])

        return s

    def dist(r1, r2):
        d = 0
        for i in input_indices:
            v1, v2 = r1[i], r2[i]
            d += (v1 - v2) ** 2
        return d ** 0.5

    # Estimate goals for the input row using 1-NN
    best_d = float('inf')
    best_row = None
    for r in rows:
        d = dist(row, r)
        if d < best_d:
            best_d = d
            best_row = r
    
    # Create a copy of the row with estimated goals
    estimated_row = list(row)
    for i in goal_indices:
        estimated_row[i] = best_row[i]

    # Calculate how many rows are worse than this one
    worse_count = 0
    for r in rows:
        if get_score(r) > est_score:
            worse_count += 1
            
    p = worse_count / len(rows)
    reason = f"This setup is estimated to outperform {p:.1%} of all historical configurations based on its input characteristics."
    
    return p, reason

def choose(header, rows, label):
    """
    Finds the best row by sampling and then exploring the neighborhood 
    of the best-performing sample.
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith(('-', '+'))]
    
    def get_score(row_goals):
        s = 0
        for i, val in enumerate(row_goals):
            idx = goal_indices[i]
            s += val if header[idx].endswith('-') else (1.0 - val)
        return s

    # 1. Initial sampling: Pick 25 random rows to establish a baseline
    indices = list(range(len(rows)))
    random.shuffle(indices)
    
    sampled_indices = indices[:25]
    best_idx = -1
    best_score = float('inf')
    
    for i in sampled_indices:
        goals = label(i)
        score = get_score(goals)
        if score < best_score:
            best_score = score
            best_idx = i
            
    # 2. Targeted exploration: Pick 25 rows closest to the best found so far
    def dist(r1_idx, r2_idx):
        d = 0
        for i, name in enumerate(header):
            if not (name.endswith('-') or name.endswith('+')):
                v1, v2 = rows[r1_idx][i], rows[r2_idx][i]
                if isinstance(v1, float) and isinstance(v2, float):
                    d += (v1 - v2) ** 2
                else:
                    d += 0 if v1 == v2 else 1
        return d ** 0.5

    # Find 25 nearest neighbors to the current best
    others = [i for i in indices if i not in sampled_indices]
    others.sort(key=lambda i: dist(i, best_idx))
    
    for i in others[:25]:
        goals = label(i)
        score = get_score(goals)
        if score < best_score:
            best_score = score
            best_idx = i
            
    return best_idx

    """

            val = float(r[i])

        return s

    # Since we only have the current row, we look for the lever that

    best_change = {}
    best_score_improvement = float('inf')

        idx = header.index(col_name)

        # Try adjusting the lever
        if isinstance(allowed, tuple):  # Numeric (low, high)

    val = levers[first_lever][0] if isinstance(levers[first_lever], tuple) else levers[first_lever][0]

import csv

def spot(header, rows):
    """
    Identifies rows that are outliers in the input space.
    We use the Mahalanobis-like distance (or simple Euclidean distance to the 
    median of the dataset) to find rows that deviate significantly from the 
    typical input patterns.
    """
    input_indices = [i for i, name in enumerate(header) if not (name.endswith(('-', '+')))]
    
    # Calculate the median for each input column to establish a 'typical' row
    medians = []
    for i in input_indices:
        vals = sorted([r[i] for r in rows if isinstance(r[i], (int, float))])
        medians.append(vals[len(vals) // 2])
    
    # Calculate distance of each row from the median row
    scores = []
    for row_idx, row in enumerate(rows):
        dist = 0
        for j, col_idx in enumerate(input_indices):
            val = row[col_idx]
            if isinstance(val, (int, float)):
                # Normalized distance from median
                dist += (val - medians[j]) ** 2
        scores.append((dist, row_idx))
    
    # Sort by distance descending (most suspicious first)
    scores.sort(key=lambda x: x[0], reverse=True)
    
    # Return the top 5% of most suspicious rows
    limit = max(1, len(rows) // 20)
    return [item[1] for item in scores[:limit]]

def race(header, rows, label):
    """
    Compares three strategies: random, greedy (nearest neighbor), and 
    a simple hill-climber, sharing a budget of 50 labels.
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith(('-', '+'))]
    input_indices = [i for i, name in enumerate(header) if not name.endswith(('-', '+'))]

    def get_score(row_goals):
        s = 0
        for i, val in enumerate(row_goals):
            idx = goal_indices[i]
            s += val if header[idx].endswith('-') else (1.0 - val)
        return s

    def dist(r1, r2):
        d = 0
        for i in input_indices:
            v1, v2 = r1[i], r2[i]
            d += (v1 - v2) ** 2
        return d ** 0.5

    # Shared budget
    budget = 50
    tried_indices = {} # index -> score

    def get_label(i):
        if i not in tried_indices:
            if len(tried_indices) >= budget:
                raise Exception("Exceeded 50 labels!")
            tried_indices[i] = get_score(label(i))
        return tried_indices[i]

    # 1. Random Strategy

            score = get_label(idx)
            if score < best_score:
                best_score, best_idx = score, idx
        return best_idx, best_score

    def strategy_greedy():
        samples = random.sample(range(len(rows)), 10)
        best_idx = min(samples, key=get_label)
        best_score = get_label(best_idx)
        # Look for 6 neighbors
        others = sorted([i for i in range(len(rows)) if i not in samples], key=lambda i: dist(rows[i], rows[best_idx]))
        for i in others[:6]:
            score = get_label(i)
            if score < best_score:
                best_score, best_idx = score, i
        return best_idx, best_score

    # 3. Hill Climber (Start random, move to better neighbor)
    def strategy_hill():
        curr = random.randint(0, len(rows) - 1)
        curr_score = get_label(curr)
        for _ in range(17): # 18 labels total
            neighbors = random.sample(range(len(rows)), 10)
            best_n = min(neighbors, key=lambda i: dist(rows[i], rows[curr]))
            n_score = get_label(best_n)
            if n_score < curr_score:
                curr, curr_score = best_n, n_score
        return curr, curr_score

    results = {
        "random": strategy_random(),
        "greedy": strategy_greedy(),
        "hill": strategy_hill()
    }

    winner = min(results, key=lambda k: results[k][1])
    return {
        "winner": winner,
        "row": results[winner][0],
        "tried": ["random", "greedy", "hill"]
    }

                # Apply the "twice as fast" logic to Dynamism distance
                if i == dynamism_idx:

import csv

def certify(xs, ys):
    """
    Returns True if the distributions of xs and ys are statistically 
    indistinguishable (the same apart from noise), False otherwise.
    Uses the A12 effect size test (Vargha-Delaney).
    """
    m, n = len(xs), len(ys)
    if m == 0 or n == 0:
        return True
    
    greater = 0
    for x in xs:
        for y in ys:
            if x > y:
                greater += 1
            elif x == y:
                greater += 0.5
    
    a12 = greater / (m * n)
    # A12 threshold: 0.56 is a small effect size.
    # If A12 is between 0.44 and 0.56, they are effectively the same.
    return 0.44 <= a12 <= 0.56

def get_score(header, row):
    """Calculates a normalized score for a row (lower is better)."""
    score = 0
    for i, name in enumerate(header):
        if name.endswith('-'):
            score += float(row[i])
        elif name.endswith('+'):
            score += (1.0 - float(row[i]))
    return score

import csv

def flow(header, row, seen):
    """
    Detects if the performance of incoming rows is starting to 'go bad'.
    We maintain a moving average of the performance of the last 20 rows.
    If the current row's score is significantly worse (e.g., > 2 standard 
    deviations) than the recent moving average, we return True to warn the user.
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith(('-', '+'))]

    def get_score(r):
        s = 0
        for i in goal_indices:
            val = float(r[i])
            s += val if header[i].endswith('-') else (1.0 - val)
        return s

    if len(seen) < 20:
        return False

    # Calculate scores of recent history
    recent_scores = [get_score(r) for r in seen[-20:]]
    avg = sum(recent_scores) / len(recent_scores)
    var = sum((x - avg) ** 2 for x in recent_scores) / len(recent_scores)
    std = var ** 0.5

    # If current row is significantly worse than the recent average
    current_score = get_score(row)
    if current_score > (avg + 2 * std):
        return True
    
    return False

    correlations.sort(reverse=True)
    return [name for _, name in correlations[:2]]

import csv

def spot(header, rows):
    """
    Identifies rows that are outliers in the input space.
    We calculate the distance of each row from the median of the dataset.
    Rows that are significantly distant from the median in the input space
    are considered 'unseen' or suspicious.
    """
    input_indices = [i for i, name in enumerate(header) if not (name.endswith(('-', '+')))]
    
    # Calculate the median for each input column to establish a 'typical' row
    medians = []
    for i in input_indices:
        vals = sorted([r[i] for r in rows if isinstance(r[i], (int, float))])
        if vals:
            medians.append(vals[len(vals) // 2])
        else:
            medians.append(0)
    
    # Calculate distance of each row from the median row
    # We use Euclidean distance on normalized inputs
    scores = []
    for row_idx, row in enumerate(rows):
        dist = 0
        for j, col_idx in enumerate(input_indices):
            val = row[col_idx]
            if isinstance(val, (int, float)):
                # Normalized distance from median
                dist += (val - medians[j]) ** 2
        scores.append((dist, row_idx))
    
    # Sort by distance descending (most suspicious/unseen first)
    scores.sort(key=lambda x: x[0], reverse=True)
    
    # Return the top 5% of most suspicious rows
    limit = max(1, len(rows) // 20)
    return [item[1] for item in scores[:limit]]
