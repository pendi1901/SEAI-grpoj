```python
import csv

def flow(header, row, seen):
    """
    Implements a Secretary Problem strategy.
    We observe a sample (the first 37% of the data) to establish a baseline 
    of the 'best' seen so far, then accept the first row that beats that baseline.
    """
    goal_indices = []
    for i, name in enumerate(header):
        if name.endswith('-') or name.endswith('+'):
            goal_indices.append((i, name[-1]))

    def score(r):
        # Lower is better: minimize '-' and maximize '+' (by negating)
        s = 0
        for idx, sign in goal_indices:
            val = float(r[idx])
            s += val if sign == '-' else (1.0 - val)
        return s

    # Strategy: 1/e (~37%) exploration phase
    n = len(seen) + 1
    limit = int(len(seen) * 0.37) + 1
    
    # Calculate best score seen during exploration
    best_seen = float('inf')
    for i in range(min(len(seen), limit)):
        best_seen = min(best_seen, score(seen[i]))
    
    # If we are past exploration, accept if better than best_seen
    if len(seen) >= limit:
        return score(row) < best_seen
    
    return False

if __name__ == "__main__":
    filename = "pom3a.n1000.train.csv"
    try:
        with open(filename, 'r') as f:
            reader = csv.reader(f)
            header = next(reader)
            rows = []
            for row in reader:
                rows.append([float(x) for x in row])
        
        seen = []
        accepted = []
        for row in rows:
            if flow(header, row, seen):
                accepted.append(row)
            seen.append(row)
            
        print(f"Processed {len(rows)} proposals.")
        print(f"Approved {len(accepted)} proposals based on the 37% threshold strategy.")
        print("This strategy balances the need to learn the market with the need to commit.")
        
    except FileNotFoundError:
        print(f"Error: {filename} not found.")
        exit(1)
```
