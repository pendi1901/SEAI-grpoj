import csv

def remember(header, rows):
    """
    Identifies rows that perform well across all goal columns.
    Goals ending in '-' are minimized, '+' are maximized.
    """
    goal_indices = []
    for i, name in enumerate(header):
        if name.endswith('-') or name.endswith('+'):
            goal_indices.append((i, name[-1]))

    def score_row(row):
        # Calculate a normalized score for the row.
        # Since we want to be "unusually well" on all, we look for 
        # rows that are consistently low (for -) or high (for +).
        # We use the raw values since they are already in [0, 1].
        scores = []
        for idx, op in goal_indices:
            val = float(row[idx])
            scores.append(val if op == '-' else (1.0 - val))
        return sum(scores) / len(scores)

    # Calculate scores for all rows
    scored_rows = []
    for i, row in enumerate(rows):
        scored_rows.append((score_row(row), i))

    # Sort by score (ascending, as we want lower values for both '-' and inverted '+')
    scored_rows.sort()

    # Return top 5% (or at least 1)
    limit = max(1, len(rows) // 20)
    return [item[1] for item in scored_rows[:limit]]

import csv

def certify(xs, ys):
    """
    Returns True if xs and ys are statistically similar (same distribution),
    False if they differ significantly. Uses a simple A12 effect size test.
    """
    def a12(lst1, lst2):
        m, n = len(lst1), len(lst2)
        greater = 0
        for x in lst1:
            for y in lst2:
                if x > y: greater += 1
                elif x == y: greater += 0.5
        return greater / (m * n)

    # If A12 is near 0.5, they are the same. 
    # Thresholds for small/medium/large effects are often 0.56/0.64/0.71.
    # We return True if the effect size is small (not significantly different).
    score = a12(xs, ys)
    return 0.44 <= score <= 0.56

def analyze_teams(header, rows):

    Splits data into small vs big teams and compares their 'Cost-' values.

    is_same = certify(small_teams, big_teams)

import csv

def flow(header, row, seen):
    """
    Decides whether to accept a row based on whether it is better than
    anything seen so far. We use a simple threshold: if the row is 
    better than the best seen so far, we accept it.
    """
    goal_indices = []
    for i, name in enumerate(header):
        if name.endswith('-') or name.endswith('+'):
            goal_indices.append((i, name[-1]))

    def get_score(r):
        scores = []
        for idx, op in goal_indices:
            val = float(r[idx])
            scores.append(val if op == '-' else (1.0 - val))
        return sum(scores) / len(scores)

    if not seen:
        return True

    current_score = get_score(row)
    best_seen_score = min(get_score(r) for r in seen)

    # Accept if this is the best we've seen so far
    return current_score < best_seen_score

import csv

def blame(header, rows):
    """
    Identifies which input columns correlate most strongly with high 'Idle-' values.
    'Idle-' is a goal to minimize, so high values represent 'worst' setups.
    """
    goal_idx = header.index('Idle-')
    input_indices = [i for i, name in enumerate(header) 
                     if not (name.endswith('-') or name.endswith('+'))]
    
    # Sort rows by 'Idle-' descending to focus on the worst setups
    sorted_rows = sorted(rows, key=lambda r: r[goal_idx], reverse=True)
    
    # Take the worst 25% of rows
    worst_rows = sorted_rows[:len(rows) // 4]
    
    # Calculate the correlation (simple difference of means)
    # We compare the mean of an input in the worst rows vs the mean of the whole set.
    # A large difference indicates that the input is a driver for the bad outcome.
    all_means = [sum(row[i] for row in rows) / len(rows) for i in input_indices]
    worst_means = [sum(row[i] for row in worst_rows) / len(worst_rows) for i in input_indices]
    
    # Calculate absolute delta to find the most influential inputs
    deltas = []
    for i, idx in enumerate(input_indices):
        delta = abs(worst_means[i] - all_means[i])
        deltas.append((delta, header[idx]))
    
    # Sort by delta descending and return the top 3 inputs
    deltas.sort(key=lambda x: x[0], reverse=True)
    return [name for delta, name in deltas[:3]]

def justify(header, rows, row):
    """
    Estimates the performance of 'row' by finding its nearest neighbor in 'rows',
    then calculates what fraction of 'rows' are worse than this estimate.
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith(('-', '+'))]
    input_indices = [i for i, name in enumerate(header) if not (name.endswith(('-', '+')))]
    
    # Estimate the missing goals using 1-Nearest Neighbor
    best_dist = float('inf')
    best_row = None
    for r in rows:
        dist = sum((float(row[i]) - float(r[i]))**2 for i in input_indices)
        if dist < best_dist:
            best_dist = dist
            best_row = r
            
    estimated_score = get_score(header, best_row)
    
    # Calculate how many rows are worse (have a higher score) than our estimate
    all_scores = [get_score(header, r) for r in rows]
    worse_count = sum(1 for s in all_scores if s > estimated_score)
    p = worse_count / len(rows)
    
    reason = f"This setup is estimated to outperform {p:.1%} of all possible configurations based on its similarity to known high-performing data."
    return p, reason

import csv

def spot(header, rows):
    """
    Identifies rows that are outliers based on the Mahalanobis-like distance
    of their input values from the column means.
    """
    input_indices = [i for i, name in enumerate(header) 
                     if not (name.endswith('-') or name.endswith('+'))]
    
    # Calculate means and standard deviations for each input column
    n = len(rows)
    means = []
    stds = []
    for i in input_indices:
        col_vals = [row[i] for row in rows]
        mean = sum(col_vals) / n
        means.append(mean)
        variance = sum((x - mean) ** 2 for x in col_vals) / n
        stds.append((variance ** 0.5) or 1.0)  # Avoid division by zero

    # Calculate an anomaly score for each row
    # The score is the average Z-score magnitude across all input columns
    scores = []
    for row_idx, row in enumerate(rows):
        z_scores = []
        for i, col_idx in enumerate(input_indices):
            z = abs(row[col_idx] - means[i]) / stds[i]
            z_scores.append(z)
        # Use the maximum Z-score as the indicator of "mistyped" data
        scores.append((max(z_scores), row_idx))
    
    # Sort by score descending (most suspicious first)
    scores.sort(key=lambda x: x[0], reverse=True)
    
    # Return only those that are significantly far from the mean (e.g., > 3 std devs)
    return [idx for score, idx in scores if score > 3.0]

def race(header, rows, label):
    """
    Compares three strategies: random, greedy, and local search.
    Budget: 50 labels total.
    """
    budget = 50
    labeled = {}
    
    def get_labeled_score(i):
        if i not in labeled:
            if len(labeled) >= budget:
                raise Exception("Budget exceeded")
            goals = label(i)
            temp_row = list(rows[i])
            goal_indices = [idx for idx, name in enumerate(header) if name.endswith(('-', '+'))]
            for j, g_idx in enumerate(goal_indices):
                temp_row[g_idx] = goals[j]
            labeled[i] = get_score(header, temp_row)
        return labeled[i]

    # 1. Random Strategy
    def strategy_random():
        indices = random.sample(range(len(rows)), 16)
        best_i = indices[0]
        best_s = get_labeled_score(best_i)
        for i in indices[1:]:
            s = get_labeled_score(i)
            if s < best_s:
                best_s, best_i = s, i
        return best_i, best_s

    # 2. Greedy Strategy (Sample 17, pick best)
    def strategy_greedy():
        indices = random.sample(range(len(rows)), 17)
        best_i = indices[0]
        best_s = get_labeled_score(best_i)
        for i in indices[1:]:
            s = get_labeled_score(i)
            if s < best_s:
                best_s, best_i = s, i
        return best_i, best_s

    # 3. Local Search Strategy (Sample 17, then local hill climb)
    def strategy_local():
        curr_i = random.choice(range(len(rows)))
        curr_s = get_labeled_score(curr_i)
        # Simple hill climb: check neighbors

            neighbor = random.choice(range(len(rows)))
            s = get_labeled_score(neighbor)
            if s < curr_s:
                curr_s, curr_i = s, neighbor
        return curr_i, curr_s

    strategies = {
        "random": strategy_random,
        "greedy": strategy_greedy,
        "local": strategy_local
    }
    
    results = {}
    for name, func in strategies.items():
        idx, score = func()
        results[name] = (score, idx)
        
    winner = min(results, key=lambda n: results[n][0])
    
    return {
        "winner": winner,
        "row": results[winner][1],

import csv

def certify(xs, ys):
    """
    Returns True if xs and ys are statistically similar (same distribution),
    False if they differ significantly. Uses the A12 effect size test.
    """
    m, n = len(xs), len(ys)
    greater = 0
    for x in xs:
        for y in ys:
            if x > y:
                greater += 1
            elif x == y:
                greater += 0.5
    
    a12 = greater / (m * n)
    # A12 threshold: 0.56 is a small effect size. 
    # If the effect size is small, we consider them the same.
    return 0.44 <= a12 <= 0.56

def flow(header, row, seen):
    """
    Warns the customer when a new row is significantly worse than the 
    historical average performance.
    """
    if not seen:
        return False
    
    # Calculate the average performance of all seen rows
    scores = [get_score(header, r) for r in seen]
    avg_score = sum(scores) / len(scores)
    
    # Calculate standard deviation of seen scores
    variance = sum((s - avg_score) ** 2 for s in scores) / len(scores)
    std_dev = variance ** 0.5
    
    # If the new row's score is more than 1 standard deviation worse 
    # than the average, we flag it as "going bad".
    current_score = get_score(header, row)
    
    # Return True if the performance is degrading (score is higher than average + threshold)
    return current_score > (avg_score + std_dev)

    """

                     if not (name.endswith('-') or name.endswith('+'))]

    scored_rows = sorted(rows, key=lambda r: get_score(header, r), reverse=True)
    worst_rows = scored_rows[:len(rows) // 4]
    
    # Calculate mean of inputs for all rows vs worst rows

    return [name for delta, name in deltas[:2]]
