```python
import csv

def get_score(header, row):
    """Calculates a normalized score [0, 1] where lower is better."""
    goal_indices = []
    for i, name in enumerate(header):
        if name.endswith('-') or name.endswith('+'):
            goal_indices.append((i, name[-1]))
    
    scores = []
    for idx, op in goal_indices:
        val = float(row[idx])
        scores.append(val if op == '-' else (1.0 - val))
    return sum(scores) / len(scores)

def remember(header, rows):
    """
    Identifies the top 5% of rows based on their goal performance.
    """
    scored = []
    for i, row in enumerate(rows):
        scored.append((get_score(header, row), i))
    
    # Sort by score ascending (lower is better)
    scored.sort()
    
    # Return top 5% (or at least 1)
    limit = max(1, len(rows) // 20)
    return [item[1] for item in scored[:limit]]

if __name__ == "__main__":
    filename = "pom3a.n1000.train.csv"
    try:
        with open(filename, 'r') as f:
            reader = csv.reader(f)
            header = next(reader)
            rows = []
            for row in reader:
                rows.append([float(x) for x in row])
        
        best_indices = remember(header, rows)
        
        print("Customer Inquiry: Past setups most like the best performers")
        print(f"Analysis: Found {len(best_indices)} top-performing historical setups.")
        print("These setups represent the configurations that achieved the best balance")
        print("of minimizing costs and maximizing scores/efficiency.")
        
        # Display summary of the top 3
        print("\nTop 3 historical setups (by index):")
        for idx in best_indices[:3]:
            row = rows[idx]
            score = get_score(header, row)
            print(f"  Index {idx}: Performance Score = {score:.4f}")
            
    except FileNotFoundError:
        print(f"Error: {filename} not found.")
        exit(1)
```
