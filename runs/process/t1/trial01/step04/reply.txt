import csv


# Module-level state that persists across calls to flow()
_state = {
    'goal_mins': {},
    'goal_maxs': {},
    'scores': [],
}


def flow(header, row, seen):
    """Approve (True) or reject (False) a proposal as it arrives."""
    goals = [c for c in header if c.endswith('-')]

    # Update running min/max from previously-seen rows
    n_seen_before = len(_state['scores'])
    for s_row in seen:
        for g in goals:
            v = float(s_row[header.index(g)])
            if g not in _state['goal_mins']:
                _state['goal_mins'][g] = v
                _state['goal_maxs'][g] = v
            else:
                _state['goal_mins'][g] = min(_state['goal_mins'][g], v)
                _state['goal_maxs'][g] = max(_state['goal_maxs'][g], v)

    # Score the current row (lower total is better for '-' goals)
    total = 0.0
    for g in goals:
        v = float(row[header.index(g)])
        mn = _state['goal_mins'].get(g, v)
        mx = _state['goal_maxs'].get(g, v)
        if mx == mn:
            norm = 0.0
        else:
            norm = (v - mn) / (mx - mn)
        total += norm

    score = total
    _state['scores'].append(score)

    # First row is always approved; thereafter use top-5% threshold
    if n_seen_before == 0:
        return True

    all_scores = sorted(_state['scores'])
    k = max(1, len(all_scores) // 20)
    threshold = all_scores[-k]

    return score <= threshold


def main():
    with open('pom3a.train.csv', newline='') as f:
        reader = csv.reader(f)
        header = next(reader)
        rows = []
        for row in reader:
            if len(row) == len(header):
                rows.append(row)

    approved = 0
    total = len(rows)
    for i, row in enumerate(rows):
        seen = [rows[j] for j in range(i)]
        if flow(header, row, seen):
            approved += 1

    print(f"Approved {approved} out of {total} proposals.")


if __name__ == "__main__":
    main()
