import csv

def flow(header, row, seen):
    """
    Decides whether to accept a row based on whether it is better than
    anything seen so far. We use a simple threshold: if the row is 
    better than the best seen so far, we accept it.
    """
    goal_indices = []
    for i, name in enumerate(header):
        if name.endswith('-') or name.endswith('+'):
            goal_indices.append((i, name[-1]))

    def get_score(r):
        scores = []
        for idx, op in goal_indices:
            val = float(r[idx])
            scores.append(val if op == '-' else (1.0 - val))
        return sum(scores) / len(scores)

    if not seen:
        return True

    current_score = get_score(row)
    best_seen_score = min(get_score(r) for r in seen)

    # Accept if this is the best we've seen so far
    return current_score < best_seen_score

if __name__ == "__main__":
    filename = "pom3a.n1000.train.csv"
    try:
        with open(filename, 'r') as f:
            reader = csv.reader(f)
            header = next(reader)
            rows = []
            for row in reader:
                rows.append([float(x) for x in row])

        accepted = []
        seen = []
        for row in rows:
            if flow(header, row, seen):
                accepted.append(row)
            seen.append(row)

        print(f"Customer Inquiry: Sequential Approval Process")
        print(f"Total proposals processed: {len(rows)}")
        print(f"Total proposals approved: {len(accepted)}")
        print("Strategy: Approved only when a proposal outperformed all previous ones.")
        
    except FileNotFoundError:
        print(f"Error: {filename} not found.")
        exit(1)
