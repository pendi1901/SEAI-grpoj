import csv

def spot(header, rows):
    """
    Identifies rows that are outliers based on the Mahalanobis-like distance
    of their input values from the column means.
    
    A row is suspicious if any of its input values are significantly far 
    from the mean (e.g., > 3 standard deviations).
    """
    input_indices = [i for i, name in enumerate(header) 
                     if not (name.endswith('-') or name.endswith('+'))]
    
    # Calculate means and standard deviations for each input column
    # We only consider rows that are not "?" (if any were masked)
    valid_rows = [r for r in rows if all(r[i] != "?" for i in input_indices)]
    n = len(valid_rows)
    if n == 0:
        return []

    means = []
    stds = []
    for i in input_indices:
        col_vals = [float(row[i]) for row in valid_rows]
        mean = sum(col_vals) / n
        means.append(mean)
        variance = sum((x - mean) ** 2 for x in col_vals) / n
        stds.append((variance ** 0.5) or 1.0)

    # Calculate an anomaly score for each row
    # The score is the maximum Z-score magnitude across all input columns
    scores = []
    for row_idx, row in enumerate(rows):
        # Skip rows with missing data
        if any(row[i] == "?" for i in input_indices):
            continue
            
        z_scores = []
        for i, col_idx in enumerate(input_indices):
            z = abs(float(row[col_idx]) - means[i]) / stds[i]
            z_scores.append(z)
        scores.append((max(z_scores), row_idx))
    
    # Sort by score descending (most suspicious first)
    scores.sort(key=lambda x: x[0], reverse=True)
    
    # Return indices of rows with Z-score > 3.0
    return [idx for score, idx in scores if score > 3.0]

if __name__ == "__main__":
    filename = "pom3a.n1000.train.csv"
    try:
        with open(filename, 'r') as f:
            reader = csv.reader(f)
            header = next(reader)
            rows = []
            for row in reader:
                processed_row = []
                for val in row:
                    try:
                        processed_row.append(float(val))
                    except ValueError:
                        processed_row.append(val)
                rows.append(processed_row)
        
        suspicious_indices = spot(header, rows)
        
        print("Customer Inquiry: Identifying Unseen/Anomalous Setups")
        if suspicious_indices:
            print(f"Analysis: Identified {len(suspicious_indices)} setups that are statistically anomalous.")
            print("These configurations contain input values significantly outside the range of historical data.")
            print(f"The most suspicious row index is: {suspicious_indices[0]}")
            print("Recommendation: These setups are 'unseen' by our model; proceed with caution.")
        else:
            print("Analysis: No highly anomalous setups were found. All data appears consistent with historical trends.")
            
    except FileNotFoundError:
        print(f"Error: {filename} not found.")
        exit(1)
