import csv
import math


def certify(xs, ys):
    """Returns True if xs and ys are the same apart from noise."""
    n1 = len(xs)
    n2 = len(ys)
    if n1 < 3 or n2 < 3:
        return False

    mean_x = sum(xs) / n1
    mean_y = sum(ys) / n2

    var_x = sum((x - mean_x) ** 2 for x in xs) / (n1 - 1) if n1 > 1 else 0.0
    var_y = sum((y - mean_y) ** 2 for y in ys) / (n2 - 1) if n2 > 1 else 0.0

    se = math.sqrt(var_x / n1 + var_y / n2)
    if se == 0:
        return True

    t_stat = abs(mean_x - mean_y) / se
    return t_stat < 2.0


def remember(header, rows):
    goals = [c for c in header if c.endswith('-')]

    goal_mins = {}
    goal_maxs = {}
    n = len(rows)
    for g in goals:
        vals = []
        for r in rows:
            v = float(r[header.index(g)])
            vals.append(v)
        goal_mins[g] = min(vals)
        goal_maxs[g] = max(vals)

    scored = []
    for i, r in enumerate(rows):
        total = 0.0
        for g in goals:
            v = float(r[header.index(g)])
            if goal_maxs[g] == goal_mins[g]:
                norm = 0.0
            else:
                norm = (v - goal_mins[g]) / (goal_maxs[g] - goal_mins[g])
            total += norm
        scored.append((total, i))

    scored.sort()
    k = max(1, len(scored) // 20)
    return [pos for _, pos in scored[:k]]


def main():
    with open('pom3a.train.csv', newline='') as f:
        reader = csv.reader(f)
        header = next(reader)
        rows = []
        for row in reader:
            if len(row) == len(header):
                rows.append(row)

    team_size_col = 'Team Size'
    cost_col = 'Cost-'

    ts_values = [float(r[header.index(team_size_col)]) for r in rows]
    median_ts = sorted(ts_values)[len(ts_values) // 2]

    small_cost = []
    big_cost = []

    for r in rows:
        ts = float(r[header.index(team_size_col)])
        c = float(r[header.index(cost_col)])
        if ts < median_ts:
            small_cost.append(c)
        else:
            big_cost.append(c)

    print(f"Small-team average Cost-: {sum(small_cost)/len(small_cost):.4f}")
    print(f"Big-team average Cost-: {sum(big_cost)/len(big_cost):.4f}")

    result = certify(small_cost, big_cost)
    if result:
        print("The difference is likely just noise - small and big teams cost about the same.")
    else:
        print("There's a real difference - one team size genuinely costs more/less than the other.")


if __name__ == "__main__":
    main()
