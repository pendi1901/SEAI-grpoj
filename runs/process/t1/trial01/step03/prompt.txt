import csv


def remember(header, rows):
    # Identify goal columns (end with '-') and input columns
    goals = [c for c in header if c.endswith('-')]
    
    # For each goal column, find min and max across all rows to normalize
    n = len(rows)
    goal_mins = {}
    goal_maxs = {}
    for g in goals:
        vals = []
        for r in rows:
            v = float(r[header.index(g)])
            vals.append(v)
        goal_mins[g] = min(vals)
        goal_maxs[g] = max(vals)
    
    # Score each row: sum of normalized (lower is better) scores
    scored = []
    for i, r in enumerate(rows):
        total = 0.0
        for g in goals:
            v = float(r[header.index(g)])
            if goal_maxs[g] == goal_mins[g]:
                norm = 0.0
            else:
                norm = (v - goal_mins[g]) / (goal_maxs[g] - goal_mins[g])
            total += norm
        scored.append((total, i))
    
    # Sort by score ascending (best first)
    scored.sort()
    
    # Return top 5% positions (at least 1)
    k = max(1, len(scored) // 20)
    return [pos for _, pos in scored[:k]]

import csv


def guess(header, train, test):
    goals = [c for c in header if c.endswith('-')]
    inputs = [c for c in header if not c.endswith('-')]
    first_goal = goals[0]

    n_inputs = len(inputs)
    m = n_inputs + 1  # rows of augmented normal-equation matrix

    X = []
    y = []
    for row in train:
        x_row = [1.0]
        for col in inputs:
            x_row.append(float(row[header.index(col)]))
        X.append(x_row)
        y.append(float(row[header.index(first_goal)]))

    # Build X^T X and X^T y
    XtX = [[0.0] * m for _ in range(m)]
    for i in range(n_inputs):
        for j in range(i, m):
            s = 0.0
            for k in range(len(train)):
                s += X[k][i] * X[k][j]
            XtX[i][j] = s
            if i != j:
                XtX[j][i] = s

    Xty = [0.0] * m
    for i in range(m):
        for k in range(len(train)):
            Xty[i] += X[k][i] * y[k]

    # Solve via Gaussian elimination with partial pivoting
    aug = []
    for i in range(m):
        aug.append(list(XtX[i]) + [Xty[i]])

    pivot_row = 0
    for col in range(m):
        if pivot_row >= m:
            break
        max_val = abs(aug[pivot_row][col])
        max_idx = pivot_row
        for row in range(pivot_row + 1, m):
            if abs(aug[row][col]) > max_val:
                max_val = abs(aug[row][col])
                max_idx = row
        aug[pivot_row], aug[max_idx] = aug[max_idx], aug[pivot_row]

        pivot = aug[pivot_row][col]
        for row in range(m):
            if row != pivot_row:
                factor = aug[row][col] / pivot
                for j in range(col, m + 1):
                    aug[row][j] -= factor * aug[pivot_row][j]

        pivot_row += 1

    beta = [aug[i][m] for i in range(m)]

    predictions = []
    for row in test:
        pred = 0.0
        for col in inputs:
            pred += float(row[header.index(col)]) * beta[header.index(col) + 1]
        pred += beta[0]
        predictions.append(pred)

    return predictions
