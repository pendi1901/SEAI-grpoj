import csv


def guess(header, train, test):
    goals = [c for c in header if c.endswith('-')]
    inputs = [c for c in header if not c.endswith('-')]
    first_goal = goals[0]

    n_inputs = len(inputs)
    m = n_inputs + 1  # rows of augmented normal-equation matrix

    X = []
    y = []
    for row in train:
        x_row = [1.0]
        for col in inputs:
            x_row.append(float(row[header.index(col)]))
        X.append(x_row)
        y.append(float(row[header.index(first_goal)]))

    # Build X^T X and X^T y
    XtX = [[0.0] * m for _ in range(m)]
    for i in range(n_inputs):
        for j in range(i, m):
            s = 0.0
            for k in range(len(train)):
                s += X[k][i] * X[k][j]
            XtX[i][j] = s
            if i != j:
                XtX[j][i] = s

    Xty = [0.0] * m
    for i in range(m):
        for k in range(len(train)):
            Xty[i] += X[k][i] * y[k]

    # Solve via Gaussian elimination with partial pivoting
    aug = []
    for i in range(m):
        aug.append(list(XtX[i]) + [Xty[i]])

    pivot_row = 0
    for col in range(m):
        if pivot_row >= m:
            break
        max_val = abs(aug[pivot_row][col])
        max_idx = pivot_row
        for row in range(pivot_row + 1, m):
            if abs(aug[row][col]) > max_val:
                max_val = abs(aug[row][col])
                max_idx = row
        aug[pivot_row], aug[max_idx] = aug[max_idx], aug[pivot_row]

        pivot = aug[pivot_row][col]
        for row in range(m):
            if row != pivot_row:
                factor = aug[row][col] / pivot
                for j in range(col, m + 1):
                    aug[row][j] -= factor * aug[pivot_row][j]

        pivot_row += 1

    beta = [aug[i][m] for i in range(m)]

    predictions = []
    for row in test:
        pred = 0.0
        for col in inputs:
            pred += float(row[header.index(col)]) * beta[header.index(col) + 1]
        pred += beta[0]
        predictions.append(pred)

    return predictions

import csv

def certify(xs, ys):
    """Returns True if xs and ys are the same apart from noise."""
    n1 = len(xs)
    n2 = len(ys)
    if n1 < 3 or n2 < 3:
        return False

    mean_x = sum(xs) / n1
    mean_y = sum(ys) / n2

    var_x = sum((x - mean_x) ** 2 for x in xs) / (n1 - 1) if n1 > 1 else 0.0
    var_y = sum((y - mean_y) ** 2 for y in ys) / (n2 - 1) if n2 > 1 else 0.0

    se = math.sqrt(var_x / n1 + var_y / n2)
    if se == 0:
        return True

    t_stat = abs(mean_x - mean_y) / se
    return t_stat < 2.0

# Module-level state that persists across calls to flow()
_state = {
    'goal_mins': {},
    'goal_maxs': {},
    'scores': [],
}


def flow(header, row, seen):
    """Approve (True) or reject (False) a proposal as it arrives."""
    goals = [c for c in header if c.endswith('-')]

    # Update running min/max from previously-seen rows
    n_seen_before = len(_state['scores'])
    for s_row in seen:
        for g in goals:
            v = float(s_row[header.index(g)])
            if g not in _state['goal_mins']:
                _state['goal_mins'][g] = v
                _state['goal_maxs'][g] = v
            else:
                _state['goal_mins'][g] = min(_state['goal_mins'][g], v)
                _state['goal_maxs'][g] = max(_state['goal_maxs'][g], v)

    # Score the current row (lower total is better for '-' goals)
    total = 0.0
    for g in goals:
        v = float(row[header.index(g)])
        mn = _state['goal_mins'].get(g, v)
        mx = _state['goal_maxs'].get(g, v)
        if mx == mn:
            norm = 0.0
        else:
            norm = (v - mn) / (mx - mn)
        total += norm

    score = total
    _state['scores'].append(score)

    # First row is always approved; thereafter use top-5% threshold
    if n_seen_before == 0:
        return True

    all_scores = sorted(_state['scores'])
    k = max(1, len(all_scores) // 20)
    threshold = all_scores[-k]

    return score <= threshold

            else:
                norm = (v - goal_mins[g]) / (goal_maxs[g] - goal_mins[g])
            total += norm
        scores.append((total, i))

    # Sort descending so worst rows come first
    scores.sort(reverse=True)

    k = max(1, n // 5)
    worst_indices = sorted([pos for _, pos in scores[:k]])
    best_indices = sorted([pos for _, pos in scores[-k:]])

    result = []
    for col in inputs:
        worst_vals = [float(r[header.index(col)]) for i, r in enumerate(rows) if i in worst_indices]
        best_vals = [float(r[header.index(col)]) for i, r in enumerate(rows) if i in best_indices]

        if not worst_vals or not best_vals:
            continue

        mean_worst = sum(worst_vals) / len(worst_vals)
        mean_best = sum(best_vals) / len(best_vals)

        var_b = sum((v - mean_best) ** 2 for v in best_vals) / max(len(best_vals) - 1, 1) if len(best_vals) > 1 else 0.0

            vals.append(v)
        goal_mins[g] = min(vals)
        goal_maxs[g] = max(vals)

    # Count how many existing rows are worse than an ideal hypothetical row (score=0)
    better_count = 0
    for r in rows:
        total_score = 0.0
        for g in goals:
            v = float(r[header.index(g)])
            if goal_maxs[g] == goal_mins[g]:
                norm = 0.0
            else:
                norm = (v - goal_mins[g]) / (goal_maxs[g] - goal_mins[g])
            total_score += norm
        if total_score > 0:
            better_count += 1

    p = better_count / n_rows

    if p >= 0.95:

    elif p >= 0.6:
        reason = "This setup performs well above average compared to existing setups."
    else:
        reason = "This setup would rank below most existing configurations."

    return (p, reason)

def fix(header, row, levers):
    """Pick the best allowed value for each lever to rescue the bad row."""
    goals = [c for c in header if c.endswith('-')]
    plus_goals = [c for c in header if c.endswith('+')]

    # Build a baseline score from the original row's goal values
    def score(vals):
        s = 0.0
        for g in goals:
            v = float(vals[header.index(g)])
            s += v
        for g in plus_goals:
            v = float(vals[header.index(g)])
            s -= v
        return s

    # Try every combination of lever values (small enough to be tractable)
    best_score = score(row)
    best_choice = {col: row[header.index(col)] for col in header}

    def try_combination(vals):
        s = 0.0
        for g in goals:
            v = float(vals[header.index(g)])
            s += v
        for g in plus_goals:
            v = float(vals[header.index(g)])
            s -= v
        return s

    # Collect all lever columns and their candidate values
    lever_cols = []
    for col, vals in levers.items():

            low, high = vals

            step = (high - low) / n_steps if n_steps > 0 else 0.0
            lever_cols.append((col, [low + i * step for i in range(n_steps)]))

    # If no levers were given, just return the original row unchanged
    if not lever_cols:
        return best_choice

    # Cartesian product of all lever values – keep it small by limiting steps
    max_combos = 200
    total = 1
    for _, vals in lever_cols:
        total *= len(vals)
    if total > max_combos:
        raise ValueError("Too many combinations; pick fewer or coarser levers.")

    combos = []
    for i in range(total):
        idx = [0] * len(lever_cols)
        # simple counter to enumerate
        vals = []
        for j, (_, cvals) in enumerate(lever_cols):
            vals.append(cvals[i % len(cvals)])

        combos.append(vals)

        if s < best_score:
            best_score = s
            for j, (col, _) in enumerate(lever_cols):

def spot(header, rows):
    """Identify rows with mistyped or made-up input values."""

    goals = [c for c in header if c.endswith('-')]
    plus_goals = [c for c in header if c.endswith('+')]
    inputs = [c for c in header if not c.endswith('-') and not c.endswith('+')]

    n = len(rows)
    suspicious = []

    # Build per-column statistics using median + MAD (robust to outliers)
    col_stats = {}
    for col in inputs:
        vals = []
        for r in rows:
            v = r[header.index(col)]
            try:
                f = float(v)
                if not (v == '?' or v == 'NA' or v == ''):
                    vals.append(f)
            except (ValueError, TypeError):
                pass

        if len(vals) < 3:
            continue

        sorted_vals = sorted(vals)
        med_idx = len(sorted_vals) // 2
        median = sorted_vals[med_idx]

        abs_devs = [abs(v - median) for v in sorted_vals]
        mad_sorted = sorted(abs_devs)
        mad_idx = len(mad_sorted) // 2
        mad = mad_sorted[mad_idx]

        col_stats[col] = (median, mad)

    # Flag rows where any input value is far from typical range
    for i, r in enumerate(rows):
        is_suspicious = False

        for col in inputs:
            v = r[header.index(col)]
            if v == '?':
                continue

            try:
                f = float(v)
            except (ValueError, TypeError):
                is_suspicious = True
                break

            med, mad = col_stats[col]

            if mad == 0:
                # All values identical; any different value is suspicious
                if f != med:
                    is_suspicious = True
                    break
            else:
                threshold = 3.0 * mad
                if abs(f - med) > threshold:
                    is_suspicious = True
                    break

        if is_suspicious:
            suspicious.append(i)

    return sorted(suspicious, reverse=True)
