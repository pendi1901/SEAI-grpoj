def race(header, rows, label):
    """Compare three strategies on up to 50 labelled rows; return winner info."""

    goals = [c for c in header if c.endswith('-')]
    plus_goals = [c for c in header if c.endswith('+')]

    def score(vals):
        s = 0.0
        for g in goals:
            v = float(vals[header.index(g)])
            s += v
        for g in plus_goals:
            v = float(vals[header.index(g)])
            s -= v
        return s

    # --- Strategy A: random (pick rows uniformly at random) ---
    def strategy_random(label):
        chosen = []
        used = set()
        while len(chosen) < 50 and len(used) < len(rows):
            i = label(len(chosen)) % len(rows)
            if i not in used:
                used.add(i)
                chosen.append(i)
        return chosen

    # --- Strategy B: greedy hill-climb (pick best unseen row by score) ---
    def strategy_greedy(label):
        chosen = []
        used = set()
        while len(chosen) < 50 and len(used) < len(rows):
            best_i = -1
            best_s = float('inf')
            for i in range(len(rows)):
                if i not in used:
                    s = score(label(i))
                    if s < best_s:
                        best_s = s
                        best_i = i
            if best_i == -1:
                break
            used.add(best_i)
            chosen.append(best_i)
        return chosen
