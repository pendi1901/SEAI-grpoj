def choose(header, rows, label):
    goals = [c for c in header if c.endswith('-')]
    plus_goals = [c for c in header if c.endswith('+')]

    best_pos = -1
    best_score = float('inf')

    n_to_try = min(50, len(rows))
    for i in range(n_to_try):
        vals = label(i)
        score = 0.0
        for g in goals:
            v = float(vals[header.index(g)])
            score += v
        for g in plus_goals:
            v = float(vals[header.index(g)])
            score -= v

        if score < best_score:
            best_score = score
            best_pos = i

    return best_pos


if __name__ == "__main__":
    import csv

    with open('pom3a.train.csv', newline='') as f:
        reader = csv.reader(f)
        header = next(reader)
        rows = []
        for row in reader:
            if len(row) == len(header):
                rows.append(row)

    goals = [c for c in header if c.endswith('-')]
    plus_goals = [c for c in header if c.endswith('+')]

    def label(i):
        vals = []
        for g in goals + plus_goals:
            vals.append(float(rows[i][header.index(g)]))
        return vals

    best_pos = choose(header, rows, label)
    print(f"Best setup position (0-based): {best_pos}")
