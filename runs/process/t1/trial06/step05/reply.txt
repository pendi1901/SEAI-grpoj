```python
import csv

def blame(header, rows):
    """
    Identifies which input columns correlate most strongly with the 'Idle-' goal.
    The 'worst' setups are those with high 'Idle-' values.
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith(('-', '+'))]
    input_indices = [i for i, name in enumerate(header) if i not in goal_indices]
    idle_idx = header.index('Idle-')

    # Sort rows by Idle- to identify the worst setups
    sorted_rows = sorted(rows, key=lambda r: r[idle_idx], reverse=True)
    
    # Take the worst 20%
    worst_rows = sorted_rows[:max(1, len(rows) // 5)]
    
    # Calculate correlation: average value of input in worst rows vs global average
    # We look for inputs that are significantly higher or lower in the worst rows
    correlations = []
    for i in input_indices:
        vals = [r[i] for r in rows]
        avg_all = sum(vals) / len(vals)
        avg_worst = sum(r[i] for r in worst_rows) / len(worst_rows)
        
        # The "blame" is the absolute difference from the mean
        # Inputs that deviate most in bad rows are the likely culprits
        diff = abs(avg_worst - avg_all)
        correlations.append((diff, header[i]))
    
    # Sort by impact and return the top 3 culprits
    correlations.sort(key=lambda x: x[0], reverse=True)
    return [name for diff, name in correlations[:3]]

if __name__ == "__main__":
    filename = "pom3a.n1000.train.csv"
    try:
        with open(filename, mode='r', encoding='utf-8') as f:
            reader = csv.reader(f)
            header = next(reader)
            rows = []
            for row in reader:
                rows.append([float(x) for x in row])
        
        culprits = blame(header, rows)
        
        print("Analysis: Why do our worst setups leave so many people waiting (high Idle-)?")
        print("The primary factors contributing to high idle time are:")
        for i, name in enumerate(culprits, 1):
            print(f"{i}. {name}")
            
    except FileNotFoundError:
        print(f"Error: {filename} not found.")
        exit(1)
```
