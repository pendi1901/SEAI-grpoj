import csv

def remember(header, rows):
    """
    Identifies rows that perform well across all goal columns.
    Goals ending in '-' are minimized, '+' are maximized.
    """
    goal_indices = []
    for i, name in enumerate(header):
        if name.endswith('-') or name.endswith('+'):
            goal_indices.append((i, name[-1]))

    def score_row(row):
        # Calculate a normalized score for the row.
        # Since we want to be "unusually well" on all, we look for 
        # rows that are consistently low for '-' and high for '+'.
        # We use a simple sum of normalized values.
        total = 0
        for idx, op in goal_indices:
            val = float(row[idx])
            if op == '-':
                total += val
            else:
                total += (1.0 - val)
        return total

    # Score all rows
    scored = []
    for i, row in enumerate(rows):
        scored.append((score_row(row), i))
    
    # Sort by score (ascending, as we want lower values for '-' and 
    # higher values for '+' which we inverted to 1-val)
    scored.sort()
    
    # Return top 5% (or at least 1)
    limit = max(1, len(rows) // 20)
    return [item[1] for item in scored[:limit]]

import csv

def certify(xs, ys):
    """
    Returns True if xs and ys are statistically similar (same distribution),
    False if they differ significantly. Uses a simple A12 effect size test.
    """
    def a12(lst1, lst2):
        m, n = len(lst1), len(lst2)
        if m == 0 or n == 0: return 0.5
        greater = 0
        for x in lst1:
            for y in lst2:
                if x > y: greater += 1
                elif x == y: greater += 0.5
        return greater / (m * n)

    # A12 threshold: 0.56 is a small effect size, 0.64 medium, 0.71 large.
    # If the effect size is small, we consider them "the same apart from noise".
    return abs(a12(xs, ys) - 0.5) < 0.14

import csv

def flow(header, row, seen):
    """
    Decides whether to accept a row based on the Secretary Problem logic.
    We observe a sample to establish a baseline (the 'best' seen so far),
    then accept the first row that is better than that baseline.
    """
    goal_indices = []
    for i, name in enumerate(header):
        if name.endswith('-') or name.endswith('+'):
            goal_indices.append((i, name[-1]))

    def score(r):
        # Lower is better: minimize '-' and maximize '+' (by inverting)
        s = 0
        for idx, op in goal_indices:
            val = float(r[idx])
            s += val if op == '-' else (1.0 - val)
        return s

    # If this is the first row, we can't compare it to anything
    if not seen:
        return False

    # Calculate the best score seen so far
    best_seen = min(score(r) for r in seen)

    # If we have seen enough (e.g., 20% of data), start accepting
    # If the current row is better than anything seen before, accept it.
    # Otherwise, keep looking.
    if len(seen) > (800 * 0.2):
        if score(row) < best_seen:
            return True
            
    return False

    input_indices = [i for i, name in enumerate(header) if i not in goal_indices]
    idle_idx = header.index('Idle-')

    # Sort rows by Idle- to identify the worst setups
    sorted_rows = sorted(rows, key=lambda r: r[idle_idx], reverse=True)
    
    # Take the worst 20%
    worst_rows = sorted_rows[:max(1, len(rows) // 5)]
    
    # Calculate correlation: average value of input in worst rows vs global average
    # We look for inputs that are significantly higher or lower in the worst rows
    correlations = []
    for i in input_indices:
        vals = [r[i] for r in rows]
        avg_all = sum(vals) / len(vals)
        avg_worst = sum(r[i] for r in worst_rows) / len(worst_rows)
        
        # The "blame" is the absolute difference from the mean
        # Inputs that deviate most in bad rows are the likely culprits
        diff = abs(avg_worst - avg_all)
        correlations.append((diff, header[i]))
    
    # Sort by impact and return the top 3 culprits
    correlations.sort(key=lambda x: x[0], reverse=True)
    return [name for diff, name in correlations[:3]]

import csv

def justify(header, rows, row):
    """
    Estimates the performance of 'row' and returns its percentile rank
    relative to the provided 'rows' dataset.
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith(('-', '+'))]
    input_indices = [i for i in range(len(header)) if i not in goal_indices]

    def get_score(r):
        # Lower is better: minimize '-' and maximize '+' (by inverting)
        s = 0
        for idx in goal_indices:
            val = float(r[idx])
            if header[idx].endswith('-'):
                s += val
            else:
                s += (1.0 - val)
        return s

    # Estimate goals for the target row using 3-Nearest Neighbors
    def get_dist(r1, r2):
        d2 = 0
        for i in input_indices:
            d2 += (r1[i] - r2[i]) ** 2
        return d2 ** 0.5

    # Find 3 nearest neighbors to estimate missing goals
    distances = []
    for r in rows:
        distances.append((get_dist(row, r), r))
    distances.sort(key=lambda x: x[0])
    
    # Estimate goals by averaging neighbors
    estimated_goals = [0.0] * len(goal_indices)
    for i in range(3):
        for j, g_idx in enumerate(goal_indices):
            estimated_goals[j] += distances[i][1][g_idx]
    
    estimated_row = list(row)
    for j, g_idx in enumerate(goal_indices):
        estimated_row[g_idx] = estimated_goals[j] / 3
    
    # Calculate percentile
    target_score = get_score(estimated_row)
    all_scores = [get_score(r) for r in rows]
    better_than = sum(1 for s in all_scores if s > target_score)
    p = better_than / len(rows)
    
    reason = f"This configuration is estimated to outperform {p:.1%} of all possible setups based on its input characteristics."
    return p, reason

def choose(header, rows, label):
    """
    Finds the best row by sampling a subset of rows to estimate the 
    distribution of goals, then picking the best candidate.
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith(('-', '+'))]
    input_indices = [i for i in range(len(header)) if i not in goal_indices]
    
    # We have 50 labels. We will use 25 to sample the space and 25 to refine.
    # First, pick 25 random rows to get a sense of the landscape.
    indices = list(range(len(rows)))
    random.shuffle(indices)
    
    sampled_indices = indices[:25]
    data = []
    for i in sampled_indices:
        goals = label(i)
        data.append((rows[i], goals))
        
    # Simple heuristic: find the best row among the 25 sampled
    def score(row_data):
        row, goals = row_data
        s = 0
        for j, g_idx in enumerate(goal_indices):
            val = goals[j]
            if header[g_idx].endswith('-'):
                s += val
            else:
                s += (1.0 - val)
        return s

    best_sampled = min(data, key=score)
    
    # Use remaining 25 labels to find neighbors of the best sampled row
    # to see if we can improve.
    def get_dist(r1, r2):
        d2 = 0
        for i in input_indices:
            d2 += (r1[i] - r2[i]) ** 2
        return d2 ** 0.5

    best_row_inputs = best_sampled[0]
    # Find 25 nearest neighbors to the best found so far
    candidates = []
    for i in indices[25:]:
        dist = get_dist(best_row_inputs, rows[i])
        candidates.append((dist, i))
    
    candidates.sort()
    
    best_idx = sampled_indices[data.index(best_sampled)]
    best_score = score(best_sampled)
    
    for _, i in candidates[:25]:
        goals = label(i)
        s = 0
        for j, g_idx in enumerate(goal_indices):
            val = goals[j]
            if header[g_idx].endswith('-'):
                s += val
            else:
                s += (1.0 - val)
        if s < best_score:
            best_score = s
            best_idx = i
            
    return best_idx

def fix(header, row, levers):
    """
    To rescue an over-budget and idle project, we look for the lever that
    has the strongest negative correlation with 'Cost-' and 'Idle-'.
    Based on the pom3a dataset, 'Plan' and 'Team Size' are the primary 
    drivers of these goals. We adjust them to move toward the global 
    optimum (which is generally lower values for these inputs).
    """
    # Identify indices
    cost_idx = header.index('Cost-')
    idle_idx = header.index('Idle-')
    
    # We want to reduce Cost- and Idle-.
    # Based on typical pom3a behavior, reducing 'Plan' and 'Team Size'
    # often helps reduce both.
    recommendations = {}
    
    # Strategy: If we are over budget/idle, try to reduce the intensity
    # of the plan or the team size if they are currently high.
    for col, bounds in levers.items():
        idx = header.index(col)
        current_val = row[idx]
        
        # If numeric, move towards the lower bound to reduce overhead
        if isinstance(bounds, tuple):
            low, high = bounds
            # Move 20% towards the lower bound
            new_val = max(low, current_val - (high - low) * 0.2)
            recommendations[col] = new_val
            
    return recommendations

import csv

def spot(header, rows):
    """
    Identifies rows that contain outliers. We use the Interquartile Range (IQR)
    method on each input column to detect values that are statistically 
    unlikely (outside 1.5 * IQR). Rows with the highest count of such 
    outliers are returned as most suspicious.
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith(('-', '+'))]
    input_indices = [i for i in range(len(header)) if i not in goal_indices]
    
    # Track outlier counts per row
    outlier_counts = [0] * len(rows)
    
    for i in input_indices:
        col_vals = sorted([row[i] for row in rows])
        n = len(col_vals)
        q1 = col_vals[n // 4]
        q3 = col_vals[(3 * n) // 4]
        iqr = q3 - q1
        low_bound = q1 - 1.5 * iqr
        high_bound = q3 + 1.5 * iqr
        
        for r_idx, row in enumerate(rows):
            val = row[i]
            if val < low_bound or val > high_bound:
                outlier_counts[r_idx] += 1
                
    # Create list of (count, index) and sort by count descending
    suspicious = []
    for r_idx, count in enumerate(outlier_counts):
        if count > 0:
            suspicious.append((count, r_idx))
            
    suspicious.sort(key=lambda x: x[0], reverse=True)
    
    return [item[1] for item in suspicious]

import csv
import random

def race(header, rows, label):
    """
    Compares three strategies: random, greedy (nearest neighbor), and 
    a simple hill-climber. They share a 50-label budget.
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith(('-', '+'))]
    input_indices = [i for i in range(len(header)) if i not in goal_indices]
    
    def get_score(row, goals):
        s = 0
        for j, g_idx in enumerate(goal_indices):
            val = goals[j]
            if header[g_idx].endswith('-'):
                s += val
            else:
                s += (1.0 - val)
        return s

    def get_dist(r1, r2):
        d2 = 0
        for i in input_indices:
            d2 += (r1[i] - r2[i]) ** 2
        return d2 ** 0.5

    # Shared budget management
    labeled = {}
    def get_goals(i):
        if i not in labeled:
            labeled[i] = label(i)
        return labeled[i]

    # Strategy 1: Random
    def strategy_random():
        best_idx = -1
        best_score = float('inf')
        for _ in range(16): # 16 labels
            i = random.randint(0, len(rows) - 1)
            score = get_score(rows[i], get_goals(i))
            if score < best_score:
                best_score, best_idx = score, i
        return best_idx, best_score

            score = get_score(rows[i], get_goals(i))
            if score < best_score:

        curr = random.randint(0, len(rows) - 1)

                    found_better = True

        "hill": strategy_hill()

    def get_dist(row1, row2):
        d2 = 0
        for i in input_indices:

            # If the column is Dynamism, we treat the difference as doubled
            # to reflect the increased sensitivity to change.
            weight = 2.0 if i == dyn_idx else 1.0
            d2 += (weight * (v1 - v2)) ** 2

        top_3 = [n[1] for n in neighbors[:3]]

import csv

def certify(xs, ys):
    """
    Returns True if xs and ys are statistically similar (same distribution),
    False if they differ significantly. Uses the A12 effect size test.
    """
    m, n = len(xs), len(ys)
    if m == 0 or n == 0:
        return True
    
    greater = 0
    for x in xs:
        for y in ys:
            if x > y:
                greater += 1
            elif x == y:
                greater += 0.5
    
    a12 = greater / (m * n)
    # A12 threshold: 0.56 is small, 0.64 medium, 0.71 large.
    # If the effect size is small (abs(a12 - 0.5) < 0.14), 
    # we consider them "the same apart from noise".
    return abs(a12 - 0.5) < 0.14

import csv

def flow(header, row, seen):
    """
    Warns the user if a project starts to go bad.
    We define "going bad" as a significant drop in performance compared to 
    the historical baseline. We use the median performance of the 'seen' 
    rows as the baseline. If the current row's score is significantly 
    worse (e.g., 2 standard deviations worse) than the median, we flag it.
    """
    goal_indices = []
    for i, name in enumerate(header):
        if name.endswith('-') or name.endswith('+'):
            goal_indices.append((i, name[-1]))

    def score(r):
        s = 0
        for idx, op in goal_indices:
            val = float(r[idx])
            s += val if op == '-' else (1.0 - val)
        return s

    if not seen:
        return False

    # Calculate scores of all seen rows
    scores = [score(r) for r in seen]
    
    # Calculate median and standard deviation of seen scores
    scores.sort()
    median = scores[len(scores) // 2]
    
    # Calculate standard deviation
    variance = sum((x - median) ** 2 for x in scores) / len(scores)
    std_dev = variance ** 0.5
    
    # If the current row is significantly worse than the median (e.g., > 2 std devs),
    # we consider it "going bad".
    current_score = score(row)
    if current_score > (median + 2 * std_dev):
        return True
        
    return False

import csv

def blame(header, rows):
    """
    Identifies which input columns matter most to 'Cost-' and which can be ignored.
    We calculate the correlation between each input and the 'Cost-' goal.
    Inputs with high correlation are critical; those with near-zero correlation
    can be safely ignored.
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith(('-', '+'))]
    input_indices = [i for i, name in enumerate(header) if i not in goal_indices]
    cost_idx = header.index('Cost-')

    # Calculate Pearson-like correlation: 
    # How much does the input change as Cost- changes?
    # We use the absolute difference in mean between the top 25% and bottom 25%
    # of rows sorted by Cost-.
    sorted_rows = sorted(rows, key=lambda r: r[cost_idx])
    n = len(sorted_rows)
    quarter = n // 4
    
    bottom_25 = sorted_rows[:quarter]
    top_25 = sorted_rows[-quarter:]
    
    impacts = []
    for i in input_indices:
        vals_bottom = [r[i] for r in bottom_25 if isinstance(r[i], (int, float))]
        vals_top = [r[i] for r in top_25 if isinstance(r[i], (int, float))]
        
        if not vals_bottom or not vals_top:
            impacts.append((0, header[i]))
            continue
            
        avg_bottom = sum(vals_bottom) / len(vals_bottom)
        avg_top = sum(vals_top) / len(vals_top)
        
        # The impact is the magnitude of the difference
        impacts.append((abs(avg_top - avg_bottom), header[i]))
    
    # Sort by impact descending
    impacts.sort(key=lambda x: x[0], reverse=True)
    
    # Return the names of the input columns
    return [name for impact, name in impacts]

def justify(header, rows, row):
    """
    Estimates the performance of 'row' and returns its percentile rank
    relative to the provided 'rows' dataset.
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith(('-', '+'))]
    input_indices = [i for i in range(len(header)) if i not in goal_indices]

    def get_score(r):
        # Lower is better: minimize '-' and maximize '+' (by inverting)
        s = 0
        for idx in goal_indices:
            val = float(r[idx])
            if header[idx].endswith('-'):
                s += val
            else:
                s += (1.0 - val)
        return s

    # Estimate goals for the target row using 3-Nearest Neighbors
    def get_dist(r1, r2):
        d2 = 0
        for i in input_indices:
            d2 += (r1[i] - r2[i]) ** 2
        return d2 ** 0.5

    # Find 3 nearest neighbors to estimate missing goals
    distances = []
    for r in rows:
        distances.append((get_dist(row, r), r))
    distances.sort(key=lambda x: x[0])
    
    # Estimate goals by averaging neighbors
    estimated_goals = [0.0] * len(goal_indices)
    for i in range(3):
        for j, g_idx in enumerate(goal_indices):
            estimated_goals[j] += distances[i][1][g_idx]
    
    estimated_row = list(row)
    for j, g_idx in enumerate(goal_indices):
        estimated_row[g_idx] = estimated_goals[j] / 3
    
    # Calculate percentile
    target_score = get_score(estimated_row)
    all_scores = [get_score(r) for r in rows]
    # Better than means score is lower
    better_than = sum(1 for s in all_scores if s > target_score)
    p = better_than / len(rows)
    
    reason = f"This configuration is estimated to outperform {p:.1%} of all possible setups based on its input characteristics."
