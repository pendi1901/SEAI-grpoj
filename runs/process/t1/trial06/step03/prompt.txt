import csv

def remember(header, rows):
    """
    Identifies rows that perform well across all goal columns.
    Goals ending in '-' are minimized, '+' are maximized.
    """
    goal_indices = []
    for i, name in enumerate(header):
        if name.endswith('-') or name.endswith('+'):
            goal_indices.append((i, name[-1]))

    def score_row(row):
        # Calculate a normalized score for the row.
        # Since we want to be "unusually well" on all, we look for 
        # rows that are consistently low for '-' and high for '+'.
        # We use a simple sum of normalized values.
        total = 0
        for idx, op in goal_indices:
            val = float(row[idx])
            if op == '-':
                total += val
            else:
                total += (1.0 - val)
        return total

    # Score all rows
    scored = []
    for i, row in enumerate(rows):
        scored.append((score_row(row), i))
    
    # Sort by score (ascending, as we want lower values for '-' and 
    # higher values for '+' which we inverted to 1-val)
    scored.sort()
    
    # Return top 5% (or at least 1)
    limit = max(1, len(rows) // 20)
    return [item[1] for item in scored[:limit]]

import csv

def guess(header, train, test):
    """
    Estimates the first goal column for each test row.
    Uses a simple k-Nearest Neighbors approach based on Euclidean distance
    of input columns (non-goal columns).
    """
    # Identify goal indices and input indices
    goal_indices = [i for i, name in enumerate(header) if name.endswith(('-', '+'))]
    first_goal_idx = goal_indices[0]
    input_indices = [i for i in range(len(header)) if i not in goal_indices]

    def get_dist(row1, row2):
        # Calculate Euclidean distance based on input columns
        d2 = 0
        for i in input_indices:
            v1, v2 = row1[i], row2[i]
            # Handle symbolic data if necessary, though pom3a is numeric
            if isinstance(v1, (int, float)) and isinstance(v2, (int, float)):
                d2 += (v1 - v2) ** 2
            else:
                d2 += 0 if v1 == v2 else 1
        return d2 ** 0.5

    predictions = []
    for t_row in test:
        # Find the nearest neighbor in training data
        best_dist = float('inf')
        best_val = 0
        for train_row in train:
            d = get_dist(t_row, train_row)
            if d < best_dist:
                best_dist = d
                best_val = train_row[first_goal_idx]
        predictions.append(float(best_val))
    
    return predictions
