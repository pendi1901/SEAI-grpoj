import csv

def spot(header, rows):
    """
    Identifies rows that are 'unlike anything we have run'.
    We calculate the distance of each row to its nearest neighbor in the 
    dataset. Rows that are significantly far from their nearest neighbor 
    (outliers in the input space) are considered 'never seen before'.
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith(('-', '+'))]
    input_indices = [i for i in range(len(header)) if i not in goal_indices]
    
    # Normalize input columns to ensure distance is not dominated by one scale
    # Since pom3a inputs are 0-1, we can use them directly.
    
    def get_dist(r1, r2):
        d2 = 0
        for i in input_indices:
            v1, v2 = r1[i], r2[i]
            d2 += (v1 - v2) ** 2
        return d2 ** 0.5

    # For each row, find the distance to its nearest neighbor
    # A high distance means the row is in a sparse region of the input space.
    distances = []
    for i, row1 in enumerate(rows):
        min_d = float('inf')
        for j, row2 in enumerate(rows):
            if i == j:
                continue
            d = get_dist(row1, row2)
            if d < min_d:
                min_d = d
        distances.append((min_d, i))
    
    # Sort by distance descending: furthest from neighbors is most suspicious
    distances.sort(key=lambda x: x[0], reverse=True)
    
    return [item[1] for item in distances]

if __name__ == "__main__":
    filename = "pom3a.n1000.train.csv"
    try:
        with open(filename, mode='r', encoding='utf-8') as f:
            reader = csv.reader(f)
            header = next(reader)
            rows = []
            for row in reader:
                processed_row = []
                for val in row:
                    try:
                        processed_row.append(float(val))
                    except ValueError:
                        processed_row.append(val)
                rows.append(processed_row)
        
        suspicious_indices = spot(header, rows)
        
        print("Analysis: Identifying setups unlike anything seen in historical data.")
        if not suspicious_indices:
            print("No unusual configurations found.")
        else:
            print(f"Found {len(suspicious_indices)} configurations that are outliers.")
            print("The top 5 most unique/unseen setups (by row index):")
            for idx in suspicious_indices[:5]:
                print(f" - Row {idx}")
            print("\nThese rows represent configurations that deviate significantly from historical norms.")
                
    except FileNotFoundError:
        print(f"Error: {filename} not found.")
        exit(1)
