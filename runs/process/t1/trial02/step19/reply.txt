import csv

def spot(header, rows):
    """
    Identifies rows that contain input values that are 'unlike anything seen'.
    We use the Interquartile Range (IQR) method to detect values that 
    are significantly outside the expected range for each input column.
    """
    input_indices = [i for i, name in enumerate(header) 
                     if not (name.endswith('-') or name.endswith('+'))]
    
    # Collect all values for each input column to calculate thresholds
    col_data = {i: [] for i in input_indices}
    for row in rows:
        for i in input_indices:
            try:
                val = float(row[i])
                col_data[i].append(val)
            except (ValueError, TypeError):
                continue
    
    # Calculate IQR bounds for each column
    # Using a standard 1.5 * IQR multiplier for detecting outliers
    bounds = {}
    for i in input_indices:
        data = sorted(col_data[i])
        if not data:
            continue
        q1 = data[len(data) // 4]
        q3 = data[3 * len(data) // 4]
        iqr = q3 - q1
        bounds[i] = (q1 - 1.5 * iqr, q3 + 1.5 * iqr)
    
    # Score rows based on how many outliers they contain and the magnitude of deviation
    suspects = []
    for idx, row in enumerate(rows):
        score = 0
        for i in input_indices:
            try:
                val = float(row[i])
                if i in bounds:
                    low, high = bounds[i]
                    if val < low:
                        score += (low - val)
                    elif val > high:
                        score += (val - high)
            except (ValueError, TypeError):
                continue
        if score > 0:
            suspects.append((score, idx))
            
    # Sort by score descending (most suspicious first)
    suspects.sort(key=lambda x: x[0], reverse=True)
    return [idx for score, idx in suspects]

if __name__ == "__main__":
    filename = "pom3a.n1000.train.csv"
    try:
        with open(filename, mode='r', encoding='utf-8') as f:
            reader = csv.reader(f)
            header = next(reader)
            rows = [row for row in reader]
            
            suspect_indices = spot(header, rows)
            
            print("Customer Inquiry: Tell me when I am asking you about something you have never seen.")
            if not suspect_indices:
                print("Analysis: All provided setups are within the expected range of historical data.")
            else:
                print(f"Analysis: Found {len(suspect_indices)} setups that contain unusual input values.")
                print("The most suspicious row indices are:", suspect_indices[:5])
                print("These rows contain parameters that deviate significantly from our historical norms.")
                
    except FileNotFoundError:
        print(f"Error: {filename} not found.")
        exit(1)
