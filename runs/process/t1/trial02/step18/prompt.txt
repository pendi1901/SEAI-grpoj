import csv

def remember(header, rows):
    """
    Identifies rows that perform well across all goal columns.
    Goals ending in '-' are minimized, '+' are maximized.
    """
    goal_indices = []
    for i, name in enumerate(header):
        if name.endswith('-') or name.endswith('+'):
            goal_indices.append((i, name[-1]))

    def score_row(row):
        # Calculate a normalized score for the row.
        # Since we want to be "unusually well" on all, we look for 
        # rows that are consistently low for '-' and high for '+'.
        # We use a simple heuristic: sum of normalized goal values.
        total = 0
        for idx, op in goal_indices:
            val = float(row[idx])
            total += val if op == '-' else (1.0 - val)
        return total

    # Score all rows
    scored = []
    for i, row in enumerate(rows):
        scored.append((score_row(row), i))
    
    # Sort by score (ascending, as we want lower values for '-' and higher for '+')
    scored.sort()
    
    # Return top 5% (or at least 1)
    limit = max(1, len(rows) // 20)
    return [item[1] for item in scored[:limit]]

def certify(xs, ys):
    """
    Returns True if xs and ys are statistically similar (same distribution),
    False if they differ significantly. Uses a simple A12 effect size test.
    """
    def a12(lst1, lst2):
        m, n = len(lst1), len(lst2)
        greater = 0
        for x in lst1:
            for y in lst2:
                if x > y: greater += 1
                elif x == y: greater += 0.5
        return greater / (m * n)

    # If the probability of one being larger than the other is close to 0.5,
    # they are likely the same. Threshold 0.56 is a standard small effect size.
    score = a12(xs, ys)
    return 0.44 <= score <= 0.56

import csv

def flow(header, row, seen):
    """
    Decides whether to accept a row based on whether it is better than
    the best row seen so far. This is a simple online optimization strategy.
    """
    goal_indices = []
    for i, name in enumerate(header):
        if name.endswith('-') or name.endswith('+'):
            goal_indices.append((i, name[-1]))

    def score_row(r):
        total = 0
        for idx, op in goal_indices:
            val = float(r[idx])
            total += val if op == '-' else (1.0 - val)
        return total

    # If this is the first row, accept it
    if not seen:
        return True

    # Calculate score for current row
    current_score = score_row(row)

    # Find the best score seen so far
    best_seen_score = min(score_row(r) for r in seen)

    # If current row is better than anything seen before, accept it
    # Note: lower score is better in our heuristic
    return current_score < best_seen_score

import csv

def blame(header, rows):
    """
    Identifies input columns that correlate most strongly with the 'Idle-' goal.
    We define 'worst' as rows where 'Idle-' is in the top 25% (highest values).
    We then find input columns that have the highest correlation with these
    high 'Idle-' values.
    """
    idle_idx = header.index('Idle-')
    input_indices = [i for i, name in enumerate(header) 
                     if not (name.endswith('-') or name.endswith('+'))]
    
    # Sort rows by Idle- descending to find the "worst"
    sorted_rows = sorted(rows, key=lambda r: float(r[idle_idx]), reverse=True)
    worst_rows = sorted_rows[:len(rows) // 4]
    
    # Calculate average of each input in the worst rows
    # and compare to the global average of that input.
    # A high difference indicates that input contributes to the "worst" state.
    scores = []
    for i in input_indices:
        worst_avg = sum(float(r[i]) for r in worst_rows) / len(worst_rows)
        global_avg = sum(float(r[i]) for r in rows) / len(rows)
        
        # We look for inputs that are significantly higher in the worst rows
        # (Assuming higher input values correlate with higher Idle- values)
        diff = worst_avg - global_avg
        scores.append((diff, header[i]))
    
    # Sort by difference descending
    scores.sort(key=lambda x: x[0], reverse=True)
    
    # Return the top 2 inputs that explain the idleness
    return [name for diff, name in scores[:2]]

def justify(header, rows, row):
    """
    Estimates the performance of a row and returns its percentile rank
    against the training data, along with a justification.
    """
    input_indices = [i for i, name in enumerate(header) 
                     if not (name.endswith('-') or name.endswith('+'))]
    goal_indices = [(i, name[-1]) for i, name in enumerate(header) 
                    if name.endswith('-') or name.endswith('+')]

    def get_score(r):
        # Calculate a normalized score: lower is better.
        # For '-', we want low values. For '+', we want high values (1-val).
        total = 0.0
        for idx, op in goal_indices:
            val = float(r[idx])
            total += val if op == '-' else (1.0 - val)
        return total

    # Estimate missing goals using 1-Nearest Neighbor
    best_dist = float('inf')
    best_row = None
    for train_row in rows:
        dist = sum((float(row[i]) - float(train_row[i]))**2 for i in input_indices)
        if dist < best_dist:
            best_dist = dist
            best_row = train_row
    
    # Create a copy of the row with estimated goals
    estimated_row = list(row)
    for idx, _ in goal_indices:
        estimated_row[idx] = best_row[idx]
    
    # Compare against all rows
    my_score = get_score(estimated_row)
    better_count = sum(1 for r in rows if get_score(r) < my_score)
    p = better_count / len(rows)
    
    reason = f"This setup is better than {p:.0%} of all known configurations based on its estimated performance metrics."
    return p, reason

            total += val if op == '-' else (1.0 - val)
        return total

    # We have 50 trials. We pick 50 random indices.
    num_rows = len(rows)
    trials = min(50, num_rows)
    indices = random.sample(range(num_rows), trials)

    best_score = float('inf')

            best_idx = i
            
    return best_idx

import csv

def spot(header, rows):
    """
    Identifies rows that contain outliers in input columns.
    We use the Interquartile Range (IQR) method to detect values that 
    are significantly outside the expected range for each input column.
    """
    input_indices = [i for i, name in enumerate(header) 
                     if not (name.endswith('-') or name.endswith('+'))]
    
    # Collect all values for each input column to calculate thresholds
    col_data = {i: [] for i in input_indices}
    for row in rows:
        for i in input_indices:
            try:
                val = float(row[i])
                col_data[i].append(val)
            except (ValueError, TypeError):
                continue
    
    # Calculate IQR bounds for each column
    bounds = {}
    for i in input_indices:
        data = sorted(col_data[i])
        q1 = data[len(data) // 4]
        q3 = data[3 * len(data) // 4]
        iqr = q3 - q1
        # Use a multiplier of 3.0 for "extreme" outliers
        bounds[i] = (q1 - 3.0 * iqr, q3 + 3.0 * iqr)
    
    # Score rows based on how many outliers they contain and how far they are
    suspects = []
    for idx, row in enumerate(rows):
        score = 0
        for i in input_indices:
            try:
                val = float(row[i])
                low, high = bounds[i]
                if val < low or val > high:
                    # Add magnitude of deviation to score
                    score += abs(val - (low if val < low else high))
            except (ValueError, TypeError):
                continue
        if score > 0:
            suspects.append((score, idx))
            
    # Sort by score descending (most suspicious first)
    suspects.sort(key=lambda x: x[0], reverse=True)
    return [idx for score, idx in suspects]

import csv
import random

def race(header, rows, label):
    """
    Compares three strategies: random, greedy, and hill-climbing.
    Budget is 50 labels total.
    """
    goal_indices = [(i, name[-1]) for i, name in enumerate(header) 
                    if name.endswith('-') or name.endswith('+')]

    def score_row(row_values):
        total = 0.0
        for idx, op in goal_indices:
            val = float(row_values[idx])
            total += val if op == '-' else (1.0 - val)
        return total

    labeled = {}
    def get_label(i):
        if i not in labeled:
            if len(labeled) >= 50:
                raise Exception("Budget exceeded")
            labeled[i] = label(i)
        return labeled[i]

    def get_score(i):
        vals = get_label(i)
        temp = list(rows[i])
        for j, (idx, _) in enumerate(goal_indices):
            temp[idx] = vals[j]
        return score_row(temp)

    # Strategy 1: Random
    def strategy_random():
        best_idx = -1
        best_score = float('inf')
        for _ in range(16): # 16 labels
            i = random.randrange(len(rows))
            s = get_score(i)
            if s < best_score:
                best_score, best_idx = s, i
        return best_idx, best_score

    # Strategy 2: Greedy (Sample 16, pick best)
    def strategy_greedy():
        indices = random.sample(range(len(rows)), 16)
        best_idx = -1
        best_score = float('inf')
        for i in indices:
            s = get_score(i)
            if s < best_score:
                best_score, best_idx = s, i
        return best_idx, best_score

    # Strategy 3: Hill Climbing (Start random, move to better neighbor)
    def strategy_hill():
        curr = random.randrange(len(rows))
        curr_score = get_score(curr)
        for _ in range(17): # 18 labels total
            neighbor = random.randrange(len(rows))
            n_score = get_score(neighbor)
            if n_score < curr_score:
                curr, curr_score = neighbor, n_score
        return curr, curr_score

    results = {
        "random": strategy_random(),
        "greedy": strategy_greedy(),
        "hill": strategy_hill()
    }

    winner = min(results, key=lambda k: results[k][1])
    return {
        "winner": winner,
        "row": results[winner][0],
        "tried": ["random", "greedy", "hill"]
    }

import csv

def certify(xs, ys):
    """
    Returns True if xs and ys are statistically similar (same distribution),
    False if they differ significantly. Uses the A12 effect size test.
    """
    m, n = len(xs), len(ys)
    if m == 0 or n == 0:
        return True
    
    greater = 0
    for x in xs:
        for y in ys:
            if x > y:
                greater += 1
            elif x == y:
                greater += 0.5
    
    a12 = greater / (m * n)
    # A12 threshold of 0.56 is a standard small effect size.
    # If the probability is between 0.44 and 0.56, they are statistically similar.
    return 0.44 <= a12 <= 0.56

    return certify(before, after), before, after

def flow(header, row, seen):
    """
    Warns if a project is going bad. We define 'bad' as a row that is 
    significantly worse than the average of the 'seen' history.
    """

    for i, name in enumerate(header):
        if name.endswith('-') or name.endswith('+'):

        total = 0.0
        for idx, op in goal_indices:
            val = float(r[idx])
            total += val if op == '-' else (1.0 - val)
        return total

    # If we have no history, we cannot judge if it is going bad
    if not seen:
        return False

    # Calculate current score
    current_score = score_row(row)
    
    # Calculate average score of history
    history_scores = [score_row(r) for r in seen]
    avg_history = sum(history_scores) / len(history_scores)
    
    # If current score is significantly worse (higher) than the average,
    # we flag it as "going bad".
    # We use a threshold of 1.5x the average to trigger a warning.
    if current_score > (avg_history * 1.5):
        return True
        
    return False

import csv

def blame(header, rows):
    """
    Identifies the two input columns that correlate most strongly with 
    the 'Cost-' goal. We define 'worst' as rows where 'Cost-' is in 
    the top 25% (highest values). We then find input columns that 
    have the highest correlation with these high 'Cost-' values.
    """
    cost_idx = header.index('Cost-')
    input_indices = [i for i, name in enumerate(header) 
                     if not (name.endswith('-') or name.endswith('+'))]
    
    # Sort rows by Cost- descending to find the "worst"
    sorted_rows = sorted(rows, key=lambda r: float(r[cost_idx]), reverse=True)
    worst_rows = sorted_rows[:len(rows) // 4]
    
    # Calculate average of each input in the worst rows
    # and compare to the global average of that input.
    # A high difference indicates that input contributes to the "worst" state.
    scores = []
    for i in input_indices:
        worst_avg = sum(float(r[i]) for r in worst_rows) / len(worst_rows)
        global_avg = sum(float(r[i]) for r in rows) / len(rows)
        
        # We look for inputs that are significantly higher in the worst rows
        # (Assuming higher input values correlate with higher Cost- values)
        diff = worst_avg - global_avg
        scores.append((diff, header[i]))
    
    # Sort by difference descending
    scores.sort(key=lambda x: x[0], reverse=True)
    
    # Return the top 2 inputs that explain the cost
    return [name for diff, name in scores[:2]]

import csv

def justify(header, rows, row):
    """
    Compares a given setup 'row' against all 'rows' in the dataset.
    Estimates the goals of 'row' using the nearest neighbor from 'rows',
    then calculates the percentile rank of 'row' based on goal performance.
    """
    input_indices = [i for i, name in enumerate(header) 
                     if not (name.endswith('-') or name.endswith('+'))]
    goal_indices = [(i, name[-1]) for i, name in enumerate(header) 
                    if name.endswith('-') or name.endswith('+')]

    def get_score(r):
        # Lower score is better: '-' minimized, '+' maximized (1 - val)
        total = 0.0
        for idx, op in goal_indices:
            val = float(r[idx])
            total += val if op == '-' else (1.0 - val)
        return total

    # Estimate missing goals using 1-Nearest Neighbor
    best_dist = float('inf')
    best_neighbor = None
    for train_row in rows:
        dist = sum((float(row[i]) - float(train_row[i]))**2 for i in input_indices)
        if dist < best_dist:
            best_dist = dist
            best_neighbor = train_row
    
    # Create a copy of the row with estimated goals
    estimated_row = list(row)
    for idx, _ in goal_indices:
        estimated_row[idx] = best_neighbor[idx]
    
    # Compare against all rows
    my_score = get_score(estimated_row)
    better_count = sum(1 for r in rows if get_score(r) < my_score)
    p = better_count / len(rows)
    
    reason = f"This setup is better than {p:.2f} of all known configurations based on its estimated performance metrics."
    return p, reason

import csv

def choose(header, rows, label):
    """
    Finds the best setup for a 'large' project (high 'Size') where failure is 
    damaging (high 'Criticality'). We prioritize rows that meet these criteria 
    and then optimize for the best goal performance within a 50-label budget.
    """
    size_idx = header.index('Size')
    crit_idx = header.index('Criticality')
    goal_indices = [(i, name[-1]) for i, name in enumerate(header) 
                    if name.endswith('-') or name.endswith('+')]

    def score_row(row_values):
        total = 0.0
        for idx, op in goal_indices:
            val = float(row_values[idx])
            total += val if op == '-' else (1.0 - val)
        return total

    # 1. Filter for 'large' and 'high criticality' projects
    # We look for rows where Size > 0.7 and Criticality > 0.7
    candidates = [i for i, row in enumerate(rows) 
                  if float(row[size_idx]) > 0.7 and float(row[crit_idx]) > 0.7]
    
    # If no candidates found, fallback to all rows
    if not candidates:
        candidates = list(range(len(rows)))

    # 2. Sample up to 50 candidates to find the best one
    trials = min(50, len(candidates))
    sample_indices = random.sample(candidates, trials)
    
    best_idx = -1
    best_score = float('inf')
    
    for i in sample_indices:
        true_goals = label(i)
        temp_row = list(rows[i])
        for j, (idx, _) in enumerate(goal_indices):
            temp_row[idx] = true_goals[j]
            
        current_score = score_row(temp_row)
        if current_score < best_score:
            best_score = current_score
            best_idx = i
            
    return best_idx
