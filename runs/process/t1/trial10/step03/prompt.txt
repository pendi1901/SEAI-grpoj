import csv

def remember(header, rows):
    """
    Identifies rows that perform well across all goal columns.
    Goals ending in '-' are minimized, '+' are maximized.
    """
    goal_indices = []
    for i, name in enumerate(header):
        if name.endswith('-') or name.endswith('+'):
            goal_indices.append((i, name[-1]))

    def score_row(row):
        # Calculate a normalized score for the row.
        # Since we want to minimize '-' and maximize '+', 
        # we can treat all as "minimize" by negating '+' values.
        # We use the relative rank or simple sum of normalized values.
        # Here, we use the sum of values (after negating '+' goals) 
        # to find rows that are generally "low" (good).
        s = 0
        for idx, op in goal_indices:
            val = float(row[idx])
            if op == '+':
                s += (1.0 - val)
            else:
                s += val
        return s

    # Score all rows
    scored = []
    for i, row in enumerate(rows):
        scored.append((score_row(row), i))
    
    # Sort by score (ascending, as lower is better)
    scored.sort()
    
    # Return top 5% (or at least 1)
    limit = max(1, len(rows) // 20)
    return [item[1] for item in scored[:limit]]

import csv

def guess(header, train, test):
    """
    Estimates the first goal column for each test row using a 
    k-Nearest Neighbors approach based on input columns.
    """
    # Identify input columns (those not ending in - or +)
    input_indices = [i for i, name in enumerate(header) 
                     if not (name.endswith('-') or name.endswith('+'))]
    
    # Identify the target column (the first goal column)
    target_idx = next(i for i, name in enumerate(header) 
                      if name.endswith('-') or name.endswith('+'))
    
    predictions = []
    
    for t_row in test:
        # Calculate Euclidean distance to all training rows based on inputs
        distances = []
        for train_row in train:
            dist = 0
            for i in input_indices:
                # Assuming numeric inputs as per problem description
                dist += (float(t_row[i]) - float(train_row[i]))**2
            distances.append((dist**0.5, float(train_row[target_idx])))
        
        # Simple k-NN: average of the 3 nearest neighbors
        distances.sort()
        k = 3
        neighbors = distances[:k]
        avg_val = sum(n[1] for n in neighbors) / k
        predictions.append(avg_val)
        
    return predictions
