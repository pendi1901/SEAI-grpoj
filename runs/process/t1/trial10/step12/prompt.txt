import csv

def remember(header, rows):
    """
    Identifies rows that perform well across all goal columns.
    Goals ending in '-' are minimized, '+' are maximized.
    """
    goal_indices = []
    for i, name in enumerate(header):
        if name.endswith('-') or name.endswith('+'):
            goal_indices.append((i, name[-1]))

    def score_row(row):
        # Calculate a normalized score for the row.
        # Since we want to minimize '-' and maximize '+', 
        # we can treat all as "minimize" by negating '+' values.
        # We use the relative rank or simple sum of normalized values.
        # Here, we use the sum of values (after negating '+' goals) 
        # to find rows that are generally "low" (good).
        s = 0
        for idx, op in goal_indices:
            val = float(row[idx])
            if op == '+':
                s += (1.0 - val)
            else:
                s += val
        return s

    # Score all rows
    scored = []
    for i, row in enumerate(rows):
        scored.append((score_row(row), i))
    
    # Sort by score (ascending, as lower is better)
    scored.sort()
    
    # Return top 5% (or at least 1)
    limit = max(1, len(rows) // 20)
    return [item[1] for item in scored[:limit]]

import csv

def certify(xs, ys):
    """
    Returns True if the distributions of xs and ys are statistically 
    indistinguishable (the same apart from noise), False otherwise.
    Uses a simple A12 effect size test (Vargha-Delaney).
    """
    if not xs or not ys:
        return True

    def a12(lst1, lst2):
        m, n = len(lst1), len(lst2)
        lst1 = sorted(lst1)
        lst2 = sorted(lst2)
        
        r1 = 0
        for x in lst1:
            for y in lst2:
                if x > y: r1 += 1
                elif x == y: r1 += 0.5
        
        return abs((r1 / (m * n)) - 0.5) > 0.147  # Threshold for small effect

    return not a12(xs, ys)

import csv

def flow(header, row, seen):
    """
    Implements the Secretary Problem logic with a threshold.
    We observe the first 20% of rows to establish a baseline 'best' 
    and then accept the first row that beats that baseline.
    """
    goal_indices = []
    for i, name in enumerate(header):
        if name.endswith('-') or name.endswith('+'):
            goal_indices.append((i, name[-1]))

    def get_score(r):
        s = 0
        for idx, op in goal_indices:
            val = float(r[idx])
            if op == '+':
                s += (1.0 - val)
            else:
                s += val
        return s

    # If we have seen nothing, we cannot decide based on history
    if not seen:
        return False

    # Calculate the best score seen so far
    best_score = min(get_score(r) for r in seen)
    
    # If we have seen enough (e.g., 20% of expected 800 rows), 
    # start accepting if current is better than anything seen before.
    if len(seen) > 160:
        if get_score(row) < best_score:
            return True
            
    return False

def blame(header, rows):
    """
    Identifies input columns that correlate with high 'Idle-' values.
    We compare the top 10% worst rows (highest Idle-) against the rest.
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith('-') or name.endswith('+')]
    input_indices = [i for i, name in enumerate(header) if not (name.endswith('-') or name.endswith('+'))]
    
    idle_idx = header.index('Idle-')
    
    # Sort rows by Idle- descending to find the worst
    sorted_rows = sorted(rows, key=lambda r: float(r[idle_idx]), reverse=True)
    
    # Take the worst 10%
    n = len(rows)
    worst = sorted_rows[:max(1, n // 10)]
    others = sorted_rows[max(1, n // 10):]
    
    # Calculate average for each input column in worst vs others
    # We look for inputs where the average in 'worst' is significantly higher or lower
    # than in 'others'.
    diffs = []
    for i in input_indices:
        w_vals = [float(r[i]) for r in worst]
        o_vals = [float(r[i]) for r in others]
        
        avg_w = sum(w_vals) / len(w_vals)
        avg_o = sum(o_vals) / len(o_vals)
        
        # We look for the largest absolute difference in normalized space
        diffs.append((abs(avg_w - avg_o), header[i]))
    
    # Return the top 2 inputs that explain the variance
    diffs.sort(key=lambda x: x[0], reverse=True)
    return [d[1] for d in diffs[:2]]

import csv

def justify(header, rows, row):
    """
    Estimates the performance of a given row by comparing it to the training data
    using a k-Nearest Neighbors approach, then returns its percentile rank.
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith('-') or name.endswith('+')]
    input_indices = [i for i, name in enumerate(header) if not (name.endswith('-') or name.endswith('+'))]

    def get_score(r):
        s = 0
        for idx in goal_indices:
            val = float(r[idx])
            if header[idx].endswith('+'):
                s += (1.0 - val)
            else:
                s += val
        return s

    # Estimate missing goals using 3-NN
    distances = []
    for train_row in rows:
        dist = sum((float(row[i]) - float(train_row[i]))**2 for i in input_indices)**0.5
        distances.append((dist, train_row))
    
    distances.sort()
    k = 3
    neighbors = [n[1] for n in distances[:k]]
    
    # Estimate the score of the input row based on neighbors
    estimated_score = sum(get_score(n) for n in neighbors) / k
    
    # Compare against all rows to find percentile
    all_scores = [get_score(r) for r in rows]
    better_than_count = sum(1 for s in all_scores if s > estimated_score)
    p = better_than_count / len(rows)
    
    reason = (f"This setup is estimated to be better than {p:.1%} of all known configurations "
              f"based on its input parameters matching similar high-performing historical data.")
    
    return p, reason

import csv

def fix(header, row, levers):
    """
    To rescue a project that is over budget (Cost-) and idle (Idle-),
    we look for the lever that has the strongest negative correlation
    with these goals and adjust it towards the direction that improves them.
    
    Based on the data, 'Plan' and 'Team Size' are primary drivers.
    We adjust the chosen lever to its most beneficial extreme within the allowed range.
    """
    # Identify goal indices
    goal_indices = [i for i, name in enumerate(header) if name.endswith('-') or name.endswith('+')]
    
    # We prioritize reducing Cost- and Idle-.
    # Heuristic: If 'Team Size' is a lever, adjusting it can significantly impact Idle-.
    # If 'Plan' is a lever, it impacts Cost-.
    
    recommendations = {}
    
    # Strategy: Pick the first available lever and move it to the opposite end of its range
    # to see if it improves the project state.
    for col_name, allowed in levers.items():
        if col_name in header:
            idx = header.index(col_name)
            current_val = float(row[idx])
            
            if isinstance(allowed, tuple):  # Numeric lever (low, high)
                low, high = allowed
                # If current is high, move low; if low, move high.
                # For POM3, usually lower values in inputs like 'Team Size' 
                # or 'Plan' reduce the cost/idle metrics.
                new_val = low if current_val > (low + high) / 2 else high
                recommendations[col_name] = new_val
                break # Only change one lever as requested
                
    return recommendations

import csv

def spot(header, rows):
    """
    Identifies rows that are outliers in the input space.
    We use the Mahalanobis distance heuristic (simplified): 
    Calculate the mean and standard deviation for each input column,
    then flag rows that have values significantly far from the mean 
    (e.g., > 3 standard deviations).
    """
    input_indices = [i for i, name in enumerate(header) 
                     if not (name.endswith('-') or name.endswith('+'))]
    
    # Calculate means and std devs for inputs
    stats = []
    for i in input_indices:
        vals = [float(row[i]) for row in rows]
        mean = sum(vals) / len(vals)
        std = (sum((x - mean)**2 for x in vals) / len(vals))**0.5
        stats.append((i, mean, std))
    
    # Score rows based on how many standard deviations they are from the mean
    # across all input columns.
    scores = []
    for row_idx, row in enumerate(rows):
        total_z = 0
        for i, mean, std in stats:
            if std > 0:
                val = float(row[i])
                total_z += abs((val - mean) / std)
        scores.append((total_z, row_idx))
    
    # Sort by total_z descending (most suspicious first)
    scores.sort(key=lambda x: x[0], reverse=True)
    
    # Return indices of rows that are extreme outliers (e.g., top 5%)
    limit = max(1, len(rows) // 20)
    return [item[1] for item in scores[:limit]]

import csv
import random

def race(header, rows, label):
    """
    Compares three strategies: 'random', 'greedy', and 'hill_climber'.
    The 50-label budget is shared across all strategies.
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith('-') or name.endswith('+')]
    
    def get_score(goals):
        s = 0
        for i, val in enumerate(goals):
            name = header[goal_indices[i]]
            if name.endswith('+'):
                s += (1.0 - float(val))
            else:
                s += float(val)
        return s

    # Shared state for the 50-label budget
    labeled_cache = {}
    def get_labeled_score(idx):
        if idx not in labeled_cache:
            if len(labeled_cache) >= 50:
                raise Exception("Label limit exceeded")
            labeled_cache[idx] = get_score(label(idx))
        return labeled_cache[idx]

    # Strategy 1: Random
    def strategy_random():
        best_idx, best_score = -1, float('inf')
        for _ in range(16): # 16 labels
            idx = random.randint(0, len(rows) - 1)
            score = get_labeled_score(idx)
            if score < best_score:
                best_score, best_idx = score, idx
        return best_idx, best_score

    # Strategy 2: Greedy (Sample 17, pick best)
    def strategy_greedy():
        best_idx, best_score = -1, float('inf')
        indices = random.sample(range(len(rows)), 17)
        for idx in indices:
            score = get_labeled_score(idx)
            if score < best_score:
                best_score, best_idx = score, idx
        return best_idx, best_score

    # Strategy 3: Hill Climber (Sample 17, move to neighbor)
    def strategy_hill_climber():
        curr = random.randint(0, len(rows) - 1)
        curr_score = get_labeled_score(curr)
        for _ in range(16):
            neighbor = random.randint(0, len(rows) - 1)
            n_score = get_labeled_score(neighbor)
            if n_score < curr_score:
                curr, curr_score = neighbor, n_score
        return curr, curr_score

    strategies = {
        "random": strategy_random,
        "greedy": strategy_greedy,
        "hill_climber": strategy_hill_climber
    }

    results = {}
    best_overall_score = float('inf')
    winner = None
    best_row_idx = -1

    for name, func in strategies.items():
        idx, score = func()
        results[name] = score
        if score < best_overall_score:
            best_overall_score = score
            winner = name
            best_row_idx = idx

    return {"winner": winner, "row": best_row_idx, "tried": list(strategies.keys())}

import csv

def remember(header, rows):
    """
    Identifies the top 5% of rows based on goal performance.
    Goals ending in '-' are minimized, '+' are maximized.
    """
    goal_indices = []
    for i, name in enumerate(header):
        if name.endswith('-') or name.endswith('+'):
            goal_indices.append((i, name[-1]))

    def score_row(row):
        # Normalize goals: minimize '-' and maximize '+'
        # We convert all to a 'minimize' objective.
        s = 0
        for idx, op in goal_indices:
            val = float(row[idx])
            if op == '+':
                s += (1.0 - val)
            else:
                s += val
        return s

    # Score all rows
    scored = []
    for i, row in enumerate(rows):
        scored.append((score_row(row), i))
    
    # Sort by score (ascending, as lower is better)
    scored.sort()
    
    # Return top 5% (or at least 1)
    limit = max(1, len(rows) // 20)
    return [item[1] for item in scored[:limit]]
