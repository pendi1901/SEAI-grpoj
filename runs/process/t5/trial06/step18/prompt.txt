#-- misc ---------------------------------------------------
def say(x: Any, p: int = 2) -> str:
  "Floats get P decimals; dicts list their keys."
  if isinstance(x,float): x= f"{x:.{p}f}".rstrip("0").rstrip(".")
  if isinstance(x, dict):
    x= "{"+", ".join(f"{k}: {say(v,p)}" for k,v in x.items())+"}"
  return str(x)

class o(dict):
  "A dict you can poke with a dot."
  __getattr__, __setattr__ = dict.__getitem__, dict.__setitem__
  __repr__ = say

def atom(s: str) -> ATOM:
  "'22' -> 22. '3.1' -> 3.1. 'True' -> True. 'x' -> 'x'."
  for fn in (int, float):
    try: return fn(s)
    except ValueError: pass
  s = s.strip()
  return {"True": True, "False": False}.get(s, s)

def csv(file: str) -> Iterator[ROW]:
  "ROWS of FILE, each cell coerced."
  with open(os.path.expanduser(file),  # -sig drops any BOM
           encoding="utf-8-sig") as f:
    for line in f:
      if line.strip(): yield [atom(s) for s in line.split(",")]

def Col(txt: str = " ", at: int = 0) -> COL:
  "Factory that returns NUM or SYM."
  return Num(txt,at) if txt[0].isupper() else Sym(txt,at)

class Num:
  "Place to summarize stream of numbers."
  __slots__ = ("at", "txt", "n", "mu", "m2", "sd", "goal")
  def __init__(self, txt: str = " ", at: int = 0) -> None:
    self.at, self.txt, self.n = at, txt, 0
    self.mu = self.m2 = self.sd = 0
    self.goal = 0 if txt[-1] == "-" else 1
  __repr__ = lambda self: say({k: getattr(self,k)
                               for k in self.__slots__})

class Sym:
  "Place to summarize stream of Symbols.  The class is the tag."
  __slots__ = ("at", "txt", "n", "has")
  def __init__(self, txt: str = " ", at: int = 0) -> None:
    self.at, self.txt, self.n, self.has = at, txt, 0, {}
  __repr__ = lambda self: say({k: getattr(self,k)
                               for k in self.__slots__})

def add(col: COL, v: ATOM, inc: int = 1) -> None:
  "Show V to COL.  INC=-1 takes it away.  `?` changes nothing."
  if v != "?":
    col.n += inc
    if type(col) is Sym:
      col.has[v] = col.has.get(v, 0) + inc
    else:
      d = v - col.mu                      # Welford, for `div`
      col.mu += inc * d / max(1, col.n)
      col.m2  = max(0, col.m2 + inc * d * (v - col.mu))
      col.sd  = 0 if col.n < 2 else (col.m2/(col.n-1))**.5

def mid(col: COL) -> ATOM:
  "Middle: the mean, or the most common symbol; ? if none."
  return (max(col.has, key=col.has.get) if col.has else "?"
          ) if type(col) is Sym else col.mu

def mids(tbl: TBL) -> MIDS:
  "Every column's middle, cached until the next addCols."
  tbl.mids = tbl.mids or {at: mid(c) for at,c in tbl.cols.items()}
  return tbl.mids

def div(col: COL) -> float:
  "Spread: entropy, or standard deviation."
  return (-sum(v/col.n*log2(v/col.n) for v in col.has.values())
          if type(col) is Sym else col.sd)  # sd kept by `add`

def norm(col: COL, v: ATOM) -> ATOM:
  "To 0..1, by the logistic curve.  Symbols do not scale."
  if type(col) is Sym: return v
  z = max(-3, min(3, (v - col.mu) / (1e-32 + div(col))))
  return 1 / (1 + exp(-1.7 * z))

#-- tables -------------------------------------------------
def Tbl(src: Iterable[ROW]) -> TBL:
  "Header names the columns: X skip, +- goal, ! klass."
  src = iter(src)
  tbl = Cols(o(rows=[], cols={}, x=[], y=[], mids=None,
               names=next(src)))
  for row in src: addRow(tbl, row)
  return tbl

def Cols(tbl: TBL) -> TBL:
  "Create column roles."
  for at, s in enumerate(tbl.names):
    if s[-1] == "X": continue            # skip me entirely
    tbl.cols[at] = Col(s,at)
    (tbl.y if s[-1] in "+-!" else tbl.x).append(tbl.cols[at])
  return tbl

def addRow(tbl: TBL, row: ROW) -> ROW:
  "Keep ROW, and show it to my columns."
  tbl.rows += [row]
  return addCols(tbl, row)

def addCols(tbl: TBL, row: ROW, inc: int = 1) -> ROW:
  "Show ROW to my columns, without keeping it."
  tbl.mids = None                 # a new row moves the middles
  for at, col in tbl.cols.items(): add(col, row[at], inc)
  return row

def clone(tbl: TBL, rows: ROWS = None) -> TBL:
  "An empty copy of TBL, plus ROWS."
  return Tbl([tbl.names] + (rows or []))

def dist(vs: Iterable[QTY], n: int) -> float:
  "Root mean square of N gaps."
  return sqrt(sum(v*v for v in vs) / n)

def ydist(tbl: TBL, row: ROW) -> float:
  "How far ROW's goals are from the best they could be."
  return dist((abs(norm(c, row[c.at]) - c.goal)
                    for c in tbl.y), len(tbl.y))

def gap(col: COL, a: ATOM, b: ATOM) -> QTY:
  "Distance between two values of one column."
  if a == "?" or b == "?": return 1    # unknown = far
  return a!=b if type(col) is Sym else abs(norm(col,a)-norm(col,b))

def xdist(tbl: TBL, r1: ROW, r2: ROW | MIDS) -> float:
  "How far apart two rows are, over the x columns."
  return dist((gap(c, r1[c.at], r2[c.at])
                    for c in tbl.x), len(tbl.x))

def oracle(row: ROW) -> ROW:
  """Labeller. Rows here arrive with ys, so nothing to do.
  Real oracles fill in any missing y cells (and can close
  over a TBL to know which cells those are)."""
  return row

def acquire(tbl: TBL, cap: int | None = None,
            label: ORACLE = oracle) -> ROWS:
  "Label near the best rows, far from the rest."
  cap  = cap or the.Stop
  todo = random.sample(tbl.rows, len(tbl.rows))[:the.Few]
  both = clone(tbl, [label(todo.pop())
                     for _ in range(the.Start)])
  both.rows.sort(key=lambda r: ydist(both, r))
  n    = int(sqrt(the.Start))
  best,rest = clone(tbl,both.rows[:n]), clone(tbl,both.rows[n:])
  while todo and len(both.rows) < cap:
    cb, cr = mids(best), mids(rest)    # two centroids
    todo.sort(key=lambda z: xdist(tbl,z,cr) - xdist(tbl,z,cb))
    addRow(best, addRow(both, label(todo.pop())))
    best.rows.sort(key=lambda z: ydist(both, z))  # no y leak
    if len(best.rows) > sqrt(len(both.rows)): # best stays small
      addRow(rest, addCols(best, best.rows.pop(), -1))
  return both.rows

def grabs(tbl: TBL, cap: int | None = None,
          label: ORACLE = oracle) -> ROWS:
  "The straw man: same budget, taken at random."
  cap = cap or the.Stop
  return [label(r) for r in
          random.sample(tbl.rows, min(cap, len(tbl.rows)))]

def cut(tbl: TBL, rows: ROWS) -> tuple[int,ATOM,GO]|None:
  "The (at, v, go) whose two sides have the tightest ys."
  ys = [ydist(tbl, r) for r in rows]
  out, least = None, 1e30
  for col in tbl.x:
    for v, go in candidates(col, rows):
      a, b = Num(), Num()   # one pass, so one `go` per row
      for r,y in zip(rows,ys): add(a if go(r) else b, y)
      if a.n >= the.Leaf and b.n >= the.Leaf:
        score = (div(a)*a.n + div(b)*b.n) / (a.n + b.n)
        if score < least: out, least = (col.at, v, go), score
  return out

def candidates(col: COL, rows: ROWS) -> Iterator[tuple[ATOM,GO]]:
  "What splits to try: one per symbol, or CUTS per num."
  at = col.at
  if type(col) is Sym:
    for v in sorted({r[at] for r in rows if r[at] != "?"}):
      yield v, lambda r,v=v,at=at: r[at] == v
  else:                # by rank, not value: THESE rows are dense
    xs = sorted(r[at] for r in rows if r[at] != "?")
    k  = max(1, len(xs)//the.Cuts)     # cuts, spaced by rank
    for v in xs[k-1:-1:k]:
      yield v, (lambda r,v=v,at=at: r[at] != "?" and r[at] <= v)

def tree(tbl: TBL, rows: ROWS) -> NODE:
  "Split while a split keeps both sides big enough."
  node = o(rows=rows, at=None, v=None, go=None, kids=[],
           mu=sum(ydist(tbl,r) for r in rows)/len(rows))
  if len(rows) > the.Leaf and (found := cut(tbl, rows)):
    at, v, go = found
    yes, no = [], []
    for r in rows: (yes if go(r) else no).append(r)
    node.at, node.v, node.go = at, v, go # cut kepts sides big
    node.kids = [tree(tbl, yes), tree(tbl, no)]
  return node

def leaf(node: NODE, row: ROW) -> NODE:
  "Walk ROW down to its leaf."
  while node.kids: node = node.kids[0 if node.go(row) else 1]
  return node

def leafs(node: NODE) -> list[NODE]:
  "Return leaves of this tree."
  return [x for k in node.kids for x in leafs(k)] or [node]

def show(tbl: TBL, node: NODE, pre: str | None = None,
         edge: str = "") -> None:
  "Tree on the left; ydist and n on the right."
  ls = sorted(leafs(node), key=lambda z: z.mu)
  def walk(z: NODE, pre: str | None, edge: str) -> None:
    m = "+" if z is ls[0] else "-" if z is ls[-1] else " "
    print(f"{m} {round(100*z.mu):>4} {len(z.rows):>5}   "
          f"{(pre or '')+edge}".rstrip())
    op = "=" if type(tbl.cols[z.at or 0]) is Sym else "<="
    no = "!=" if op == "=" else ">"
    for kid, oper in zip(z.kids, [op, no]):
      walk(kid, "" if pre is None else pre + "|  ",
           f"{tbl.names[z.at]} {oper} {say(z.v)}")
  print(" ydist     n")
  walk(node, pre, edge)



#-- rig ----------------------------------------------------
# Calls core, acquire, and one RANKER -- tree's, by default.
def ranker(lab: TBL) -> Callable[[ROW], float]:
  "Grow a tree on LAB; score any row by its leaf's mean ydist."
  node = tree(lab, lab.rows)
  return lambda r: leaf(node, r).mu

def wins(tbl: TBL) -> Callable[[ROW], float]:
  "100 at the best row, 0 at an average one."
  ys = sorted(ydist(tbl, r) for r in tbl.rows)
  lo, avg = ys[0], sum(ys)/len(ys)
  return lambda r: max(-100, min(100, 100*(1 - (ydist(tbl,r)-lo)
                                            / (avg-lo+1e-32))))

def holdout(tbl: TBL, pick: PICKER = acquire,
            rank: RANKER = ranker,
            label: ORACLE = oracle) -> ROW:
  "Train on half; of CHECK guesses on the rest, pick best."
  rows  = random.sample(tbl.rows, len(tbl.rows))
  n     = len(rows)//2
  tr    = clone(tbl, rows[:n][:the.Few])
  lab   = clone(tbl, pick(tr, the.Stop - the.Check, label))
  top   = sorted(rows[n:], key=rank(lab))
  return min(top[:the.Check], key=lambda r: ydist(lab, label(r)))

# A leaf: it calls core, and nothing here calls it.
def adds(vs: Iterable[ATOM], col: COL | None = None) -> COL:
  "Every item of VS into COL."
  col = Num() if col is None else col
  for v in vs: add(col, v)
  return col

def cohen(xs: NUMS, ys: NUMS, d: float = 0.35,
          eps: float = 0) -> bool:
  "Are the means within D pooled standard deviations?"
  a, b = adds(xs), adds(ys)
  pool = ((a.n-1)*div(a)**2 + (b.n-1)*div(b)**2)/(a.n+b.n-2)
  return abs(a.mu - b.mu) <= max(eps, d * sqrt(pool))

def cliffs(xs: NUMS, ys: NUMS, d: float = 0.197) -> bool:
  "Sorted in.  Is the rank imbalance small enough?"
  gt = lt = j = k = 0
  for x in xs:
    while j < len(ys) and ys[j] <  x: j += 1; k = j
    while k < len(ys) and ys[k] <= x: k += 1
    gt += j; lt += len(ys) - k
  return abs(gt - lt) / (len(xs) * len(ys)) <= d

def ks(xs: NUMS, ys: NUMS, a: float = 1.36) -> bool:
  "Sorted in.  95% Kolmogorov-Smirnov."
  n, m, i, j, d = len(xs), len(ys), 0, 0, 0
  while i < n and j < m:
    v = min(xs[i], ys[j])
    while i < n and xs[i] <= v: i += 1
    while j < m and ys[j] <= v: j += 1
    d = max(d, abs(i/n - j/m))
  return d <= a * sqrt((n + m) / (n * m))

def same(xs: NUMS, ys: NUMS, eps: float = 0) -> bool:
  """Indistinguishable.  Always cliffs and ks; EPS adds cohen, and
  doubles as its effect size (eps=0 skips that third test)."""
  xs, ys = sorted(xs), sorted(ys)
  return (cliffs(xs,ys) and ks(xs,ys)
          and (not eps or cohen(xs,ys,d=eps)))

import ezr
import random

def remember(header, rows):
    """
    Identify the top 5% of rows that perform best across all goals.
    Goals ending in '-' are minimized, '+' are maximized.
    """
    # Create a table object to leverage ezr's normalization and distance logic
    # We construct the table by combining the header and the rows
    data = [header] + rows
    tbl = ezr.Tbl(data)
    
    # Calculate ydist for every row. 
    # ydist in ezr calculates the distance of a row's goals from the ideal.
    # Lower ydist is better.
    scored_rows = []
    for i, row in enumerate(rows):
        d = ezr.ydist(tbl, row)
        scored_rows.append((d, i))
    
    # Sort by distance (ascending, since lower is better)
    scored_rows.sort(key=lambda x: x[0])
    
    # Return the indices of the top 5%
    limit = max(1, len(rows) // 20)
    return [item[1] for item in scored_rows[:limit]]

import ezr
import random

def guess(header, train, test):
    """
    Estimates the first goal column (Cost-) for each test row.
    Uses the tree-based ranker from ezr to find the leaf node 
    corresponding to the test row and returns the mean of the 
    first goal column in that leaf.
    """
    # Create a table for the training data
    data = [header] + train
    tbl = ezr.Tbl(data)
    
    # Build a tree on the training data
    root = ezr.tree(tbl, tbl.rows)
    
    # Identify the index of the first goal column (Cost-)
    # Based on the header, it is the first column ending in '-'
    goal_idx = -1
    for i, name in enumerate(header):
        if name.endswith("-"):
            goal_idx = i
            break
            
    results = []
    for row in test:
        # Walk the tree to find the leaf for this test row
        node = ezr.leaf(root, row)
        
        # Calculate the mean of the goal column in this leaf
        # We use the column object from the table to get the mean
        col = tbl.cols[goal_idx]
        
        # If the leaf has rows, return the average of the goal column
        if node.rows:
            vals = [r[goal_idx] for r in node.rows]
            avg = sum(vals) / len(vals)
            results.append(avg)
        else:
            # Fallback to global mean if leaf is somehow empty
            results.append(col.mu)
            
    return results

if __name__ == "__main__":
    # Load data
    filename = "pom3a.n1000.train.csv"
    data = list(ezr.csv(filename))
    header = data[0]
    all_rows = data[1:]
    
    # Split into train and test (80/20 split)
    random.seed(1234567891)
    random.shuffle(all_rows)
    split = int(len(all_rows) * 0.8)
    train = all_rows[:split]
    test_full = all_rows[split:]
    
    # Mask the goals in the test set to simulate "unknown"
    # Goals are the last 3 columns: Cost-, Score-, Idle-
    test_masked = []

        for i in range(len(header) - 3, len(header)):
            masked_row[i] = "?"
        test_masked.append(masked_row)
        
    # Get estimates

    for i in range(min(5, len(estimates))):

import ezr

def certify(xs, ys):
    """
    Returns True if the distributions are statistically indistinguishable,
    False if they are significantly different.
    """
    # Using the same logic as ezr.same, which checks for statistical 
    # significance using Cliff's Delta and the Kolmogorov-Smirnov test.
    return ezr.same(xs, ys)

import ezr
import random

def flow(header, row, seen):
    """
    Decide whether to approve a row immediately.
    We compare the current row's potential against the best seen so far.
    If the current row is significantly better than the average of the 
    best 10% seen so far, we approve it.
    """
    if not seen:
        return True
    
    # Create a table to handle normalization and goal distance
    data = [header] + seen
    tbl = ezr.Tbl(data)
    
    # Calculate ydist for all seen rows
    # Lower ydist is better
    seen_scores = [ezr.ydist(tbl, r) for r in seen]
    
    # Identify the threshold: the average of the best 10% seen so far
    seen_scores.sort()
    limit = max(1, len(seen_scores) // 10)
    threshold = sum(seen_scores[:limit]) / limit
    
    # Calculate ydist for the current row
    # We use the existing table's column logic to normalize the new row
    current_score = ezr.ydist(tbl, row)
    
    # Approve if the current row is better (lower distance) than the threshold
    return current_score <= threshold

import ezr
import random

def blame(header, rows):
    """
    Identify input columns that best explain why the worst rows are bad.
    We define 'worst' as the rows with the highest ydist.
    We then compare the distribution of input values in the worst rows
    against the rest of the data to find the most significant differences.
    """
    # Create a table to handle normalization and distance
    data = [header] + rows
    tbl = ezr.Tbl(data)
    
    # Calculate ydist for all rows
    scored = [(ezr.ydist(tbl, r), r) for r in rows]
    scored.sort(key=lambda x: x[0], reverse=True)
    
    # Define worst as the bottom 25%
    n_worst = len(rows) // 4
    worst = [x[1] for x in scored[:n_worst]]
    rest = [x[1] for x in scored[n_worst:]]
    
    # For each input column, check if the distribution in 'worst' 
    # is statistically different from 'rest' using the same logic as ezr.same
    # We look for columns where the values are significantly different.
    # Since we want to explain why they are bad, we look for columns 
    # that show a clear shift in distribution.
    
    explanations = []
    for col in tbl.x:
        w_vals = [r[col.at] for r in worst if r[col.at] != "?"]
        r_vals = [r[col.at] for r in rest if r[col.at] != "?"]
        
        if not w_vals or not r_vals:
            continue
            
        # If they are NOT the same, this column contributes to the difference
        if not ezr.same(sorted(w_vals), sorted(r_vals)):
            explanations.append(col.txt)
            
    return explanations

    """

    # The ranker returns a function that estimates ydist

    estimated_ydist = ranker_func(row)
    
    # Calculate ydist for all existing rows to compare
    all_ydists = [ezr.ydist(tbl, r) for r in rows]

    better_than_count = sum(1 for d in all_ydists if d > estimated_ydist)
    p = better_than_count / len(rows)
    
    reason = (f"This setup is estimated to be in the top {int((1-p)*100)}% "
              f"of all possible configurations based on its input characteristics.")
    
    return p, reason

import ezr
import random

def choose(header, rows, label):
    """
    Find the best row by using the acquire strategy from ezr.
    We have a budget of 50 labels.
    """
    # Create a table object to manage the data.
    # We must ensure the table knows about the columns before we start.
    # We use a copy of the rows to avoid modifying the original list structure
    # while we perform the acquisition.
    tbl = ezr.Tbl([header])
    for row in rows:
        tbl.rows.append(row)
    
    # Re-initialize column stats based on the rows provided
    # (ezr.Tbl constructor already does this, but we ensure it's clean)
    tbl.mids = None
    for row in rows:
        ezr.addCols(tbl, row)

    # Track labeled indices to satisfy the budget constraint
    labeled_data = {}

    def oracle_wrapper(row):
        # Find the index of the row in the original list
        idx = -1
        for i, r in enumerate(rows):
            if r is row:
                idx = i
                break
        
        if idx != -1 and idx not in labeled_data:
            # Get the true values from the oracle
            true_row = label(idx)
            labeled_data[idx] = true_row
            # Update the row with true values
            for j, val in enumerate(true_row):
                row[j] = val
            # Update the table's column stats with the new information
            ezr.addCols(tbl, row)
        return row

    # Perform acquisition
    # We use the acquire function from ezr.
    # Note: acquire expects the table to have rows populated.
    final_rows = ezr.acquire(tbl, cap=50, label=oracle_wrapper)
    
    # After acquisition, find the best row among the ones we have labeled
    # The best row is the one with the lowest ydist.
    # We only consider rows that were actually labeled.
    best_row = min(final_rows, key=lambda r: ezr.ydist(tbl, r))
    
    # Return the index of the best row
    for i, r in enumerate(rows):
        if r is best_row:
            return i
    return 0

import ezr
import random

def fix(header, row, levers):
    """
    Finds the best change to a 'bad' row by testing all levers.
    For each lever, we try a value that moves the row towards the 
    global median of the best-performing rows.
    """
    # Load data to establish context for "good" performance
    data = list(ezr.csv("pom3a.n1000.train.csv"))
    tbl = ezr.Tbl(data)
    
    # Sort rows by ydist to find the best
    rows_data = data[1:]
    sorted_rows = sorted(rows_data, key=lambda r: ezr.ydist(tbl, r))
    best_rows = sorted_rows[:max(1, len(rows_data) // 10)]
    
    # Calculate the target values (midpoint of best rows)
    targets = {}
    for col in tbl.x:
        vals = [r[col.at] for r in best_rows]
        if isinstance(col, ezr.Num):
            targets[col.txt] = sum(vals) / len(vals)
        else:
            targets[col.txt] = max(set(vals), key=vals.count)
            
    # Evaluate each lever
    best_change = {}
    min_ydist = ezr.ydist(tbl, row)
    
    for col_name, allowed in levers.items():
        if col_name not in header:
            continue
        idx = header.index(col_name)
        original_val = row[idx]
        
        # Determine candidate values to test
        candidates = []
        if isinstance(allowed, tuple): # Numeric (low, high)
            low, high = allowed
            # Test the target value and the boundaries
            candidates = [max(low, min(high, targets.get(col_name, (low+high)/2)))]
        else: # Symbolic
            candidates = [v for v in allowed if v in allowed]
            
        for val in candidates:
            row[idx] = val
            current_ydist = ezr.ydist(tbl, row)
            
            if current_ydist < min_ydist:
                min_ydist = current_ydist
                best_change = {col_name: val}
            
        # Reset row
        row[idx] = original_val
            
    return best_change

def spot(header, rows):
    """
    Identifies rows that are 'suspicious' by checking for outliers in 
    the input space. A row is suspicious if its input values are 
    statistically distant from the rest of the population, as measured 
    by the xdist (distance across input columns) to the median row.
    """
    # Create a table to handle normalization and distance
    data = [header] + rows
    tbl = ezr.Tbl(data)
    
    # Calculate the median row (the centroid of the input space)
    # ezr.mids(tbl) returns a dictionary of column medians
    median_row = ezr.mids(tbl)
    
    # Calculate the xdist of every row to the median row
    # Rows with high xdist are outliers in the input space
    outlier_scores = []
    for i, row in enumerate(rows):
        d = ezr.xdist(tbl, row, median_row)
        outlier_scores.append((d, i))
    
    # Sort by distance descending (most suspicious first)
    outlier_scores.sort(key=lambda x: x[0], reverse=True)
    
    # Return the indices of the top 5% most suspicious rows
    limit = max(1, len(rows) // 20)
    return [item[1] for item in outlier_scores[:limit]]

import ezr
import random

def race(header, rows, label):
    """
    Compares three strategies: random, greedy (nearest to best), 
    and the tree-based ranker.
    """
    # Shared budget management
    labeled_cache = {}
    
    def oracle_wrapper(idx):
        if idx not in labeled_cache:
            if len(labeled_cache) >= 50:
                raise Exception("Budget exceeded")
            labeled_cache[idx] = label(idx)
        return labeled_cache[idx]

    # Helper to update row with true values
    def reveal(idx):
        true_vals = oracle_wrapper(idx)
        for j, val in enumerate(true_vals):
            if header[j].endswith(("-", "+")):
                rows[idx][j] = val
        return rows[idx]

    # Strategy 1: Random
    def strategy_random(budget):
        indices = random.sample(range(len(rows)), budget)
        for i in indices:
            reveal(i)
        # Find best among labeled
        return min(indices, key=lambda i: ezr.ydist(tbl, rows[i]))

    # Strategy 2: Greedy (Nearest to current best)
    def strategy_greedy(budget):
        # Start with one random
        idx = random.randint(0, len(rows) - 1)
        reveal(idx)
        best_idx = idx
        
        for _ in range(budget - 1):
            best_row = rows[best_idx]
            # Find row closest to current best in input space
            unlabeled = [i for i in range(len(rows)) if i not in labeled_cache]
            if not unlabeled: break
            next_idx = min(unlabeled, key=lambda i: ezr.xdist(tbl, rows[i], best_row))
            reveal(next_idx)
            if ezr.ydist(tbl, rows[next_idx]) < ezr.ydist(tbl, rows[best_idx]):
                best_idx = next_idx
        return best_idx

    # Strategy 3: Tree-based (using ezr.acquire)
    def strategy_tree(budget):
        # We need a local table for acquire to work correctly
        local_tbl = ezr.clone(tbl, [rows[i] for i in range(len(rows)) if i in labeled_cache or random.random() < 0.1])
        # acquire expects to label rows. We wrap the oracle.
        def local_oracle(r):
            # Find index of row object
            for i, row in enumerate(rows):
                if row is r:
                    reveal(i)
                    return rows[i]
            return r
        
        labeled = ezr.acquire(tbl, cap=len(labeled_cache) + budget, label=local_oracle)
        return min(range(len(rows)), key=lambda i: ezr.ydist(tbl, rows[i]))

    # Setup
    tbl = ezr.Tbl([header] + rows)
    
    # Run races
    results = {}
    strategies = {"random": strategy_random, "greedy": strategy_greedy, "tree": strategy_tree}
    
    # Reset rows to masked state for each strategy
    for name, func in strategies.items():
        for r in rows:
            for j, h in enumerate(header):
                if h.endswith(("-", "+")): r[j] = "?"
        labeled_cache.clear()
        results[name] = func(16) # 16 * 3 = 48 < 50

    # Determine winner based on ydist
    winner = min(results, key=lambda n: ezr.ydist(tbl, rows[results[n]]))
    
    return {"winner": winner, "row": results[winner], "tried": list(strategies.keys())}

import ezr
import random

def guess(header, train, test):
    """
    Estimates the first goal column (Cost-) for each test row.
    To account for the "changing requirements" scenario, we split the training
    data into two halves: the first half represents stable requirements,
    and the second half represents the "twice as fast" changing requirements.
    We weight the second half more heavily when building the tree.
    """
    # Identify the index of the first goal column (Cost-)
    goal_idx = -1
    for i, name in enumerate(header):
        if name.endswith("-"):
            goal_idx = i
            break
            
    # Split training data to simulate the shift in requirements
    # The second half is treated as the "new" environment
    mid = len(train) // 2
    part1 = train[:mid]
    part2 = train[mid:]
    
    # Weight the second half by duplicating it to reflect the "faster change"
    weighted_train = part1 + (part2 * 2)
    
    # Create a table for the weighted training data
    data = [header] + weighted_train
    tbl = ezr.Tbl(data)
    
    # Build a tree on the weighted training data
    root = ezr.tree(tbl, tbl.rows)
    
    results = []
    for row in test:
        # Walk the tree to find the leaf for this test row
        node = ezr.leaf(root, row)
        
        # Calculate the mean of the goal column in this leaf
        if node.rows:
            vals = [r[goal_idx] for r in node.rows]
            avg = sum(vals) / len(vals)
            results.append(avg)
        else:
            # Fallback to global mean if leaf is empty
            results.append(tbl.cols[goal_idx].mu)
            
    return results

import ezr
import random

def certify(xs, ys):
    """
    Returns True if the distributions are statistically indistinguishable,
    False if they are significantly different.
    Uses the same logic as ezr.same, which checks for statistical 
    significance using Cliff's Delta and the Kolmogorov-Smirnov test.
    """
    return ezr.same(xs, ys)

def analyze():
    """
    Compares the performance (ydist) of the first half of the data 
    (pre-change) vs the second half (post-change).
    """
    filename = "pom3a.n1000.train.csv"
    data = list(ezr.csv(filename))
    header = data[0]
    rows = data[1:]
    
    # Create a table to calculate ydist
    tbl = ezr.Tbl(data)
    
    # Split data into two halves
    mid = len(rows) // 2
    pre_change = rows[:mid]
    post_change = rows[mid:]
    
    # Calculate ydist for each group
    pre_ydists = [ezr.ydist(tbl, r) for r in pre_change]
    post_ydists = [ezr.ydist(tbl, r) for r in post_change]
    
    # Perform the certification
    is_same = certify(pre_ydists, post_ydists)
    
    print("--- Analysis: Did the results shift after the spring change? ---")
    print(f"Pre-change sample size:  {len(pre_ydists)}")
    print(f"Post-change sample size: {len(post_ydists)}")
    print(f"Pre-change avg ydist:    {sum(pre_ydists)/len(pre_ydists):.4f}")
    print(f"Post-change avg ydist:   {sum(post_ydists)/len(post_ydists):.4f}")
    print(f"Are the distributions statistically the same? {is_same}")
    
    if not is_same:
        print("Conclusion: The results have shifted significantly.")

def flow(header, row, seen):
    """
    Decide whether to approve a row immediately.
    We monitor for performance degradation. If the current row's ydist
    is significantly worse than the recent trend (the average ydist 
    of the last 50 rows), we return False to signal a potential issue.
    Otherwise, we return True.
    """
    if not seen:
        return True
    
    # Create a table to handle normalization and goal distance
    # We use the full history to establish the global context for normalization
    data = [header] + seen
    tbl = ezr.Tbl(data)
    
    # Calculate ydist for the current row
    current_score = ezr.ydist(tbl, row)
    
    # Calculate the average ydist of the last 50 rows (the recent trend)
    recent_rows = seen[-50:]
    recent_scores = [ezr.ydist(tbl, r) for r in recent_rows]
    avg_recent = sum(recent_scores) / len(recent_scores)
    
    # If the current row is significantly worse (higher ydist) than the 
    # recent average (e.g., 20% worse), we flag it as "going bad".
    # We return False to indicate this row should not be approved.
    if current_score > (avg_recent * 1.2):
        return False
        
    return True

import ezr
import random

def blame(header, rows):
    """
    Identifies which input columns matter most to cost by comparing the 
    distribution of inputs in the 'worst' rows (high ydist) vs the 'best' rows.
    Columns that show the largest statistical difference are the ones that 
    'matter most'.
    """
    # Create a table to handle normalization and distance
    data = [header] + rows
    tbl = ezr.Tbl(data)
    
    # Calculate ydist for all rows to rank them
    scored = [(ezr.ydist(tbl, r), r) for r in rows]
    scored.sort(key=lambda x: x[0])
    
    # Define 'best' and 'worst' as the top and bottom 25%
    n = len(rows) // 4
    best = [x[1] for x in scored[:n]]
    worst = [x[1] for x in scored[-n:]]
    
    # For each input column, measure the difference in distribution
    # between the best and worst rows.
    differences = []
    for col in tbl.x:
        b_vals = [r[col.at] for r in best if r[col.at] != "?"]
        w_vals = [r[col.at] for r in worst if r[col.at] != "?"]
        
        if not b_vals or not w_vals:
            continue
            
        # Use the same logic as ezr.same to check if distributions are different.
        # If they are NOT the same, the column is a candidate for 'mattering'.
        # We rank by the magnitude of the difference (using standard deviation 
        # or simple mean difference for numeric columns).
        if not ezr.same(sorted(b_vals), sorted(w_vals)):
            # Calculate a simple effect size: difference in means
            b_mid = sum(b_vals) / len(b_vals)
            w_mid = sum(w_vals) / len(w_vals)
            differences.append((abs(b_mid - w_mid), col.txt))
            
    # Sort by the magnitude of the difference
    differences.sort(key=lambda x: x[0], reverse=True)
    
    # Return the names of the top 2 most influential columns
    return [item[1] for item in differences[:2]]

import ezr
import random

def justify(header, rows, row):
    """
    Returns (p, reason): p is the fraction of rows that 'row' is better than,
    and reason is a sentence explaining why 'row' is better or worse than 
    the average configuration.
    """
    # Create a table to handle normalization and distance
    data = [header] + rows
    tbl = ezr.Tbl(data)
    
    # Estimate the goals for the input row using the tree-based ranker
    # The ranker returns a function that estimates ydist
    ranker_func = ezr.ranker(tbl)
    estimated_ydist = ranker_func(row)
    
    # Calculate ydist for all existing rows to compare
    all_ydists = [ezr.ydist(tbl, r) for r in rows]
    
    # p is the fraction of rows that have a higher (worse) ydist than our estimate
    # (i.e., our row is better than them)
    better_than_count = sum(1 for d in all_ydists if d > estimated_ydist)
    p = better_than_count / len(rows)
    
    # Construct a reason based on the performance rank
    if p > 0.7:
        reason = f"This setup is highly efficient, outperforming {int(p*100)}% of configurations."
    elif p > 0.4:
        reason = f"This setup performs reasonably well, better than {int(p*100)}% of configurations."
    else:
        reason = f"This setup is suboptimal, performing better than only {int(p*100)}% of configurations."
    
    return p, reason

import ezr
import random

def choose(header, rows, label):
    """
    Finds the best setup for a large, high-stakes project.
    We use the 'acquire' strategy from ezr to find the best configuration
    within a budget of 50 labels. We avoid the 'holdout' function as it 
    internally uses sets that conflict with the list-based row structure.
    """
    # Create a table object to manage the data.
    # We use a copy of the rows to avoid modifying the original list structure.
    tbl = ezr.Tbl([header] + rows)
    
    # We need to track which rows have been labeled to ensure we don't 
    # exceed the budget and to correctly map the oracle results.
    labeled_cache = {}

    def oracle_wrapper(row):
        # Find the index of the row in the original list
        idx = -1
        for i, r in enumerate(rows):
            if r is row:
                idx = i
                break
        
        if idx != -1 and idx not in labeled_cache:
            # Get the true values from the oracle
            true_row = label(idx)
            labeled_cache[idx] = true_row
            # Update the row with true values
            for j, val in enumerate(true_row):
                row[j] = val
            # Update the table's column stats with the new information
            ezr.addCols(tbl, row)
        return row

    # Perform acquisition using the acquire function from ezr.
    # We set the budget to 50.
    final_rows = ezr.acquire(tbl, cap=50, label=oracle_wrapper)
    
    # After acquisition, find the best row among the ones we have labeled.
    # The best row is the one with the lowest ydist.
    best_row = min(final_rows, key=lambda r: ezr.ydist(tbl, r))
    
    # Return the index of the best row
    for i, r in enumerate(rows):
        if r is best_row:
            return i
    return 0
