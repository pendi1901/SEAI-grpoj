import ezr
import random

def flow(header, row, seen):
    """
    Decide whether to approve a row immediately.
    We monitor for performance degradation. If the current row's ydist
    is significantly worse than the recent trend (the average ydist 
    of the last 50 rows), we return False to signal a potential issue.
    Otherwise, we return True.
    """
    if not seen:
        return True
    
    # Create a table to handle normalization and goal distance
    # We use the full history to establish the global context for normalization
    data = [header] + seen
    tbl = ezr.Tbl(data)
    
    # Calculate ydist for the current row
    current_score = ezr.ydist(tbl, row)
    
    # Calculate the average ydist of the last 50 rows (the recent trend)
    recent_rows = seen[-50:]
    recent_scores = [ezr.ydist(tbl, r) for r in recent_rows]
    avg_recent = sum(recent_scores) / len(recent_scores)
    
    # If the current row is significantly worse (higher ydist) than the 
    # recent average (e.g., 20% worse), we flag it as "going bad".
    # We return False to indicate this row should not be approved.
    if current_score > (avg_recent * 1.2):
        return False
        
    return True

if __name__ == "__main__":
    # Load the data
    filename = "pom3a.n1000.train.csv"
    data = list(ezr.csv(filename))
    header = data[0]
    rows = data[1:]
    
    # Shuffle to simulate random arrival
    random.seed(1234567891)
    random.shuffle(rows)
    
    seen = []
    flagged_count = 0
    
    for row in rows:
        if not flow(header, row, seen):
            flagged_count += 1
        seen.append(row)
    
    print("--- Performance Monitoring ---")
    print(f"Total rows processed: {len(rows)}")
    print(f"Rows flagged as 'going bad': {flagged_count}")
    print("The system monitors the recent trend of performance (ydist).")
    print("Rows that deviate significantly from the recent average are flagged.")
    
    exit(0)
