```python
import ezr
import random

def remember(header, rows):
    """
    Finds the top 5% of rows that perform best across all goals.
    A row is better when its goals are closer to their best (low for '-', high for '+').
    """
    # Create a table object to leverage ezr's normalization and distance logic
    # We construct the table by combining the header and the rows
    data = [header] + rows
    tbl = ezr.Tbl(data)
    
    # Calculate ydist for every row. 
    # ydist in ezr calculates the distance of a row's goals from the ideal.
    # Lower ydist is better.
    scored_rows = []
    for i, row in enumerate(rows):
        d = ezr.ydist(tbl, row)
        scored_rows.append((d, i))
    
    # Sort by distance (ascending, since lower is better)
    scored_rows.sort(key=lambda x: x[0])
    
    # Return the indices of the top 5%
    limit = max(1, len(rows) // 20)
    return [item[1] for item in scored_rows[:limit]]

if __name__ == "__main__":
    # Load the data
    filename = "pom3a.n1000.train.csv"
    data = list(ezr.csv(filename))
    header = data[0]
    rows = data[1:]
    
    # Get the best indices
    best_indices = remember(header, rows)
    
    # Print a summary for the customer
    print(f"Analysis complete. Found {len(best_indices)} top-performing setups.")
    print(f"{'Index':<10} | {'Goal Distance (ydist)':<20}")
    print("-" * 35)
    
    # Calculate table for ydist reference
    tbl = ezr.Tbl(data)
    for idx in best_indices:
        dist = ezr.ydist(tbl, rows[idx])
        print(f"{idx:<10} | {dist:.4f}")
        
    exit(0)
```
