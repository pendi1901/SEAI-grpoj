import ezr
import random

def race(header, rows, label):
    """
    Compares three strategies: random, greedy (nearest to best), 
    and the tree-based ranker.
    """
    # Shared budget management
    labeled_cache = {}
    
    def oracle_wrapper(idx):
        if idx not in labeled_cache:
            if len(labeled_cache) >= 50:
                raise Exception("Budget exceeded")
            labeled_cache[idx] = label(idx)
        return labeled_cache[idx]

    # Helper to update row with true values
    def reveal(idx):
        true_vals = oracle_wrapper(idx)
        for j, val in enumerate(true_vals):
            if header[j].endswith(("-", "+")):
                rows[idx][j] = val
        return rows[idx]

    # Strategy 1: Random
    def strategy_random(budget):
        indices = random.sample(range(len(rows)), budget)
        for i in indices:
            reveal(i)
        # Find best among labeled
        return min(indices, key=lambda i: ezr.ydist(tbl, rows[i]))

    # Strategy 2: Greedy (Nearest to current best)
    def strategy_greedy(budget):
        # Start with one random
        idx = random.randint(0, len(rows) - 1)
        reveal(idx)
        best_idx = idx
        
        for _ in range(budget - 1):
            best_row = rows[best_idx]
            # Find row closest to current best in input space
            unlabeled = [i for i in range(len(rows)) if i not in labeled_cache]
            if not unlabeled: break
            next_idx = min(unlabeled, key=lambda i: ezr.xdist(tbl, rows[i], best_row))
            reveal(next_idx)
            if ezr.ydist(tbl, rows[next_idx]) < ezr.ydist(tbl, rows[best_idx]):
                best_idx = next_idx
        return best_idx

    # Strategy 3: Tree-based (using ezr.acquire)
    def strategy_tree(budget):
        # We need a local table for acquire to work correctly
        local_tbl = ezr.clone(tbl, [rows[i] for i in range(len(rows)) if i in labeled_cache or random.random() < 0.1])
        # acquire expects to label rows. We wrap the oracle.
        def local_oracle(r):
            # Find index of row object
            for i, row in enumerate(rows):
                if row is r:
                    reveal(i)
                    return rows[i]
            return r
        
        labeled = ezr.acquire(tbl, cap=len(labeled_cache) + budget, label=local_oracle)
        return min(range(len(rows)), key=lambda i: ezr.ydist(tbl, rows[i]))

    # Setup
    tbl = ezr.Tbl([header] + rows)
    
    # Run races
    results = {}
    strategies = {"random": strategy_random, "greedy": strategy_greedy, "tree": strategy_tree}
    
    # Reset rows to masked state for each strategy
    for name, func in strategies.items():
        for r in rows:
            for j, h in enumerate(header):
                if h.endswith(("-", "+")): r[j] = "?"
        labeled_cache.clear()
        results[name] = func(16) # 16 * 3 = 48 < 50

    # Determine winner based on ydist
    winner = min(results, key=lambda n: ezr.ydist(tbl, rows[results[n]]))
    
    return {"winner": winner, "row": results[winner], "tried": list(strategies.keys())}

if __name__ == "__main__":
    filename = "pom3a.n1000.train.csv"
    data = list(ezr.csv(filename))
    header = data[0]
    rows = data[1:]
    
    ground_truth = [row[:] for row in rows]
    
    def label_func(i):
        return ground_truth[i]
        
    result = race(header, rows, label_func)
    print(f"Race Results: Winner is {result['winner']} strategy.")
    print(f"Best row found at index {result['row']}.")
    print(f"Strategies compared: {', '.join(result['tried'])}")
    exit(0)
