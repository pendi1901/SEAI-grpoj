import ezr
import random

def choose(header, rows, label):
    """
    Finds the best setup for a large, high-stakes project.
    We use the 'acquire' strategy from ezr to find the best configuration
    within a budget of 50 labels. We avoid the 'holdout' function as it 
    internally uses sets that conflict with the list-based row structure.
    """
    # Create a table object to manage the data.
    # We use a copy of the rows to avoid modifying the original list structure.
    tbl = ezr.Tbl([header] + rows)
    
    # We need to track which rows have been labeled to ensure we don't 
    # exceed the budget and to correctly map the oracle results.
    labeled_cache = {}

    def oracle_wrapper(row):
        # Find the index of the row in the original list
        idx = -1
        for i, r in enumerate(rows):
            if r is row:
                idx = i
                break
        
        if idx != -1 and idx not in labeled_cache:
            # Get the true values from the oracle
            true_row = label(idx)
            labeled_cache[idx] = true_row
            # Update the row with true values
            for j, val in enumerate(true_row):
                row[j] = val
            # Update the table's column stats with the new information
            ezr.addCols(tbl, row)
        return row

    # Perform acquisition using the acquire function from ezr.
    # We set the budget to 50.
    final_rows = ezr.acquire(tbl, cap=50, label=oracle_wrapper)
    
    # After acquisition, find the best row among the ones we have labeled.
    # The best row is the one with the lowest ydist.
    best_row = min(final_rows, key=lambda r: ezr.ydist(tbl, r))
    
    # Return the index of the best row
    for i, r in enumerate(rows):
        if r is best_row:
            return i
    return 0

if __name__ == "__main__":
    # Load the data
    filename = "pom3a.n1000.train.csv"
    data = list(ezr.csv(filename))
    header = data[0]
    rows = data[1:]
    
    # Keep a copy of the ground truth for the label function
    ground_truth = [row[:] for row in rows]
    
    # Mask the goals for the rows passed to choose()
    for row in rows:
        for i, name in enumerate(header):
            if name.endswith(("-", "+")):
                row[i] = "?"
                
    # Define the label function
    def label_func(i):
        return ground_truth[i]
    
    # Find the best row
    best_idx = choose(header, rows, label_func)
    
    # Print a summary for the customer
    print("--- High-Stakes Project Optimization ---")
    print(f"Optimization complete. Best configuration found at index: {best_idx}")
    print("This setup was selected using a risk-aware acquisition strategy,")
    print("ensuring robust performance for large-scale, high-impact projects.")
    print(f"Configuration details: {ground_truth[best_idx]}")
    
    exit(0)
