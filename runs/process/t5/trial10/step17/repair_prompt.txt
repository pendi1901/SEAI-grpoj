def say(x: Any, p: int = 2) -> str:
  "Floats get P decimals; dicts list their keys."
  if isinstance(x,float): x= f"{x:.{p}f}".rstrip("0").rstrip(".")
  if isinstance(x, dict):
    x= "{"+", ".join(f"{k}: {say(v,p)}" for k,v in x.items())+"}"
  return str(x)

class o(dict):
  "A dict you can poke with a dot."
  __getattr__, __setattr__ = dict.__getitem__, dict.__setitem__
  __repr__ = say

def atom(s: str) -> ATOM:
  "'22' -> 22. '3.1' -> 3.1. 'True' -> True. 'x' -> 'x'."
  for fn in (int, float):
    try: return fn(s)
    except ValueError: pass
  s = s.strip()
  return {"True": True, "False": False}.get(s, s)

def csv(file: str) -> Iterator[ROW]:
  "ROWS of FILE, each cell coerced."
  with open(os.path.expanduser(file),  # -sig drops any BOM
           encoding="utf-8-sig") as f:
    for line in f:
      if line.strip(): yield [atom(s) for s in line.split(",")]

def Col(txt: str = " ", at: int = 0) -> COL:
  "Factory that returns NUM or SYM."
  return Num(txt,at) if txt[0].isupper() else Sym(txt,at)

class Num:
  "Place to summarize stream of numbers."
  __slots__ = ("at", "txt", "n", "mu", "m2", "sd", "goal")
  def __init__(self, txt: str = " ", at: int = 0) -> None:
    self.at, self.txt, self.n = at, txt, 0
    self.mu = self.m2 = self.sd = 0
    self.goal = 0 if txt[-1] == "-" else 1
  __repr__ = lambda self: say({k: getattr(self,k)
                               for k in self.__slots__})

class Sym:
  "Place to summarize stream of Symbols.  The class is the tag."
  __slots__ = ("at", "txt", "n", "has")
  def __init__(self, txt: str = " ", at: int = 0) -> None:
    self.at, self.txt, self.n, self.has = at, txt, 0, {}
  __repr__ = lambda self: say({k: getattr(self,k)
                               for k in self.__slots__})

def add(col: COL, v: ATOM, inc: int = 1) -> None:
  "Show V to COL.  INC=-1 takes it away.  `?` changes nothing."
  if v != "?":
    col.n += inc
    if type(col) is Sym:
      col.has[v] = col.has.get(v, 0) + inc
    else:
      d = v - col.mu                      # Welford, for `div`
      col.mu += inc * d / max(1, col.n)
      col.m2  = max(0, col.m2 + inc * d * (v - col.mu))
      col.sd  = 0 if col.n < 2 else (col.m2/(col.n-1))**.5

def mid(col: COL) -> ATOM:
  "Middle: the mean, or the most common symbol; ? if none."
  return (max(col.has, key=col.has.get) if col.has else "?"
          ) if type(col) is Sym else col.mu

def mids(tbl: TBL) -> MIDS:
  "Every column's middle, cached until the next addCols."
  tbl.mids = tbl.mids or {at: mid(c) for at,c in tbl.cols.items()}
  return tbl.mids

def div(col: COL) -> float:
  "Spread: entropy, or standard deviation."
  return (-sum(v/col.n*log2(v/col.n) for v in col.has.values())
          if type(col) is Sym else col.sd)  # sd kept by `add`

def norm(col: COL, v: ATOM) -> ATOM:
  "To 0..1, by the logistic curve.  Symbols do not scale."
  if type(col) is Sym: return v
  z = max(-3, min(3, (v - col.mu) / (1e-32 + div(col))))
  return 1 / (1 + exp(-1.7 * z))

#-- tables -------------------------------------------------
def Tbl(src: Iterable[ROW]) -> TBL:
  "Header names the columns: X skip, +- goal, ! klass."
  src = iter(src)
  tbl = Cols(o(rows=[], cols={}, x=[], y=[], mids=None,
               names=next(src)))
  for row in src: addRow(tbl, row)
  return tbl

def Cols(tbl: TBL) -> TBL:
  "Create column roles."
  for at, s in enumerate(tbl.names):
    if s[-1] == "X": continue            # skip me entirely
    tbl.cols[at] = Col(s,at)
    (tbl.y if s[-1] in "+-!" else tbl.x).append(tbl.cols[at])
  return tbl

def addRow(tbl: TBL, row: ROW) -> ROW:
  "Keep ROW, and show it to my columns."
  tbl.rows += [row]
  return addCols(tbl, row)

def addCols(tbl: TBL, row: ROW, inc: int = 1) -> ROW:
  "Show ROW to my columns, without keeping it."
  tbl.mids = None                 # a new row moves the middles
  for at, col in tbl.cols.items(): add(col, row[at], inc)
  return row

def clone(tbl: TBL, rows: ROWS = None) -> TBL:
  "An empty copy of TBL, plus ROWS."
  return Tbl([tbl.names] + (rows or []))

def dist(vs: Iterable[QTY], n: int) -> float:
  "Root mean square of N gaps."
  return sqrt(sum(v*v for v in vs) / n)

def ydist(tbl: TBL, row: ROW) -> float:
  "How far ROW's goals are from the best they could be."
  return dist((abs(norm(c, row[c.at]) - c.goal)
                    for c in tbl.y), len(tbl.y))

def gap(col: COL, a: ATOM, b: ATOM) -> QTY:
  "Distance between two values of one column."
  if a == "?" or b == "?": return 1    # unknown = far
  return a!=b if type(col) is Sym else abs(norm(col,a)-norm(col,b))

def xdist(tbl: TBL, r1: ROW, r2: ROW | MIDS) -> float:
  "How far apart two rows are, over the x columns."
  return dist((gap(c, r1[c.at], r2[c.at])
                    for c in tbl.x), len(tbl.x))

def oracle(row: ROW) -> ROW:
  """Labeller. Rows here arrive with ys, so nothing to do.
  Real oracles fill in any missing y cells (and can close
  over a TBL to know which cells those are)."""
  return row

def acquire(tbl: TBL, cap: int | None = None,
            label: ORACLE = oracle) -> ROWS:
  "Label near the best rows, far from the rest."
  cap  = cap or the.Stop
  todo = random.sample(tbl.rows, len(tbl.rows))[:the.Few]
  both = clone(tbl, [label(todo.pop())
                     for _ in range(the.Start)])
  both.rows.sort(key=lambda r: ydist(both, r))
  n    = int(sqrt(the.Start))
  best,rest = clone(tbl,both.rows[:n]), clone(tbl,both.rows[n:])
  while todo and len(both.rows) < cap:
    cb, cr = mids(best), mids(rest)    # two centroids
    todo.sort(key=lambda z: xdist(tbl,z,cr) - xdist(tbl,z,cb))
    addRow(best, addRow(both, label(todo.pop())))
    best.rows.sort(key=lambda z: ydist(both, z))  # no y leak
    if len(best.rows) > sqrt(len(both.rows)): # best stays small
      addRow(rest, addCols(best, best.rows.pop(), -1))
  return both.rows

def grabs(tbl: TBL, cap: int | None = None,
          label: ORACLE = oracle) -> ROWS:
  "The straw man: same budget, taken at random."
  cap = cap or the.Stop
  return [label(r) for r in
          random.sample(tbl.rows, min(cap, len(tbl.rows)))]

def cut(tbl: TBL, rows: ROWS) -> tuple[int,ATOM,GO]|None:
  "The (at, v, go) whose two sides have the tightest ys."
  ys = [ydist(tbl, r) for r in rows]
  out, least = None, 1e30
  for col in tbl.x:
    for v, go in candidates(col, rows):
      a, b = Num(), Num()   # one pass, so one `go` per row
      for r,y in zip(rows,ys): add(a if go(r) else b, y)
      if a.n >= the.Leaf and b.n >= the.Leaf:
        score = (div(a)*a.n + div(b)*b.n) / (a.n + b.n)
        if score < least: out, least = (col.at, v, go), score
  return out

def candidates(col: COL, rows: ROWS) -> Iterator[tuple[ATOM,GO]]:
  "What splits to try: one per symbol, or CUTS per num."
  at = col.at
  if type(col) is Sym:
    for v in sorted({r[at] for r in rows if r[at] != "?"}):
      yield v, lambda r,v=v,at=at: r[at] == v
  else:                # by rank, not value: THESE rows are dense
    xs = sorted(r[at] for r in rows if r[at] != "?")
    k  = max(1, len(xs)//the.Cuts)     # cuts, spaced by rank
    for v in xs[k-1:-1:k]:
      yield v, (lambda r,v=v,at=at: r[at] != "?" and r[at] <= v)

def tree(tbl: TBL, rows: ROWS) -> NODE:
  "Split while a split keeps both sides big enough."
  node = o(rows=rows, at=None, v=None, go=None, kids=[],
           mu=sum(ydist(tbl,r) for r in rows)/len(rows))
  if len(rows) > the.Leaf and (found := cut(tbl, rows)):
    at, v, go = found
    yes, no = [], []
    for r in rows: (yes if go(r) else no).append(r)
    node.at, node.v, node.go = at, v, go # cut kepts sides big
    node.kids = [tree(tbl, yes), tree(tbl, no)]
  return node

def leaf(node: NODE, row: ROW) -> NODE:
  "Walk ROW down to its leaf."
  while node.kids: node = node.kids[0 if node.go(row) else 1]
  return node

def leafs(node: NODE) -> list[NODE]:
  "Return leaves of this tree."
  return [x for k in node.kids for x in leafs(k)] or [node]

def show(tbl: TBL, node: NODE, pre: str | None = None,
         edge: str = "") -> None:
  "Tree on the left; ydist and n on the right."
  ls = sorted(leafs(node), key=lambda z: z.mu)
  def walk(z: NODE, pre: str | None, edge: str) -> None:
    m = "+" if z is ls[0] else "-" if z is ls[-1] else " "
    print(f"{m} {round(100*z.mu):>4} {len(z.rows):>5}   "
          f"{(pre or '')+edge}".rstrip())
    op = "=" if type(tbl.cols[z.at or 0]) is Sym else "<="
    no = "!=" if op == "=" else ">"
    for kid, oper in zip(z.kids, [op, no]):
      walk(kid, "" if pre is None else pre + "|  ",
           f"{tbl.names[z.at]} {oper} {say(z.v)}")
  print(" ydist     n")
  walk(node, pre, edge)



#-- rig ----------------------------------------------------
# Calls core, acquire, and one RANKER -- tree's, by default.
def ranker(lab: TBL) -> Callable[[ROW], float]:
  "Grow a tree on LAB; score any row by its leaf's mean ydist."
  node = tree(lab, lab.rows)
  return lambda r: leaf(node, r).mu

def wins(tbl: TBL) -> Callable[[ROW], float]:
  "100 at the best row, 0 at an average one."
  ys = sorted(ydist(tbl, r) for r in tbl.rows)
  lo, avg = ys[0], sum(ys)/len(ys)
  return lambda r: max(-100, min(100, 100*(1 - (ydist(tbl,r)-lo)
                                            / (avg-lo+1e-32))))

def holdout(tbl: TBL, pick: PICKER = acquire,
            rank: RANKER = ranker,
            label: ORACLE = oracle) -> ROW:
  "Train on half; of CHECK guesses on the rest, pick best."
  rows  = random.sample(tbl.rows, len(tbl.rows))
  n     = len(rows)//2
  tr    = clone(tbl, rows[:n][:the.Few])
  lab   = clone(tbl, pick(tr, the.Stop - the.Check, label))
  top   = sorted(rows[n:], key=rank(lab))
  return min(top[:the.Check], key=lambda r: ydist(lab, label(r)))

# A leaf: it calls core, and nothing here calls it.
def adds(vs: Iterable[ATOM], col: COL | None = None) -> COL:
  "Every item of VS into COL."
  col = Num() if col is None else col
  for v in vs: add(col, v)
  return col

def cohen(xs: NUMS, ys: NUMS, d: float = 0.35,
          eps: float = 0) -> bool:
  "Are the means within D pooled standard deviations?"
  a, b = adds(xs), adds(ys)
  pool = ((a.n-1)*div(a)**2 + (b.n-1)*div(b)**2)/(a.n+b.n-2)
  return abs(a.mu - b.mu) <= max(eps, d * sqrt(pool))

def cliffs(xs: NUMS, ys: NUMS, d: float = 0.197) -> bool:
  "Sorted in.  Is the rank imbalance small enough?"
  gt = lt = j = k = 0
  for x in xs:
    while j < len(ys) and ys[j] <  x: j += 1; k = j
    while k < len(ys) and ys[k] <= x: k += 1
    gt += j; lt += len(ys) - k
  return abs(gt - lt) / (len(xs) * len(ys)) <= d

def ks(xs: NUMS, ys: NUMS, a: float = 1.36) -> bool:
  "Sorted in.  95% Kolmogorov-Smirnov."
  n, m, i, j, d = len(xs), len(ys), 0, 0, 0
  while i < n and j < m:
    v = min(xs[i], ys[j])
    while i < n and xs[i] <= v: i += 1
    while j < m and ys[j] <= v: j += 1
    d = max(d, abs(i/n - j/m))
  return d <= a * sqrt((n + m) / (n * m))

def same(xs: NUMS, ys: NUMS, eps: float = 0) -> bool:
  """Indistinguishable.  Always cliffs and ks; EPS adds cohen, and
  doubles as its effect size (eps=0 skips that third test)."""
  xs, ys = sorted(xs), sorted(ys)
  return (cliffs(xs,ys) and ks(xs,ys)
          and (not eps or cohen(xs,ys,d=eps)))

import ezr
import random

def remember(header, rows):
    """
    Identify the top 5% of rows that perform best across all goals.
    Goals ending in '-' are minimized, '+' are maximized.
    """
    # Create a table object using the provided header and rows
    # The ezr.Tbl function expects an iterable of rows including the header
    data = [header] + rows
    tbl = ezr.Tbl(data)
    
    # Calculate the ydist for every row.
    # ydist measures how far a row's goals are from the ideal (0 for -, 1 for +).
    # Lower ydist is better.
    scored_rows = []
    for i, row in enumerate(rows):
        d = ezr.ydist(tbl, row)
        scored_rows.append((d, i))
    
    # Sort by distance (ascending, since lower is better)
    scored_rows.sort(key=lambda x: x[0])
    
    # Return the top 5% of rows (800 rows * 0.05 = 40 rows)
    limit = max(1, int(len(rows) * 0.05))
    return [item[1] for item in scored_rows[:limit]]

import ezr
import random

def guess(header, train, test):
    """
    Estimates the first goal column for each test row.
    Uses the tree-based ranker to find the leaf node for a test row,
    then returns the mean value of the first goal column in that leaf.
    """
    # Build a table from the training data
    data = [header] + train
    tbl = ezr.Tbl(data)
    
    # Build a tree on the training data
    node = ezr.tree(tbl, tbl.rows)
    
    # Identify the index of the first goal column
    # The first goal column is the first column in tbl.y
    goal_col = tbl.y[0]
    goal_idx = goal_col.at
    
    results = []
    for row in test:
        # Walk the tree to find the leaf for this test row
        leaf_node = ezr.leaf(node, row)
        
        # Calculate the mean of the first goal column in this leaf
        vals = [r[goal_idx] for r in leaf_node.rows if r[goal_idx] != "?"]
        if not vals:
            # Fallback to global mean if leaf is empty or has no data
            mean_val = sum(r[goal_idx] for r in tbl.rows) / len(tbl.rows)
        else:
            mean_val = sum(vals) / len(vals)
        results.append(mean_val)
        
    return results

if __name__ == "__main__":
    # Load the data
    filename = "pom3a.n1000.train.csv"
    data = list(ezr.csv(filename))
    header = data[0]
    rows = data[1:]
    
    # Split into train and test (80/20)
    random.seed(1234567891)
    random.shuffle(rows)
    n = int(len(rows) * 0.8)
    train = rows[:n]
    test_full = rows[n:]
    
    # Mask the goals in the test set to simulate "?"
    # Goals are the last 3 columns: Cost-, Score-, Idle-
    test_masked = []
    for row in test_full:
        new_row = list(row)
        new_row[-3] = "?"
        new_row[-2] = "?"
        new_row[-1] = "?"
        test_masked.append(new_row)
        
    # Get estimates for the first goal (Cost-)
    estimates = guess(header, train, test_masked)
    
    # Calculate actuals for comparison
    actuals = [r[-3] for r in test_full]
    
    # Print summary
    print(f"Estimated Cost- for {len(estimates)} test rows.")

def certify(xs, ys):
    """
    Returns True if the two distributions are statistically indistinguishable 
    (same apart from noise), False if they are significantly different.
    Uses the same() function from ezr.stats.
    """
    return ezr.same(xs, ys)

import ezr
import random

def flow(header, row, seen):
    """
    Decide whether to accept a row based on whether it is better than 
    the current best seen so far.
    """
    # If this is the first row, we have no baseline, so we accept it.
    if not seen:
        return True
    
    # Create a table to handle normalization and goal distance calculations
    # We use the header and the history of seen rows to define the context
    data = [header] + seen
    tbl = ezr.Tbl(data)
    
    # Calculate the quality of the current row and the best row seen so far
    # ydist returns 0 for ideal, higher for worse.
    current_ydist = ezr.ydist(tbl, row)
    best_ydist = min(ezr.ydist(tbl, r) for r in seen)
    
    # If the current row is better (lower ydist) than the best seen, accept it.
    # We use a small threshold to avoid accepting rows that are effectively identical.
    return current_ydist < (best_ydist * 0.95)

import ezr

def blame(header, rows):
    """
    Identify input columns that best explain why the worst rows are bad.
    We define 'worst' as the rows with the highest ydist.
    We compare the distribution of input values in the worst 20% 
    against the rest, looking for features that correlate with high ydist.
    """
    # Create table to calculate ydist
    data = [header] + rows
    tbl = ezr.Tbl(data)
    
    # Calculate ydist for all rows
    scored = []
    for row in rows:
        scored.append((ezr.ydist(tbl, row), row))
    
    # Sort by ydist descending to get the worst rows
    scored.sort(key=lambda x: x[0], reverse=True)
    
    # Split into worst 20% and the rest
    n_worst = max(1, len(rows) // 5)
    worst = [x[1] for x in scored[:n_worst]]
    rest = [x[1] for x in scored[n_worst:]]
    
    # For each input column, check if the distribution of values in 'worst'
    # is significantly different from 'rest' using the same() statistical test.
    # If they are different, the column is a candidate for 'blame'.
    blamed = []
    for col in tbl.x:
        at = col.at
        # Extract values for this column
        vals_worst = [r[at] for r in worst if r[at] != "?"]
        vals_rest = [r[at] for r in rest if r[at] != "?"]
        
        if not vals_worst or not vals_rest:
            continue
            
        # If they are NOT the same, this column contributes to the difference
        if not ezr.same(vals_worst, vals_rest):
            blamed.append(header[at])
            
    return blamed

def justify(header, rows, row):
    """

    and reason is a human-readable explanation.
    """

    data = [header] + rows
    tbl = ezr.Tbl(data)

    # We build a tree on all known rows to predict the performance of the input row
    ranker = ezr.ranker(tbl)
    estimated_ydist = ranker(row)
    
    # Calculate ydist for all known rows to establish a distribution
    all_ydists = [ezr.ydist(tbl, r) for r in rows]
    
    # Calculate p: fraction of rows that this row is better than
    # A row is better if its ydist is lower.
    # We count how many rows have a higher ydist than our estimate.
    better_than_count = sum(1 for d in all_ydists if d > estimated_ydist)
    p = better_than_count / len(rows)
    
    reason = (f"This setup is better than {int(p * 100)}% of all known configurations "
              f"based on its predicted performance across all goals.")
    
    return p, reason

import ezr
import random

def choose(header, rows, label):
    """
    Find the best row among the provided rows by using a budget of 50 labels.
    We use the 'acquire' strategy from ezr to intelligently select rows to label.
    """
    # Create a table object.
    # Note: rows currently have '?' for goals.
    data = [header] + rows
    tbl = ezr.Tbl(data)
    
    # The oracle function to pass to ezr.acquire.
    # The harness label(i) returns the full row (including goals).
    # We need to map the row object back to its index to call the harness.
    # We store the original index in the row object temporarily if needed,
    # but here we can use a lookup dictionary.
    row_to_idx = {id(r): i for i, r in enumerate(rows)}

    def oracle_wrapper(row):
        idx = row_to_idx[id(row)]
        real_row = label(idx)
        # Update the row in place so the TBL reflects the new data
        for i in range(len(real_row)):
            row[i] = real_row[i]
        return row

    # Use ezr.acquire to intelligently pick 50 rows to label.
    # ezr.acquire returns the rows that were labeled.
    labeled_rows = ezr.acquire(tbl, cap=50, label=oracle_wrapper)
    
    # After labeling, find the best row among all rows that were labeled.
    # We use ydist to determine the best.
    best_row = min(labeled_rows, key=lambda r: ezr.ydist(tbl, r))
    
    # Return the index of the best row.
    return row_to_idx[id(best_row)]

    for col_name, allowed in levers.items():
        if col_name not in header:
            continue
            
        col_idx = header.index(col_name)
        original_val = row[col_idx]
        
        # Generate candidates based on lever type
        candidates = []
        if isinstance(allowed, tuple):  # Numeric (low, high)
            low, high = allowed
            # Try 10 points in the range to increase chance of finding improvement
            for i in range(10):
                val = low + (high - low) * (i / 9)

                    candidates.append(val)
        else:  # Symbolic (list of strings)

                    candidates.append(val)

        for val in candidates:
            new_row = list(row)
            new_row[col_idx] = val
            
            # Use ydist to see if this improves the row
            d = ezr.ydist(tbl, new_row)
            if d < best_ydist:
                best_ydist = d
                best_change = {col_name: val}

import ezr

def spot(header, rows):
    """
    Identify rows that contain suspicious input values.
    We define 'suspicious' as rows that have an input value (X column)
    that is an outlier relative to the distribution of that column.
    We use the standard deviation (sd) from the Num class to detect
    values that are more than 3 standard deviations from the mean.
    """
    # Create a table to get column summaries
    data = [header] + rows
    tbl = ezr.Tbl(data)
    
    # Calculate suspiciousness score for each row
    # A row's score is the maximum number of standard deviations 
    # any of its input values is from the column mean.
    scores = []
    for i, row in enumerate(rows):
        max_z = 0
        for col in tbl.x:
            val = row[col.at]
            # Only check numeric columns
            if isinstance(col, ezr.Num) and isinstance(val, (int, float)):
                # Avoid division by zero
                sd = col.sd if col.sd > 0 else 1e-32
                z = abs(val - col.mu) / sd
                if z > max_z:
                    max_z = z
        scores.append((max_z, i))
    
    # Sort by suspiciousness descending
    scores.sort(key=lambda x: x[0], reverse=True)
    
    # Return indices of rows that are significantly suspicious (e.g., z > 3)
    # We return all rows that meet this criteria, sorted by most suspicious.
    return [i for z, i in scores if z > 3]

import ezr
import random

def race(header, rows, label):
    """
    Compare three strategies for finding the best row within a shared 50-label budget.
    Strategies:
    1. random: Picks rows at random.
    2. acquire: Uses the intelligent acquisition strategy from ezr.
    3. greedy: Picks rows that are closest to the current best-known centroid.
    """
    budget = 50
    labeled_cache = {}

    def oracle_wrapper(idx):
        if idx not in labeled_cache:
            labeled_cache[idx] = label(idx)
        return labeled_cache[idx]

    # Create a base table for normalization context
    tbl = ezr.Tbl([header] + rows)

    # Strategy 1: Random
    def strategy_random(budget_left):
        indices = random.sample(range(len(rows)), min(budget_left, len(rows)))
        for i in indices:
            oracle_wrapper(i)
        return indices

    # Strategy 2: Acquire
    def strategy_acquire(budget_left):
        # We use a local table to track the acquisition process
        # We need to ensure the oracle returns the full row
        def local_oracle(r):
            # Find index by matching row content (or object identity)
            for i, row in enumerate(rows):
                if row is r:
                    return oracle_wrapper(i)
            return r
        
        labeled_rows = ezr.acquire(tbl, cap=budget_left, label=local_oracle)
        # Map back to indices
        return [i for i, r in enumerate(rows) if r in labeled_rows]

    # Strategy 3: Greedy
    def strategy_greedy(budget_left):
        indices = []
        idx = random.randint(0, len(rows) - 1)
        indices.append(idx)
        oracle_wrapper(idx)
        
        while len(indices) < budget_left:
            # Find best labeled row
            best_idx = min(indices, key=lambda i: ezr.ydist(tbl, oracle_wrapper(i)))
            best_row = oracle_wrapper(best_idx)
            
            # Find unlabelled row closest to best_row
            unlabeled = [i for i in range(len(rows)) if i not in indices]
            if not unlabeled: break
            next_idx = min(unlabeled, key=lambda i: ezr.xdist(tbl, rows[i], best_row))
            oracle_wrapper(next_idx)
            indices.append(next_idx)
        return indices

    # Run the race
    strategies = {
        "random": strategy_random,
        "acquire": strategy_acquire,
        "greedy": strategy_greedy
    }
    
    budget_per = budget // len(strategies)
    results = {}
    
    for name, func in strategies.items():
        indices = func(budget_per)
        # Find best row found by this strategy
        best_i = min(indices, key=lambda i: ezr.ydist(tbl, oracle_wrapper(i)))
        results[name] = (best_i, ezr.ydist(tbl, oracle_wrapper(best_i)))

    winner = min(results, key=lambda n: results[n][1])
    
    return {
        "winner": winner,
        "row": results[winner][0],
        "tried": list(strategies.keys())
    }

import ezr
import random

def guess(header, train, test):
    """
    Estimates the first goal column for each test row.
    To account for the 'changing requirements' scenario, we weight the training
    data: rows that appear later in the training set are given more influence
    on the tree's leaf predictions, simulating the impact of increased dynamism.
    """
    # Build a table from the training data
    data = [header] + train
    tbl = ezr.Tbl(data)
    
    # Build a tree on the training data
    node = ezr.tree(tbl, tbl.rows)
    
    # Identify the index of the first goal column
    goal_col = tbl.y[0]
    goal_idx = goal_col.at
    
    # To simulate requirements changing twice as fast halfway through,
    # we calculate a weighted mean for each leaf.
    # Rows in the second half of the training set are weighted 2x.
    n_train = len(train)
    midpoint = n_train // 2
    
    results = []
    for row in test:
        leaf_node = ezr.leaf(node, row)
        
        # Calculate weighted mean of the first goal column in this leaf
        weighted_sum = 0
        total_weight = 0
        found = False
        
        for r in leaf_node.rows:
            # Find original index in training set to apply weight
            # We assume the order in leaf_node.rows matches the input order
            # or we can check identity
            try:
                idx = train.index(r)
                weight = 2.0 if idx >= midpoint else 1.0
                if r[goal_idx] != "?":
                    weighted_sum += r[goal_idx] * weight
                    total_weight += weight
                    found = True
            except ValueError:
                continue
        
        if not found or total_weight == 0:
            # Fallback to global mean
            mean_val = sum(r[goal_idx] for r in tbl.rows if r[goal_idx] != "?") / len(tbl.rows)
        else:
            mean_val = weighted_sum / total_weight
        results.append(mean_val)
        
    return results

import ezr

def flow(header, row, seen):
    """
    Warn the customer if the incoming row is 'bad'.
    We define 'bad' as a row that is significantly worse than the 
    historical performance baseline. We use the 95th percentile of 
    the 'seen' rows as a threshold for 'good' performance. If the 
    new row's ydist is significantly higher than this threshold, 
    we flag it as a potential warning (returning True).
    """
    if not seen:
        return False
    
    # Create a table to handle normalization and ydist calculations
    data = [header] + seen
    tbl = ezr.Tbl(data)
    
    # Calculate ydist for all seen rows
    seen_ydists = [ezr.ydist(tbl, r) for r in seen]

    # Calculate ydist for the new row
    current_ydist = ezr.ydist(tbl, row)
    
    # Return True if the row is significantly worse than the threshold
    return current_ydist > threshold

import ezr

def blame(header, rows):
    """
    Identify input columns that best explain why the worst rows are bad.
    We define 'worst' as the rows with the highest ydist.
    We compare the distribution of input values in the worst 20% 
    against the rest, looking for features that correlate with high ydist.
    """
    # Create table to calculate ydist
    data = [header] + rows
    tbl = ezr.Tbl(data)
    
    # Calculate ydist for all rows
    scored = []
    for row in rows:
        scored.append((ezr.ydist(tbl, row), row))
    
    # Sort by ydist descending to get the worst rows
    scored.sort(key=lambda x: x[0], reverse=True)
    
    # Split into worst 20% and the rest
    n_worst = max(1, len(rows) // 5)
    worst = [x[1] for x in scored[:n_worst]]
    rest = [x[1] for x in scored[n_worst:]]
    
    # For each input column, check if the distribution of values in 'worst'
    # is significantly different from 'rest' using the same() statistical test.
    # If they are different, the column is a candidate for 'blame'.
    blamed = []
    for col in tbl.x:
        at = col.at
        # Extract values for this column
        vals_worst = [r[at] for r in worst if r[at] != "?"]
        vals_rest = [r[at] for r in rest if r[at] != "?"]
        
        if not vals_worst or not vals_rest:
            continue
            
        # If they are NOT the same, this column contributes to the difference
        if not ezr.same(vals_worst, vals_rest):
            blamed.append(header[at])
            
    return blamed
