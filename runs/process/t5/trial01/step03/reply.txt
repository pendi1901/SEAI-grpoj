#!/usr/bin/env python3 -B
"""step03.py: check whether small-team setups really cost less."""
import ezr


def certify(xs, ys):
    """Return True if xs and ys are the same apart from noise."""
    return ezr.same(xs, ys)


if __name__ == "__main__":
    import os

    path = os.path.join(os.path.dirname(__file__), "pom3a.train.csv")
    with open(path, encoding="utf-8-sig") as f:
        lines = [l.strip() for l in f if l.strip()]

    header = lines[0].split(",")
    rows = [[ezr.atom(s) for s in line.split(",")] for line in lines[1:]]

    team_idx = header.index("Team Size")
    cost_idx = header.index("Cost-")

    small_cost, big_cost = [], []
    for row in rows:
        ts = float(row[team_idx])
        c = float(row[cost_idx])
        if ts < 0.5:
            small_cost.append(c)
        else:
            big_cost.append(c)

    result = certify(small_cost, big_cost)
    print(f"Small-team Cost- vs Big-team Cost-: same={result}")
