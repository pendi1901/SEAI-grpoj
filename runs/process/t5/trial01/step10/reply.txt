import ezr
import random

def race(header, rows, label):
    """
    Compares three strategies: random, greedy (acquire), and a simple 
    distance-based search. Total budget of 50 labels shared across all.
    """
    # Shared state for the budget
    budget = 50
    labeled_indices = {}

    def get_label(i):
        nonlocal budget
        if i not in labeled_indices:
            if budget <= 0:
                raise Exception("Budget exceeded")
            labeled_indices[i] = label(i)
            budget -= 1
        return labeled_indices[i]

    # Strategies
    def strategy_random(tbl, rows_with_hidden):
        indices = random.sample(range(len(rows_with_hidden)), min(16, budget))
        for i in indices:
            get_label(i)
        return min(indices, key=lambda i: ezr.ydist(tbl, get_label(i)))

    def strategy_acquire(tbl, rows_with_hidden):
        # Use ezr.acquire logic
        # We need to pass a wrapper that uses our shared budget
        def oracle_wrapper(row):
            # Find index of this row object
            idx = next(i for i, r in enumerate(rows_with_hidden) if r is row)
            return get_label(idx)
        
        best_rows = ezr.acquire(tbl, cap=17, label=oracle_wrapper)
        return next(i for i, r in enumerate(rows_with_hidden) if r in best_rows)

    def strategy_distance(tbl, rows_with_hidden):
        # Search for rows closest to the centroid (mids)
        centroid = ezr.mids(tbl)
        scored = []
        for i, row in enumerate(rows_with_hidden):
            scored.append((ezr.xdist(tbl, row, centroid), i))
        scored.sort()
        indices = [x[1] for x in scored[:17]]
        for i in indices:
            get_label(i)
        return min(indices, key=lambda i: ezr.ydist(tbl, get_label(i)))

    # Setup
    tbl = ezr.Tbl([header] + rows)
    strategies = {
        "random": strategy_random,
        "acquire": strategy_acquire,
        "distance": strategy_distance
    }

    results = {}
    for name, func in strategies.items():
        try:
            best_idx = func(tbl, rows)
            results[name] = (best_idx, ezr.ydist(tbl, get_label(best_idx)))
        except:
            results[name] = (None, float('inf'))

    winner = min(results, key=lambda n: results[n][1])
    return {
        "winner": winner,
        "row": results[winner][0],
        "tried": list(strategies.keys())
    }

if __name__ == "__main__":
    filename = "pom3a.n1000.train.csv"
    rows_iter = ezr.csv(filename)
    header = next(rows_iter)
    all_rows = list(rows_iter)
    
    # Mask goals
    hidden_rows = []
    for row in all_rows:
        new_row = list(row)
        for i, name in enumerate(header):
            if name[-1] in "+-":
                new_row[i] = "?"
        hidden_rows.append(new_row)
        
    def label_func(i):
        return all_rows[i]
    
    random.seed(1234567891)
    result = race(header, hidden_rows, label_func)
    
    print("Race Results:")
    print(f"Winner: {result['winner']}")
    print(f"Best row index found: {result['row']}")
    print(f"Strategies tried: {', '.join(result['tried'])}")
    exit(0)
