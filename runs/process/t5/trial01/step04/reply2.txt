#!/usr/bin/env python3 -B
"""step04.py: approve or reject each proposal on arrival."""


def flow(header, row, seen):
    """Return True to approve `row`, False to reject it.

    We keep the best ydist (lowest) seen so far; a new row is approved only if
    its predicted ydist beats every previously approved row.  This gives an
    online, irrevocable selection that favours setups with low Cost- and high
    Score-, Idle-.
    """
    tbl = ezr.Tbl([header] + seen)
    ranker_fn = ezr.ranker(tbl)

    # Predict ydist for the current row (goals hidden as "?")
    pred_row = list(row)
    for i, col_name in enumerate(header):
        if col_name[-1] in "+-":
            if pred_row[i] != "?":
                pred_row[i] = "?"

    new_ydist = ranker_fn(pred_row)

    # Track the best ydist among approved rows (initially None)
    if not seen:
        return True  # first row is always approved

    # We need to know the best ydist of previously approved rows.
    # Since we can't go back, we maintain a running threshold from `seen`.
    # But `seen` includes both approved and rejected rows — we don't have that info.
    # Strategy: use the minimum predicted ydist across all seen rows as our bar.
    best_so_far = min(
        (ranker_fn(list(r)) for r in seen),
        default=float("inf"),
    )

    return new_ydist < best_so_far


if __name__ == "__main__":
    import os

    path = os.path.join(os.path.dirname(__file__), "pom3a.train.csv")
    with open(path, encoding="utf-8-sig") as f:
        lines = [l.strip() for l in f if l.strip()]

    header = lines[0].split(",")
    rows = [[ezr.atom(s) for s in line.split(",")] for line in lines[1:]]

    approvals = 0
    current_row = None
    seen_rows = []

    for row in rows:
        if flow(header, row, seen_rows):
            approvals += 1
            print(f"APPROVED  Row {approvals}")
            # Reset: after approval we start fresh (irrevocable choice)
            current_row = row
            seen_rows = []
        else:
            seen_rows.append(row)

    if not seen_rows and approvals > 0:
        print(f"\nFinal decision: approved {approvals} of {len(rows)} proposals.")
    elif seen_rows:
        # If we end with unprocessed rows, the last one is implicitly rejected.
        print(f"\nRejected trailing {len(seen_rows)} proposal(s).")

    print(f"Summary: {approvals}/{len(rows)} approved.")
