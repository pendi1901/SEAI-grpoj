import ezr
import random

def race(header, rows, label):
    """
    Compare three strategies: random, greedy, and tree-based.
    Budget: 50 labels total.
    """
    tbl = ezr.Tbl([header] + rows)
    goal_indices = [i for i, name in enumerate(header) if name.endswith(('-', '+'))]
    
    # Shared state for the budget
    labeled_cache = {}
    def get_label(i):
        if i not in labeled_cache:
            if len(labeled_cache) >= 50:
                raise Exception("Budget exceeded")
            labeled_cache[i] = label(i)
        return labeled_cache[i]

    def get_row_with_goals(i):
        row = list(rows[i])
        goals = get_label(i)
        for idx, val in zip(goal_indices, goals):
            row[idx] = val
        return row

    # Strategy 1: Random
    def strategy_random():
        indices = random.sample(range(len(rows)), 16)
        best_idx = indices[0]
        for i in indices:
            if ezr.ydist(tbl, get_row_with_goals(i)) < ezr.ydist(tbl, get_row_with_goals(best_idx)):
                best_idx = i
        return best_idx

    # Strategy 2: Greedy (Nearest to best known)
    def strategy_greedy():
        indices = [random.choice(range(len(rows)))]
        for _ in range(15):
            best_so_far = get_row_with_goals(indices[-1])
            candidates = random.sample(range(len(rows)), 50)
            next_idx = min(candidates, key=lambda i: ezr.xdist(tbl, rows[i], best_so_far))
            indices.append(next_idx)
        return indices[-1]

    # Strategy 3: Tree-based (using ezr.acquire)
    def strategy_tree():
        # We use a wrapper to map the acquire process to our labeled_cache
        def oracle_wrapper(row):
            # Find index by matching row content (since rows are unique)
            for i, r in enumerate(rows):
                if r == row:
                    get_label(i)
                    return get_row_with_goals(i)
            return row
        
        labeled_rows = ezr.acquire(tbl, cap=16, label=oracle_wrapper)
        best_row = min(labeled_rows, key=lambda r: ezr.ydist(tbl, r))
        # Find index of best_row
        for i, r in enumerate(rows):
            if r == best_row: return i
        return 0

    results = {
        "random": strategy_random(),
        "greedy": strategy_greedy(),
        "tree": strategy_tree()
    }

    # Determine winner (lowest ydist)
    best_strategy = min(results, key=lambda s: ezr.ydist(tbl, get_row_with_goals(results[s])))
    
    return {
        "winner": best_strategy,
        "row": results[best_strategy],
        "tried": list(results.keys())
    }

if __name__ == "__main__":
    filename = "pom3a.n1000.train.csv"
    data_gen = ezr.csv(filename)
    header = next(data_gen)
    all_rows = list(data_gen)
    
    goal_indices = [i for i, name in enumerate(header) if name.endswith(('-', '+'))]
    masked_rows = []
    for row in all_rows:
        new_row = list(row)
        for idx in goal_indices:
            new_row[idx] = "?"
        masked_rows.append(new_row)
        
    def label_func(i):
        return [all_rows[i][idx] for idx in goal_indices]
        
    result = race(header, masked_rows, label_func)
    
    print("Strategy Race Results")
    print("-" * 40)
    print(f"Winner: {result['winner']}")
    print(f"Best row index found: {result['row']}")
    print(f"Strategies compared: {', '.join(result['tried'])}")
    print("-" * 40)
    exit(0)
