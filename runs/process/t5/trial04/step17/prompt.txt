Faster, simpler AI. Better maps, not bigger boots.

Options:

      -Stop=50         acquire: total labelling budget   
      -Few=128         holdout: max training rows   
      -Cuts=8          tree: splits to try, per num

def say(x: Any, p: int = 2) -> str:
  "Floats get P decimals; dicts list their keys."
  if isinstance(x,float): x= f"{x:.{p}f}".rstrip("0").rstrip(".")
  if isinstance(x, dict):
    x= "{"+", ".join(f"{k}: {say(v,p)}" for k,v in x.items())+"}"
  return str(x)

class o(dict):
  "A dict you can poke with a dot."
  __getattr__, __setattr__ = dict.__getitem__, dict.__setitem__
  __repr__ = say

def atom(s: str) -> ATOM:
  "'22' -> 22. '3.1' -> 3.1. 'True' -> True. 'x' -> 'x'."
  for fn in (int, float):
    try: return fn(s)
    except ValueError: pass
  s = s.strip()
  return {"True": True, "False": False}.get(s, s)

def csv(file: str) -> Iterator[ROW]:
  "ROWS of FILE, each cell coerced."
  with open(os.path.expanduser(file),  # -sig drops any BOM
           encoding="utf-8-sig") as f:
    for line in f:
      if line.strip(): yield [atom(s) for s in line.split(",")]

def Col(txt: str = " ", at: int = 0) -> COL:
  "Factory that returns NUM or SYM."
  return Num(txt,at) if txt[0].isupper() else Sym(txt,at)

class Num:
  "Place to summarize stream of numbers."
  __slots__ = ("at", "txt", "n", "mu", "m2", "sd", "goal")
  def __init__(self, txt: str = " ", at: int = 0) -> None:
    self.at, self.txt, self.n = at, txt, 0
    self.mu = self.m2 = self.sd = 0
    self.goal = 0 if txt[-1] == "-" else 1
  __repr__ = lambda self: say({k: getattr(self,k)
                               for k in self.__slots__})

class Sym:
  "Place to summarize stream of Symbols.  The class is the tag."
  __slots__ = ("at", "txt", "n", "has")
  def __init__(self, txt: str = " ", at: int = 0) -> None:
    self.at, self.txt, self.n, self.has = at, txt, 0, {}
  __repr__ = lambda self: say({k: getattr(self,k)
                               for k in self.__slots__})

def add(col: COL, v: ATOM, inc: int = 1) -> None:
  "Show V to COL.  INC=-1 takes it away.  `?` changes nothing."
  if v != "?":
    col.n += inc
    if type(col) is Sym:
      col.has[v] = col.has.get(v, 0) + inc
    else:
      d = v - col.mu                      # Welford, for `div`
      col.mu += inc * d / max(1, col.n)
      col.m2  = max(0, col.m2 + inc * d * (v - col.mu))
      col.sd  = 0 if col.n < 2 else (col.m2/(col.n-1))**.5

def mid(col: COL) -> ATOM:
  "Middle: the mean, or the most common symbol; ? if none."
  return (max(col.has, key=col.has.get) if col.has else "?"
          ) if type(col) is Sym else col.mu

def mids(tbl: TBL) -> MIDS:
  "Every column's middle, cached until the next addCols."
  tbl.mids = tbl.mids or {at: mid(c) for at,c in tbl.cols.items()}
  return tbl.mids

def div(col: COL) -> float:
  "Spread: entropy, or standard deviation."
  return (-sum(v/col.n*log2(v/col.n) for v in col.has.values())
          if type(col) is Sym else col.sd)  # sd kept by `add`

def norm(col: COL, v: ATOM) -> ATOM:
  "To 0..1, by the logistic curve.  Symbols do not scale."
  if type(col) is Sym: return v
  z = max(-3, min(3, (v - col.mu) / (1e-32 + div(col))))
  return 1 / (1 + exp(-1.7 * z))

#-- tables -------------------------------------------------
def Tbl(src: Iterable[ROW]) -> TBL:
  "Header names the columns: X skip, +- goal, ! klass."
  src = iter(src)
  tbl = Cols(o(rows=[], cols={}, x=[], y=[], mids=None,
               names=next(src)))
  for row in src: addRow(tbl, row)
  return tbl

def Cols(tbl: TBL) -> TBL:
  "Create column roles."
  for at, s in enumerate(tbl.names):
    if s[-1] == "X": continue            # skip me entirely
    tbl.cols[at] = Col(s,at)
    (tbl.y if s[-1] in "+-!" else tbl.x).append(tbl.cols[at])
  return tbl

def addRow(tbl: TBL, row: ROW) -> ROW:
  "Keep ROW, and show it to my columns."
  tbl.rows += [row]
  return addCols(tbl, row)

def addCols(tbl: TBL, row: ROW, inc: int = 1) -> ROW:
  "Show ROW to my columns, without keeping it."
  tbl.mids = None                 # a new row moves the middles
  for at, col in tbl.cols.items(): add(col, row[at], inc)
  return row

def clone(tbl: TBL, rows: ROWS = None) -> TBL:
  "An empty copy of TBL, plus ROWS."
  return Tbl([tbl.names] + (rows or []))

def dist(vs: Iterable[QTY], n: int) -> float:
  "Root mean square of N gaps."
  return sqrt(sum(v*v for v in vs) / n)

def ydist(tbl: TBL, row: ROW) -> float:
  "How far ROW's goals are from the best they could be."
  return dist((abs(norm(c, row[c.at]) - c.goal)
                    for c in tbl.y), len(tbl.y))

def gap(col: COL, a: ATOM, b: ATOM) -> QTY:
  "Distance between two values of one column."
  if a == "?" or b == "?": return 1    # unknown = far
  return a!=b if type(col) is Sym else abs(norm(col,a)-norm(col,b))

def xdist(tbl: TBL, r1: ROW, r2: ROW | MIDS) -> float:
  "How far apart two rows are, over the x columns."
  return dist((gap(c, r1[c.at], r2[c.at])
                    for c in tbl.x), len(tbl.x))

def oracle(row: ROW) -> ROW:
  """Labeller. Rows here arrive with ys, so nothing to do.
  Real oracles fill in any missing y cells (and can close
  over a TBL to know which cells those are)."""
  return row

def acquire(tbl: TBL, cap: int | None = None,
            label: ORACLE = oracle) -> ROWS:
  "Label near the best rows, far from the rest."
  cap  = cap or the.Stop
  todo = random.sample(tbl.rows, len(tbl.rows))[:the.Few]
  both = clone(tbl, [label(todo.pop())
                     for _ in range(the.Start)])
  both.rows.sort(key=lambda r: ydist(both, r))
  n    = int(sqrt(the.Start))
  best,rest = clone(tbl,both.rows[:n]), clone(tbl,both.rows[n:])
  while todo and len(both.rows) < cap:
    cb, cr = mids(best), mids(rest)    # two centroids
    todo.sort(key=lambda z: xdist(tbl,z,cr) - xdist(tbl,z,cb))
    addRow(best, addRow(both, label(todo.pop())))
    best.rows.sort(key=lambda z: ydist(both, z))  # no y leak
    if len(best.rows) > sqrt(len(both.rows)): # best stays small
      addRow(rest, addCols(best, best.rows.pop(), -1))
  return both.rows

def grabs(tbl: TBL, cap: int | None = None,
          label: ORACLE = oracle) -> ROWS:
  "The straw man: same budget, taken at random."
  cap = cap or the.Stop
  return [label(r) for r in
          random.sample(tbl.rows, min(cap, len(tbl.rows)))]

def cut(tbl: TBL, rows: ROWS) -> tuple[int,ATOM,GO]|None:
  "The (at, v, go) whose two sides have the tightest ys."
  ys = [ydist(tbl, r) for r in rows]
  out, least = None, 1e30
  for col in tbl.x:
    for v, go in candidates(col, rows):
      a, b = Num(), Num()   # one pass, so one `go` per row
      for r,y in zip(rows,ys): add(a if go(r) else b, y)
      if a.n >= the.Leaf and b.n >= the.Leaf:
        score = (div(a)*a.n + div(b)*b.n) / (a.n + b.n)
        if score < least: out, least = (col.at, v, go), score
  return out

def candidates(col: COL, rows: ROWS) -> Iterator[tuple[ATOM,GO]]:
  "What splits to try: one per symbol, or CUTS per num."
  at = col.at
  if type(col) is Sym:
    for v in sorted({r[at] for r in rows if r[at] != "?"}):
      yield v, lambda r,v=v,at=at: r[at] == v
  else:                # by rank, not value: THESE rows are dense
    xs = sorted(r[at] for r in rows if r[at] != "?")
    k  = max(1, len(xs)//the.Cuts)     # cuts, spaced by rank
    for v in xs[k-1:-1:k]:
      yield v, (lambda r,v=v,at=at: r[at] != "?" and r[at] <= v)

def tree(tbl: TBL, rows: ROWS) -> NODE:
  "Split while a split keeps both sides big enough."
  node = o(rows=rows, at=None, v=None, go=None, kids=[],
           mu=sum(ydist(tbl,r) for r in rows)/len(rows))
  if len(rows) > the.Leaf and (found := cut(tbl, rows)):
    at, v, go = found
    yes, no = [], []
    for r in rows: (yes if go(r) else no).append(r)
    node.at, node.v, node.go = at, v, go # cut kepts sides big
    node.kids = [tree(tbl, yes), tree(tbl, no)]
  return node

def leaf(node: NODE, row: ROW) -> NODE:
  "Walk ROW down to its leaf."
  while node.kids: node = node.kids[0 if node.go(row) else 1]
  return node

def leafs(node: NODE) -> list[NODE]:
  "Return leaves of this tree."
  return [x for k in node.kids for x in leafs(k)] or [node]

def show(tbl: TBL, node: NODE, pre: str | None = None,
         edge: str = "") -> None:
  "Tree on the left; ydist and n on the right."
  ls = sorted(leafs(node), key=lambda z: z.mu)
  def walk(z: NODE, pre: str | None, edge: str) -> None:
    m = "+" if z is ls[0] else "-" if z is ls[-1] else " "
    print(f"{m} {round(100*z.mu):>4} {len(z.rows):>5}   "
          f"{(pre or '')+edge}".rstrip())
    op = "=" if type(tbl.cols[z.at or 0]) is Sym else "<="
    no = "!=" if op == "=" else ">"
    for kid, oper in zip(z.kids, [op, no]):
      walk(kid, "" if pre is None else pre + "|  ",
           f"{tbl.names[z.at]} {oper} {say(z.v)}")
  print(" ydist     n")
  walk(node, pre, edge)



#-- rig ----------------------------------------------------
# Calls core, acquire, and one RANKER -- tree's, by default.
def ranker(lab: TBL) -> Callable[[ROW], float]:
  "Grow a tree on LAB; score any row by its leaf's mean ydist."
  node = tree(lab, lab.rows)
  return lambda r: leaf(node, r).mu

def wins(tbl: TBL) -> Callable[[ROW], float]:
  "100 at the best row, 0 at an average one."
  ys = sorted(ydist(tbl, r) for r in tbl.rows)
  lo, avg = ys[0], sum(ys)/len(ys)
  return lambda r: max(-100, min(100, 100*(1 - (ydist(tbl,r)-lo)
                                            / (avg-lo+1e-32))))

def holdout(tbl: TBL, pick: PICKER = acquire,
            rank: RANKER = ranker,
            label: ORACLE = oracle) -> ROW:
  "Train on half; of CHECK guesses on the rest, pick best."
  rows  = random.sample(tbl.rows, len(tbl.rows))
  n     = len(rows)//2
  tr    = clone(tbl, rows[:n][:the.Few])
  lab   = clone(tbl, pick(tr, the.Stop - the.Check, label))
  top   = sorted(rows[n:], key=rank(lab))
  return min(top[:the.Check], key=lambda r: ydist(lab, label(r)))

# A leaf: it calls core, and nothing here calls it.
def adds(vs: Iterable[ATOM], col: COL | None = None) -> COL:
  "Every item of VS into COL."
  col = Num() if col is None else col
  for v in vs: add(col, v)
  return col

def cohen(xs: NUMS, ys: NUMS, d: float = 0.35,
          eps: float = 0) -> bool:
  "Are the means within D pooled standard deviations?"
  a, b = adds(xs), adds(ys)
  pool = ((a.n-1)*div(a)**2 + (b.n-1)*div(b)**2)/(a.n+b.n-2)
  return abs(a.mu - b.mu) <= max(eps, d * sqrt(pool))

def cliffs(xs: NUMS, ys: NUMS, d: float = 0.197) -> bool:
  "Sorted in.  Is the rank imbalance small enough?"
  gt = lt = j = k = 0
  for x in xs:
    while j < len(ys) and ys[j] <  x: j += 1; k = j
    while k < len(ys) and ys[k] <= x: k += 1
    gt += j; lt += len(ys) - k
  return abs(gt - lt) / (len(xs) * len(ys)) <= d

def ks(xs: NUMS, ys: NUMS, a: float = 1.36) -> bool:
  "Sorted in.  95% Kolmogorov-Smirnov."
  n, m, i, j, d = len(xs), len(ys), 0, 0, 0
  while i < n and j < m:
    v = min(xs[i], ys[j])
    while i < n and xs[i] <= v: i += 1
    while j < m and ys[j] <= v: j += 1
    d = max(d, abs(i/n - j/m))
  return d <= a * sqrt((n + m) / (n * m))

def same(xs: NUMS, ys: NUMS, eps: float = 0) -> bool:
  """Indistinguishable.  Always cliffs and ks; EPS adds cohen, and
  doubles as its effect size (eps=0 skips that third test)."""
  xs, ys = sorted(xs), sorted(ys)
  return (cliffs(xs,ys) and ks(xs,ys)
          and (not eps or cohen(xs,ys,d=eps)))

import ezr
import random

def remember(header, rows):
    """
    Identify the top 5% of rows based on multi-objective optimization.
    A row is better if its goals (ending in - or +) are closer to their ideal.
    """
    # Create a table object using the ezr library
    # We prepend the header to the rows to match the expected input format
    data = [header] + rows
    tbl = ezr.Tbl(data)
    
    # Calculate ydist for every row.
    # ydist measures how far a row's goals are from the best possible values.
    # Lower ydist is better.
    scored_rows = []
    for i, row in enumerate(rows):
        score = ezr.ydist(tbl, row)
        scored_rows.append((score, i))
    
    # Sort by score (ascending, as lower ydist is better)
    scored_rows.sort(key=lambda x: x[0])
    
    # Return the top 5% of rows (800 rows * 0.05 = 40 rows)
    limit = max(1, int(len(rows) * 0.05))
    return [item[1] for item in scored_rows[:limit]]

import ezr
import random

def guess(header, train, test):
    """
    Estimates the first goal column for each row in test.
    Uses the ezr tree to find the leaf node for each test row
    and returns the mean of the first goal column in that leaf.
    """
    # Create table from training data
    data = [header] + train
    tbl = ezr.Tbl(data)
    
    # Build a tree on the training data
    node = ezr.tree(tbl, tbl.rows)
    
    # Identify the index of the first goal column
    # The first goal column is the first column with a name ending in '-' or '+'
    goal_indices = [i for i, name in enumerate(header) if name.endswith(('-', '+'))]
    first_goal_idx = goal_indices[0]
    
    results = []
    for row in test:
        # Walk the tree to find the leaf
        leaf_node = ezr.leaf(node, row)
        
        # Calculate the mean of the first goal column in this leaf
        vals = [r[first_goal_idx] for r in leaf_node.rows if r[first_goal_idx] != "?"]
        if not vals:
            # Fallback to global mean if leaf is empty or missing data
            vals = [r[first_goal_idx] for r in train if r[first_goal_idx] != "?"]
            
        results.append(sum(vals) / len(vals))
        
    return results

if __name__ == "__main__":
    # Load the data
    filename = "pom3a.n1000.train.csv"
    data_gen = ezr.csv(filename)
    
    try:
        header = next(data_gen)
        all_rows = list(data_gen)
    except StopIteration:
        print("Error: File is empty.")
        exit(1)
        
    # Split data: 80% train, 20% test
    random.seed(1234567891)
    random.shuffle(all_rows)
    split = int(len(all_rows) * 0.8)
    train = all_rows[:split]
    test_raw = all_rows[split:]
    
    # Mask goals in test set
    goal_indices = [i for i, name in enumerate(header) if name.endswith(('-', '+'))]
    test = []
    for row in test_raw:
        new_row = list(row)
        for idx in goal_indices:
            new_row[idx] = "?"
        test.append(new_row)
        
    # Get estimates
    estimates = guess(header, train, test)
    
    # Print summary
    print("Estimation Results (First Goal Column):")
    print("-" * 40)
    print(f"Estimated average for {len(estimates)} test cases: {sum(estimates)/len(estimates):.4f}")
    print("-" * 40)
    print("Sample estimates:")
    for i in range(min(5, len(estimates))):
        print(f"Row {i+1}: {estimates[i]:.4f}")
        
    exit(0)

import ezr
import random

def certify(xs, ys):
    """
    Returns True if xs and ys are statistically indistinguishable,
    False if they differ significantly.
    """

import ezr

def flow(header, row, seen):
    """
    Decide whether to approve a row.
    We approve if the row is significantly better than the average 
    of what we have seen so far.
    """
    if not seen:
        return True
    
    # Create a table to handle normalization and distance calculations
    # We use the seen rows to establish the context of "good"
    data = [header] + seen
    tbl = ezr.Tbl(data)
    
    # Calculate the ydist of the current row relative to the seen data
    # Note: We need to handle the fact that the current row might have '?'
    # but the goal columns are usually provided in the input stream.
    current_ydist = ezr.ydist(tbl, row)
    
    # Calculate the average ydist of all seen rows
    seen_ydists = [ezr.ydist(tbl, r) for r in seen]
    avg_ydist = sum(seen_ydists) / len(seen_ydists)
    
    # Approve if the current row is better (lower ydist) than the average
    # We add a small buffer to be selective
    return current_ydist < (avg_ydist * 0.95)

import ezr
import random

def blame(header, rows):
    """
    Identify input columns that best explain why the worst rows are bad.
    We define 'worst' as the rows with the highest ydist.
    We compare the distribution of input values in the worst 25% 
    against the rest, using the difference in means/modes as a proxy for blame.
    """
    tbl = ezr.Tbl([header] + rows)
    
    # Calculate ydist for all rows
    scored = [(ezr.ydist(tbl, r), r) for r in rows]
    scored.sort(key=lambda x: x[0], reverse=True)
    
    # Split into worst 25% and the rest
    n = len(rows) // 4
    worst = [x[1] for x in scored[:n]]
    others = [x[1] for x in scored[n:]]
    
    # Identify input columns (those not ending in - or +)
    input_indices = [i for i, name in enumerate(header) 
                     if not (name.endswith('-') or name.endswith('+'))]
    
    scores = []
    for i in input_indices:
        col = tbl.cols[i]
        # Calculate difference in "mid" (mean for Num, mode for Sym)
        # We normalize the difference to make them comparable
        w_vals = [r[i] for r in worst if r[i] != "?"]
        o_vals = [r[i] for r in others if r[i] != "?"]
        
        if not w_vals or not o_vals:
            continue
            
        if isinstance(col, ezr.Num):
            w_mid = sum(w_vals) / len(w_vals)
            o_mid = sum(o_vals) / len(o_vals)
            diff = abs(w_mid - o_mid)
        else:
            # For symbols, use simple frequency difference of the mode
            w_mode = max(set(w_vals), key=w_vals.count)
            o_mode = max(set(o_vals), key=o_vals.count)
            diff = 1.0 if w_mode != o_mode else 0.0
            
        scores.append((header[i], diff))
    
    # Return names of top 3 columns that contribute most to the difference
    scores.sort(key=lambda x: x[1], reverse=True)
    return [name for name, score in scores[:3]]

import ezr

def justify(header, rows, row):
    """
    Returns (p, reason): p is the fraction of rows that 'row' is better than.
    'row' has '?' in goal columns; we estimate its performance using a tree.
    """
    # Build a model on all known rows
    tbl = ezr.Tbl([header] + rows)
    node = ezr.tree(tbl, rows)
    
    # Estimate the ydist of the input row by finding its leaf
    leaf_node = ezr.leaf(node, row)
    estimated_ydist = leaf_node.mu
    
    # Calculate how many rows are worse than this estimate
    # A row is worse if its ydist is higher than our estimated ydist
    worse_count = 0
    for r in rows:
        if ezr.ydist(tbl, r) > estimated_ydist:
            worse_count += 1
            
    p = worse_count / len(rows)
    
    reason = (f"This setup is estimated to be better than {int(p * 100)}% of all "
              "possible configurations based on its input parameters.")
    
    return p, reason

import ezr
import random

def choose(header, rows, label):
    """
    Find the best row by using the acquire strategy to intelligently
    sample the search space within a budget of 50 labels.
    """
    # Create a table structure to manage the data
    # We use the provided rows (which have '?' in goals)
    tbl = ezr.Tbl([header] + rows)
    
    # The acquire function in ezr is designed to find the best rows
    # by balancing exploration and exploitation.
    # It returns a list of rows that have been labeled.
    # We set the budget to 50 as requested.
    
    # We need to wrap the label function to track indices
    # because acquire expects an oracle that returns a full row.
    labeled_data = []
    
    def oracle_wrapper(row):
        # Find the index of the row in the original list
        # Since rows are unique in this dataset, we can match by content
        for i, r in enumerate(rows):
            if r == row:
                # Call the provided label function
                true_goals = label(i)
                # Update the row with the true goal values
                new_row = list(row)
                goal_indices = [j for j, name in enumerate(header) 
                                if name.endswith(('-', '+'))]
                for idx, val in zip(goal_indices, true_goals):
                    new_row[idx] = val
                return new_row
        return row

    # Use ezr's acquire to perform the search
    # It will call our oracle_wrapper up to 50 times
    labeled_rows = ezr.acquire(tbl, cap=50, label=oracle_wrapper)
    
    # After labeling, find the best row among the labeled ones
    # ydist measures distance to the ideal goal values (lower is better)
    best_row = min(labeled_rows, key=lambda r: ezr.ydist(tbl, r))
    
    # Return the index of the best row found
    for i, r in enumerate(rows):
        if r == best_row:
            return i
    return 0

import ezr
import random

def fix(header, row, levers):
    """
    Finds the best change to a bad row by exploring the neighborhood
    defined by the provided levers.
    """
    # We need a reference table to calculate ydist.
    # We use the row as a single-row table to get the column definitions.
    # The normalization (norm) relies on the column's mu/sd.
    # Since we don't have the full training set here, we rely on the 
    # fact that ezr.norm uses column stats.
    tbl = ezr.Tbl([header, row])
    
    best_ydist = ezr.ydist(tbl, row)
    best_change = {}

    # Try changing each lever to improve the row
    for col_name, allowed in levers.items():
        if col_name not in header:
            continue
        at = header.index(col_name)
        
        # Generate candidates based on lever type
        candidates = []
        if isinstance(allowed, list):  # Symbolic
            candidates = [v for v in allowed if v != row[at]]
        else:  # Numeric (low, high)
            low, high = allowed
            # Try a few points in the range to ensure we find an improvement
            candidates = [low, (low + high) / 2, high]
            
        for val in candidates:
            test_row = list(row)
            test_row[at] = val
            
            # Calculate ydist for the modified row
            current_ydist = ezr.ydist(tbl, test_row)
            
            if current_ydist < best_ydist:
                best_ydist = current_ydist
                best_change = {col_name: val}
                
    return best_change

import ezr
import random

def race(header, rows, label):
    """
    Compare three strategies: random, greedy, and tree-based.
    Budget: 50 labels total.
    """
    tbl = ezr.Tbl([header] + rows)
    goal_indices = [i for i, name in enumerate(header) if name.endswith(('-', '+'))]
    
    # Shared state for the budget
    labeled_cache = {}
    def get_label(i):
        if i not in labeled_cache:
            if len(labeled_cache) >= 50:
                raise Exception("Budget exceeded")
            labeled_cache[i] = label(i)
        return labeled_cache[i]

    def get_row_with_goals(i):
        row = list(rows[i])
        goals = get_label(i)
        for idx, val in zip(goal_indices, goals):
            row[idx] = val
        return row

    # Strategy 1: Random
    def strategy_random():
        indices = random.sample(range(len(rows)), 16)
        best_idx = indices[0]
        for i in indices:
            if ezr.ydist(tbl, get_row_with_goals(i)) < ezr.ydist(tbl, get_row_with_goals(best_idx)):
                best_idx = i
        return best_idx

    # Strategy 2: Greedy (Nearest to best known)
    def strategy_greedy():
        indices = [random.choice(range(len(rows)))]
        for _ in range(15):
            best_so_far = get_row_with_goals(indices[-1])
            candidates = random.sample(range(len(rows)), 50)
            next_idx = min(candidates, key=lambda i: ezr.xdist(tbl, rows[i], best_so_far))
            indices.append(next_idx)
        return indices[-1]

    # Strategy 3: Tree-based (using ezr.acquire)
    def strategy_tree():
        # We use a wrapper to map the acquire process to our labeled_cache
        def oracle_wrapper(row):
            # Find index by matching row content (since rows are unique)
            for i, r in enumerate(rows):
                if r == row:
                    get_label(i)
                    return get_row_with_goals(i)
            return row
        
        labeled_rows = ezr.acquire(tbl, cap=16, label=oracle_wrapper)
        best_row = min(labeled_rows, key=lambda r: ezr.ydist(tbl, r))
        # Find index of best_row
        for i, r in enumerate(rows):
            if r == best_row: return i
        return 0

    results = {
        "random": strategy_random(),
        "greedy": strategy_greedy(),
        "tree": strategy_tree()
    }

    # Determine winner (lowest ydist)
    best_strategy = min(results, key=lambda s: ezr.ydist(tbl, get_row_with_goals(results[s])))
    
    return {
        "winner": best_strategy,
        "row": results[best_strategy],
        "tried": list(results.keys())
    }

import ezr

def guess(header, train, test):
    """
    Estimates the first goal column for each row in test.
    To simulate 'requirements changing twice as fast', we weight the 
    training data: the second half of the training set is given more 
    influence on the tree's leaf predictions.
    """
    # Identify the index of the first goal column
    goal_indices = [i for i, name in enumerate(header) if name.endswith(('-', '+'))]
    first_goal_idx = goal_indices[0]
    
    # Split training data into two halves
    mid = len(train) // 2
    first_half = train[:mid]
    second_half = train[mid:]
    
    # To simulate "changing twice as fast", we oversample the second half
    # so the tree model learns patterns more representative of the later state.
    weighted_train = first_half + (second_half * 2)
    
    # Create table from weighted training data
    data = [header] + weighted_train
    tbl = ezr.Tbl(data)
    
    # Build a tree on the weighted training data
    node = ezr.tree(tbl, tbl.rows)
    
    results = []
    for row in test:
        # Walk the tree to find the leaf
        leaf_node = ezr.leaf(node, row)
        
        # Calculate the mean of the first goal column in this leaf
        vals = [r[first_goal_idx] for r in leaf_node.rows if r[first_goal_idx] != "?"]
        
        if not vals:
            # Fallback to global mean of weighted training set
            vals = [r[first_goal_idx] for r in weighted_train if r[first_goal_idx] != "?"]
            
        results.append(sum(vals) / len(vals))
        
    return results

import ezr
import random

def flow(header, row, seen):
    """
    Warn the customer if a row is 'going bad'.
    We define 'going bad' as a significant drop in performance (ydist)
    compared to the historical trend of the 'seen' rows.
    """
    if len(seen) < 10:
        return False
    
    # Create a table from seen rows to establish the performance baseline
    tbl = ezr.Tbl([header] + seen)
    
    # Calculate the ydist of the current row
    # If the row has '?' in goals, we estimate its performance using the tree
    if any(cell == "?" for cell in row):
        node = ezr.tree(tbl, seen)
        current_ydist = ezr.leaf(node, row).mu
    else:
        current_ydist = ezr.ydist(tbl, row)
    
    # Calculate the moving average of ydist for the last 10 rows
    recent_rows = seen[-10:]
    recent_ydists = [ezr.ydist(tbl, r) for r in recent_rows]
    avg_recent_ydist = sum(recent_ydists) / len(recent_ydists)
    
    # Warn (return True) if the current performance is significantly worse
    # than the recent average (e.g., 20% worse).
    # Note: Higher ydist is worse.
    return current_ydist > (avg_recent_ydist * 1.2)

import ezr
import random

def blame(header, rows):
    """
    Identify the top two input columns that contribute most to poor performance
    (high ydist) and the columns that matter least.
    """
    tbl = ezr.Tbl([header] + rows)
    
    # Calculate ydist for all rows to identify the "worst"
    scored = [(ezr.ydist(tbl, r), r) for r in rows]
    scored.sort(key=lambda x: x[0], reverse=True)
    
    # Split into worst 25% and the rest
    n = len(rows) // 4
    worst = [x[1] for x in scored[:n]]
    others = [x[1] for x in scored[n:]]
    
    # Identify input columns (those not ending in - or +)
    input_indices = [i for i, name in enumerate(header) 
                     if not (name.endswith('-') or name.endswith('+'))]
    
    scores = []
    for i in input_indices:
        col = tbl.cols[i]
        w_vals = [r[i] for r in worst if r[i] != "?"]
        o_vals = [r[i] for r in others if r[i] != "?"]
        
        if not w_vals or not o_vals:
            continue
            
        # Use the difference in means/modes as a proxy for impact
        if isinstance(col, ezr.Num):
            w_mid = sum(w_vals) / len(w_vals)
            o_mid = sum(o_vals) / len(o_vals)
            diff = abs(w_mid - o_mid)
        else:
            w_mode = max(set(w_vals), key=w_vals.count)
            o_mode = max(set(o_vals), key=o_vals.count)
            diff = 1.0 if w_mode != o_mode else 0.0
            
        scores.append((header[i], diff))
    
    # Sort by impact
    scores.sort(key=lambda x: x[1], reverse=True)
    
    # Return the names of the top two (most impact) and the bottom one (least impact)
    # The prompt asks for "one or two facts that matter most" and "what to stop worrying about"
    return [scores[0][0], scores[1][0], scores[-1][0]]

import ezr
import random

def justify(header, rows, row):
    """
    Compares a target row against all known rows.
    Returns (p, reason): p is the fraction of rows that 'row' is better than.
    'row' has '?' in goal columns; we estimate its performance using a tree.
    """
    # Build a model on all known rows to estimate the performance of the target row
    tbl = ezr.Tbl([header] + rows)
    node = ezr.tree(tbl, rows)
    
    # Estimate the ydist of the input row by finding its leaf
    # ydist is the distance to the ideal, so lower is better.
    leaf_node = ezr.leaf(node, row)
    estimated_ydist = leaf_node.mu
    
    # Calculate how many rows are worse than this estimate
    # A row is worse if its ydist is higher than our estimated ydist
    worse_count = 0
    for r in rows:
        if ezr.ydist(tbl, r) > estimated_ydist:
            worse_count += 1
            
    p = worse_count / len(rows)
    
    reason = (f"Your chosen setup is estimated to be better than {p:.2%} "
              "of all historical configurations based on its input parameters.")
    
    return p, reason
