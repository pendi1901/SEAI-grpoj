import ezr
import random

def spot(header, rows):
    """
    Identify rows that are 'mistyped' or 'made up'.
    We define suspicious rows as those that are outliers in the input space
    (x-columns) relative to the rest of the data. We use the distance from
    the centroid of the input space as a measure of suspicion.
    """
    # Create a table to handle column roles
    tbl = ezr.Tbl([header] + rows)
    
    # Calculate the centroid of the input space (x-columns)
    # We use the 'mids' function from ezr which provides the median/mean
    # for each column.
    centroid = ezr.mids(tbl)
    
    # Calculate the distance of each row from the centroid
    # Higher distance = more suspicious (outlier)
    suspicious_scores = []
    for i, row in enumerate(rows):
        # xdist calculates the distance between a row and a set of values (the centroid)
        d = ezr.xdist(tbl, row, centroid)
        suspicious_scores.append((d, i))
    
    # Sort by distance descending (most suspicious first)
    suspicious_scores.sort(key=lambda x: x[0], reverse=True)
    
    # Return the indices of the top 5% most suspicious rows
    limit = max(1, int(len(rows) * 0.05))
    return [item[1] for item in suspicious_scores[:limit]]

if __name__ == "__main__":
    filename = "pom3a.n1000.train.csv"
    data_gen = ezr.csv(filename)
    
    try:
        header = next(data_gen)
        rows = list(data_gen)
    except StopIteration:
        print("Error: File is empty or missing.")
        exit(1)
        
    suspicious_indices = spot(header, rows)
    
    print(f"Analysis: Found {len(suspicious_indices)} potentially mistyped or anomalous records.")
    print("The following row indices are the most suspicious (most anomalous first):")
    print(suspicious_indices[:10])
    
    print("\nThese rows contain input values that deviate significantly from the")
    print("patterns observed in the rest of the dataset, suggesting they may")
    print("be outliers or data entry errors.")
    
    exit(0)
