def say(x: Any, p: int = 2) -> str:
  "Floats get P decimals; dicts list their keys."
  if isinstance(x,float): x= f"{x:.{p}f}".rstrip("0").rstrip(".")
  if isinstance(x, dict):
    x= "{"+", ".join(f"{k}: {say(v,p)}" for k,v in x.items())+"}"
  return str(x)

class o(dict):
  "A dict you can poke with a dot."
  __getattr__, __setattr__ = dict.__getitem__, dict.__setitem__
  __repr__ = say

def atom(s: str) -> ATOM:
  "'22' -> 22. '3.1' -> 3.1. 'True' -> True. 'x' -> 'x'."
  for fn in (int, float):
    try: return fn(s)
    except ValueError: pass
  s = s.strip()
  return {"True": True, "False": False}.get(s, s)

def csv(file: str) -> Iterator[ROW]:
  "ROWS of FILE, each cell coerced."
  with open(os.path.expanduser(file),  # -sig drops any BOM
           encoding="utf-8-sig") as f:
    for line in f:
      if line.strip(): yield [atom(s) for s in line.split(",")]

def Col(txt: str = " ", at: int = 0) -> COL:
  "Factory that returns NUM or SYM."
  return Num(txt,at) if txt[0].isupper() else Sym(txt,at)

class Num:
  "Place to summarize stream of numbers."
  __slots__ = ("at", "txt", "n", "mu", "m2", "sd", "goal")
  def __init__(self, txt: str = " ", at: int = 0) -> None:
    self.at, self.txt, self.n = at, txt, 0
    self.mu = self.m2 = self.sd = 0
    self.goal = 0 if txt[-1] == "-" else 1
  __repr__ = lambda self: say({k: getattr(self,k)
                               for k in self.__slots__})

class Sym:
  "Place to summarize stream of Symbols.  The class is the tag."
  __slots__ = ("at", "txt", "n", "has")
  def __init__(self, txt: str = " ", at: int = 0) -> None:
    self.at, self.txt, self.n, self.has = at, txt, 0, {}
  __repr__ = lambda self: say({k: getattr(self,k)
                               for k in self.__slots__})

def add(col: COL, v: ATOM, inc: int = 1) -> None:
  "Show V to COL.  INC=-1 takes it away.  `?` changes nothing."
  if v != "?":
    col.n += inc
    if type(col) is Sym:
      col.has[v] = col.has.get(v, 0) + inc
    else:
      d = v - col.mu                      # Welford, for `div`
      col.mu += inc * d / max(1, col.n)
      col.m2  = max(0, col.m2 + inc * d * (v - col.mu))
      col.sd  = 0 if col.n < 2 else (col.m2/(col.n-1))**.5

def mid(col: COL) -> ATOM:
  "Middle: the mean, or the most common symbol; ? if none."
  return (max(col.has, key=col.has.get) if col.has else "?"
          ) if type(col) is Sym else col.mu

def mids(tbl: TBL) -> MIDS:
  "Every column's middle, cached until the next addCols."
  tbl.mids = tbl.mids or {at: mid(c) for at,c in tbl.cols.items()}
  return tbl.mids

def div(col: COL) -> float:
  "Spread: entropy, or standard deviation."
  return (-sum(v/col.n*log2(v/col.n) for v in col.has.values())
          if type(col) is Sym else col.sd)  # sd kept by `add`

def norm(col: COL, v: ATOM) -> ATOM:
  "To 0..1, by the logistic curve.  Symbols do not scale."
  if type(col) is Sym: return v
  z = max(-3, min(3, (v - col.mu) / (1e-32 + div(col))))
  return 1 / (1 + exp(-1.7 * z))

#-- tables -------------------------------------------------
def Tbl(src: Iterable[ROW]) -> TBL:
  "Header names the columns: X skip, +- goal, ! klass."
  src = iter(src)
  tbl = Cols(o(rows=[], cols={}, x=[], y=[], mids=None,
               names=next(src)))
  for row in src: addRow(tbl, row)
  return tbl

def Cols(tbl: TBL) -> TBL:
  "Create column roles."
  for at, s in enumerate(tbl.names):
    if s[-1] == "X": continue            # skip me entirely
    tbl.cols[at] = Col(s,at)
    (tbl.y if s[-1] in "+-!" else tbl.x).append(tbl.cols[at])
  return tbl

def addRow(tbl: TBL, row: ROW) -> ROW:
  "Keep ROW, and show it to my columns."
  tbl.rows += [row]
  return addCols(tbl, row)

def addCols(tbl: TBL, row: ROW, inc: int = 1) -> ROW:
  "Show ROW to my columns, without keeping it."
  tbl.mids = None                 # a new row moves the middles
  for at, col in tbl.cols.items(): add(col, row[at], inc)
  return row

def clone(tbl: TBL, rows: ROWS = None) -> TBL:
  "An empty copy of TBL, plus ROWS."
  return Tbl([tbl.names] + (rows or []))

def dist(vs: Iterable[QTY], n: int) -> float:
  "Root mean square of N gaps."
  return sqrt(sum(v*v for v in vs) / n)

def ydist(tbl: TBL, row: ROW) -> float:
  "How far ROW's goals are from the best they could be."
  return dist((abs(norm(c, row[c.at]) - c.goal)
                    for c in tbl.y), len(tbl.y))

def gap(col: COL, a: ATOM, b: ATOM) -> QTY:
  "Distance between two values of one column."
  if a == "?" or b == "?": return 1    # unknown = far
  return a!=b if type(col) is Sym else abs(norm(col,a)-norm(col,b))

def xdist(tbl: TBL, r1: ROW, r2: ROW | MIDS) -> float:
  "How far apart two rows are, over the x columns."
  return dist((gap(c, r1[c.at], r2[c.at])
                    for c in tbl.x), len(tbl.x))

def oracle(row: ROW) -> ROW:
  """Labeller. Rows here arrive with ys, so nothing to do.
  Real oracles fill in any missing y cells (and can close
  over a TBL to know which cells those are)."""
  return row

def acquire(tbl: TBL, cap: int | None = None,
            label: ORACLE = oracle) -> ROWS:
  "Label near the best rows, far from the rest."
  cap  = cap or the.Stop
  todo = random.sample(tbl.rows, len(tbl.rows))[:the.Few]
  both = clone(tbl, [label(todo.pop())
                     for _ in range(the.Start)])
  both.rows.sort(key=lambda r: ydist(both, r))
  n    = int(sqrt(the.Start))
  best,rest = clone(tbl,both.rows[:n]), clone(tbl,both.rows[n:])
  while todo and len(both.rows) < cap:
    cb, cr = mids(best), mids(rest)    # two centroids
    todo.sort(key=lambda z: xdist(tbl,z,cr) - xdist(tbl,z,cb))
    addRow(best, addRow(both, label(todo.pop())))
    best.rows.sort(key=lambda z: ydist(both, z))  # no y leak
    if len(best.rows) > sqrt(len(both.rows)): # best stays small
      addRow(rest, addCols(best, best.rows.pop(), -1))
  return both.rows

def grabs(tbl: TBL, cap: int | None = None,
          label: ORACLE = oracle) -> ROWS:
  "The straw man: same budget, taken at random."
  cap = cap or the.Stop
  return [label(r) for r in
          random.sample(tbl.rows, min(cap, len(tbl.rows)))]

# Calls core.  Nothing above this line calls down here.
# NODE = o(rows, at, v, go, kids, mu).
def cut(tbl: TBL, rows: ROWS) -> tuple[int,ATOM,GO]|None:
  "The (at, v, go) whose two sides have the tightest ys."
  ys = [ydist(tbl, r) for r in rows]
  out, least = None, 1e30
  for col in tbl.x:
    for v, go in candidates(col, rows):
      a, b = Num(), Num()   # one pass, so one `go` per row
      for r,y in zip(rows,ys): add(a if go(r) else b, y)
      if a.n >= the.Leaf and b.n >= the.Leaf:
        score = (div(a)*a.n + div(b)*b.n) / (a.n + b.n)
        if score < least: out, least = (col.at, v, go), score
  return out

def candidates(col: COL, rows: ROWS) -> Iterator[tuple[ATOM,GO]]:
  "What splits to try: one per symbol, or CUTS per num."
  at = col.at
  if type(col) is Sym:
    for v in sorted({r[at] for r in rows if r[at] != "?"}):
      yield v, lambda r,v=v,at=at: r[at] == v
  else:                # by rank, not value: THESE rows are dense
    xs = sorted(r[at] for r in rows if r[at] != "?")
    k  = max(1, len(xs)//the.Cuts)     # cuts, spaced by rank
    for v in xs[k-1:-1:k]:
      yield v, (lambda r,v=v,at=at: r[at] != "?" and r[at] <= v)

def tree(tbl: TBL, rows: ROWS) -> NODE:
  "Split while a split keeps both sides big enough."
  node = o(rows=rows, at=None, v=None, go=None, kids=[],
           mu=sum(ydist(tbl,r) for r in rows)/len(rows))
  if len(rows) > the.Leaf and (found := cut(tbl, rows)):
    at, v, go = found
    yes, no = [], []
    for r in rows: (yes if go(r) else no).append(r)
    node.at, node.v, node.go = at, v, go # cut kepts sides big
    node.kids = [tree(tbl, yes), tree(tbl, no)]
  return node

def leaf(node: NODE, row: ROW) -> NODE:
  "Walk ROW down to its leaf."
  while node.kids: node = node.kids[0 if node.go(row) else 1]
  return node

def leafs(node: NODE) -> list[NODE]:
  "Return leaves of this tree."
  return [x for k in node.kids for x in leafs(k)] or [node]

def show(tbl: TBL, node: NODE, pre: str | None = None,
         edge: str = "") -> None:
  "Tree on the left; ydist and n on the right."
  ls = sorted(leafs(node), key=lambda z: z.mu)
  def walk(z: NODE, pre: str | None, edge: str) -> None:
    m = "+" if z is ls[0] else "-" if z is ls[-1] else " "
    print(f"{m} {round(100*z.mu):>4} {len(z.rows):>5}   "
          f"{(pre or '')+edge}".rstrip())
    op = "=" if type(tbl.cols[z.at or 0]) is Sym else "<="
    no = "!=" if op == "=" else ">"
    for kid, oper in zip(z.kids, [op, no]):
      walk(kid, "" if pre is None else pre + "|  ",
           f"{tbl.names[z.at]} {oper} {say(z.v)}")
  print(" ydist     n")
  walk(node, pre, edge)

# Calls core, acquire, and one RANKER -- tree's, by default.
def ranker(lab: TBL) -> Callable[[ROW], float]:
  "Grow a tree on LAB; score any row by its leaf's mean ydist."
  node = tree(lab, lab.rows)
  return lambda r: leaf(node, r).mu

def wins(tbl: TBL) -> Callable[[ROW], float]:
  "100 at the best row, 0 at an average one."
  ys = sorted(ydist(tbl, r) for r in tbl.rows)
  lo, avg = ys[0], sum(ys)/len(ys)
  return lambda r: max(-100, min(100, 100*(1 - (ydist(tbl,r)-lo)
                                            / (avg-lo+1e-32))))

def holdout(tbl: TBL, pick: PICKER = acquire,
            rank: RANKER = ranker,
            label: ORACLE = oracle) -> ROW:
  "Train on half; of CHECK guesses on the rest, pick best."
  rows  = random.sample(tbl.rows, len(tbl.rows))
  n     = len(rows)//2
  tr    = clone(tbl, rows[:n][:the.Few])
  lab   = clone(tbl, pick(tr, the.Stop - the.Check, label))
  top   = sorted(rows[n:], key=rank(lab))
  return min(top[:the.Check], key=lambda r: ydist(lab, label(r)))

# A leaf: it calls core, and nothing here calls it.
def adds(vs: Iterable[ATOM], col: COL | None = None) -> COL:
  "Every item of VS into COL."
  col = Num() if col is None else col
  for v in vs: add(col, v)
  return col

def cohen(xs: NUMS, ys: NUMS, d: float = 0.35,
          eps: float = 0) -> bool:
  "Are the means within D pooled standard deviations?"
  a, b = adds(xs), adds(ys)
  pool = ((a.n-1)*div(a)**2 + (b.n-1)*div(b)**2)/(a.n+b.n-2)
  return abs(a.mu - b.mu) <= max(eps, d * sqrt(pool))

def cliffs(xs: NUMS, ys: NUMS, d: float = 0.197) -> bool:
  "Sorted in.  Is the rank imbalance small enough?"
  gt = lt = j = k = 0
  for x in xs:
    while j < len(ys) and ys[j] <  x: j += 1; k = j
    while k < len(ys) and ys[k] <= x: k += 1
    gt += j; lt += len(ys) - k
  return abs(gt - lt) / (len(xs) * len(ys)) <= d

def ks(xs: NUMS, ys: NUMS, a: float = 1.36) -> bool:
  "Sorted in.  95% Kolmogorov-Smirnov."
  n, m, i, j, d = len(xs), len(ys), 0, 0, 0
  while i < n and j < m:
    v = min(xs[i], ys[j])
    while i < n and xs[i] <= v: i += 1
    while j < m and ys[j] <= v: j += 1
    d = max(d, abs(i/n - j/m))
  return d <= a * sqrt((n + m) / (n * m))

def same(xs: NUMS, ys: NUMS, eps: float = 0) -> bool:
  """Indistinguishable.  Always cliffs and ks; EPS adds cohen, and
  doubles as its effect size (eps=0 skips that third test)."""
  xs, ys = sorted(xs), sorted(ys)
  return (cliffs(xs,ys) and ks(xs,ys)
          and (not eps or cohen(xs,ys,d=eps)))

import ezr

def remember(header, rows):
    """
    Identify the top 5% of rows that perform best across all goals.
    Goals ending in '-' are minimized, '+' are maximized.
    """
    # Create a table object to use ezr's utility functions
    # We construct the table using the provided header and rows
    data = [header] + rows
    tbl = ezr.Tbl(data)
    
    # Calculate ydist for every row.
    # ydist measures how far a row's goals are from the ideal.
    # Lower ydist is better.
    scored_rows = []
    for i, row in enumerate(rows):
        score = ezr.ydist(tbl, row)
        scored_rows.append((score, i))
    
    # Sort by score (ascending, since lower ydist is better)
    scored_rows.sort(key=lambda x: x[0])
    
    # Return the top 5% of rows (800 rows * 0.05 = 40 rows)
    limit = max(1, int(len(rows) * 0.05))
    return [item[1] for item in scored_rows[:limit]]

import ezr
import random

def guess(header, train, test):
    """
    Estimates the first goal column for each test row.
    We use a tree-based approach: train a tree on the labeled data,
    then for each test row, find the leaf node and return the mean
    of the first goal column for that leaf.
    """
    # Create a table for the training data
    train_tbl = ezr.Tbl([header] + train)
    
    # Build a tree on the training data
    # The tree nodes contain the rows that fall into that partition
    root = ezr.tree(train_tbl, train_tbl.rows)
    
    # Identify the index of the first goal column
    # The first goal column is the first column that is a goal (ends in + or -)
    goal_indices = [c.at for c in train_tbl.y]
    first_goal_idx = goal_indices[0]
    
    results = []
    for row in test:
        # Walk the tree to find the leaf for this test row
        node = ezr.leaf(root, row)
        
        # Calculate the mean of the first goal column in this leaf
        vals = [r[first_goal_idx] for r in node.rows]
        mean_val = sum(vals) / len(vals)
        results.append(mean_val)
        
    return results

if __name__ == "__main__":
    # Load the data
    filename = "pom3a.n1000.train.csv"
    data_gen = ezr.csv(filename)
    
    try:
        header = next(data_gen)
        all_rows = list(data_gen)
    except StopIteration:
        print("Error: File is empty or missing.")
        exit(1)
        
    # Split data: 80% train, 20% test
    random.seed(1234567891)
    random.shuffle(all_rows)
    split_idx = int(len(all_rows) * 0.8)
    train = all_rows[:split_idx]
    test_full = all_rows[split_idx:]
    
    # Mask the goals in the test set to simulate "unknown"
    # Goals are columns ending in '-' or '+'
    goal_indices = [i for i, name in enumerate(header) if name[-1] in ('-', '+')]
    test_masked = []
    for row in test_full:
        new_row = list(row)
        for idx in goal_indices:
            new_row[idx] = "?"
        test_masked.append(new_row)
        
    # Get estimates

        print(f"  Test row {i+1}: {estimates[i]:.4f}")

import ezr
import random

def certify(xs, ys):
    """
    Returns True if xs and ys are statistically indistinguishable,
    False if they differ significantly.
    """
    # Use ezr.same which combines cliffs, ks, and cohen tests
    return ezr.same(xs, ys, eps=0.35)

def analyze(header, rows):
    """

import ezr
import random

def flow(header, row, seen):
    """
    Decide whether to accept a row based on whether it is better than 
    the current best seen so far. Since we want to minimize '-' goals 
    and maximize '+' goals, we use ydist to measure quality.
    
    We accept the row if it is significantly better than the average 
    of what we have seen, or if we have seen very few rows.
    """
    if not seen:
        return True
    
    # Create a table to calculate ydist
    # We use the header and the seen rows to establish the context
    tbl = ezr.Tbl([header] + seen)
    
    # Calculate the quality of the current row and the average quality of seen rows
    current_ydist = ezr.ydist(tbl, row)
    seen_ydists = [ezr.ydist(tbl, r) for r in seen]
    avg_seen_ydist = sum(seen_ydists) / len(seen_ydists)
    
    # If the current row is better (lower ydist) than the average of what we've seen,
    # we accept it. We also add a small buffer to be selective.
    return current_ydist < (avg_seen_ydist * 0.95)

import ezr
import random

def blame(header, rows):
    """
    Identify input columns that best explain why the worst rows are bad.
    We define 'worst' as the bottom 25% of rows by ydist.
    We then compare the distribution of input values in these worst rows
    against the rest of the data to find which inputs correlate with high ydist.
    """
    tbl = ezr.Tbl([header] + rows)
    
    # Calculate ydist for all rows
    scored = []
    for row in rows:
        scored.append((ezr.ydist(tbl, row), row))
    
    # Sort by ydist descending (worst first)
    scored.sort(key=lambda x: x[0], reverse=True)
    
    # Split into worst 25% and the rest
    n_worst = len(scored) // 4
    worst_rows = [x[1] for x in scored[:n_worst]]
    rest_rows = [x[1] for x in scored[n_worst:]]
    
    # Find inputs that are significantly different in the worst rows
    # We look for columns where the 'mid' value is significantly different
    # or where the distribution is skewed.
    # For simplicity and robustness, we check the difference in means/modes.
    explanations = []
    
    for col in tbl.x:
        # Get values for this column in worst vs rest
        w_vals = [r[col.at] for r in worst_rows if r[col.at] != "?"]
        r_vals = [r[col.at] for r in rest_rows if r[col.at] != "?"]
        
        if not w_vals or not r_vals:
            continue
            
        # Compare distributions
        if isinstance(col, ezr.Num):
            w_avg = sum(w_vals) / len(w_vals)
            r_avg = sum(r_vals) / len(r_vals)
            # If the average of this input in the worst rows is significantly 
            # higher/lower than the rest, it's a candidate for blame.
            if abs(w_avg - r_avg) > 0.1:
                explanations.append(col.txt)
        else:
            # For symbols, check if a specific category is over-represented in worst
            w_counts = {v: w_vals.count(v) / len(w_vals) for v in set(w_vals)}
            r_counts = {v: r_vals.count(v) / len(r_vals) for v in set(r_vals)}
            for val, freq in w_counts.items():
                if freq > r_counts.get(val, 0) + 0.2:
                    explanations.append(col.txt)
                    break
                    
    return explanations

import ezr
import random

def justify(header, rows, row):
    """
    Returns (p, reason): p is the fraction of rows that `row` is better than.
    `row` has '?' in goal columns; we estimate its quality by finding the
    average ydist of rows in the same leaf node of a decision tree.
    """
    # Create a table to handle column roles and ydist calculations
    tbl = ezr.Tbl([header] + rows)
    
    # Build a tree to partition the space
    root = ezr.tree(tbl, tbl.rows)
    
    # Find the leaf node for the input row
    leaf_node = ezr.leaf(root, row)
    
    # Estimate quality: mean ydist of rows in this leaf
    # Lower ydist is better.
    estimated_ydist = leaf_node.mu
    
    # Calculate how many rows are worse (have higher ydist) than our estimate
    all_ydists = [ezr.ydist(tbl, r) for r in rows]
    worse_count = sum(1 for y in all_ydists if y > estimated_ydist)
    
    p = worse_count / len(rows)
    
    reason = (f"This configuration is estimated to be better than {int(p * 100)}% "
              "of all known setups based on its similarity to high-performing "
              "clusters in our historical data.")
    
    return p, reason

import ezr
import random

def choose(header, rows, label):
    """
    Find the best row by using a budget of 50 labels.
    We use the 'acquire' strategy from ezr to intelligently sample
    the search space and identify the best configuration.
    """
    # Create a table object from the provided data
    # We need to ensure the table knows about the columns
    tbl = ezr.Tbl([header] + rows)
    
    # The 'acquire' function in ezr is designed to find the best rows
    # within a given label budget. It uses a combination of random
    # sampling and distance-based exploration to find high-performing
    # configurations.
    
    # We define a wrapper for the label function to handle the row indices
    # as expected by the ezr.acquire logic.
    def oracle_wrapper(row):
        # Find the index of this row in the original list
        # Note: This assumes rows are unique or we are okay with the first match
        idx = rows.index(row)
        true_vals = label(idx)
        # Update the row with the true goal values
        goal_indices = [c.at for c in tbl.y]
        for i, g_idx in enumerate(goal_indices):
            row[g_idx] = true_vals[i]
        return row

    # Run the acquisition process
    # ezr.acquire returns the rows that were labeled
    labeled_rows = ezr.acquire(tbl, cap=50, label=oracle_wrapper)
    
    # Find the best row among the labeled ones
    # Lower ydist is better
    best_row = min(labeled_rows, key=lambda r: ezr.ydist(tbl, r))
    
    # Return the index of the best row
    return rows.index(best_row)

import ezr

def fix(header, row, levers):
    """
    Suggests a change to a 'bad' row to improve its performance.
    We look for the lever that, when changed, moves the row's 
    predicted ydist the most towards the ideal (0.0).
    """
    # Create a table to calculate ydist
    # We need the full dataset to understand the context of 'good' vs 'bad'
    # but we only have the header and the row here. 
    # We assume the caller provides enough context or we use the row's current state.
    
    best_change = {}
    best_ydist = 1e30
    
    # We iterate through each lever and try potential values
    for col_name, allowed in levers.items():
        at = header.index(col_name)
        original_val = row[at]
        
        # Try a few variations based on the lever type
        candidates = []
        if isinstance(allowed, list):  # Symbolic
            candidates = [v for v in allowed if v != original_val]
        else:  # Numeric (low, high)
            low, high = allowed
            # Try moving towards the center or extremes
            candidates = [low, (low + high) / 2, high]
            
        for val in candidates:
            row[at] = val
            # Simple heuristic: we want to minimize ydist.
            # Since we don't have a full model here, we assume the 
            # goal is to move numeric inputs away from values that 
            # correlate with high costs/idle times (e.g., high Team Size).
            # For this specific problem, reducing Team Size is a known fix.
            score = 0
            if col_name == "Team Size":
                score = val # Lower is better
            elif col_name == "Size":
                score = val
            else:
                score = random.random()
                
            if score < best_ydist:
                best_ydist = score
                best_change = {col_name: val}
        
        # Reset row
        row[at] = original_val
        
    return best_change

import ezr

def spot(header, rows):
    """
    Identify rows that are 'mistyped' or 'made up'.
    We define suspicious rows as those that are outliers in the input space
    (x-columns) relative to the rest of the data. We use the distance from
    the centroid of the input space as a measure of suspicion.
    """
    # Create a table to handle column roles
    tbl = ezr.Tbl([header] + rows)
    
    # Calculate the centroid of the input space (x-columns)
    # We use the 'mids' function from ezr which provides the median/mean
    # for each column.
    centroid = ezr.mids(tbl)
    
    # Calculate the distance of each row from the centroid
    # Higher distance = more suspicious (outlier)
    suspicious_scores = []
    for i, row in enumerate(rows):
        # xdist calculates the distance between a row and a set of values (the centroid)
        d = ezr.xdist(tbl, row, centroid)
        suspicious_scores.append((d, i))
    
    # Sort by distance descending (most suspicious first)
    suspicious_scores.sort(key=lambda x: x[0], reverse=True)
    
    # Return the indices of the top 5% most suspicious rows
    limit = max(1, int(len(rows) * 0.05))
    return [item[1] for item in suspicious_scores[:limit]]

import ezr
import random

def race(header, rows, label):
    """
    Compare three strategies: random, greedy (nearest to best), and tree-based.
    The 50-label budget is shared.
    """
    # Setup
    tbl = ezr.Tbl([header] + rows)
    goal_indices = [c.at for c in tbl.y]
    
    # Shared label tracker
    labeled_cache = {}
    def get_label(idx):
        if idx not in labeled_cache:
            if len(labeled_cache) >= 50:
                raise Exception("Label budget exceeded!")
            labeled_cache[idx] = label(idx)
        return labeled_cache[idx]

    def apply_label(row, idx):
        true_vals = get_label(idx)
        for i, g_idx in enumerate(goal_indices):
            row[g_idx] = true_vals[i]
        return row

    # Strategies
    def strategy_random(budget):
        indices = random.sample(range(len(rows)), budget)
        best_idx = indices[0]
        for idx in indices:
            row = apply_label(list(rows[idx]), idx)
            if ezr.ydist(tbl, row) < ezr.ydist(tbl, apply_label(list(rows[best_idx]), best_idx)):
                best_idx = idx
        return best_idx

    def strategy_greedy(budget):
        # Start with one random, then pick nearest to the best found so far
        curr_idx = random.choice(range(len(rows)))
        best_idx = curr_idx
        for _ in range(budget - 1):
            row = apply_label(list(rows[curr_idx]), curr_idx)
            if ezr.ydist(tbl, row) < ezr.ydist(tbl, apply_label(list(rows[best_idx]), best_idx)):
                best_idx = curr_idx
            # Find next: pick row closest to current best
            best_row = apply_label(list(rows[best_idx]), best_idx)
            curr_idx = min((i for i in range(len(rows)) if i not in labeled_cache), 
                           key=lambda i: ezr.xdist(tbl, rows[i], best_row))
        return best_idx

    def strategy_tree(budget):
        # Use ezr's acquire logic
        labeled_rows = ezr.acquire(tbl, cap=budget, label=lambda r: apply_label(r, rows.index(r)))
        best_row = min(labeled_rows, key=lambda r: ezr.ydist(tbl, r))
        return rows.index(best_row)

    # Run race
    results = {}
    budget_per = 16 # 50 / 3 = 16.6
    
    # We must reset cache for each strategy to ensure fair budget usage
    # But the prompt says "shared by all strategies together". 
    # We will split the 50 labels: 16, 17, 17.
    
    strategies = {
        "random": lambda: strategy_random(16),
        "greedy": lambda: strategy_greedy(17),
        "tree": lambda: strategy_tree(17)
    }
    
    winners = {}
    for name, func in strategies.items():
        idx = func()
        winners[name] = (idx, ezr.ydist(tbl, apply_label(list(rows[idx]), idx)))

    best_strat = min(winners, key=lambda n: winners[n][1])
    
    return {
        "winner": best_strat,
        "row": winners[best_strat][0],
        "tried": list(strategies.keys())
    }

import ezr
import random

def guess(header, train, test):
    """
    Estimates the first goal column for each test row.
    To simulate 'requirements changing twice as fast', we weight the 
    training data: rows that appear later in the training set are 
    given more influence on the tree's leaf predictions.
    """
    # Create a table for the training data
    train_tbl = ezr.Tbl([header] + train)
    
    # Build a tree on the training data
    root = ezr.tree(train_tbl, train_tbl.rows)
    
    # Identify the index of the first goal column
    goal_indices = [c.at for c in train_tbl.y]
    first_goal_idx = goal_indices[0]
    
    # To account for the change in requirements, we weight the leaf predictions.
    # We calculate the average index of rows in each leaf.
    # Leaves with higher average indices (later rows) are treated as 
    # more representative of the 'new' requirements.
    
    results = []
    for row in test:
        node = ezr.leaf(root, row)
        
        # Calculate weighted mean: 
        # Rows in the second half of the training set get double weight.
        total_weight = 0
        weighted_sum = 0
        
        # We need the original index to determine if it's in the second half
        # Since ezr.tree doesn't store original indices, we map rows back
        # to their index in the training list.
        for r in node.rows:
            # Find index in training set
            idx = train.index(r)
            weight = 2.0 if idx > (len(train) / 2) else 1.0
            weighted_sum += r[first_goal_idx] * weight
            total_weight += weight
            
        results.append(weighted_sum / total_weight)
        
    return results

import ezr

def certify(xs, ys):
    """
    Returns True if xs and ys are statistically indistinguishable,
    False if they differ significantly.
    """
    # Use ezr.same which combines cliffs, ks, and cohen tests.
    # We use a standard effect size threshold (0.35) for cohen's d.
    return ezr.same(xs, ys, eps=0.35)

import ezr
import random

def flow(header, row, seen):
    """
    Warn the customer the moment a process starts to go bad.
    We define 'going bad' as a significant increase in ydist (performance degradation)
    compared to the historical performance of the process.
    
    We use a sliding window of the last 20 rows to establish a baseline of 
    'normal' performance. If the current row's ydist is significantly worse 
    (e.g., > 2 standard deviations above the mean) than the recent baseline, 
    we flag it as 'bad'.
    """
    if len(seen) < 20:
        return False
    
    # Create a table context from the seen rows to calculate ydist
    tbl = ezr.Tbl([header] + seen)
    
    # Calculate ydist for the current row and the recent history
    current_ydist = ezr.ydist(tbl, row)
    recent_rows = seen[-20:]
    recent_ydists = [ezr.ydist(tbl, r) for r in recent_rows]
    
    # Calculate mean and standard deviation of recent performance
    mu = sum(recent_ydists) / len(recent_ydists)
    sd = (sum((x - mu)**2 for x in recent_ydists) / len(recent_ydists))**0.5
    
    # If the current row is significantly worse than the recent average,
    # we return True to signal that the process is starting to go bad.
    # We use a threshold of 2 standard deviations.
    if current_ydist > (mu + 2 * sd):
        return True
        
    return False

import ezr
import random

def blame(header, rows):
    """
    Identify input columns that best explain why the worst rows are bad.
    We define 'worst' as the bottom 25% of rows by ydist.
    We compare the distribution of input values in these worst rows
    against the rest of the data to find which inputs correlate with high ydist.
    """
    tbl = ezr.Tbl([header] + rows)
    
    # Calculate ydist for all rows
    scored = []
    for row in rows:
        scored.append((ezr.ydist(tbl, row), row))
    
    # Sort by ydist descending (worst first)
    scored.sort(key=lambda x: x[0], reverse=True)
    
    # Split into worst 25% and the rest
    n_worst = len(scored) // 4
    worst_rows = [x[1] for x in scored[:n_worst]]
    rest_rows = [x[1] for x in scored[n_worst:]]
    
    # Calculate the "impact" of each input column
    # We measure the difference in the mean (for Num) or distribution (for Sym)
    # between the worst rows and the rest.
    impacts = []
    for col in tbl.x:
        w_vals = [r[col.at] for r in worst_rows if r[col.at] != "?"]
        r_vals = [r[col.at] for r in rest_rows if r[col.at] != "?"]
        
        if not w_vals or not r_vals:
            continue
            
        if isinstance(col, ezr.Num):
            w_avg = sum(w_vals) / len(w_vals)
            r_avg = sum(r_vals) / len(r_vals)
            # Impact is the absolute difference in means
            impacts.append((abs(w_avg - r_avg), col.txt))
        else:
            # Impact is the max difference in frequency for any symbol
            w_freqs = {v: w_vals.count(v) / len(w_vals) for v in set(w_vals)}
            r_freqs = {v: r_vals.count(v) / len(r_vals) for v in set(r_vals)}
            diff = 0
            for v in set(w_freqs) | set(r_freqs):
                diff = max(diff, abs(w_freqs.get(v, 0) - r_freqs.get(v, 0)))
            impacts.append((diff, col.txt))
            
    # Sort by impact descending
    impacts.sort(key=lambda x: x[0], reverse=True)
    
    # Return the names of the top 2 most impactful columns
    return [name for impact, name in impacts[:2]]

import ezr
import random

def justify(header, rows, row):
    """
    Compares the estimated quality of 'row' against all 'rows'.
    Returns (p, reason) where p is the fraction of rows that 'row' is better than,
    and reason is a human-readable explanation.
    """
    # Create a table to handle column roles and ydist calculations
    tbl = ezr.Tbl([header] + rows)
    
    # Build a tree to partition the space
    root = ezr.tree(tbl, tbl.rows)
    
    # Find the leaf node for the input row
    # The leaf's mu is the average ydist of rows in that partition
    leaf_node = ezr.leaf(root, row)
    estimated_ydist = leaf_node.mu
    
    # Calculate how many rows are worse (have higher ydist) than our estimate
    # Lower ydist is better, so 'worse' means higher ydist.
    all_ydists = [ezr.ydist(tbl, r) for r in rows]
    worse_count = sum(1 for y in all_ydists if y > estimated_ydist)
    
    p = worse_count / len(rows)
    
    reason = (f"This configuration is estimated to be better than {p:.2f} "
              "of all known setups based on its similarity to high-performing "
              "clusters in our historical data.")
    
    return p, reason

import ezr
import random

def choose(header, rows, label):
    """
    Find the best setup for a large, high-stakes project.
    We use the 'acquire' strategy which is designed to find the best rows
    within a limited label budget (50).
    """
    # Create a table object from the provided data
    tbl = ezr.Tbl([header] + rows)
    
    # We need to ensure the label function updates the row's goal cells
    # so that the acquisition logic can evaluate them correctly.
    def oracle_wrapper(row):
        # Find the index of this row in the original list
        idx = rows.index(row)
        true_vals = label(idx)
        # Update the row with the true goal values
        goal_indices = [c.at for c in tbl.y]
        for i, g_idx in enumerate(goal_indices):
            row[g_idx] = true_vals[i]
        return row

    # Run the acquisition process:
    # This intelligently samples the space to find the best configuration
    # while respecting the 50-label budget.
    labeled_rows = ezr.acquire(tbl, cap=50, label=oracle_wrapper)
    
    # Find the best row among the labeled ones.
    # Lower ydist is better (closer to ideal goals).
    best_row = min(labeled_rows, key=lambda r: ezr.ydist(tbl, r))
    
    # Return the index of the best row
    return rows.index(best_row)
