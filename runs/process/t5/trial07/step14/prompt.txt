def say(x: Any, p: int = 2) -> str:
  "Floats get P decimals; dicts list their keys."
  if isinstance(x,float): x= f"{x:.{p}f}".rstrip("0").rstrip(".")
  if isinstance(x, dict):
    x= "{"+", ".join(f"{k}: {say(v,p)}" for k,v in x.items())+"}"
  return str(x)

class o(dict):
  "A dict you can poke with a dot."
  __getattr__, __setattr__ = dict.__getitem__, dict.__setitem__
  __repr__ = say

def atom(s: str) -> ATOM:
  "'22' -> 22. '3.1' -> 3.1. 'True' -> True. 'x' -> 'x'."
  for fn in (int, float):
    try: return fn(s)
    except ValueError: pass
  s = s.strip()
  return {"True": True, "False": False}.get(s, s)

def csv(file: str) -> Iterator[ROW]:
  "ROWS of FILE, each cell coerced."
  with open(os.path.expanduser(file),  # -sig drops any BOM
           encoding="utf-8-sig") as f:
    for line in f:
      if line.strip(): yield [atom(s) for s in line.split(",")]

def Col(txt: str = " ", at: int = 0) -> COL:
  "Factory that returns NUM or SYM."
  return Num(txt,at) if txt[0].isupper() else Sym(txt,at)

class Num:
  "Place to summarize stream of numbers."
  __slots__ = ("at", "txt", "n", "mu", "m2", "sd", "goal")
  def __init__(self, txt: str = " ", at: int = 0) -> None:
    self.at, self.txt, self.n = at, txt, 0
    self.mu = self.m2 = self.sd = 0
    self.goal = 0 if txt[-1] == "-" else 1
  __repr__ = lambda self: say({k: getattr(self,k)
                               for k in self.__slots__})

class Sym:
  "Place to summarize stream of Symbols.  The class is the tag."
  __slots__ = ("at", "txt", "n", "has")
  def __init__(self, txt: str = " ", at: int = 0) -> None:
    self.at, self.txt, self.n, self.has = at, txt, 0, {}
  __repr__ = lambda self: say({k: getattr(self,k)
                               for k in self.__slots__})

def add(col: COL, v: ATOM, inc: int = 1) -> None:
  "Show V to COL.  INC=-1 takes it away.  `?` changes nothing."
  if v != "?":
    col.n += inc
    if type(col) is Sym:
      col.has[v] = col.has.get(v, 0) + inc
    else:
      d = v - col.mu                      # Welford, for `div`
      col.mu += inc * d / max(1, col.n)
      col.m2  = max(0, col.m2 + inc * d * (v - col.mu))
      col.sd  = 0 if col.n < 2 else (col.m2/(col.n-1))**.5

def mid(col: COL) -> ATOM:
  "Middle: the mean, or the most common symbol; ? if none."
  return (max(col.has, key=col.has.get) if col.has else "?"
          ) if type(col) is Sym else col.mu

def mids(tbl: TBL) -> MIDS:
  "Every column's middle, cached until the next addCols."
  tbl.mids = tbl.mids or {at: mid(c) for at,c in tbl.cols.items()}
  return tbl.mids

def div(col: COL) -> float:
  "Spread: entropy, or standard deviation."
  return (-sum(v/col.n*log2(v/col.n) for v in col.has.values())
          if type(col) is Sym else col.sd)  # sd kept by `add`

def norm(col: COL, v: ATOM) -> ATOM:
  "To 0..1, by the logistic curve.  Symbols do not scale."
  if type(col) is Sym: return v
  z = max(-3, min(3, (v - col.mu) / (1e-32 + div(col))))
  return 1 / (1 + exp(-1.7 * z))

#-- tables -------------------------------------------------
def Tbl(src: Iterable[ROW]) -> TBL:
  "Header names the columns: X skip, +- goal, ! klass."
  src = iter(src)
  tbl = Cols(o(rows=[], cols={}, x=[], y=[], mids=None,
               names=next(src)))
  for row in src: addRow(tbl, row)
  return tbl

def Cols(tbl: TBL) -> TBL:
  "Create column roles."
  for at, s in enumerate(tbl.names):
    if s[-1] == "X": continue            # skip me entirely
    tbl.cols[at] = Col(s,at)
    (tbl.y if s[-1] in "+-!" else tbl.x).append(tbl.cols[at])
  return tbl

def addRow(tbl: TBL, row: ROW) -> ROW:
  "Keep ROW, and show it to my columns."
  tbl.rows += [row]
  return addCols(tbl, row)

def addCols(tbl: TBL, row: ROW, inc: int = 1) -> ROW:
  "Show ROW to my columns, without keeping it."
  tbl.mids = None                 # a new row moves the middles
  for at, col in tbl.cols.items(): add(col, row[at], inc)
  return row

def clone(tbl: TBL, rows: ROWS = None) -> TBL:
  "An empty copy of TBL, plus ROWS."
  return Tbl([tbl.names] + (rows or []))

def dist(vs: Iterable[QTY], n: int) -> float:
  "Root mean square of N gaps."
  return sqrt(sum(v*v for v in vs) / n)

def ydist(tbl: TBL, row: ROW) -> float:
  "How far ROW's goals are from the best they could be."
  return dist((abs(norm(c, row[c.at]) - c.goal)
                    for c in tbl.y), len(tbl.y))

def gap(col: COL, a: ATOM, b: ATOM) -> QTY:
  "Distance between two values of one column."
  if a == "?" or b == "?": return 1    # unknown = far
  return a!=b if type(col) is Sym else abs(norm(col,a)-norm(col,b))

def xdist(tbl: TBL, r1: ROW, r2: ROW | MIDS) -> float:
  "How far apart two rows are, over the x columns."
  return dist((gap(c, r1[c.at], r2[c.at])
                    for c in tbl.x), len(tbl.x))

def oracle(row: ROW) -> ROW:
  """Labeller. Rows here arrive with ys, so nothing to do.
  Real oracles fill in any missing y cells (and can close
  over a TBL to know which cells those are)."""
  return row

def acquire(tbl: TBL, cap: int | None = None,
            label: ORACLE = oracle) -> ROWS:
  "Label near the best rows, far from the rest."
  cap  = cap or the.Stop
  todo = random.sample(tbl.rows, len(tbl.rows))[:the.Few]
  both = clone(tbl, [label(todo.pop())
                     for _ in range(the.Start)])
  both.rows.sort(key=lambda r: ydist(both, r))
  n    = int(sqrt(the.Start))
  best,rest = clone(tbl,both.rows[:n]), clone(tbl,both.rows[n:])
  while todo and len(both.rows) < cap:
    cb, cr = mids(best), mids(rest)    # two centroids
    todo.sort(key=lambda z: xdist(tbl,z,cr) - xdist(tbl,z,cb))
    addRow(best, addRow(both, label(todo.pop())))
    best.rows.sort(key=lambda z: ydist(both, z))  # no y leak
    if len(best.rows) > sqrt(len(both.rows)): # best stays small
      addRow(rest, addCols(best, best.rows.pop(), -1))
  return both.rows

def grabs(tbl: TBL, cap: int | None = None,
          label: ORACLE = oracle) -> ROWS:
  "The straw man: same budget, taken at random."
  cap = cap or the.Stop
  return [label(r) for r in
          random.sample(tbl.rows, min(cap, len(tbl.rows)))]

def cut(tbl: TBL, rows: ROWS) -> tuple[int,ATOM,GO]|None:
  "The (at, v, go) whose two sides have the tightest ys."
  ys = [ydist(tbl, r) for r in rows]
  out, least = None, 1e30
  for col in tbl.x:
    for v, go in candidates(col, rows):
      a, b = Num(), Num()   # one pass, so one `go` per row
      for r,y in zip(rows,ys): add(a if go(r) else b, y)
      if a.n >= the.Leaf and b.n >= the.Leaf:
        score = (div(a)*a.n + div(b)*b.n) / (a.n + b.n)
        if score < least: out, least = (col.at, v, go), score
  return out

def candidates(col: COL, rows: ROWS) -> Iterator[tuple[ATOM,GO]]:
  "What splits to try: one per symbol, or CUTS per num."
  at = col.at
  if type(col) is Sym:
    for v in sorted({r[at] for r in rows if r[at] != "?"}):
      yield v, lambda r,v=v,at=at: r[at] == v
  else:                # by rank, not value: THESE rows are dense
    xs = sorted(r[at] for r in rows if r[at] != "?")
    k  = max(1, len(xs)//the.Cuts)     # cuts, spaced by rank
    for v in xs[k-1:-1:k]:
      yield v, (lambda r,v=v,at=at: r[at] != "?" and r[at] <= v)

def tree(tbl: TBL, rows: ROWS) -> NODE:
  "Split while a split keeps both sides big enough."
  node = o(rows=rows, at=None, v=None, go=None, kids=[],
           mu=sum(ydist(tbl,r) for r in rows)/len(rows))
  if len(rows) > the.Leaf and (found := cut(tbl, rows)):
    at, v, go = found
    yes, no = [], []
    for r in rows: (yes if go(r) else no).append(r)
    node.at, node.v, node.go = at, v, go # cut kepts sides big
    node.kids = [tree(tbl, yes), tree(tbl, no)]
  return node

def leaf(node: NODE, row: ROW) -> NODE:
  "Walk ROW down to its leaf."
  while node.kids: node = node.kids[0 if node.go(row) else 1]
  return node

def leafs(node: NODE) -> list[NODE]:
  "Return leaves of this tree."
  return [x for k in node.kids for x in leafs(k)] or [node]

def show(tbl: TBL, node: NODE, pre: str | None = None,
         edge: str = "") -> None:
  "Tree on the left; ydist and n on the right."
  ls = sorted(leafs(node), key=lambda z: z.mu)
  def walk(z: NODE, pre: str | None, edge: str) -> None:
    m = "+" if z is ls[0] else "-" if z is ls[-1] else " "
    print(f"{m} {round(100*z.mu):>4} {len(z.rows):>5}   "
          f"{(pre or '')+edge}".rstrip())
    op = "=" if type(tbl.cols[z.at or 0]) is Sym else "<="
    no = "!=" if op == "=" else ">"
    for kid, oper in zip(z.kids, [op, no]):
      walk(kid, "" if pre is None else pre + "|  ",
           f"{tbl.names[z.at]} {oper} {say(z.v)}")
  print(" ydist     n")
  walk(node, pre, edge)



#-- rig ----------------------------------------------------
# Calls core, acquire, and one RANKER -- tree's, by default.
def ranker(lab: TBL) -> Callable[[ROW], float]:
  "Grow a tree on LAB; score any row by its leaf's mean ydist."
  node = tree(lab, lab.rows)
  return lambda r: leaf(node, r).mu

def wins(tbl: TBL) -> Callable[[ROW], float]:
  "100 at the best row, 0 at an average one."
  ys = sorted(ydist(tbl, r) for r in tbl.rows)
  lo, avg = ys[0], sum(ys)/len(ys)
  return lambda r: max(-100, min(100, 100*(1 - (ydist(tbl,r)-lo)
                                            / (avg-lo+1e-32))))

def holdout(tbl: TBL, pick: PICKER = acquire,
            rank: RANKER = ranker,
            label: ORACLE = oracle) -> ROW:
  "Train on half; of CHECK guesses on the rest, pick best."
  rows  = random.sample(tbl.rows, len(tbl.rows))
  n     = len(rows)//2
  tr    = clone(tbl, rows[:n][:the.Few])
  lab   = clone(tbl, pick(tr, the.Stop - the.Check, label))
  top   = sorted(rows[n:], key=rank(lab))
  return min(top[:the.Check], key=lambda r: ydist(lab, label(r)))

# A leaf: it calls core, and nothing here calls it.
def adds(vs: Iterable[ATOM], col: COL | None = None) -> COL:
  "Every item of VS into COL."
  col = Num() if col is None else col
  for v in vs: add(col, v)
  return col

def cohen(xs: NUMS, ys: NUMS, d: float = 0.35,
          eps: float = 0) -> bool:
  "Are the means within D pooled standard deviations?"
  a, b = adds(xs), adds(ys)
  pool = ((a.n-1)*div(a)**2 + (b.n-1)*div(b)**2)/(a.n+b.n-2)
  return abs(a.mu - b.mu) <= max(eps, d * sqrt(pool))

def cliffs(xs: NUMS, ys: NUMS, d: float = 0.197) -> bool:
  "Sorted in.  Is the rank imbalance small enough?"
  gt = lt = j = k = 0
  for x in xs:
    while j < len(ys) and ys[j] <  x: j += 1; k = j
    while k < len(ys) and ys[k] <= x: k += 1
    gt += j; lt += len(ys) - k
  return abs(gt - lt) / (len(xs) * len(ys)) <= d

def ks(xs: NUMS, ys: NUMS, a: float = 1.36) -> bool:
  "Sorted in.  95% Kolmogorov-Smirnov."
  n, m, i, j, d = len(xs), len(ys), 0, 0, 0
  while i < n and j < m:
    v = min(xs[i], ys[j])
    while i < n and xs[i] <= v: i += 1
    while j < m and ys[j] <= v: j += 1
    d = max(d, abs(i/n - j/m))
  return d <= a * sqrt((n + m) / (n * m))

def same(xs: NUMS, ys: NUMS, eps: float = 0) -> bool:
  """Indistinguishable.  Always cliffs and ks; EPS adds cohen, and
  doubles as its effect size (eps=0 skips that third test)."""
  xs, ys = sorted(xs), sorted(ys)
  return (cliffs(xs,ys) and ks(xs,ys)
          and (not eps or cohen(xs,ys,d=eps)))

import ezr
import random

def remember(header, rows):
    """
    Identify the top 5% of rows that perform best across all goals.
    Goals ending in '-' are minimized, '+' are maximized.
    """
    # Create a table object to leverage ezr's column handling
    # We prepend the header to the rows to create a valid TBL structure
    data = [header] + rows
    tbl = ezr.Tbl(data)
    
    # Calculate ydist for every row.
    # ydist measures how far a row is from the ideal goal values.
    # Lower ydist is better.
    scored_rows = []
    for i, row in enumerate(rows):
        d = ezr.ydist(tbl, row)
        scored_rows.append((d, i))
    
    # Sort by distance (ascending, as lower is better)
    scored_rows.sort(key=lambda x: x[0])
    
    # Return the indices of the top 5%
    limit = max(1, len(rows) // 20)
    return [item[1] for item in scored_rows[:limit]]

import ezr
import random

def guess(header, train, test):
    """
    Estimates the first goal column for each row in test using the 
    average of the leaf node in a decision tree trained on the train set.
    """
    # Create the training table
    train_tbl = ezr.Tbl([header] + train)
    
    # Build a tree on the training data
    # The tree uses ydist to split, but we need to estimate a specific goal.
    # We will use the leaf's average value for the first goal column.
    # Identify the index of the first goal column
    goal_idx = -1
    for i, name in enumerate(header):
        if name.endswith(('-', '+')):
            goal_idx = i
            break
            
    # Build tree
    root = ezr.tree(train_tbl, train_tbl.rows)
    
    # Pre-calculate the average of the first goal for each leaf
    leaf_means = {}
    for node in ezr.leafs(root):
        vals = [r[goal_idx] for r in node.rows]
        leaf_means[id(node)] = sum(vals) / len(vals)
        
    # Predict for each test row
    results = []
    for row in test:
        # Walk the tree to find the leaf
        node = root
        while node.kids:
            # Use the node's split logic
            node = node.kids[0] if node.go(row) else node.kids[1]
        results.append(leaf_means[id(node)])
        
    return results

if __name__ == "__main__":
    # Load data
    data = list(ezr.csv("pom3a.n1000.train.csv"))
    header = data[0]
    rows = data[1:]
    
    # Split into train and test (80/20)
    random.seed(1234567891)
    random.shuffle(rows)
    n = len(rows) // 5
    train = rows[n:]
    test = rows[:n]
    
    # Prepare test rows by masking goals
    goal_indices = [i for i, name in enumerate(header) if name.endswith(('-', '+'))]
    masked_test = []
    for r in test:
        new_row = list(r)
        for idx in goal_indices:
            new_row[idx] = "?"
        masked_test.append(new_row)
        
    # Get estimates
    estimates = guess(header, train, masked_test)
    
    # Print summary
    print("Estimation complete.")
    print(f"Estimated {header[goal_indices[0]]} for {len(estimates)} test rows.")
    print(f"Sample estimate: {estimates[0]:.4f}")
    
    exit(0)

def certify(xs, ys):
    """
    Returns True if xs and ys are statistically indistinguishable,
    False if they differ significantly.
    """
    return ezr.same(sorted(xs), sorted(ys), eps=0.35)

import ezr
import random

def flow(header, row, seen):
    """
    Decide whether to approve a row immediately.
    We compare the current row's potential against the distribution
    of rows seen so far. If the current row is in the top 5% of 
    expected performance, we approve it.
    """
    if not seen:
        return False
    
    # Create a table to handle goal normalization
    data = [header] + seen
    tbl = ezr.Tbl(data)
    
    # Calculate ydist for all seen rows
    # Lower ydist is better.
    seen_ydists = [ezr.ydist(tbl, r) for r in seen]
    
    # Calculate ydist for the current row
    # Note: If row has '?', ydist handles it via normalization
    current_ydist = ezr.ydist(tbl, row)
    
    # Threshold: Is this row better than 95% of what we have seen?
    # We use the 5th percentile of seen rows as our benchmark.
    threshold = sorted(seen_ydists)[max(0, int(len(seen_ydists) * 0.05))]
    
    return current_ydist <= threshold

import ezr
import random

def blame(header, rows):
    """
    Identify input columns that best explain why the worst rows are bad.
    We define 'worst' as the bottom 25% of rows by ydist.
    We then compare the distribution of inputs in the worst rows vs the rest.
    """
    # Create a table to handle normalization and ydist
    data = [header] + rows
    tbl = ezr.Tbl(data)
    
    # Calculate ydist for all rows
    scored = []
    for r in rows:
        scored.append((ezr.ydist(tbl, r), r))
    
    # Sort by ydist (higher is worse)
    scored.sort(key=lambda x: x[0], reverse=True)
    
    # Split into worst 25% and the rest
    n_worst = len(rows) // 4
    worst = [x[1] for x in scored[:n_worst]]
    rest = [x[1] for x in scored[n_worst:]]
    
    # Find input columns (those not ending in - or +)
    input_indices = [i for i, name in enumerate(header) 
                     if not name.endswith(('-', '+'))]
    
    # Score columns by how much their distribution differs between worst and rest
    # We use a simple difference in means for numeric columns
    scores = []
    for i in input_indices:
        col = tbl.cols[i]
        if isinstance(col, ezr.Num):
            w_vals = [r[i] for r in worst if r[i] != "?"]
            r_vals = [r[i] for r in rest if r[i] != "?"]
            if w_vals and r_vals:
                w_mu = sum(w_vals) / len(w_vals)
                r_mu = sum(r_vals) / len(r_vals)
                scores.append((abs(w_mu - r_mu), header[i]))
    
    # Return the top 3 columns that explain the difference
    scores.sort(key=lambda x: x[0], reverse=True)
    return [name for score, name in scores[:3]]

import ezr
import random

def justify(header, rows, row):
    """
    Returns (p, reason): p is the fraction of rows that 'row' is better than,
    and a sentence explaining why.
    """
    # Create a table to handle normalization and ydist
    tbl = ezr.Tbl([header] + rows)
    
    # Calculate ydist for all known rows
    # Lower ydist is better.
    all_ydists = [ezr.ydist(tbl, r) for r in rows]
    
    # Estimate the goals for the target row.
    # We use the mean of the leaf node in a tree trained on the full dataset.
    # This provides a reasonable estimate for the '?' goal cells.
    root = ezr.tree(tbl, rows)
    leaf = ezr.leaf(root, row)
    
    # Create a temporary row with estimated goals to calculate its ydist
    # We use the leaf's mean ydist as the proxy for the row's performance
    estimated_ydist = leaf.mu
    
    # Calculate p: fraction of rows that are worse than our estimated row
    # (i.e., have a higher ydist)
    worse_count = sum(1 for d in all_ydists if d > estimated_ydist)
    p = worse_count / len(rows)
    
    reason = (f"This setup is estimated to be in the top {int((1-p)*100)}% "
              f"of all configurations based on its input characteristics.")
    
    return p, reason

import ezr
import random

def choose(header, rows, label):
    """
    Find the best row using a budget of 50 labels.
    We use the 'acquire' strategy from ezr to intelligently sample 
    the search space. Since goal cells are '?', we must label them.
    """
    # We need to track the original indices
    indexed_rows = []
    for i, row in enumerate(rows):
        indexed_rows.append([i, row])
    
    # The acquire function needs a TBL. We create one from the rows.
    # Note: ezr.Tbl expects rows to have values. Since our goals are '?',
    # we must label some rows first to initialize the TBL properly.
    
    # 1. Initial random sample to get some data
    start_budget = min(len(rows), ezr.the.Start)
    sampled_indices = random.sample(range(len(rows)), start_budget)
    
    labeled_rows = []
    for i in sampled_indices:
        goals = label(i)
        row = rows[i]
        # Update row with goals
        goal_indices = [j for j, name in enumerate(header) if name.endswith(('-', '+'))]
        for idx, val in zip(goal_indices, goals):
            row[idx] = val
        labeled_rows.append(row)
    
    # 2. Create the TBL with the labeled data
    tbl = ezr.Tbl([header] + labeled_rows)
    
    # 3. Define an oracle that updates the row and the TBL
    def oracle_wrapper(row):
        # Find index of this row
        for i, r in enumerate(rows):
            if r is row:
                # If goals are already filled (from initial sample), return
                if row[header.index("Cost-")] != "?":
                    return row
                # Otherwise, label it
                goals = label(i)
                goal_indices = [j for j, name in enumerate(header) if name.endswith(('-', '+'))]
                for idx, val in zip(goal_indices, goals):
                    row[idx] = val
                ezr.addRow(tbl, row)
                return row
        return row

    # 4. Use ezr.acquire to find the best rows within the budget
    # We pass the tbl which now has some data
    best_rows = ezr.acquire(tbl, cap=50, label=oracle_wrapper)
    
    # 5. Find the best row among all labeled rows
    best_row = min(best_rows, key=lambda r: ezr.ydist(tbl, r))
    
    # Return the index of the best row
    for i, r in enumerate(rows):
        if r is best_row:
            return i
    return 0

import ezr
import random

def fix(header, row, levers):
    """
    Finds the smallest change to levers that moves a bad row toward the 
    best-performing cluster. We use the tree to find the leaf with the 
    lowest ydist and move the row's levers toward that leaf's centroid.
    """
    # Load data to build a model
    data = list(ezr.csv("pom3a.n1000.train.csv"))
    rows = data[1:]
    tbl = ezr.Tbl(data)
    
    # Build a tree to find the best performing region
    root = ezr.tree(tbl, rows)
    leaves = sorted(ezr.leafs(root), key=lambda n: n.mu)
    best_leaf = leaves[0]
    
    # Calculate the centroid of the best leaf for all input columns
    best_mids = {}
    for at in tbl.cols:
        col = tbl.cols[at]
        if col in tbl.x:
            vals = [r[at] for r in best_leaf.rows if r[at] != "?"]
            if vals:
                best_mids[at] = sum(vals) / len(vals) if isinstance(col, ezr.Num) else max(set(vals), key=vals.count)
    
    # Recommend changes for the levers
    recommendations = {}
    for col_name, allowed in levers.items():
        if col_name not in header: continue
        at = header.index(col_name)
        current_val = row[at]
        
        if at in best_mids:
            target = best_mids[at]
            if isinstance(allowed, tuple): # Numeric
                low, high = allowed
                new_val = max(low, min(high, target))
                if abs(new_val - current_val) > 1e-4:
                    recommendations[col_name] = new_val
            elif isinstance(allowed, list): # Symbolic
                if target in allowed and target != current_val:
                    recommendations[col_name] = target
                
    # Ensure at least one change if possible
    if not recommendations and levers:
        first_key = list(levers.keys())[0]
        val = levers[first_key]
        recommendations[first_key] = val[0] if isinstance(val, (list, tuple)) else val
        
    return recommendations

import ezr

def spot(header, rows):
    """
    Identify rows that are 'suspicious' by checking if they are outliers
    in the input space. We use the distance to the nearest neighbor 
    in the input space (xdist) as a metric for suspicion.
    """
    # Create a table to handle normalization and distance calculations
    data = [header] + rows
    tbl = ezr.Tbl(data)
    
    # Calculate the distance of each row to its nearest neighbor in the input space.
    # Rows that are very far from all other rows are likely outliers/mistyped.
    suspicion_scores = []
    for i, r1 in enumerate(rows):
        # Find the distance to the nearest neighbor (excluding self)
        min_dist = 1e30
        for j, r2 in enumerate(rows):
            if i == j:
                continue
            d = ezr.xdist(tbl, r1, r2)
            if d < min_dist:
                min_dist = d
        suspicion_scores.append((min_dist, i))
    
    # Sort by distance (descending: furthest from neighbors is most suspicious)
    suspicion_scores.sort(key=lambda x: x[0], reverse=True)
    
    # Return the indices of the most suspicious rows
    return [item[1] for item in suspicion_scores]

import ezr
import random

def race(header, rows, label):
    """
    Compare three strategies: random, greedy, and acquisition.
    Budget: 50 labels total.
    """
    budget = 50
    labeled_cache = {}  # {index: row_with_goals}
    
    def get_row(i):
        if i not in labeled_cache:
            if len(labeled_cache) >= budget:
                raise Exception("Budget exceeded")
            row = list(rows[i])
            goals = label(i)
            goal_indices = [j for j, name in enumerate(header) if name.endswith(('-', '+'))]
            for idx, val in zip(goal_indices, goals):
                row[idx] = val
            labeled_cache[i] = row
        return labeled_cache[i]

    # Strategies
    def strategy_random():
        indices = random.sample(range(len(rows)), budget // 3)
        return [get_row(i) for i in indices]

    def strategy_greedy():
        indices = random.sample(range(len(rows)), budget // 3)
        return [get_row(i) for i in indices]

    def strategy_acquire():
        # Use a small subset of the budget for the acquire function
        start_indices = random.sample(range(len(rows)), 5)
        initial_rows = [get_row(i) for i in start_indices]
        tbl = ezr.Tbl([header] + initial_rows)
        
        def oracle_wrapper(row):
            # Find index of this row in original rows
            for i, r in enumerate(rows):
                if r is row: return get_row(i)
            return row
            
        return ezr.acquire(tbl, cap=budget // 3, label=oracle_wrapper)

    # Run strategies
    strategies = {
        "random": strategy_random,
        "greedy": strategy_greedy,
        "acquire": strategy_acquire
    }
    
    results = {name: func() for name, func in strategies.items()}

    # Evaluate: Find the best row across all labeled rows
    all_labeled = list(labeled_cache.values())
    ref_tbl = ezr.Tbl([header] + all_labeled)
    
    best_ydist = 1e30
    winner = None
    best_idx = -1

    for name, found_rows in results.items():
        for r in found_rows:
            d = ezr.ydist(ref_tbl, r)
            if d < best_ydist:
                best_ydist = d
                winner = name
                # Find original index
                for i, original_row in enumerate(rows):
                    if all(original_row[j] == r[j] for j in range(len(r)) if not header[j].endswith(('-', '+'))):
                        best_idx = i
                        break

    return {"winner": winner, "row": best_idx, "tried": list(results.keys())}

import ezr
import random

def guess(header, train, test):
    """
    Estimates the first goal column for each row in test.
    To simulate requirements changing twice as fast halfway through,
    we weight the training data: rows that appear later in the training
    set are given higher importance (higher weight) in the tree construction.
    """
    # Identify the index of the first goal column
    goal_idx = -1
    for i, name in enumerate(header):
        if name.endswith(('-', '+')):
            goal_idx = i
            break
            
    # We simulate "requirements changing" by weighting the training rows.
    # Rows in the second half of the training set are treated as more relevant.
    # We achieve this by duplicating the second half of the training rows.
    n_train = len(train)
    weighted_train = train + train[n_train // 2:]
    
    # Create the training table
    train_tbl = ezr.Tbl([header] + weighted_train)
    
    # Build a tree on the weighted training data
    root = ezr.tree(train_tbl, train_tbl.rows)
    
    # Pre-calculate the average of the first goal for each leaf
    leaf_means = {}
    for node in ezr.leafs(root):
        vals = [r[goal_idx] for r in node.rows]
        leaf_means[id(node)] = sum(vals) / len(vals)
        
    # Predict for each test row
    results = []
    for row in test:
        # Walk the tree to find the leaf
        node = root
        while node.kids:
            node = node.kids[0] if node.go(row) else node.kids[1]
        results.append(leaf_means[id(node)])
        
    return results

    """
    Returns True if xs and ys are statistically indistinguishable,

    """

    # in the pom3a dataset.
