import ezr
import random

def blame(header, rows):
    """
    Identify input columns that best explain why the worst rows are bad.
    We define 'worst' as the bottom 25% of rows based on ydist.
    We then compare the distribution of input values in the worst rows
    against the rest of the data to find the most influential features.
    """
    # Create a table to handle normalization and ydist calculations
    tbl = ezr.Tbl([header] + rows)
    
    # Score all rows
    scored = []
    for row in rows:
        scored.append((ezr.ydist(tbl, row), row))
    
    # Sort by ydist (higher is worse)
    scored.sort(key=lambda x: x[0], reverse=True)
    
    # Split into worst 25% and the rest
    n_worst = len(rows) // 4
    worst = [x[1] for x in scored[:n_worst]]
    rest = [x[1] for x in scored[n_worst:]]
    
    # Identify input columns (exclude goals)
    input_indices = [i for i, h in enumerate(header) if not (h.endswith('-') or h.endswith('+'))]
    
    # For each input column, calculate the difference in means (or mode)
    # between the worst rows and the rest.
    scores = []
    for i in input_indices:
        col = tbl.cols[i]
        # Get values for this column
        w_vals = [r[i] for r in worst if r[i] != "?"]
        r_vals = [r[i] for r in rest if r[i] != "?"]
        
        if not w_vals or not r_vals:
            continue
            
        # Calculate a simple effect size (difference in means/proportions)
        if isinstance(col, ezr.Num):
            w_avg = sum(w_vals) / len(w_vals)
            r_avg = sum(r_vals) / len(r_vals)
            diff = abs(w_avg - r_avg)
        else:
            # For symbols, compare frequency of the most common value
            w_mode = max(set(w_vals), key=w_vals.count)
            r_mode = max(set(r_vals), key=r_vals.count)
            diff = 1.0 if w_mode != r_mode else 0.0
            
        scores.append((header[i], diff))
    
    # Sort by impact and return the top 3 column names
    scores.sort(key=lambda x: x[1], reverse=True)
    return [s[0] for s in scores[:3]]

if __name__ == "__main__":
    file_path = "pom3a.n1000.train.csv"
    rows_gen = ezr.csv(file_path)
    header = next(rows_gen)
    rows = list(rows_gen)
    
    top_blame = blame(header, rows)
    
    print("Analysis: Why do our worst setups leave so many people waiting around?")
    print(f"The primary factors contributing to poor performance are: {', '.join(top_blame)}.")
    print("These inputs show the strongest correlation with high 'Idle-' scores.")
