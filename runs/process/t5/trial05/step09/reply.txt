import ezr
import random

def spot(header, rows):
    """
    Identify rows that are 'suspicious' by checking if their input values
    are outliers relative to the rest of the data. We use the distance
    from the median of each column as a measure of suspicion.
    """
    # Create a table to handle column statistics
    tbl = ezr.Tbl([header] + rows)
    
    # Identify input columns (exclude goals)
    input_indices = [i for i, h in enumerate(header) if not (h.endswith('-') or h.endswith('+'))]
    
    # Calculate the median for each input column
    medians = {}
    for i in input_indices:
        col = tbl.cols[i]
        vals = [r[i] for r in rows if r[i] != "?"]
        if not vals:
            continue
        if isinstance(col, ezr.Num):
            sorted_vals = sorted(vals)
            medians[i] = sorted_vals[len(sorted_vals) // 2]
        else:
            # For symbols, use the mode
            medians[i] = max(set(vals), key=vals.count)
            
    # Score rows based on how far they deviate from the medians
    # A higher deviation score indicates a more "suspicious" row
    suspicion_scores = []
    for idx, row in enumerate(rows):
        score = 0
        for i in input_indices:
            val = row[i]
            if val == "?" or i not in medians:
                continue
            
            col = tbl.cols[i]
            if isinstance(col, ezr.Num):
                # Normalize the distance from the median
                diff = abs(val - medians[i])
                # Divide by range if possible, else use raw diff
                score += diff
            else:
                # Symbolic: 1 if different from mode, 0 otherwise
                if val != medians[i]:
                    score += 1
        suspicion_scores.append((score, idx))
        
    # Sort by suspicion score descending
    suspicion_scores.sort(key=lambda x: x[0], reverse=True)
    
    # Return the indices of the top 10 most suspicious rows
    return [item[1] for item in suspicion_scores[:10]]

if __name__ == "__main__":
    file_path = "pom3a.n1000.train.csv"
    rows_gen = ezr.csv(file_path)
    header = next(rows_gen)
    rows = list(rows_gen)
    
    suspicious_indices = spot(header, rows)
    
    print("Anomaly Detection Report:")
    print(f"Identified {len(suspicious_indices)} potentially mistyped or anomalous records.")
    print("Most suspicious row indices (descending order of suspicion):")
    print(suspicious_indices)
    
    print("\nSample of suspicious row data:")
    for idx in suspicious_indices[:3]:
        print(f"Row {idx}: {rows[idx]}")
