def say(x: Any, p: int = 2) -> str:
  "Floats get P decimals; dicts list their keys."
  if isinstance(x,float): x= f"{x:.{p}f}".rstrip("0").rstrip(".")
  if isinstance(x, dict):
    x= "{"+", ".join(f"{k}: {say(v,p)}" for k,v in x.items())+"}"
  return str(x)

class o(dict):
  "A dict you can poke with a dot."
  __getattr__, __setattr__ = dict.__getitem__, dict.__setitem__
  __repr__ = say

def atom(s: str) -> ATOM:
  "'22' -> 22. '3.1' -> 3.1. 'True' -> True. 'x' -> 'x'."
  for fn in (int, float):
    try: return fn(s)
    except ValueError: pass
  s = s.strip()
  return {"True": True, "False": False}.get(s, s)

def csv(file: str) -> Iterator[ROW]:
  "ROWS of FILE, each cell coerced."
  with open(os.path.expanduser(file),  # -sig drops any BOM
           encoding="utf-8-sig") as f:
    for line in f:
      if line.strip(): yield [atom(s) for s in line.split(",")]

def Col(txt: str = " ", at: int = 0) -> COL:
  "Factory that returns NUM or SYM."
  return Num(txt,at) if txt[0].isupper() else Sym(txt,at)

class Num:
  "Place to summarize stream of numbers."
  __slots__ = ("at", "txt", "n", "mu", "m2", "sd", "goal")
  def __init__(self, txt: str = " ", at: int = 0) -> None:
    self.at, self.txt, self.n = at, txt, 0
    self.mu = self.m2 = self.sd = 0
    self.goal = 0 if txt[-1] == "-" else 1
  __repr__ = lambda self: say({k: getattr(self,k)
                               for k in self.__slots__})

class Sym:
  "Place to summarize stream of Symbols.  The class is the tag."
  __slots__ = ("at", "txt", "n", "has")
  def __init__(self, txt: str = " ", at: int = 0) -> None:
    self.at, self.txt, self.n, self.has = at, txt, 0, {}
  __repr__ = lambda self: say({k: getattr(self,k)
                               for k in self.__slots__})

def add(col: COL, v: ATOM, inc: int = 1) -> None:
  "Show V to COL.  INC=-1 takes it away.  `?` changes nothing."
  if v != "?":
    col.n += inc
    if type(col) is Sym:
      col.has[v] = col.has.get(v, 0) + inc
    else:
      d = v - col.mu                      # Welford, for `div`
      col.mu += inc * d / max(1, col.n)
      col.m2  = max(0, col.m2 + inc * d * (v - col.mu))
      col.sd  = 0 if col.n < 2 else (col.m2/(col.n-1))**.5

def mid(col: COL) -> ATOM:
  "Middle: the mean, or the most common symbol; ? if none."
  return (max(col.has, key=col.has.get) if col.has else "?"
          ) if type(col) is Sym else col.mu

def mids(tbl: TBL) -> MIDS:
  "Every column's middle, cached until the next addCols."
  tbl.mids = tbl.mids or {at: mid(c) for at,c in tbl.cols.items()}
  return tbl.mids

def div(col: COL) -> float:
  "Spread: entropy, or standard deviation."
  return (-sum(v/col.n*log2(v/col.n) for v in col.has.values())
          if type(col) is Sym else col.sd)  # sd kept by `add`

def norm(col: COL, v: ATOM) -> ATOM:
  "To 0..1, by the logistic curve.  Symbols do not scale."
  if type(col) is Sym: return v
  z = max(-3, min(3, (v - col.mu) / (1e-32 + div(col))))
  return 1 / (1 + exp(-1.7 * z))

#-- tables -------------------------------------------------
def Tbl(src: Iterable[ROW]) -> TBL:
  "Header names the columns: X skip, +- goal, ! klass."
  src = iter(src)
  tbl = Cols(o(rows=[], cols={}, x=[], y=[], mids=None,
               names=next(src)))
  for row in src: addRow(tbl, row)
  return tbl

def Cols(tbl: TBL) -> TBL:
  "Create column roles."
  for at, s in enumerate(tbl.names):
    if s[-1] == "X": continue            # skip me entirely
    tbl.cols[at] = Col(s,at)
    (tbl.y if s[-1] in "+-!" else tbl.x).append(tbl.cols[at])
  return tbl

def addRow(tbl: TBL, row: ROW) -> ROW:
  "Keep ROW, and show it to my columns."
  tbl.rows += [row]
  return addCols(tbl, row)

def addCols(tbl: TBL, row: ROW, inc: int = 1) -> ROW:
  "Show ROW to my columns, without keeping it."
  tbl.mids = None                 # a new row moves the middles
  for at, col in tbl.cols.items(): add(col, row[at], inc)
  return row

def clone(tbl: TBL, rows: ROWS = None) -> TBL:
  "An empty copy of TBL, plus ROWS."
  return Tbl([tbl.names] + (rows or []))

def dist(vs: Iterable[QTY], n: int) -> float:
  "Root mean square of N gaps."
  return sqrt(sum(v*v for v in vs) / n)

def ydist(tbl: TBL, row: ROW) -> float:
  "How far ROW's goals are from the best they could be."
  return dist((abs(norm(c, row[c.at]) - c.goal)
                    for c in tbl.y), len(tbl.y))

def gap(col: COL, a: ATOM, b: ATOM) -> QTY:
  "Distance between two values of one column."
  if a == "?" or b == "?": return 1    # unknown = far
  return a!=b if type(col) is Sym else abs(norm(col,a)-norm(col,b))

def xdist(tbl: TBL, r1: ROW, r2: ROW | MIDS) -> float:
  "How far apart two rows are, over the x columns."
  return dist((gap(c, r1[c.at], r2[c.at])
                    for c in tbl.x), len(tbl.x))

def oracle(row: ROW) -> ROW:
  """Labeller. Rows here arrive with ys, so nothing to do.
  Real oracles fill in any missing y cells (and can close
  over a TBL to know which cells those are)."""
  return row

def acquire(tbl: TBL, cap: int | None = None,
            label: ORACLE = oracle) -> ROWS:
  "Label near the best rows, far from the rest."
  cap  = cap or the.Stop
  todo = random.sample(tbl.rows, len(tbl.rows))[:the.Few]
  both = clone(tbl, [label(todo.pop())
                     for _ in range(the.Start)])
  both.rows.sort(key=lambda r: ydist(both, r))
  n    = int(sqrt(the.Start))
  best,rest = clone(tbl,both.rows[:n]), clone(tbl,both.rows[n:])
  while todo and len(both.rows) < cap:
    cb, cr = mids(best), mids(rest)    # two centroids
    todo.sort(key=lambda z: xdist(tbl,z,cr) - xdist(tbl,z,cb))
    addRow(best, addRow(both, label(todo.pop())))
    best.rows.sort(key=lambda z: ydist(both, z))  # no y leak
    if len(best.rows) > sqrt(len(both.rows)): # best stays small
      addRow(rest, addCols(best, best.rows.pop(), -1))
  return both.rows

def grabs(tbl: TBL, cap: int | None = None,
          label: ORACLE = oracle) -> ROWS:
  "The straw man: same budget, taken at random."
  cap = cap or the.Stop
  return [label(r) for r in
          random.sample(tbl.rows, min(cap, len(tbl.rows)))]

def cut(tbl: TBL, rows: ROWS) -> tuple[int,ATOM,GO]|None:
  "The (at, v, go) whose two sides have the tightest ys."
  ys = [ydist(tbl, r) for r in rows]
  out, least = None, 1e30
  for col in tbl.x:
    for v, go in candidates(col, rows):
      a, b = Num(), Num()   # one pass, so one `go` per row
      for r,y in zip(rows,ys): add(a if go(r) else b, y)
      if a.n >= the.Leaf and b.n >= the.Leaf:
        score = (div(a)*a.n + div(b)*b.n) / (a.n + b.n)
        if score < least: out, least = (col.at, v, go), score
  return out

def candidates(col: COL, rows: ROWS) -> Iterator[tuple[ATOM,GO]]:
  "What splits to try: one per symbol, or CUTS per num."
  at = col.at
  if type(col) is Sym:
    for v in sorted({r[at] for r in rows if r[at] != "?"}):
      yield v, lambda r,v=v,at=at: r[at] == v
  else:                # by rank, not value: THESE rows are dense
    xs = sorted(r[at] for r in rows if r[at] != "?")
    k  = max(1, len(xs)//the.Cuts)     # cuts, spaced by rank
    for v in xs[k-1:-1:k]:
      yield v, (lambda r,v=v,at=at: r[at] != "?" and r[at] <= v)

def tree(tbl: TBL, rows: ROWS) -> NODE:
  "Split while a split keeps both sides big enough."
  node = o(rows=rows, at=None, v=None, go=None, kids=[],
           mu=sum(ydist(tbl,r) for r in rows)/len(rows))
  if len(rows) > the.Leaf and (found := cut(tbl, rows)):
    at, v, go = found
    yes, no = [], []
    for r in rows: (yes if go(r) else no).append(r)
    node.at, node.v, node.go = at, v, go # cut kepts sides big
    node.kids = [tree(tbl, yes), tree(tbl, no)]
  return node

def leaf(node: NODE, row: ROW) -> NODE:
  "Walk ROW down to its leaf."
  while node.kids: node = node.kids[0 if node.go(row) else 1]
  return node

def leafs(node: NODE) -> list[NODE]:
  "Return leaves of this tree."
  return [x for k in node.kids for x in leafs(k)] or [node]

def show(tbl: TBL, node: NODE, pre: str | None = None,
         edge: str = "") -> None:
  "Tree on the left; ydist and n on the right."
  ls = sorted(leafs(node), key=lambda z: z.mu)
  def walk(z: NODE, pre: str | None, edge: str) -> None:
    m = "+" if z is ls[0] else "-" if z is ls[-1] else " "
    print(f"{m} {round(100*z.mu):>4} {len(z.rows):>5}   "
          f"{(pre or '')+edge}".rstrip())
    op = "=" if type(tbl.cols[z.at or 0]) is Sym else "<="
    no = "!=" if op == "=" else ">"
    for kid, oper in zip(z.kids, [op, no]):
      walk(kid, "" if pre is None else pre + "|  ",
           f"{tbl.names[z.at]} {oper} {say(z.v)}")
  print(" ydist     n")
  walk(node, pre, edge)

# Calls core, acquire, and one RANKER -- tree's, by default.
def ranker(lab: TBL) -> Callable[[ROW], float]:
  "Grow a tree on LAB; score any row by its leaf's mean ydist."
  node = tree(lab, lab.rows)
  return lambda r: leaf(node, r).mu

def wins(tbl: TBL) -> Callable[[ROW], float]:
  "100 at the best row, 0 at an average one."
  ys = sorted(ydist(tbl, r) for r in tbl.rows)
  lo, avg = ys[0], sum(ys)/len(ys)
  return lambda r: max(-100, min(100, 100*(1 - (ydist(tbl,r)-lo)
                                            / (avg-lo+1e-32))))

def holdout(tbl: TBL, pick: PICKER = acquire,
            rank: RANKER = ranker,
            label: ORACLE = oracle) -> ROW:
  "Train on half; of CHECK guesses on the rest, pick best."
  rows  = random.sample(tbl.rows, len(tbl.rows))
  n     = len(rows)//2
  tr    = clone(tbl, rows[:n][:the.Few])
  lab   = clone(tbl, pick(tr, the.Stop - the.Check, label))
  top   = sorted(rows[n:], key=rank(lab))
  return min(top[:the.Check], key=lambda r: ydist(lab, label(r)))

# A leaf: it calls core, and nothing here calls it.
def adds(vs: Iterable[ATOM], col: COL | None = None) -> COL:
  "Every item of VS into COL."
  col = Num() if col is None else col
  for v in vs: add(col, v)
  return col

def cohen(xs: NUMS, ys: NUMS, d: float = 0.35,
          eps: float = 0) -> bool:
  "Are the means within D pooled standard deviations?"
  a, b = adds(xs), adds(ys)
  pool = ((a.n-1)*div(a)**2 + (b.n-1)*div(b)**2)/(a.n+b.n-2)
  return abs(a.mu - b.mu) <= max(eps, d * sqrt(pool))

def cliffs(xs: NUMS, ys: NUMS, d: float = 0.197) -> bool:
  "Sorted in.  Is the rank imbalance small enough?"
  gt = lt = j = k = 0
  for x in xs:
    while j < len(ys) and ys[j] <  x: j += 1; k = j
    while k < len(ys) and ys[k] <= x: k += 1
    gt += j; lt += len(ys) - k
  return abs(gt - lt) / (len(xs) * len(ys)) <= d

def ks(xs: NUMS, ys: NUMS, a: float = 1.36) -> bool:
  "Sorted in.  95% Kolmogorov-Smirnov."
  n, m, i, j, d = len(xs), len(ys), 0, 0, 0
  while i < n and j < m:
    v = min(xs[i], ys[j])
    while i < n and xs[i] <= v: i += 1
    while j < m and ys[j] <= v: j += 1
    d = max(d, abs(i/n - j/m))
  return d <= a * sqrt((n + m) / (n * m))

def same(xs: NUMS, ys: NUMS, eps: float = 0) -> bool:
  """Indistinguishable.  Always cliffs and ks; EPS adds cohen, and
  doubles as its effect size (eps=0 skips that third test)."""
  xs, ys = sorted(xs), sorted(ys)
  return (cliffs(xs,ys) and ks(xs,ys)
          and (not eps or cohen(xs,ys,d=eps)))

import ezr

def remember(header, rows):
    """
    Identify the top 5% of rows based on multi-objective optimization.
    A row is better if its goals (ending in - or +) are closer to their ideal.
    """
    # Create a table object using the provided data
    # The ezr.Tbl function expects an iterable of rows including the header
    data = [header] + rows
    tbl = ezr.Tbl(data)
    
    # Calculate ydist for every row. 
    # ydist measures how far a row's goals are from the best possible values.
    # Lower ydist is better.
    scored_rows = []
    for i, row in enumerate(rows):
        score = ezr.ydist(tbl, row)
        scored_rows.append((score, i))
    
    # Sort by score (ascending, since lower ydist is better)
    scored_rows.sort(key=lambda x: x[0])
    
    # Return the indices of the top 5% of rows
    limit = max(1, len(rows) // 20)
    return [item[1] for item in scored_rows[:limit]]

import ezr
import random

def guess(header, train, test):
    """
    Estimates the first goal column for each test row by training a tree
    on the provided training data and predicting the mean value of the
    first goal column at the corresponding leaf node.
    """
    # Create a table object for the training data
    data = [header] + train
    tbl = ezr.Tbl(data)
    
    # Identify the index of the first goal column
    # Goal columns end in '-' or '+'
    goal_indices = [i for i, h in enumerate(header) if h.endswith(('-', '+'))]
    first_goal_idx = goal_indices[0]
    
    # Build a tree on the training data
    # We use the tree structure to group similar rows
    root = ezr.tree(tbl, tbl.rows)
    
    # For each test row, find the leaf node and return the mean of the 
    # first goal column for all training rows that fell into that leaf.
    predictions = []
    for row in test:
        leaf_node = ezr.leaf(root, row)
        # Calculate the mean of the first goal column for rows in this leaf
        vals = [r[first_goal_idx] for r in leaf_node.rows if r[first_goal_idx] != "?"]
        if vals:
            predictions.append(sum(vals) / len(vals))
        else:
            # Fallback to global mean if leaf is empty or has no valid data
            all_vals = [r[first_goal_idx] for r in train if r[first_goal_idx] != "?"]
            predictions.append(sum(all_vals) / len(all_vals))
            
    return predictions

if __name__ == "__main__":
    # Load the data
    file_path = "pom3a.n1000.train.csv"
    rows_gen = ezr.csv(file_path)
    header = next(rows_gen)
    all_rows = list(rows_gen)
    
    # Split data: 80% train, 20% test
    random.seed(1234567891)
    random.shuffle(all_rows)
    split = int(len(all_rows) * 0.8)
    train = all_rows[:split]
    test_raw = all_rows[split:]
    
    # Mask the goals in the test set to simulate "unknown"
    goal_indices = [i for i, h in enumerate(header) if h.endswith(('-', '+'))]
    test = []
    for row in test_raw:
        new_row = list(row)
        for idx in goal_indices:
            new_row[idx] = "?"
        test.append(new_row)
        
    # Get predictions
    preds = guess(header, train, test)

    print(f"Sample predictions: {[round(p, 4) for p in preds[:5]]}")

import ezr

def certify(xs, ys):
    """
    Returns True if the two distributions are statistically indistinguishable,
    False if they are significantly different.
    """
    # Use the same logic as ezr.same, which checks for statistical significance
    # using Cliff's Delta and the Kolmogorov-Smirnov test.
    return ezr.same(xs, ys)

import ezr

def flow(header, row, seen):
    """
    Decide whether to accept a row based on whether it is significantly 
    better than the current 'best' seen so far.
    """
    # Create a table to handle column definitions and normalization
    data = [header] + (seen if seen else [row])
    tbl = ezr.Tbl(data)
    
    # If this is the first row, we have no baseline, so we accept it
    if not seen:
        return True
    
    # Calculate the ydist (distance from ideal goals) for the new row
    # and the best row seen so far.
    current_ydist = ezr.ydist(tbl, row)
    best_ydist = min(ezr.ydist(tbl, r) for r in seen)
    
    # We accept the row if it is better than the current best.
    # Since ydist measures distance from ideal, lower is better.
    # We use a small threshold to ensure we only stop for meaningful improvements.
    if current_ydist < best_ydist * 0.95:
        return True
    
    return False

import ezr
import random

def blame(header, rows):
    """
    Identify input columns that best explain why the worst rows are bad.
    We define 'worst' as the bottom 25% of rows based on ydist.
    We then compare the distribution of input values in the worst rows
    against the rest of the data to find the most influential features.
    """
    # Create a table to handle normalization and ydist calculations
    tbl = ezr.Tbl([header] + rows)
    
    # Score all rows
    scored = []
    for row in rows:
        scored.append((ezr.ydist(tbl, row), row))
    
    # Sort by ydist (higher is worse)
    scored.sort(key=lambda x: x[0], reverse=True)
    
    # Split into worst 25% and the rest
    n_worst = len(rows) // 4
    worst = [x[1] for x in scored[:n_worst]]
    rest = [x[1] for x in scored[n_worst:]]
    
    # Identify input columns (exclude goals)
    input_indices = [i for i, h in enumerate(header) if not (h.endswith('-') or h.endswith('+'))]
    
    # For each input column, calculate the difference in means (or mode)
    # between the worst rows and the rest.
    scores = []
    for i in input_indices:
        col = tbl.cols[i]
        # Get values for this column
        w_vals = [r[i] for r in worst if r[i] != "?"]
        r_vals = [r[i] for r in rest if r[i] != "?"]
        
        if not w_vals or not r_vals:
            continue
            
        # Calculate a simple effect size (difference in means/proportions)
        if isinstance(col, ezr.Num):
            w_avg = sum(w_vals) / len(w_vals)
            r_avg = sum(r_vals) / len(r_vals)
            diff = abs(w_avg - r_avg)
        else:
            # For symbols, compare frequency of the most common value
            w_mode = max(set(w_vals), key=w_vals.count)
            r_mode = max(set(r_vals), key=r_vals.count)
            diff = 1.0 if w_mode != r_mode else 0.0
            
        scores.append((header[i], diff))
    
    # Sort by impact and return the top 3 column names
    scores.sort(key=lambda x: x[1], reverse=True)
    return [s[0] for s in scores[:3]]

import ezr
import random

def justify(header, rows, row):
    """
    Returns (p, reason): p is the fraction of rows that 'row' is better than,
    and reason is a sentence explaining the justification.
    """
    # Create a table to handle normalization and distance calculations
    tbl = ezr.Tbl([header] + rows)
    
    # Estimate the goal values for the input 'row' using the tree's leaf mean
    # We build a tree on the known rows to predict the performance of the input
    root = ezr.tree(tbl, tbl.rows)
    leaf_node = ezr.leaf(root, row)
    
    # Calculate the predicted ydist for the input row
    # We use the mean ydist of the leaf node as the estimate
    predicted_ydist = leaf_node.mu
    
    # Calculate ydist for all known rows
    all_ydists = [ezr.ydist(tbl, r) for r in rows]
    
    # Calculate p: fraction of rows that are worse than our predicted row
    # (i.e., have a higher ydist)
    better_than_count = sum(1 for d in all_ydists if d > predicted_ydist)
    p = better_than_count / len(rows)
    
    reason = (f"This configuration is estimated to be in the top {int((1-p)*100)}% "
              f"of all possible setups based on its input characteristics.")
    
    return p, reason

import ezr
import random

def choose(header, rows, label):
    """
    Find the best row by using the acquire strategy to explore the space
    within a budget of 50 labels.
    """
    # Create a table object. 
    # We must ensure the table is initialized with rows that have numeric values
    # where possible, but since we are given rows with '?', we must handle
    # the fact that ezr.Tbl will see '?' as a string.
    # We will use a subset of rows to initialize the table's column statistics
    # or simply use the provided rows and handle the '?' in the label function.
    data = [header] + rows
    tbl = ezr.Tbl(data)
    
    # Track labeled rows
    labeled_data = []

    def tracked_label(row):
        # Find the index of the row in the original list
        # We use the row object identity to find the index
        idx = -1
        for i, r in enumerate(rows):
            if r is row:
                idx = i
                break
        
        # Get the true values from the oracle
        true_vals = label(idx)
        
        # Update the row in place
        for i, val in enumerate(true_vals):
            row[i] = val
            
        # Update the table's column statistics with the new information
        ezr.addCols(tbl, row, 1)
        
        labeled_data.append((idx, row))
        return row

    # Use the acquire logic to explore the space.
    # We limit the budget to 50.
    ezr.acquire(tbl, cap=50, label=tracked_label)
    
    # Find the best row among those we labeled
    best_idx = -1
    best_ydist = float('inf')
    
    for idx, row in labeled_data:
        current_ydist = ezr.ydist(tbl, row)
        if current_ydist < best_ydist:
            best_ydist = current_ydist
            best_idx = idx
            
    return best_idx

        if col_name not in header:

import ezr
import random

def spot(header, rows):
    """
    Identify rows that are 'suspicious' by checking if their input values
    are outliers relative to the rest of the data. We use the distance
    from the median of each column as a measure of suspicion.
    """
    # Create a table to handle column statistics
    tbl = ezr.Tbl([header] + rows)
    
    # Identify input columns (exclude goals)
    input_indices = [i for i, h in enumerate(header) if not (h.endswith('-') or h.endswith('+'))]
    
    # Calculate the median for each input column
    medians = {}
    for i in input_indices:
        col = tbl.cols[i]
        vals = [r[i] for r in rows if r[i] != "?"]
        if not vals:
            continue
        if isinstance(col, ezr.Num):
            sorted_vals = sorted(vals)
            medians[i] = sorted_vals[len(sorted_vals) // 2]
        else:
            # For symbols, use the mode
            medians[i] = max(set(vals), key=vals.count)
            
    # Score rows based on how far they deviate from the medians
    # A higher deviation score indicates a more "suspicious" row
    suspicion_scores = []
    for idx, row in enumerate(rows):
        score = 0
        for i in input_indices:
            val = row[i]
            if val == "?" or i not in medians:
                continue
            
            col = tbl.cols[i]
            if isinstance(col, ezr.Num):
                # Normalize the distance from the median
                diff = abs(val - medians[i])
                # Divide by range if possible, else use raw diff
                score += diff
            else:
                # Symbolic: 1 if different from mode, 0 otherwise
                if val != medians[i]:
                    score += 1
        suspicion_scores.append((score, idx))
        
    # Sort by suspicion score descending
    suspicion_scores.sort(key=lambda x: x[0], reverse=True)
    
    # Return the indices of the top 10 most suspicious rows
    return [item[1] for item in suspicion_scores[:10]]

import ezr
import random

def race(header, rows, label):
    """
    Compares three strategies: random, greedy, and tree-based.
    The shared budget of 50 labels is strictly enforced.
    """
    budget = 50
    labeled_data = {} # index -> row (with goals filled)

    def get_label(i):
        if i not in labeled_data:
            if len(labeled_data) >= budget:
                return None
            true_vals = label(i)
            row = list(rows[i])
            for j, val in enumerate(true_vals):
                row[j] = val
            labeled_data[i] = row
        return labeled_data[i]

    # Helper to calculate ydist for a row that might have '?'
    def get_ydist(i):
        row = get_label(i)
        if not row: return 1e30
        # Create a temporary table to get correct normalization for this row
        # based on the current knowledge
        temp_tbl = ezr.Tbl([header] + [r for r in labeled_data.values()])
        return ezr.ydist(temp_tbl, row)

    # Strategy 1: Random
    def strategy_random():
        indices = random.sample(range(len(rows)), 50)
        for i in indices:
            get_label(i)

    # Strategy 2: Greedy (Nearest to best)
    def strategy_greedy():
        # Start with 5 random
        for i in random.sample(range(len(rows)), 5):
            get_label(i)
        # Hill climb
        while len(labeled_data) < 50:
            best_i = min(labeled_data.keys(), key=get_ydist)
            # Find unlabelled row closest to current best
            unlabeled = [i for i in range(len(rows)) if i not in labeled_data]
            if not unlabeled: break
            # Use xdist to find nearest neighbor
            tbl = ezr.Tbl([header] + [r for r in labeled_data.values()])
            next_i = min(unlabeled, key=lambda i: ezr.xdist(tbl, rows[i], labeled_data[best_i]))
            get_label(next_i)

    # Strategy 3: Tree-based
    def strategy_tree():
        # Use ezr.acquire logic
        tbl = ezr.Tbl([header] + rows)
        def oracle_wrapper(row):
            # Find index
            for i, r in enumerate(rows):
                if r is row:
                    return get_label(i)
            return row
        ezr.acquire(tbl, cap=50, label=oracle_wrapper)

    # Run the race
    # We must reset labeled_data for each strategy to ensure fair budget usage
    results = {}
    strategies = {"random": strategy_random, "greedy": strategy_greedy, "tree": strategy_tree}
    
    for name, func in strategies.items():
        labeled_data = {}
        func()
        if labeled_data:
            best_i = min(labeled_data.keys(), key=get_ydist)
            results[name] = (best_i, get_ydist(best_i))
        else:
            results[name] = (-1, 1e30)

    winner = min(results, key=lambda n: results[n][1])
    return {"winner": winner, "row": results[winner][0], "tried": list(strategies.keys())}
