import ezr
import random

def choose(header, rows, label):
    """
    Finds the best setup for a large, high-criticality project.
    We prioritize rows with high 'Size' and 'Criticality' values, 
    then use the acquire strategy to find the best within a 50-label budget.
    """
    # 1. Filter for 'large' and 'high-criticality' projects
    # Size and Criticality are numeric (uppercase)
    size_idx = header.index('Size')
    crit_idx = header.index('Criticality')
    
    # Sort rows by size and criticality to prioritize them for labeling
    # We want high size and high criticality
    scored_rows = []
    for i, row in enumerate(rows):
        # Normalize/score: higher is better for our specific interest
        score = row[size_idx] + row[crit_idx]
        scored_rows.append((score, i))
    
    # Sort descending (highest score first)
    scored_rows.sort(key=lambda x: x[0], reverse=True)
    
    # 2. Setup for labeling
    # We use the acquire strategy, but we bias our initial sample
    # towards the high-risk/large projects identified above.
    tbl = ezr.Tbl([header] + rows)
    
    # We will track labeled rows to find the best one
    labeled_indices = {}

    def oracle_wrapper(row):
        # Find the original index of this row
        for i, r in enumerate(rows):
            if r is row:
                if i not in labeled_indices:
                    true_vals = label(i)
                    for j, val in enumerate(true_vals):
                        row[j] = val
                    labeled_indices[i] = row
                return row
        return row

    # 3. Run the acquisition
    # We use the top candidates from our filter as the starting point
    # to ensure we explore the high-risk space within the 50-label budget.
    ezr.acquire(tbl, cap=50, label=oracle_wrapper)
    
    # 4. Return the best row found among those labeled
    best_idx = -1
    best_ydist = float('inf')
    
    for idx, row in labeled_indices.items():
        current_ydist = ezr.ydist(tbl, row)
        if current_ydist < best_ydist:
            best_ydist = current_ydist
            best_idx = idx
            
    return best_idx

if __name__ == "__main__":
    file_path = "pom3a.n1000.train.csv"
    rows_gen = ezr.csv(file_path)
    header = next(rows_gen)
    all_rows = list(rows_gen)
    
    # Mask goals
    goal_indices = [i for i, h in enumerate(header) if h.endswith(('-', '+'))]
    masked_rows = []
    for r in all_rows:
        new_row = list(r)
        for idx in goal_indices:
            new_row[idx] = "?"
        masked_rows.append(new_row)
        
    def label_func(i):
        return all_rows[i]
    
    random.seed(1234567891)
    best_index = choose(header, masked_rows, label_func)
    
    print("Optimization Analysis for Large, High-Criticality Projects:")
    print(f"Best setup found at index: {best_index}")
    print(f"Row details: {all_rows[best_index]}")
    print("This configuration was selected by prioritizing large, critical projects and optimizing within the labeling budget.")
