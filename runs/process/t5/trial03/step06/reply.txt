import ezr
import random

def justify(header, rows, row):
    """
    Returns (p, reason): p is the fraction of rows that 'row' is better than.
    'row' has goals as '?', so we estimate its goals using a tree-based ranker
    trained on the provided rows.
    """
    # Create the training table
    tbl = ezr.Tbl([header] + rows)
    
    # Build a ranker (tree-based) to estimate the quality of the input row
    # The ranker returns the mean ydist of the leaf node the row falls into.
    rank_func = ezr.ranker(tbl)
    
    # Estimate the ydist of the target row
    # We need to pass a row with '?' to the ranker, which handles it via the tree
    estimated_ydist = rank_func(row)
    
    # Calculate ydist for all known rows to establish a baseline
    all_ydists = [ezr.ydist(tbl, r) for r in rows]
    
    # Calculate how many rows are worse than our estimate
    # A row is better if its ydist is lower.
    # We are better than a row if our estimated_ydist < row_ydist.
    better_than_count = sum(1 for d in all_ydists if estimated_ydist < d)
    p = better_than_count / len(rows)
    
    reason = (f"This setup is estimated to be better than {int(p * 100)}% of "
              "all known configurations based on its input characteristics.")
    
    return p, reason

if __name__ == "__main__":
    filename = "pom3a.n1000.train.csv"
    data = list(ezr.csv(filename))
    header = data[0]
    rows = data[1:]
    
    # Pick a random row to justify
    random.seed(1234567891)
    target_row = random.choice(rows)
    
    # Mask the goals to simulate the requirement
    masked_row = list(target_row)
    goal_indices = [i for i, h in enumerate(header) if h.endswith(('-', '+'))]
    for idx in goal_indices:
        masked_row[idx] = "?"
        
    # Get justification
    p, reason = justify(header, rows, masked_row)
    
    print(f"Justification for setup:")
    print(reason)
    print(f"Rank percentile: {p:.4f}")
