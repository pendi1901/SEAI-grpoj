import csv

def spot(header, rows):
    """
    Identifies suspicious rows by checking for outliers in numeric columns.
    A row is considered suspicious if it contains values that are statistically
    improbable (e.g., more than 3 standard deviations from the mean).
    """
    suspicious = []
    
    # Identify numeric columns
    numeric_indices = [i for i, h in enumerate(header) if h[0].isupper()]
    
    # Calculate mean and std dev for each numeric column
    stats = {}
    for i in numeric_indices:
        vals = [r[i] for r in rows if isinstance(r[i], (int, float))]
        if not vals:
            continue
        mean = sum(vals) / len(vals)
        variance = sum((x - mean) ** 2 for x in vals) / len(vals)
        std = variance ** 0.5
        stats[i] = (mean, std)
    
    # Score rows based on how many values are extreme outliers
    row_scores = []
    for idx, row in enumerate(rows):
        score = 0
        for i in numeric_indices:
            if i in stats and isinstance(row[i], (int, float)):
                mean, std = stats[i]
                if std > 0:
                    # Z-score check
                    z = abs(row[i] - mean) / std
                    if z > 3.5:  # Threshold for "wrong" entry
                        score += z
        if score > 0:
            row_scores.append((idx, score))
            
    # Sort by score descending (most suspicious first)
    row_scores.sort(key=lambda x: x[1], reverse=True)
    return [item[0] for item in row_scores]

    filename = 'WA_Fn-UseC_-HR-Employee-Attrition.n1000.train.csv'
    rows = []
    header = []
    
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = []
            for i, val in enumerate(row):
                if header[i][0].isupper():
                    try:
                        processed_row.append(float(val))
                    except ValueError:
                        processed_row.append(0.0)
                else:
                    processed_row.append(val)
            rows.append(processed_row)

    suspicious_indices = spot(header, rows)
    
    print("Analysis: Identifying potentially erroneous data entries.")
    print(f"Found {len(suspicious_indices)} rows with suspicious numeric values.")
    print("Top 5 most suspicious rows:")
    for i in suspicious_indices[:5]:
        print(f"- Row {i} (Check for data entry errors in numeric fields)")
