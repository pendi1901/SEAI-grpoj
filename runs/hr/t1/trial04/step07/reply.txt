import csv
import random

def choose(header, rows, label):
    """
    Finds the best working arrangement by sampling 50 rows.
    We use a simple strategy: sample 50 random rows, query their true goal values,
    and return the index of the row with the best performance.
    """
    goal_indices = [i for i, h in enumerate(header) if h.endswith('+') or h.endswith('-')]
    
    # Select 50 random indices to query
    indices_to_query = random.sample(range(len(rows)), min(50, len(rows)))
    
    best_idx = -1
    best_score = -float('inf')
    
    for i in indices_to_query:
        # Get true values for this row
        true_goals = label(i)
        
        # Calculate a score for this row
        # Goal +: higher is better, Goal -: lower is better
        score = 0.0
        for j, g_idx in enumerate(goal_indices):
            val = true_goals[j]
            # Normalize based on global min/max if possible, 
            # but here we use relative performance within our sample
            if header[g_idx].endswith('-'):
                score -= val
            else:
                score += val
        
        if score > best_score:
            best_score = score
            best_idx = i
            
    return best_idx

if __name__ == "__main__":
    filename = 'WA_Fn-UseC_-HR-Employee-Attrition.n1000.train.csv'
    rows = []
    header = []
    
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = []
            for i, val in enumerate(row):
                if header[i][0].isupper():
                    try:
                        processed_row.append(float(val))
                    except ValueError:
                        processed_row.append(0.0)
                else:
                    processed_row.append(val)
            rows.append(processed_row)

    # Prepare goal indices and mask them
    goal_indices = [i for i, h in enumerate(header) if h.endswith('+') or h.endswith('-')]
    
    # Define the label function for the harness
    def label(i):
        return [rows[i][g] for g in goal_indices]

    # Mask the rows
    for row in rows:
        for g in goal_indices:
            row[g] = "?"

    best_row_idx = choose(header, rows, label)
    
    print(f"Analysis complete.")
    print(f"The best working arrangement found is at row index: {best_row_idx}")
    print("This selection was based on sampling 50 staff members to identify the highest performance on retention-related metrics.")
