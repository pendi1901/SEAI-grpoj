import csv

def guess(header, train, test):
    """
    Estimates the first goal column ('YearsAtCompany+') for test rows.
    Uses a K-Nearest Neighbors approach based on numeric input columns.
    """
    # Identify goal column (first one ending in + or -)
    goal_idx = -1
    for i, col in enumerate(header):
        if col.endswith('+') or col.endswith('-'):
            goal_idx = i
            break
    
    if goal_idx == -1:
        return [0.0] * len(test)

    # Identify numeric input columns (excluding the goal column)
    # Ensure we only use columns that are actually numeric (float)
    input_indices = [
        i for i, col in enumerate(header) 
        if col[0].isupper() and i != goal_idx
    ]

    predictions = []
    for t_row in test:
        distances = []
        for r_row in train:
            dist = 0.0
            for i in input_indices:
                # Ensure both values are numeric before subtraction
                val_t = t_row[i]
                val_r = r_row[i]
                if isinstance(val_t, (int, float)) and isinstance(val_r, (int, float)):
                    dist += abs(val_t - val_r)
                else:
                    # If data is missing or non-numeric, add a penalty
                    dist += 1.0
            distances.append((dist, r_row[goal_idx]))
        
        # Find the average of the goal value for the 5 nearest neighbors
        distances.sort(key=lambda x: x[0])
        k = 5
        neighbors = distances[:k]
        avg_val = sum(n[1] for n in neighbors) / k
        predictions.append(float(avg_val))
        
    return predictions

import csv

def certify(xs, ys):
    """
    Performs a simple two-sample t-test (Welch's) to determine if two
    distributions are significantly different.
    Returns True if they are statistically similar (p > 0.05),
    False if they are significantly different (p <= 0.05).
    """
    n1, n2 = len(xs), len(ys)
    if n1 < 2 or n2 < 2:
        return True
    
    m1, m2 = sum(xs) / n1, sum(ys) / n2
    v1 = sum((x - m1) ** 2 for x in xs) / (n1 - 1)
    v2 = sum((y - m2) ** 2 for y in ys) / (n2 - 1)
    
    # Standard error
    se = ((v1 / n1) + (v2 / n2)) ** 0.5
    if se == 0:
        return m1 == m2
    
    # t-statistic
    t_stat = abs(m1 - m2) / se
    
    # For a large sample size (n=800), a t-stat > 1.96 roughly corresponds to p < 0.05
    return t_stat < 1.96

def flow(header, row, seen):
    """
    Decides if the current employee needs a 'what would keep you here' conversation.
    We prioritize employees who have high 'YearsAtCompany+' but are currently
    showing signs of attrition risk (e.g., low JobSatisfaction or high OverTime).
    
    Strategy: If this is the first row, accept it. Otherwise, compare the current
    row's 'YearsAtCompany+' against the average of 'seen' rows. If the current
    row has a high tenure but low satisfaction, flag them for retention.
    """
    # Identify indices
    try:
        tenure_idx = header.index('YearsAtCompany+')
        sat_idx = header.index('JobSatisfaction')
        ot_idx = header.index('overTime')
    except ValueError:
        return False

    # If no history, accept the first one
    if not seen:
        return True

    # Extract current values
    current_tenure = row[tenure_idx]
    current_sat = row[sat_idx]
    current_ot = row[ot_idx]

    # Calculate baseline from seen data
    seen_tenures = [s[tenure_idx] for s in seen if isinstance(s[tenure_idx], (int, float))]
    if not seen_tenures:
        return False
    
    avg_tenure = sum(seen_tenures) / len(seen_tenures)

    # who is either dissatisfied or working overtime.
    is_long_tenure = current_tenure > avg_tenure
    is_at_risk = (current_sat < 3) or (current_ot == 'Yes')

    return is_long_tenure and is_at_risk

import csv

def blame(header, rows):
    """
    Identifies input columns that correlate with low tenure (YearsAtCompany+).
    We define 'worst' rows as those in the bottom 25% of tenure.
    We then find input columns that have the highest variance or distinct 
    distribution in these 'worst' rows compared to the rest.
    """
    tenure_idx = header.index('YearsAtCompany+')
    
    # Filter out goal columns and identify input columns
    input_indices = []
    for i, name in enumerate(header):
        if not (name.endswith('+') or name.endswith('-')):
            input_indices.append(i)
            
    # Sort rows by tenure to find the 'worst' (lowest tenure)
    sorted_rows = sorted(rows, key=lambda r: r[tenure_idx])
    worst_rows = sorted_rows[:len(rows) // 4]
    
    # Calculate average tenure for the whole dataset vs worst rows
    # We look for columns where the distribution in 'worst' rows is 
    # significantly different from the global average.
    scores = []
    for i in input_indices:
        col_vals = [r[i] for r in rows if isinstance(r[i], (int, float))]
        if not col_vals:
            continue
        
        global_avg = sum(col_vals) / len(col_vals)
        worst_vals = [r[i] for r in worst_rows if isinstance(r[i], (int, float))]
        
        if not worst_vals:
            continue
            
        worst_avg = sum(worst_vals) / len(worst_vals)
        # Score based on how much the worst rows deviate from the global average
        scores.append((abs(worst_avg - global_avg), header[i]))
        
    # Return top 3 columns that explain the short stays
    scores.sort(key=lambda x: x[0], reverse=True)
    return [s[1] for s in scores[:3]]

import csv

def justify(header, rows, row):
    """
    Returns (p, reason) where p is the percentile rank of the row
    based on the average of its goal columns.
    """
    goal_indices = [i for i, h in enumerate(header) if h.endswith('+') or h.endswith('-')]
    
    # Estimate goal values for the test row using simple mean of all rows
    # (as a baseline estimator since we need to compare it to the population)
    estimated_goals = []
    for g_idx in goal_indices:
        vals = [r[g_idx] for r in rows if isinstance(r[g_idx], (int, float))]
        estimated_goals.append(sum(vals) / len(vals))

    # Calculate a score for every row in the dataset
    # Goal +: higher is better, Goal -: lower is better
    def get_score(r):
        score = 0.0
        for i, g_idx in enumerate(goal_indices):
            val = r[g_idx]
            # Normalize to 0-1 range roughly
            col_vals = [x[g_idx] for x in rows if isinstance(x[g_idx], (int, float))]
            min_v, max_v = min(col_vals), max(col_vals)
            norm = (val - min_v) / (max_v - min_v + 1e-9)
            if header[g_idx].endswith('-'):
                score += (1.0 - norm)
            else:
                score += norm
        return score / len(goal_indices)

    # Compare test row score to all rows
    test_score = 0.0
    for i, g_idx in enumerate(goal_indices):
        norm = (estimated_goals[i] - min(r[g_idx] for r in rows)) / (max(r[g_idx] for r in rows) - min(r[g_idx] for r in rows) + 1e-9)
        test_score += (1.0 - norm) if header[g_idx].endswith('-') else norm
    test_score /= len(goal_indices)

    all_scores = [get_score(r) for r in rows]
    better_than = sum(1 for s in all_scores if test_score > s)
    p = better_than / len(rows)
    
    reason = f"This team arrangement ranks in the top {int((1-p)*100)}% of all records based on its performance across all retention-related metrics."
    return (p, reason)

    """
    Finds the best working arrangement by sampling 50 rows.
    We use a simple strategy: sample 50 random rows, query their true goal values,
    and return the index of the row with the best performance.
    """

    # Select 50 random indices to query
    indices_to_query = random.sample(range(len(rows)), min(50, len(rows)))
    
    best_idx = -1
    best_score = -float('inf')
    
    for i in indices_to_query:
        # Get true values for this row
        true_goals = label(i)
        
        # Calculate a score for this row
        # Goal +: higher is better, Goal -: lower is better
        score = 0.0
        for j, g_idx in enumerate(goal_indices):
            val = true_goals[j]
            # Normalize based on global min/max if possible, 
            # but here we use relative performance within our sample
            if header[g_idx].endswith('-'):
                score -= val
            else:
                score += val
        
        if score > best_score:
            best_score = score
            best_idx = i
            
    return best_idx

import csv

    """
    Recommends changes to improve retention-related goals.
    Strategy: For a 'bad' row (e.g., high attrition risk), we adjust levers
    to move towards values associated with higher tenure/satisfaction.
    """

    # Simple heuristic:
    # 1. If 'overTime' is 'Yes', change to 'No'.
    # 2. If 'PercentSalaryHike' is low, increase it.
    # 3. If 'StockOptionLevel' is low, increase it.
    # 4. If 'TrainingTimesLastYear' is low, increase it.
    
    if 'overTime' in levers and 'overTime' in header:
        idx = header.index('overTime')
        if row[idx] == 'Yes' and 'No' in levers['overTime']:
            recommendations['overTime'] = 'No'

        idx = header.index('PercentSalaryHike')

        if row[idx] < high:

    if 'StockOptionLevel' in levers and 'StockOptionLevel' in header:
        idx = header.index('StockOptionLevel')
        low, high = levers['StockOptionLevel']
        if row[idx] < high:
            recommendations['StockOptionLevel'] = high

    if 'TrainingTimesLastYear' in levers and 'TrainingTimesLastYear' in header:
        idx = header.index('TrainingTimesLastYear')
        low, high = levers['TrainingTimesLastYear']
        if row[idx] < high:
            recommendations['TrainingTimesLastYear'] = high

    # Ensure at least one recommendation is made if possible
    if not recommendations and levers:
        for col, val in levers.items():
            if isinstance(val, list):
                recommendations[col] = val[0]
            else:
                recommendations[col] = val[1]
                
    return recommendations

    """

    # Identify numeric columns
    numeric_indices = [i for i, h in enumerate(header) if h[0].isupper()]
    
    # Calculate mean and std dev for each numeric column
    stats = {}
    for i in numeric_indices:
        vals = [r[i] for r in rows if isinstance(r[i], (int, float))]

            continue

import csv
import random

def race(header, rows, label):
    """
    Compares three strategies for finding the best employee arrangement.
    Budget: 50 labels total.
    """
    goal_indices = [i for i, h in enumerate(header) if h.endswith('+') or h.endswith('-')]
    
    def get_score(row_idx):
        true_goals = label(row_idx)
        score = 0.0
        for j, g_idx in enumerate(goal_indices):
            val = true_goals[j]
            # Normalize: goal+ (higher is better), goal- (lower is better)
            # Since we don't have global min/max, we treat raw values as relative
            if header[g_idx].endswith('-'):
                score -= val
            else:
                score += val
        return score

    # Strategies
    def strategy_random(budget):
        indices = random.sample(range(len(rows)), budget)
        best_idx = max(indices, key=get_score)
        return best_idx, get_score(best_idx)

    def strategy_first_n(budget):
        indices = list(range(budget))
        best_idx = max(indices, key=get_score)
        return best_idx, get_score(best_idx)

    def strategy_last_n(budget):
        indices = list(range(len(rows) - budget, len(rows)))
        best_idx = max(indices, key=get_score)
        return best_idx, get_score(best_idx)

    # Budget allocation: 16, 17, 17 = 50
    res_rand = strategy_random(16)
    res_first = strategy_first_n(17)
    res_last = strategy_last_n(17)

    results = {
        "random": res_rand,
        "first_n": res_first,
        "last_n": res_last
    }

    winner = max(results, key=lambda k: results[k][1])
    
    return {
        "winner": winner,
        "row": results[winner][0],
        "tried": ["random", "first_n", "last_n"]
    }

        i for i, col in enumerate(header)

    target = rows[0]

    # Return top 5% (at least 1, max 5% of total rows)

    return [item[0] for item in distances[:limit]]

import csv

def guess(header, train, test):
    """
    Estimates the first goal column ('YearsAtCompany+') for test rows.
    To answer the customer's question: "If we cut overtime across Sales, 
    what happens to how long people stay?", we use a K-Nearest Neighbors 
    approach that considers the 'department' and 'overTime' status.
    """
    goal_idx = next(i for i, h in enumerate(header) if h.endswith('+') or h.endswith('-'))
    dept_idx = header.index('department')
    ot_idx = header.index('overTime')
    
    # Identify numeric input columns for distance calculation
    # Ensure we only use columns that are actually numeric (float)
    input_indices = [i for i, h in enumerate(header) if h[0].isupper() and i != goal_idx]

    predictions = []
    for t_row in test:
        # We simulate the "cut overtime" scenario by forcing 'overTime' to 'No'
        # if the employee is in 'Sales'.
        simulated_ot = t_row[ot_idx]
        if t_row[dept_idx] == 'Sales':
            simulated_ot = 'No'
            
        distances = []
        for r_row in train:
            # Calculate distance based on numeric features
            dist = 0.0
            for i in input_indices:
                val_t = t_row[i]
                val_r = r_row[i]
                # Ensure both are numeric before subtraction
                if isinstance(val_t, (int, float)) and isinstance(val_r, (int, float)):
                    dist += (val_t - val_r) ** 2
                else:
                    dist += 1.0 # Penalty for non-numeric mismatch
            
            # Add penalty for categorical mismatch
            if t_row[dept_idx] != r_row[dept_idx]:
                dist += 100.0
            if simulated_ot != r_row[ot_idx]:
                dist += 50.0
                
            distances.append((dist**0.5, r_row[goal_idx]))
        
        # Average of 5 nearest neighbors
        distances.sort(key=lambda x: x[0])
        k = 5
        neighbors = distances[:k]
        avg_val = sum(n[1] for n in neighbors) / k
        predictions.append(float(avg_val))
        
    return predictions
