import csv

def certify(xs, ys):
    """
    Performs a simple two-sample t-test (Welch's) to determine if two
    distributions are significantly different.
    Returns True if they are statistically similar (p > 0.05),
    False if they are significantly different (p <= 0.05).
    """
    n1, n2 = len(xs), len(ys)
    if n1 < 2 or n2 < 2:
        return True
    
    m1, m2 = sum(xs) / n1, sum(ys) / n2
    v1 = sum((x - m1) ** 2 for x in xs) / (n1 - 1)
    v2 = sum((y - m2) ** 2 for y in ys) / (n1 - 1)
    
    se = ((v1 / n1) + (v2 / n2)) ** 0.5
    if se == 0:
        return m1 == m2
    
    t_stat = abs(m1 - m2) / se
    # For large samples, t-stat > 1.96 corresponds to p < 0.05
    return t_stat < 1.96

if __name__ == "__main__":
    filename = 'WA_Fn-UseC_-HR-Employee-Attrition.n1000.train.csv'
    rows = []
    header = []
    
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = []
            for i, val in enumerate(row):
                if header[i][0].isupper():
                    try:
                        processed_row.append(float(val))
                    except ValueError:
                        processed_row.append(0.0)
                else:
                    processed_row.append(val)
            rows.append(processed_row)

    # Goal: YearsWithCurrManager+
    # Input: TrainingTimesLastYear
    # Split into groups: Low training (0-2) vs High training (3+)
    train_idx = header.index('TrainingTimesLastYear')
    mgr_idx = header.index('YearsWithCurrManager+')
    
    low_train = [r[mgr_idx] for r in rows if r[train_idx] <= 2]
    high_train = [r[mgr_idx] for r in rows if r[train_idx] > 2]
    
    is_noise = certify(low_train, high_train)
    
    print("Analysis: Do people with more training stay longer with their manager?")
    print(f"Average tenure (Low Training): {sum(low_train)/len(low_train):.2f} years")
    print(f"Average tenure (High Training): {sum(high_train)/len(high_train):.2f} years")
    
    if is_noise:
        print("Conclusion: The observed difference is likely just noise; training does not significantly change tenure with the manager.")
    else:
        print("Conclusion: There is a statistically significant difference in tenure based on training levels.")
