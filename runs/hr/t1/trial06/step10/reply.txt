import csv
import random

def race(header, rows, label):
    """
    Compares three strategies for finding the best employee based on goal columns.
    Budget: 50 labels total.
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith('+') or name.endswith('-')]
    
    def get_score(row_idx):
        vals = label(row_idx)
        score = 0
        for i, val in enumerate(vals):
            name = header[goal_indices[i]]
            if name.endswith('+'):
                score += val
            else:
                score -= val
        return score

    # Strategies
    # 1. Random: Pick 17 random rows
    # 2. Greedy: Pick 17 rows with highest 'TotalWorkingYears' (proxy for experience)
    # 3. Diversity: Pick 16 rows with highest 'DistanceFromHome' (proxy for variety)
    
    strategies = ["random", "experience", "diversity"]
    labeled = {}
    
    def get_label(i):
        if i not in labeled:
            labeled[i] = get_score(i)
        return labeled[i]

    results = {}
    
    # Random
    indices_rand = random.sample(range(len(rows)), 17)
    results["random"] = max(indices_rand, key=get_label)
    
    # Experience (TotalWorkingYears index is 28)
    idx_exp = header.index('TotalWorkingYears')
    indices_exp = sorted(range(len(rows)), key=lambda i: rows[i][idx_exp], reverse=True)[:17]
    results["experience"] = max(indices_exp, key=get_label)
    
    # Diversity (DistanceFromHome index is 5)
    idx_dist = header.index('DistanceFromHome')
    indices_dist = sorted(range(len(rows)), key=lambda i: rows[i][idx_dist], reverse=True)[:16]
    results["diversity"] = max(indices_dist, key=get_label)
    
    # Find winner
    best_strategy = max(results, key=lambda s: get_label(results[s]))
    
    return {
        "winner": best_strategy,
        "row": results[best_strategy],
        "tried": strategies
    }

if __name__ == "__main__":
    filename = 'WA_Fn-UseC_-HR-Employee-Attrition.n1000.train.csv'
    rows = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = []
            for i, val in enumerate(row):
                if header[i][0].isupper():
                    processed_row.append(float(val))
                else:
                    processed_row.append(val)
            rows.append(processed_row)

    goal_indices = [i for i, name in enumerate(header) if name.endswith('+') or name.endswith('-')]
    
    def label(i):
        return [rows[i][idx] for idx in goal_indices]

    # Mask goals
    for row in rows:
        for idx in goal_indices:
            row[idx] = "?"

    outcome = race(header, rows, label)
    
    print("Strategy Race Results:")
    print(f"The winning strategy was: {outcome['winner']}")
    print(f"Best row found at index: {outcome['row']}")
    print(f"Strategies compared: {', '.join(outcome['tried'])}")
