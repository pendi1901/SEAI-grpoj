import csv

def guess(header, train, test):
    """
    Estimates the first goal column ('YearsAtCompany+') for test rows
    using a K-Nearest Neighbors approach based on numeric input features.
    """
    # Identify goal columns: those ending in '+' or '-'
    # The first goal column is 'YearsAtCompany+'
    goal_idx = 31
    
    # Identify input columns (numeric ones that are not goals)
    # Numeric columns start with uppercase.
    input_indices = []
    for i, name in enumerate(header):
        if name[0].isupper() and i != goal_idx and not name.endswith('+') and not name.endswith('-'):
            input_indices.append(i)

    # Normalize training data for distance calculation
    mins = [min(row[i] for row in train) for i in input_indices]
    maxs = [max(row[i] for row in train) for i in input_indices]
    ranges = [maxs[i] - mins[i] if maxs[i] != mins[i] else 1.0 for i in range(len(input_indices))]

    def get_features(row):
        return [(row[idx] - mins[i]) / ranges[i] for i, idx in enumerate(input_indices)]

    train_features = [get_features(row) for row in train]
    predictions = []

    for t_row in test:
        t_feat = get_features(t_row)
        # Find nearest neighbor (simple Euclidean distance)
        best_dist = float('inf')
        best_val = 0.0
        
        for i, feat in enumerate(train_features):
            dist = sum((t_feat[j] - feat[j])**2 for j in range(len(feat)))
            if dist < best_dist:
                best_dist = dist
                best_val = train[i][goal_idx]
        
        predictions.append(float(best_val))
        
    return predictions

import csv

def certify(xs, ys):
    """
    Performs a simple two-sample t-test (Welch's) to determine if two 
    distributions are significantly different. Returns True if they are 
    statistically similar (p > 0.05), False if they differ significantly.
    """
    n1, n2 = len(xs), len(ys)
    if n1 < 2 or n2 < 2:
        return True
    
    m1, m2 = sum(xs) / n1, sum(ys) / n2
    v1 = sum((x - m1) ** 2 for x in xs) / (n1 - 1)
    v2 = sum((y - m2) ** 2 for y in ys) / (n2 - 1)
    
    # Standard error
    se = ((v1 / n1) + (v2 / n2)) ** 0.5
    if se == 0:
        return m1 == m2
    
    t_stat = abs(m1 - m2) / se
    # For large samples, t-distribution approaches normal. 
    # Critical value for 95% confidence is ~1.96.
    return t_stat < 1.96

import csv

def flow(header, row, seen):
    """
    Decides if the current employee needs a 'what would keep you here' conversation.
    We target employees who are high-performers but have high risk of attrition.
    We use a simple threshold strategy: if the employee has high 'YearsAtCompany+'
    and 'YearsInCurrentRole+' but low 'JobSatisfaction', they are a priority.
    
    Since we cannot go back, we use a 'Secretary Problem' inspired threshold:
    we observe the first 20% of employees to establish a baseline for 
    'JobSatisfaction' and 'YearsAtCompany+', then pick the first one that 
    exceeds our baseline criteria.
    """
    idx_sat = header.index('JobSatisfaction')
    idx_years = header.index('YearsAtCompany+')
    
    # Baseline: average of the first 20% of arrivals
    n_seen = len(seen)
    if n_seen < 160:
        return False
    
    # Calculate baseline from seen data
    avg_sat = sum(r[idx_sat] for r in seen) / n_seen
    avg_years = sum(r[idx_years] for r in seen) / n_seen
    
    # Criteria: If current employee has lower satisfaction than average
    # but higher tenure than average, they are at risk and need a conversation.
    if row[idx_sat] < avg_sat and row[idx_years] > avg_years:
        return True
        
    return False

import csv

def blame(header, rows):
    """
    Identifies input columns that correlate with short stays.
    'YearsAtCompany+' is a goal to maximize. 'Worst' rows are those with 
    the lowest 'YearsAtCompany+' values.
    """
    goal_idx = header.index('YearsAtCompany+')
    
    # Identify input columns (numeric or symbolic, excluding goals)
    input_indices = [i for i, name in enumerate(header) 
                     if not name.endswith('+') and not name.endswith('-')]
    
    # Sort rows by YearsAtCompany+ ascending to find the 'worst' (shortest stays)
    sorted_rows = sorted(rows, key=lambda r: r[goal_idx])
    worst_rows = sorted_rows[:len(rows) // 10]  # Bottom 10%
    
    # Calculate average of input features for the worst rows
    # and compare against the global average to find 'blame'
    def get_val(row, i):
        return row[i]

    scores = []
    for i in input_indices:
        col_data = [r[i] for r in rows if isinstance(r[i], (int, float))]
        if not col_data:
            continue
            
        global_avg = sum(col_data) / len(col_data)
        worst_avg = sum(r[i] for r in worst_rows if isinstance(r[i], (int, float))) / len(worst_rows)

        # e.g., lower JobSatisfaction or higher DistanceFromHome
        diff = abs(worst_avg - global_avg)
        scores.append((diff, header[i]))
    
    # Return top 3 columns that explain the short stays
    scores.sort(key=lambda x: x[0], reverse=True)
    return [s[1] for s in scores[:3]]

    """
    Estimates the rank of a given row by comparing its goal performance 
    against all other rows in the dataset.
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith('+') or name.endswith('-')]
    
    # Simple estimation for goal columns: use the average of the dataset
    # (as a baseline since we don't have a complex model here)
    estimated_row = list(row)
    for idx in goal_indices:
        col_vals = [r[idx] for r in rows if isinstance(r[idx], (int, float))]
        estimated_row[idx] = sum(col_vals) / len(col_vals)
    
    def is_better(r1, r2):
        """Returns True if r1 is better than r2."""
        for idx in goal_indices:

                if r1[idx] < r2[idx]: return False

    reason = f"This arrangement ranks in the top {int((1-p)*100)}% of teams because it outperforms {int(p*100)}% of the historical records in retention-focused metrics."
    
    return (p, reason)

def choose(header, rows, label):
    """
    Finds the best row by sampling 50 rows, labeling them, and picking the one
    with the highest aggregate score for goal columns.
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith('+') or name.endswith('-')]
    
    # We have 800 rows. We can label 50.
    # We pick 50 random indices to evaluate.
    indices_to_label = random.sample(range(len(rows)), 50)
    
    best_idx = -1
    best_score = -float('inf')
    
    for i in indices_to_label:
        # Get true values for this row
        true_goals = label(i)
        
        # Calculate score: 
        # For '+', higher is better. For '-', lower is better.
        # We normalize by assuming a rough range or just summing relative to goals.
        # Since we don't have global min/max, we use the values themselves.
        # To make them comparable, we treat '-' as negative and '+' as positive.
        current_score = 0
        for j, goal_idx in enumerate(goal_indices):
            val = true_goals[j]
            if header[goal_idx].endswith('+'):
                current_score += val
            else:
                current_score -= val
        
        if current_score > best_score:
            best_score = current_score
            best_idx = i
            
    return best_idx

import csv

def spot(header, rows):
    """
    Identifies suspicious rows by checking for logical inconsistencies 
    between related numeric fields.
    
    Common HR data inconsistencies:
    1. YearsInCurrentRole > TotalWorkingYears
    2. YearsWithCurrManager > TotalWorkingYears
    3. YearsSinceLastPromotion > TotalWorkingYears
    4. YearsAtCompany > TotalWorkingYears
    """
    idx_total = header.index('TotalWorkingYears')
    idx_at_co = header.index('YearsAtCompany+')
    idx_role = header.index('YearsInCurrentRole+')
    idx_promo = header.index('YearsSinceLastPromotion')
    idx_mgr = header.index('YearsWithCurrManager+')

    suspicious = []
    
    for i, row in enumerate(rows):
        score = 0
        # Check if any tenure metric exceeds total career length
        if row[idx_at_co] > row[idx_total]: score += 1
        if row[idx_role] > row[idx_total]: score += 1
        if row[idx_promo] > row[idx_total]: score += 1
        if row[idx_mgr] > row[idx_total]: score += 1
        
        # Also check if role tenure exceeds company tenure
        if row[idx_role] > row[idx_at_co]: score += 1
        
        if score > 0:
            suspicious.append((score, i))
            
    # Sort by score descending (most suspicious first)
    suspicious.sort(key=lambda x: x[0], reverse=True)
    
    return [item[1] for item in suspicious]

import csv

def remember(header, rows):
    """
    Finds the 5% of employees most similar to the first employee in the dataset
    (the 'target' employee). Similarity is calculated using Euclidean distance
    on normalized numeric input features.
    """
    # Target is the first row
    target = rows[0]
    
    # Identify input columns (numeric, not goals)
    input_indices = [
        i for i, name in enumerate(header) 
        if name[0].isupper() and not name.endswith('+') and not name.endswith('-')
    ]
    
    # Normalize data for distance calculation
    mins = [min(row[i] for row in rows) for i in input_indices]
    maxs = [max(row[i] for row in rows) for i in input_indices]
    ranges = [maxs[i] - mins[i] if maxs[i] != mins[i] else 1.0 for i in range(len(input_indices))]
    
    def get_features(row):
        return [(row[i] - mins[j]) / ranges[j] for j, i in enumerate(input_indices)]
    
    target_feat = get_features(target)
    
    # Calculate distances to all other rows
    distances = []
    for i, row in enumerate(rows):
        if i == 0: continue  # Skip self
        row_feat = get_features(row)
        dist = sum((target_feat[j] - row_feat[j])**2 for j in range(len(target_feat)))
        distances.append((dist, i))
    
    # Sort by distance ascending
    distances.sort(key=lambda x: x[0])
    
    # Return top 5% (800 rows * 0.05 = 40 rows)
    limit = max(1, int(len(rows) * 0.05))
    return [item[1] for item in distances[:limit]]

import csv

def certify(xs, ys):
    """
    Performs a simple two-sample t-test (Welch's) to determine if two 
    distributions are significantly different. Returns True if they are 
    statistically similar (p > 0.05), False if they differ significantly.
    """
    n1, n2 = len(xs), len(ys)
    if n1 < 2 or n2 < 2:
        return True
    
    m1, m2 = sum(xs) / n1, sum(ys) / n2
    v1 = sum((x - m1) ** 2 for x in xs) / (n1 - 1)
    v2 = sum((y - m2) ** 2 for y in ys) / (n2 - 1)
    
    # Standard error
    se = ((v1 / n1) + (v2 / n2)) ** 0.5
    if se == 0:
        return m1 == m2
    
    t_stat = abs(m1 - m2) / se
    # For large samples, t-distribution approaches normal. 
    # Critical value for 95% confidence is ~1.96.
    return t_stat < 1.96

import csv

def flow(header, row, seen):
    """
    Detects if a team's composition has drifted significantly from the 
    historical baseline. We monitor the average 'JobSatisfaction' and 
    'TotalWorkingYears'. If the current row deviates significantly 
    (more than 2 standard deviations) from the moving average of the 
    'seen' population, we flag it as a potential shift in team dynamics.
    """
    if len(seen) < 50:
        return False

    idx_sat = header.index('JobSatisfaction')
    idx_exp = header.index('TotalWorkingYears')

    def get_stats(data, idx):
        vals = [r[idx] for r in data]
        mean = sum(vals) / len(vals)
        var = sum((x - mean) ** 2 for x in vals) / len(vals)
        return mean, var ** 0.5

    mean_sat, std_sat = get_stats(seen, idx_sat)
    mean_exp, std_exp = get_stats(seen, idx_exp)

    # Flag if the new row is an outlier (beyond 2 standard deviations)
    # in either satisfaction or experience.
    is_outlier_sat = abs(row[idx_sat] - mean_sat) > (2 * std_sat) if std_sat > 0 else False
    is_outlier_exp = abs(row[idx_exp] - mean_exp) > (2 * std_exp) if std_exp > 0 else False

    return is_outlier_sat or is_outlier_exp

import csv

def blame(header, rows):
    """
    Identifies input columns that have the least correlation with the goal 
    columns (tenure). Columns that show no significant difference between 
    the 'best' and 'worst' performers are candidates for removal.
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith('+') or name.endswith('-')]
    input_indices = [i for i, name in enumerate(header) if not name.endswith('+') and not name.endswith('-')]
    
    # Define 'best' and 'worst' based on a simple aggregate of goal columns
    def get_score(row):
        score = 0
        for idx in goal_indices:
            val = row[idx]
            if header[idx].endswith('+'):
                score += val
            else:
                score -= val
        return score

    sorted_rows = sorted(rows, key=get_score)
    worst = sorted_rows[:len(rows) // 5]
    best = sorted_rows[-len(rows) // 5:]
    
    # For each input column, calculate the difference in means between best and worst
    # If the difference is negligible, the column has little to do with the goal.
    useless_cols = []
    for i in input_indices:
        col_data = [r[i] for r in rows if isinstance(r[i], (int, float))]
        if not col_data:
            continue
            
        worst_vals = [r[i] for r in worst if isinstance(r[i], (int, float))]
        best_vals = [r[i] for r in best if isinstance(r[i], (int, float))]
        
        if not worst_vals or not best_vals:
            continue
            
        w_avg = sum(worst_vals) / len(worst_vals)
        b_avg = sum(best_vals) / len(best_vals)
        
        # Normalize difference by range
        col_range = max(col_data) - min(col_data)
        if col_range == 0:
            useless_cols.append((0, header[i]))
        else:
            diff = abs(w_avg - b_avg) / col_range
            useless_cols.append((diff, header[i]))
            
    # Sort by difference ascending (smallest difference = least impact)
    useless_cols.sort(key=lambda x: x[0])
    
    # Return the names of the 5 columns with the least impact
    return [item[1] for item in useless_cols[:5]]

import csv

def spot(header, rows):
    """
    Identifies rows that are 'unlike anyone we have employed' by checking
    if their input features fall outside the observed range of the historical data.
    
    A row is suspicious if any of its numeric input features are outside the 
    min/max range established by the rest of the population.
    """
    # Identify numeric input columns (not goals)
    input_indices = [
        i for i, name in enumerate(header)
        if name[0].isupper() and not name.endswith('+') and not name.endswith('-')
    ]
    
    # Calculate min/max for each input column
    mins = {}
    maxs = {}
    for i in input_indices:
        vals = [r[i] for r in rows if isinstance(r[i], (int, float))]
        if vals:
            mins[i] = min(vals)
            maxs[i] = max(vals)
            
    suspicious = []
    for idx, row in enumerate(rows):
        anomaly_score = 0
        for i in input_indices:
            val = row[i]
            if isinstance(val, (int, float)):
                # If the value is outside the historical range, it's suspicious
                if val < mins[i] or val > maxs[i]:
                    anomaly_score += 1
        
        if anomaly_score > 0:
            suspicious.append((anomaly_score, idx))
            
    # Sort by anomaly score descending (most suspicious first)
    suspicious.sort(key=lambda x: x[0], reverse=True)
    
    return [item[1] for item in suspicious]
