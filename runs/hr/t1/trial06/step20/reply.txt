import csv
import random

def race(header, rows, label):
    """
    Compares three strategies for finding the best employee based on goal columns.
    Budget: 50 labels total.
    Strategies:
    1. 'random': Pick 16 random rows.
    2. 'rule_of_thumb': Pick 17 rows with the lowest 'YearsAtCompany+' (newest staff).
    3. 'tenure_focus': Pick 17 rows with the highest 'YearsAtCompany+' (longest tenure).
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith('+') or name.endswith('-')]
    
    def get_score(row_idx):
        vals = label(row_idx)
        score = 0
        for i, val in enumerate(vals):
            name = header[goal_indices[i]]
            if name.endswith('+'):
                score += val
            else:
                score -= val
        return score

    strategies = ["random", "rule_of_thumb", "tenure_focus"]
    labeled = {}
    
    def get_label(i):
        if i not in labeled:
            labeled[i] = get_score(i)
        return labeled[i]

    results = {}
    
    # 1. Random (16 labels)
    indices_rand = random.sample(range(len(rows)), 16)
    results["random"] = max(indices_rand, key=get_label)
    
    # 2. Rule of thumb: Newest staff (17 labels)
    # We use 'YearsAtCompany+' as a proxy for tenure.
    idx_tenure = header.index('YearsAtCompany+')
    indices_new = sorted(range(len(rows)), key=lambda i: rows[i][idx_tenure])[:17]
    results["rule_of_thumb"] = max(indices_new, key=get_label)
    
    # 3. Tenure focus: Longest staff (17 labels)
    indices_old = sorted(range(len(rows)), key=lambda i: rows[i][idx_tenure], reverse=True)[:17]
    results["tenure_focus"] = max(indices_old, key=get_label)
    
    # Find winner
    best_strategy = max(results, key=lambda s: get_label(results[s]))
    
    return {
        "winner": best_strategy,
        "row": results[best_strategy],
        "tried": strategies
    }

    rows = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = []
            for i, val in enumerate(row):
                if header[i][0].isupper():
                    processed_row.append(float(val))
                else:
                    processed_row.append(val)
            rows.append(processed_row)

    goal_indices = [i for i, name in enumerate(header) if name.endswith('+') or name.endswith('-')]
    labeled_data = {i: [rows[i][idx] for idx in goal_indices] for i in range(len(rows))}
    
    def label(i):
        return labeled_data[i]

    # Mask goals
    for row in rows:
        for idx in goal_indices:
            row[idx] = "?"

    outcome = race(header, rows, label)
    
    print("Comparison: Is the 'newest staff' rule of thumb effective?")
    print(f"The winning strategy was: {outcome['winner']}")
    if outcome['winner'] == "rule_of_thumb":
        print("Result: Your rule of thumb is effective for this dataset.")
    else:
        print("Result: Your rule of thumb is outperformed by other strategies.")
    print(f"Best row found at index: {outcome['row']}")
