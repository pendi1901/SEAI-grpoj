import csv

def spot(header, rows):
    """
    Identifies rows that are 'unlike anyone we have employed' by checking
    if the input values are outside the range of historical data (outliers).
    
    A row is suspicious if its numeric input values are significantly outside
    the observed min/max range of the training set.
    """
    # Identify numeric input columns (exclude goals)
    input_indices = [
        i for i, name in enumerate(header)
        if name[0].isupper() and not (name.endswith('+') or name.endswith('-'))
    ]
    
    # Calculate min and max for each input column
    mins = {}
    maxs = {}
    for i in input_indices:
        vals = [row[i] for row in rows if isinstance(row[i], (int, float))]
        if vals:
            mins[i] = min(vals)
            maxs[i] = max(vals)
    
    suspicious = []
    for idx, row in enumerate(rows):
        score = 0
        for i in input_indices:
            val = row[i]
            if isinstance(val, (int, float)):
                # If a value is outside the historical range, it's 'unlike anyone'
                if val < mins[i] or val > maxs[i]:
                    score += 1
        
        if score > 0:
            suspicious.append((score, idx))
            
    # Sort by score descending (most suspicious first)
    suspicious.sort(key=lambda x: x[0], reverse=True)
    
    return [item[1] for item in suspicious]

if __name__ == "__main__":
    filename = 'WA_Fn-UseC_-HR-Employee-Attrition.n1000.train.csv'
    rows = []
    header = []
    
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = []
            for i, val in enumerate(row):
                if header[i][0].isupper():
                    try:
                        processed_row.append(float(val))
                    except ValueError:
                        processed_row.append(0.0)
                else:
                    processed_row.append(val)
            rows.append(processed_row)
            
    suspects = spot(header, rows)
    
    print("Outlier Analysis: Employees unlike historical records identified:")
    if not suspects:
        print("No suspicious records found.")
    else:
        for score, idx in suspects[:10]:
            print(f"- Row {idx}: Contains {score} input value(s) outside historical ranges.")
        print(f"\nTotal of {len(suspects)} records flagged as potentially unreliable for estimation.")
