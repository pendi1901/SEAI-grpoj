def guess(header, train, test):
    """
    Estimates the value of the first goal column ('YearsAtCompany+')
    for test rows using a K-Nearest Neighbors approach based on 
    numeric input columns.
    """
    # Find the first goal column index
    goal_idx = -1
    for i, name in enumerate(header):
        if name.endswith('+') or name.endswith('-'):
            goal_idx = i
            break
    
    if goal_idx == -1:
        return [0.0] * len(test)

    # Identify numeric input columns (excluding the goal column)
    # Only include columns that are numeric (start with uppercase)
    input_indices = [
        i for i, name in enumerate(header) 
        if name[0].isupper() and i != goal_idx
    ]

    # Calculate ranges for numeric inputs for normalization
    mins = [float('inf')] * len(header)
    maxs = [float('-inf')] * len(header)
    for row in train:
        for i in input_indices:
            val = row[i]
            if isinstance(val, (int, float)):
                if val < mins[i]: mins[i] = val
                if val > maxs[i]: maxs[i] = val
    
    ranges = [maxs[i] - mins[i] if (maxs[i] != float('-inf') and maxs[i] != mins[i]) else 1.0 for i in range(len(header))]

    predictions = []
    for t_row in test:
        best_dist = float('inf')
        best_val = 0.0
        
        for train_row in train:
            dist = 0.0
            for i in input_indices:
                # Ensure we only compare numeric values
                if isinstance(t_row[i], (int, float)) and isinstance(train_row[i], (int, float)):
                    diff = (t_row[i] - train_row[i]) / ranges[i]
                    dist += diff * diff
            
            if dist < best_dist:
                best_dist = dist
                best_val = train_row[goal_idx]

import csv

def certify(xs, ys):
    """
    Performs a simple two-sample t-test (Welch's) to check if two 
    distributions are significantly different. Returns True if they 
    are statistically similar (p > 0.05), False if they differ.
    """
    n1, n2 = len(xs), len(ys)
    if n1 < 2 or n2 < 2:
        return True
    
    m1, m2 = sum(xs) / n1, sum(ys) / n2
    v1 = sum((x - m1) ** 2 for x in xs) / (n1 - 1)
    v2 = sum((y - m2) ** 2 for y in ys) / (n2 - 1)
    
    # Standard error
    se = ((v1 / n1) + (v2 / n2)) ** 0.5
    if se == 0:
        return m1 == m2
    
    # t-statistic
    t = abs(m1 - m2) / se
    
    # For a large enough sample, t > 1.96 corresponds to p < 0.05
    return t < 1.96

import csv

def flow(header, row, seen):
    """
    Decides if the current employee needs a 'what would keep you here' 
    conversation. We target employees who are high-performers or have 
    high potential (long tenure) but might be at risk.
    
    Strategy: Use a simple threshold based on 'YearsAtCompany+' and 
    'YearsInCurrentRole+'. If the current row is in the top 20% of 
    historical data seen so far, flag it.
    """
    # Identify goal indices
    yac_idx = header.index('YearsAtCompany+')
    yicr_idx = header.index('YearsInCurrentRole+')
    
    # Current values
    yac = row[yac_idx]
    yicr = row[yicr_idx]
    
    # If no history, we can't compare, so default to False
    if not seen:
        return False
    
    # Calculate thresholds from seen data (top 20%)
    seen_yac = sorted([r[yac_idx] for r in seen], reverse=True)
    seen_yicr = sorted([r[yicr_idx] for r in seen], reverse=True)
    
    limit = max(1, len(seen) // 5)
    yac_threshold = seen_yac[min(limit, len(seen_yac) - 1)]
    yicr_threshold = seen_yicr[min(limit, len(seen_yicr) - 1)]
    
    # Flag if the employee is a long-tenured veteran (top 20%)
    # These are the most valuable to retain.
    if yac >= yac_threshold and yicr >= yicr_threshold:
        return True
        
    return False

import csv

def blame(header, rows):
    """
    Identifies input columns that correlate with low 'YearsAtCompany+' 
    (the goal to maximize). We look for columns where the values in the 
    bottom 20% of tenure differ significantly from the rest.
    """
    yac_idx = header.index('YearsAtCompany+')
    
    # Identify input columns (exclude goals)
    input_indices = [
        i for i, name in enumerate(header)
        if not (name.endswith('+') or name.endswith('-'))
    ]
    
    # Sort rows by YearsAtCompany+ (ascending) to find the "worst" (shortest stay)
    sorted_rows = sorted(rows, key=lambda r: r[yac_idx])
    limit = max(1, len(rows) // 5)
    worst = sorted_rows[:limit]
    rest = sorted_rows[limit:]
    
    scores = []
    for i in input_indices:
        col_name = header[i]
        val_worst = [r[i] for r in worst]
        val_rest = [r[i] for r in rest]
        
        # Calculate a simple "difference" score
        # For numeric: difference in means
        # For symbolic: difference in frequency of the most common value
        if isinstance(val_worst[0], (int, float)):
            m1 = sum(val_worst) / len(val_worst)
            m2 = sum(val_rest) / len(val_rest)
            score = abs(m1 - m2)
        else:
            # Frequency of the most common value in 'worst'
            mode_w = max(set(val_worst), key=val_worst.count)
            freq_w = val_worst.count(mode_w) / len(val_worst)
            freq_r = val_rest.count(mode_w) / len(val_rest)
            score = abs(freq_w - freq_r)
            
        scores.append((score, col_name))
    
    # Return top 3 columns that explain the short stays
    scores.sort(key=lambda x: x[0], reverse=True)
    return [s[1] for s in scores[:3]]
