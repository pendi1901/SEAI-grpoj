import csv

def remember(header, rows):
    """
    Finds the most similar employees to the first employee in the dataset
    and returns their indices. Similarity is calculated using Euclidean 
    distance on normalized numeric input columns.
    """
    # Identify goal columns and input columns
    goal_indices = [i for i, name in enumerate(header) if name.endswith('+') or name.endswith('-')]
    input_indices = [i for i, name in enumerate(header) if name[0].isupper() and i not in goal_indices]
    
    # Target row is the first one
    target = rows[0]
    
    # Normalize numeric inputs
    mins = [min(r[i] for r in rows) for i in input_indices]
    maxs = [max(r[i] for r in rows) for i in input_indices]
    ranges = [maxs[i] - mins[i] if maxs[i] != mins[i] else 1.0 for i in range(len(input_indices))]
    
    def get_dist(r1, r2):
        d = 0
        for idx, i in enumerate(input_indices):
            d += ((r1[i] - r2[i]) / ranges[idx]) ** 2
        return d**0.5

    # Calculate distances to all other rows
    distances = []
    for i in range(1, len(rows)):
        dist = get_dist(target, rows[i])
        distances.append((dist, i))
    
    # Sort by distance (ascending)
    distances.sort(key=lambda x: x[0])
    
    # Return top 5% (max 40 for 800 rows)
    limit = max(1, len(rows) // 20)
    return [item[1] for item in distances[:limit]]

if __name__ == "__main__":
    filename = 'WA_Fn-UseC_-HR-Employee-Attrition.n1000.train.csv'
    rows = []
    header = []
    
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = []
            for i, val in enumerate(row):
                if header[i][0].isupper():
                    try:
                        processed_row.append(float(val))
                    except ValueError:
                        processed_row.append(0.0)
                else:
                    processed_row.append(val)
            rows.append(processed_row)

    yac_idx = header.index('YearsAtCompany+')
    indices = remember(header, rows)
    
    print(f"Employees most similar to the first record (Row 0):")
    for idx in indices:
        years = rows[idx][yac_idx]
        print(f"- Row {idx}: Stayed for {years} years.")
