import csv

def guess(header, train, test):
    """
    Estimates the first goal column ('YearsAtCompany+') for test rows.
    Uses a simple K-Nearest Neighbors approach based on numeric input columns.
    """
    # Identify goal columns and input columns
    goal_indices = [i for i, name in enumerate(header) if name.endswith('+') or name.endswith('-')]
    first_goal_idx = goal_indices[0]
    
    # Use numeric input columns for distance calculation
    input_indices = [i for i, name in enumerate(header) 
                     if name[0].isupper() and i not in goal_indices]
    
    # Normalize numeric inputs to handle different scales
    def get_stats(data):
        mins = [min(row[i] for row in data) for i in input_indices]
        maxs = [max(row[i] for row in data) for i in input_indices]
        return mins, maxs

    mins, maxs = get_stats(train)
    
    def dist(r1, r2):
        d = 0
        for i in input_indices:
            idx = input_indices.index(i)
            range_val = maxs[idx] - mins[idx]
            if range_val == 0: continue
            d += ((r1[i] - r2[i]) / range_val) ** 2
        return d ** 0.5

    predictions = []
    for t_row in test:
        # Find 5 nearest neighbors in training set
        distances = []
        for train_row in train:
            distances.append((dist(t_row, train_row), train_row[first_goal_idx]))
        
        distances.sort(key=lambda x: x[0])
        # Average the goal value of the 5 nearest neighbors
        k = 5
        avg = sum(d[1] for d in distances[:k]) / k
        predictions.append(float(avg))
        
    return predictions

import csv

def certify(xs, ys):
    """
    Performs a simple two-sample t-test (Welch's) to determine if two groups
    are statistically different. Returns True if they are likely the same 
    (null hypothesis not rejected), False if they differ significantly.
    """
    def mean(data):
        return sum(data) / len(data)

    def variance(data):
        m = mean(data)
        return sum((x - m) ** 2 for x in data) / (len(data) - 1)

    n1, n2 = len(xs), len(ys)
    m1, m2 = mean(xs), mean(ys)
    v1, v2 = variance(xs), variance(ys)

    # Calculate t-statistic
    t_stat = (m1 - m2) / ((v1 / n1 + v2 / n2) ** 0.5 + 1e-9)
    
    # For a 95% confidence level, critical value is approx 1.96
    # If absolute t-stat > 1.96, we reject the null hypothesis (they differ)
    return abs(t_stat) < 1.96

def flow(header, row, seen):
    """
    Decides if the current employee needs a 'what would keep you here' conversation.
    We identify employees at risk of attrition (low tenure/satisfaction) 
    compared to the 'seen' population.
    """
    # Identify indices
    try:
        yac_idx = header.index('YearsAtCompany+')
        sat_idx = header.index('JobSatisfaction')
        ot_idx = header.index('overTime')
    except ValueError:
        return False

    # If no history, we can't compare, so default to False
    if not seen:
        return False

    # Current employee metrics
    current_yac = row[yac_idx]
    current_sat = row[sat_idx]
    current_ot = row[ot_idx]

    # Calculate averages from seen data to establish a baseline
    seen_yac = [r[yac_idx] for r in seen]
    seen_sat = [r[sat_idx] for r in seen]
    
    avg_yac = sum(seen_yac) / len(seen_yac)
    avg_sat = sum(seen_sat) / len(seen_sat)

    # Logic: If the employee has lower-than-average tenure, 
    # lower-than-average satisfaction, and works overtime, 
    # they are a high-priority candidate for retention.
    is_low_tenure = current_yac < avg_yac
    is_low_sat = current_sat < avg_sat
    is_overworked = (current_ot == 'Yes')

    return is_low_tenure and is_low_sat and is_overworked

import csv

def blame(header, rows):
    """
    Identifies input columns that correlate with low tenure (YearsAtCompany+).
    We define 'worst rows' as those in the bottom 25% of tenure.
    We then identify input columns where the average value for these 'worst'
    rows deviates most significantly from the global average.
    """
    yac_idx = header.index('YearsAtCompany+')
    
    # Identify input columns (numeric or symbolic, excluding goals)
    input_indices = [i for i, name in enumerate(header) 
                     if not name.endswith('+') and not name.endswith('-')]
    
    # Sort rows by tenure to find the 'worst' (bottom 25%)
    sorted_rows = sorted(rows, key=lambda r: r[yac_idx])
    worst_rows = sorted_rows[:len(rows) // 4]
    
    # Calculate global averages/modes for inputs
    def get_avg(idx):
        col_data = [r[idx] for r in rows]
        if isinstance(col_data[0], float):
            return sum(col_data) / len(col_data)
        else:
            # For symbolic, return the most frequent
            return max(set(col_data), key=col_data.count)

    # Calculate deviation for each input column
    deviations = []
    for i in input_indices:
        worst_val = [r[i] for r in worst_rows]
        if isinstance(worst_val[0], float):
            avg_all = get_avg(i)
            avg_worst = sum(worst_val) / len(worst_val)
            # Score based on how much the worst group differs from the mean
            score = abs(avg_worst - avg_all)
        else:
            # For symbolic, score based on frequency difference
            mode_all = get_avg(i)
            mode_worst = max(set(worst_val), key=worst_val.count)
            score = 1.0 if mode_worst != mode_all else 0.0
        
        deviations.append((header[i], score))
    
    # Return top 3 columns that explain the short stays
    deviations.sort(key=lambda x: x[1], reverse=True)
    return [d[0] for d in deviations[:3]]

import csv

def justify(header, rows, row):
    """
    Estimates the rank of a given row by comparing its predicted goal performance
    against the actual performance of the provided population.
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith('+') or name.endswith('-')]
    input_indices = [i for i, name in enumerate(header) if i not in goal_indices]

    # Simple KNN to estimate the missing goal values in 'row'
    def get_dist(r1, r2):
        d = 0
        for i in input_indices:
            if isinstance(r1[i], float) and isinstance(r2[i], float):
                d += (r1[i] - r2[i])**2
            else:
                d += 1 if r1[i] != r2[i] else 0
        return d**0.5

    # Estimate goals for the input row
    k = 7
    distances = sorted([(get_dist(row, r), r) for r in rows], key=lambda x: x[0])
    neighbors = distances[:k]
    
    estimates = []
    for g_idx in goal_indices:
        vals = [n[1][g_idx] for n in neighbors]
        estimates.append(sum(vals) / k)

    # Calculate score for the row (higher is better for + goals, lower for - goals)
    def score(r):
        s = 0
        for i, g_idx in enumerate(goal_indices):
            val = r[g_idx]
            if header[g_idx].endswith('+'):
                s += val
            else:
                s -= val
        return s

    target_score = sum(estimates[i] if header[goal_indices[i]].endswith('+') else -estimates[i] 
                       for i in range(len(goal_indices)))
    
    all_scores = [score(r) for r in rows]
    better_than = [s for s in all_scores if s < target_score]
    p = len(better_than) / len(rows)
    
    reason = f"This arrangement ranks in the top {int((1-p)*100)}% of all teams because it outperforms {int(p*100)}% of the existing staff records in tenure and retention metrics."
    
    return (p, reason)

def choose(header, rows, label):

    sample_indices = random.sample(range(len(rows)), 50)
    
    best_idx = -1
    best_score = -float('inf')
    
    for idx in sample_indices:
        # Get true goal values
        goals = label(idx)
        
        # Calculate score: sum of '+' goals minus sum of '-' goals
        current_score = 0
        for i, g_idx in enumerate(goal_indices):
            val = goals[i]
            if header[g_idx].endswith('+'):
                current_score += val
            else:
                current_score -= val
        
        if current_score > best_score:
            best_score = current_score
            best_idx = idx
            
    return best_idx

import csv

def fix(header, row, levers):
    """
    Recommends changes to a 'bad' row (low tenure) to improve retention.
    We prioritize increasing pay (PercentSalaryHike), stock options, 
    and reducing overtime or travel frequency.
    """
    recommendations = {}
    
    # Map indices for quick lookup
    h_map = {name: i for i, name in enumerate(header)}
    
    # Strategy:
    # 1. If overtime is 'Yes' and we can change it, set to 'No'.
    if 'overTime' in levers and row[h_map['overTime']] == 'Yes':
        if 'No' in levers['overTime']:
            recommendations['overTime'] = 'No'
            
    # 2. If PercentSalaryHike is low, increase it to the max allowed.
    if 'PercentSalaryHike' in levers:
        low, high = levers['PercentSalaryHike']
        if row[h_map['PercentSalaryHike']] < high:
            recommendations['PercentSalaryHike'] = high
            
    # 3. If StockOptionLevel is low, increase it.
    if 'StockOptionLevel' in levers:
        low, high = levers['StockOptionLevel']
        if row[h_map['StockOptionLevel']] < high:
            recommendations['StockOptionLevel'] = high
            
    # 4. If businessTravel is high frequency, reduce it.
    if 'businessTravel' in levers and row[h_map['businessTravel']] == 'Travel_Frequently':
        if 'Travel_Rarely' in levers['businessTravel']:
            recommendations['businessTravel'] = 'Travel_Rarely'
            
    # Ensure at least one change is made if possible
    if not recommendations and levers:
        # Fallback: pick the first available lever and set to a reasonable value
        first_key = list(levers.keys())[0]
        val = levers[first_key]
        recommendations[first_key] = val[0] if isinstance(val, list) else val[1]
        
    return recommendations
