import csv

def blame(header, rows):
    """
    Identifies input columns that have 'nothing to do' with tenure.
    We define 'nothing to do' as columns where the distribution in the 
    bottom 25% of tenure (worst) is statistically indistinguishable 
    from the distribution in the top 25% of tenure (best).
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith('+') or name.endswith('-')]
    yac_idx = header.index('YearsAtCompany+')
    
    # Identify input columns (excluding goals)
    input_indices = [i for i, name in enumerate(header) if i not in goal_indices]
    
    # Sort rows by tenure to find the 'best' (top 25%) and 'worst' (bottom 25%)
    sorted_rows = sorted(rows, key=lambda r: r[yac_idx])
    worst_rows = sorted_rows[:len(rows) // 4]
    best_rows = sorted_rows[-(len(rows) // 4):]
    
    def get_dist_score(idx):
        """
        Calculates how different the distributions are between best and worst.
        For numeric: difference in means.
        For symbolic: difference in frequency of the most common category.
        """
        w_vals = [r[idx] for r in worst_rows]
        b_vals = [r[idx] for r in best_rows]
        
        if isinstance(w_vals[0], float):
            m_w = sum(w_vals) / len(w_vals)
            m_b = sum(b_vals) / len(b_vals)
            return abs(m_w - m_b)
        else:
            # Symbolic: compare frequency of the most common item in the whole set
            all_vals = [r[idx] for r in rows]
            mode = max(set(all_vals), key=all_vals.count)
            freq_w = w_vals.count(mode) / len(w_vals)
            freq_b = b_vals.count(mode) / len(b_vals)
            return abs(freq_w - freq_b)

    # Calculate scores for all inputs
    scores = []
    for i in input_indices:
        scores.append((header[i], get_dist_score(i)))
    
    # Sort by score ascending (lowest difference means least impact on tenure)
    scores.sort(key=lambda x: x[1])
    
    # Return the bottom 5 columns that have the least impact
    return [s[0] for s in scores[:5]]

if __name__ == "__main__":
    filename = 'WA_Fn-UseC_-HR-Employee-Attrition.n1000.train.csv'
    rows = []
    header = []
    
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = []
            for i, val in enumerate(row):
                if header[i][0].isupper():
                    processed_row.append(float(val))
                else:
                    processed_row.append(val)
            rows.append(processed_row)
            
    useless_columns = blame(header, rows)
    
    print("Analysis: Columns with the least impact on employee tenure.")
    print("These columns show the smallest difference between our best and worst performers,")
    print("suggesting they may not be necessary to collect:")
    for col in useless_columns:
        print(f" - {col}")
