import csv

def spot(header, rows):
    """
    Identifies rows that are 'unlike anyone we have employed'.
    We define 'unlike' as rows where input values are outliers (Z-score > 3)
    compared to the historical population.
    """
    h_map = {name: i for i, name in enumerate(header)}
    goal_indices = [i for i, name in enumerate(header) if name.endswith('+') or name.endswith('-')]
    input_indices = [i for i in range(len(header)) if i not in goal_indices and isinstance(rows[0][i], float)]
    
    # Calculate means and standard deviations for numeric inputs
    stats = {}
    for i in input_indices:
        vals = [r[i] for r in rows]
        mean = sum(vals) / len(vals)
        var = sum((x - mean) ** 2 for x in vals) / len(vals)
        stats[i] = (mean, var ** 0.5)
        
    suspicious = []
    for idx, row in enumerate(rows):
        # Calculate an anomaly score based on how many standard deviations
        # the inputs are from the mean.
        score = 0
        for i in input_indices:
            mean, sd = stats[i]
            if sd > 0:
                z = abs(row[i] - mean) / sd
                if z > 3:
                    score += z
        
        if score > 0:
            suspicious.append((score, idx))
            
    # Sort by score descending (most suspicious first)
    suspicious.sort(key=lambda x: x[0], reverse=True)
    return [item[1] for item in suspicious]

if __name__ == "__main__":
    filename = 'WA_Fn-UseC_-HR-Employee-Attrition.n1000.train.csv'
    rows = []
    header = []
    
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = []
            for i, val in enumerate(row):
                if header[i][0].isupper():
                    processed_row.append(float(val))
                else:
                    processed_row.append(val)
            rows.append(processed_row)
            
    suspect_indices = spot(header, rows)
    
    print("Anomaly Detection Analysis:")
    print(f"We identified {len(suspect_indices)} records that are statistically 'unlike' our historical population.")
    print("These records contain input values that deviate significantly (Z-score > 3) from the norm.")
    print("Our estimates for these specific employees cannot be trusted.")
    
    if suspect_indices:
        print(f"\nTop 3 most suspicious row indices: {suspect_indices[:3]}")
