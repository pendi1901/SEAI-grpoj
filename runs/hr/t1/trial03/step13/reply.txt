import csv

def certify(xs, ys):
    """
    Performs a simple Welch's t-test approximation to check if two samples
    are statistically different. Returns True if they are likely the same
    (null hypothesis not rejected), False if they differ significantly.
    """
    n1, n2 = len(xs), len(ys)
    if n1 < 2 or n2 < 2:
        return True
    
    m1, m2 = sum(xs) / n1, sum(ys) / n2
    v1 = sum((x - m1) ** 2 for x in xs) / (n1 - 1)
    v2 = sum((y - m2) ** 2 for y in ys) / (n2 - 1)
    
    # Standard error of the difference
    se = ((v1 / n1) + (v2 / n2)) ** 0.5
    if se == 0:
        return m1 == m2
    
    t_stat = abs(m1 - m2) / se
    # Threshold for 95% confidence (approx 1.96 for large samples)
    return t_stat < 1.96

def analyze_training_impact(header, rows):
    """
    Compares 'YearsWithCurrManager+' for employees with high training 
    (>= 3 days) vs low training (< 3 days).
    """
    train_idx = header.index('TrainingTimesLastYear')
    mgr_idx = header.index('YearsWithCurrManager+')
    
    high_training = [row[mgr_idx] for row in rows if row[train_idx] >= 3]
    low_training = [row[mgr_idx] for row in rows if row[train_idx] < 3]
    
    is_same = certify(high_training, low_training)
    return is_same, len(high_training), len(low_training)

def load_data(filename):
    rows = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = []
            for i, val in enumerate(row):
                if header[i][0].isupper():
                    try:
                        processed_row.append(float(val))
                    except ValueError:
                        processed_row.append(0.0)
                else:
                    processed_row.append(val)
            rows.append(processed_row)
    return header, rows

if __name__ == "__main__":
    filename = 'WA_Fn-UseC_-HR-Employee-Attrition.n1000.train.csv'
    header, rows = load_data(filename)
    
    same, n_high, n_low = analyze_training_impact(header, rows)
    
    print("Analysis: Do people with more training stay longer with their manager?")
    if same:
        print("Result: No, the difference in tenure with the manager is statistically insignificant.")
    else:
        print("Result: Yes, there is a statistically significant difference in tenure based on training.")
    print(f"(Sample sizes: High Training={n_high}, Low Training={n_low})")
