import csv

def guess(header, train, test):
    """
    Estimates the first goal column ('YearsAtCompany+') for test rows.
    Uses a simple K-Nearest Neighbors approach based on numeric input columns.
    """
    # Identify goal column (first one ending in + or -)
    goal_idx = -1
    for i, name in enumerate(header):
        if name.endswith('+') or name.endswith('-'):
            goal_idx = i
            break
    
    if goal_idx == -1:
        return [0.0] * len(test)

    # Identify numeric input columns (excluding the goal column)
    input_indices = [
        i for i, name in enumerate(header) 
        if name[0].isupper() and i != goal_idx
    ]

    # Normalize training data for distance calculation
    # Using simple mean/std normalization
    train_inputs = [[row[i] for i in input_indices] for row in train]
    goals = [row[goal_idx] for row in train]
    
    # Calculate simple average as a baseline
    avg_goal = sum(goals) / len(goals)
    
    results = []
    for t_row in test:
        # Find nearest neighbor in training set based on Euclidean distance
        best_dist = float('inf')
        best_val = avg_goal
        
        for i, train_row in enumerate(train_inputs):
            dist = 0
            for j, idx in enumerate(input_indices):
                # Handle missing values in test row by skipping or using 0
                val = t_row[idx] if t_row[idx] != '?' else 0.0
                dist += (val - train_row[j])**2
            
            if dist < best_dist:
                best_dist = dist
                best_val = goals[i]
        
        results.append(float(best_val))
        
    return results

import csv

def certify(xs, ys):
    """
    Performs a simple Welch's t-test approximation to check if two samples
    are statistically different. Returns True if they are likely the same
    (null hypothesis not rejected), False if they differ significantly.
    """
    n1, n2 = len(xs), len(ys)
    if n1 < 2 or n2 < 2:
        return True
    
    m1, m2 = sum(xs) / n1, sum(ys) / n2
    v1 = sum((x - m1) ** 2 for x in xs) / (n1 - 1)
    v2 = sum((y - m2) ** 2 for y in ys) / (n2 - 1)
    
    # Standard error of the difference
    se = ((v1 / n1) + (v2 / n2)) ** 0.5
    if se == 0:
        return m1 == m2
    
    t_stat = abs(m1 - m2) / se
    # Threshold for 95% confidence (approx 1.96 for large samples)
    return t_stat < 1.96

def flow(header, row, seen):
    """
    Decides if the current employee needs a 'what would keep you here' conversation.
    We use a simple threshold strategy: if the employee's 'YearsAtCompany+' 
    is significantly lower than the average of those seen so far, 
    or if they are in a high-risk category (e.g., low JobSatisfaction), 
    we flag them.
    
    Since we must stop eventually, we use a 1/e (approx 37%) stopping rule 
    to establish a baseline, then pick the next one that is 'better' 
    than our observed baseline.
    """
    yac_idx = header.index('YearsAtCompany+')
    js_idx = header.index('JobSatisfaction')
    
    # If this is the first row, we can't compare
    if not seen:
        return False
    
    # 1/e stopping rule: explore first 37% of data to set a benchmark
    n = len(seen) + 1
    limit = 800 // 3  # Approx 266
    
    # Calculate baseline from seen data
    avg_yac = sum(r[yac_idx] for r in seen) / len(seen)
    
    # Decision logic:
    # If we are past the exploration phase, flag if:
    # 1. YearsAtCompany is below average (potential churn risk)
    # 2. JobSatisfaction is low (<= 2)
    if n > limit:
        if row[yac_idx] < avg_yac or row[js_idx] <= 2:
            return True
            
    return False

import csv

def blame(header, rows):
    """
    Identifies input columns that correlate with low values in goal columns
    (YearsAtCompany+, YearsInCurrentRole+, YearsWithCurrManager+).
    We define 'bad' rows as those in the bottom 25% of the average of 
    normalized goal values.
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith('+') or name.endswith('-')]
    input_indices = [i for i, name in enumerate(header) if not (name.endswith('+') or name.endswith('-'))]

    # Normalize goals to [0, 1] range to calculate a composite score
    # For '+', higher is better. For '-', lower is better.
    def get_score(row):
        scores = []
        for i in goal_indices:
            col_data = [r[i] for r in rows]
            min_v, max_v = min(col_data), max(col_data)
            rng = max_v - min_v if max_v != min_v else 1
            norm = (row[i] - min_v) / rng
            if header[i].endswith('-'):
                norm = 1 - norm
            scores.append(norm)
        return sum(scores) / len(scores)

    # Rank rows by composite score
    scored_rows = sorted([(get_score(row), row) for row in rows])
    worst_rows = scored_rows[:len(rows) // 4]
    best_rows = scored_rows[-len(rows) // 4:]

    # Find inputs that differ most between worst and best rows
    # We look for the highest absolute difference in mean values
    diffs = []
    for i in input_indices:
        if isinstance(rows[0][i], (int, float)):
            worst_vals = [r[1][i] for r in worst_rows]
            best_vals = [r[1][i] for r in best_rows]
            avg_w = sum(worst_vals) / len(worst_vals)
            avg_b = sum(best_vals) / len(best_vals)
            diffs.append((abs(avg_w - avg_b), header[i]))
        else:
            # For symbolic, check frequency difference
            worst_vals = [r[1][i] for r in worst_rows]
            best_vals = [r[1][i] for r in best_rows]
            # Simple heuristic: difference in mode frequency
            mode_w = max(set(worst_vals), key=worst_vals.count)
            freq_w = worst_vals.count(mode_w) / len(worst_vals)
            freq_b = best_vals.count(mode_w) / len(best_vals)
            diffs.append((abs(freq_w - freq_b), header[i]))

    diffs.sort(key=lambda x: x[0], reverse=True)
    return [d[1] for d in diffs[:3]]

import csv

def get_goal_indices(header):
    return [i for i, name in enumerate(header) if name.endswith('+') or name.endswith('-')]

def get_score(header, row, goal_indices, min_vals, max_vals):
    """Calculates a normalized score [0, 1] for a row based on goal columns."""
    scores = []
    for i in goal_indices:
        rng = max_vals[i] - min_vals[i]
        rng = rng if rng != 0 else 1
        norm = (row[i] - min_vals[i]) / rng
        if header[i].endswith('-'):
            norm = 1 - norm
        scores.append(norm)
    return sum(scores) / len(scores)

def justify(header, rows, row):
    """
    Estimates the performance of a row by comparing it to the training set
    using a simple K-Nearest Neighbors approach, then ranks it.
    """
    goal_indices = get_goal_indices(header)
    input_indices = [i for i, name in enumerate(header) if not (name.endswith('+') or name.endswith('-'))]
    
    # Precompute min/max for normalization
    min_vals = {i: min(r[i] for r in rows) for i in goal_indices}
    max_vals = {i: max(r[i] for r in rows) for i in goal_indices}
    
    # Estimate missing goals using nearest neighbor
    best_dist = float('inf')
    best_row = None
    for train_row in rows:
        dist = sum((row[i] - train_row[i])**2 for i in input_indices if isinstance(row[i], (int, float)))
        if dist < best_dist:
            best_dist = dist
            best_row = train_row
            
    # Fill in the row with estimated values
    estimated_row = list(row)
    for i in goal_indices:
        estimated_row[i] = best_row[i]
        
    # Calculate score for the estimated row
    score = get_score(header, estimated_row, goal_indices, min_vals, max_vals)
    
    # Calculate percentile rank
    all_scores = [get_score(header, r, goal_indices, min_vals, max_vals) for r in rows]
    better_than = sum(1 for s in all_scores if s < score)
    p = better_than / len(rows)
    
    reason = f"This arrangement ranks in the top {int((1-p)*100)}% of all teams because its estimated tenure and role stability metrics outperform {int(p*100)}% of the historical data."
    
    return p, reason

import csv

def fix(header, row, levers):
    """
    Recommends changes to a 'bad' row to improve its goal metrics.
    We prioritize increasing 'PercentSalaryHike' and 'StockOptionLevel',
    and adjusting 'overTime' or 'businessTravel' to improve retention.
    """
    recommendations = {}
    
    for col, allowed in levers.items():
        idx = header.index(col)
        current_val = row[idx]
        
        if isinstance(allowed, tuple):  # Numeric lever (low, high)
            low, high = allowed
            # If current is low, push towards high
            if current_val < high:
                recommendations[col] = min(current_val + (high - low) * 0.2, high)
        else:  # Symbolic lever (list of options)
            # If current is 'Yes'/'No' or specific travel, try to switch
            if current_val in allowed:
                # Simple heuristic: pick the first option that isn't the current one
                for option in allowed:
                    if option != current_val:
                        recommendations[col] = option
                        break
    
    # Ensure at least one change is made if possible
    if not recommendations and levers:
        first_key = list(levers.keys())[0]
        if isinstance(levers[first_key], tuple):
            recommendations[first_key] = levers[first_key][1]
        else:
            recommendations[first_key] = levers[first_key][0]
            
    return recommendations

def spot(header, rows):
    """
    Identifies suspicious rows by checking for outliers in numeric columns.
    A value is considered suspicious if it is more than 3 standard deviations
    away from the mean of its column.
    """
    numeric_indices = [i for i, name in enumerate(header) if name[0].isupper()]
    
    # Calculate mean and std for each numeric column
    stats = {}
    for i in numeric_indices:
        col_vals = [row[i] for row in rows]
        mean = sum(col_vals) / len(col_vals)
        variance = sum((x - mean) ** 2 for x in col_vals) / len(col_vals)
        std = variance ** 0.5
        stats[i] = (mean, std)
    
    # Score rows based on how many outliers they contain and the magnitude
    suspicion_scores = []
    for r_idx, row in enumerate(rows):
        score = 0
        for i in numeric_indices:
            mean, std = stats[i]
            if std > 0:
                z_score = abs(row[i] - mean) / std
                if z_score > 3:
                    score += z_score
        if score > 0:
            suspicion_scores.append((score, r_idx))
            
    # Sort by suspicion score descending
    suspicion_scores.sort(key=lambda x: x[0], reverse=True)
    
    return [item[1] for item in suspicion_scores]

def guess(header, train, test):

    Estimates 'YearsAtCompany+' for test rows.
    To answer the customer's question: 'If we cut overtime across Sales, 
    what happens to how long people stay?', we model the relationship 
    between 'overTime', 'department', and 'YearsAtCompany+'.
    
    We use a conditional mean approach: for each test row, we estimate 
    the goal by calculating the average 'YearsAtCompany+' for training 
    rows that share the same 'department' and 'overTime' status.
    """
    goal_idx = next(i for i, name in enumerate(header) if name.endswith('+') or name.endswith('-'))
    ot_idx = header.index('overTime')
    dept_idx = header.index('department')
    
    # Precompute averages for (department, overTime) groups
    groups = {}
    for row in train:
        key = (row[dept_idx], row[ot_idx])
        if key not in groups:
            groups[key] = []
        groups[key].append(row[goal_idx])
    
    group_means = {k: sum(v) / len(v) for k, v in groups.items()}
    global_mean = sum(row[goal_idx] for row in train) / len(train)
    
    predictions = []
    for t_row in test:

        dept = t_row[dept_idx]

        # Predict based on the group mean
        val = group_means.get((dept, ot), global_mean)
        predictions.append(float(val))
        
    return predictions

import csv

def certify(xs, ys):
    """
    Performs a simple Welch's t-test approximation to check if two samples
    are statistically different. Returns True if they are likely the same
    (null hypothesis not rejected), False if they differ significantly.
    """
    n1, n2 = len(xs), len(ys)
    if n1 < 2 or n2 < 2:
        return True
    
    m1, m2 = sum(xs) / n1, sum(ys) / n2
    v1 = sum((x - m1) ** 2 for x in xs) / (n1 - 1)
    v2 = sum((y - m2) ** 2 for y in ys) / (n2 - 1)
    
    # Standard error of the difference
    se = ((v1 / n1) + (v2 / n2)) ** 0.5
    if se == 0:
        return m1 == m2
    
    t_stat = abs(m1 - m2) / se
    # Threshold for 95% confidence (approx 1.96 for large samples)
    return t_stat < 1.96

def flow(header, row, seen):
    """
    Detects if a team's composition has shifted significantly.
    We monitor the mean of 'YearsAtCompany+' and 'JobSatisfaction'.
    If the current row's metrics deviate by more than 2 standard deviations
    from the historical mean of the 'seen' rows, we flag it as an anomaly.
    """
    if not seen:
        return False

    # Identify goal and key metrics
    yac_idx = header.index('YearsAtCompany+')
    js_idx = header.index('JobSatisfaction')
    
    # Only consider numeric metrics for drift detection
    metrics = [yac_idx, js_idx]
    
    # Calculate stats for seen data
    for m_idx in metrics:
        vals = [r[m_idx] for r in seen]
        mean = sum(vals) / len(vals)
        var = sum((x - mean) ** 2 for x in vals) / len(vals)
        std = var ** 0.5
        
        # If std is 0, we can't detect drift on this metric
        if std == 0:
            continue
            
        # Check if current row is an outlier (> 2 std devs)
        if abs(row[m_idx] - mean) > (2 * std):
            return True
            
    return False

import csv

def blame(header, rows):
    """
    Identifies input columns that have the least correlation with the goal columns.
    We define 'bad' as columns that show no significant difference in distribution
    between the top 25% and bottom 25% of employees (ranked by composite goal score).
    Columns with the smallest difference are candidates for removal.
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith('+') or name.endswith('-')]
    input_indices = [i for i, name in enumerate(header) if not (name.endswith('+') or name.endswith('-'))]

    # Normalize goals to [0, 1] range to calculate a composite score
    def get_score(row):
        scores = []
        for i in goal_indices:
            col_data = [r[i] for r in rows]
            min_v, max_v = min(col_data), max(col_data)
            rng = max_v - min_v if max_v != min_v else 1
            norm = (row[i] - min_v) / rng
            if header[i].endswith('-'):
                norm = 1 - norm
            scores.append(norm)
        return sum(scores) / len(scores)

    # Rank rows by composite score
    scored_rows = sorted([(get_score(row), row) for row in rows])
    worst_rows = [r[1] for r in scored_rows[:len(rows) // 4]]
    best_rows = [r[1] for r in scored_rows[-len(rows) // 4:]]

    # Find inputs that differ LEAST between worst and best rows
    # These are the columns that have 'nothing to do' with the outcome
    diffs = []
    for i in input_indices:
        if isinstance(rows[0][i], (int, float)):
            worst_vals = [r[i] for r in worst_rows]
            best_vals = [r[i] for r in best_rows]
            avg_w = sum(worst_vals) / len(worst_vals)
            avg_b = sum(best_vals) / len(best_vals)
            # Normalize difference by range to make it comparable
            col_data = [r[i] for r in rows]
            rng = max(col_data) - min(col_data)
            rng = rng if rng != 0 else 1
            diffs.append((abs(avg_w - avg_b) / rng, header[i]))
        else:
            # For symbolic, check frequency difference (Jensen-Shannon-like heuristic)
            worst_vals = [r[i] for r in worst_rows]
            best_vals = [r[i] for r in best_rows]
            all_cats = set(worst_vals) | set(best_vals)
            score = 0
            for cat in all_cats:
                score += abs(worst_vals.count(cat)/len(worst_vals) - best_vals.count(cat)/len(best_vals))
            diffs.append((score, header[i]))

    # Sort by smallest difference
    diffs.sort(key=lambda x: x[0])
    return [d[1] for d in diffs[:5]]
