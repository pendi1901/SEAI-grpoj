import csv

def spot(header, rows):
    """
    Identifies suspicious rows by checking for outliers in numeric columns.
    A value is considered suspicious if it is more than 3 standard deviations
    away from the mean of its column.
    """
    numeric_indices = [i for i, name in enumerate(header) if name[0].isupper()]
    
    # Calculate mean and std for each numeric column
    stats = {}
    for i in numeric_indices:
        col_vals = [row[i] for row in rows]
        mean = sum(col_vals) / len(col_vals)
        variance = sum((x - mean) ** 2 for x in col_vals) / len(col_vals)
        std = variance ** 0.5
        stats[i] = (mean, std)
    
    # Score rows based on how many outliers they contain and the magnitude
    suspicion_scores = []
    for r_idx, row in enumerate(rows):
        score = 0
        for i in numeric_indices:
            mean, std = stats[i]
            if std > 0:
                z_score = abs(row[i] - mean) / std
                if z_score > 3:
                    score += z_score
        if score > 0:
            suspicion_scores.append((score, r_idx))
            
    # Sort by suspicion score descending
    suspicion_scores.sort(key=lambda x: x[0], reverse=True)
    
    return [item[1] for item in suspicion_scores]

def load_data(filename):
    rows = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = []
            for i, val in enumerate(row):
                if header[i][0].isupper():
                    try:
                        processed_row.append(float(val))
                    except ValueError:
                        processed_row.append(0.0)
                else:
                    processed_row.append(val)
            rows.append(processed_row)
    return header, rows

if __name__ == "__main__":
    filename = 'WA_Fn-UseC_-HR-Employee-Attrition.n1000.train.csv'
    header, rows = load_data(filename)
    
    suspects = spot(header, rows)
    
    print("Suspicious rows identified (potential data entry errors):")
    for i in suspects[:10]:
        print(f"- Row {i} (Suspicion Score: High)")
        
    print("\nStrategy: We identified rows containing numeric values that are statistical outliers "
          "(>3 standard deviations from the mean), which often indicates manual entry errors.")
