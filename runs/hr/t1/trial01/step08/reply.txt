def fix(header, row, levers):
    """Recommend changes to keep a bad employee."""
    goals_min = [c for c in header if c.endswith('-')]
    goals_max = [c for c in header if c.endswith('+')]

    def score(r):
        s = 0.0
        for g in goals_min:
            idx = header.index(g)
            v = float(r[idx]) if r[idx] != '?' else 0.0
            s += v
        for g in goals_max:
            idx = header.index(g)
            v = float(r[idx]) if r[idx] != '?' else 0.0
            s -= v
        return s

    changes = {}

    for col, allowed in levers.items():
        idx = header.index(col)

        if col == 'businessTravel':
            # Prefer Non-Travel or Travel_Rarely over Travel_Frequently
            preferred = ['Non-Travel', 'Travel_Rarely']
            chosen = next((v for v in preferred if v in allowed), allowed[0])
            changes[col] = chosen

        elif col == 'overTime':
            # N is better than Y (overtime correlates with leaving sooner)
            if 'N' in allowed:
                changes[col] = 'N'

        elif col == 'TrainingTimesLastYear':
            low, high = allowed
            target = (low + high) * 0.75
            step = max(1, int((high - low) / 4))
            val = round(target / step) * step
            changes[col] = val

        elif col == 'PercentSalaryHike':
            low, high = allowed
            target = (low + high) * 0.25
            step = max(1, int((high - low) / 4))
            val = round(target / step) * step
            changes[col] = val

        elif col == 'StockOptionLevel':
            low, high = allowed
            target = (low + high) * 0.75
            step = max(1, int((high - low) / 4))
            val = round(target / step) * step
            changes[col] = val

    return changes


if __name__ == "__main__":
    import csv

    with open("WA_Fn-UseC_-HR-Employee-Attrition.train.csv", newline="") as f:
        reader = csv.reader(f)
        header = next(reader)

    attr_idx = header.index('attrition')
    bad_row = None
    for r in rows:
        if r[attr_idx] == 'Yes':
            bad_row = r
            break

    changes = fix(header, bad_row, {
        'businessTravel': ['Non-Travel', 'Travel_Rarely', 'Travel_Frequently'],
        'overTime': ['Y', 'N'],
        'TrainingTimesLastYear': (0, 25),
        'PercentSalaryHike': (0, 100),
        'StockOptionLevel': (0, 4)
    })

    print("Recommended changes to keep this employee:")
    for col, val in changes.items():
        print(f"  {col}: {val}")
