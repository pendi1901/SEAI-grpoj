import csv

def fix(header, row, levers):
    """
    Recommends changes to improve retention for a 'bad' row (low tenure).
    We prioritize increasing pay (PercentSalaryHike), stock options, 
    and reducing overtime or travel if they are high.
    """
    recommendations = {}
    
    # Map indices for quick lookup
    idx_map = {name: i for i, name in enumerate(header)}
    
    # 1. If OverTime is 'Yes', try to change to 'No'
    if 'overTime' in levers and row[idx_map['overTime']] == 'Yes':
        if 'No' in levers['overTime']:
            recommendations['overTime'] = 'No'
            
    # 2. If StockOptionLevel is low, try to increase it
    if 'StockOptionLevel' in levers:
        low, high = levers['StockOptionLevel']
        if row[idx_map['StockOptionLevel']] < high:
            recommendations['StockOptionLevel'] = high
            
    # 3. If PercentSalaryHike is low, try to increase it
    if 'PercentSalaryHike' in levers:
        low, high = levers['PercentSalaryHike']
        if row[idx_map['PercentSalaryHike']] < high:
            recommendations['PercentSalaryHike'] = high
            
    # 4. If BusinessTravel is frequent, try to reduce it
    if 'businessTravel' in levers and row[idx_map['businessTravel']] == 'Travel_Frequently':
        if 'Travel_Rarely' in levers['businessTravel']:
            recommendations['businessTravel'] = 'Travel_Rarely'
            
    # Ensure at least one change is made if possible
    if not recommendations and levers:
        # Fallback: pick the first available lever and set to a reasonable value
        first_key = list(levers.keys())[0]
        val = levers[first_key]
        recommendations[first_key] = val[0] if isinstance(val, list) else val[1]
        
    return recommendations

if __name__ == "__main__":
    filename = 'WA_Fn-UseC_-HR-Employee-Attrition.n1000.train.csv'
    rows = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = []
            for i, val in enumerate(row):
                if header[i][0].isupper():
                    try:
                        processed_row.append(float(val))
                    except ValueError:
                        processed_row.append(0.0)
                else:
                    processed_row.append(val)
            rows.append(processed_row)

    # Define levers based on the customer's request
    levers = {
        'businessTravel': ['Non-Travel', 'Travel_Rarely', 'Travel_Frequently'],
        'overTime': ['No', 'Yes'],
        'TrainingTimesLastYear': (0.0, 6.0),
        'PercentSalaryHike': (11.0, 25.0),
        'StockOptionLevel': (0.0, 3.0)
    }

    # Simulate a 'bad' row (e.g., low tenure)
    bad_row = rows[0]
    changes = fix(header, bad_row, levers)

    print("Retention Strategy Recommendations:")
    for col, val in changes.items():
        print(f"- Change '{col}' to: {val}")
    print("\nThese adjustments are recommended to improve employee retention based on identified levers.")
