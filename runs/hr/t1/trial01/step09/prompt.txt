def guess(header, train, test):
    goals_min = [c for c in header if c.endswith('-')]
    goals_max = [c for c in header if c.endswith('+')]

    # Build a simple linear model: predict each goal column from all numeric inputs.
    # We use only the non-goal columns as features (symbolic + numeric).
    feature_cols = []
    for c in header:
        if c not in goals_min and c not in goals_max:
            feature_cols.append(c)

    n_train = len(train)
    n_features = len(feature_cols)

    # Collect training data per goal column.
    train_data = {c: [] for c in goals_min + goals_max}
    for row in train:
        feats = [float(row[c]) if c in feature_cols else 0.0 for c in feature_cols]
        for g in goals_min:
            train_data[g].append((feats, float(row[g])))
        for g in goals_max:
            train_data[g].append((feats, float(row[g])))

    # Fit a simple linear regression per goal using normal equations.
    # y = X @ w + b  ->  minimize sum((y - Xw - b)^2)
    def fit_linear(data):
        n = len(data)
        if n == 0:
            return [0.0] * n_features, 0.0

        # Compute sums for normal equations: (X^T X + reg*I) w = X^T y
        reg = 1e-6
        XtX = [[reg] * n_features for _ in range(n_features)]
        Xty = [0.0] * n_features
        sum_y = 0.0

        for feats, y in data:
            sum_y += y
            for j in range(n_features):
                xj = feats[j] if j < len(feats) else 0.0
                Xty[j] += xj * y
                for i in range(j + 1):
                    Xi = feats[i] if i < len(feats) else 0.0
                    XtX[i][j] += xj * Xi

        # Solve using Gaussian elimination with partial pivoting.
        m = n_features
        aug = [XtX[i][:] + [Xty[i]] for i in range(m)]
        for col in range(m):
            max_row = col
            for row in range(col + 1, m):
                if abs(aug[row][col]) > abs(aug[max_row][col]):
                    max_row = row
            aug[col], aug[max_row] = aug[max_row], aug[col]
            pivot = aug[col][col]
            if abs(pivot) < 1e-12:
                continue
            for row in range(col + 1, m):
                factor = aug[row][col] / pivot
                for k in range(col, m + 1):
                    aug[row][k] -= factor * aug[col][k]

        w = [0.0] * m
        b = sum_y / n if n > 0 else 0.0
        for i in range(m - 1, -1, -1):
            s = aug[i][m]
            for j in range(i + 1, m):
                s -= aug[i][j] * w[j]
            if abs(aug[i][i]) < 1e-12:
                w[i] = 0.0
            else:
                w[i] = s / aug[i][i]

        return w, b

    models = {}
    for g in goals_min + goals_max:
        w, b = fit_linear(train_data[g])
        models[g] = (w, b)

    # Predict first goal column for each test row.
    results = []
    for row in test:
        feats = [float(row[c]) if c in feature_cols else 0.0 for c in feature_cols]
        g = goals_min[0]  # first goal column
        w, b = models[g]
        pred = sum(w[j] * feats[j] for j in range(len(feats))) + b
        results.append(pred)

    return results

import csv


def certify(xs, ys):
    """Returns True if xs and ys are the same apart from noise, False if they really differ."""
    mean_x = sum(xs) / len(xs) if xs else 0.0
    mean_y = sum(ys) / len(ys) if ys else 0.0

    # Use standard deviations to gauge natural variability in each group.
    def sd(vals):
        m = sum(vals) / len(vals)
        return (sum((v - m) ** 2 for v in vals) / len(vals)) ** 0.5 if len(vals) > 1 else 0.0

    sx, sy = sd(xs), sd(ys)
    # If both groups vary a lot, allow more tolerance; otherwise be stricter.
    noise_scale = (sx + sy) * 0.3
    return abs(mean_x - mean_y) <= max(noise_scale, 1.5)

def flow(header, row, seen):
    """Returns True if this person needs a 'what would keep you here' conversation."""
    goals_max = [c for c in header if c.endswith('+')]

    current_score = sum(float(row[header.index(g)]) for g in goals_max)

    if not seen:
        return False  # first row — defer decision until we have context

    scores = [sum(float(s[header.index(g)]) for g in goals_max) for s in seen]
    mean_score = sum(scores) / len(scores)
    variance = sum((s - mean_score) ** 2 for s in scores) / len(scores)
    std_dev = variance ** 0.5

    return current_score < mean_score - std_dev

    import math

    def sd(xs):
        m = mean(xs)
        var = sum((x - m)**2 for x in xs)/len(xs) if len(xs)>1 else 0.0
        return math.sqrt(var)

        std_val = sd(vals)

def justify(header, rows, row):

    goals_max = [c for c in header if c.endswith('+')]

        s = sum(float(r[header.index(g)]) for g in goals_min) + \
             sum(-float(r[header.index(g)]) for g in goals_max)
        return s

    scores = [(score_row(r), r) for r in rows]
    scores.sort(reverse=True)
    threshold = scores[int(len(scores) * 0.25)][0]

    p = sum(1 for _, r in scores if score_row(row) >= threshold) / len(rows)
    reason = (
        f"Among {len(rows)} employees, this person's goal alignment is at the top "
        f"{int(p * 100)}% — they rank better than most of their peers."
    )
    return p, reason

    # Since all goal cells are "?", we use the first non-goal cell to infer types.
    non_goal_cols = [c for c in header if c not in goals_min and c not in goals_max]

    col_types = {}
    for c in header:
        val = rows[0][header.index(c)]
        if val == "?":
            col_types[c] = "symbolic"
        else:
            try:
                float(val)

    # For each row, compute a score based on goal alignment.

        s = 0.0

            idx = header.index(g)
            val = float(row[idx]) if col_types[g] == "numeric" else 0.0
            s -= val
        return s

    # We don't know the true scores since goals are "?", so we use a proxy:
    # score based on non-goal columns that correlate with retention.
    # From step05, we know attrition and goal columns; remaining numeric cols are features.
    feature_cols = [c for c in non_goal_cols if col_types[c] == "numeric"]

    # Compute a proxy score using feature columns (these capture the underlying patterns).
    def proxy_score(row):
        s = 0.0

            val = float(row[idx]) if col_types[c] == "numeric" else 0.0
            # Simple heuristic: weight by column index to break ties, but primarily use the values.
            s += val * (1.0 / (len(feature_cols) + idx))

            best_idx = i

    if best_idx == -1:

    return best_idx
