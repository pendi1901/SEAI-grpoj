import csv

def certify(xs, ys):
    """
    Returns True if the two distributions are statistically similar (noise),
    False if they differ significantly. Uses a simple difference of means
    relative to the pooled standard deviation (Cohen's d approximation).
    """
    if not xs or not ys:
        return True
    
    n1, n2 = len(xs), len(ys)
    mu1, mu2 = sum(xs) / n1, sum(ys) / n2
    
    var1 = sum((x - mu1) ** 2 for x in xs) / (n1 - 1) if n1 > 1 else 0
    var2 = sum((y - mu2) ** 2 for y in ys) / (n2 - 1) if n1 > 1 else 0
    
    pooled_std = ((var1 + var2) / 2) ** 0.5
    
    if pooled_std == 0:
        return mu1 == mu2
    
    # Cohen's d threshold: 0.5 is a standard medium effect size
    d = abs(mu1 - mu2) / pooled_std
    return d < 0.5

def analyze_training_impact(header, rows):
    """
    Compares 'YearsWithCurrManager+' for employees with high training
    vs low training (split by median).
    """
    train_idx = header.index('TrainingTimesLastYear')
    tenure_idx = header.index('YearsWithCurrManager+')
    
    training_vals = [row[train_idx] for row in rows]
    median_train = sorted(training_vals)[len(training_vals) // 2]
    
    high_train = [row[tenure_idx] for row in rows if row[train_idx] > median_train]
    low_train = [row[tenure_idx] for row in rows if row[train_idx] <= median_train]
    
    is_noise = certify(high_train, low_train)
    
    avg_high = sum(high_train) / len(high_train) if high_train else 0
    avg_low = sum(low_train) / len(low_train) if low_train else 0
    
    return is_noise, avg_high, avg_low

if __name__ == "__main__":
    filename = 'WA_Fn-UseC_-HR-Employee-Attrition.n1000.train.csv'
    rows = []
    header = []
    
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = []
            for i, val in enumerate(row):
                if header[i][0].isupper():
                    try:
                        processed_row.append(float(val))
                    except ValueError:
                        processed_row.append(0.0)
                else:
                    processed_row.append(val)
            rows.append(processed_row)
            
    is_noise, avg_high, avg_low = analyze_training_impact(header, rows)
    
    print("Analysis: Do more training days lead to longer tenure with the manager?")
    print(f"Average tenure (High Training): {avg_high:.2f} years")
    print(f"Average tenure (Low/Med Training): {avg_low:.2f} years")
    
    if is_noise:
        print("Conclusion: The difference is likely just noise; training frequency does not significantly change tenure with the manager.")
    else:
        print("Conclusion: There is a statistically significant difference in tenure with the manager based on training frequency.")
