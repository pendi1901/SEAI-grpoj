import csv
import random

def choose(header, rows, label):
    """
    Finds the best row by sampling 50 rows, labeling them to get their true
    tenure values, and selecting the one with the highest tenure score.
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith('+') or name.endswith('-')]
    
    # We have 800 rows, we can label 50. 
    # We pick 50 random indices to evaluate.
    indices_to_label = random.sample(range(len(rows)), min(50, len(rows)))
    
    best_idx = -1
    best_score = -float('inf')
    
    for idx in indices_to_label:
        # Get true values for the goals
        true_goals = label(idx)
        
        # Calculate score: sum of '+' goals minus sum of '-' goals
        score = 0
        for i, goal_idx in enumerate(goal_indices):
            val = true_goals[i]
            if header[goal_idx].endswith('+'):
                score += val
            else:
                score -= val
        
        if score > best_score:
            best_score = score
            best_idx = idx
            
    return best_idx

    filename = 'WA_Fn-UseC_-HR-Employee-Attrition.n1000.train.csv'
    rows = []

    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = []
            for i, val in enumerate(row):
                if header[i][0].isupper():
                    try:
                        processed_row.append(float(val))
                    except ValueError:
                        processed_row.append(0.0)
                else:
                    processed_row.append(val)
            rows.append(processed_row)
            
    # Mask goal columns
    goal_indices = [i for i, name in enumerate(header) if name.endswith('+') or name.endswith('-')]
    for row in rows:
        for idx in goal_indices:
            row[idx] = "?"
            
    # Define the label function for the harness
    def label(i):
        # In this local execution, we reload the original data to provide the truth
        with open(filename, 'r') as f:
            reader = csv.reader(f)
            next(reader)
            data = list(reader)
            row = data[i]
            return [float(row[idx]) for idx in goal_indices]

    best_row_idx = choose(header, rows, label)
    
    print(f"Analysis complete. The best working arrangement found is at row index {best_row_idx}.")
