import csv

def certify(xs, ys):
    """
    Returns True if the two distributions are statistically similar (noise),
    False if they differ significantly. Uses a simple difference of means
    relative to the pooled standard deviation (Cohen's d approximation).
    """
    if not xs or not ys:
        return True
    
    n1, n2 = len(xs), len(ys)
    mu1, mu2 = sum(xs) / n1, sum(ys) / n2
    
    var1 = sum((x - mu1) ** 2 for x in xs) / (n1 - 1) if n1 > 1 else 0
    var2 = sum((y - mu2) ** 2 for y in ys) / (n2 - 1) if n2 > 1 else 0
    
    pooled_std = ((var1 + var2) / 2) ** 0.5
    
    if pooled_std == 0:
        return mu1 == mu2
    
    # Effect size threshold: Cohen's d > 0.5 is often considered a medium effect
    d = abs(mu1 - mu2) / pooled_std
    return d < 0.5

    current_tenure = row[tenure_idx]

    if current_tenure >= threshold and row[attrition_idx] == 'No':

import csv

def blame(header, rows):
    """
    Identifies input columns that correlate with low tenure (YearsAtCompany+).
    We define 'worst rows' as those in the bottom 20% of tenure.
    We then identify which input columns have the highest variance or 
    distinctive values in these low-tenure rows compared to the rest.
    """
    tenure_idx = header.index('YearsAtCompany+')
    
    # Filter out goal columns and identify input columns
    input_indices = [
        i for i, name in enumerate(header) 
        if not (name.endswith('+') or name.endswith('-'))
    ]
    
    # Sort rows by tenure to find the 'worst' (lowest tenure)
    sorted_rows = sorted(rows, key=lambda r: r[tenure_idx])
    worst_rows = sorted_rows[:int(len(rows) * 0.2)]
    
    # For each input column, calculate how much the values in 'worst' rows
    # differ from the global average/distribution.
    # We look for columns where the values are most concentrated or distinct.
    scores = []
    for i in input_indices:
        col_name = header[i]
        values = [r[i] for r in worst_rows]
        
        # Simple heuristic: if numeric, check if the mean is significantly lower/higher
        # than the global mean. If symbolic, check for high frequency of a specific value.
        if isinstance(values[0], float):
            global_mean = sum(r[i] for r in rows) / len(rows)
            worst_mean = sum(values) / len(values)
            diff = abs(global_mean - worst_mean)
            scores.append((diff, col_name))
        else:
            # For symbolic, check if one value dominates the worst rows
            counts = {}
            for v in values:
                counts[v] = counts.get(v, 0) + 1
            max_freq = max(counts.values()) / len(values)
            scores.append((max_freq, col_name))
            
    # Return the top 3 columns that best explain the short stays
    scores.sort(key=lambda x: x[0], reverse=True)
    return [s[1] for s in scores[:3]]

def justify(header, rows, row):

def choose(header, rows, label):
    """
    Finds the best row by sampling 50 rows, labeling them to get their true
    tenure values, and selecting the one with the highest tenure score.
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith('+') or name.endswith('-')]
    
    # We have 800 rows, we can label 50. 
    # We pick 50 random indices to evaluate.
    indices_to_label = random.sample(range(len(rows)), min(50, len(rows)))
    
    best_idx = -1
    best_score = -float('inf')
    
    for idx in indices_to_label:
        # Get true values for the goals
        true_goals = label(idx)
        
        # Calculate score: sum of '+' goals minus sum of '-' goals
        score = 0
        for i, goal_idx in enumerate(goal_indices):
            val = true_goals[i]
            if header[goal_idx].endswith('+'):
                score += val
            else:
                score -= val
        
        if score > best_score:
            best_score = score
            best_idx = idx
            
    return best_idx

import csv

def spot(header, rows):
    """
    Identifies suspicious rows by checking for outliers in numeric columns.
    A row is considered suspicious if it contains values that are statistically
    improbable (e.g., outside 3 standard deviations from the mean).
    """
    suspicious = []
    num_indices = [i for i, name in enumerate(header) if name[0].isupper()]
    
    # Calculate stats for each numeric column
    stats = {}
    for i in num_indices:
        vals = [r[i] for r in rows if isinstance(r[i], (int, float))]
        if not vals:
            continue
        mu = sum(vals) / len(vals)
        sigma = (sum((x - mu) ** 2 for x in vals) / len(vals)) ** 0.5
        stats[i] = (mu, sigma)
    
    # Score rows based on how many values are extreme outliers
    row_scores = []
    for idx, row in enumerate(rows):
        score = 0
        for i in num_indices:
            if i in stats:
                mu, sigma = stats[i]
                if sigma > 0:
                    val = row[i]
                    # Z-score check: values > 3 standard deviations are highly suspicious
                    z = abs(val - mu) / sigma
                    if z > 3:
                        score += z
        if score > 0:
            row_scores.append((score, idx))
            
    # Sort by score descending
    row_scores.sort(key=lambda x: x[0], reverse=True)
    return [item[1] for item in row_scores]
