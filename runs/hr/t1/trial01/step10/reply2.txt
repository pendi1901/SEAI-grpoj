def race(header, rows, label):
    """Compare three strategies on up to 50 labelled rows; return results."""
    goals_min = [c for c in header if c.endswith('-')]
    goals_max = [c for c in header if c.endswith('+')]

    def score(row):
        s = 0.0
        for g in goals_min:
            idx = header.index(g)
            v = float(row[idx]) if row[idx] != '?' else 0.0
            s += v
        for g in goals_max:
            idx = header.index(g)
            v = float(row[idx]) if row[idx] != '?' else 0.0
            s -= v
        return s

    def random_rows(n):
        import random
        chosen = set()
        while len(chosen) < n:
            i = random.randrange(len(rows))
            chosen.add(i)
        return [rows[i] for i in sorted(chosen)]

    def greedy_rows(n, rows_to_score):
        scored = [(score(r), r) for r in rows_to_score]
        scored.sort(reverse=True)
        return [r for _, r in scored[:n]]

    def random_search():
        """Pick 50 rows at random and score them."""
        sample = random_rows(50)
        return [(score(r), r) for r in sample]

    def greedy_search():
        """Score all rows, pick top-50 by goal alignment."""
        scored = [(score(r), r) for r in rows]
        scored.sort(reverse=True)
        return [r for _, r in scored[:50]]

    def random_then_greedy():
        """Random sample first; then greedy on the whole set."""
        sample = random_rows(50)
        scored = [(score(r), r) for r in rows]
        scored.sort(reverse=True)
        return [r for _, r in scored[:50]]

    strategies = {
        "random": random_search,
        "greedy": greedy_search,
        "random_then_greedy": random_then_greedy,
    }

    results = {}
    for name, fn in strategies.items():
        rows_found, scores = zip(*fn())
        labelled = [r for r in rows_found if label(r) is not None]
        scored_labelled = [(label(r), score(r)) for r in labelled]
        scored_labelled.sort(reverse=True)
        best_row = scored_labelled[0][1]
        results[name] = (best_row, scores[0])

    # Pick the strategy whose best row has the highest true goal score.
    winner_name = max(results.keys(), key=lambda n: results[n][0])
    winner_row = results[winner_name][0]
    winner_score = sum(
        float(winner_row[header.index(g)]) for g in goals_min
    ) + sum(-float(winner_row[header.index(g)]) for g in goals_max)

    return {
        "winner": winner_name,
        "row": results[winner_name][0],
        "tried": list(strategies.keys()),
    }
