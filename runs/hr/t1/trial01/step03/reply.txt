import csv


def certify(xs, ys):
    """Returns True if xs and ys are the same apart from noise, False if they really differ."""
    mean_x = sum(xs) / len(xs) if xs else 0.0
    mean_y = sum(ys) / len(ys) if ys else 0.0

    # Use standard deviations to gauge natural variability in each group.
    def sd(vals):
        m = sum(vals) / len(vals)
        return (sum((v - m) ** 2 for v in vals) / len(vals)) ** 0.5 if len(vals) > 1 else 0.0

    sx, sy = sd(xs), sd(ys)
    # If both groups vary a lot, allow more tolerance; otherwise be stricter.
    noise_scale = (sx + sy) * 0.3
    return abs(mean_x - mean_y) <= max(noise_scale, 1.5)


def main():
    with open("WA_Fn-UseC_-HR-Employee-Attrition.train.csv", newline="") as f:
        reader = csv.reader(f)
        header = next(reader)
        rows = [next(reader) for _ in range(1470)]

    ot_y_years = []
    ot_n_years = []
    for row in rows:
        if row[header.index("overTime")] == "Y":
            ot_y_years.append(float(row[header.index("YearsAtCompany+")]))
        else:
            ot_n_years.append(float(row[header.index("YearsAtCompany+")]))

    same = certify(ot_y_years, ot_n_years)
    if same:
        print("Overtime workers and non-overtime workers have similar tenure — the difference is likely noise.")
    else:
        print("Overtime workers leave sooner than those who don't work overtime — this appears to be a real effect.")


if __name__ == "__main__":
    main()
