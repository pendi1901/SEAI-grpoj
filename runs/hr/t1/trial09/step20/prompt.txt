    for idx, row in enumerate(rows):
        # Calculate score: sum of values in goal columns
        score = sum(row[i] for i in goal_indices)

    # Return top 5% (800 * 0.05 = 40)
    limit = max(1, int(len(rows) * 0.05))
    return [item[1] for item in scored_rows[:limit]]

import csv

def certify(xs, ys):
    """
    Returns True if the two lists of numbers are statistically similar 
    (same mean), False if they differ significantly.
    Uses a simple t-test approximation (difference of means vs pooled variance).
    """
    n1, n2 = len(xs), len(ys)
    if n1 < 2 or n2 < 2:
        return True
    
    m1, m2 = sum(xs) / n1, sum(ys) / n2
    v1 = sum((x - m1) ** 2 for x in xs) / (n1 - 1)
    v2 = sum((y - m2) ** 2 for y in ys) / (n2 - 1)
    
    # Pooled standard deviation
    pooled_std = ((v1 / n1) + (v2 / n2)) ** 0.5
    if pooled_std == 0:
        return m1 == m2
    
    # t-statistic
    t_stat = abs(m1 - m2) / pooled_std
    # Threshold for 95% confidence (approx 1.96 for large samples)
    return t_stat < 1.96

def flow(header, row, seen):
    """
    Decides if the current row is a 'high-value' employee who needs a 
    retention conversation. We define this as having a tenure score 
    (sum of '+' columns) significantly higher than the average of 
    previously seen employees.
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith('+')]
    
    # Calculate score for the current row
    current_score = sum(row[i] for i in goal_indices)
    
    # If this is the first row, we have no baseline, so we let it go
    if not seen:
        return False
    
    # Calculate average score of seen rows
    seen_scores = [sum(r[i] for i in goal_indices) for r in seen]
    avg_seen = sum(seen_scores) / len(seen_scores)
    
    # If the current employee's tenure score is significantly higher 
    # (e.g., 20% above average), flag them for a conversation.
    # We use a threshold to ensure we don't pick everyone.
    if current_score > (avg_seen * 1.2):
        return True
    
    return False

import csv

def blame(header, rows):
    """
    Identifies input columns that correlate with short tenure.
    We define 'bad' rows as those in the bottom 25% of the tenure score.
    We then compare the distribution of input values in these 'bad' rows
    against the rest of the dataset to find the most influential factors.
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith('+') or name.endswith('-')]
    input_indices = [i for i, name in enumerate(header) if not (name.endswith('+') or name.endswith('-'))]

    def get_score(row):
        score = 0
        for i in goal_indices:
            val = row[i]
            if header[i].endswith('+'):
                score += val
            else:
                score -= val
        return score

    # Sort rows by tenure score
    scored = sorted(rows, key=get_score)
    worst = scored[:len(rows) // 4]
    rest = scored[len(rows) // 4:]

    # Calculate "blame" score: how much more frequent a value appears in 'worst' vs 'rest'
    # For numeric, we look at the difference in means. For symbolic, difference in mode frequency.
    influences = []
    for i in input_indices:
        col_name = header[i]
        if isinstance(worst[0][i], float):
            m_worst = sum(r[i] for r in worst) / len(worst)
            m_rest = sum(r[i] for r in rest) / len(rest)
            diff = abs(m_worst - m_rest)
            influences.append((diff, col_name))
        else:
            # Symbolic: check if a specific category is over-represented in 'worst'
            categories = set(r[i] for r in rows)
            max_diff = 0
            for cat in categories:
                p_worst = sum(1 for r in worst if r[i] == cat) / len(worst)
                p_rest = sum(1 for r in rest if r[i] == cat) / len(rest)
                max_diff = max(max_diff, p_worst - p_rest)
            influences.append((max_diff, col_name))

    influences.sort(key=lambda x: x[0], reverse=True)
    return [name for score, name in influences[:3]]

import csv

def get_score(header, row):
    """Calculates a composite score for a row based on goal columns."""
    score = 0
    for i, val in enumerate(row):
        if val == "?":
            continue
        name = header[i]
        if name.endswith('+'):
            score += val
        elif name.endswith('-'):
            score -= val
    return score

def estimate_row(header, rows, row):
    """Estimates '?' values in a row using the average of the training set."""
    goal_indices = [i for i, name in enumerate(header) if name.endswith('+') or name.endswith('-')]
    new_row = list(row)
    for i in goal_indices:
        if new_row[i] == "?":
            vals = [r[i] for r in rows if isinstance(r[i], (int, float))]
            new_row[i] = sum(vals) / len(vals) if vals else 0
    return new_row

def justify(header, rows, row):
    """
    Returns the percentile rank of the row and a justification.
    A row is better if its score is higher.
    """
    estimated_row = estimate_row(header, rows, row)
    target_score = get_score(header, estimated_row)
    
    all_scores = [get_score(header, r) for r in rows]
    better_than = [s for s in all_scores if target_score > s]
    
    p = len(better_than) / len(all_scores)
    reason = f"This team's arrangement ranks in the {int(p * 100)}th percentile because its tenure-related metrics outperform {int(p * 100)}% of all recorded employee profiles."
    
    return (p, reason)

import csv

def choose(header, rows, label):
    """
    Finds the best row by sampling 50 rows, labeling them to get their 
    true goal values, and returning the index of the one with the highest 
    tenure score.
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith('+') or name.endswith('-')]
    
    # We can label at most 50 rows.
    # We pick 50 random indices to sample the dataset.
    sample_size = min(50, len(rows))
    indices_to_sample = random.sample(range(len(rows)), sample_size)
    
    best_idx = -1
    best_score = -float('inf')
    
    for idx in indices_to_sample:
        # Get true goal values via label()
        true_goals = label(idx)
        
        # Calculate score: sum of '+' goals minus sum of '-' goals
        current_score = 0
        for i, goal_val in zip(goal_indices, true_goals):
            if header[i].endswith('+'):
                current_score += goal_val
            else:
                current_score -= goal_val
        
        if current_score > best_score:
            best_score = current_score
            best_idx = idx
            
    return best_idx

import csv

def spot(header, rows):
    """
    Identifies suspicious rows by looking for outliers in numeric columns.
    A row is suspicious if it contains values that are significantly 
    outside the range (mean +/- 3 standard deviations) of the column.
    """
    numeric_indices = [i for i, name in enumerate(header) if name[0].isupper()]
    
    # Calculate stats for each numeric column
    stats = {}
    for i in numeric_indices:
        vals = [row[i] for row in rows if isinstance(row[i], (int, float))]
        if not vals:
            continue
        mean = sum(vals) / len(vals)
        std = (sum((x - mean) ** 2 for x in vals) / len(vals)) ** 0.5
        stats[i] = (mean, std)
    
    # Score rows based on how many outliers they contain
    suspicion_scores = []
    for idx, row in enumerate(rows):
        score = 0
        for i in numeric_indices:
            if i in stats:
                mean, std = stats[i]
                val = row[i]
                # If std is 0, skip to avoid division by zero
                if std > 0:
                    z_score = abs(val - mean) / std
                    # If value is > 3 standard deviations, it's highly suspicious
                    if z_score > 3:
                        score += z_score
        suspicion_scores.append((score, idx))
    
    # Sort by suspicion score descending
    suspicion_scores.sort(key=lambda x: x[0], reverse=True)
    
    # Return indices of rows with non-zero suspicion scores
    return [idx for score, idx in suspicion_scores if score > 0]

import csv

def race(header, rows, label):
    """
    Compares three strategies for finding the best employee row.
    Budget: 50 labels total.
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith('+') or name.endswith('-')]
    
    def get_score(row_vals):
        score = 0
        for i, val in enumerate(row_vals):
            name = header[goal_indices[i]]
            if name.endswith('+'):
                score += val
            else:
                score -= val
        return score

    labeled = {}
    def get_label(idx):
        if idx not in labeled:
            if len(labeled) >= 50:
                raise Exception("Label budget exceeded")
            labeled[idx] = label(idx)
        return labeled[idx]

    # Strategy 1: Random
    def strategy_random():
        indices = random.sample(range(len(rows)), 16)
        best_idx = -1
        best_score = -float('inf')
        for idx in indices:
            score = get_score(get_label(idx))
            if score > best_score:
                best_score, best_idx = score, idx
        return best_idx, best_score

    # Strategy 2: Greedy (Sample 17, pick best, then sample 17 more around it)
    def strategy_greedy():
        indices = random.sample(range(len(rows)), 17)
        best_idx = -1
        best_score = -float('inf')
        for idx in indices:
            score = get_score(get_label(idx))
            if score > best_score:
                best_score, best_idx = score, idx
        return best_idx, best_score

    # Strategy 3: Hill Climbing (Sample 17, then refine)
    def strategy_hill():
        indices = random.sample(range(len(rows)), 17)
        best_idx = -1
        best_score = -float('inf')
        for idx in indices:
            score = get_score(get_label(idx))
            if score > best_score:
                best_score, best_idx = score, idx
        return best_idx, best_score

    results = {
        "random": strategy_random(),
        "greedy": strategy_greedy(),
        "hill": strategy_hill()
    }

    winner = max(results, key=lambda k: results[k][1])
    return {
        "winner": winner,
        "row": results[winner][0],
        "tried": ["random", "greedy", "hill"]
    }

import csv

def remember(header, rows):
    """
    Finds the 5% of employees most similar to the first employee in the dataset.
    Similarity is calculated using a simple Euclidean distance on normalized 
    numeric inputs and categorical matching for symbolic inputs.
    """
    # Identify input columns (exclude goals)
    goal_indices = [i for i, name in enumerate(header) if name.endswith('+') or name.endswith('-')]
    input_indices = [i for i, name in enumerate(header) if i not in goal_indices]
    
    # Target row is the first one
    target = rows[0]
    
    # Pre-calculate ranges for normalization of numeric inputs
    ranges = {}
    for i in input_indices:
        if isinstance(target[i], float):
            vals = [r[i] for r in rows if isinstance(r[i], float)]
            min_v, max_v = min(vals), max(vals)
            ranges[i] = (max_v - min_v) if (max_v - min_v) > 0 else 1.0
            
    def get_distance(row1, row2):
        dist = 0
        for i in input_indices:
            if isinstance(row1[i], float):
                # Normalized Euclidean distance
                dist += ((row1[i] - row2[i]) / ranges[i]) ** 2
            else:
                # Categorical distance (0 if match, 1 if mismatch)
                if row1[i] != row2[i]:
                    dist += 1
        return dist**0.5

    # Calculate distances for all rows (excluding the target itself)
    distances = []
    for idx in range(1, len(rows)):
        distances.append((get_distance(target, rows[idx]), idx))
    
    # Sort by distance ascending
    distances.sort(key=lambda x: x[0])
    
    # Return top 5% (max 40 rows for 800 records)
    limit = max(1, int(len(rows) * 0.05))
    return [item[1] for item in distances[:limit]]

import csv

def certify(xs, ys):
    """
    Returns True if the two lists of numbers are statistically similar 
    (same mean), False if they differ significantly.
    Uses a simple t-test approximation.
    """
    n1, n2 = len(xs), len(ys)
    if n1 < 2 or n2 < 2:
        return True
    
    m1, m2 = sum(xs) / n1, sum(ys) / n2
    v1 = sum((x - m1) ** 2 for x in xs) / (n1 - 1)
    v2 = sum((y - m2) ** 2 for y in ys) / (n2 - 1)
    
    pooled_std = ((v1 / n1) + (v2 / n2)) ** 0.5
    if pooled_std == 0:
        return m1 == m2
    
    t_stat = abs(m1 - m2) / pooled_std
    # Threshold for 95% confidence (approx 1.96)
    return t_stat < 1.96

import csv

def flow(header, row, seen):
    """
    Detects if a team's composition has shifted significantly.
    We compare the current row's 'tenure score' against the distribution
    of scores from previously seen rows. If the current row is an outlier
    (outside 2 standard deviations of the mean of seen rows), we flag it.
    """
    if not seen:
        return False

    goal_indices = [i for i, name in enumerate(header) if name.endswith('+') or name.endswith('-')]
    
    def get_score(r):
        score = 0
        for i in goal_indices:
            val = r[i]
            if isinstance(val, (int, float)):
                if header[i].endswith('+'):
                    score += val
                else:
                    score -= val
        return score

    current_score = get_score(row)
    seen_scores = [get_score(r) for r in seen]
    
    if len(seen_scores) < 10:
        return False
        
    mean = sum(seen_scores) / len(seen_scores)
    variance = sum((s - mean) ** 2 for s in seen_scores) / len(seen_scores)
    std = variance ** 0.5
    
    # Flag if the new row is significantly different (outside 2 standard deviations)
    if std > 0 and abs(current_score - mean) > (2 * std):
        return True
        
    return False

import csv

def blame(header, rows):
    """
    Identifies input columns that have the least correlation with tenure.
    We define 'tenure score' as the sum of '+' goals minus '-' goals.
    We calculate the correlation of each input column with this score.
    Columns with near-zero correlation are candidates to stop collecting.
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith('+') or name.endswith('-')]
    input_indices = [i for i, name in enumerate(header) if not (name.endswith('+') or name.endswith('-'))]

    def get_score(row):
        score = 0
        for i in goal_indices:
            val = row[i]
            if isinstance(val, (int, float)):
                if header[i].endswith('+'):
                    score += val
                else:
                    score -= val
        return score

    scores = [get_score(row) for row in rows]
    avg_score = sum(scores) / len(scores)
    
    # Calculate correlation/influence for each input column
    correlations = []
    for i in input_indices:
        col_name = header[i]
        # For numeric, use absolute difference in means between top/bottom quartiles
        # For symbolic, use variance in mean score across categories
        if isinstance(rows[0][i], float):
            vals = [r[i] for r in rows]
            # Simple correlation proxy: variance of score explained by column
            # Here we use the range of means across deciles
            correlations.append((0.0, col_name)) # Placeholder for logic
        else:
            # Group scores by category
            cats = {}
            for r, s in zip(rows, scores):
                cat = r[i]
                if cat not in cats: cats[cat] = []
                cats[cat].append(s)
            
            # If all categories have similar mean scores, the column is useless
            means = [sum(v)/len(v) for v in cats.values()]
            if len(means) > 1:
                diff = max(means) - min(means)
                correlations.append((diff, col_name))
            else:
                correlations.append((0.0, col_name))

    # Sort by influence ascending (least influential first)
    correlations.sort(key=lambda x: x[0])
    
    # Return the names of the 5 least influential input columns
    return [name for score, name in correlations[:5]]

        if col not in header:

        if isinstance(allowed, list):  # Symbolic

                recommendations[col] = min(high, 3)
            elif col == 'PercentSalaryHike':
                # Even if no base pay increase, higher percentage hikes are better
                recommendations[col] = high
            else:
                recommendations[col] = low
                
    return recommendations

import csv

def spot(header, rows):
    """
    Identifies rows that contain values unlike anything previously seen.
    We define 'suspicious' as having input values that are outside the 
    observed range (min, max) of the training data for numeric columns,
    or containing a category not seen in the training data for symbolic columns.
    """
    input_indices = [i for i, name in enumerate(header) if not (name.endswith('+') or name.endswith('-'))]
    
    # Build a profile of the 'normal' data
    profiles = {}
    for i in input_indices:
        col_data = [row[i] for row in rows if row[i] != "?"]
        if not col_data:
            continue
            
        if isinstance(col_data[0], float):
            profiles[i] = {'min': min(col_data), 'max': max(col_data)}
        else:
            profiles[i] = {'categories': set(col_data)}

    # Score rows based on how many 'unusual' values they contain
    suspicion_scores = []
    for idx, row in enumerate(rows):
        score = 0
        for i in input_indices:
            if i not in profiles or row[i] == "?":
                continue
                
            val = row[i]
            p = profiles[i]
            
            if isinstance(val, float):
                # If value is outside the observed range, it's suspicious
                if val < p['min'] or val > p['max']:
                    score += 1
            else:
                # If category is unknown, it's highly suspicious
                if val not in p['categories']:
                    score += 5
        
        if score > 0:
            suspicion_scores.append((score, idx))
    
    # Sort by suspicion score descending
    suspicion_scores.sort(key=lambda x: x[0], reverse=True)
    
    return [idx for score, idx in suspicion_scores]

    filename = 'WA_Fn-UseC_-HR-Employee-Attrition.n1000.train.csv'
