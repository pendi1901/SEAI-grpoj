import csv

def remember(header, rows):
    """
    Identifies staff who stay 'unusually long'.
    We define 'staying long' by the goal columns:
    YearsAtCompany+, YearsInCurrentRole+, YearsWithCurrManager+
    We calculate a score based on the sum of these values and return
    the indices of the top 5% (max 40 rows for 800 records).
    """
    # Identify indices for goal columns ending in '+'
    goal_indices = [i for i, name in enumerate(header) if name.endswith('+')]
    
    scored_rows = []
    for idx, row in enumerate(rows):
        # Calculate score: sum of values in goal columns
        score = sum(row[i] for i in goal_indices)
        scored_rows.append((score, idx))
    
    # Sort by score descending
    scored_rows.sort(key=lambda x: x[0], reverse=True)
    
    # Return top 5% (800 * 0.05 = 40)
    limit = max(1, int(len(rows) * 0.05))
    return [item[1] for item in scored_rows[:limit]]

import csv

def guess(header, train, test):
    """
    Estimates the value of the first goal column (YearsAtCompany+)
    for test rows based on the average value of that column in the training set.
    """
    # Find the index of the first goal column
    goal_idx = -1
    for i, name in enumerate(header):
        if name.endswith('+') or name.endswith('-'):
            goal_idx = i
            break
    
    if goal_idx == -1:
        return [0.0] * len(test)
    
    # Calculate the average of the first goal column from training data
    total = sum(row[goal_idx] for row in train)
    avg = total / len(train)
    
    # Return the average as the estimate for every test row
    return [float(avg)] * len(test)

import csv

def certify(xs, ys):
    """
    Returns True if the two lists of numbers are statistically similar 
    (same mean), False if they differ significantly.
    Uses a simple t-test approximation (difference of means vs pooled variance).
    """
    n1, n2 = len(xs), len(ys)
    if n1 < 2 or n2 < 2:
        return True
    
    m1, m2 = sum(xs) / n1, sum(ys) / n2
    v1 = sum((x - m1) ** 2 for x in xs) / (n1 - 1)
    v2 = sum((y - m2) ** 2 for y in ys) / (n2 - 1)
    
    # Pooled standard deviation
    pooled_std = ((v1 / n1) + (v2 / n2)) ** 0.5
    if pooled_std == 0:
        return m1 == m2
    
    # t-statistic
    t_stat = abs(m1 - m2) / pooled_std
    # Threshold for 95% confidence (approx 1.96 for large samples)
    return t_stat < 1.96

import csv

def flow(header, row, seen):
    """
    Decides if the current row is a 'high-value' employee who needs a 
    retention conversation. We define this as having a tenure score 
    (sum of '+' columns) significantly higher than the average of 
    previously seen employees.
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith('+')]
    
    # Calculate score for the current row
    current_score = sum(row[i] for i in goal_indices)
    
    # If this is the first row, we have no baseline, so we let it go
    if not seen:
        return False
    
    # Calculate average score of seen rows
    seen_scores = [sum(r[i] for i in goal_indices) for r in seen]
    avg_seen = sum(seen_scores) / len(seen_scores)
    
    # If the current employee's tenure score is significantly higher 
    # (e.g., 20% above average), flag them for a conversation.
    # We use a threshold to ensure we don't pick everyone.
    if current_score > (avg_seen * 1.2):
        return True
    
    return False
