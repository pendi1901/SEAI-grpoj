    first_goal_idx = goal_indices[0]
    
    # Identify numeric input columns (excluding goals)
    input_indices = [
        i for i, name in enumerate(header) 
        if name[0].isupper() and i not in goal_indices
    ]
    
    # Normalize numeric inputs for distance calculation
    def get_features(row):
        return [row[i] for i in input_indices]

    # Simple KNN: find the average of the first goal column for the 5 nearest neighbors
    predictions = []
    for t_row in test:
        t_feats = get_features(t_row)
        distances = []
        for tr_row in train:
            tr_feats = get_features(tr_row)

import csv

def certify(xs, ys):
    """
    Performs a simple two-sample t-test (Welch's) to determine if 
    two populations are statistically different.
    Returns True if they are likely the same (cannot reject null hypothesis),
    False if they are significantly different.
    """
    n1, n2 = len(xs), len(ys)
    if n1 < 2 or n2 < 2:
        return True
    
    m1, m2 = sum(xs) / n1, sum(ys) / n2
    v1 = sum((x - m1) ** 2 for x in xs) / (n1 - 1)
    v2 = sum((y - m2) ** 2 for y in ys) / (n2 - 1)
    
    # Standard error
    se = ((v1 / n1) + (v2 / n2)) ** 0.5
    if se == 0:
        return m1 == m2
    
    t_stat = abs(m1 - m2) / se
    # For a 95% confidence level, critical t-value is approx 1.96
    return t_stat < 1.96

def flow(header, row, seen):
    """
    Decides if a row needs a 'what would keep you here' conversation.
    We identify 'at-risk' employees as those with low tenure metrics 
    (YearsAtCompany+, YearsInCurrentRole+, YearsWithCurrManager+) 
    compared to the historical average of the 'seen' population.
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith('+')]
    
    # If this is the first row, we have no baseline, so we don't pick it.
    if not seen:
        return False
    
    # Calculate current row's tenure score
    current_score = sum(row[i] for i in goal_indices)
    
    # Calculate average tenure score of seen rows
    seen_scores = [sum(r[i] for i in goal_indices) for r in seen]
    avg_seen_score = sum(seen_scores) / len(seen_scores)
    
    # If the current employee's tenure is significantly lower than the average 
    # (e.g., less than 50% of the average), they are a candidate for retention.
    # We use a threshold to avoid picking everyone.
    if current_score < (avg_seen_score * 0.5):
        return True
    
    return False

import csv

def blame(header, rows):
    """
    Identifies input columns that correlate with 'bad' performance.
    'Bad' is defined as having low values in goal columns ending in '+'.
    We calculate the correlation between each input column and the 
    average of the goal columns to find what explains the short stays.
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith('+')]
    input_indices = [i for i, name in enumerate(header) if not (name.endswith('+') or name.endswith('-'))]

    # Calculate a normalized score for each row (higher is better)
    def get_score(row):
        return sum(row[i] for i in goal_indices)

    # Calculate average score of the population to find 'bad' rows
    scores = [get_score(row) for row in rows]
    avg_score = sum(scores) / len(scores)
    
    # Focus on rows that are below average (the 'short stayers')
    bad_rows = [row for row in rows if get_score(row) < avg_score]
    if not bad_rows:
        return []

    # For each input column, calculate the average value in 'bad' rows
    # vs the average in the whole population.
    # We look for columns where the 'bad' rows have a distinct distribution.
    insights = []
    for i in input_indices:
        col_name = header[i]
        
        # Only consider numeric inputs for simple correlation
        if isinstance(rows[0][i], float):
            bad_avg = sum(r[i] for r in bad_rows) / len(bad_rows)
            all_avg = sum(r[i] for r in rows) / len(rows)
            # If the average in bad rows is significantly different, it's a candidate
            if all_avg != 0:
                diff = abs(bad_avg - all_avg) / all_avg
                insights.append((diff, col_name))
    
    # Sort by the magnitude of difference and return the top 3
    insights.sort(key=lambda x: x[0], reverse=True)
    return [item[1] for item in insights[:3]]

import csv

def justify(header, rows, row):
    """
    Estimates the performance of a row by comparing its predicted goal values 
    against the historical population, returning the percentile rank and a 
    justification string.
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith('+') or name.endswith('-')]
    input_indices = [i for i, name in enumerate(header) if i not in goal_indices]

    # Simple KNN estimation for the hidden goal values
    def estimate_goals(target_row):
        estimates = []
        for g_idx in goal_indices:
            # Find 5 nearest neighbors based on numeric input columns
            distances = []
            for r in rows:
                dist = 0
                for i in input_indices:
                    if isinstance(r[i], float) and isinstance(target_row[i], float):
                        dist += (r[i] - target_row[i])**2
                distances.append((dist**0.5, r[g_idx]))
            
            distances.sort(key=lambda x: x[0])
            neighbors = [d[1] for d in distances[:5]]
            estimates.append(sum(neighbors) / len(neighbors))
        return estimates

    # Calculate a composite score: (sum of + goals) - (sum of - goals)
    def get_score(goals):
        score = 0
        for i, g_idx in enumerate(goal_indices):
            val = goals[i]
            if header[g_idx].endswith('+'):
                score += val
            else:
                score -= val
        return score

    # Estimate goals for the input row
    target_estimates = estimate_goals(row)
    target_score = get_score(target_estimates)

    # Calculate scores for all existing rows
    all_scores = []
    for r in rows:
        all_scores.append(get_score([r[g] for g in goal_indices]))

    # Calculate percentile (fraction of rows that this row is better than)
    better_than = [s for s in all_scores if target_score > s]
    p = len(better_than) / len(all_scores)

    reason = f"This arrangement ranks in the top {int((1-p)*100)}% of all teams because its projected tenure metrics outperform {int(p*100)}% of historical records."
    
    return (p, reason)

import csv
import random

def choose(header, rows, label):
    """
    Finds the best working arrangement by sampling 50 rows, 
    labeling them to get their true goal values, and picking the 
    one with the highest composite score.
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith('+') or name.endswith('-')]
    
    def get_score(goals):
        score = 0
        for i, g_idx in enumerate(goal_indices):
            val = goals[i]
            # + goals: maximize, - goals: minimize
            if header[g_idx].endswith('+'):
                score += val
            else:
                score -= val
        return score

    # Randomly sample 50 indices to label
    num_to_sample = min(50, len(rows))
    indices_to_label = random.sample(range(len(rows)), num_to_sample)
    
    best_idx = -1
    best_score = float('-inf')
    
    for idx in indices_to_label:
        true_goals = label(idx)
        score = get_score(true_goals)
        
        if score > best_score:
            best_score = score
            best_idx = idx
            
    return best_idx

def fix(header, row, levers):
    """
    Recommends a change to a 'bad' row to improve its goal metrics.
    We evaluate the impact of changing each lever by checking the 
    average goal performance of rows that match the new lever value.
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith('+') or name.endswith('-')]
    
    def get_score(r):
        score = 0
        for g_idx in goal_indices:
            val = r[g_idx]
            if header[g_idx].endswith('+'):
                score += val
            else:
                score -= val
        return score

    # We need a reference to the global rows to calculate the impact of changes.
    # Since the harness passes only the row, we assume the caller or context 
    # provides a way to evaluate, but per requirements, we use the provided 
    # levers to make a heuristic choice.
    
    recommendations = {}
    for col_name, allowed in levers.items():
        idx = header.index(col_name)
        if isinstance(allowed, list):  # Symbolic
            # Pick a random valid value different from current
            choices = [v for v in allowed if v != row[idx]]

            # Move towards the higher end of the range
            low, high = allowed
            recommendations[col_name] = high
            
    # Return at least one change
    return {list(recommendations.keys())[0]: list(recommendations.values())[0]}
