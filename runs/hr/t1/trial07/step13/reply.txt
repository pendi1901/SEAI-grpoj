import csv

def certify(xs, ys):
    """
    Performs a Welch's t-test to determine if two samples have significantly
    different means. Returns True if they are statistically similar (p > 0.05),
    False if they are significantly different (p <= 0.05).
    """
    n1, n2 = len(xs), len(ys)
    if n1 < 2 or n2 < 2:
        return True
    
    m1, m2 = sum(xs) / n1, sum(ys) / n2
    v1 = sum((x - m1) ** 2 for x in xs) / (n1 - 1)
    v2 = sum((y - m2) ** 2 for y in ys) / (n2 - 1)
    
    se = ((v1 / n1) + (v2 / n2)) ** 0.5
    if se == 0:
        return m1 == m2
    
    t_stat = abs(m1 - m2) / se
    # Critical value for t-distribution (approx for alpha=0.05, two-tailed)
    # 2.0 is a standard approximation for large samples
    return t_stat < 2.0

def analyze_training_impact(header, rows):
    """
    Compares 'YearsWithCurrManager+' for employees with high vs low training.
    """
    train_idx = header.index('TrainingTimesLastYear')
    mgr_idx = header.index('YearsWithCurrManager+')
    
    # Split into two groups: High training (>= 3) vs Low training (< 3)
    high_train = [r[mgr_idx] for r in rows if r[train_idx] >= 3]
    low_train = [r[mgr_idx] for r in rows if r[train_idx] < 3]
    
    is_similar = certify(high_train, low_train)
    
    avg_high = sum(high_train) / len(high_train) if high_train else 0
    avg_low = sum(low_train) / len(low_train) if low_train else 0
    
    print("Analysis: Do more training days lead to longer tenure with the manager?")
    print(f"  Avg years with manager (High Training >= 3): {avg_high:.2f}")
    print(f"  Avg years with manager (Low Training < 3):   {avg_low:.2f}")
    
    if not is_similar:
        print("  Result: The difference is statistically significant.")
        if avg_high > avg_low:
            print("  Conclusion: Yes, more training correlates with longer tenure with the manager.")
        else:
            print("  Conclusion: No, more training correlates with shorter tenure with the manager.")
    else:
        print("  Result: The difference is likely just noise; training does not significantly change tenure with the manager.")

if __name__ == "__main__":
    filename = 'WA_Fn-UseC_-HR-Employee-Attrition.n1000.train.csv'
    rows = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = []
            for i, val in enumerate(row):
                if header[i][0].isupper():
                    try:
                        processed_row.append(float(val))
                    except ValueError:
                        processed_row.append(0.0)
                else:
                    processed_row.append(val)
            rows.append(processed_row)
            
    analyze_training_impact(header, rows)
