        return []

    # Create a list of (index, value)
    indexed_rows = []
    for i, row in enumerate(rows):
        val = row[idx]
        if isinstance(val, (int, float)):
            indexed_rows.append((i, val))

    # Sort by value descending
    indexed_rows.sort(key=lambda x: x[1], reverse=True)

    # Calculate 5% limit (at least 1, max 5% of total rows)
    n = len(rows)
    limit = max(1, int(n * 0.05))
    
    # Return the indices of the top performers
    return [item[0] for item in indexed_rows[:limit]]

import csv

def guess(header, train, test):
    """
    Estimates the first goal column ('YearsAtCompany+') for test rows
    using a k-Nearest Neighbors approach, considering only numeric inputs.
    """
    # Identify the first goal column
    goal_cols = [i for i, h in enumerate(header) if h.endswith('+') or h.endswith('-')]
    goal_idx = goal_cols[0]
    
    # Identify numeric input columns (excluding goals and non-numeric columns)
    # A column is numeric if the header starts with an uppercase letter
    input_indices = [
        i for i, h in enumerate(header) 
        if h[0].isupper() and i != goal_idx
    ]
    
    # Normalize data for distance calculation (min-max)
    mins = [min(row[i] for row in train) for i in input_indices]
    maxs = [max(row[i] for row in train) for i in input_indices]
    ranges = [maxs[j] - mins[j] if maxs[j] != mins[j] else 1.0 for j in range(len(input_indices))]
    
    def get_dist(r1, r2):
        d = 0.0
        for j, idx in enumerate(input_indices):
            d += ((r1[idx] - r2[idx]) / ranges[j])**2
        return d**0.5

    predictions = []
    for t_row in test:
        # Find nearest neighbor in training set
        best_dist = float('inf')
        best_val = 0.0
        for train_row in train:
            d = get_dist(t_row, train_row)
            if d < best_dist:
                best_dist = d
                best_val = train_row[goal_idx]
        predictions.append(float(best_val))
        
    return predictions

import csv
import math

def certify(xs, ys):
    """
    Performs a Welch's t-test to determine if two samples have significantly
    different means. Returns True if they are statistically similar (p > 0.05),
    False if they are significantly different (p <= 0.05).
    """
    n1, n2 = len(xs), len(ys)
    if n1 < 2 or n2 < 2:
        return True
    
    m1, m2 = sum(xs) / n1, sum(ys) / n2
    v1 = sum((x - m1) ** 2 for x in xs) / (n1 - 1)
    v2 = sum((y - m2) ** 2 for y in ys) / (n2 - 1)
    
    # Welch's t-statistic
    se = math.sqrt((v1 / n1) + (v2 / n2))
    if se == 0:
        return m1 == m2
    
    t_stat = abs(m1 - m2) / se
    
    # Degrees of freedom (Welch-Satterthwaite)
    df = ((v1 / n1 + v2 / n2) ** 2) / (((v1 / n1) ** 2 / (n1 - 1)) + ((v2 / n2) ** 2 / (n2 - 1)))
    
    # Critical value for t-distribution (approx for alpha=0.05, two-tailed)
    # For df > 30, 2.0 is a reasonable approximation for 95% confidence
    critical_val = 2.0
    
    return t_stat < critical_val

import csv

def flow(header, row, seen):
    """
    Decides if the current employee needs a 'what would keep you here' conversation.
    We target employees who are high performers (high 'YearsAtCompany+') but
    might be at risk (e.g., low 'JobSatisfaction' or 'EnvironmentSatisfaction').
    
    This is an online selection problem. We use a threshold-based approach:
    if the current row's tenure is significantly better than the average 
    tenure seen so far, and they have low satisfaction, we flag them.
    """
    tenure_idx = header.index('YearsAtCompany+')
    sat_idx = header.index('JobSatisfaction')
    
    # If this is the first row, we can't compare, so let it go
    if not seen:
        return False
    
    # Calculate average tenure of seen rows
    avg_tenure = sum(r[tenure_idx] for r in seen) / len(seen)
    
    # Criteria:
    # 1. Tenure is above average (they are experienced)
    # 2. Job Satisfaction is low (<= 2)
    # 3. We only pick one person (the first one that meets the criteria)
    # Since the harness asks to stop once we return True, this acts as a 
    # "first-best-fit" selector.
    
    current_tenure = row[tenure_idx]
    current_sat = row[sat_idx]
    
    if current_tenure > avg_tenure and current_sat <= 2:
        return True
    
    return False

import csv

def blame(header, rows):
    """
    Identifies input columns that correlate with low tenure (YearsAtCompany+).
    We define 'worst rows' as the bottom 25% of employees by tenure.
    We then look for input columns where the values in these worst rows
    deviate significantly from the population average.
    """
    tenure_idx = header.index('YearsAtCompany+')
    
    # Identify input columns (exclude goals)
    input_indices = [
        i for i, h in enumerate(header) 
        if not (h.endswith('+') or h.endswith('-'))
    ]
    
    # Sort rows by tenure to find the worst 25%
    sorted_rows = sorted(rows, key=lambda r: r[tenure_idx])
    n = len(sorted_rows)
    worst_rows = sorted_rows[:max(1, n // 4)]
    
    # Calculate global averages/modes for comparison
    def get_stats(col_idx):
        vals = [r[col_idx] for r in rows]
        if isinstance(vals[0], (int, float)):
            return sum(vals) / len(vals)
        else:
            # Return most frequent for symbolic
            return max(set(vals), key=vals.count)

    # Score columns based on how much the worst rows differ from the norm
    scores = []
    for i in input_indices:
        col_name = header[i]
        worst_vals = [r[i] for r in worst_rows]
        
        if isinstance(worst_vals[0], (int, float)):
            avg_all = get_stats(i)
            avg_worst = sum(worst_vals) / len(worst_vals)
            # Score is the absolute difference normalized
            diff = abs(avg_all - avg_worst)
            scores.append((diff, col_name))
        else:
            # For symbolic, score by how much the distribution shifts
            mode_all = get_stats(i)
            mode_worst = max(set(worst_vals), key=worst_vals.count)
            score = 1.0 if mode_all != mode_worst else 0.0
            scores.append((score, col_name))
            
    # Return top 3 columns that explain the short stays
    scores.sort(key=lambda x: x[0], reverse=True)
    return [s[1] for s in scores[:3]]

import csv

def justify(header, rows, row):
    """
    Estimates the rank of a given row by comparing its predicted goals
    against the actual goals of the provided training rows.
    """
    goal_indices = [i for i, h in enumerate(header) if h.endswith('+') or h.endswith('-')]
    input_indices = [i for i, h in enumerate(header) if i not in goal_indices]
    
    # Simple KNN-based estimation for the goal
    def get_dist(r1, r2):
        d = 0.0
        for i in input_indices:
            if isinstance(r1[i], (int, float)) and isinstance(r2[i], (int, float)):
                d += (r1[i] - r2[i])**2
            else:
                d += 1.0 if r1[i] != r2[i] else 0.0
        return d**0.5

    # Estimate goal values for the input row
    predictions = {}
    for g_idx in goal_indices:
        # Find nearest neighbor
        best_dist = float('inf')
        best_val = 0.0
        for r in rows:
            dist = get_dist(row, r)
            if dist < best_dist:
                best_dist = dist
                best_val = r[g_idx]
        predictions[g_idx] = float(best_val)

    # Calculate how many rows this row is "better than"
    better_count = 0
    for r in rows:
        is_better = True
        for g_idx in goal_indices:
            val_row = predictions[g_idx]
            val_r = r[g_idx]
            # If goal ends in +, higher is better. If -, lower is better.
            if header[g_idx].endswith('+'):
                if not (val_row >= val_r):
                    is_better = False
            else:
                if not (val_row <= val_r):
                    is_better = False
        if is_better:
            better_count += 1
            
    p = better_count / len(rows)
    reason = f"This employee's predicted performance metrics outperform {p:.1%} of the existing workforce."
    return (p, reason)

def choose(header, rows, label):
    """
    Finds the best working arrangement by sampling 50 rows.
    We define 'best' as the highest 'YearsAtCompany+' (a goal to maximize).
    """
    tenure_idx = header.index('YearsAtCompany+')
    
    # We have a budget of 50 labels.
    # We will sample 50 random rows to estimate the best working arrangement.
    n = len(rows)
    sample_size = min(n, 50)
    indices = random.sample(range(n), sample_size)
    
    best_idx = -1
    max_tenure = -1.0
    
    for i in indices:
        # Label the row to get the true goal values
        goals = label(i)
        # The goal 'YearsAtCompany+' is the value we want to maximize
        # Based on the header, it is the 32nd column (index 31)

        # Note: The harness label() returns the list of goal values.
        # Since 'YearsAtCompany+' is the only goal ending in '+', 
        # we find its specific value.
        
        if current_tenure > max_tenure:
            max_tenure = current_tenure
            best_idx = i
            
    return best_idx

                recommendations[col] = float(val[1])

import csv

def spot(header, rows):
    """
    Identifies suspicious rows by looking for outliers in numeric columns.
    A row is considered suspicious if it contains values that are statistically
    improbable (e.g., beyond 3 standard deviations from the mean).
    """
    suspicious_scores = {}
    
    # Identify numeric columns
    numeric_indices = [i for i, h in enumerate(header) if h[0].isupper()]
    
    # Calculate mean and std for each numeric column
    stats = {}
    for i in numeric_indices:
        vals = [r[i] for r in rows if isinstance(r[i], (int, float))]
        if not vals:

def race(header, rows, label):
    """
    Compares three strategies for finding the best employee (maximizing tenure).
    Budget: 50 total labels.
    """
    goal_indices = [i for i, h in enumerate(header) if h.endswith('+') or h.endswith('-')]
    tenure_idx = header.index('YearsAtCompany+')
    
    # Track labels used
    labeled = {}
    def get_label(i):
        if i not in labeled:
            labeled[i] = label(i)
        return labeled[i]

    def get_tenure(i):
        # tenure_idx is the index in the row. 
        # The label() function returns values for goal_indices.
        goals = get_label(i)
        # Find which position in goal_indices corresponds to tenure_idx
        for pos, g_idx in enumerate(goal_indices):
            if g_idx == tenure_idx:
                return goals[pos]
        return 0.0

    # Strategies
    def random_strat():
        indices = random.sample(range(len(rows)), 16)
        best_i = indices[0]
        for i in indices:
            if get_tenure(i) > get_tenure(best_i):
                best_i = i
        return best_i

    def greedy_strat():
        # Sample 17 rows, pick the best
        indices = random.sample(range(len(rows)), 17)
        best_i = indices[0]
        for i in indices:
            if get_tenure(i) > get_tenure(best_i):
                best_i = i
        return best_i

    def exploration_strat():
        # Sample 17 rows, pick the best
        indices = random.sample(range(len(rows)), 17)
        best_i = indices[0]
        for i in indices:
            if get_tenure(i) > get_tenure(best_i):
                best_i = i
        return best_i

    strategies = {
        "random": random_strat,
        "greedy": greedy_strat,
        "exploration": exploration_strat
    }

    results = {}
    for name, func in strategies.items():
        results[name] = func()

    # Determine winner
    winner = max(results, key=lambda n: get_tenure(results[n]))
    
    return {
        "winner": winner,
        "row": results[winner],
        "tried": list(strategies.keys())
    }

import csv

def remember(header, rows):
    """
    Finds the 5% of employees most similar to the first employee in the dataset.
    Similarity is calculated using Euclidean distance on normalized numeric inputs.
    """
    # Identify goal columns and input columns
    goal_indices = [i for i, h in enumerate(header) if h.endswith('+') or h.endswith('-')]
    input_indices = [i for i, h in enumerate(header) if i not in goal_indices and header[i][0].isupper()]
    
    # Target is the first row
    target = rows[0]
    
    # Normalize numeric inputs for distance calculation
    mins = [min(row[i] for row in rows) for i in input_indices]
    maxs = [max(row[i] for row in rows) for i in input_indices]
    ranges = [maxs[j] - mins[j] if maxs[j] != mins[j] else 1.0 for j in range(len(input_indices))]
    
    def get_dist(r1, r2):
        d = 0.0
        for j, idx in enumerate(input_indices):
            d += ((r1[idx] - r2[idx]) / ranges[j])**2
        return d**0.5

    # Calculate distances to target for all rows
    distances = []
    for i, row in enumerate(rows):
        if i == 0: continue  # Skip self
        distances.append((i, get_dist(target, row)))
    
    # Sort by distance ascending
    distances.sort(key=lambda x: x[1])
    
    # Return top 5%
    limit = max(1, int(len(rows) * 0.05))
    return [item[0] for item in distances[:limit]]

import csv

def certify(xs, ys):
    """
    Performs a Welch's t-test to determine if two samples have significantly
    different means. Returns True if they are statistically similar (p > 0.05),
    False if they are significantly different (p <= 0.05).
    """
    n1, n2 = len(xs), len(ys)
    if n1 < 2 or n2 < 2:
        return True
    
    m1, m2 = sum(xs) / n1, sum(ys) / n2
    v1 = sum((x - m1) ** 2 for x in xs) / (n1 - 1)
    v2 = sum((y - m2) ** 2 for y in ys) / (n2 - 1)
    
    se = ((v1 / n1) + (v2 / n2)) ** 0.5
    if se == 0:
        return m1 == m2
    
    t_stat = abs(m1 - m2) / se
    # Critical value for t-distribution (approx for alpha=0.05, two-tailed)
    # 2.0 is a standard approximation for large samples
    return t_stat < 2.0

import csv

def flow(header, row, seen):
    """
    Detects if the current row represents a 'drift' in team composition.
    We compare the current row's numeric distribution to the distribution
    of the 'seen' rows. If the current row's average numeric value is 
    statistically distant (e.g., > 2 standard deviations) from the 
    historical average of the team, we flag it as a change.
    """
    if len(seen) < 30:
        return False

    # Identify numeric columns
    numeric_indices = [i for i, h in enumerate(header) if h[0].isupper()]
    
    # Calculate historical means and standard deviations
    means = {}
    stds = {}
    for i in numeric_indices:
        vals = [r[i] for r in seen if isinstance(r[i], (int, float))]
        if len(vals) > 1:
            m = sum(vals) / len(vals)
            s = (sum((x - m) ** 2 for x in vals) / len(vals)) ** 0.5
            means[i] = m
            stds[i] = s if s > 0 else 1.0

    # Calculate Z-score for the current row
    z_scores = []
    for i in numeric_indices:
        if i in means and isinstance(row[i], (int, float)):
            z = abs(row[i] - means[i]) / stds[i]
            z_scores.append(z)
    
    # If the average Z-score across all numeric fields is high, 
    # the team composition has drifted significantly.
    if z_scores and (sum(z_scores) / len(z_scores)) > 2.5:
        return True
        
    return False

import csv

def blame(header, rows):
    """
    Identifies input columns that have the least correlation with the goal columns.
    We define 'least correlation' as columns where the distribution of values
    is statistically similar between the 'best' and 'worst' performers.
    
    Returns a list of column names that are candidates for removal.
    """
    goal_indices = [i for i, h in enumerate(header) if h.endswith('+') or h.endswith('-')]
    input_indices = [i for i, h in enumerate(header) if i not in goal_indices]
    
    # Define 'best' and 'worst' rows based on the first goal column
    goal_idx = goal_indices[0]
    is_max = header[goal_idx].endswith('+')
    
    sorted_rows = sorted(rows, key=lambda r: r[goal_idx], reverse=is_max)
    n = len(sorted_rows)
    # Compare top 25% vs bottom 25%
    top_25 = sorted_rows[:max(1, n // 4)]
    bot_25 = sorted_rows[-max(1, n // 4):]
    
    def get_dist_diff(col_idx):
        vals_top = [r[col_idx] for r in top_25]
        vals_bot = [r[col_idx] for r in bot_25]
        
        if isinstance(vals_top[0], (int, float)):
            m_top = sum(vals_top) / len(vals_top)
            m_bot = sum(vals_bot) / len(vals_bot)
            # Normalize difference by range
            all_vals = [r[col_idx] for r in rows]
            r_range = max(all_vals) - min(all_vals)
            return abs(m_top - m_bot) / (r_range if r_range != 0 else 1.0)
        else:
            # For symbolic, check if the distribution of categories is similar
            set_top = set(vals_top)
            set_bot = set(vals_bot)
            # Jaccard-like distance
            intersection = set_top.intersection(set_bot)
            union = set_top.union(set_bot)
            return 1.0 - (len(intersection) / len(union) if union else 0)

    # Calculate scores for all inputs
    scores = []
    for i in input_indices:
        scores.append((get_dist_diff(i), header[i]))
    
    # Sort by score ascending (lowest difference = least impact)
    scores.sort(key=lambda x: x[0])
    
    # Return the bottom 5 columns that have the least impact
    return [s[1] for s in scores[:5]]
