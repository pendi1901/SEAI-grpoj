import csv

def spot(header, rows):
    """
    Identifies rows that are 'unlike anyone we have employed'.
    We use a distance-based outlier detection: if a row is significantly 
    far from the nearest neighbors in the training set, the estimate 
    cannot be trusted.
    """
    # Identify numeric input columns
    goal_indices = [i for i, col in enumerate(header) if col.endswith('+') or col.endswith('-')]
    input_indices = [i for i, col in enumerate(header) if i not in goal_indices and isinstance(rows[0][i], (int, float))]
    
    # Normalize numeric inputs for distance calculation
    mins = [min(row[i] for row in rows) for i in input_indices]
    maxs = [max(row[i] for row in rows) for i in input_indices]
    ranges = [maxs[i] - mins[i] if maxs[i] != mins[i] else 1 for i in range(len(input_indices))]
    
    def get_dist(r1, r2):
        d = 0
        for i, idx in enumerate(input_indices):
            d += ((r1[idx] - r2[idx]) / ranges[i]) ** 2
        return d ** 0.5

    # For each row, find the distance to its nearest neighbor
    # If the distance is very large, the row is an outlier (unlike anyone else)
    outlier_scores = []
    for i, row in enumerate(rows):
        distances = []
        for j, other in enumerate(rows):
            if i == j: continue
            distances.append(get_dist(row, other))
        
        # The score is the distance to the closest neighbor
        # High distance = isolated = untrustworthy estimate
        min_dist = min(distances)
        outlier_scores.append((min_dist, i))
        
    # Sort by distance descending (most isolated/suspicious first)
    outlier_scores.sort(key=lambda x: x[0], reverse=True)
    
    return [s[1] for s in outlier_scores]

if __name__ == "__main__":
    filename = 'WA_Fn-UseC_-HR-Employee-Attrition.n1000.train.csv'
    rows = []
    header = None
    
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = []
            for i, val in enumerate(row):
                if header[i][0].isupper():
                    try:
                        processed_row.append(float(val))
                    except ValueError:
                        processed_row.append(val)
                else:
                    processed_row.append(val)
            rows.append(processed_row)
            
    suspects = spot(header, rows)
    
    print("Outlier Detection Analysis:")
    print("The following rows represent employees who are statistically unlike others in our records.")
    print("Estimates for these individuals are likely to be unreliable due to lack of similar historical data:")
    for i in suspects[:5]:
        print(f"- Row {i}: High distance from existing employee profiles.")
    
    exit(0)
