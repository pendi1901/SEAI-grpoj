import csv

def guess(header, train, test):
    """
    Estimates the first goal column ('YearsAtCompany+') for test rows.
    Uses a K-Nearest Neighbors approach based on numeric input columns.
    """
    # Identify goal column (first one ending in + or -)
    goal_idx = -1
    for i, col in enumerate(header):
        if col.endswith('+') or col.endswith('-'):
            goal_idx = i
            break
    
    # Identify numeric input columns (starting with uppercase, excluding the goal)
    # We must ensure we only use columns that are actually numeric (floats)
    input_indices = [
        i for i, col in enumerate(header) 
        if col[0].isupper() and i != goal_idx and isinstance(train[0][i], (int, float))
    ]
    
    # Normalize training data for distance calculation
    mins = [min(row[i] for row in train) for i in input_indices]
    maxs = [max(row[i] for row in train) for i in input_indices]
    ranges = [maxs[i] - mins[i] if maxs[i] != mins[i] else 1 for i in range(len(input_indices))]
    
    def get_dist(r1, r2):
        d = 0
        for i, idx in enumerate(input_indices):
            d += ((r1[idx] - r2[idx]) / ranges[i]) ** 2
        return d ** 0.5

    predictions = []
    for t_row in test:
        # Find 5 nearest neighbors in training set
        distances = []
        for train_row in train:
            distances.append((get_dist(t_row, train_row), train_row[goal_idx]))
        
        distances.sort(key=lambda x: x[0])
        
        # Average the goal value of the 5 nearest neighbors
        k = 5
        avg = sum(d[1] for d in distances[:k]) / k
        predictions.append(float(avg))
        
    return predictions

import csv

def certify(xs, ys):
    """
    Performs a simple two-sample t-test (Welch's) to determine if two 
    distributions are significantly different. Returns True if they are 
    statistically similar (p > 0.05), False if they differ.
    """
    n1, n2 = len(xs), len(ys)
    if n1 < 2 or n2 < 2:
        return True
    
    m1, m2 = sum(xs) / n1, sum(ys) / n2
    v1 = sum((x - m1) ** 2 for x in xs) / (n1 - 1)
    v2 = sum((y - m2) ** 2 for y in ys) / (n2 - 1)
    
    # Standard error
    se = ((v1 / n1) + (v2 / n2)) ** 0.5
    if se == 0:
        return m1 == m2
    
    t_stat = abs(m1 - m2) / se
    # For large samples, t-distribution approaches normal. 
    # Critical value for 95% confidence is approx 1.96.
    return t_stat < 1.96

import csv

def flow(header, row, seen):
    """
    Decides if the current employee needs a 'what would keep you here' conversation.
    We target employees who are high-performers or have high potential but are 
    at risk of leaving (e.g., low JobSatisfaction or high YearsSinceLastPromotion).
    
    This is an online selection problem (Secretary Problem variant).
    We use a threshold strategy: observe the first 20% of employees to establish 
    a baseline for 'YearsAtCompany+' and 'JobSatisfaction', then select the 
    first employee who exceeds these benchmarks.
    """
    # Identify indices
    try:
        sat_idx = header.index('JobSatisfaction')
        years_idx = header.index('YearsAtCompany+')
        promo_idx = header.index('YearsSinceLastPromotion')
    except ValueError:
        return False

    # Baseline phase: 20% of 800 is 160
    n = len(seen) + 1
    if n < 160:
        return False
    
    # Calculate thresholds from seen data
    avg_sat = sum(r[sat_idx] for r in seen) / len(seen)
    avg_years = sum(r[years_idx] for r in seen) / len(seen)
    
    # Logic: If current employee has lower satisfaction than average 
    # but has been here longer than average (high value/risk), 
    # or has been waiting for a promotion longer than average.
    current_sat = row[sat_idx]
    current_years = row[years_idx]
    current_promo = row[promo_idx]
    
    if current_sat <= avg_sat and (current_years >= avg_years or current_promo > 2):
        return True
        
    return False

import csv

def blame(header, rows):
    """
    Identifies input columns that correlate with short stays ('YearsAtCompany+').
    We define 'worst rows' as the bottom 20% of employees by tenure.
    We then compare the average of input features in these 'worst' rows 
    against the global average to see which features deviate most.
    """
    # Identify goal and input columns
    years_idx = header.index('YearsAtCompany+')
    input_indices = [
        i for i, col in enumerate(header) 
        if not (col.endswith('+') or col.endswith('-'))
    ]
    
    # Sort rows by tenure to find the bottom 20%
    sorted_rows = sorted(rows, key=lambda r: r[years_idx])
    limit = max(1, int(len(rows) * 0.2))
    worst_rows = sorted_rows[:limit]
    
    # Calculate global averages for numeric inputs and modes for symbolic
    def get_stats(data_rows):
        stats = {}
        for i in input_indices:
            col_data = [r[i] for r in data_rows]
            if isinstance(col_data[0], (int, float)):
                stats[i] = sum(col_data) / len(col_data)
            else:
                # Mode for symbolic
                stats[i] = max(set(col_data), key=col_data.count)
        return stats

    global_stats = get_stats(rows)
    worst_stats = get_stats(worst_rows)
    
    # Identify features that differ most (for numeric) or differ at all (for symbolic)
    # We look for features where the 'worst' group deviates from the norm
    diffs = []
    for i in input_indices:
        col_name = header[i]
        if isinstance(global_stats[i], (int, float)):
            # Normalize difference
            val_diff = abs(worst_stats[i] - global_stats[i])
            diffs.append((val_diff, col_name))
        else:
            # Symbolic: if the mode is different, it's a strong indicator
            if worst_stats[i] != global_stats[i]:
                diffs.append((1.0, col_name))
    
    # Return top 3 columns that explain the short stays
    diffs.sort(key=lambda x: x[0], reverse=True)
    return [d[1] for d in diffs[:3]]

import csv

def justify(header, rows, row):
    """
    Estimates the performance of a given row against the dataset and returns
    its percentile rank and a justification.
    """
    # Identify goal columns
    goal_indices = [i for i, col in enumerate(header) if col.endswith('+') or col.endswith('-')]
    input_indices = [i for i, col in enumerate(header) if i not in goal_indices]

    # Simple KNN estimation for the missing goal values in 'row'
    def estimate_goal(target_row, goal_idx):
        # Use numeric inputs for distance
        numeric_inputs = [i for i in input_indices if isinstance(rows[0][i], (int, float))]
        
        def dist(r1, r2):
            d = 0
            for i in numeric_inputs:
                d += (r1[i] - r2[i])**2
            return d**0.5

        # Find 5 nearest neighbors
        neighbors = sorted(rows, key=lambda r: dist(target_row, r))[:5]
        return sum(n[goal_idx] for n in neighbors) / 5

    # Fill in the '?' values
    estimated_row = list(row)
    for idx in goal_indices:
        estimated_row[idx] = estimate_goal(row, idx)

    # Calculate how many rows this row is 'better' than
    better_count = 0
    for r in rows:
        is_better = True
        for idx in goal_indices:
            val_r = r[idx]
            val_row = estimated_row[idx]
            # If goal ends in +, higher is better. If -, lower is better.
            if header[idx].endswith('+'):
                if not (val_row >= val_r):
                    is_better = False
            else:
                if not (val_row <= val_r):
                    is_better = False
        if is_better:
            better_count += 1
            
    p = better_count / len(rows)
    reason = f"This employee's arrangement ranks in the top {int((1-p)*100)}% of the company based on tenure and retention metrics."
    return (p, reason)

import csv
import random

def choose(header, rows, label):
    """
    Finds the best working arrangement by sampling 50 rows.
    We use a simple strategy: sample 50 random rows, label them,
    and pick the one with the best performance on the goal columns.
    """
    goal_indices = [i for i, col in enumerate(header) if col.endswith('+') or col.endswith('-')]
    
    # We can label at most 50 rows.
    n_samples = min(50, len(rows))
    indices = random.sample(range(len(rows)), n_samples)
    
    best_idx = -1
    best_score = None
    
    for i in indices:
        # Get true values for the goals
        goals = label(i)
        
        # Calculate a normalized score for this row
        # For '+', higher is better. For '-', lower is better.
        # We normalize by the range of the observed samples to make them comparable.
        current_score = 0
        for j, goal_idx in enumerate(goal_indices):
            val = goals[j]
            # Simple heuristic: if +, maximize; if -, minimize.
            # We use a basic sign-based weight.
            if header[goal_idx].endswith('+'):
                current_score += val
            else:
                current_score -= val
        
        if best_score is None or current_score > best_score:
            best_score = current_score
            best_idx = i
            
    return best_idx

import csv

def fix(header, row, levers):
    """
    Recommends changes to a 'bad' row (short tenure) to improve retention.
    We prioritize increasing pay, reducing overtime, and increasing training.
    """
    recommendations = {}
    
    # Map indices for easier access
    idx_map = {name: i for i, name in enumerate(header)}
    
    # 1. If Overtime is 'Yes', change to 'No'
    if 'overTime' in levers and row[idx_map['overTime']] == 'Yes':
        recommendations['overTime'] = 'No'
    
    # 2. If PercentSalaryHike is low, increase it
    if 'PercentSalaryHike' in levers:
        low, high = levers['PercentSalaryHike']
        # If current is below the midpoint, boost it
        if row[idx_map['PercentSalaryHike']] < (low + high) / 2:
            recommendations['PercentSalaryHike'] = high
            
    # 3. If TrainingTimesLastYear is low, increase it
    if 'TrainingTimesLastYear' in levers:
        low, high = levers['TrainingTimesLastYear']
        if row[idx_map['TrainingTimesLastYear']] < high:
            recommendations['TrainingTimesLastYear'] = high
            
    # 4. If StockOptionLevel is low, increase it
    if 'StockOptionLevel' in levers:
        low, high = levers['StockOptionLevel']
        if row[idx_map['StockOptionLevel']] < high:
            recommendations['StockOptionLevel'] = high

    # Ensure at least one change is made if possible
    if not recommendations and levers:
        first_key = list(levers.keys())[0]
        val = levers[first_key]
        recommendations[first_key] = val[1] if isinstance(val, tuple) else val[0]
        
    return recommendations

def race(header, rows, label):
    """
    Compares three strategies for finding the best employee arrangement.
    Budget: 50 labels total.
    """
    goal_indices = [i for i, col in enumerate(header) if col.endswith('+') or col.endswith('-')]
    
    def get_score(row_idx):
        goals = label(row_idx)
        score = 0
        for j, goal_idx in enumerate(goal_indices):
            val = goals[j]
            # Normalize: + is maximize, - is minimize
            if header[goal_idx].endswith('+'):
                score += val
            else:
                score -= val
        return score

    # Strategies
    def strategy_random(budget):
        indices = random.sample(range(len(rows)), budget)
        return {i: get_score(i) for i in indices}

    def strategy_tenure(budget):
        # Focus on high tenure (YearsAtCompany+)
        idx = header.index('YearsAtCompany+')
        sorted_indices = sorted(range(len(rows)), key=lambda i: rows[i][idx], reverse=True)
        indices = sorted_indices[:budget]
        return {i: get_score(i) for i in indices}

    def strategy_satisfaction(budget):
        # Focus on high JobSatisfaction
        idx = header.index('JobSatisfaction')
        sorted_indices = sorted(range(len(rows)), key=lambda i: rows[i][idx], reverse=True)
        indices = sorted_indices[:budget]
        return {i: get_score(i) for i in indices}

    # Budget allocation: 16 labels per strategy (48 total)
    budget_per = 16
    results = {}
    results["random"] = strategy_random(budget_per)
    results["tenure"] = strategy_tenure(budget_per)
    results["satisfaction"] = strategy_satisfaction(budget_per)

    best_overall = None
    best_row = -1
    best_strategy = ""

    for name, data in results.items():
        for row_idx, score in data.items():
            if best_overall is None or score > best_overall:
                best_overall = score
                best_row = row_idx
                best_strategy = name

    return {
        "winner": best_strategy,
        "row": best_row,
        "tried": ["random", "tenure", "satisfaction"]
    }

import csv

def remember(header, rows):
    """
    Finds the 5% of employees most similar to the 'average' employee 
    (or a representative profile) and returns their indices.
    Similarity is based on Euclidean distance of numeric input features.
    """
    # Identify goal columns and input columns
    goal_indices = [i for i, col in enumerate(header) if col.endswith('+') or col.endswith('-')]
    input_indices = [i for i, col in enumerate(header) if i not in goal_indices and isinstance(rows[0][i], (int, float))]
    
    # Calculate the 'average' employee profile to compare against
    avg_profile = []
    for i in input_indices:
        col_vals = [r[i] for r in rows]
        avg_profile.append(sum(col_vals) / len(col_vals))
    
    # Calculate distance of each row to the average profile
    def get_dist(row):
        d = 0
        for j, idx in enumerate(input_indices):
            d += (row[idx] - avg_profile[j]) ** 2
        return d ** 0.5
    
    # Rank rows by similarity (smallest distance is most similar)
    indexed_rows = [(get_dist(row), i) for i, row in enumerate(rows)]
    indexed_rows.sort(key=lambda x: x[0])
    
    # Return top 5%
    limit = max(1, int(len(rows) * 0.05))
    return [item[1] for item in indexed_rows[:limit]]

def guess(header, train, test):

    """

    # Identify numeric input columns (starting with uppercase, excluding the goal)
    input_indices = [
        i for i, col in enumerate(header) 
        if col[0].isupper() and i != goal_idx and isinstance(train[0][i], (int, float))
    ]
    
    # Normalize numeric inputs
    mins = [min(row[i] for row in train) for i in input_indices]
    maxs = [max(row[i] for row in train) for i in input_indices]
    ranges = [maxs[i] - mins[i] if maxs[i] != mins[i] else 1 for i in range(len(input_indices))]
    
    def get_dist(r1, r2):
        d = 0
        # Only use numeric columns for distance calculation to avoid TypeError
        for i, idx in enumerate(input_indices):
            d += ((r1[idx] - r2[idx]) / ranges[i]) ** 2
        return d ** 0.5

        distances = sorted([(get_dist(t_row, tr), tr[goal_idx]) for tr in train], key=lambda x: x[0])
        avg = sum(d[1] for d in distances[:5]) / 5
        predictions.append(float(avg))
        
    return predictions

import csv

def certify(xs, ys):
    """
    Performs a simple two-sample t-test (Welch's) to determine if two 
    distributions are significantly different. Returns True if they are 
    statistically similar (p > 0.05), False if they differ.
    """
    n1, n2 = len(xs), len(ys)
    if n1 < 2 or n2 < 2:
        return True
    
    m1, m2 = sum(xs) / n1, sum(ys) / n2
    v1 = sum((x - m1) ** 2 for x in xs) / (n1 - 1)
    v2 = sum((y - m2) ** 2 for y in ys) / (n2 - 1)
    
    se = ((v1 / n1) + (v2 / n2)) ** 0.5
    if se == 0:
        return m1 == m2
    
    t_stat = abs(m1 - m2) / se
    # Critical value for 95% confidence (two-tailed) is approx 1.96
    return t_stat < 1.96

import csv

def flow(header, row, seen):
    """
    Detects if the current team composition has drifted from the 'normal' 
    baseline established by the 'seen' rows.
    
    We monitor the distribution of 'department' and the average 'YearsAtCompany+'.
    If the current row significantly deviates from the established baseline 
    (e.g., a sudden influx of a specific department or a shift in average tenure),
    we flag it as a change in team dynamics.
    """
    if len(seen) < 30:
        return False

    dept_idx = header.index('department')
    tenure_idx = header.index('YearsAtCompany+')
    
    # Baseline stats
    seen_depts = [r[dept_idx] for r in seen]
    seen_tenures = [r[tenure_idx] for r in seen]
    
    avg_tenure = sum(seen_tenures) / len(seen_tenures)
    
    # Current row metrics
    current_dept = row[dept_idx]
    current_tenure = row[tenure_idx]
    
    # Detect drift:
    # 1. If the department frequency is very low (rare arrival)
    dept_counts = {d: seen_depts.count(d) for d in set(seen_depts)}
    dept_freq = dept_counts.get(current_dept, 0) / len(seen)
    
    # 2. If the tenure is an outlier (more than 2x the average)
    tenure_drift = abs(current_tenure - avg_tenure) > (avg_tenure * 2)
    
    # Flag if the team structure is changing (rare department or tenure spike)
    if dept_freq < 0.05 or tenure_drift:
        return True
        
    return False
