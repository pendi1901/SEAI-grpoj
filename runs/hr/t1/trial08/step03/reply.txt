import csv

def certify(xs, ys):
    """
    Performs a simple two-sample t-test (Welch's) to determine if two 
    distributions are significantly different. Returns True if they are 
    statistically similar (p > 0.05), False if they differ.
    """
    n1, n2 = len(xs), len(ys)
    if n1 < 2 or n2 < 2:
        return True
    
    m1, m2 = sum(xs) / n1, sum(ys) / n2
    v1 = sum((x - m1) ** 2 for x in xs) / (n1 - 1)
    v2 = sum((y - m2) ** 2 for y in ys) / (n2 - 1)
    
    # Standard error
    se = ((v1 / n1) + (v2 / n2)) ** 0.5
    if se == 0:
        return m1 == m2
    
    t_stat = abs(m1 - m2) / se
    # For large samples, t-distribution approaches normal. 
    # Critical value for 95% confidence is approx 1.96.
    return t_stat < 1.96

def analyze_overtime_impact(header, rows):
    """
    Compares 'YearsInCurrentRole+' for employees who work overtime vs those who don't.
    """
    ot_idx = header.index('overTime')
    role_idx = header.index('YearsInCurrentRole+')
    
    with_ot = [row[role_idx] for row in rows if row[ot_idx] == 'Yes']
    without_ot = [row[role_idx] for row in rows if row[ot_idx] == 'No']
    
    avg_ot = sum(with_ot) / len(with_ot)
    avg_no_ot = sum(without_ot) / len(without_ot)
    
    is_same = certify(with_ot, without_ot)
    
    print(f"Average Years in Current Role (Overtime): {avg_ot:.2f}")
    print(f"Average Years in Current Role (No Overtime): {avg_no_ot:.2f}")
    
    if not is_same:
        print("Conclusion: The difference is statistically significant. Overtime correlates with shorter tenure in role.")
    else:
        print("Conclusion: The observed difference is likely just noise; no significant impact found.")

if __name__ == "__main__":
    filename = 'WA_Fn-UseC_-HR-Employee-Attrition.n1000.train.csv'
    rows = []
    header = None
    
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = []
            for i, val in enumerate(row):
                if header[i][0].isupper():
                    try:
                        processed_row.append(float(val))
                    except ValueError:
                        processed_row.append(val)
                else:
                    processed_row.append(val)
            rows.append(processed_row)
            
    analyze_overtime_impact(header, rows)
