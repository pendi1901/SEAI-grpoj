import csv
import random

def race(header, rows, label):
    """
    Compares three strategies for finding the best row (maximizing '+' goals).
    Budget: 50 labels total.
    """
    # Identify goal column index
    goal_idx = -1
    for i, name in enumerate(header):
        if name.endswith('+') or name.endswith('-'):
            goal_idx = i
            break
    
    is_maximize = header[goal_idx].endswith('+')
    
    def get_score(row):
        return row[goal_idx]

    # Budget management
    tried = ["random", "seniority", "wealth"]
    results = {name: {"best_idx": -1, "best_val": -float('inf') if is_maximize else float('inf')} for name in tried}
    
    # Strategy 1: Random (16 labels)
    rand_indices = random.sample(range(len(rows)), 16)
    for i in rand_indices:
        val = get_score(label(i))
        if (is_maximize and val > results["random"]["best_val"]) or (not is_maximize and val < results["random"]["best_val"]):
            results["random"]["best_val"] = val
            results["random"]["best_idx"] = i
            
    # Strategy 2: Seniority (17 labels) - Look at TotalWorkingYears
    twy_idx = header.index('TotalWorkingYears')
    sorted_by_twy = sorted(range(len(rows)), key=lambda i: rows[i][twy_idx], reverse=True)
    for i in sorted_by_twy[:17]:
        val = get_score(label(i))
        if (is_maximize and val > results["seniority"]["best_val"]) or (not is_maximize and val < results["seniority"]["best_val"]):
            results["seniority"]["best_val"] = val
            results["seniority"]["best_idx"] = i
            
    # Strategy 3: Wealth (17 labels) - Look at MonthlyIncome
    inc_idx = header.index('MonthlyIncome')
    sorted_by_inc = sorted(range(len(rows)), key=lambda i: rows[i][inc_idx], reverse=True)
    for i in sorted_by_inc[:17]:
        val = get_score(label(i))
        if (is_maximize and val > results["wealth"]["best_val"]) or (not is_maximize and val < results["wealth"]["best_val"]):
            results["wealth"]["best_val"] = val
            results["wealth"]["best_idx"] = i
            
    # Determine winner
    if is_maximize:
        winner = max(tried, key=lambda name: results[name]["best_val"])
    else:
        winner = min(tried, key=lambda name: results[name]["best_val"])
    
    return {
        "winner": winner,
        "row": results[winner]["best_idx"],
        "tried": tried
    }

if __name__ == "__main__":
    filename = 'WA_Fn-UseC_-HR-Employee-Attrition.n1000.train.csv'
    rows = []
    header = []
    
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = []
            for i, val in enumerate(row):
                if header[i][0].isupper():
                    processed_row.append(float(val))
                else:
                    processed_row.append(val)
            rows.append(processed_row)
            
    goal_idx = next(i for i, name in enumerate(header) if name.endswith('+') or name.endswith('-'))
    
    # Mask rows for the harness
    masked_rows = []
    for row in rows:
        new_row = list(row)
        new_row[goal_idx] = "?"
        masked_rows.append(new_row)
        
    def label(i):
        return rows[i]
        
    outcome = race(header, masked_rows, label)
    
    print(f"Race Results:")
    print(f"Strategies tried: {', '.join(outcome['tried'])}")
    print(f"The winning strategy is: {outcome['winner']}")
    print(f"Best row found at index: {outcome['row']}")
    print(f"Goal value at that row: {rows[outcome['row']][goal_idx]}")
