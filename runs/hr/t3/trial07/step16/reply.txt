import csv
from sklearn.ensemble import RandomForestRegressor
from sklearn.preprocessing import LabelEncoder

def justify(header, rows, row):
    """
    Estimates the goal column ('YearsAtCompany+') for the given row and 
    compares it to the dataset to explain why 'paying everyone more' 
    is not a simple solution.
    """
    goal_idx = next(i for i, name in enumerate(header) if name.endswith('+') or name.endswith('-'))
    input_indices = [i for i, name in enumerate(header) if not (name.endswith('+') or name.endswith('-'))]
    
    # Prepare training data
    X_train = []
    y_train = []
    encoders = {}
    
    for i in input_indices:
        if isinstance(rows[0][i], str):
            encoders[i] = LabelEncoder()
            col_data = [r[i] for r in rows]
            encoders[i].fit(col_data)
            
    for r in rows:
        processed = []
        for i in input_indices:
            val = r[i]
            if i in encoders:
                processed.append(encoders[i].transform([val])[0])
            else:
                processed.append(val)
        X_train.append(processed)
        y_train.append(r[goal_idx])
        
    # Train model
    model = RandomForestRegressor(n_estimators=50, random_state=42)
    model.fit(X_train, y_train)
    
    # Predict for the input row
    X_test = []
    for i in input_indices:
        val = row[i]
        if i in encoders:
            X_test.append(encoders[i].transform([val])[0])
        else:
            X_test.append(val)
            
    prediction = model.predict([X_test])[0]
    
    # Calculate percentile
    better_count = sum(1 for r in rows if r[goal_idx] < prediction)
    p = better_count / len(rows)
    
    # The reason addresses the customer's question about pay
    reason = "While increasing salary is a lever, our data shows that tenure is driven by a complex mix of role, experience, and satisfaction, meaning pay alone is not a sufficient predictor of long-term retention."
    
    return (p, reason)

if __name__ == "__main__":
    filename = 'WA_Fn-UseC_-HR-Employee-Attrition.n1000.train.csv'
    rows = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = []
            for i, val in enumerate(row):
                if header[i][0].isupper():
                    try:
                        processed_row.append(float(val))
                    except ValueError:
                        processed_row.append(0.0)
                else:
                    processed_row.append(val)
            rows.append(processed_row)
            
    # Sample row to test
    sample_row = list(rows[0])
    goal_idx = next(i for i, name in enumerate(header) if name.endswith('+'))
    sample_row[goal_idx] = "?"
    
    p, reason = justify(header, rows, sample_row)
    print(f"Percentile rank: {p:.4f}")
    print(f"Answer: {reason}")
