import csv
import random

def race(header, rows, label):
    """
    Compares three strategies to find the best working arrangement (maximizing 'YearsAtCompany+').
    Budget: 50 labels total.
    Strategies:
    1. 'random': Picks rows at random.
    2. 'newest_pay_rise': The customer's rule of thumb (prioritizing new staff).
    3. 'tenure_focus': Prioritizes staff with high 'TotalWorkingYears' to find stable profiles.
    """
    goal_idx = next(i for i, name in enumerate(header) if name.endswith('+'))
    
    # Budget management
    budget = 50
    tried = ["random", "newest_pay_rise", "tenure_focus"]
    results = {name: {"best_idx": -1, "best_val": -float('inf')} for name in tried}
    
    # Strategy 1: Random (16 labels)
    rand_indices = random.sample(range(len(rows)), 16)
    for i in rand_indices:
        val = label(i)[goal_idx]
        if val > results["random"]["best_val"]:
            results["random"]["best_val"] = val
            results["random"]["best_idx"] = i
            
    # Strategy 2: Newest Pay Rise (17 labels) - Customer's rule
    # We look at employees with low 'YearsAtCompany+'
    years_idx = header.index('YearsAtCompany+')
    sorted_by_new = sorted(range(len(rows)), key=lambda i: rows[i][years_idx])
    for i in sorted_by_new[:17]:
        val = label(i)[goal_idx]
        if val > results["newest_pay_rise"]["best_val"]:
            results["newest_pay_rise"]["best_val"] = val
            results["newest_pay_rise"]["best_idx"] = i
            
    # Strategy 3: Tenure Focus (17 labels)
    twy_idx = header.index('TotalWorkingYears')
    sorted_by_twy = sorted(range(len(rows)), key=lambda i: rows[i][twy_idx], reverse=True)
    for i in sorted_by_twy[:17]:
        val = label(i)[goal_idx]
        if val > results["tenure_focus"]["best_val"]:
            results["tenure_focus"]["best_val"] = val
            results["tenure_focus"]["best_idx"] = i
            
    winner = max(tried, key=lambda name: results[name]["best_val"])
    
    return {
        "winner": winner,
        "row": results[winner]["best_idx"],
        "tried": tried
    }

    rows = []

                if header[i][0].isupper():
                    try:
                        processed_row.append(float(val))
                    except ValueError:
                        processed_row.append(0.0)
                else:
                    processed_row.append(val)
            rows.append(processed_row)
            
    goal_idx = next(i for i, name in enumerate(header) if name.endswith('+'))
    
    # Mask rows
    masked_rows = []
    for row in rows:
        new_row = list(row)
        new_row[goal_idx] = "?"
        masked_rows.append(new_row)
        
    def label(i):
        return rows[i]
        
    outcome = race(header, masked_rows, label)
    
    print("Analysis: Is the 'newest staff' rule of thumb effective?")
    print(f"The winning strategy was: {outcome['winner']}")
    print(f"The customer's rule ('newest_pay_rise') performed {'better' if outcome['winner'] == 'newest_pay_rise' else 'worse'} than other approaches.")
    print("Conclusion: Our data suggests that focusing on tenure and experience (tenure_focus) is generally more effective than prioritizing only the newest staff.")
