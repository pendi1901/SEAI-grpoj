            goal_idx = i

import csv
from scipy import stats

def certify(xs, ys):
    """
    Performs a Welch's t-test to determine if two groups (xs and ys)
    are statistically different. Returns True if they are not 
    significantly different (p > 0.05), False otherwise.
    """
    if len(xs) < 2 or len(ys) < 2:
        return True
    
    t_stat, p_value = stats.ttest_ind(xs, ys, equal_var=False)
    
    # If p > 0.05, we fail to reject the null hypothesis (no real difference)
    return p_value > 0.05

import csv

def flow(header, row, seen):
    """
    Decides if the current employee needs a 'what would keep you here' conversation.
    We target employees who are high-performers or long-tenured but might be at risk.
    Using a Secretary Problem-inspired approach: we observe the first 20% of 
    employees to establish a baseline for 'YearsAtCompany+' and then pick the 
    first one that exceeds that baseline.
    """
    # Identify indices
    try:
        years_idx = header.index('YearsAtCompany+')
    except ValueError:
        return False

    # If this is the first row, we can't compare
    if not seen:
        return False

    # Calculate baseline from seen data (first 20% or all if fewer)
    n = len(seen)
    limit = max(1, 800 // 5)
    
    # If we are in the observation phase, don't pick
    if n < limit:
        return False
    
    # Calculate the max 'YearsAtCompany+' seen so far as our threshold
    threshold = max(r[years_idx] for r in seen)
    
    # If current row exceeds the threshold, flag for conversation
    if row[years_idx] > threshold:
        return True
        
    return False

def blame(header, rows):
    """
    Identifies input columns that best explain short stays (low 'YearsAtCompany+').
    We define 'worst rows' as those in the bottom 20% of 'YearsAtCompany+'.
    We then train a Random Forest to predict 'YearsAtCompany+' and extract
    feature importances to see which inputs correlate with tenure.
    """
    goal_name = 'YearsAtCompany+'
    goal_idx = header.index(goal_name)
    
    # Identify input columns (exclude goals)
    input_indices = [i for i, name in enumerate(header) 
                     if not (name.endswith('+') or name.endswith('-'))]
    
    # Prepare data
    X = []
    y = []
    encoders = {}
    
    for i in input_indices:
        if isinstance(rows[0][i], str):
            encoders[i] = LabelEncoder()
            col_data = [row[i] for row in rows]
            encoders[i].fit(col_data)
            
    for row in rows:
        processed = []
        for i in input_indices:
            val = row[i]
            if i in encoders:
                processed.append(encoders[i].transform([val])[0])
            else:
                processed.append(val)
        X.append(processed)
        y.append(row[goal_idx])
        
    # Train a regressor to find feature importance
    model = RandomForestRegressor(n_estimators=50, random_state=42)
    model.fit(X, y)
    
    # Get feature importances
    importances = model.feature_importances_
    
    # Map importances back to column names
    feature_scores = []
    for i, idx in enumerate(input_indices):
        feature_scores.append((header[idx], importances[i]))
        
    # Sort by importance
    feature_scores.sort(key=lambda x: x[1], reverse=True)
    
    # Return top 3 columns that explain the variance in tenure
    return [name for name, score in feature_scores[:3]]

def choose(header, rows, label):
    """
    Finds the working arrangement that keeps people longest by sampling 50 rows.
    The label(i) function returns the full row (including goals) for index i.
    """
    # Identify goal column index
    goal_idx = -1
    for i, name in enumerate(header):
        if name.endswith('+') or name.endswith('-'):
            goal_idx = i
            break
            
    # Select 50 random indices to sample
    num_samples = min(50, len(rows))
    indices = random.sample(range(len(rows)), num_samples)
    
    best_val = -float('inf')
    best_idx = -1
    
    # Determine if we are maximizing (+) or minimizing (-)
    is_maximize = header[goal_idx].endswith('+')
    
    for i in indices:
        # Query the true row data
        full_row = label(i)
        val = full_row[goal_idx]
        
        # Update best based on goal type
        if is_maximize:
            if val > best_val:
                best_val = val
                best_idx = i
        else:
            if val < best_val or best_idx == -1:
                best_val = val
                best_idx = i
            
    return best_idx

import csv

def fix(header, row, levers):
    """
    Recommends changes for a 'bad' row (low tenure) by adjusting levers
    to move the employee toward a more stable profile.
    """
    recommendations = {}
    
    # We aim to improve 'YearsAtCompany+' (a + goal).
    # Strategy: Increase numeric levers (pay, training) and optimize symbolic ones.
    for col, allowed in levers.items():
        idx = header.index(col)
        current_val = row[idx]
        
        if isinstance(allowed, tuple):  # Numeric lever (low, high)
            low, high = allowed
            # If current is low, push towards high
            if current_val < high:
                recommendations[col] = min(high, current_val + (high - low) * 0.2)
        
        elif isinstance(allowed, list):  # Symbolic lever
            # Simple heuristic: pick a value that isn't the current one
            options = [val for val in allowed if val != current_val]
            if options:
                recommendations[col] = random.choice(options)
                
    # Ensure at least one change is made
    if not recommendations:
        col = list(levers.keys())[0]
        if isinstance(levers[col], list):
            recommendations[col] = levers[col][0]
        else:
            recommendations[col] = levers[col][1]
            
    return recommendations

import csv

def spot(header, rows):
    """
    Identifies rows that look suspicious based on internal consistency.
    We look for rows where numeric values are extreme outliers compared to 
    the rest of the dataset (e.g., Age > 100, YearsAtCompany > TotalWorkingYears).
    """
    suspicious = []
    
    # Indices for sanity checks
    try:
        age_idx = header.index('Age')
        years_comp_idx = header.index('YearsAtCompany+')
        total_years_idx = header.index('TotalWorkingYears')
        years_role_idx = header.index('YearsInCurrentRole+')
    except ValueError:
        return []

    for i, row in enumerate(rows):
        score = 0
        
        # Rule 1: Age sanity check (e.g., Age < 18 or > 90)
        if row[age_idx] < 18 or row[age_idx] > 90:
            score += 5
            
        # Rule 2: YearsAtCompany cannot exceed TotalWorkingYears
        if row[years_comp_idx] > row[total_years_idx]:
            score += 10
            
        # Rule 3: YearsInCurrentRole cannot exceed YearsAtCompany
        if row[years_role_idx] > row[years_comp_idx]:
            score += 8
            
        # Rule 4: StandardHours is usually 80 in this dataset
        std_hours_idx = header.index('StandardHours')
        if row[std_hours_idx] != 80:
            score += 3
            
        if score > 0:
            suspicious.append((i, score))
            
    # Sort by score descending (most suspicious first)
    suspicious.sort(key=lambda x: x[1], reverse=True)
    
    return [item[0] for item in suspicious]

import csv
import random

def race(header, rows, label):
    """
    Compares three strategies for finding the best row (maximizing '+' goals).
    Budget: 50 labels total.
    """
    # Identify goal column index
    goal_idx = -1
    for i, name in enumerate(header):
        if name.endswith('+') or name.endswith('-'):
            goal_idx = i
            break
    
    is_maximize = header[goal_idx].endswith('+')
    
    def get_score(row):
        return row[goal_idx]

    # Budget management
    tried = ["random", "seniority", "wealth"]
    results = {name: {"best_idx": -1, "best_val": -float('inf') if is_maximize else float('inf')} for name in tried}
    
    # Strategy 1: Random (16 labels)
    rand_indices = random.sample(range(len(rows)), 16)
    for i in rand_indices:
        val = get_score(label(i))
        if (is_maximize and val > results["random"]["best_val"]) or (not is_maximize and val < results["random"]["best_val"]):
            results["random"]["best_val"] = val
            results["random"]["best_idx"] = i
            
    # Strategy 2: Seniority (17 labels) - Look at TotalWorkingYears
    twy_idx = header.index('TotalWorkingYears')
    sorted_by_twy = sorted(range(len(rows)), key=lambda i: rows[i][twy_idx], reverse=True)
    for i in sorted_by_twy[:17]:
        val = get_score(label(i))
        if (is_maximize and val > results["seniority"]["best_val"]) or (not is_maximize and val < results["seniority"]["best_val"]):
            results["seniority"]["best_val"] = val
            results["seniority"]["best_idx"] = i
            
    # Strategy 3: Wealth (17 labels) - Look at MonthlyIncome
    inc_idx = header.index('MonthlyIncome')
    sorted_by_inc = sorted(range(len(rows)), key=lambda i: rows[i][inc_idx], reverse=True)
    for i in sorted_by_inc[:17]:
        val = get_score(label(i))
        if (is_maximize and val > results["wealth"]["best_val"]) or (not is_maximize and val < results["wealth"]["best_val"]):
            results["wealth"]["best_val"] = val
            results["wealth"]["best_idx"] = i
            
    # Determine winner
    if is_maximize:
        winner = max(tried, key=lambda name: results[name]["best_val"])
    else:
        winner = min(tried, key=lambda name: results[name]["best_val"])
    
    return {
        "winner": winner,
        "row": results[winner]["best_idx"],
        "tried": tried
    }

import csv
from math import sqrt

def remember(header, rows):
    """
    Finds the 5% of employees most similar to the first employee (index 0)
    based on Euclidean distance of normalized numeric features.
    """
    # Identify numeric columns for similarity calculation
    numeric_indices = [i for i, name in enumerate(header) if name[0].isupper()]
    
    # Normalize numeric columns
    normalized_rows = []
    for row in rows:
        norm_row = []
        for i in numeric_indices:
            # Simple min-max normalization based on column data
            col_vals = [r[i] for r in rows if isinstance(r[i], (int, float))]
            min_v, max_v = min(col_vals), max(col_vals)
            val = (row[i] - min_v) / (max_v - min_v + 1e-9)
            norm_row.append(val)
        normalized_rows.append(norm_row)
    
    # Calculate Euclidean distance from the first employee (index 0)
    target = normalized_rows[0]
    distances = []
    for i in range(1, len(rows)):
        dist = sqrt(sum((target[j] - normalized_rows[i][j])**2 for j in range(len(target))))
        distances.append((i, dist))
    
    # Sort by distance (ascending)
    distances.sort(key=lambda x: x[1])
    
    # Return top 5% (40 rows out of 800)
    limit = max(1, len(rows) // 20)
    return [d[0] for d in distances[:limit]]

def guess(header, train, test):
    """
    Estimates the first goal column ('YearsAtCompany+') for test rows.
    To answer the customer's question about cutting overtime in Sales,
    we train a model that includes 'department' and 'overTime' as features.
    """
    # Identify goal column (first column ending in '+' or '-')
    goal_idx = next(i for i, name in enumerate(header) if name.endswith('+') or name.endswith('-'))
    
    # Identify input columns
    input_indices = [i for i, name in enumerate(header) 
                     if not (name.endswith('+') or name.endswith('-'))]
    
    # Prepare data
    X_train = []
    y_train = []
    encoders = {}
    
    # Fit encoders on all data to handle categories
    for i in input_indices:
        if isinstance(train[0][i], str):
            encoders[i] = LabelEncoder()
            col_data = [row[i] for row in train] + [row[i] for row in test if row[i] != '?']
            encoders[i].fit(col_data)
            
    for row in train:
        processed = []
        for i in input_indices:
            val = row[i]
            if i in encoders:
                processed.append(encoders[i].transform([val])[0])
            else:
                processed.append(val)
        X_train.append(processed)
        y_train.append(row[goal_idx])
        
    # Train model
    model = RandomForestRegressor(n_estimators=100, random_state=42)
    model.fit(X_train, y_train)
    
    # Predict
    X_test = []
    for row in test:
        processed = []
        for i in input_indices:
            val = row[i]
            if i in encoders:
                processed.append(encoders[i].transform([val])[0])
            else:
                processed.append(val)
        X_test.append(processed)
        
    return model.predict(X_test).tolist()
