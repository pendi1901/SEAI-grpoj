    We calculate the 95th percentile of 'YearsAtCompany+' and return

    try:
        idx = header.index('YearsAtCompany+')
    except ValueError:
        return [0]

    # Extract all values for the goal column
    years = [row[idx] for row in rows]
    
    # Calculate threshold for "unusually long" (top 5%)
    sorted_years = sorted(years, reverse=True)
    limit = max(1, len(rows) // 20)
    threshold = sorted_years[limit - 1]
    
    # Find indices of rows meeting the criteria
    indices = [i for i, val in enumerate(years) if val >= threshold]
    
    # Ensure we return at most 5% of the rows (limit is already 5%)
    return indices[:limit]

import csv
from scipy import stats

def certify(xs, ys):
    """
    Performs a Welch's t-test to determine if two samples have significantly 
    different means. Returns True if the difference is not statistically 
    significant (p > 0.05), False otherwise.
    """
    if len(xs) < 2 or len(ys) < 2:
        return True
    
    t_stat, p_value = stats.ttest_ind(xs, ys, equal_var=False)
    
    # If p > 0.05, we fail to reject the null hypothesis (they are the same apart from noise)
    return p_value > 0.05

import csv

def flow(header, row, seen):
    """
    Decides if the current employee needs a 'what would keep you here' conversation.
    We target employees who are high-performers or have high tenure but might be 
    at risk of attrition. We use a simple threshold-based strategy: if the 
    employee's 'YearsAtCompany+' is in the top 10% of what we have seen so far, 
    or if they have high 'JobSatisfaction' (>= 3) but are 'Single' (often 
    higher turnover risk), we flag them.
    """
    # Identify indices
    try:
        idx_years = header.index('YearsAtCompany+')
        idx_sat = header.index('JobSatisfaction')
        idx_status = header.index('maritalStatus')
    except ValueError:
        return False

    current_years = row[idx_years]
    current_sat = row[idx_sat]
    current_status = row[idx_status]

    # Strategy:
    # 1. If they have been here longer than 90% of the people seen so far, 
    #    they are a valuable asset worth retaining.
    # 2. If they are 'Single' and have high satisfaction, they are high-value 
    #    targets for retention before they decide to move on.
    
    if not seen:
        return True
    
    seen_years = [r[idx_years] for r in seen]
    threshold = sorted(seen_years, reverse=True)[max(0, len(seen_years) // 10)]
    
    if current_years >= threshold:
        return True
    
    if current_status == 'Single' and current_sat >= 3:
        return True
        
    return False

import csv

def blame(header, rows):
    """
    Identifies input columns that correlate with short stays ('YearsAtCompany+').
    We define 'worst rows' as the bottom 20% of employees by tenure.
    We then compare the distribution of input features in these short-stay 
    employees against the rest of the population to find the most 
    discriminating factors.
    """
    goal_idx = header.index('YearsAtCompany+')
    
    # Sort rows by tenure to find the "worst" (shortest stay)
    sorted_rows = sorted(rows, key=lambda x: x[goal_idx])
    cutoff = max(1, len(rows) // 5)
    worst = sorted_rows[:cutoff]
    rest = sorted_rows[cutoff:]
    
    # Identify input columns (exclude goals)
    input_indices = [i for i, name in enumerate(header) 
                     if not (name.endswith('+') or name.endswith('-'))]
    
    scores = []
    for i in input_indices:
        col_name = header[i]
        # Calculate simple difference in means/proportions
        if isinstance(worst[0][i], float):
            w_val = sum(r[i] for r in worst) / len(worst)
            r_val = sum(r[i] for r in rest) / len(rest)
            diff = abs(w_val - r_val)
        else:
            # For symbolic, compare frequency of the most common value in 'worst'
            w_vals = [r[i] for r in worst]
            r_vals = [r[i] for r in rest]
            mode_w = max(set(w_vals), key=w_vals.count)
            diff = abs(w_vals.count(mode_w) / len(w_vals) - r_vals.count(mode_w) / len(r_vals))
        
        scores.append((diff, col_name))
    
    # Return top 3 columns that explain the difference
    scores.sort(key=lambda x: x[0], reverse=True)
    return [name for _, name in scores[:3]]

def justify(header, rows, row):
    """
    Estimates the performance of a given row relative to the population
    using a Random Forest Regressor to predict goal columns.
    Returns (percentile, reason).
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith('+') or name.endswith('-')]
    feature_indices = [i for i in range(len(header)) if i not in goal_indices]
    
    # Prepare training data
    encoders = {}
    def encode(data):
        encoded = []
        for r in data:
            new_row = []
            for i in feature_indices:
                val = r[i]
                if isinstance(val, str):
                    if i not in encoders:
                        encoders[i] = LabelEncoder().fit([row_i[i] for row_i in rows])
                    new_row.append(float(encoders[i].transform([val])[0]))
                else:
                    new_row.append(float(val))
            encoded.append(new_row)
        return encoded

    X_train = encode(rows)
    X_test = [encode([row])[0]]
    
    # Predict goals
    predictions = []
    for g_idx in goal_indices:
        y_train = [r[g_idx] for r in rows]
        model = RandomForestRegressor(n_estimators=50, random_state=42)
        model.fit(X_train, y_train)
        predictions.append(model.predict(X_test)[0])
    
    # Calculate percentile
    better_count = 0
    for r in rows:
        is_better = True
        for i, g_idx in enumerate(goal_indices):
            val_r = r[g_idx]
            val_p = predictions[i]
            # If +, higher is better. If -, lower is better.
            if header[g_idx].endswith('+'):
                if not (val_p >= val_r): is_better = False
            else:
                if not (val_p <= val_r): is_better = False
        if is_better:
            better_count += 1
            
    p = better_count / len(rows)
    reason = f"This arrangement ranks in the top {int((1-p)*100)}% of all teams based on predicted performance metrics."
    return p, reason

def fix(header, row, levers):
    """
    Recommends changes for a 'bad' row (short tenure) to improve retention.
    We prioritize increasing 'PercentSalaryHike' or 'StockOptionLevel'
    and reducing 'overTime' or 'businessTravel' frequency.
    """
    recommendations = {}
    
    # Strategy:
    # 1. If 'overTime' is 'Yes', change to 'No' if allowed.
    # 2. If 'PercentSalaryHike' is low, increase it.
    # 3. If 'StockOptionLevel' is low, increase it.
    
    if 'overTime' in levers and row[header.index('overTime')] == 'Yes':
        if 'No' in levers['overTime']:
            recommendations['overTime'] = 'No'
            return recommendations

    if 'PercentSalaryHike' in levers:

        current = row[header.index('PercentSalaryHike')]
        if current < high:
            recommendations['PercentSalaryHike'] = min(high, current + 5.0)

        current = row[header.index('StockOptionLevel')]
        if current < high:
            recommendations['StockOptionLevel'] = min(high, current + 1.0)
            return recommendations

    # Default: pick first available lever and set to a reasonable value
    for col, val in levers.items():
        if isinstance(val, list):
            recommendations[col] = val[0]
        else:
            recommendations[col] = val[1]
        break
        
    return recommendations

import csv

def spot(header, rows):
    """
    Identifies rows that look 'wrong' by checking for outliers in numeric columns.
    A row is considered suspicious if it contains values that are statistically
    improbable (e.g., 3 standard deviations from the mean).
    """
    suspicious = {}
    
    # Identify numeric columns
    numeric_indices = [i for i, name in enumerate(header) if name[0].isupper()]
    
    # Calculate mean and std for each numeric column
    stats = {}
    for i in numeric_indices:
        vals = [row[i] for row in rows if isinstance(row[i], (int, float))]
        if not vals:
            continue
        mean = sum(vals) / len(vals)
        std = (sum((x - mean) ** 2 for x in vals) / len(vals)) ** 0.5
        stats[i] = (mean, std)
    
    # Flag rows with values > 3 standard deviations from the mean
    for row_idx, row in enumerate(rows):
        score = 0
        for i in numeric_indices:
            if i in stats:
                mean, std = stats[i]
                if std > 0:
                    val = row[i]
                    z_score = abs(val - mean) / std
                    if z_score > 3:
                        score += z_score
        
        if score > 0:
            suspicious[row_idx] = score
            
    # Sort by suspicion score descending
    sorted_rows = sorted(suspicious.items(), key=lambda x: x[1], reverse=True)
    return [item[0] for item in sorted_rows]

import csv
import random

def race(header, rows, label):
    """
    Compares three strategies to find the best row using a shared budget of 50 labels.
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith('+') or name.endswith('-')]
    
    def get_score(row_idx):
        """Returns a normalized score where higher is better."""
        goals = label(row_idx)
        score = 0.0
        for i, g_idx in enumerate(goal_indices):
            val = goals[i]
            # For simplicity, we assume goal columns are numeric.
            # Normalize: if +, higher is better; if -, lower is better.
            # This is a heuristic to compare rows.
            if header[g_idx].endswith('+'):
                score += val
            else:
                score -= val
        return score

    # Strategies
    def random_strategy(budget):
        indices = random.sample(range(len(rows)), budget)
        best_idx = -1
        best_val = -float('inf')
        for idx in indices:
            val = get_score(idx)
            if val > best_val:
                best_val = val
                best_idx = idx
        return best_idx

    def greedy_strategy(budget):
        # Sample a few, then pick the best
        indices = random.sample(range(len(rows)), budget)
        best_idx = -1
        best_val = -float('inf')
        for idx in indices:
            val = get_score(idx)
            if val > best_val:
                best_val = val
                best_idx = idx
        return best_idx

    def extreme_strategy(budget):
        # Focus on rows with high 'TotalWorkingYears' as a proxy for 'best'
        years_idx = header.index('TotalWorkingYears')
        sorted_indices = sorted(range(len(rows)), key=lambda i: rows[i][years_idx], reverse=True)
        best_idx = -1
        best_val = -float('inf')
        for idx in sorted_indices[:budget]:
            val = get_score(idx)
            if val > best_val:
                best_val = val
                best_idx = idx
        return best_idx

    # Budget allocation: 16 + 17 + 17 = 50
    strategies = {
        "random": (random_strategy, 16),
        "greedy": (greedy_strategy, 17),
        "extreme": (extreme_strategy, 17)
    }
    
    results = {}
    best_overall_idx = -1
    best_overall_val = -float('inf')
    
    for name, (func, budget) in strategies.items():
        idx = func(budget)
        results[name] = idx
        val = get_score(idx)
        if val > best_overall_val:
            best_overall_val = val
            best_overall_idx = idx
            
    winner = max(results, key=lambda n: get_score(results[n]))
    
    return {
        "winner": winner,
        "row": results[winner],
        "tried": list(strategies.keys())
    }

import csv

def remember(header, rows):
    """
    Finds the 5 most similar employees to the first employee in the dataset
    (excluding the first employee themselves) based on Euclidean distance
    of normalized numeric features.
    """
    # Identify numeric columns (excluding goals)
    goal_indices = [i for i, name in enumerate(header) if name.endswith('+') or name.endswith('-')]
    numeric_indices = [i for i, name in enumerate(header) if name[0].isupper() and i not in goal_indices]
    
    # Normalize numeric data for distance calculation
    data_matrix = []
    for row in rows:
        data_matrix.append([row[i] for i in numeric_indices])
    
    # Simple min-max normalization
    mins = [min(col) for col in zip(*data_matrix)]
    maxs = [max(col) for col in zip(*data_matrix)]
    
    def normalize(row_vals):
        norm = []
        for i, val in enumerate(row_vals):
            denom = (maxs[i] - mins[i]) if (maxs[i] - mins[i]) != 0 else 1
            norm.append((val - mins[i]) / denom)
        return norm

    target_norm = normalize(data_matrix[0])
    
    # Calculate distances
    distances = []
    for idx, row_vals in enumerate(data_matrix):
        if idx == 0: continue
        row_norm = normalize(row_vals)
        dist = sum((a - b) ** 2 for a, b in zip(target_norm, row_norm)) ** 0.5
        distances.append((dist, idx))
    
    # Return 5 closest indices
    distances.sort()
    limit = max(1, len(rows) // 20)
    return [idx for _, idx in distances[:limit]]

import csv
from scipy import stats

def certify(xs, ys):
    """
    Performs a Welch's t-test to determine if two samples have significantly 
    different means. Returns True if the difference is not statistically 
    significant (p > 0.05), False otherwise.
    """
    if len(xs) < 2 or len(ys) < 2:
        return True
    
    t_stat, p_value = stats.ttest_ind(xs, ys, equal_var=False)
    
    # If p > 0.05, we fail to reject the null hypothesis (they are the same apart from noise)
    return p_value > 0.05
