```python
import csv
from scipy import stats

def certify(xs, ys):
    """
    Performs a Welch's t-test to determine if two samples have significantly 
    different means. Returns True if the difference is not statistically 
    significant (p > 0.05), False otherwise.
    """
    if len(xs) < 2 or len(ys) < 2:
        return True
    
    t_stat, p_value = stats.ttest_ind(xs, ys, equal_var=False)
    
    # If p > 0.05, we fail to reject the null hypothesis (they are the same apart from noise)
    return p_value > 0.05

def analyze_overtime_impact(header, rows):
    """
    Compares 'YearsInCurrentRole+' for employees who work overtime vs those who don't.
    """
    ot_idx = header.index('overTime')
    years_idx = header.index('YearsInCurrentRole+')
    
    yes_ot = [row[years_idx] for row in rows if row[ot_idx] == 'Yes']
    no_ot = [row[years_idx] for row in rows if row[ot_idx] == 'No']
    
    is_noise = certify(yes_ot, no_ot)
    
    print(f"Analysis of 'YearsInCurrentRole+' by Overtime status:")
    print(f"Average years (Overtime): {sum(yes_ot)/len(yes_ot):.2f}")
    print(f"Average years (No Overtime): {sum(no_ot)/len(no_ot):.2f}")
    
    if is_noise:
        print("Conclusion: The observed difference is likely just noise (not statistically significant).")
    else:
        print("Conclusion: There is a statistically significant difference in tenure based on overtime.")
    
    return is_noise

if __name__ == "__main__":
    filename = 'WA_Fn-UseC_-HR-Employee-Attrition.n1000.train.csv'
    rows = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = []
            for i, val in enumerate(row):
                if header[i][0].isupper():
                    processed_row.append(float(val))
                else:
                    processed_row.append(val)
            rows.append(processed_row)
    
    analyze_overtime_impact(header, rows)
```
