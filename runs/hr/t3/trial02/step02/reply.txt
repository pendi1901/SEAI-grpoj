import csv
from sklearn.ensemble import RandomForestRegressor
from sklearn.preprocessing import LabelEncoder

def guess(header, train, test):
    """
    Estimates the value of the first goal column ('YearsAtCompany+')
    for test rows using a Random Forest Regressor.
    """
    # Identify goal column (first column ending in + or -)
    goal_idx = -1
    for i, col in enumerate(header):
        if col.endswith('+') or col.endswith('-'):
            goal_idx = i
            break
    
    # Prepare data
    # We use all columns except the goal columns as features
    feature_indices = [i for i in range(len(header)) if not (header[i].endswith('+') or header[i].endswith('-'))]
    
    # Encode symbolic columns
    encoders = {}
    def encode_data(data_rows):
        encoded = []
        for row in data_rows:
            new_row = []
            for i in feature_indices:
                val = row[i]
                if isinstance(val, str):
                    if i not in encoders:
                        encoders[i] = LabelEncoder()
                        # Fit on all available data to ensure consistency
                        all_vals = [r[i] for r in train]
                        encoders[i].fit(all_vals)
                    new_row.append(encoders[i].transform([val])[0])
                else:
                    new_row.append(val)
            encoded.append(new_row)
        return encoded

    X_train = encode_data(train)
    y_train = [row[goal_idx] for row in train]
    X_test = encode_data(test)
    
    # Train model
    model = RandomForestRegressor(n_estimators=100, random_state=42)
    model.fit(X_train, y_train)
    
    # Predict
    return model.predict(X_test).tolist()

if __name__ == "__main__":
    filename = 'WA_Fn-UseC_-HR-Employee-Attrition.n1000.train.csv'
    rows = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = []
            for i, val in enumerate(row):
                if header[i][0].isupper():
                    processed_row.append(float(val))
                else:
                    processed_row.append(val)
            rows.append(processed_row)
    
    # Split for demonstration: use last 10 rows as 'test'
    train_data = rows[:-10]
    test_data = [row[:] for row in rows[-10:]]
    
    # Mask the goal column in test data
    goal_idx = next(i for i, h in enumerate(header) if h.endswith('+') or h.endswith('-'))
    for row in test_data:
        row[goal_idx] = "?"
        
    predictions = guess(header, train_data, test_data)
    
    print("Estimation of 'YearsAtCompany+' for new hires:")
    print(f"{'Actual':<10} | {'Predicted':<10}")
    print("-" * 25)
    for i, pred in enumerate(predictions):
        actual = rows[-(10-i)][goal_idx]
        print(f"{actual:<10.1f} | {pred:<10.2f}")
