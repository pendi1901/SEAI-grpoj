```python
import csv
from scipy import stats

def certify(xs, ys):
    """
    Returns True if the difference between two groups is not statistically 
    significant (p > 0.05), suggesting the difference is likely noise.
    Returns False if the difference is statistically significant (p <= 0.05).
    """
    t_stat, p_value = stats.ttest_ind(xs, ys, equal_var=False)
    return p_value > 0.05

def analyze_attrition_and_overtime(header, rows):
    """
    Compares 'YearsAtCompany+' for employees who work overtime vs those who don't.
    """
    overtime_idx = header.index('overTime')
    years_idx = header.index('YearsAtCompany+')
    
    overtime_yes = [row[years_idx] for row in rows if row[overtime_idx] == 'Yes']
    overtime_no = [row[years_idx] for row in rows if row[overtime_idx] == 'No']
    
    avg_yes = sum(overtime_yes) / len(overtime_yes)
    avg_no = sum(overtime_no) / len(overtime_no)
    
    is_noise = certify(overtime_yes, overtime_no)
    
    return avg_yes, avg_no, is_noise

if __name__ == "__main__":
    filename = 'WA_Fn-UseC_-HR-Employee-Attrition.n1000.train.csv'
    rows = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = []
            for i, val in enumerate(row):
                if header[i][0].isupper():
                    processed_row.append(float(val))
                else:
                    processed_row.append(val)
            rows.append(processed_row)
            
    avg_yes, avg_no, is_noise = analyze_attrition_and_overtime(header, rows)
    
    print("Analysis: Do employees who work overtime leave sooner?")
    print(f"Average years at company (Overtime): {avg_yes:.2f}")
    print(f"Average years at company (No Overtime): {avg_no:.2f}")
    
    if is_noise:
        print("Conclusion: The observed difference is likely just noise (not statistically significant).")
    else:
        print("Conclusion: The difference is statistically significant; overtime appears to correlate with tenure.")
```
