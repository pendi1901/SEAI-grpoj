        return [0]

    years = [row[idx] for row in rows]

    sorted_years = sorted(years, reverse=True)
    limit = max(1, int(len(rows) * 0.05))

    # Find indices of rows meeting the threshold
    best_indices = [i for i, val in enumerate(years) if val >= threshold]
    
    # Ensure we return at most 5% of rows (handle ties at the threshold)
    return best_indices[:limit]

def guess(header, train, test):

import csv
from scipy import stats

def certify(xs, ys):
    """
    Returns True if the difference between two groups is not statistically 
    significant (p > 0.05), suggesting the difference is likely noise.
    Returns False if the difference is statistically significant (p <= 0.05).
    """
    t_stat, p_value = stats.ttest_ind(xs, ys, equal_var=False)
    return p_value > 0.05

def analyze_attrition_and_overtime(header, rows):
    """
    Compares 'YearsAtCompany+' for employees who work overtime vs those who don't.
    """
    overtime_idx = header.index('overTime')
    years_idx = header.index('YearsAtCompany+')
    
    overtime_yes = [row[years_idx] for row in rows if row[overtime_idx] == 'Yes']
    overtime_no = [row[years_idx] for row in rows if row[overtime_idx] == 'No']
    
    avg_yes = sum(overtime_yes) / len(overtime_yes)
    avg_no = sum(overtime_no) / len(overtime_no)
    
    is_noise = certify(overtime_yes, overtime_no)
    
    return avg_yes, avg_no, is_noise

import csv

def flow(header, row, seen):
    """
    Decides if the current employee needs a 'what would keep you here' conversation.
    Strategy: Identify 'high-value' employees who are at risk of attrition.
    We look for employees with high 'YearsAtCompany+' (tenure) and 
    'JobSatisfaction' who are currently 'No' for 'attrition' but might be 
    approaching a threshold where they could leave.
    
    If the current row's 'YearsAtCompany+' is in the top 10% of the 'seen' 
    data and they have high 'JobSatisfaction', we flag them.
    """
    years_idx = header.index('YearsAtCompany+')
    sat_idx = header.index('JobSatisfaction')
    attrition_idx = header.index('attrition')
    
    # If this is the first row, we can't compare, so let it go
    if not seen:
        return False
    
    # Extract tenure of seen employees
    seen_years = [r[years_idx] for r in seen]
    threshold = sorted(seen_years, reverse=True)[max(0, int(len(seen) * 0.1))]
    
    current_years = row[years_idx]
    current_sat = row[sat_idx]
    current_attrition = row[attrition_idx]
    
    # Logic: If they are a long-tenured employee (top 10%) with high satisfaction
    # but we want to ensure they stay (proactive retention), flag them.
    if current_years >= threshold and current_sat >= 3 and current_attrition == 'No':
        return True
    
    return False

def blame(header, rows):
    """
    Identifies input columns that best explain why employees have short stays.
    'YearsAtCompany+' is a goal to maximize. We define 'worst' rows as those
    with the lowest 'YearsAtCompany+' values. We use a Random Forest to 
    determine feature importance relative to this goal.
    """
    goal_col = 'YearsAtCompany+'
    goal_idx = header.index(goal_col)
    
    # Identify input columns (exclude goals)
    input_indices = [i for i, col in enumerate(header) if not (col.endswith('+') or col.endswith('-'))]
    
    # Prepare data
    X = []
    y = []
    for row in rows:
        processed_row = []
        for i in input_indices:
            val = row[i]
            if isinstance(val, str):
                # Simple label encoding for symbolic columns
                unique_vals = sorted(list(set(r[i] for r in rows)))
                processed_row.append(unique_vals.index(val))
            else:
                processed_row.append(val)
        X.append(processed_row)
        y.append(row[goal_idx])
    
    # Train a model to see what predicts tenure
    model = RandomForestRegressor(n_estimators=50, random_state=42)
    model.fit(X, y)
    
    # Get feature importances
    importances = model.feature_importances_
    
    # Map importances back to column names
    feature_importance_map = []
    for i, idx in enumerate(input_indices):
        feature_importance_map.append((header[idx], importances[i]))
    
    # Sort by importance descending
    feature_importance_map.sort(key=lambda x: x[1], reverse=True)
    
    # Return the top 3 columns that explain the variance in tenure
    return [item[0] for item in feature_importance_map[:3]]

def justify(header, rows, row):
    """
    Estimates the performance of a given row relative to the dataset
    and returns its percentile rank and a justification.
    """
    # Identify goal columns
    goal_indices = [i for i, h in enumerate(header) if h.endswith('+') or h.endswith('-')]
    input_indices = [i for i in range(len(header)) if i not in goal_indices]
    
    # Prepare training data
    encoders = {}
    def encode(data):
        encoded = []
        for r in data:
            new_r = []
            for i in input_indices:
                val = r[i]
                if isinstance(val, str):
                    if i not in encoders:
                        encoders[i] = LabelEncoder().fit([x[i] for x in rows])
                    new_r.append(encoders[i].transform([val])[0])
                else:
                    new_r.append(val)
            encoded.append(new_r)
        return encoded

    X = encode(rows)
    
    # We focus on the first goal column for the ranking
    goal_idx = goal_indices[0]
    y = [r[goal_idx] for r in rows]
    
    # Train model to predict the goal
    model = RandomForestRegressor(n_estimators=50, random_state=42)
    model.fit(X, y)
    
    # Predict for the target row
    target_X = encode([row])
    prediction = model.predict(target_X)[0]
    
    # Calculate percentile rank (fraction of rows worse than prediction)
    # If goal ends in +, higher is better. If -, lower is better.
    is_maximize = header[goal_idx].endswith('+')
    if is_maximize:
        better_than = [val for val in y if val < prediction]
    else:
        better_than = [val for val in y if val > prediction]
        
    p = len(better_than) / len(y)
    
    reason = (f"This arrangement is estimated to be in the top {int((1-p)*100)}% "
              f"of all teams based on its predicted {header[goal_idx].strip('+').strip('-')}.")
    
    return p, reason

def choose(header, rows, label):

    labeling them to find the true goal values, and returning

    num_to_sample = min(len(rows), 50)
    indices_to_sample = random.sample(range(len(rows)), num_to_sample)
    
    best_idx = -1
    best_score = None
    
    for idx in indices_to_sample:
        # Get true goal values
        goals = label(idx)
        
        # Calculate a normalized score for this row
        # Since we want to maximize '+' and minimize '-', we normalize
        # For simplicity, we use the first goal column provided

            score = -val
        else:
            score = val
            
        if best_score is None or score > best_score:
            best_score = score
            best_idx = idx
            
    return best_idx

import csv

def spot(header, rows):
    """
    Identifies suspicious rows by checking for outliers in numeric columns.
    A row is considered suspicious if it contains values that are statistically
    improbable (e.g., outside 3 standard deviations from the mean).
    """
    suspicious_scores = {}

    # Identify numeric columns
    numeric_indices = [i for i, col in enumerate(header) if col[0].isupper()]

    # Calculate mean and std for each numeric column
    stats = {}
    for i in numeric_indices:
        vals = [row[i] for row in rows if isinstance(row[i], (int, float))]
        if not vals:
            continue
        mean = sum(vals) / len(vals)
        std = (sum((x - mean) ** 2 for x in vals) / len(vals)) ** 0.5
        stats[i] = (mean, std)

    # Score rows based on how many standard deviations values are from the mean
    for row_idx, row in enumerate(rows):
        score = 0
        for i in numeric_indices:
            if i in stats:
                mean, std = stats[i]
                if std > 0:
                    val = row[i]
                    z_score = abs(val - mean) / std
                    # If a value is more than 4 standard deviations away, it's highly suspicious
                    if z_score > 4:
                        score += z_score
        
        if score > 0:
            suspicious_scores[row_idx] = score

    # Return indices sorted by suspicion score descending
    return sorted(suspicious_scores, key=suspicious_scores.get, reverse=True)

import csv
import random

def race(header, rows, label):
    """
    Compares three strategies for finding the best employee record:
    1. random: Picks rows at random.
    2. extreme: Picks rows with the highest 'TotalWorkingYears'.
    3. balanced: Picks rows with 'JobSatisfaction' >= 3.
    """
    goal_indices = [i for i, h in enumerate(header) if h.endswith('+') or h.endswith('-')]
    
    def get_score(row_idx):
        goals = label(row_idx)
        score = 0.0
        for i, g_idx in enumerate(goal_indices):
            val = goals[i]
            # Normalize: + is better high, - is better low
            if header[g_idx].endswith('-'):
                score -= val
            else:
                score += val
        return score

    strategies = {
        "random": lambda: random.sample(range(len(rows)), 16),
        "extreme": lambda: sorted(range(len(rows)), key=lambda i: rows[i][header.index('TotalWorkingYears')], reverse=True)[:17],
        "balanced": lambda: [i for i, r in enumerate(rows) if r[header.index('JobSatisfaction')] >= 3][:17]
    }

    results = {}
    best_overall_score = float('-inf')
    winner = None
    best_row_idx = -1

    for name, selector in strategies.items():
        indices = selector()
        best_strat_score = float('-inf')
        best_strat_row = -1
        
        for idx in indices:
            score = get_score(idx)
            if score > best_strat_score:
                best_strat_score = score
                best_strat_row = idx
            
            if score > best_overall_score:
                best_overall_score = score
                winner = name
                best_row_idx = idx
        
        results[name] = best_strat_score

    return {"winner": winner, "row": best_row_idx, "tried": list(strategies.keys())}

import csv

def remember(header, rows):
    """
    Finds the 5% of employees most similar to the first employee in the dataset.
    Similarity is calculated using a simple Euclidean distance on normalized 
    numeric inputs.
    """
    # Identify numeric input columns (excluding goals)
    numeric_indices = [
        i for i, col in enumerate(header) 
        if col[0].isupper() and not (col.endswith('+') or col.endswith('-'))
    ]
    
    # Normalize numeric columns to [0, 1] for distance calculation
    normalized_rows = []
    for row in rows:
        norm_row = []
        for i in numeric_indices:
            # Find min/max for this column across all rows
            col_vals = [r[i] for r in rows]
            min_v, max_v = min(col_vals), max(col_vals)
            if max_v == min_v:
                norm_row.append(0.0)
            else:
                norm_row.append((row[i] - min_v) / (max_v - min_v))
        normalized_rows.append(norm_row)
    
    # Target is the first employee
    target = normalized_rows[0]
    
    # Calculate distances
    distances = []
    for i, row in enumerate(normalized_rows):
        dist = sum((row[j] - target[j])**2 for j in range(len(target)))**0.5
        distances.append((i, dist))
    
    # Sort by distance (closest first), skip the target itself (index 0)
    distances.sort(key=lambda x: x[1])
    
    # Return top 5% (excluding the target itself)
    limit = max(1, int(len(rows) * 0.05))
    return [d[0] for d in distances[1:limit+1]]

import csv
from scipy import stats

def certify(xs, ys):
    """
    Returns True if the difference between two groups is not statistically 
    significant (p > 0.05), suggesting the difference is likely noise.
    Returns False if the difference is statistically significant (p <= 0.05).
    """
    if not xs or not ys:
        return True
    t_stat, p_value = stats.ttest_ind(xs, ys, equal_var=False)
    return p_value > 0.05

def analyze_training_impact(header, rows):

    avg_low = sum(low_training) / len(low_training) if low_training else 0
    
    is_noise = certify(high_training, low_training)
    
    return avg_high, avg_low, is_noise

import csv

def flow(header, row, seen):
    """
    Detects if a team's composition has drifted significantly from the 'normal'
    distribution of the seen data. We monitor the distribution of 'JobSatisfaction'
    and 'MonthlyIncome' using a simple threshold on the mean.
    
    Returns True if the current row causes the team's average for these metrics
    to deviate by more than 2 standard deviations from the historical mean.
    """
    if len(seen) < 30:
        return False

    # Identify indices
    sat_idx = header.index('JobSatisfaction')
    inc_idx = header.index('MonthlyIncome')
    
    # Calculate historical stats
    seen_sat = [r[sat_idx] for r in seen]
    seen_inc = [r[inc_idx] for r in seen]
    
    mean_sat = sum(seen_sat) / len(seen_sat)
    std_sat = (sum((x - mean_sat) ** 2 for x in seen_sat) / len(seen_sat)) ** 0.5
    
    mean_inc = sum(seen_inc) / len(seen_inc)
    std_inc = (sum((x - mean_inc) ** 2 for x in seen_inc) / len(seen_inc)) ** 0.5
    
    # Check if current row is an outlier (drift)
    # If the new row is significantly different, flag the team
    drift_sat = abs(row[sat_idx] - mean_sat) > (2 * std_sat) if std_sat > 0 else False
    drift_inc = abs(row[inc_idx] - mean_inc) > (2 * std_inc) if std_inc > 0 else False
    
    return drift_sat or drift_inc

def blame(header, rows):
    """
    Identifies input columns that have the least impact on the goal columns.
    We train a Random Forest Regressor to predict the goal columns and 
    identify features with near-zero importance.
    """
    # Identify goal columns
    goal_indices = [i for i, h in enumerate(header) if h.endswith('+') or h.endswith('-')]
    input_indices = [i for i in range(len(header)) if i not in goal_indices]
    
    # Prepare data
    X = []
    y = []
    
    # We aggregate goals into a single score for simplicity
    # Normalize goals: + is better high, - is better low
    for row in rows:
        processed_row = []
        for i in input_indices:
            val = row[i]
            if isinstance(val, str):
                # Simple label encoding
                unique_vals = sorted(list(set(r[i] for r in rows)))
                processed_row.append(unique_vals.index(val))
            else:
                processed_row.append(val)
        X.append(processed_row)
        
        # Calculate a composite score for the row
        score = 0
        for g_idx in goal_indices:
            val = row[g_idx]
            if header[g_idx].endswith('-'):
                score -= val
            else:
                score += val
        y.append(score)
    
    # Train model
    model = RandomForestRegressor(n_estimators=50, random_state=42)
    model.fit(X, y)
    
    # Get feature importances
    importances = model.feature_importances_
    
    # Map importances back to column names
    feature_importance_map = []
    for i, idx in enumerate(input_indices):
        feature_importance_map.append((header[idx], importances[i]))
    
    # Sort by importance ascending (least important first)
    feature_importance_map.sort(key=lambda x: x[1])
    
    # Return columns with very low importance (bottom 5)
    return [item[0] for item in feature_importance_map[:5]]

import csv

def fix(header, row, levers):
    """
    Recommends changes to a 'bad' row (short tenure) to improve it,
    specifically avoiding 'MonthlyIncome' or 'PercentSalaryHike' 
    since there is no money for pay increases.
    """
    recommendations = {}
    
    # Strategy:
    # 1. If 'overTime' is 'Yes', change to 'No' (improves work-life balance).
    # 2. If 'JobSatisfaction' is low, try to increase it via 'TrainingTimesLastYear'.
    # 3. If 'StockOptionLevel' is a lever, set it to the high end (non-monetary incentive).
    
    # Avoid pay-related levers
    forbidden = ['MonthlyIncome', 'PercentSalaryHike']
    
    # 1. Reduce Overtime
    if 'overTime' in levers and row[header.index('overTime')] == 'Yes':
        recommendations['overTime'] = 'No'
    
    # 2. Increase Training (Non-monetary growth)
    if 'TrainingTimesLastYear' in levers:
        low, high = levers['TrainingTimesLastYear']
        recommendations['TrainingTimesLastYear'] = float(high)
        
    # 3. Increase Stock Options (Equity, not cash)
    if 'StockOptionLevel' in levers:
        low, high = levers['StockOptionLevel']
        recommendations['StockOptionLevel'] = float(high)
        
    # Filter out forbidden levers if they were accidentally included
    final_recommendations = {k: v for k, v in recommendations.items() if k not in forbidden}
    
    # Ensure at least one change is made
    if not final_recommendations:
        for col, val in levers.items():
            if col not in forbidden:
                if isinstance(val, list):
                    final_recommendations[col] = val[0]
                else:
                    final_recommendations[col] = float(val[1])
                break
            
    return final_recommendations

import csv

def spot(header, rows):
    """
    Identifies rows that are 'unlike anyone we have employed' by measuring
    the distance of each row from the centroid of the existing data.
    Rows with a high distance (outliers) are considered 'untrusted' because
    the model has no historical precedent for such an employee.
    """
    # Identify numeric input columns (excluding goals)
    numeric_indices = [
        i for i, col in enumerate(header) 
        if col[0].isupper() and not (col.endswith('+') or col.endswith('-'))
    ]
    
    # Calculate mean and std for each numeric column to normalize
    stats = {}
    for i in numeric_indices:
        vals = [row[i] for row in rows if isinstance(row[i], (int, float))]
        if not vals:
            continue
        mean = sum(vals) / len(vals)
        std = (sum((x - mean) ** 2 for x in vals) / len(vals)) ** 0.5
        stats[i] = (mean, std)

    # Score rows based on their Mahalanobis-like distance (Euclidean in normalized space)
    # A higher score means the row is further from the "average" employee
    suspicious_scores = {}
    for row_idx, row in enumerate(rows):
        score = 0
        for i in numeric_indices:
            if i in stats:
                mean, std = stats[i]
                if std > 0:
                    # Z-score squared contribution
                    z_score = (row[i] - mean) / std
                    score += z_score ** 2
        
        # We only care about rows that are significantly different
        if score > 0:
            suspicious_scores[row_idx] = score

    # Return indices sorted by suspicion score descending
    return sorted(suspicious_scores, key=suspicious_scores.get, reverse=True)
