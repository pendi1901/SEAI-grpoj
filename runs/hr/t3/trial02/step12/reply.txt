import csv
from sklearn.ensemble import RandomForestRegressor
from sklearn.preprocessing import LabelEncoder

def guess(header, train, test):
    """
    Estimates the first goal column ('YearsAtCompany+') for test rows.
    To answer the customer's question about cutting overtime in Sales,
    we train a model on the existing data and predict the tenure for
    the test rows where 'overTime' is set to 'No' and 'department' is 'Sales'.
    """
    # Identify goal column (first column ending in + or -)
    goal_idx = next(i for i, col in enumerate(header) if col.endswith('+') or col.endswith('-'))
    
    # Identify feature indices (all columns except goal columns)
    feature_indices = [i for i in range(len(header)) if not (header[i].endswith('+') or header[i].endswith('-'))]
    
    # Encode symbolic columns
    encoders = {}
    def encode_data(data_rows):
        encoded = []
        for row in data_rows:
            new_row = []
            for i in feature_indices:
                val = row[i]
                if isinstance(val, str):
                    if i not in encoders:
                        encoders[i] = LabelEncoder()
                        # Fit on all available data to ensure consistency
                        all_vals = [r[i] for r in train]
                        encoders[i].fit(all_vals)
                    new_row.append(encoders[i].transform([val])[0])
                else:
                    new_row.append(val)
            encoded.append(new_row)
        return encoded

    X_train = encode_data(train)
    y_train = [row[goal_idx] for row in train]
    X_test = encode_data(test)
    
    # Train model
    model = RandomForestRegressor(n_estimators=100, random_state=42)
    model.fit(X_train, y_train)
    
    # Predict
    return model.predict(X_test).tolist()

if __name__ == "__main__":
    filename = 'WA_Fn-UseC_-HR-Employee-Attrition.n1000.train.csv'
    rows = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = []
            for i, val in enumerate(row):
                if header[i][0].isupper():
                    processed_row.append(float(val))
                else:
                    processed_row.append(val)
            rows.append(processed_row)
    
    # Customer scenario: What happens if we cut overtime in Sales?
    # We take Sales employees who currently work overtime and simulate them NOT working overtime.
    dept_idx = header.index('department')
    ot_idx = header.index('overTime')
    goal_idx = next(i for i, h in enumerate(header) if h.endswith('+') or h.endswith('-'))
    
    sales_ot_rows = [r[:] for r in rows if r[dept_idx] == 'Sales' and r[ot_idx] == 'Yes']
    
    # Create test set: modify overtime to 'No'
    test_data = [r[:] for r in sales_ot_rows]
    for row in test_data:
        row[ot_idx] = 'No'
        row[goal_idx] = "?"
        
    predictions = guess(header, rows, test_data)
    
    avg_original = sum(r[goal_idx] for r in sales_ot_rows) / len(sales_ot_rows)
    avg_predicted = sum(predictions) / len(predictions)
    
    print("Analysis: Impact of cutting overtime in Sales on employee tenure.")
    print(f"Average tenure (Sales + Overtime): {avg_original:.2f} years")
    print(f"Predicted average tenure (Sales + No Overtime): {avg_predicted:.2f} years")
    print("\nConclusion: Based on the model, cutting overtime for Sales staff is "
          f"{'expected to increase' if avg_predicted > avg_original else 'expected to decrease'} "
          "average tenure.")
