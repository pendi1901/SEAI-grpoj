import csv
import random

def fix(header, row, levers):
    """
    Recommends changes to a 'bad' row (short tenure) to improve it.
    We prioritize increasing 'PercentSalaryHike' or 'StockOptionLevel' 
    if available, or adjusting 'overTime' to 'No'.
    """
    recommendations = {}
    
    # Strategy:
    # 1. If 'overTime' is 'Yes', change to 'No' (often associated with better retention).
    # 2. If 'PercentSalaryHike' is a lever, set it to the high end of the range.
    # 3. If 'StockOptionLevel' is a lever, set it to the high end of the range.
    
    if 'overTime' in levers and row[header.index('overTime')] == 'Yes':
        recommendations['overTime'] = 'No'
    
    if 'PercentSalaryHike' in levers:
        low, high = levers['PercentSalaryHike']
        recommendations['PercentSalaryHike'] = float(high)
        
    if 'StockOptionLevel' in levers:
        low, high = levers['StockOptionLevel']
        recommendations['StockOptionLevel'] = float(high)
        
    # Ensure at least one change is made
    if not recommendations:
        for col, val in levers.items():
            if isinstance(val, list):
                recommendations[col] = val[0]
            else:
                recommendations[col] = float(val[1])
            break
            
    return recommendations

if __name__ == "__main__":
    filename = 'WA_Fn-UseC_-HR-Employee-Attrition.n1000.train.csv'
    rows = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = []
            for i, val in enumerate(row):
                if header[i][0].isupper():
                    processed_row.append(float(val))
                else:
                    processed_row.append(val)
            rows.append(processed_row)

    # Define levers based on customer request
    levers = {
        'businessTravel': ['Non-Travel', 'Travel_Frequently', 'Travel_Rarely'],
        'overTime': ['No', 'Yes'],
        'TrainingTimesLastYear': (0.0, 6.0),
        'PercentSalaryHike': (11.0, 25.0),
        'StockOptionLevel': (0.0, 3.0)
    }

    # Simulate a 'bad' row (short tenure)
    tenure_idx = header.index('YearsAtCompany+')
    bad_rows = [r for r in rows if r[tenure_idx] < 2.0]
    
    if bad_rows:
        sample_bad_row = bad_rows[0]
        recommendation = fix(header, sample_bad_row, levers)
        
        print("Customer Request: How to improve retention for short-tenure staff?")
        print(f"Original Tenure: {sample_bad_row[tenure_idx]} years")
        print("Recommended changes:")
        for col, val in recommendation.items():
            print(f" - Change '{col}' to {val}")
        print("\nStrategy: Reducing overtime and increasing financial/growth incentives (pay/stock) "
              "are the most effective levers for improving employee tenure.")
    else:
        print("No short-tenure employees found to analyze.")
