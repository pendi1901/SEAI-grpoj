import csv

def remember(header, rows):
    """
    Identifies staff who stay 'unusually long'.
    'YearsAtCompany+' is a goal to maximize.
    We calculate the 95th percentile of 'YearsAtCompany+' and return
    the indices of rows that meet or exceed this threshold.
    """
    # Identify column index for 'YearsAtCompany+'
    try:
        idx = header.index('YearsAtCompany+')
    except ValueError:
        return [0]

    # Extract all values for this column
    years = [row[idx] for row in rows]
    
    # Calculate threshold for top 5% (unusually long)
    sorted_years = sorted(years, reverse=True)
    limit = max(1, len(rows) // 20)
    threshold = sorted_years[limit - 1]
    
    # Find indices of rows meeting the criteria
    best_indices = [i for i, val in enumerate(years) if val >= threshold]
    
    # Ensure we return at most 5% (limit is already calculated as 5%)
    return best_indices[:limit]

import csv
from sklearn.ensemble import RandomForestRegressor
from sklearn.preprocessing import LabelEncoder

def guess(header, train, test):
    """
    Estimates the value of the first goal column (YearsAtCompany+) 
    for test rows based on input features.
    """
    # Identify goal column (first column ending in + or -)
    goal_idx = -1
    for i, name in enumerate(header):
        if name.endswith('+') or name.endswith('-'):
            goal_idx = i
            break
    
    # Prepare training data
    # We use all columns except the goal columns as features
    feature_indices = [i for i, name in enumerate(header) if not (name.endswith('+') or name.endswith('-'))]
    
    # Encode symbolic columns
    encoders = {}
    def encode_data(data_rows, is_train=True):
        processed = []
        for row in data_rows:
            new_row = []
            for i in feature_indices:
                val = row[i]
                if isinstance(val, str):
                    if i not in encoders:
                        encoders[i] = LabelEncoder().fit([r[i] for r in train])
                    new_row.append(encoders[i].transform([val])[0])
                else:
                    new_row.append(val)
            processed.append(new_row)
        return processed

    X_train = encode_data(train)
    y_train = [row[goal_idx] for row in train]
    X_test = encode_data(test, is_train=False)
    
    # Train model
    model = RandomForestRegressor(n_estimators=100, random_state=42)
    model.fit(X_train, y_train)
    
    # Predict
    return model.predict(X_test).tolist()

import csv
from scipy import stats

def certify(xs, ys):
    """
    Returns True if the two groups are statistically indistinguishable 
    (p-value > 0.05), False if they are significantly different.
    """
    t_stat, p_value = stats.ttest_ind(xs, ys, equal_var=False)
    return p_value > 0.05

def analyze_attrition_impact(header, rows):
    """
    Compares 'YearsInCurrentRole+' for employees who work overtime vs those who don't.
    """
    overtime_idx = header.index('overTime')
    role_idx = header.index('YearsInCurrentRole+')
    
    yes_overtime = [row[role_idx] for row in rows if row[overtime_idx] == 'Yes']
    no_overtime = [row[role_idx] for row in rows if row[overtime_idx] == 'No']
    
    is_noise = certify(yes_overtime, no_overtime)
    
    return is_noise, yes_overtime, no_overtime

import csv

def flow(header, row, seen):
    """
    Decides if an employee needs a 'what would keep you here' conversation.
    We identify employees who are high-performers but potentially at risk 
    (e.g., low JobSatisfaction or low EnvironmentSatisfaction) 
    compared to the 'seen' population.
    """
    # Identify indices
    try:
        sat_idx = header.index('JobSatisfaction')
        env_idx = header.index('EnvironmentSatisfaction')
        years_idx = header.index('YearsAtCompany+')
    except ValueError:
        return False

    # If this is the first row, we can't compare, so let it go
    if not seen:
        return False

    # Calculate averages of the 'seen' population
    avg_sat = sum(r[sat_idx] for r in seen) / len(seen)
    avg_env = sum(r[env_idx] for r in seen) / len(seen)
    avg_years = sum(r[years_idx] for r in seen) / len(seen)

    # Criteria: If the employee has been here longer than average, 
    # but their satisfaction is below the average of those seen so far,
    # they are a candidate for a retention conversation.
    if row[years_idx] > avg_years:
        if row[sat_idx] < avg_sat or row[env_idx] < avg_env:
            return True

    return False
