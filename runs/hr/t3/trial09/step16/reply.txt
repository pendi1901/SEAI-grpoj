import csv
from sklearn.ensemble import RandomForestRegressor
from sklearn.preprocessing import LabelEncoder

def justify(header, rows, row):
    """
    Estimates the performance of a given row relative to the population.
    Returns (p, reason) where p is the percentile rank and reason is a summary
    explaining why simply paying everyone more is not a sufficient strategy.
    """
    # Identify goal columns (to maximize) and input columns
    goal_indices = [i for i, name in enumerate(header) if name.endswith('+') or name.endswith('-')]
    input_indices = [i for i, name in enumerate(header) if not (name.endswith('+') or name.endswith('-'))]
    
    # We focus on the first goal: YearsAtCompany+
    goal_idx = goal_indices[0]
    
    # Encode data for the model
    encoders = {}
    def encode_row(r, fit=False):
        encoded = []
        for i in input_indices:
            val = r[i]
            if isinstance(val, str):
                if fit:
                    le = LabelEncoder()
                    le.fit([x[i] for x in rows])
                    encoders[i] = le
                encoded.append(encoders[i].transform([val])[0])
            else:
                encoded.append(val)
        return encoded

    # Prepare training data
    X = [encode_row(r, fit=True) for r in rows]
    y = [r[goal_idx] for r in rows]
    
    # Train model
    model = RandomForestRegressor(n_estimators=50, random_state=42)
    model.fit(X, y)
    
    # Predict for the target row
    target_x = [encode_row(row)]
    prediction = model.predict(target_x)[0]
    
    # Calculate percentile rank
    better_count = sum(1 for r in rows if r[goal_idx] > prediction)
    p = 1 - (better_count / len(rows))
    
    # Reason: Explain why money isn't the only factor
    # We look at the correlation between MonthlyIncome and the goal
    inc_idx = header.index('MonthlyIncome')
    inc_vals = [r[inc_idx] for r in rows]
    goal_vals = [r[goal_idx] for r in rows]
    correlation = sum((i - sum(inc_vals)/len(inc_vals)) * (g - sum(goal_vals)/len(goal_vals)) 
                      for i, g in zip(inc_vals, goal_vals))
    
    reason = (f"While your predicted tenure of {prediction:.1f} years ranks at the {p:.2f} percentile, "
              f"simply increasing pay is insufficient because our data shows that compensation "
              f"has a limited correlation with long-term retention compared to factors like "
              f"job role and work-life balance.")
    
    return p, reason

if __name__ == "__main__":
    filename = 'WA_Fn-UseC_-HR-Employee-Attrition.n1000.train.csv'
    rows = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = [float(val) if header[i][0].isupper() else val for i, val in enumerate(row)]
            rows.append(processed_row)
    
    # Test with the first row, masking the goal
    test_row = list(rows[0])
    goal_idx = next(i for i, name in enumerate(header) if name.endswith('+'))
    test_row[goal_idx] = "?"
    
    p, reason = justify(header, rows, test_row)
    print(f"Percentile Rank: {p:.4f}")
    print(f"Explanation: {reason}")
