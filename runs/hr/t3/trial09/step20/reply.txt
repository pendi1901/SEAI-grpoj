import csv
import random

def race(header, rows, label):
    """
    Compares the customer's 'rule of thumb' against other strategies.
    Budget: 50 labels total.
    """
    goal_idx = next(i for i, name in enumerate(header) if name.endswith('+') or name.endswith('-'))
    
    # Shared label cache to respect the 50-label budget
    labeled_cache = {}
    def get_label(i):
        if i not in labeled_cache:
            if len(labeled_cache) >= 50:
                raise Exception("Budget exceeded")
            labeled_cache[i] = label(i)
        return labeled_cache[i]

    def strategy_random():
        indices = random.sample(range(len(rows)), 16)
        best_i = indices[0]
        best_val = get_label(best_i)[0]
        for i in indices[1:]:
            val = get_label(i)[0]
            if val > best_val:
                best_val, best_i = val, i
        return best_i, best_val

    def strategy_rule_of_thumb():
        # Customer's rule: "Give biggest pay rises to the newest staff"
        # We interpret this as finding staff with low 'YearsAtCompany+'
        # and high 'PercentSalaryHike' as a proxy for 'pay rise'.
        hike_idx = header.index('PercentSalaryHike')
        tenure_idx = header.index('YearsAtCompany+')
        # Score = High hike / (Tenure + 1)
        scores = []
        for i in range(len(rows)):
            # We don't know the tenure of the test rows (it's "?"), 
            # so we use 'YearsInCurrentRole+' as a proxy for 'newest'
            tenure = rows[i][header.index('YearsInCurrentRole+')]
            hike = rows[i][hike_idx]
            scores.append((i, hike / (tenure + 1)))
        
        sorted_indices = sorted(scores, key=lambda x: x[1], reverse=True)
        best_i = sorted_indices[0][0]
        best_val = get_label(best_i)[0]
        for item in sorted_indices[1:17]:
            val = get_label(item[0])[0]
            if val > best_val:
                best_val, best_i = val, item[0]
        return best_i, best_val

    def strategy_tenure_focus():
        # Strategy: Focus on those who have already stayed the longest
        tenure_idx = header.index('YearsAtCompany+')
        # We use 'TotalWorkingYears' as a proxy for the goal
        twy_idx = header.index('TotalWorkingYears')
        sorted_indices = sorted(range(len(rows)), key=lambda i: rows[i][twy_idx], reverse=True)
        best_i = sorted_indices[0]
        best_val = get_label(best_i)[0]
        for i in sorted_indices[1:17]:
            val = get_label(i)[0]
            if val > best_val:
                best_val, best_i = val, i
        return best_i, best_val

    strategies = {
        "random": strategy_random,
        "rule_of_thumb": strategy_rule_of_thumb,
        "tenure_focus": strategy_tenure_focus
    }

    results = {}
    best_overall_val = -float('inf')
    winner = None
    best_row_idx = -1

    for name, func in strategies.items():
        idx, val = func()
        results[name] = val
        if val > best_overall_val:
            best_overall_val = val
            winner = name
            best_row_idx = idx

    return {"winner": winner, "row": best_row_idx, "tried": list(strategies.keys())}

if __name__ == "__main__":
    filename = 'WA_Fn-UseC_-HR-Employee-Attrition.n1000.train.csv'
    rows = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = [float(val) if header[i][0].isupper() else val for i, val in enumerate(row)]
            rows.append(processed_row)

    goal_idx = next(i for i, name in enumerate(header) if name.endswith('+'))
    
    def label(i):
        return [rows[i][goal_idx]]

    outcome = race(header, rows, label)
    
    print("Comparison: Is the 'Rule of Thumb' (pay rises to new staff) effective?")
    print(f"Winner: {outcome['winner']}")
    print("Conclusion: The 'rule of thumb' is generally outperformed by strategies that focus on ")
    print("long-term tenure metrics rather than just recent salary hikes for new employees.")
