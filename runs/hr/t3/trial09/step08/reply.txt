```python
import csv

def fix(header, row, levers):
    """
    Recommends changes to a 'bad' row (short tenure) to improve it.
    We prioritize increasing 'MonthlyIncome' (if available) or adjusting 
    'OverTime'/'BusinessTravel' to improve retention.
    """
    recommendations = {}
    
    # Strategy: 
    # 1. If 'overTime' is 'Yes', try changing to 'No'.
    # 2. If 'MonthlyIncome' is a lever, try increasing it.
    # 3. If 'businessTravel' is a lever, try 'Non-Travel'.
    
    if 'overTime' in levers and row[header.index('overTime')] == 'Yes':
        if 'No' in levers['overTime']:
            recommendations['overTime'] = 'No'
            
    if 'MonthlyIncome' in levers:
        low, high = levers['MonthlyIncome']
        current = row[header.index('MonthlyIncome')]
        # Suggest a 10% increase if possible
        new_val = min(high, current * 1.1)
        if new_val > current:
            recommendations['MonthlyIncome'] = new_val
            
    if 'businessTravel' in levers and not recommendations:
        # Try to reduce travel stress
        if 'Non-Travel' in levers['businessTravel']:
            recommendations['businessTravel'] = 'Non-Travel'
            
    # Ensure at least one change is made if possible
    if not recommendations and levers:
        for col, val in levers.items():
            if isinstance(val, list):
                recommendations[col] = val[0]
            else:
                recommendations[col] = val[1]
                
    return recommendations

if __name__ == "__main__":
    filename = 'WA_Fn-UseC_-HR-Employee-Attrition.n1000.train.csv'
    rows = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = []
            for i, val in enumerate(row):
                if header[i][0].isupper():
                    processed_row.append(float(val))
                else:
                    processed_row.append(val)
            rows.append(processed_row)

    # Define levers based on customer request
    levers = {
        'businessTravel': ['Non-Travel', 'Travel_Frequently', 'Travel_Rarely'],
        'overTime': ['No', 'Yes'],
        'TrainingTimesLastYear': (0, 6),
        'MonthlyIncome': (1000.0, 20000.0),
        'StockOptionLevel': (0.0, 3.0)
    }

    # Pick a 'bad' row (e.g., one with low YearsAtCompany+)
    goal_idx = header.index('YearsAtCompany+')
    bad_row = min(rows, key=lambda r: r[goal_idx])
    
    changes = fix(header, bad_row, levers)
    
    print("Retention Strategy for Short-Tenure Staff:")
    print(f"Current Tenure: {bad_row[goal_idx]} years")
    print("Recommended adjustments to improve retention:")
    for col, val in changes.items():
        print(f"- Change '{col}' to: {val}")
    print("\nLogic: Reducing overtime and increasing compensation are primary levers to improve employee longevity.")
```
