import csv


def certify(xs, ys):
    # xs = predicted YearsAtCompany+, ys = actual YearsAtCompany+
    # Return True if predictions match reality within noise.
    n = len(xs)
    errors = [abs(x - y) for x, y in zip(xs, ys)]
    mae = sum(errors) / n
    scale = max(max(ys), max(xs)) if ys and xs else 1.0
    return mae <= 0.15 * scale


if __name__ == "__main__":
    with open("WA_Fn-UseC_-HR-Employee-Attrition.train.csv", "r") as f:
        reader = csv.reader(f)
        header = next(reader)
        rows = []
        for row in reader:
            parsed = []
            for cell in row:
                try:
                    parsed.append(float(cell))
                except ValueError:
                    parsed.append(cell)
            rows.append(parsed)

    target_col = "YearsAtCompany+"
    overtime_idx = header.index("overTime")
    over_yes = [float(r[header.index(target_col)]) for r in rows if r[overtime_idx] == "Y"]
    over_no = [float(r[header.index(target_col)]) for r in rows if r[overtime_idx] == "N"]

    avg_yes = sum(over_yes) / len(over_yes) if over_yes else 0.0
    avg_no = sum(over_no) / len(over_no) if over_no else 0.0
    diff = avg_yes - avg_no

    print(f"Overtime workers: avg YearsAtCompany+ = {avg_yes:.1f}")
    print(f"Non-overtime workers: avg YearsAtCompany+ = {avg_no:.1f}")
    print(f"Difference (overtime − non-overtime): {diff:+.1f} years")

    if diff < 0:
        print("Overtime workers leave sooner — the signal is real.")
    else:
        print("No clear difference found; overtime may not matter for tenure.")
