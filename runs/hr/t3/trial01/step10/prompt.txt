def remember(header, rows):
    # Columns ending in '+' are goals to maximise (tenure-related metrics).
    plus_goals = [c for c in header if c.endswith('+')]

    scores = []
    for row in rows:
        score = 0.0
        for col in plus_goals:
            val = float(row[header.index(col)])
            score += val
        scores.append(score)

    # Return top rows (at most 5% of total, at least one).
    n_top = max(1, int(len(rows) * 0.05))
    best_indices = np.argsort(scores)[-n_top:][::-1]
    return sorted(best_indices.tolist())

def flow(header, row, seen):
    plus_goals = [c for c in header if c.endswith('+')]

    if not seen:
        return False

    # Compute average of each '+' goal from previously seen rows
    goal_sums = {}
    for col in plus_goals:
        idx = header.index(col)
        total = 0.0
        count = 0
        for r in seen:
            val = float(r[idx])
            total += val
            count += 1
        goal_sums[col] = (total, count)

    # Evaluate current row's '+' goals against the running averages
    above_count = 0
    for col in plus_goals:
        avg_seen, _ = goal_sums[col]
        if float(row[header.index(col)]) > avg_seen:
            above_count += 1

    return above_count >= len(plus_goals) // 2 + 1

import csv


def blame(header, rows):
    """Return names of input columns that best explain why the worst rows are bad."""
    # Identify goals: '-' -> minimise, '+' -> maximise; everything else is input.
    goal_cols = [c for c in header if c.endswith('-') or c.endswith('+')]
    input_cols = [c for c in header if c not in goal_cols]

    n = len(rows)
    # Compute a score per row: sum of (goal - best_possible) normalised.
    scores = []
    for row in rows:
        s = 0.0
        for col in goal_cols:
            idx = header.index(col)
            val = float(row[idx])
            if col.endswith('-'):
                # lower is better -> penalty proportional to value
                s += val
            else:
                # higher is better -> penalty proportional to (max - val)
                s -= val
        scores.append(s)

    # Worst rows are those with highest score.
    worst = sorted(range(n), key=lambda i: scores[i], reverse=True)[:n // 5]

    # For each input column, compute correlation of its values with the "badness" (score).
    # We want columns where high value -> bad outcome (positive correlation with score).
    import numpy as np
    arr = np.array(scores)
    worst_mask = np.zeros(n, dtype=bool)
    for i in worst:
        worst_mask[i] = True

    result = []
    for col in input_cols:
        idx = header.index(col)
        vals = [float(row[idx]) if row[idx].strip() != '?' else 0.0 for row in rows]
        arr_col = np.array(vals, dtype=float)
        # Use only non-zero variance columns to avoid spurious correlations.
        if arr_col.std() < 1e-9:
            continue
        corr = np.corrcoef(arr_col, arr)[0, 1]
        result.append((col, float(corr)))

    # Sort by absolute correlation descending; pick top contributors with positive sign (high value -> bad).
    result.sort(key=lambda x: -abs(x[1]))
    return [c for c, _ in result[:5]]

def choose(header, rows, label):
    """Pick the best row among up to 50 candidates (labelled)."""
    plus = [c for c in header if c.endswith('+')]
    minus = [c for c in header if c.endswith('-')]

    # Collect all labelled indices.
    labelled = []
    for i, vals in enumerate(label):
        if any(v != "?" for v in vals):
            labelled.append(i)

    n = len(labelled)
    best_i = 0
    best_score = float("-inf")

    for idx in labelled:
        row = rows[idx]
        score = 0.0
        for col in plus:
            v = float(row[header.index(col)])
            score += v
        for col in minus:
            v = float(row[header.index(col)])
            score -= v

        if score > best_score:
            best_score = score
            best_i = idx

    return best_i

import csv


def fix(header, row, levers):
    """Recommend changes for an underperforming employee."""
    plus = [c for c in header if c.endswith('+')]
    minus = [c for c in header if c.endswith('-')]

    # Identify which columns are goals and which are inputs.
    goal_cols = plus + minus
    input_cols = [c for c in header if c not in goal_cols]

    # For each lever, pick the value that best improves the row's score.
    # We evaluate: for a candidate set of changes, compute the resulting score.
    # Since we can't easily simulate all combinations, use a greedy approach:
    # try each allowed value for each lever and pick the one that most improves the score.

    def eval_score(cells):
        s = 0.0
        for col in plus:
            idx = header.index(col)
            v = float(cells[idx])
            s += v
        for col in minus:
            idx = header.index(col)
            v = float(cells[idx])
            s -= v
        return s

    # Current score (with original row values, but we'll use levers to override).
    best_cells = list(row)
    best_score = eval_score(best_cells)

    for lever_col in levers:
        if lever_col not in input_cols and lever_col not in goal_cols:
            continue  # skip non-actionable columns

        allowed = levers[lever_col]
        is_numeric = isinstance(allowed, tuple)

        current_val = row[header.index(lever_col)]

        candidates = []
        if is_numeric:
            low, high = allowed
            # Try a few evenly spaced values between low and high.
            import numpy as np
            vals = np.linspace(low, high, 20)
            for v in vals:
                cells = list(row)
                idx = header.index(lever_col)
                cells[idx] = float(v)
                s = eval_score(cells)
                candidates.append((s, v))
        else:
            # Symbolic column: try each allowed value.
            for val in allowed:
                cells = list(row)
                idx = header.index(lever_col)
                cells[idx] = val
                s = eval_score(cells)
                candidates.append((s, val))

        if not candidates:
            continue

        # Pick the candidate with highest score.
        best_cand = max(candidates, key=lambda x: x[0])
        new_val = best_cand[1]

        # Only apply if it actually improves things (or is neutral).
        cells_test = list(row)
        idx = header.index(lever_col)
        cells_test[idx] = new_val
        test_score = eval_score(cells_test)

        if test_score > best_score:
            best_cells[idx] = new_val
            best_score = test_score

    return {col: val for col, val in zip(header, best_cells)}

def spot(header, rows):
    # Goals: '+' -> maximise, '-' -> minimise; everything else is input.
    plus = [c for c in header if c.endswith('+')]
    minus = [c for c in header if c.endswith('-')]
    input_cols = [c for c in header if c not in plus and c not in minus]

    n = len(rows)
    # Per-row suspicion score: number of suspicious cells.
    row_scores = {i: 0 for i in range(n)}

    # Known valid categories for symbolic columns (domain knowledge).
    VALID_DEPT = {'Sales', 'Research & Development', 'Operations',
                  'Human Resources', 'Engineering'}
    VALID_GENDER = {'Male', 'Female'}
    VALID_MARITAL = {'Single', 'Married', 'Divorced', 'Widowed'}
    VALID_ROLE = {'Laboratory Technician', 'Sales Executive',
                  'Sales Representative', 'Research Scientist',
                  'Manufacturing Director', 'Human Resources'}
    VALID_EDU = {'Other', 'Life Sciences', 'Medical',
                 'Marketing', 'Sales', 'Technical'}
    VALID_TRAVEL = {'Non-Travel', 'Travel_Rarely', 'Travel_Frequently'}

    for col in input_cols:
        idx = header.index(col)
        try:
            vals = [float(row[idx]) for row in rows]
            arr = __import__('numpy').array(vals)
            mean_v, std_v = float(arr.mean()), float(arr.std())

            if std_v < 1e-9:
                # Constant column — any deviation is wrong.
                target = mean_v
                for i, v in enumerate(rows):
                    if abs(float(v[idx]) - target) > 0.5:
                        row_scores[i] += 1
            else:
                threshold = 3.0 * std_v
                for i, v in enumerate(rows):
                    diff = abs(float(v[idx]) - mean_v)
                    if diff > threshold:
                        row_scores[i] += 1

        except (ValueError, TypeError):
            # Symbolic column — validate against known categories.
            vals = [row[idx] for row in rows]
            valid_set = None
            if col == 'department':
                valid_set = VALID_DEPT
            elif col == 'gender':
                valid_set = VALID_GENDER
            elif col == 'maritalStatus':
                valid_set = VALID_MARITAL
            elif col == 'jobRole':
                valid_set = VALID_ROLE
            elif col == 'educationField':
                valid_set = VALID_EDU
            elif col == 'businessTravel':
                valid_set = VALID_TRAVEL

            if valid_set is not None:
                for i, v in enumerate(vals):
                    if v.strip() not in valid_set:
                        row_scores[i] += 1

    # Return rows sorted by suspicion (most suspicious first).
    return [i for i in range(n) if row_scores[i] > 0], sorted(
        [i for i in range(n)], key=lambda i: -row_scores[i])
