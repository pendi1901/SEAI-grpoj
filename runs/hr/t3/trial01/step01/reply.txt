import numpy as np


def remember(header, rows):
    # Columns ending in '+' are goals to maximise (tenure-related metrics).
    plus_goals = [c for c in header if c.endswith('+')]

    scores = []
    for row in rows:
        score = 0.0
        for col in plus_goals:
            val = float(row[header.index(col)])
            score += val
        scores.append(score)

    # Return top rows (at most 5% of total, at least one).
    n_top = max(1, int(len(rows) * 0.05))
    best_indices = np.argsort(scores)[-n_top:][::-1]
    return sorted(best_indices.tolist())


if __name__ == "__main__":
    import csv

    with open("WA_Fn-UseC_-HR-Employee-Attrition.train.csv", "r") as f:
        reader = csv.reader(f)
        header = next(reader)
        rows = []
        for row in reader:
            parsed = []
            for cell in row:
                try:
                    parsed.append(float(cell))
                except ValueError:
                    parsed.append(cell)
            rows.append(parsed)

    result = remember(header, rows)
    print(f"Top {len(result)} employees with longest tenure:")
    for idx in result[:5]:
        age = int(rows[idx][header.index("Age")])
        years = float(rows[idx][header.index("YearsAtCompany+")])
        role = rows[idx][header.index("jobRole")]
        print(f"  Position {idx}: Age={age}, YearsAtCompany+={years:.1f}, Role={role}")
