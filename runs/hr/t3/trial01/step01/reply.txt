import csv

def remember(header, rows):
    """
    Identifies staff who stay 'unusually long'.
    We look for rows where 'YearsAtCompany+' is in the top percentile.
    'YearsAtCompany+' is at index 31.
    """
    target_idx = 31
    
    # Extract values for the goal column
    values = []
    for i, row in enumerate(rows):
        val = row[target_idx]
        values.append((val, i))
    
    # Sort by YearsAtCompany+ descending (higher is better for + goal)
    values.sort(key=lambda x: x[0], reverse=True)
    
    # Return top 5% of rows (800 rows * 0.05 = 40 rows)
    limit = max(1, int(len(rows) * 0.05))
    return [item[1] for item in values[:limit]]

if __name__ == "__main__":
    filename = 'WA_Fn-UseC_-HR-Employee-Attrition.n1000.train.csv'
    rows = []
    header = []
    
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            # Convert numeric columns (starting with uppercase) to floats
            processed_row = []
            for i, val in enumerate(row):
                if header[i][0].isupper():
                    processed_row.append(float(val))
                else:
                    processed_row.append(val)
            rows.append(processed_row)
            
    indices = remember(header, rows)
    
    print("Staff members who stay unusually long (Top 5% by YearsAtCompany+):")
    print(f"{'Index':<10} | {'YearsAtCompany':<15} | {'JobRole':<20}")
    print("-" * 50)
    
    # Indices for reference
    role_idx = header.index('jobRole')
    years_idx = header.index('YearsAtCompany+')
    
    for idx in indices:
        print(f"{idx:<10} | {rows[idx][years_idx]:<15} | {rows[idx][role_idx]}")
