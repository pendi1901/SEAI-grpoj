def flow(header, row, seen):
    plus_goals = [c for c in header if c.endswith('+')]

    if not seen:
        return False

    # Compute average of each '+' goal from previously seen rows
    goal_sums = {}
    for col in plus_goals:
        idx = header.index(col)
        total = 0.0
        count = 0
        for r in seen:
            val = float(r[idx])
            total += val
            count += 1
        goal_sums[col] = (total, count)

    # Evaluate current row's '+' goals against the running averages
    above_count = 0
    for col in plus_goals:
        avg_seen, _ = goal_sums[col]
        if float(row[header.index(col)]) > avg_seen:
            above_count += 1

    return above_count >= len(plus_goals) // 2 + 1


if __name__ == "__main__":
    import csv

    with open("WA_Fn-UseC_-HR-Employee-Attrition.train.csv", "r") as f:
        reader = csv.reader(f)
        header = next(reader)
        rows = []
        for row in reader:
            parsed = []
            for cell in row:
                try:
                    parsed.append(float(cell))
                except ValueError:
                    parsed.append(cell)
            rows.append(parsed)

    result = []
    seen = []
    for row in rows:
        keep = flow(header, row, seen)
        if keep:
            result.append(row)
        seen.append(row)

    print(f"Kept {len(result)} out of {len(rows)} employees as high-tenure candidates.")
