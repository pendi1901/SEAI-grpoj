```python
import csv
from sklearn.ensemble import RandomForestRegressor
from sklearn.preprocessing import LabelEncoder

def justify(header, rows, row):
    """
    Estimates the performance of a given row relative to the population.
    A row is 'better' if its goal columns (ending in +) are higher than others.
    """
    goal_indices = [i for i, h in enumerate(header) if h.endswith('+') or h.endswith('-')]
    input_indices = [i for i in range(len(header)) if i not in goal_indices]
    
    # Prepare data for training
    encoders = {}
    for i in input_indices:
        if not header[i][0].isupper():
            le = LabelEncoder()
            le.fit([str(r[i]) for r in rows])
            encoders[i] = le

    def transform(r):
        features = []
        for i in input_indices:
            val = r[i]
            if i in encoders:
                features.append(float(encoders[i].transform([str(val)])[0]))
            else:
                features.append(float(val))
        return features

    X = [transform(r) for r in rows]
    
    # We evaluate based on all goal columns. 
    # For simplicity, we focus on the first goal column found.
    target_idx = goal_indices[0]
    y = [float(r[target_idx]) for r in rows]
    
    model = RandomForestRegressor(n_estimators=50, random_state=42)
    model.fit(X, y)
    
    # Predict the value for the input row
    prediction = model.predict([transform(row)])[0]
    
    # Calculate how many rows are worse than this prediction
    # If goal ends in +, higher is better.
    better_than_count = sum(1 for val in y if val < prediction)
    p = better_than_count / len(rows)
    
    reason = f"This arrangement ranks in the top {int((1-p)*100)}% of all teams because its predicted {header[target_idx].strip('+')} is higher than {p:.1%} of the existing staff records."
    
    return p, reason

if __name__ == "__main__":
    filename = 'WA_Fn-UseC_-HR-Employee-Attrition.n1000.train.csv'
    rows = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = []
            for i, val in enumerate(row):
                if header[i][0].isupper():
                    processed_row.append(float(val))
                else:
                    processed_row.append(val)
            rows.append(processed_row)
    
    # Test with the first row, masking its goal columns
    test_row = list(rows[0])
    goal_indices = [i for i, h in enumerate(header) if h.endswith('+') or h.endswith('-')]
    for idx in goal_indices:
        test_row[idx] = "?"
        
    p, reason = justify(header, rows, test_row)
    print(f"Justification: {reason}")
```
