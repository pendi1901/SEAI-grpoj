import csv
from scipy import stats

def certify(xs, ys):
    """
    Returns True if the two groups are statistically similar (noise),
    False if they are significantly different.
    Uses a t-test to compare means.
    """
    t_stat, p_value = stats.ttest_ind(xs, ys, equal_var=False)
    # If p-value > 0.05, we fail to reject the null hypothesis (they are the same)
    return p_value > 0.05

def analyze_attrition(header, rows):
    """
    Compares 'YearsInCurrentRole+' for employees who work overtime vs those who don't.
    """
    idx_overtime = header.index('overTime')
    idx_years_role = header.index('YearsInCurrentRole+')
    
    overtime_yes = [row[idx_years_role] for row in rows if row[idx_overtime] == 'Yes']
    overtime_no = [row[idx_years_role] for row in rows if row[idx_overtime] == 'No']
    
    is_noise = certify(overtime_yes, overtime_no)
    
    return is_noise, overtime_yes, overtime_no

if __name__ == "__main__":
    filename = 'WA_Fn-UseC_-HR-Employee-Attrition.n1000.train.csv'
    rows = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = []
            for i, val in enumerate(row):
                if header[i][0].isupper():
                    processed_row.append(float(val))
                else:
                    processed_row.append(val)
            rows.append(processed_row)
    
    is_noise, yes_group, no_group = analyze_attrition(header, rows)
    
    print("Analysis: Does working overtime affect years in current role?")
    if is_noise:
        print("Result: The difference is statistically insignificant (noise).")
    else:
        avg_yes = sum(yes_group) / len(yes_group)
        avg_no = sum(no_group) / len(no_group)
        print(f"Result: The difference is real.")
        print(f"Average years in role (Overtime): {avg_yes:.2f}")
        print(f"Average years in role (No Overtime): {avg_no:.2f}")
