import csv
from sklearn.ensemble import RandomForestRegressor
from sklearn.preprocessing import LabelEncoder

def guess(header, train, test):
    """
    Estimates the first goal column ('YearsAtCompany+') for test rows
    using a Random Forest Regressor based on input features.
    """
    # Identify goal column (first one ending in + or -)
    goal_idx = -1
    for i, name in enumerate(header):
        if name.endswith('+') or name.endswith('-'):
            goal_idx = i
            break
    
    # Prepare data
    # We use all columns except the goal columns as features
    feature_indices = [i for i, name in enumerate(header) 
                       if not (name.endswith('+') or name.endswith('-'))]
    
    # Encode symbolic data
    encoders = {}
    def encode_data(data_rows):
        encoded = []
        for row in data_rows:
            new_row = []
            for i in feature_indices:
                val = row[i]
                if isinstance(val, str):
                    if i not in encoders:
                        encoders[i] = LabelEncoder()
                        # Fit on all available data to ensure consistency
                        all_vals = [r[i] for r in train] + [r[i] for r in test]
                        encoders[i].fit(all_vals)
                    new_row.append(encoders[i].transform([val])[0])
                else:
                    new_row.append(val)
            encoded.append(new_row)
        return encoded

    X_train = encode_data(train)
    y_train = [row[goal_idx] for row in train]
    X_test = encode_data(test)
    
    # Train model
    model = RandomForestRegressor(n_estimators=100, random_state=42)
    model.fit(X_train, y_train)
    
    # Predict
    return model.predict(X_test).tolist()

if __name__ == "__main__":
    filename = 'WA_Fn-UseC_-HR-Employee-Attrition.n1000.train.csv'
    rows = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = []
            for i, val in enumerate(row):
                if header[i][0].isupper():
                    processed_row.append(float(val))
                else:
                    processed_row.append(val)
            rows.append(processed_row)
    
    # Split data: use 80% for training, 20% as a test set
    split = int(len(rows) * 0.8)
    train_data = rows[:split]
    test_data = rows[split:]
    
    # Mask goals in test data
    goal_indices = [i for i, name in enumerate(header) if name.endswith('+') or name.endswith('-')]
    masked_test = []
    for row in test_data:
        new_row = list(row)
        for idx in goal_indices:
            new_row[idx] = "?"
        masked_test.append(new_row)
    
    predictions = guess(header, train_data, masked_test)
    
    print("Estimation of 'YearsAtCompany+' for the first 5 test cases:")
    for i in range(min(5, len(predictions))):
        print(f"Test Row {i}: Predicted {predictions[i]:.2f} years")
