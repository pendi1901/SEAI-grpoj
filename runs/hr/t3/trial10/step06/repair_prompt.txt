def guess(header, train, test):
    """
    Estimates the value of the first goal column (YearsAtCompany+)
    using a Random Forest Regressor.
    """
    # Identify the first goal column (ends with + or -)
    goal_cols = [i for i, h in enumerate(header) if h.endswith('+') or h.endswith('-')]
    target_idx = goal_cols[0]
    
    # Prepare encoders for symbolic columns
    encoders = {}
    for i, col_name in enumerate(header):
        if i != target_idx and not header[i][0].isupper():
            encoders[i] = LabelEncoder()
            all_vals = [r[i] for r in train]
            encoders[i].fit(all_vals)

    def transform_row(row, is_test=False):
        new_row = []
        for i, val in enumerate(row):
            if i == target_idx and is_test:
                # Placeholder for the goal column in test data
                new_row.append(0.0)
            elif i in encoders:
                # Handle potential unseen labels in test data by mapping to 0
                try:
                    new_row.append(float(encoders[i].transform([val])[0]))
                except ValueError:
                    new_row.append(0.0)
            else:
                new_row.append(float(val))
        return new_row

    # Prepare training features and target
    X_train = [transform_row(r) for r in train]
    y_train = [float(r[target_idx]) for r in train]
    
    # Prepare test features
    X_test = [transform_row(r, is_test=True) for r in test]
    
    # Train model
    model = RandomForestRegressor(n_estimators=100, random_state=42)
    model.fit(X_train, y_train)
    
    # Predict
    return model.predict(X_test).tolist()

import csv
from scipy import stats

def certify(xs, ys):
    """
    Performs a Welch's t-test to determine if two samples are significantly different.
    Returns True if the difference is not statistically significant (p > 0.05),
    suggesting the difference is likely noise.
    """
    t_stat, p_value = stats.ttest_ind(xs, ys, equal_var=False)
    return p_value > 0.05

def analyze_overtime_impact(header, rows):
    """
    Compares 'YearsInCurrentRole+' for employees who work overtime vs those who don't.
    """
    ot_idx = header.index('overTime')
    role_idx = header.index('YearsInCurrentRole+')
    
    overtime_group = [r[role_idx] for r in rows if r[ot_idx] == 'Yes']
    no_overtime_group = [r[role_idx] for r in rows if r[ot_idx] == 'No']
    
    is_noise = certify(overtime_group, no_overtime_group)
    
    avg_ot = sum(overtime_group) / len(overtime_group)
    avg_no_ot = sum(no_overtime_group) / len(no_overtime_group)
    
    return is_noise, avg_ot, avg_no_ot

import csv

def flow(header, row, seen):
    """
    Decides if the current employee needs a 'what would keep you here' conversation.
    We prioritize employees who have high 'YearsAtCompany+' and 'YearsInCurrentRole+'
    but are currently experiencing 'overTime' (Yes), as these are high-value 
    employees at risk of burnout.
    
    We use a simple thresholding strategy based on the distribution of the 
    seen data to decide if the current row is in the top 10% of value.
    """
    # Identify indices
    try:
        years_idx = header.index('YearsAtCompany+')
        role_idx = header.index('YearsInCurrentRole+')
        ot_idx = header.index('overTime')
    except ValueError:
        return False

    # If this is the first row, we can't compare, so wait
    if not seen:
        return False

    # Calculate a simple "retention score"
    # Higher years = higher value. Overtime = higher risk.
    def get_score(r):
        score = r[years_idx] + r[role_idx]
        if r[ot_idx] == 'Yes':
            score += 5  # Bonus for being a high-value employee working overtime
        return score

    current_score = get_score(row)
    seen_scores = [get_score(r) for r in seen]
    
    # Threshold: Is this employee in the top 10% of the seen population?
    threshold = sorted(seen_scores, reverse=True)[max(0, len(seen_scores) // 10)]
    
    return current_score >= threshold

import csv

def blame(header, rows):
    """
    Identifies input columns that correlate with short stays (low 'YearsAtCompany+').
    We define 'worst rows' as the bottom 20% by 'YearsAtCompany+'.
    We then identify input columns that show the highest variance or 
    distinctive values in this group compared to the rest.
    """
    years_idx = header.index('YearsAtCompany+')
    
    # Sort rows by YearsAtCompany+ to find the worst performers
    sorted_rows = sorted(rows, key=lambda x: x[years_idx])
    worst_rows = sorted_rows[:len(rows) // 5]
    
    # Identify input columns (exclude goals)
    input_cols = [i for i, h in enumerate(header) if not (h.endswith('+') or h.endswith('-'))]
    
    # Simple heuristic: find columns where the distribution in 'worst' rows 
    # is most distinct from the global average.
    # For symbolic, we look for high frequency of specific categories.
    # For numeric, we look for low values.
    scores = []
    for i in input_cols:
        col_name = header[i]
        vals = [r[i] for r in worst_rows]
        
        if isinstance(vals[0], float):
            # For numeric, low values might explain short stays
            avg = sum(vals) / len(vals)
            scores.append((avg, col_name))
        else:
            # For symbolic, count the most frequent value
            counts = {}
            for v in vals:
                counts[v] = counts.get(v, 0) + 1
            top_val = max(counts, key=counts.get)
            # Score based on how dominant this value is in the worst group
            scores.append((counts[top_val] / len(vals), col_name))
            
    # Return the top 3 columns that seem to characterize the short-stay group
    # We prioritize low numeric values and high categorical frequency
    scores.sort()
    return [s[1] for s in scores[:3]]
