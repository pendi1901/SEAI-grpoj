import csv
from sklearn.ensemble import RandomForestRegressor
from sklearn.preprocessing import LabelEncoder

def guess(header, train, test):
    """
    Estimates the value of the first goal column (YearsAtCompany+)
    using a Random Forest Regressor.
    """
    # Identify the first goal column (ends with + or -)
    goal_cols = [i for i, h in enumerate(header) if h.endswith('+') or h.endswith('-')]
    target_idx = goal_cols[0]
    
    # Prepare data: convert symbolic columns to numeric for sklearn
    # We use LabelEncoder for all symbolic columns
    encoders = {}
    
    def transform_data(data_rows, is_train=True):
        processed = []
        for row in data_rows:
            new_row = []
            for i, val in enumerate(row):
                if isinstance(val, str):
                    if i not in encoders:
                        encoders[i] = LabelEncoder()
                        # Fit on all unique values found in train
                        all_vals = [r[i] for r in train]
                        encoders[i].fit(all_vals)
                    new_row.append(encoders[i].transform([val])[0])
                else:
                    new_row.append(val)
            processed.append(new_row)
        return processed

    # Prepare training features and target
    train_processed = transform_data(train)
    X_train = [r[:target_idx] + r[target_idx+1:] for r in train_processed]
    y_train = [r[target_idx] for r in train_processed]
    
    # Prepare test features
    test_processed = transform_data(test, is_train=False)
    X_test = [r[:target_idx] + r[target_idx+1:] for r in test_processed]
    
    # Train model
    model = RandomForestRegressor(n_estimators=100, random_state=42)
    model.fit(X_train, y_train)
    
    # Predict
    return model.predict(X_test).tolist()

if __name__ == "__main__":
    filename = 'WA_Fn-UseC_-HR-Employee-Attrition.n1000.train.csv'
    rows = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = []
            for i, val in enumerate(row):
                if header[i][0].isupper():
                    processed_row.append(float(val))
                else:
                    processed_row.append(val)
            rows.append(processed_row)
    
    # Split data for demonstration: use last 10 rows as 'test'
    train_data = rows[:-10]
    test_data = [r[:] for r in rows[-10:]]
    
    # Mask the goal column in test data
    goal_idx = next(i for i, h in enumerate(header) if h.endswith('+'))
    for row in test_data:
        row[goal_idx] = "?"
        
    predictions = guess(header, train_data, test_data)
    
    print("Estimates for 'YearsAtCompany+' for the last 10 employees:")
    for i, pred in enumerate(predictions):
        print(f"Employee {i+1}: Predicted {pred:.2f} years")
