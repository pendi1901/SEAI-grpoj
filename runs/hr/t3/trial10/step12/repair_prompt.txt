import csv
from scipy import stats

def certify(xs, ys):
    """
    Performs a Welch's t-test to determine if two samples are significantly different.
    Returns True if the difference is not statistically significant (p > 0.05),
    suggesting the difference is likely noise.
    """
    t_stat, p_value = stats.ttest_ind(xs, ys, equal_var=False)
    return p_value > 0.05

def analyze_overtime_impact(header, rows):
    """
    Compares 'YearsInCurrentRole+' for employees who work overtime vs those who don't.
    """
    ot_idx = header.index('overTime')
    role_idx = header.index('YearsInCurrentRole+')
    
    overtime_group = [r[role_idx] for r in rows if r[ot_idx] == 'Yes']
    no_overtime_group = [r[role_idx] for r in rows if r[ot_idx] == 'No']
    
    is_noise = certify(overtime_group, no_overtime_group)
    
    avg_ot = sum(overtime_group) / len(overtime_group)
    avg_no_ot = sum(no_overtime_group) / len(no_overtime_group)
    
    return is_noise, avg_ot, avg_no_ot

import csv

def flow(header, row, seen):
    """
    Decides if the current employee needs a 'what would keep you here' conversation.
    We prioritize employees who have high 'YearsAtCompany+' and 'YearsInCurrentRole+'
    but are currently experiencing 'overTime' (Yes), as these are high-value 
    employees at risk of burnout.
    
    We use a simple thresholding strategy based on the distribution of the 
    seen data to decide if the current row is in the top 10% of value.
    """
    # Identify indices
    try:
        years_idx = header.index('YearsAtCompany+')
        role_idx = header.index('YearsInCurrentRole+')
        ot_idx = header.index('overTime')
    except ValueError:
        return False

    # If this is the first row, we can't compare, so wait
    if not seen:
        return False

    # Calculate a simple "retention score"
    # Higher years = higher value. Overtime = higher risk.
    def get_score(r):
        score = r[years_idx] + r[role_idx]
        if r[ot_idx] == 'Yes':
            score += 5  # Bonus for being a high-value employee working overtime
        return score

    current_score = get_score(row)
    seen_scores = [get_score(r) for r in seen]
    
    # Threshold: Is this employee in the top 10% of the seen population?
    threshold = sorted(seen_scores, reverse=True)[max(0, len(seen_scores) // 10)]
    
    return current_score >= threshold

import csv

def blame(header, rows):
    """
    Identifies input columns that correlate with short stays (low 'YearsAtCompany+').
    We define 'worst rows' as the bottom 20% by 'YearsAtCompany+'.
    We then identify input columns that show the highest variance or 
    distinctive values in this group compared to the rest.
    """
    years_idx = header.index('YearsAtCompany+')
    
    # Sort rows by YearsAtCompany+ to find the worst performers
    sorted_rows = sorted(rows, key=lambda x: x[years_idx])
    worst_rows = sorted_rows[:len(rows) // 5]
    
    # Identify input columns (exclude goals)
    input_cols = [i for i, h in enumerate(header) if not (h.endswith('+') or h.endswith('-'))]
    
    # Simple heuristic: find columns where the distribution in 'worst' rows 
    # is most distinct from the global average.
    # For symbolic, we look for high frequency of specific categories.
    # For numeric, we look for low values.
    scores = []
    for i in input_cols:
        col_name = header[i]
        vals = [r[i] for r in worst_rows]
        
        if isinstance(vals[0], float):
            # For numeric, low values might explain short stays
            avg = sum(vals) / len(vals)
            scores.append((avg, col_name))
        else:
            # For symbolic, count the most frequent value
            counts = {}
            for v in vals:
                counts[v] = counts.get(v, 0) + 1
            top_val = max(counts, key=counts.get)
            # Score based on how dominant this value is in the worst group
            scores.append((counts[top_val] / len(vals), col_name))
            
    # Return the top 3 columns that seem to characterize the short-stay group
    # We prioritize low numeric values and high categorical frequency
    scores.sort()
    return [s[1] for s in scores[:3]]

def justify(header, rows, row):
    """
    Estimates the performance of a given row relative to the dataset.
    Returns (p, reason) where p is the percentile rank and reason is the explanation.
    """
    # Identify goal columns
    goal_indices = [i for i, h in enumerate(header) if h.endswith('+') or h.endswith('-')]
    target_idx = goal_indices[0]
    
    # Prepare encoders for symbolic columns
    encoders = {}
    for i, h in enumerate(header):
        if not h[0].isupper() and i not in goal_indices:
            encoders[i] = LabelEncoder()
            encoders[i].fit([str(r[i]) for r in rows])

    def process_row(r, is_input=False):
        features = []
        for i, val in enumerate(r):
            if i == target_idx and is_input:
                # Use a dummy value for the goal column during prediction
                features.append(0.0)
            elif i in encoders:
                features.append(float(encoders[i].transform([str(val)])[0]))
            else:
                features.append(float(val))
        return features

    # Train a model to predict the first goal (YearsAtCompany+)
    X = [process_row(r) for r in rows]
    y = [float(r[target_idx]) for r in rows]
    
    model = RandomForestRegressor(n_estimators=50, random_state=42)
    model.fit(X, y)
    
    # Predict the value for the input row
    input_features = process_row(row, is_input=True)
    prediction = model.predict([input_features])[0]
    
    # Calculate how many rows are worse than this prediction
    # Since YearsAtCompany+ is a "maximize" goal, higher is better
    better_than_count = sum(1 for r in rows if float(r[target_idx]) < prediction)
    p = better_than_count / len(rows)
    
    reason = (f"This employee's estimated tenure of {prediction:.1f} years ranks them "
              f"better than {p:.2%} of the staff in the company.")
    
    return p, reason

import csv

def choose(header, rows, label):
    """
    Finds the working arrangement that keeps people longest by sampling 50 rows.
    'YearsAtCompany+' is the goal to maximize.

    work_cols = ['businessTravel', 'department', 'jobRole', 'overTime']
    work_indices = [header.index(c) for c in work_cols]
    
    # Randomly sample 50 indices to label
    sample_size = min(50, len(rows))
    indices_to_sample = random.sample(range(len(rows)), sample_size)
    
    best_val = -1.0
    best_idx = -1
    
    # Track performance by arrangement
    arrangement_performance = {}
    
    for i in indices_to_sample:
        # label(i) returns the full row (as per instructions, 
        # the harness provides the full row including the goal value)
        full_row = label(i)
        val = float(full_row[goal_idx])

        if arrangement not in arrangement_performance:
            arrangement_performance[arrangement] = []
        arrangement_performance[arrangement].append(val)
        
        if val > best_val:
            best_val = val
            best_idx = i
            
    return best_idx

    We prioritize increasing pay-related metrics and reducing overtime.
    """

    # Map indices for easier access
    idx_map = {h: i for i, h in enumerate(header)}
    
    # Strategy:
    # 1. If 'overTime' is 'Yes', change to 'No' (if allowed)
    if 'overTime' in levers and row[idx_map['overTime']] == 'Yes':
        if 'No' in levers['overTime']:
            recommendations['overTime'] = 'No'

    if 'PercentSalaryHike' in levers:

        first_key = list(levers.keys())[0]
        val = levers[first_key]
        recommendations[first_key] = val[1] if isinstance(val, tuple) else val[0]

import csv

def spot(header, rows):
    """
    Identifies suspicious rows by checking for logical inconsistencies in 
    tenure-related columns. Specifically, we look for rows where 
    'YearsInCurrentRole+' or 'YearsWithCurrManager+' exceed 'YearsAtCompany+'.
    These are physically impossible and suggest data entry errors.
    """
    idx_at = header.index('YearsAtCompany+')
    idx_role = header.index('YearsInCurrentRole+')
    idx_mgr = header.index('YearsWithCurrManager+')
    
    suspicious = []
    
    for i, row in enumerate(rows):
        # Skip rows with missing data
        if row[idx_at] == "?" or row[idx_role] == "?" or row[idx_mgr] == "?":
            continue
            
        # Logic check: Years in role or with manager cannot exceed total years at company
        # We assign a "suspicion score" based on the magnitude of the violation
        violation_score = 0
        if row[idx_role] > row[idx_at]:
            violation_score += (row[idx_role] - row[idx_at])
        if row[idx_mgr] > row[idx_at]:
            violation_score += (row[idx_mgr] - row[idx_at])
            
        if violation_score > 0:
            suspicious.append((violation_score, i))
            
    # Sort by suspicion score descending
    suspicious.sort(key=lambda x: x[0], reverse=True)
    
    return [item[1] for item in suspicious]

import csv
import random

def race(header, rows, label):
    """
    Compares three strategies for finding the best employee (maximizing 'YearsAtCompany+').
    Budget: 50 labels total.
    """
    # Identify the goal column index
    goal_idx = next(i for i, h in enumerate(header) if h.endswith('+'))
    
    # Strategies:
    # 1. random: Pick 16 random rows.
    # 2. tenure_focus: Pick 17 rows with highest 'TotalWorkingYears'.
    # 3. satisfaction_focus: Pick 17 rows with highest 'JobSatisfaction'.
    
    strategies = ["random", "tenure_focus", "satisfaction_focus"]
    indices = range(len(rows))
    
    # Pre-calculate indices for strategies
    tenure_idx = header.index('TotalWorkingYears')
    tenure_sorted = sorted(indices, key=lambda i: rows[i][tenure_idx], reverse=True)
    
    sat_idx = header.index('JobSatisfaction')
    sat_sorted = sorted(indices, key=lambda i: rows[i][sat_idx], reverse=True)
    
    selection = {
        "random": random.sample(indices, 16),
        "tenure_focus": tenure_sorted[:17],
        "satisfaction_focus": sat_sorted[:17]
    }
    
    best_overall_val = -float('inf')
    best_overall_idx = -1
    strategy_bests = {name: -float('inf') for name in strategies}
    
    # Evaluate
    for name in strategies:
        for idx in selection[name]:
            full_row = label(idx)
            val = float(full_row[goal_idx])
            
            if val > strategy_bests[name]:
                strategy_bests[name] = val
            
            if val > best_overall_val:
                best_overall_val = val
                best_overall_idx = idx
    
    # Determine winner based on which strategy found the highest value
    winner = max(strategy_bests, key=strategy_bests.get)
                
    return {"winner": winner, "row": best_overall_idx, "tried": strategies}

import csv
import math

def remember(header, rows):
    """
    Finds the 5% of employees most similar to the first employee in the dataset.
    Similarity is calculated using Euclidean distance on normalized numeric features.
    """
    # Identify numeric columns for distance calculation
    numeric_indices = [i for i, h in enumerate(header) if h[0].isupper()]
    
    # Normalize numeric columns to [0, 1] range
    normalized_rows = []
    for row in rows:
        norm_row = [0.0] * len(row)
        for i in numeric_indices:
            norm_row[i] = float(row[i])
        normalized_rows.append(norm_row)
        
    for i in numeric_indices:
        col_vals = [r[i] for r in normalized_rows]
        min_v, max_v = min(col_vals), max(col_vals)
        diff = max_v - min_v
        if diff > 0:
            for r in normalized_rows:
                r[i] = (r[i] - min_v) / diff
        else:
            for r in normalized_rows:
                r[i] = 0.0

    # Target: First employee
    target = normalized_rows[0]
    
    # Calculate Euclidean distance to all other rows
    distances = []
    for idx, row in enumerate(normalized_rows):
        dist = math.sqrt(sum((row[i] - target[i])**2 for i in numeric_indices))
        distances.append((dist, idx))
    
    # Sort by distance (ascending)
    distances.sort(key=lambda x: x[0])
    
    # Return top 5% (excluding the target itself at index 0)
    limit = max(1, len(rows) // 20)
    return [item[1] for item in distances[1:limit+1]]
