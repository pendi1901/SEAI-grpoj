import csv

def remember(header, rows):
    """
    Identifies staff who stay 'unusually long'.
    We focus on 'YearsAtCompany+' (maximize) and 'YearsWithCurrManager+' (maximize).
    We calculate a score based on these two columns and return the top 5% indices.
    """
    # Identify indices for goal columns
    idx_years_at_company = header.index('YearsAtCompany+')
    idx_years_with_manager = header.index('YearsWithCurrManager+')
    
    # Calculate a composite score for each row
    # We want to maximize these values
    scores = []
    for i, row in enumerate(rows):
        score = row[idx_years_at_company] + row[idx_years_with_manager]
        scores.append((score, i))
    
    # Sort by score descending
    scores.sort(key=lambda x: x[0], reverse=True)
    
    # Return top 5% (800 rows * 0.05 = 40 rows)
    limit = max(1, int(len(rows) * 0.05))
    return [item[1] for item in scores[:limit]]

if __name__ == "__main__":
    filename = 'WA_Fn-UseC_-HR-Employee-Attrition.n1000.train.csv'
    rows = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            # Convert numeric columns (starting with uppercase) to float
            processed_row = []
            for i, val in enumerate(row):
                if header[i][0].isupper():
                    processed_row.append(float(val))
                else:
                    processed_row.append(val)
            rows.append(processed_row)
    
    best_indices = remember(header, rows)
    
    print("Staff members who stay unusually long (Top 5% by tenure):")
    print(f"{'Index':<10} | {'YearsAtCompany':<15} | {'YearsWithManager':<15}")
    print("-" * 45)
    for idx in best_indices:
        row = rows[idx]
        print(f"{idx:<10} | {row[header.index('YearsAtCompany+')]:<15} | {row[header.index('YearsWithCurrManager+')]:<15}")
