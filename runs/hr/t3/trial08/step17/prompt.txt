def guess(header, train, test):
    """
    Estimates the value of the first goal column (YearsAtCompany+) 
    for test rows using a Random Forest Regressor.
    """
    # Identify goal column (first column ending in '+' or '-')
    goal_idx = -1
    for i, name in enumerate(header):
        if name.endswith('+') or name.endswith('-'):
            goal_idx = i
            break
    
    # Identify symbolic columns
    symbolic_indices = [i for i, name in enumerate(header) if not name[0].isupper()]
    
    # Fit encoders on all available data
    encoders = {}
    for i in symbolic_indices:
        encoders[i] = LabelEncoder()
        all_vals = [row[i] for row in train] + [row[i] for row in test if row[i] != '?']
        encoders[i].fit(all_vals)
            
    def transform_row(row, is_test=False):
        new_row = []
        for i, val in enumerate(row):
            if i == goal_idx:
                continue
            if i in symbolic_indices:
                new_row.append(float(encoders[i].transform([val])[0]))
            else:
                # If it's a test row and the value is '?', we shouldn't be using it 
                # as an input feature anyway, but here we ensure numeric conversion
                new_row.append(float(val))
        return new_row

    X_train = [transform_row(row) for row in train]
    y_train = [float(row[goal_idx]) for row in train]
    X_test = [transform_row(row) for row in test]
        
    # Train model
    model = RandomForestRegressor(n_estimators=100, random_state=42)
    model.fit(X_train, y_train)
    
    # Predict
    return model.predict(X_test).tolist()

from scipy import stats

def certify(xs, ys):
    """
    Returns True if the two groups are statistically similar (p > 0.05),
    False if they differ significantly (p <= 0.05).
    """
    t_stat, p_value = stats.ttest_ind(xs, ys, equal_var=False)
    return p_value > 0.05

def analyze_overtime_impact(header, rows):

    Compares 'YearsInCurrentRole+' for employees who work overtime vs those who don't.

import csv

def flow(header, row, seen):
    """
    Decides if the current employee needs a 'what would keep you here' conversation.
    We target employees who are high-performers (YearsAtCompany+ is high) 
    but might be at risk (e.g., low JobSatisfaction or high OverTime).
    
    Strategy: Use a simple threshold based on the distribution of seen data.
    If the current row's 'YearsAtCompany+' is in the top 10% of what we've seen
    AND they have low JobSatisfaction (<= 2), flag them for a conversation.
    """
    years_idx = header.index('YearsAtCompany+')
    sat_idx = header.index('JobSatisfaction')
    
    current_years = row[years_idx]
    current_sat = row[sat_idx]
    
    # If this is the first row, we can't compare, so let it go
    if not seen:
        return False
    
    # Get all seen years to determine the threshold
    seen_years = [r[years_idx] for r in seen]
    threshold = sorted(seen_years, reverse=True)[max(0, len(seen_years) // 10)]
    
    # Criteria: High tenure (top 10%) + Low satisfaction = High risk of attrition
    if current_years >= threshold and current_sat <= 2:
        return True
    
    return False

import csv

def blame(header, rows):
    """
    Identifies input columns that correlate with short stays (low YearsAtCompany+).
    We define 'worst rows' as those in the bottom 25% of 'YearsAtCompany+'.
    We then identify which input columns have the strongest correlation with 
    these low values.
    """
    years_idx = header.index('YearsAtCompany+')
    
    # Filter out goal columns
    input_indices = [
        i for i, name in enumerate(header) 
        if not (name.endswith('+') or name.endswith('-'))
    ]
    
    # Sort rows by YearsAtCompany+ ascending (lower is worse)
    sorted_rows = sorted(rows, key=lambda r: r[years_idx])
    
    # Define 'worst' as the bottom 25%
    limit = len(rows) // 4
    worst_rows = sorted_rows[:limit]
    
    # Calculate average for each input column in the worst group
    # and compare to the global average to find "drivers" of short stays
    scores = []
    for i in input_indices:
        col_data = [r[i] for r in rows if isinstance(r[i], (int, float))]
        if not col_data:
            continue
            
        global_avg = sum(col_data) / len(col_data)
        worst_avg = sum(r[i] for r in worst_rows if isinstance(r[i], (int, float))) / len(worst_rows)
        
        # We look for features where the worst group deviates significantly from the mean
        # For numeric, we look at the absolute difference
        diff = abs(worst_avg - global_avg)
        scores.append((diff, header[i]))
    
    # Sort by difference and return the top 3 column names
    scores.sort(key=lambda x: x[0], reverse=True)
    return [name for diff, name in scores[:3]]

def justify(header, rows, row):
    """
    Estimates the performance of a given row relative to the dataset.
    A row is 'better' if its goal columns (ending in +) are higher than others.
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith('+') or name.endswith('-')]
    symbolic_indices = [i for i, name in enumerate(header) if not name[0].isupper()]
    
    # Prepare data for regression
    X = []
    y = []
    # We focus on the first goal column for ranking as per previous steps
    target_idx = goal_indices[0]
    
    encoders = {}
    for i in symbolic_indices:
        encoders[i] = LabelEncoder()
        all_vals = [r[i] for r in rows]
        encoders[i].fit(all_vals)

    def process(r):
        features = []
        for i, val in enumerate(r):
            if i in goal_indices: continue
            if i in symbolic_indices:
                features.append(float(encoders[i].transform([val])[0]))
            else:
                features.append(float(val))
        return features

    for r in rows:
        X.append(process(r))
        y.append(float(r[target_idx]))

    model = RandomForestRegressor(n_estimators=50, random_state=42)
    model.fit(X, y)
    
    # Predict the value for the input row
    pred = model.predict([process(row)])[0]
    
    # Calculate fraction of rows that this row is better than
    # Higher value is better for '+' columns
    better_count = sum(1 for r in rows if pred > r[target_idx])
    p = better_count / len(rows)
    
    reason = f"This arrangement ranks in the top {int((1-p)*100)}% of all teams because its predicted tenure is {pred:.1f} years, which outperforms most other configurations in our records."
    
    return p, reason

import csv
import random

def choose(header, rows, label):
    """
    Finds the best working arrangement by sampling 50 rows, 
    labeling them to find their true goal values, and picking the best.
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith('+') or name.endswith('-')]
    
    # We want to maximize columns ending in '+' and minimize columns ending in '-'
    def score_row(goals):
        score = 0
        for i, val in enumerate(goals):
            name = header[goal_indices[i]]
            if name.endswith('+'):
                score += val
            else:
                score -= val
        return score

    # Randomly sample 50 indices to label
    indices = list(range(len(rows)))
    sample_indices = random.sample(indices, min(50, len(rows)))
    
    best_idx = -1
    best_score = float('-inf')
    
    for idx in sample_indices:
        # Get true values for the goals
        true_goals = label(idx)
        current_score = score_row(true_goals)
        
        if current_score > best_score:
            best_score = current_score
            best_idx = idx
            
    return best_idx

import csv

def fix(header, row, levers):
    """
    Recommends changes to a 'bad' row (short tenure) to improve it.
    We prioritize increasing pay (PercentSalaryHike), increasing stock options,
    and adjusting work-life balance/overtime.
    """
    recommendations = {}
    
    # Identify indices for levers
    lever_indices = {name: header.index(name) for name in levers if name in header}
    
    # Strategy:
    # 1. If OverTime is 'Yes', change to 'No' (often associated with better retention)
    if 'overTime' in lever_indices:
        idx = lever_indices['overTime']
        if row[idx] == 'Yes':
            recommendations['overTime'] = 'No'
            
    # 2. If PercentSalaryHike is low, push to the high end of the allowed range
    if 'PercentSalaryHike' in lever_indices:
        idx = lever_indices['PercentSalaryHike']
        low, high = levers['PercentSalaryHike']
        if row[idx] < high:
            recommendations['PercentSalaryHike'] = high
            
    # 3. If StockOptionLevel is low, push to the high end
    if 'StockOptionLevel' in lever_indices:
        idx = lever_indices['StockOptionLevel']
        low, high = levers['StockOptionLevel']
        if row[idx] < high:
            recommendations['StockOptionLevel'] = high
            
    # 4. If TrainingTimesLastYear is low, increase it
    if 'TrainingTimesLastYear' in lever_indices:
        idx = lever_indices['TrainingTimesLastYear']
        low, high = levers['TrainingTimesLastYear']
        if row[idx] < high:
            recommendations['TrainingTimesLastYear'] = high

    # Ensure at least one change is made if possible
    if not recommendations and levers:
        # Fallback: pick the first lever and set to its max/first option
        first_key = list(levers.keys())[0]
        val = levers[first_key]
        recommendations[first_key] = val[1] if isinstance(val, tuple) else val[0]
        
    return recommendations

import csv

def spot(header, rows):
    """
    Identifies suspicious rows by checking for logical inconsistencies 
    between related columns (e.g., YearsAtCompany vs TotalWorkingYears).
    """
    # Indices for columns that should have logical relationships
    try:
        years_at_co_idx = header.index('YearsAtCompany+')
        total_years_idx = header.index('TotalWorkingYears')
        years_in_role_idx = header.index('YearsInCurrentRole+')
        years_with_mgr_idx = header.index('YearsWithCurrManager+')
    except ValueError:
        return []

    suspicious = []
    
    for i, row in enumerate(rows):
        # Rule 1: Years at company cannot exceed total working years
        # Rule 2: Years in current role cannot exceed years at company
        # Rule 3: Years with manager cannot exceed years at company
        
        score = 0
        
        # Check numeric consistency
        if row[years_at_co_idx] > row[total_years_idx]:
            score += 3
        if row[years_in_role_idx] > row[years_at_co_idx]:
            score += 2
        if row[years_with_mgr_idx] > row[years_at_co_idx]:
            score += 2
            
        # Check for impossible values (e.g., negative years)
        if any(row[idx] < 0 for idx in [years_at_co_idx, total_years_idx, years_in_role_idx]):
            score += 5
            
        if score > 0:
            suspicious.append((score, i))
            
    # Sort by score descending (most suspicious first)
    suspicious.sort(key=lambda x: x[0], reverse=True)
    
    return [item[1] for item in suspicious]

import csv
import random

def race(header, rows, label):
    """
    Compares three strategies to find the best employee row based on goal columns.
    Budget: 50 labels total.
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith('+') or name.endswith('-')]
    
    def get_score(row_idx):
        goals = label(row_idx)
        score = 0
        for i, val in enumerate(goals):
            name = header[goal_indices[i]]
            if name.endswith('+'):
                score += val
            else:
                score -= val
        return score

    # Strategies
    # 1. Random: Picks 16 random rows
    # 2. Tenure-based: Picks 17 rows with highest 'TotalWorkingYears'
    # 3. Satisfaction-based: Picks 17 rows with highest 'JobSatisfaction'
    
    strategies = ["random", "tenure", "satisfaction"]
    results = {}
    
    # Strategy 1: Random
    random_indices = random.sample(range(len(rows)), 16)
    best_random_idx = -1
    best_random_score = float('-inf')
    for idx in random_indices:
        s = get_score(idx)
        if s > best_random_score:
            best_random_score = s
            best_random_idx = idx
    results["random"] = (best_random_idx, best_random_score)
    
    # Strategy 2: Tenure (TotalWorkingYears)
    twy_idx = header.index('TotalWorkingYears')
    tenure_sorted = sorted(range(len(rows)), key=lambda i: rows[i][twy_idx], reverse=True)
    best_tenure_idx = -1
    best_tenure_score = float('-inf')
    for idx in tenure_sorted[:17]:
        s = get_score(idx)
        if s > best_tenure_score:
            best_tenure_score = s
            best_tenure_idx = idx
    results["tenure"] = (best_tenure_idx, best_tenure_score)
    
    # Strategy 3: Satisfaction (JobSatisfaction)
    js_idx = header.index('JobSatisfaction')
    sat_sorted = sorted(range(len(rows)), key=lambda i: rows[i][js_idx], reverse=True)
    best_sat_idx = -1
    best_sat_score = float('-inf')
    for idx in sat_sorted[:17]:
        s = get_score(idx)
        if s > best_sat_score:
            best_sat_score = s
            best_sat_idx = idx
    results["satisfaction"] = (best_sat_idx, best_sat_score)
    
    # Determine winner
    winner = max(results, key=lambda k: results[k][1])
    
    return {
        "winner": winner,
        "row": results[winner][0],
        "tried": strategies
    }

import csv
import math

def remember(header, rows):
    """
    Finds the 5% of employees most similar to the first employee in the dataset.
    Similarity is calculated using Euclidean distance on normalized numeric features.
    """
    # Identify numeric columns (excluding goals)
    goal_indices = [i for i, name in enumerate(header) if name.endswith('+') or name.endswith('-')]
    numeric_indices = [i for i, name in enumerate(header) if name[0].isupper() and i not in goal_indices]
    
    # Normalize numeric columns
    data_matrix = []
    for row in rows:
        data_matrix.append([float(row[i]) for i in numeric_indices])
    
    mins = [min(col) for col in zip(*data_matrix)]
    maxs = [max(col) for col in zip(*data_matrix)]
    
    def normalize(row_data):
        norm = []
        for i, val in enumerate(row_data):
            denom = (maxs[i] - mins[i])
            norm.append((val - mins[i]) / denom if denom != 0 else 0)
        return norm

    norm_rows = [normalize(r) for r in data_matrix]
    
    # Target is the first row
    target = norm_rows[0]
    
    # Calculate distances
    distances = []
    for i in range(1, len(norm_rows)):
        dist = math.sqrt(sum((target[j] - norm_rows[i][j])**2 for j in range(len(target))))
        distances.append((dist, i))
    
    # Sort by distance (ascending)
    distances.sort(key=lambda x: x[0])
    
    # Return top 5% (max 40 rows for 800 total)
    limit = max(1, len(rows) // 20)
    return [item[1] for item in distances[:limit]]

def certify(xs, ys):
    """
    Returns True if the two groups are statistically similar (p > 0.05),
    False if they differ significantly (p <= 0.05).
    """
    if len(xs) < 2 or len(ys) < 2:
        return True
    t_stat, p_value = stats.ttest_ind(xs, ys, equal_var=False)
    return p_value > 0.05

import csv

def flow(header, row, seen):
    """
    Detects if a team's composition has shifted significantly.
    We compare the current row's numeric profile (mean of all numeric inputs)
    against the distribution of the 'seen' rows. If the current row is 
    an outlier (e.g., beyond 2 standard deviations from the mean of seen rows),
    we flag it as a shift in team composition.
    """
    if len(seen) < 30:
        return False

    # Identify numeric input columns (excluding goals)
    goal_indices = [i for i, name in enumerate(header) if name.endswith('+') or name.endswith('-')]
    numeric_indices = [i for i, name in enumerate(header) if name[0].isupper() and i not in goal_indices]
    
    def get_numeric_features(r):
        return [float(r[i]) for i in numeric_indices if isinstance(r[i], (int, float))]

    current_features = get_numeric_features(row)
    if not current_features:
        return False

    # Calculate mean and std of the 'seen' data for the first numeric feature (Age)
    # as a proxy for team demographic shift
    age_idx = header.index('Age')
    seen_ages = [r[age_idx] for r in seen if isinstance(r[age_idx], (int, float))]
    
    if len(seen_ages) < 30:
        return False
        
    mean_age = sum(seen_ages) / len(seen_ages)
    std_age = (sum((x - mean_age) ** 2 for x in seen_ages) / len(seen_ages)) ** 0.5
    
    # If the current employee's age is more than 2 standard deviations away,
    # we flag this as a potential shift in team composition.
    if std_age > 0 and abs(row[age_idx] - mean_age) > (2 * std_age):
        return True
        
    return False

import csv

def blame(header, rows):
    """
    Identifies input columns that have the least correlation with the goal 
    'YearsAtCompany+'. We identify columns that are 'useless' by checking 
    if their values are statistically independent of the goal.
    
    We return the names of columns that have the lowest impact on the goal,
    suggesting they are candidates to stop collecting.
    """
    goal_idx = -1
    for i, name in enumerate(header):
        if name.endswith('+') or name.endswith('-'):
            goal_idx = i
            break
            
    input_indices = [i for i, name in enumerate(header) if not (name.endswith('+') or name.endswith('-'))]
    
    # Calculate correlation between each input and the goal
    # For numeric inputs, use absolute correlation. For symbolic, use variance in goal.
    scores = []
    goal_vals = [float(r[goal_idx]) for r in rows]
    
    for i in input_indices:
        col_vals = [r[i] for r in rows]
        
        if isinstance(col_vals[0], (int, float)):
            # Numeric: Correlation
            mean_x = sum(col_vals) / len(col_vals)
            mean_y = sum(goal_vals) / len(goal_vals)
            num = sum((col_vals[j] - mean_x) * (goal_vals[j] - mean_y) for j in range(len(rows)))
            den = (sum((col_vals[j] - mean_x)**2 for j in range(len(rows))) * 
                   sum((goal_vals[j] - mean_y)**2 for j in range(len(rows))))**0.5
            corr = abs(num / den) if den != 0 else 0
            scores.append((corr, header[i]))
        else:
            # Symbolic: Group by category and check mean goal difference
            categories = set(col_vals)
            if len(categories) > 1:
                cat_means = {cat: [] for cat in categories}
                for j, cat in enumerate(col_vals):
                    cat_means[cat].append(goal_vals[j])
                # Score is the variance of the means
                means = [sum(v)/len(v) for v in cat_means.values()]
                var_means = sum((m - sum(means)/len(means))**2 for m in means)
                scores.append((var_means / 1000, header[i])) # Scale down
            else:
                scores.append((0, header[i]))
                
    # Sort by score ascending (lowest correlation is most useless)
    scores.sort(key=lambda x: x[0])
    
    # Return the bottom 5 columns that have the least to do with the goal
    return [name for score, name in scores[:5]]
