        idx = header.index('YearsAtCompany+')

        return [0]

    # Extract values and track original indices
    indexed_data = []
    for i, row in enumerate(rows):
        val = row[idx]
        indexed_data.append((val, i))

    # Calculate 5% limit
    limit = max(1, len(rows) // 20)
    
    # Return the indices of the top 5%
    return [item[1] for item in indexed_data[:limit]]

import csv
from sklearn.ensemble import RandomForestRegressor
from sklearn.preprocessing import LabelEncoder

def guess(header, train, test):
    """
    Estimates the value of the first goal column (YearsAtCompany+) 
    for test rows using a Random Forest Regressor.
    """
    # Identify goal column (first column ending in '+' or '-')
    goal_idx = -1
    for i, name in enumerate(header):
        if name.endswith('+') or name.endswith('-'):
            goal_idx = i
            break
    
    # Identify symbolic columns
    symbolic_indices = [i for i, name in enumerate(header) if not name[0].isupper()]
    
    # Fit encoders on all available data
    encoders = {}
    for i in symbolic_indices:
        encoders[i] = LabelEncoder()
        all_vals = [row[i] for row in train] + [row[i] for row in test if row[i] != '?']
        encoders[i].fit(all_vals)
            
    def transform_row(row, is_test=False):
        new_row = []
        for i, val in enumerate(row):
            if i == goal_idx:
                continue
            if i in symbolic_indices:
                new_row.append(float(encoders[i].transform([val])[0]))
            else:
                # If it's a test row and the value is '?', we shouldn't be using it 
                # as an input feature anyway, but here we ensure numeric conversion
                new_row.append(float(val))
        return new_row

    X_train = [transform_row(row) for row in train]
    y_train = [float(row[goal_idx]) for row in train]
    X_test = [transform_row(row) for row in test]
        
    # Train model
    model = RandomForestRegressor(n_estimators=100, random_state=42)
    model.fit(X_train, y_train)
    
    # Predict
    return model.predict(X_test).tolist()

import csv
from scipy import stats

def certify(xs, ys):
    """
    Returns True if the two groups are statistically similar (p > 0.05),
    False if they differ significantly (p <= 0.05).
    """
    t_stat, p_value = stats.ttest_ind(xs, ys, equal_var=False)
    return p_value > 0.05

def analyze_overtime_impact(header, rows):
    """
    Compares 'YearsInCurrentRole+' for employees who work overtime vs those who don't.
    """
    overtime_idx = header.index('overTime')
    years_idx = header.index('YearsInCurrentRole+')
    
    overtime_group = [row[years_idx] for row in rows if row[overtime_idx] == 'Yes']
    no_overtime_group = [row[years_idx] for row in rows if row[overtime_idx] == 'No']
    
    is_same = certify(overtime_group, no_overtime_group)
    
    avg_ot = sum(overtime_group) / len(overtime_group)
    avg_no_ot = sum(no_overtime_group) / len(no_overtime_group)
    
    return is_same, avg_ot, avg_no_ot

import csv

def flow(header, row, seen):
    """
    Decides if the current employee needs a 'what would keep you here' conversation.
    We target employees who are high-performers (YearsAtCompany+ is high) 
    but might be at risk (e.g., low JobSatisfaction or high OverTime).
    
    Strategy: Use a simple threshold based on the distribution of seen data.
    If the current row's 'YearsAtCompany+' is in the top 10% of what we've seen
    AND they have low JobSatisfaction (<= 2), flag them for a conversation.
    """
    years_idx = header.index('YearsAtCompany+')
    sat_idx = header.index('JobSatisfaction')
    
    current_years = row[years_idx]
    current_sat = row[sat_idx]
    
    # If this is the first row, we can't compare, so let it go
    if not seen:
        return False
    
    # Get all seen years to determine the threshold
    seen_years = [r[years_idx] for r in seen]
    threshold = sorted(seen_years, reverse=True)[max(0, len(seen_years) // 10)]
    
    # Criteria: High tenure (top 10%) + Low satisfaction = High risk of attrition
    if current_years >= threshold and current_sat <= 2:
        return True
    
    return False
