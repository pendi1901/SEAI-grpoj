import ezr
import random

def spot(header, rows):
    """
    Finds rows that look wrong by identifying outliers.
    We use the distance of each row from the 'mids' (centroids) of the 
    entire dataset. Rows that are significantly distant from the 
    typical values in their respective columns are flagged as suspicious.
    """
    # Create a table to leverage ezr's column statistics
    tbl = ezr.Tbl([header] + rows)
    
    # Calculate the 'mids' (centroids) for all columns
    centroids = ezr.mids(tbl)
    
    # Calculate a 'suspicion score' for each row.
    # A row is suspicious if its x-columns are far from the global centroid.
    # We use xdist to measure this distance.
    scores = []
    for i, row in enumerate(rows):
        # Calculate distance from the centroid
        d = ezr.xdist(tbl, row, centroids)
        scores.append((d, i))
    
    # Sort by distance descending (most suspicious first)
    scores.sort(key=lambda x: x[0], reverse=True)
    
    # Return the indices of the top 10 most suspicious rows
    return [item[1] for item in scores[:10]]

if __name__ == "__main__":
    filename = "WA_Fn-UseC_-HR-Employee-Attrition.n1000.train.csv"
    data = list(ezr.csv(filename))
    header = data[0]
    rows = data[1:]
    
    # Identify suspicious rows
    suspicious_indices = spot(header, rows)
    
    print(f"Analysis: Found {len(suspicious_indices)} potentially erroneous records.")
    print("These records contain input values that are statistical outliers compared to the rest of the dataset.")
    print("\nTop 5 most suspicious row indices:")
    for i in suspicious_indices[:5]:
        print(f"- Row index: {i}")
    
    print("\nNote: These rows should be manually reviewed for data entry errors.")
