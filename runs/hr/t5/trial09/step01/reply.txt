import ezr
import random

def remember(header, rows):
    """
    Finds the best rows based on the goal columns.
    Goals: YearsAtCompany+, YearsInCurrentRole+, YearsWithCurrManager+ (maximize)
    Others ending in - (minimize).
    """
    # Create a table object to use the library's utility functions
    tbl = ezr.Tbl([header] + rows)
    
    # Calculate ydist for all rows. 
    # ydist measures how far a row is from the ideal goal values.
    # Lower ydist is better.
    scored = []
    for i, row in enumerate(rows):
        d = ezr.ydist(tbl, row)
        scored.append((d, i))
    
    # Sort by distance (ascending, so best first)
    scored.sort()
    
    # Return top 5% of rows (800 rows * 0.05 = 40)
    limit = max(1, int(len(rows) * 0.05))
    return [item[1] for item in scored[:limit]]

if __name__ == "__main__":
    # Load the data
    filename = "WA_Fn-UseC_-HR-Employee-Attrition.n1000.train.csv"
    data = list(ezr.csv(filename))
    header = data[0]
    rows = data[1:]
    
    # Get the indices of the best rows
    best_indices = remember(header, rows)
    
    # Print a summary for the customer
    print(f"Found {len(best_indices)} employees who stay unusually long.")
    print("Top 3 examples (YearsAtCompany, YearsInCurrentRole, YearsWithCurrManager):")
    
    # Identify column indices for the requested metrics
    idx_comp = header.index("YearsAtCompany+")
    idx_role = header.index("YearsInCurrentRole+")
    idx_mgr = header.index("YearsWithCurrManager+")
    
    for i in best_indices[:3]:
        r = rows[i]
        print(f"- Years at company: {r[idx_comp]}, "
              f"In role: {r[idx_role]}, "
              f"With manager: {r[idx_mgr]}")
