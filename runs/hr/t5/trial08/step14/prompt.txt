def say(x: Any, p: int = 2) -> str:
  "Floats get P decimals; dicts list their keys."
  if isinstance(x,float): x= f"{x:.{p}f}".rstrip("0").rstrip(".")
  if isinstance(x, dict):
    x= "{"+", ".join(f"{k}: {say(v,p)}" for k,v in x.items())+"}"
  return str(x)

class o(dict):
  "A dict you can poke with a dot."
  __getattr__, __setattr__ = dict.__getitem__, dict.__setitem__
  __repr__ = say

def atom(s: str) -> ATOM:
  "'22' -> 22. '3.1' -> 3.1. 'True' -> True. 'x' -> 'x'."
  for fn in (int, float):
    try: return fn(s)
    except ValueError: pass
  s = s.strip()
  return {"True": True, "False": False}.get(s, s)

def csv(file: str) -> Iterator[ROW]:

  with open(os.path.expanduser(file),  # -sig drops any BOM
           encoding="utf-8-sig") as f:
    for line in f:
      if line.strip(): yield [atom(s) for s in line.split(",")]

def Col(txt: str = " ", at: int = 0) -> COL:
  "Factory that returns NUM or SYM."
  return Num(txt,at) if txt[0].isupper() else Sym(txt,at)

class Num:
  "Place to summarize stream of numbers."
  __slots__ = ("at", "txt", "n", "mu", "m2", "sd", "goal")
  def __init__(self, txt: str = " ", at: int = 0) -> None:
    self.at, self.txt, self.n = at, txt, 0
    self.mu = self.m2 = self.sd = 0
    self.goal = 0 if txt[-1] == "-" else 1
  __repr__ = lambda self: say({k: getattr(self,k)
                               for k in self.__slots__})

class Sym:
  "Place to summarize stream of Symbols.  The class is the tag."
  __slots__ = ("at", "txt", "n", "has")
  def __init__(self, txt: str = " ", at: int = 0) -> None:
    self.at, self.txt, self.n, self.has = at, txt, 0, {}
  __repr__ = lambda self: say({k: getattr(self,k)
                               for k in self.__slots__})

def add(col: COL, v: ATOM, inc: int = 1) -> None:
  "Show V to COL.  INC=-1 takes it away.  `?` changes nothing."
  if v != "?":
    col.n += inc
    if type(col) is Sym:
      col.has[v] = col.has.get(v, 0) + inc
    else:
      d = v - col.mu                      # Welford, for `div`
      col.mu += inc * d / max(1, col.n)
      col.m2  = max(0, col.m2 + inc * d * (v - col.mu))
      col.sd  = 0 if col.n < 2 else (col.m2/(col.n-1))**.5

def mid(col: COL) -> ATOM:
  "Middle: the mean, or the most common symbol; ? if none."
  return (max(col.has, key=col.has.get) if col.has else "?"
          ) if type(col) is Sym else col.mu

def mids(tbl: TBL) -> MIDS:
  "Every column's middle, cached until the next addCols."
  tbl.mids = tbl.mids or {at: mid(c) for at,c in tbl.cols.items()}
  return tbl.mids

def div(col: COL) -> float:
  "Spread: entropy, or standard deviation."
  return (-sum(v/col.n*log2(v/col.n) for v in col.has.values())
          if type(col) is Sym else col.sd)  # sd kept by `add`

def norm(col: COL, v: ATOM) -> ATOM:
  "To 0..1, by the logistic curve.  Symbols do not scale."
  if type(col) is Sym: return v
  z = max(-3, min(3, (v - col.mu) / (1e-32 + div(col))))
  return 1 / (1 + exp(-1.7 * z))

#-- tables -------------------------------------------------
def Tbl(src: Iterable[ROW]) -> TBL:
  "Header names the columns: X skip, +- goal, ! klass."
  src = iter(src)
  tbl = Cols(o(rows=[], cols={}, x=[], y=[], mids=None,
               names=next(src)))
  for row in src: addRow(tbl, row)
  return tbl

def Cols(tbl: TBL) -> TBL:
  "Create column roles."
  for at, s in enumerate(tbl.names):
    if s[-1] == "X": continue            # skip me entirely
    tbl.cols[at] = Col(s,at)
    (tbl.y if s[-1] in "+-!" else tbl.x).append(tbl.cols[at])
  return tbl

def addRow(tbl: TBL, row: ROW) -> ROW:
  "Keep ROW, and show it to my columns."
  tbl.rows += [row]
  return addCols(tbl, row)

def addCols(tbl: TBL, row: ROW, inc: int = 1) -> ROW:
  "Show ROW to my columns, without keeping it."
  tbl.mids = None                 # a new row moves the middles
  for at, col in tbl.cols.items(): add(col, row[at], inc)
  return row

def clone(tbl: TBL, rows: ROWS = None) -> TBL:
  "An empty copy of TBL, plus ROWS."
  return Tbl([tbl.names] + (rows or []))

def dist(vs: Iterable[QTY], n: int) -> float:
  "Root mean square of N gaps."
  return sqrt(sum(v*v for v in vs) / n)

def ydist(tbl: TBL, row: ROW) -> float:
  "How far ROW's goals are from the best they could be."
  return dist((abs(norm(c, row[c.at]) - c.goal)
                    for c in tbl.y), len(tbl.y))

def gap(col: COL, a: ATOM, b: ATOM) -> QTY:
  "Distance between two values of one column."
  if a == "?" or b == "?": return 1    # unknown = far
  return a!=b if type(col) is Sym else abs(norm(col,a)-norm(col,b))

def xdist(tbl: TBL, r1: ROW, r2: ROW | MIDS) -> float:
  "How far apart two rows are, over the x columns."
  return dist((gap(c, r1[c.at], r2[c.at])
                    for c in tbl.x), len(tbl.x))

def oracle(row: ROW) -> ROW:
  """Labeller. Rows here arrive with ys, so nothing to do.
  Real oracles fill in any missing y cells (and can close
  over a TBL to know which cells those are)."""
  return row

def acquire(tbl: TBL, cap: int | None = None,
            label: ORACLE = oracle) -> ROWS:
  "Label near the best rows, far from the rest."
  cap  = cap or the.Stop
  todo = random.sample(tbl.rows, len(tbl.rows))[:the.Few]
  both = clone(tbl, [label(todo.pop())
                     for _ in range(the.Start)])
  both.rows.sort(key=lambda r: ydist(both, r))
  n    = int(sqrt(the.Start))
  best,rest = clone(tbl,both.rows[:n]), clone(tbl,both.rows[n:])
  while todo and len(both.rows) < cap:
    cb, cr = mids(best), mids(rest)    # two centroids
    todo.sort(key=lambda z: xdist(tbl,z,cr) - xdist(tbl,z,cb))
    addRow(best, addRow(both, label(todo.pop())))
    best.rows.sort(key=lambda z: ydist(both, z))  # no y leak
    if len(best.rows) > sqrt(len(both.rows)): # best stays small
      addRow(rest, addCols(best, best.rows.pop(), -1))
  return both.rows

def grabs(tbl: TBL, cap: int | None = None,
          label: ORACLE = oracle) -> ROWS:
  "The straw man: same budget, taken at random."
  cap = cap or the.Stop
  return [label(r) for r in
          random.sample(tbl.rows, min(cap, len(tbl.rows)))]

# Calls core.  Nothing above this line calls down here.
# NODE = o(rows, at, v, go, kids, mu).
def cut(tbl: TBL, rows: ROWS) -> tuple[int,ATOM,GO]|None:
  "The (at, v, go) whose two sides have the tightest ys."
  ys = [ydist(tbl, r) for r in rows]
  out, least = None, 1e30
  for col in tbl.x:
    for v, go in candidates(col, rows):
      a, b = Num(), Num()   # one pass, so one `go` per row
      for r,y in zip(rows,ys): add(a if go(r) else b, y)
      if a.n >= the.Leaf and b.n >= the.Leaf:
        score = (div(a)*a.n + div(b)*b.n) / (a.n + b.n)
        if score < least: out, least = (col.at, v, go), score
  return out

def candidates(col: COL, rows: ROWS) -> Iterator[tuple[ATOM,GO]]:
  "What splits to try: one per symbol, or CUTS per num."
  at = col.at
  if type(col) is Sym:
    for v in sorted({r[at] for r in rows if r[at] != "?"}):
      yield v, lambda r,v=v,at=at: r[at] == v
  else:                # by rank, not value: THESE rows are dense
    xs = sorted(r[at] for r in rows if r[at] != "?")
    k  = max(1, len(xs)//the.Cuts)     # cuts, spaced by rank
    for v in xs[k-1:-1:k]:
      yield v, (lambda r,v=v,at=at: r[at] != "?" and r[at] <= v)

def tree(tbl: TBL, rows: ROWS) -> NODE:
  "Split while a split keeps both sides big enough."
  node = o(rows=rows, at=None, v=None, go=None, kids=[],
           mu=sum(ydist(tbl,r) for r in rows)/len(rows))
  if len(rows) > the.Leaf and (found := cut(tbl, rows)):
    at, v, go = found
    yes, no = [], []
    for r in rows: (yes if go(r) else no).append(r)
    node.at, node.v, node.go = at, v, go # cut kepts sides big
    node.kids = [tree(tbl, yes), tree(tbl, no)]
  return node

def leaf(node: NODE, row: ROW) -> NODE:
  "Walk ROW down to its leaf."
  while node.kids: node = node.kids[0 if node.go(row) else 1]
  return node

def leafs(node: NODE) -> list[NODE]:
  "Return leaves of this tree."
  return [x for k in node.kids for x in leafs(k)] or [node]

def show(tbl: TBL, node: NODE, pre: str | None = None,
         edge: str = "") -> None:
  "Tree on the left; ydist and n on the right."
  ls = sorted(leafs(node), key=lambda z: z.mu)
  def walk(z: NODE, pre: str | None, edge: str) -> None:
    m = "+" if z is ls[0] else "-" if z is ls[-1] else " "
    print(f"{m} {round(100*z.mu):>4} {len(z.rows):>5}   "
          f"{(pre or '')+edge}".rstrip())
    op = "=" if type(tbl.cols[z.at or 0]) is Sym else "<="
    no = "!=" if op == "=" else ">"
    for kid, oper in zip(z.kids, [op, no]):
      walk(kid, "" if pre is None else pre + "|  ",
           f"{tbl.names[z.at]} {oper} {say(z.v)}")
  print(" ydist     n")
  walk(node, pre, edge)

# Calls core, acquire, and one RANKER -- tree's, by default.
def ranker(lab: TBL) -> Callable[[ROW], float]:
  "Grow a tree on LAB; score any row by its leaf's mean ydist."
  node = tree(lab, lab.rows)
  return lambda r: leaf(node, r).mu

def wins(tbl: TBL) -> Callable[[ROW], float]:
  "100 at the best row, 0 at an average one."
  ys = sorted(ydist(tbl, r) for r in tbl.rows)
  lo, avg = ys[0], sum(ys)/len(ys)
  return lambda r: max(-100, min(100, 100*(1 - (ydist(tbl,r)-lo)
                                            / (avg-lo+1e-32))))

def holdout(tbl: TBL, pick: PICKER = acquire,
            rank: RANKER = ranker,
            label: ORACLE = oracle) -> ROW:
  "Train on half; of CHECK guesses on the rest, pick best."
  rows  = random.sample(tbl.rows, len(tbl.rows))
  n     = len(rows)//2
  tr    = clone(tbl, rows[:n][:the.Few])
  lab   = clone(tbl, pick(tr, the.Stop - the.Check, label))
  top   = sorted(rows[n:], key=rank(lab))
  return min(top[:the.Check], key=lambda r: ydist(lab, label(r)))

# A leaf: it calls core, and nothing here calls it.
def adds(vs: Iterable[ATOM], col: COL | None = None) -> COL:
  "Every item of VS into COL."
  col = Num() if col is None else col
  for v in vs: add(col, v)
  return col

def cohen(xs: NUMS, ys: NUMS, d: float = 0.35,
          eps: float = 0) -> bool:
  "Are the means within D pooled standard deviations?"
  a, b = adds(xs), adds(ys)
  pool = ((a.n-1)*div(a)**2 + (b.n-1)*div(b)**2)/(a.n+b.n-2)
  return abs(a.mu - b.mu) <= max(eps, d * sqrt(pool))

def cliffs(xs: NUMS, ys: NUMS, d: float = 0.197) -> bool:
  "Sorted in.  Is the rank imbalance small enough?"
  gt = lt = j = k = 0
  for x in xs:
    while j < len(ys) and ys[j] <  x: j += 1; k = j
    while k < len(ys) and ys[k] <= x: k += 1
    gt += j; lt += len(ys) - k
  return abs(gt - lt) / (len(xs) * len(ys)) <= d

def ks(xs: NUMS, ys: NUMS, a: float = 1.36) -> bool:
  "Sorted in.  95% Kolmogorov-Smirnov."
  n, m, i, j, d = len(xs), len(ys), 0, 0, 0
  while i < n and j < m:
    v = min(xs[i], ys[j])
    while i < n and xs[i] <= v: i += 1
    while j < m and ys[j] <= v: j += 1
    d = max(d, abs(i/n - j/m))
  return d <= a * sqrt((n + m) / (n * m))

def same(xs: NUMS, ys: NUMS, eps: float = 0) -> bool:
  """Indistinguishable.  Always cliffs and ks; EPS adds cohen, and
  doubles as its effect size (eps=0 skips that third test)."""
  xs, ys = sorted(xs), sorted(ys)
  return (cliffs(xs,ys) and ks(xs,ys)
          and (not eps or cohen(xs,ys,d=eps)))

import ezr
import random

def guess(header, train, test):
    """
    Estimates the value of the first goal column for test rows.
    Uses a tree-based approach: train a tree on the labeled data,
    then for each test row, find the leaf and return the mean
    of the first goal column for rows in that leaf.
    """
    # Create a table for training data
    train_tbl = ezr.Tbl([header] + train)
    
    # Build a tree on the training data
    # The tree nodes contain the rows that fell into that leaf
    root = ezr.tree(train_tbl, train_tbl.rows)
    
    # Identify the index of the first goal column
    # Goals are columns ending in '+' or '-'
    goal_indices = [i for i, name in enumerate(header) if name[-1] in "+-"]
    first_goal_idx = goal_indices[0]
    
    results = []
    for row in test:
        # Walk the tree to find the leaf for this test row
        node = ezr.leaf(root, row)
        
        # Calculate the mean of the first goal column for all rows in this leaf
        vals = [r[first_goal_idx] for r in node.rows if r[first_goal_idx] != "?"]
        if vals:
            results.append(sum(vals) / len(vals))
        else:
            # Fallback to global mean if leaf is empty or has no valid data
            all_vals = [r[first_goal_idx] for r in train if r[first_goal_idx] != "?"]
            results.append(sum(all_vals) / len(all_vals))
            
    return results

import ezr

def flow(header, row, seen):
    """
    Decide if the current row is worth a 'retention' conversation.
    We use the 'seen' rows to build a model of what 'good' looks like
    (using the ydist metric) and compare the current row against that.
    If we have enough data, we pick the top 5% of candidates.
    """
    # Need at least some history to make a relative decision
    if len(seen) < 20:
        return False
    
    # Create a table from seen rows to calculate ydist
    # We filter out rows that might have missing goals if necessary
    tbl = ezr.Tbl([header] + seen)
    
    # Calculate ydist for the current row
    # ydist measures how far a row is from the ideal goal values
    # Lower ydist is better
    current_ydist = ezr.ydist(tbl, row)
    
    # Calculate ydist for all seen rows
    seen_ydists = [ezr.ydist(tbl, r) for r in seen]
    
    # If the current row is in the top 5% of what we have seen so far,
    # flag it for a conversation.
    threshold = sorted(seen_ydists)[int(len(seen_ydists) * 0.05)]
    
    return current_ydist <= threshold

import ezr
import random

def blame(header, rows):
    """
    Identifies input columns that best explain why the worst rows are bad.
    We define 'bad' as having a high ydist (distance from ideal goals).
    We compare the distribution of values in the worst 25% of rows 
    against the best 25% of rows for each input column.
    """
    tbl = ezr.Tbl([header] + rows)
    
    # Sort rows by ydist (lower is better)
    sorted_rows = sorted(rows, key=lambda r: ezr.ydist(tbl, r))
    
    n = len(sorted_rows) // 4
    best = sorted_rows[:n]
    worst = sorted_rows[-n:]
    
    # Score columns by how much their distributions differ between best and worst
    # We use a simple heuristic: for numeric, difference in means; 
    # for symbolic, difference in frequency of the most common value.
    scores = []
    for col in tbl.x:
        at = col.at
        if isinstance(col, ezr.Num):
            best_vals = [r[at] for r in best if r[at] != "?"]
            worst_vals = [r[at] for r in worst if r[at] != "?"]
            if best_vals and worst_vals:
                diff = abs((sum(worst_vals)/len(worst_vals)) - (sum(best_vals)/len(best_vals)))
                scores.append((diff, header[at]))
        else:
            # Symbolic: compare frequency of the most common value in 'worst'
            worst_vals = [r[at] for r in worst if r[at] != "?"]
            if worst_vals:
                counts = {}
                for v in worst_vals: counts[v] = counts.get(v, 0) + 1
                mode = max(counts, key=counts.get)
                # Score is how much more common this value is in worst vs best
                best_vals = [r[at] for r in best if r[at] != "?"]
                b_count = best_vals.count(mode) / len(best_vals) if best_vals else 0
                w_count = counts[mode] / len(worst_vals)
                scores.append((w_count - b_count, header[at]))
    
    # Return top 3 columns that correlate with being 'bad'
    scores.sort(key=lambda x: x[0], reverse=True)
    return [name for score, name in scores[:3]]

import ezr
import random

def justify(header, rows, row):
    """
    Returns (p, reason): p is the fraction of rows that 'row' is better than,
    and reason is a sentence explaining the ranking.
    """
    # Create a table to use ydist for ranking
    tbl = ezr.Tbl([header] + rows)
    
    # Estimate the goal values for the input row.
    # We use the tree-based approach: find the leaf for the input row
    # and use the median ydist of that leaf as the estimate.
    root = ezr.tree(tbl, tbl.rows)
    leaf_node = ezr.leaf(root, row)
    
    # The estimated ydist for the input row is the mean ydist of its leaf
    estimated_ydist = leaf_node.mu
    
    # Calculate ydist for all existing rows
    all_ydists = [ezr.ydist(tbl, r) for r in rows]
    
    # p is the fraction of rows that have a higher (worse) ydist than our estimate
    # (i.e., our row is better than them)
    better_than = [y for y in all_ydists if y > estimated_ydist]
    p = len(better_than) / len(all_ydists)
    
    reason = (f"This arrangement ranks in the top {int((1-p)*100)}% of all observed "
              f"retention outcomes, based on its alignment with key tenure metrics.")
    
    return p, reason

import ezr
import random

def choose(header, rows, label):
    """
    Find the best working arrangement using at most 50 labels.
    We use the 'acquire' strategy from ezr to intelligently select
    rows to label, then return the index of the best one found.
    """
    # The error in the previous version occurred because ezr.ydist 
    # was called on rows containing '?' (strings).
    # We must only call ydist on rows that have been fully labeled.
    
    # Create a table object. We use a copy of rows to avoid modifying the input
    # until we have the true values.
    tbl = ezr.Tbl([header] + rows)
    
    # Track which rows have been labeled to avoid re-labeling
    labeled_indices = {}

    def oracle_wrapper(row):
        # Find the index of this row in the input list
        # We use id() or index matching to find the original row
        idx = -1
        for i, r in enumerate(rows):
            if r is row:
                idx = i
                break
        
        if idx not in labeled_indices:
            true_vals = label(idx)
            # Update the row in place with true values
            for i, val in enumerate(true_vals):
                row[i] = val
            labeled_indices[idx] = row
        return row

    # Use ezr.acquire to select and label rows.
    # ezr.acquire returns the rows that were labeled.
    # Note: ezr.acquire uses ydist internally, so it will only work
    # correctly once rows are labeled. We ensure the initial set 
    # is labeled before the main loop logic in acquire kicks in.
    labeled_rows = ezr.acquire(tbl, cap=50, label=oracle_wrapper)
    
    # After labeling, find the best row among those labeled.
    # A row is better when its ydist is lower.
    best_row = min(labeled_rows, key=lambda r: ezr.ydist(tbl, r))
    
    # Return the index of the best row
    for i, r in enumerate(rows):
        if r is best_row:
            return i
    return 0

import ezr

def fix(header, row, levers):
    """
    Given a 'bad' row, suggest changes to levers to improve its ydist.
    We simulate the impact of changing each lever by creating a modified row
    and checking if it moves closer to the ideal goal values.
    """
    # Load data to build the table context for ydist
    data = list(ezr.csv("WA_Fn-UseC_-HR-Employee-Attrition.n1000.train.csv"))
    tbl = ezr.Tbl(data)
    
    best_change = {}
    best_ydist = ezr.ydist(tbl, row)
    
    # Try each lever
    for col_name, allowed in levers.items():
        if col_name not in header:
            continue
        col_idx = header.index(col_name)
        col = tbl.cols[col_idx]
        
        # Generate candidates based on type
        candidates = []
        if isinstance(col, ezr.Sym):
            # For symbolic, try all allowed values
            candidates = [v for v in allowed if v != row[col_idx]]
        else:
            # For numeric, try the boundaries and midpoint
            low, high = allowed
            candidates = [v for v in [low, (low + high) / 2, high] if v != row[col_idx]]
            
        for val in candidates:
            new_row = list(row)
            new_row[col_idx] = val
            
            # Check if this change improves the ydist
            current_ydist = ezr.ydist(tbl, new_row)
            if current_ydist < best_ydist:
                best_ydist = current_ydist
                best_change = {col_name: val}
                
    return best_change

def spot(header, rows):
    """
    Finds rows that look wrong by identifying outliers in the input space.
    We use the distance from the centroid of the data (mids) as a measure
    of how 'unusual' a row is. Rows that are significantly distant from 
    the central tendency of the dataset are flagged as suspicious.
    """
    # Create a table to get column summaries
    tbl = ezr.Tbl([header] + rows)
    
    # Calculate the centroid (mids) of the dataset
    centroid = ezr.mids(tbl)
    
    # Calculate the distance of each row from the centroid
    # We use xdist because we are looking for suspicious input values
    # (not goal values, which might be naturally extreme)
    distances = []
    for i, row in enumerate(rows):
        d = ezr.xdist(tbl, row, centroid)
        distances.append((d, i))
    
    # Sort by distance descending (most suspicious first)
    distances.sort(key=lambda x: x[0], reverse=True)
    
    # Return the indices of the top 10 most suspicious rows
    return [idx for d, idx in distances[:10]]

import ezr
import random

def race(header, rows, label):
    """
    Compare three strategies for finding the best employee arrangement.
    Budget: 50 labels total.
    """
    # Shared budget tracker
    labeled_data = {}
    
    def oracle_wrapper(i):
        if i not in labeled_data:
            if len(labeled_data) >= 50:
                return None
            labeled_data[i] = label(i)
        return labeled_data[i]

    # We need a table that knows the true goal values to calculate ydist.
    # Since we only have 50 labels, we build a 'lab' table from those.
    # To calculate ydist, we need a table that has seen the full distribution.
    # We use the initial rows to build the column definitions.
    full_tbl = ezr.Tbl([header] + rows)

    def get_ydist(row_idx):
        # If we haven't labeled this row, we can't calculate its true ydist.
        # We return a high value for unlabelled rows to discourage picking them
        # as the 'best' unless they are labeled.
        if row_idx not in labeled_data:
            return 1e30
        return ezr.ydist(full_tbl, labeled_data[row_idx])

    # Strategies
    def strategy_random(rows):
        indices = random.sample(range(len(rows)), 16) # 16 * 3 = 48 < 50
        for i in indices: oracle_wrapper(i)
        return min(indices, key=get_ydist)

    def strategy_acquire(rows):
        # Use ezr.acquire logic
        # We need to map the oracle to the index
        def local_oracle(row):
            for i, r in enumerate(rows):
                if r is row: return oracle_wrapper(i)
            return row
        
        # We use a subset of the budget
        labeled = ezr.acquire(full_tbl, cap=17, label=local_oracle)
        # Find index of best labeled row
        best_row = min(labeled, key=lambda r: ezr.ydist(full_tbl, r))
        for i, r in enumerate(rows):
            if r is best_row: return i
        return 0

    def strategy_greedy(rows):
        # Simple greedy: pick one, then pick the one closest to it
        curr = random.randint(0, len(rows)-1)
        oracle_wrapper(curr)
        for _ in range(16):
            best_next = -1
            min_dist = 1e30
            for i in range(len(rows)):
                if i not in labeled_data:
                    d = ezr.xdist(full_tbl, rows[i], rows[curr])
                    if d < min_dist:
                        min_dist = d
                        best_next = i
            if best_next == -1: break
            oracle_wrapper(best_next)
            curr = best_next
        return min(labeled_data.keys(), key=get_ydist)

    # Run race
    results = {
        "random": strategy_random(rows),
        "acquire": strategy_acquire(rows),
        "greedy": strategy_greedy(rows)
    }
    
    # Determine winner (lowest ydist)
    winner = min(results, key=lambda k: get_ydist(results[k]))
    
    return {
        "winner": winner,
        "row": results[winner],
        "tried": list(results.keys())
    }

def guess(header, train, test):

    'no overtime' policy across the 'Sales' department.
    """

    goal_indices = [i for i, name in enumerate(header) if name[-1] in "+-"]
    first_goal_idx = goal_indices[0]
    dept_idx = header.index("department")
    overtime_idx = header.index("overTime")
    
    # Filter training data: only keep rows where department is Sales
    # and overtime is 'No' to learn the behavior of that specific group.
    sales_no_ot = [r for r in train if r[dept_idx] == "Sales" and r[overtime_idx] == "No"]
    
    # If we don't have enough specific data, fall back to all training data
    if len(sales_no_ot) < 10:
        training_set = train
    else:
        training_set = sales_no_ot
        
    # Build a tree on the relevant training data
    tbl = ezr.Tbl([header] + training_set)
    root = ezr.tree(tbl, tbl.rows)
    
    results = []
    for row in test:
        # Create a copy and apply the 'no overtime' policy
        modified_row = list(row)
        if modified_row[dept_idx] == "Sales":
            modified_row[overtime_idx] = "No"
            
        # Walk the tree to find the leaf
        node = ezr.leaf(root, modified_row)
        
        # Estimate based on the mean of the first goal in that leaf
        vals = [r[first_goal_idx] for r in node.rows if r[first_goal_idx] != "?"]
        if vals:
            results.append(sum(vals) / len(vals))
        else:
            # Fallback to global mean of the training set
            all_vals = [r[first_goal_idx] for r in training_set if r[first_goal_idx] != "?"]
            results.append(sum(all_vals) / len(all_vals) if all_vals else 0.0)
            
    return results
