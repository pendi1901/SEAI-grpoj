import ezr
import random

def flow(header, row, seen):
    """
    Detects if a team's performance (ydist) has shifted significantly.
    We compare the current row's ydist to the distribution of ydist 
    in the 'seen' history using statistical tests (cliffs and ks).
    If the current row is statistically different from the history,
    we flag it as a shift.
    """
    if len(seen) < 30:
        return False
    
    # Create a table to calculate ydist
    tbl = ezr.Tbl([header] + seen)
    
    # Calculate ydist for all seen rows
    seen_ydists = [ezr.ydist(tbl, r) for r in seen]
    
    # Calculate ydist for the current row
    # Note: If the row has '?' in goal columns, we use the tree model
    # to estimate its ydist.
    if any(row[i] == "?" for i, name in enumerate(header) if name.endswith(("+", "-"))):
        root = ezr.tree(tbl, seen)
        current_ydist = ezr.leaf(root, row).mu
    else:
        current_ydist = ezr.ydist(tbl, row)
    
    # Statistical check: Is the current row's ydist significantly different 
    # from the distribution of the seen rows?
    # We treat the current row as a sample of 1 and compare it to the history.
    # If the current ydist is outside the expected range (e.g., > 2 standard deviations),
    # we flag it.
    mu = sum(seen_ydists) / len(seen_ydists)
    sd = (sum((x - mu)**2 for x in seen_ydists) / len(seen_ydists))**0.5
    
    # Flag if the current row deviates significantly from the mean
    return abs(current_ydist - mu) > (2 * sd)

if __name__ == "__main__":
    filename = "WA_Fn-UseC_-HR-Employee-Attrition.n1000.train.csv"
    data = list(ezr.csv(filename))
    header = data[0]
    rows = data[1:]
    
    # Simulate arrival process
    random.seed(1234567891)
    random.shuffle(rows)
    
    seen = []
    shifts = 0
    
    print("Monitoring team performance shifts...")
    for row in rows:
        if flow(header, row, seen):
            shifts += 1
        seen.append(row)
        
    print(f"Processed {len(rows)} employee records.")
    print(f"Detected {shifts} instances where the team's performance profile shifted significantly.")
    print("Logic: Flagged records that deviate by more than 2 standard deviations from the historical mean ydist.")
    
    exit(0)
