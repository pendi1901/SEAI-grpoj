import ezr
import random

def choose(header, rows, label):
    """
    Find the best working arrangement using at most 50 labels.
    We use an active learning approach to identify the best row.
    Since the library's 'holdout' function expects to work with internal
    data structures, we implement a manual selection loop to stay
    within the 50-label budget and avoid unhashable type errors.
    """
    # Create a table to manage column definitions and ydist calculations
    tbl = ezr.Tbl([header] + rows)
    
    # We will track which rows we have labeled to avoid re-labeling
    # and to stay within the budget.
    labeled_indices = {}
    
    def get_labeled_row(idx):
        if idx not in labeled_indices:
            # The label function provided by the harness reveals the goals
            # for the row at the given index.
            labeled_indices[idx] = label(rows[idx])
        return labeled_indices[idx]

    # Initial random sample to start the model
    # We use a small number to bootstrap the search
    budget = 50
    indices = list(range(len(rows)))
    random.shuffle(indices)
    
    # Start by labeling a few random rows
    initial_samples = indices[:ezr.the.Start]
    for i in initial_samples:
        get_labeled_row(i)
        
    # Build a model with the labeled data
    labeled_rows = [labeled_indices[i] for i in labeled_indices]
    lab_tbl = ezr.clone(tbl, labeled_rows)
    
    # Iteratively pick the next best row to label based on the current model
    # We use the ranker to find the most promising rows
    while len(labeled_indices) < budget:
        rank_fn = ezr.ranker(lab_tbl)
        # Score all unlabeled rows
        unlabeled = [i for i in indices if i not in labeled_indices]
        if not unlabeled:
            break
        
        # Pick the row that the model thinks is best
        best_unlabeled = min(unlabeled, key=lambda i: rank_fn(rows[i]))
        get_labeled_row(best_unlabeled)
        
        # Update the model
        labeled_rows = [labeled_indices[i] for i in labeled_indices]
        lab_tbl = ezr.clone(tbl, labeled_rows)
        
    # After exhausting the budget, find the best row among all labeled ones
    best_idx = min(labeled_indices.keys(), key=lambda i: ezr.ydist(tbl, labeled_indices[i]))
    
    return best_idx

if __name__ == "__main__":
    filename = "WA_Fn-UseC_-HR-Employee-Attrition.n1000.train.csv"
    data = list(ezr.csv(filename))
    header = data[0]
    rows = data[1:]
    
    # Mask the goals for the simulation
    goal_indices = [i for i, name in enumerate(header) if name.endswith('+') or name.endswith('-')]
    
    def masked_label(row):
        # In the real scenario, this would be an interview.
        # Here we return the full row as the ground truth.
        return row

    # Create masked rows for the choose function
    masked_rows = []
    for row in rows:
        new_row = list(row)
        for i in goal_indices:
            new_row[i] = "?"
        masked_rows.append(new_row)
        
    best_idx = choose(header, masked_rows, masked_label)
    
    print(f"Analysis complete. Identified optimal employee record at index: {best_idx}")
    print("This record represents the working arrangement that maximizes")
    print("tenure and satisfaction based on 50 targeted interviews.")
    
    best_row = rows[best_idx]
    print("\nKey characteristics of the optimal arrangement:")
    for i, name in enumerate(header):
        if name in ["Age", "MonthlyIncome", "TotalWorkingYears", "YearsAtCompany+", "jobRole"]:
            print(f"  {name:20}: {best_row[i]}")
