import ezr
import random

def spot(header, rows):
    """
    Identify rows that are 'untrustworthy' by measuring their distance
    from the population norm in the input space. If a row is an extreme
    outlier (high xdist from the centroid), our tree-based estimates
    for that row cannot be trusted.
    """
    # Create a table to handle column definitions and distance calculations
    # We use the full dataset to establish the population norm
    tbl = ezr.Tbl([header] + rows)
    
    # Calculate the centroid (mids) of the input columns
    centroid = ezr.mids(tbl)
    
    # Calculate the distance of each row from the centroid
    # We use xdist to measure distance across input columns
    distances = []
    for i, row in enumerate(rows):
        d = ezr.xdist(tbl, row, centroid)
        distances.append((i, d))
    
    # Sort by distance descending (most suspicious/untrustworthy first)
    distances.sort(key=lambda x: x[1], reverse=True)
    
    # Return the indices of the rows that are most suspicious
    # We return the top 10 as the most likely candidates for "untrustworthy" estimates
    return [item[0] for item in distances[:10]]

if __name__ == "__main__":
    filename = "WA_Fn-UseC_-HR-Employee-Attrition.n1000.train.csv"
    data = list(ezr.csv(filename))
    header = data[0]
    rows = data[1:]
    
    suspicious_indices = spot(header, rows)
    
    print("Untrustworthy Estimate Analysis")
    print("-" * 60)
    print("The following rows are the most statistically unusual")
    print("based on their input values. Estimates for these employees")
    print("cannot be trusted as they deviate from our known population:")
    print("-" * 60)
    print(f"{'Row Index':<10} | {'Suspicion Score (Distance)'}")
    print("-" * 60)
    
    # Re-calculate distances for display purposes
    tbl = ezr.Tbl([header] + rows)
    centroid = ezr.mids(tbl)
    
    for idx in suspicious_indices:
        dist = ezr.xdist(tbl, rows[idx], centroid)
        print(f"{idx:<10} | {dist:.4f}")
        
    print("-" * 60)
    print("Logic: Flagged rows with high input-space distance from the centroid.")
    print("These records are too unique for our current model to predict reliably.")
