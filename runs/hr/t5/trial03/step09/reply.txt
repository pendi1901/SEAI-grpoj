import ezr
import random

def spot(header, rows):
    """
    Identifies suspicious rows by finding outliers.
    A row is suspicious if its input values are far from the 
    expected distribution of the rest of the data (using xdist).
    """
    tbl = ezr.Tbl([header] + rows)
    
    # Calculate the centroid of the entire dataset
    # The library's mids() provides the central tendency for all columns
    centroid = ezr.mids(tbl)
    
    # Calculate the distance of each row from the centroid
    # Rows that are very far from the centroid are likely outliers/errors
    scores = []
    for i, row in enumerate(rows):
        # xdist measures distance across input columns
        d = ezr.xdist(tbl, row, centroid)
        scores.append((d, i))
    
    # Sort by distance descending (most suspicious first)
    scores.sort(key=lambda x: x[0], reverse=True)
    
    # Return the indices of the top 10 most suspicious rows
    return [idx for d, idx in scores[:10]]

if __name__ == "__main__":
    filename = "WA_Fn-UseC_-HR-Employee-Attrition.n1000.train.csv"
    data = list(ezr.csv(filename))
    header = data[0]
    rows = data[1:]
    
    suspicious_indices = spot(header, rows)
    
    print("--- Suspicious Record Analysis ---")
    print(f"Identified {len(suspicious_indices)} potentially erroneous records.")
    print("Most suspicious row indices (most to least):")
    print(suspicious_indices)
    
    print("\nSample of suspicious data (Row Index 1):")
    # Show a few columns for the most suspicious row
    top_idx = suspicious_indices[0]
    row = rows[top_idx]
    for i in range(min(5, len(header))):
        print(f"  {header[i]}: {row[i]}")
    
    exit(0)
