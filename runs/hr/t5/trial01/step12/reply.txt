import ezr
import random

def guess(header, train, test):
    """
    Estimates the value of the first goal column for each row in test.
    We use a tree-based approach: train a tree on the labeled data,
    then for each test row, walk the tree to find the leaf and return
    the mean value of the first goal column in that leaf.
    """
    # Create table from training data
    tbl = ezr.Tbl([header] + train)
    
    # Build the tree using the training data
    root = ezr.tree(tbl, tbl.rows)
    
    # Identify the index of the first goal column
    # Goals are columns ending in '+' or '-'
    goal_indices = [i for i, name in enumerate(header) if name.endswith('+') or name.endswith('-')]
    first_goal_idx = goal_indices[0]
    
    # Pre-calculate the mean of the first goal for each leaf
    def cache_leaf_means(node):
        if not node.kids:
            vals = [r[first_goal_idx] for r in node.rows if r[first_goal_idx] != "?"]
            node.mean_goal = sum(vals) / len(vals) if vals else 0
        else:
            for kid in node.kids:
                cache_leaf_means(kid)
    
    cache_leaf_means(root)
    
    # For each test row, find its leaf and return the cached mean
    results = []
    for row in test:
        leaf_node = ezr.leaf(root, row)
        results.append(float(leaf_node.mean_goal))
        
    return results

    filename = "WA_Fn-UseC_-HR-Employee-Attrition.n1000.train.csv"
    data = list(ezr.csv(filename))
    header = data[0]
    rows = data[1:]
    
    # Identify indices
    dept_idx = header.index('department')
    ot_idx = header.index('overTime')
    goal_idx = [i for i, name in enumerate(header) if name.endswith('+') or name.endswith('-')][0]
    
    # Filter for Sales department
    sales_rows = [r for r in rows if r[dept_idx] == 'Sales']
    
    # Create two scenarios:
    # 1. Current state (keep as is)
    # 2. Cut overtime (force 'No' for all)
    test_rows_current = [list(r) for r in sales_rows]
    test_rows_no_ot = [list(r) for r in sales_rows]
    for r in test_rows_no_ot:
        r[ot_idx] = 'No'
        
    # Mask goals for estimation
    for r in test_rows_current: r[goal_idx] = "?"
    for r in test_rows_no_ot: r[goal_idx] = "?"
    
    # Estimate
    est_current = guess(header, rows, test_rows_current)
    est_no_ot = guess(header, rows, test_rows_no_ot)
    
    avg_current = sum(est_current) / len(est_current)
    avg_no_ot = sum(est_no_ot) / len(est_no_ot)
    
    print("Analysis: Impact of cutting overtime in Sales on retention (YearsAtCompany+)")
    print(f"  Average years at company (Current): {avg_current:.2f}")
    print(f"  Average years at company (No Overtime): {avg_no_ot:.2f}")
    print(f"  Predicted change: {avg_no_ot - avg_current:+.2f} years")
    print("\nConclusion: Cutting overtime in Sales is predicted to change the average")
