import ezr
import csv


def race(header, rows, label):
    """Three strategies compete for the best row among up to 50 labelled."""

    tbl = ezr.Tbl([header] + rows)

    def ydist(r):
        return ezr.ydist(tbl, r)

    # ---- Strategy: random -------------------------------------------------
    def _random():
        chosen = set()
        while len(chosen) < 50 and any(c == "?" for c in tbl.x.values()):
            i = label(random.randint(0, len(rows) - 1))
            chosen.add(i)
        return list(chosen)[:50]

    # ---- Strategy: greedy (closest to current best) -----------------------
    def _greedy():
        chosen = set()
        seeded = set()
        for i in range(min(10, len(rows))):
            label(i)
            seeded.add(i)
        while len(seeded) < 50 and any(c == "?" for c in tbl.x.values()):
            candidates = [i for i in range(len(rows)) if i not in seeded]
            scored = [(ydist(tbl.rows[i]), i) for i in candidates]
            scored.sort(key=lambda x: x[0])
            pick = scored[0][1]
            label(pick)
            seeded.add(pick)
        return list(seeded)[:50]

    # ---- Strategy: tree-guided (pick rows near the best centroid) ---------
    def _tree():
        chosen = set()
        seeded = set()
        for i in range(min(10, len(rows))):
            label(i)
            seeded.add(i)
        while len(seeded) < 50 and any(c == "?" for c in tbl.x.values()):
            cb = ezr.mids(ezr.clone(tbl, [tbl.rows[i] for i in seeded]))
            candidates = [i for i in range(len(rows)) if i not in seeded]
            scored = [(ezr.xdist(tbl, tbl.rows[i], cb), i) for i in candidates]
            scored.sort(key=lambda x: -x[0])
            pick = scored[0][1]
            label(pick)
            seeded.add(pick)
        return list(seeded)[:50]

    # Run all three, collect the best row across them.
    strategies = {
        "random": _random(),
        "greedy": _greedy(),
        "tree":   _tree(),
    }

    best_row = None
    best_score = float("inf")
    winner_name = ""
    tried = list(strategies.keys())

    for name, chosen in strategies.items():
        for i in chosen:
            score = ydist(tbl.rows[i])
            if score < best_score:
                best_score = score
                best_row = i
                winner_name = name

    return {
        "winner": winner_name,
        "row": best_row,
        "tried": tried,
    }


if __name__ == "__main__":
    import random
    with open("WA_Fn-UseC_-HR-Employee-Attrition.train.csv", encoding="utf-8-sig") as f:
        reader = csv.reader(f)
        header = next(reader)
        rows = [next(reader) for _ in range(1470)]

    result = race(header, rows, label)
    print(f"Winner: {result['winner']}  |  Best row: {result['row']}")
