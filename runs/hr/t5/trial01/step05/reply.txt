import ezr

def blame(header, rows):
    """Return column names (not goals) that best explain why the worst rows are bad."""
    tbl = ezr.Tbl([header] + rows)

    def ydist(row):
        return ezr.ydist(tbl, row)

    scored = [(ydist(r), r) for r in rows]
    scored.sort(key=lambda x: -x[0])  # worst first

    half = len(scored) // 2
    worst = [r for _, r in scored[:half]]

    tree = ezr.tree(tbl, worst)

    blame_cols = []
    if tree.at is not None:
        col_name = header[tree.at]
        blame_cols.append(col_name)
        for kid in tree.kids:
            if kid.at is not None and kid.at != tree.at:
                col_name = header[kid.at]
                blame_cols.append(col_name)

    return blame_cols


if __name__ == "__main__":
    import csv
    with open("WA_Fn-UseC_-HR-Employee-Attrition.train.csv", encoding="utf-8-sig") as f:
        reader = csv.reader(f)
        header = next(reader)
        rows = [next(reader) for _ in range(1470)]

    result = blame(header, rows)
    print("The jobs that explain short stays are:")
    for col in result:
        print(f"  {col}")
