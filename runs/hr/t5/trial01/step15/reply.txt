import ezr
import random

def blame(header, rows):
    """
    Identifies input columns that have the least impact on goal outcomes.
    We compare the distribution of input values in the worst 25% of rows
    against the rest. Columns with the smallest difference in normalized 
    space are the ones that have 'nothing to do' with the outcomes.
    """
    tbl = ezr.Tbl([header] + rows)
    
    # Calculate ydist for all rows to identify the 'worst'
    scored = []
    for r in rows:
        scored.append((ezr.ydist(tbl, r), r))
    
    # Sort by ydist descending (worst first)
    scored.sort(key=lambda x: x[0], reverse=True)
    
    n_worst = max(1, len(rows) // 4)
    worst = [x[1] for x in scored[:n_worst]]
    rest = [x[1] for x in scored[n_worst:]]
    
    # For each input column, calculate a 'relevance' score
    # Relevance is the magnitude of difference between worst and rest groups
    relevance = []
    for col in tbl.x:
        w_vals = [r[col.at] for r in worst if r[col.at] != "?"]
        r_vals = [r[col.at] for r in rest if r[col.at] != "?"]
        
        if not w_vals or not r_vals:
            continue
            
        def get_val(col, v):
            res = ezr.norm(col, v)
            return float(res) if isinstance(res, (int, float)) else 0.0

        w_mu = sum(get_val(col, v) for v in w_vals) / len(w_vals)
        r_mu = sum(get_val(col, v) for v in r_vals) / len(r_vals)
        
        relevance.append((abs(w_mu - r_mu), col.txt))
    
    # Sort by relevance ascending (least relevant first)
    relevance.sort(key=lambda x: x[0])
    
    # Return the names of the 3 least relevant input columns
    return [s[1] for s in relevance[:3]]

if __name__ == "__main__":
    filename = "WA_Fn-UseC_-HR-Employee-Attrition.n1000.train.csv"
    data = list(ezr.csv(filename))
    header = data[0]
    rows = data[1:]
    
    results = blame(header, rows)
    
    print("Analysis: Which columns have the least impact on retention/performance?")
    print("The following input columns show the smallest difference between")
    print("the worst-performing records and the rest of the dataset:")
    for i, col_name in enumerate(results, 1):
        print(f"{i}. {col_name}")
    
    print("\nThese columns are candidates for removal as they appear to have")
    print("little correlation with the goal outcomes.")
