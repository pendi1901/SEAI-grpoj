def guess(header, train, test):
    """Estimate YearsAtCompany+ for each test row using tree-based regression on training data."""
    goals = [header[i] for i in range(len(header)) if header[i][-1] in '+-']
    goal_idx = {g: header.index(g) for g in goals}

    def score(row):
        s = 0.0
        for col_name, idx in goal_idx.items():
            v = row[idx]
            if col_name[-1] == '+':
                target = 1.0
            else:
                target = 0.0
            s += abs(v - target)
        return s

    def ydist(row):
        return score(row)

    x_cols = [header[i] for i in range(len(header)) if header[i][-1] not in '+-!X']

    # Build training table structure (minimal, no imports from ezr needed beyond what's allowed)
    class Col:
        def __init__(self, name):
            self.name = name
            self.n = 0
            self.mu = 0.0
            self.m2 = 0.0
            self.sd = 0.0
            self.goal = 1.0 if name[-1] == '+' else 0.0

        def add(self, v):
            self.n += 1
            d = v - self.mu
            self.mu += d / self.n
            self.m2 += (v - self.mu) * (v - self.mu + d * (self.mu - v))
            if self.n >= 2:
                self.sd = (self.m2 / (self.n - 1)) ** 0.5

    class Sym:
        def __init__(self, name):
            self.name = name
            self.n = 0
            self.has = {}

        def add(self, v):
            self.n += 1
            if v not in self.has:
                self.has[v] = 0
            self.has[v] += 1

    class Tbl:
        def __init__(self, header):
            self.names = list(header)
            self.cols = {}
            for i, name in enumerate(self.names):
                if name[-1] == 'X': continue
                c = Col(name) if name[0].isupper() else Sym(name)
                self.cols[i] = c

        def addRow(self, row):
            for i, col in self.cols.items():
                col.add(row[i])

    tbl = Tbl(header)
    for r in train:
        tbl.addRow(r)

    # Simple decision tree implementation
    class Node:
        def __init__(self):
            self.rows = []
            self.at = None
            self.v = None
            self.go = None
            self.kids = []
            self.mu = 0.0

    def cut(tbl, rows):
        ys = [ydist(r) for r in rows]
        best_split = None
        least_score = float('inf')
        for col_idx in tbl.cols:
            col = tbl.cols[col_idx]
            if type(col).__name__ == 'Sym':
                vals = sorted(set(r[col_idx] for r in rows))
                for v in vals[:-1]:
                    go = lambda r, vv=v, ii=col_idx: r[ii] == vv
                    a_rows = [r for r in rows if go(r)]
                    b_rows = [r for r in rows if not go(r)]
                    if len(a_rows) >= 2 and len(b_rows) >= 2:
                        a_ys = [ydist(r) for r in a_rows]
                        b_ys = [ydist(r) for r in b_rows]
                        score = (sum((y - sum(a_ys)/len(a_ys))**2 for y in a_ys) +
                                 sum((y - sum(b_ys)/len(b_ys))**2 for y in b_ys)) / len(rows)
                        if score < least_score:
                            least_score = score
                            best_split = (col_idx, v, go)
            else:
                vals = sorted(r[col_idx] for r in rows)
                k = max(1, len(vals) // 8)
                for i in range(k - 1, len(vals) - 1, k):
                    v = vals[i]
                    go = lambda r, vv=v, ii=col_idx: r[ii] <= vv
                    a_rows = [r for r in rows if go(r)]
                    b_rows = [r for r in rows if not go(r)]
                    if len(a_rows) >= 2 and len(b_rows) >= 2:
                        a_ys = [ydist(r) for r in a_rows]
                        b_ys = [ydist(r) for r in b_rows]
                        score = (sum((y - sum(a_ys)/len(a_ys))**2 for y in a_ys) +
                                 sum((y - sum(b_ys)/len(b_ys))**2 for y in b_ys)) / len(rows)
                        if score < least_score:
                            least_score = score
                            best_split = (col_idx, v, go)
        return best_split

    def tree(tbl, rows):
        node = Node()
        node.rows = list(rows)
        node.mu = sum(ydist(r) for r in rows) / len(rows) if rows else 0.0
        if len(rows) > 4 and (split := cut(tbl, rows)):
            col_idx, v, go = split
            yes = [r for r in rows if go(r)]
            no = [r for r in rows if not go(r)]
            node.at = col_idx
            node.v = v
            node.go = go
            node.kids = [tree(tbl, yes), tree(tbl, no)]
        return node

    train_tree = tree(tbl, train)

    def predict(row):
        node = train_tree
        while node.kids:
            if node.go(row):
                node = node.kids[0]
            else:
                node = node.kids[1]
        return node.mu

    results = [predict(r) for r in test]
    return results
