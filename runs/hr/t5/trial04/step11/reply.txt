import ezr
import random

def remember(header, rows):
    """
    Finds the 5% of rows most similar to the 'best' performing employees.
    'Best' is defined by minimizing ydist (distance to optimal goals).
    """
    # Create a table to leverage ezr's ydist calculations
    tbl = ezr.Tbl([header] + rows)
    
    # Calculate ydist for all rows. 
    # ydist measures how far a row is from the ideal goal values.
    # We want to minimize ydist.
    scored = []
    for i, row in enumerate(rows):
        d = ezr.ydist(tbl, row)
        scored.append((d, i))
    
    # Sort by distance (ascending) to find the "best" rows
    scored.sort()
    
    # Return the top 5% of rows (or at least 1)
    limit = max(1, int(len(rows) * 0.05))
    return [item[1] for item in scored[:limit]]

if __name__ == "__main__":
    # Load data
    filename = "WA_Fn-UseC_-HR-Employee-Attrition.n1000.train.csv"
    data = list(ezr.csv(filename))
    header = data[0]
    rows = data[1:]
    
    # Get indices of the best rows
    best_indices = remember(header, rows)
    
    # Print a summary for the customer
    print(f"Analysis of {len(rows)} employee records.")
    print(f"Identified {len(best_indices)} employees with the most optimal working arrangements.")
    
    # Display details of the top 3 identified records
    print("\nTop 3 records most similar to ideal performance:")
    for i, idx in enumerate(best_indices[:3]):
        row = rows[idx]
        # YearsAtCompany+ (31), YearsInCurrentRole+ (32), YearsWithCurrManager+ (34)
        print(f"{i+1}. Row {idx}: YearsAtCompany={row[31]}, YearsInRole={row[32]}, YearsWithManager={row[34]}")
    
    print("\nThese employees represent the top 5% of the dataset based on goal metrics.")
