import ezr
import random

def race(header, rows, label):
    """
    Compares three strategies for finding the best employee record
    within a shared budget of 50 labels.
    """
    # Shared budget management
    labeled_indices = set()
    
    def get_label(idx):
        if len(labeled_indices) >= 50 and idx not in labeled_indices:
            raise Exception("Budget exceeded")
        labeled_indices.add(idx)
        return label(idx)

    # Strategy 1: Random
    def strategy_random(tbl):
        indices = random.sample(range(len(rows)), 16)
        for i in indices:
            get_label(i)
        return min(indices, key=lambda i: ezr.ydist(tbl, rows[i]))

    # Strategy 2: Active Learning (ezr.acquire)
    def strategy_active(tbl):
        # We wrap the label function to track budget
        def oracle_wrapper(row):
            # Find index by matching row object
            idx = next(i for i, r in enumerate(rows) if r is row)
            true_vals = get_label(idx)
            for i, name in enumerate(header):
                if name.endswith('+') or name.endswith('-'):
                    row[i] = true_vals[i]
            return row
        
        labeled = ezr.acquire(tbl, cap=17, label=oracle_wrapper)
        return min(range(len(rows)), key=lambda i: ezr.ydist(tbl, rows[i]))

    # Strategy 3: Greedy (Top 17 by initial heuristic)
    def strategy_greedy(tbl):
        # Sort by initial distance (using current knowns)
        scored = sorted(range(len(rows)), key=lambda i: ezr.ydist(tbl, rows[i]))
        for i in scored[:17]:
            get_label(i)
        return min(scored[:17], key=lambda i: ezr.ydist(tbl, rows[i]))

    # Run race
    tbl = ezr.Tbl([header] + rows)
    results = {
        "random": strategy_random(tbl),
        "active": strategy_active(tbl),
        "greedy": strategy_greedy(tbl)
    }
    
    # Determine winner (lowest ydist)
    winner = min(results, key=lambda name: ezr.ydist(tbl, rows[results[name]]))
    
    return {
        "winner": winner,
        "row": results[winner],
        "tried": ["random", "active", "greedy"]
    }

if __name__ == "__main__":
    filename = "WA_Fn-UseC_-HR-Employee-Attrition.n1000.train.csv"
    data = list(ezr.csv(filename))

                row[i] = "?"
                
    # Mock label function
    full_data = list(ezr.csv(filename))
    def mock_label(i):
        return full_data[i+1]
    
    result = race(header, rows, mock_label)
    
    print("Race Results:")
    print(f"Winner: {result['winner']}")
    print(f"Best row found at index: {result['row']}")
    print(f"Strategies compared: {', '.join(result['tried'])}")
    print("\nThe active learning strategy typically outperforms random sampling by focusing on high-value areas.")
