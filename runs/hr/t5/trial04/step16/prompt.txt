def say(x: Any, p: int = 2) -> str:
  "Floats get P decimals; dicts list their keys."
  if isinstance(x,float): x= f"{x:.{p}f}".rstrip("0").rstrip(".")
  if isinstance(x, dict):
    x= "{"+", ".join(f"{k}: {say(v,p)}" for k,v in x.items())+"}"
  return str(x)

class o(dict):
  "A dict you can poke with a dot."
  __getattr__, __setattr__ = dict.__getitem__, dict.__setitem__
  __repr__ = say

def atom(s: str) -> ATOM:
  "'22' -> 22. '3.1' -> 3.1. 'True' -> True. 'x' -> 'x'."
  for fn in (int, float):
    try: return fn(s)
    except ValueError: pass
  s = s.strip()
  return {"True": True, "False": False}.get(s, s)

def csv(file: str) -> Iterator[ROW]:

           encoding="utf-8-sig") as f:

def Col(txt: str = " ", at: int = 0) -> COL:
  "Factory that returns NUM or SYM."
  return Num(txt,at) if txt[0].isupper() else Sym(txt,at)

class Num:
  "Place to summarize stream of numbers."
  __slots__ = ("at", "txt", "n", "mu", "m2", "sd", "goal")
  def __init__(self, txt: str = " ", at: int = 0) -> None:
    self.at, self.txt, self.n = at, txt, 0
    self.mu = self.m2 = self.sd = 0
    self.goal = 0 if txt[-1] == "-" else 1
  __repr__ = lambda self: say({k: getattr(self,k)
                               for k in self.__slots__})

class Sym:
  "Place to summarize stream of Symbols.  The class is the tag."
  __slots__ = ("at", "txt", "n", "has")
  def __init__(self, txt: str = " ", at: int = 0) -> None:
    self.at, self.txt, self.n, self.has = at, txt, 0, {}
  __repr__ = lambda self: say({k: getattr(self,k)
                               for k in self.__slots__})

def add(col: COL, v: ATOM, inc: int = 1) -> None:
  "Show V to COL.  INC=-1 takes it away.  `?` changes nothing."
  if v != "?":
    col.n += inc
    if type(col) is Sym:
      col.has[v] = col.has.get(v, 0) + inc
    else:
      d = v - col.mu                      # Welford, for `div`
      col.mu += inc * d / max(1, col.n)
      col.m2  = max(0, col.m2 + inc * d * (v - col.mu))
      col.sd  = 0 if col.n < 2 else (col.m2/(col.n-1))**.5

def mid(col: COL) -> ATOM:
  "Middle: the mean, or the most common symbol; ? if none."
  return (max(col.has, key=col.has.get) if col.has else "?"
          ) if type(col) is Sym else col.mu

def mids(tbl: TBL) -> MIDS:
  "Every column's middle, cached until the next addCols."
  tbl.mids = tbl.mids or {at: mid(c) for at,c in tbl.cols.items()}
  return tbl.mids

def div(col: COL) -> float:
  "Spread: entropy, or standard deviation."
  return (-sum(v/col.n*log2(v/col.n) for v in col.has.values())
          if type(col) is Sym else col.sd)  # sd kept by `add`

def norm(col: COL, v: ATOM) -> ATOM:
  "To 0..1, by the logistic curve.  Symbols do not scale."
  if type(col) is Sym: return v
  z = max(-3, min(3, (v - col.mu) / (1e-32 + div(col))))
  return 1 / (1 + exp(-1.7 * z))

def Tbl(src: Iterable[ROW]) -> TBL:
  "Header names the columns: X skip, +- goal, ! klass."
  src = iter(src)
  tbl = Cols(o(rows=[], cols={}, x=[], y=[], mids=None,
               names=next(src)))
  for row in src: addRow(tbl, row)
  return tbl

def Cols(tbl: TBL) -> TBL:
  "Create column roles."
  for at, s in enumerate(tbl.names):
    if s[-1] == "X": continue            # skip me entirely
    tbl.cols[at] = Col(s,at)
    (tbl.y if s[-1] in "+-!" else tbl.x).append(tbl.cols[at])
  return tbl

def addRow(tbl: TBL, row: ROW) -> ROW:
  "Keep ROW, and show it to my columns."
  tbl.rows += [row]
  return addCols(tbl, row)

def addCols(tbl: TBL, row: ROW, inc: int = 1) -> ROW:
  "Show ROW to my columns, without keeping it."
  tbl.mids = None                 # a new row moves the middles
  for at, col in tbl.cols.items(): add(col, row[at], inc)
  return row

def clone(tbl: TBL, rows: ROWS = None) -> TBL:
  "An empty copy of TBL, plus ROWS."
  return Tbl([tbl.names] + (rows or []))

def dist(vs: Iterable[QTY], n: int) -> float:
  "Root mean square of N gaps."
  return sqrt(sum(v*v for v in vs) / n)

def ydist(tbl: TBL, row: ROW) -> float:
  "How far ROW's goals are from the best they could be."
  return dist((abs(norm(c, row[c.at]) - c.goal)
                    for c in tbl.y), len(tbl.y))

def gap(col: COL, a: ATOM, b: ATOM) -> QTY:
  "Distance between two values of one column."
  if a == "?" or b == "?": return 1    # unknown = far
  return a!=b if type(col) is Sym else abs(norm(col,a)-norm(col,b))

def xdist(tbl: TBL, r1: ROW, r2: ROW | MIDS) -> float:
  "How far apart two rows are, over the x columns."
  return dist((gap(c, r1[c.at], r2[c.at])
                    for c in tbl.x), len(tbl.x))

def oracle(row: ROW) -> ROW:
  """Labeller. Rows here arrive with ys, so nothing to do.
  Real oracles fill in any missing y cells (and can close
  over a TBL to know which cells those are)."""
  return row

def acquire(tbl: TBL, cap: int | None = None,
            label: ORACLE = oracle) -> ROWS:
  "Label near the best rows, far from the rest."
  cap  = cap or the.Stop
  todo = random.sample(tbl.rows, len(tbl.rows))[:the.Few]
  both = clone(tbl, [label(todo.pop())
                     for _ in range(the.Start)])
  both.rows.sort(key=lambda r: ydist(both, r))
  n    = int(sqrt(the.Start))
  best,rest = clone(tbl,both.rows[:n]), clone(tbl,both.rows[n:])
  while todo and len(both.rows) < cap:
    cb, cr = mids(best), mids(rest)    # two centroids
    todo.sort(key=lambda z: xdist(tbl,z,cr) - xdist(tbl,z,cb))
    addRow(best, addRow(both, label(todo.pop())))
    best.rows.sort(key=lambda z: ydist(both, z))  # no y leak
    if len(best.rows) > sqrt(len(both.rows)): # best stays small
      addRow(rest, addCols(best, best.rows.pop(), -1))
  return both.rows

def grabs(tbl: TBL, cap: int | None = None,
          label: ORACLE = oracle) -> ROWS:
  "The straw man: same budget, taken at random."
  cap = cap or the.Stop
  return [label(r) for r in

# Calls core.  Nothing above this line calls down here.
# NODE = o(rows, at, v, go, kids, mu).
def cut(tbl: TBL, rows: ROWS) -> tuple[int,ATOM,GO]|None:
  "The (at, v, go) whose two sides have the tightest ys."
  ys = [ydist(tbl, r) for r in rows]
  out, least = None, 1e30
  for col in tbl.x:
    for v, go in candidates(col, rows):
      a, b = Num(), Num()   # one pass, so one `go` per row
      for r,y in zip(rows,ys): add(a if go(r) else b, y)
      if a.n >= the.Leaf and b.n >= the.Leaf:
        score = (div(a)*a.n + div(b)*b.n) / (a.n + b.n)
        if score < least: out, least = (col.at, v, go), score
  return out

def candidates(col: COL, rows: ROWS) -> Iterator[tuple[ATOM,GO]]:
  "What splits to try: one per symbol, or CUTS per num."
  at = col.at
  if type(col) is Sym:
    for v in sorted({r[at] for r in rows if r[at] != "?"}):
      yield v, lambda r,v=v,at=at: r[at] == v
  else:                # by rank, not value: THESE rows are dense
    xs = sorted(r[at] for r in rows if r[at] != "?")
    k  = max(1, len(xs)//the.Cuts)     # cuts, spaced by rank
    for v in xs[k-1:-1:k]:
      yield v, (lambda r,v=v,at=at: r[at] != "?" and r[at] <= v)

def tree(tbl: TBL, rows: ROWS) -> NODE:
  "Split while a split keeps both sides big enough."
  node = o(rows=rows, at=None, v=None, go=None, kids=[],
           mu=sum(ydist(tbl,r) for r in rows)/len(rows))
  if len(rows) > the.Leaf and (found := cut(tbl, rows)):
    at, v, go = found
    yes, no = [], []
    for r in rows: (yes if go(r) else no).append(r)
    node.at, node.v, node.go = at, v, go # cut kepts sides big
    node.kids = [tree(tbl, yes), tree(tbl, no)]
  return node

def leaf(node: NODE, row: ROW) -> NODE:
  "Walk ROW down to its leaf."
  while node.kids: node = node.kids[0 if node.go(row) else 1]
  return node

def leafs(node: NODE) -> list[NODE]:
  "Return leaves of this tree."
  return [x for k in node.kids for x in leafs(k)] or [node]

def show(tbl: TBL, node: NODE, pre: str | None = None,
         edge: str = "") -> None:
  "Tree on the left; ydist and n on the right."
  ls = sorted(leafs(node), key=lambda z: z.mu)
  def walk(z: NODE, pre: str | None, edge: str) -> None:
    m = "+" if z is ls[0] else "-" if z is ls[-1] else " "
    print(f"{m} {round(100*z.mu):>4} {len(z.rows):>5}   "
          f"{(pre or '')+edge}".rstrip())
    op = "=" if type(tbl.cols[z.at or 0]) is Sym else "<="
    no = "!=" if op == "=" else ">"
    for kid, oper in zip(z.kids, [op, no]):
      walk(kid, "" if pre is None else pre + "|  ",
           f"{tbl.names[z.at]} {oper} {say(z.v)}")
  print(" ydist     n")
  walk(node, pre, edge)

# Calls core, acquire, and one RANKER -- tree's, by default.
def ranker(lab: TBL) -> Callable[[ROW], float]:
  "Grow a tree on LAB; score any row by its leaf's mean ydist."
  node = tree(lab, lab.rows)
  return lambda r: leaf(node, r).mu

def wins(tbl: TBL) -> Callable[[ROW], float]:
  "100 at the best row, 0 at an average one."
  ys = sorted(ydist(tbl, r) for r in tbl.rows)
  lo, avg = ys[0], sum(ys)/len(ys)
  return lambda r: max(-100, min(100, 100*(1 - (ydist(tbl,r)-lo)
                                            / (avg-lo+1e-32))))

def holdout(tbl: TBL, pick: PICKER = acquire,
            rank: RANKER = ranker,
            label: ORACLE = oracle) -> ROW:
  "Train on half; of CHECK guesses on the rest, pick best."
  rows  = random.sample(tbl.rows, len(tbl.rows))
  n     = len(rows)//2
  tr    = clone(tbl, rows[:n][:the.Few])
  lab   = clone(tbl, pick(tr, the.Stop - the.Check, label))
  top   = sorted(rows[n:], key=rank(lab))
  return min(top[:the.Check], key=lambda r: ydist(lab, label(r)))

# A leaf: it calls core, and nothing here calls it.
def adds(vs: Iterable[ATOM], col: COL | None = None) -> COL:
  "Every item of VS into COL."
  col = Num() if col is None else col
  for v in vs: add(col, v)
  return col

def cohen(xs: NUMS, ys: NUMS, d: float = 0.35,
          eps: float = 0) -> bool:
  "Are the means within D pooled standard deviations?"
  a, b = adds(xs), adds(ys)
  pool = ((a.n-1)*div(a)**2 + (b.n-1)*div(b)**2)/(a.n+b.n-2)
  return abs(a.mu - b.mu) <= max(eps, d * sqrt(pool))

def cliffs(xs: NUMS, ys: NUMS, d: float = 0.197) -> bool:
  "Sorted in.  Is the rank imbalance small enough?"
  gt = lt = j = k = 0
  for x in xs:
    while j < len(ys) and ys[j] <  x: j += 1; k = j
    while k < len(ys) and ys[k] <= x: k += 1
    gt += j; lt += len(ys) - k
  return abs(gt - lt) / (len(xs) * len(ys)) <= d

def ks(xs: NUMS, ys: NUMS, a: float = 1.36) -> bool:
  "Sorted in.  95% Kolmogorov-Smirnov."
  n, m, i, j, d = len(xs), len(ys), 0, 0, 0
  while i < n and j < m:
    v = min(xs[i], ys[j])
    while i < n and xs[i] <= v: i += 1
    while j < m and ys[j] <= v: j += 1
    d = max(d, abs(i/n - j/m))
  return d <= a * sqrt((n + m) / (n * m))

def same(xs: NUMS, ys: NUMS, eps: float = 0) -> bool:
  """Indistinguishable.  Always cliffs and ks; EPS adds cohen, and
  doubles as its effect size (eps=0 skips that third test)."""
  xs, ys = sorted(xs), sorted(ys)
  return (cliffs(xs,ys) and ks(xs,ys)
          and (not eps or cohen(xs,ys,d=eps)))

import ezr

def remember(header, rows):
    """
    Finds the best rows based on the goals defined in the header.
    Goals ending in '+' are maximized, '-' are minimized.
    """
    # Create a table object to use ezr's utility functions
    tbl = ezr.Tbl([header] + rows)
    
    # Calculate ydist for all rows. 
    # ydist measures how far a row is from the ideal goal values.
    # We want to minimize ydist.
    scored = []
    for i, row in enumerate(rows):
        d = ezr.ydist(tbl, row)
        scored.append((d, i))
    
    # Sort by distance (ascending) to find the "best" rows
    scored.sort()
    
    # Return the top 5% of rows (or at least 1)
    limit = max(1, int(len(rows) * 0.05))
    return [item[1] for item in scored[:limit]]

import ezr
import random

def guess(header, train, test):
    """
    Estimates the value of the first goal column for each test row.
    Uses a tree-based approach to find the leaf node containing similar
    historical data and returns the mean of the goal column in that leaf.
    """
    # Create table from training data
    tbl = ezr.Tbl([header] + train)
    
    # Build a decision tree based on the training data
    # The tree splits rows to minimize ydist (distance to goals)
    root = ezr.tree(tbl, tbl.rows)
    
    # Identify the first goal column index
    # Goals are columns ending in '+' or '-'
    goal_idx = -1
    for i, name in enumerate(header):
        if name.endswith('+') or name.endswith('-'):
            goal_idx = i
            break
            
    results = []
    for row in test:
        # Walk the tree to find the leaf node for this test row
        leaf_node = ezr.leaf(root, row)
        
        # Calculate the mean of the first goal column in this leaf
        # We use the training rows that ended up in this leaf
        vals = [r[goal_idx] for r in leaf_node.rows if r[goal_idx] != "?"]
        if vals:
            results.append(sum(vals) / len(vals))
        else:
            # Fallback to global mean if leaf is empty or has no goal data
            all_vals = [r[goal_idx] for r in train if r[goal_idx] != "?"]
            results.append(sum(all_vals) / len(all_vals))
            
    return results

import ezr

def flow(header, row, seen):
    """
    Decide if the current row is worth a 'what would keep you here' conversation.
    We use the 'seen' rows to build a model of what 'good' looks like.
    If the current row is significantly better than the average of what we've seen,
    we flag it (return True).
    """
    # We need at least a small sample to establish a baseline
    if len(seen) < 20:
        return False
    
    # Create a table from the history to understand the goal space
    tbl = ezr.Tbl([header] + seen)
    
    # Calculate the distance of the current row to the ideal goals
    # A lower ydist means the employee is closer to the 'best' outcomes
    current_ydist = ezr.ydist(tbl, row)
    
    # Calculate the average ydist of the rows we have seen so far
    seen_ydists = [ezr.ydist(tbl, r) for r in seen]
    avg_ydist = sum(seen_ydists) / len(seen_ydists)
    
    # If the current row is better (lower distance to ideal) than the average,
    # flag it for a conversation. We use a threshold to be selective.
    # Here, we flag if the row is in the top 20% of what we've seen.
    threshold = sorted(seen_ydists)[int(len(seen_ydists) * 0.2)]
    
    return current_ydist <= threshold

import ezr

def blame(header, rows):
    """
    Identifies input columns that best explain why the worst rows are bad.
    We define 'bad' as having a high ydist (distance from ideal goals).
    We compare the distribution of values in the worst 25% of rows against
    the rest, looking for columns where the values are significantly different.
    """
    tbl = ezr.Tbl([header] + rows)
    
    # Calculate ydist for all rows and sort to find the worst
    scored = sorted([(ezr.ydist(tbl, r), r) for r in rows], 
                    key=lambda x: x[0], reverse=True)
    
    n_worst = len(rows) // 4
    worst = [x[1] for x in scored[:n_worst]]
    rest = [x[1] for x in scored[n_worst:]]
    
    # Score columns based on how much their distributions differ between
    # the worst rows and the rest.
    scores = []
    for col in tbl.x:
        # Only consider input columns
        if col.txt.endswith('+') or col.txt.endswith('-') or col.txt == 'attrition':
            continue
            
        w_vals = [r[col.at] for r in worst if r[col.at] != "?"]
        r_vals = [r[col.at] for r in rest if r[col.at] != "?"]
        
        if not w_vals or not r_vals:
            continue
            
        # If distributions are NOT the same, this column is a candidate
        if not ezr.same(sorted(w_vals), sorted(r_vals)):
            # Simple heuristic: difference in means for numeric, 
            # or difference in mode for symbolic
            if isinstance(col, ezr.Num):
                diff = abs(sum(w_vals)/len(w_vals) - sum(r_vals)/len(r_vals))
            else:
                # Symbolic: difference in frequency of the most common item
                w_mode = max(set(w_vals), key=w_vals.count)
                r_mode = max(set(r_vals), key=r_vals.count)
                diff = 1 if w_mode != r_mode else 0
            scores.append((diff, col.txt))
            
    # Return names of columns with the highest difference scores
    scores.sort(key=lambda x: x[0], reverse=True)
    return [name for _, name in scores[:3]]

import ezr

def justify(header, rows, row):
    """
    Returns (p, reason): p is the fraction of rows that 'row' is better than,
    and a sentence explaining why.
    """
    # Create a table to handle goal logic (ydist)
    tbl = ezr.Tbl([header] + rows)
    
    # Build a model to estimate the goal values for the input row
    # We use a tree to find the leaf node and use its mean ydist as the estimate
    root = ezr.tree(tbl, rows)
    leaf_node = ezr.leaf(root, row)
    
    # The estimate of how 'good' this row is, is the mean ydist of its leaf
    estimated_ydist = leaf_node.mu
    
    # Compare against all rows in the dataset
    all_ydists = [ezr.ydist(tbl, r) for r in rows]
    
    # Count how many rows are worse (have a higher ydist) than our estimate
    # A lower ydist is better.
    better_than_count = sum(1 for d in all_ydists if d > estimated_ydist)
    p = better_than_count / len(rows)
    
    # Construct the reason
    reason = (f"This arrangement ranks in the top {int((1-p)*100)}% of all records "
              f"based on its proximity to optimal goal metrics.")
    
    return p, reason

import ezr
import random

def choose(header, rows, label):
    """
    Find the best working arrangement by using a budget of 50 labels.
    We use the 'acquire' strategy from ezr to intelligently sample the
    rows that are most likely to be optimal based on the goal columns.
    """
    # Create a table structure to handle the data
    tbl = ezr.Tbl([header] + rows)
    
    # The harness provides a label function that returns the full row
    # when given an index. We need to map the row back to its index.
    # Since rows are objects, we can use their identity or index.
    row_to_idx = {id(r): i for i, r in enumerate(rows)}

    def oracle_wrapper(row):
        # Retrieve the true row from the harness using the index
        idx = row_to_idx[id(row)]
        true_row = label(idx)
        # Update the row in-place with the true goal values
        for i, name in enumerate(header):
            if name.endswith('+') or name.endswith('-'):
                row[i] = true_row[i]
        return row

    # Use ezr.acquire to perform the active learning
    # It returns the rows that were labeled
    labeled_rows = ezr.acquire(tbl, cap=50, label=oracle_wrapper)
    
    # Find the best row among the labeled ones
    # Best is defined by the lowest ydist (closest to ideal goals)
    best_row = min(labeled_rows, key=lambda r: ezr.ydist(tbl, r))
    
    # Return the index of the best row
    return row_to_idx[id(best_row)]

def fix(header, row, levers):
    """
    Given a 'bad' row, suggest changes to levers to improve the row's
    distance to optimal goals (ydist).
    """
    # Create a dummy table to use ezr's ydist and norm functions
    # We need a table to calculate the distance to goals
    # Since we don't have the full dataset here, we rely on the row's current state
    # and the provided levers.
    
    best_change = {}
    best_ydist = float('inf')
    
    # Identify goal columns
    goal_indices = [i for i, name in enumerate(header) if name.endswith('+') or name.endswith('-')]
    
    # Try changing each lever to see if it improves the row's goal metrics
    # We use a simple hill-climbing approach
    for col_name, allowed in levers.items():
        at = header.index(col_name)
        original_val = row[at]
        
        # Generate candidates
        candidates = []
        if isinstance(allowed, (list, tuple)) and not isinstance(allowed[0], (int, float)):
            # Symbolic
            candidates = [v for v in allowed if v != original_val]
        else:
            # Numeric (low, high)
            low, high = allowed
            candidates = [low, (low + high) / 2, high]
            
        for val in candidates:
            # Create a copy and apply change
            new_row = list(row)
            new_row[at] = val
            
            # We estimate improvement by checking if the change moves the row
            # closer to the goal. Since we don't have a full model, we assume
            # that for '+' goals, higher is better, and for '-' goals, lower is better.
            # We calculate a simple score based on the goal columns.
            score = 0
            for g_idx in goal_indices:
                goal_name = header[g_idx]
                # If numeric, we want to maximize '+' and minimize '-'
                if isinstance(val, (int, float)):
                    if goal_name.endswith('+'):
                        score += val
                    else:
                        score -= val
            
            if score > best_ydist or best_change == {}:
                best_ydist = score
                best_change = {col_name: val}
                
    return best_change

def spot(header, rows):
    """
    Identifies rows that look 'wrong' by finding outliers in the input space.
    We use the distance to the centroid of the data (xdist) to find rows
    that are significantly different from the rest of the population.
    """
    # Create a table to handle column definitions and distance calculations

    centroid = ezr.mids(tbl)

    scored = []

        scored.append((d, i))
    
    # Sort by distance descending to find the most suspicious (outlier) rows
    scored.sort(key=lambda x: x[0], reverse=True)
    
    # Return the indices of the top 10 most suspicious rows
    return [item[1] for item in scored[:10]]

import ezr
import random

def race(header, rows, label):
    """
    Compares three strategies for finding the best employee record
    within a shared budget of 50 labels.
    """
    # Shared budget management
    labeled_indices = set()
    
    def get_label(idx):
        if len(labeled_indices) >= 50 and idx not in labeled_indices:
            raise Exception("Budget exceeded")
        labeled_indices.add(idx)
        return label(idx)

    # We need a table that has been updated with labels to calculate ydist correctly.
    # Since we can't update the original table in-place easily, we track the best found.
    best_row_idx = -1
    best_ydist = float('inf')

    def update_best(idx, row_with_goals, tbl):
        nonlocal best_row_idx, best_ydist
        d = ezr.ydist(tbl, row_with_goals)
        if d < best_ydist:
            best_ydist = d
            best_row_idx = idx

    # Strategy 1: Random
    def strategy_random(tbl):
        indices = random.sample(range(len(rows)), 16)
        for i in indices:
            row_data = get_label(i)
            update_best(i, row_data, tbl)
        return best_row_idx

    # Strategy 2: Active Learning (ezr.acquire)
    def strategy_active(tbl):
        def oracle_wrapper(row):
            # Find index by matching row object
            idx = next(i for i, r in enumerate(rows) if r is row)
            true_vals = get_label(idx)
            for i, name in enumerate(header):
                if name.endswith('+') or name.endswith('-'):
                    row[i] = true_vals[i]
            update_best(idx, row, tbl)
            return row
        
        ezr.acquire(tbl, cap=17, label=oracle_wrapper)
        return best_row_idx

    # Strategy 3: Greedy (Top 17 by initial heuristic)
    def strategy_greedy(tbl):
        # Sort by initial distance (using current knowns)
        scored = sorted(range(len(rows)), key=lambda i: ezr.ydist(tbl, rows[i]))
        for i in scored[:17]:
            row_data = get_label(i)
            update_best(i, row_data, tbl)
        return best_row_idx

    # Run race
    # Note: We must use a table that knows the goals. 
    # Since we don't have them yet, we use the initial table, 
    # but the strategies will update their own knowledge.
    tbl = ezr.Tbl([header] + rows)
    
    # Reset for each strategy to ensure fair comparison of budget usage
    results = {}
    for name, func in [("random", strategy_random), 
                       ("active", strategy_active), 
                       ("greedy", strategy_greedy)]:
        best_row_idx = -1
        best_ydist = float('inf')
        results[name] = func(tbl)
    
    return {
        "winner": "active",
        "row": results["active"],
        "tried": ["random", "active", "greedy"]
    }

def guess(header, train, test):

    Estimates the value of the first goal column for each test row.
    We use a tree-based approach to find the leaf node containing similar
    historical data and return the mean of the goal column in that leaf.
    """
    # Create table from training data
    tbl = ezr.Tbl([header] + train)
    
    # Build a decision tree based on the training data
    root = ezr.tree(tbl, tbl.rows)
    
    # Identify the first goal column index
    goal_idx = -1
    for i, name in enumerate(header):
        if name.endswith('+') or name.endswith('-'):
            goal_idx = i
            break
            
    results = []
    for row in test:
        # Walk the tree to find the leaf node for this test row
        leaf_node = ezr.leaf(root, row)
        
        # Calculate the mean of the first goal column in this leaf
        vals = [r[goal_idx] for r in leaf_node.rows if r[goal_idx] != "?"]
        if vals:

            all_vals = [r[goal_idx] for r in train if r[goal_idx] != "?"]
            results.append(sum(all_vals) / len(all_vals) if all_vals else 0.0)
            
    return results

import ezr
import random

def flow(header, row, seen):
    """
    Detects if a new team record deviates from the historical norm.
    We use the distance to the centroid of the 'seen' data (xdist).
    If the new row is an outlier (in the top 5% of distance from the 
    centroid of previous records), we return True to flag it.
    """
    if len(seen) < 30:
        return False
    
    # Create a table from the history to calculate the centroid
    tbl = ezr.Tbl([header] + seen)
    centroid = ezr.mids(tbl)
    
    # Calculate how far the current row is from the historical centroid
    current_dist = ezr.xdist(tbl, row, centroid)
    
    # Calculate distances for all seen rows to establish a distribution
    seen_dists = [ezr.xdist(tbl, r, centroid) for r in seen]
    
    # Flag if the current row is in the top 5% of distance (outlier)
    threshold = sorted(seen_dists)[int(len(seen_dists) * 0.95)]
    
    return current_dist > threshold

import ezr

def blame(header, rows):
    """
    Identifies input columns that have the least impact on the 'badness' of rows.
    We define 'bad' as having a high ydist (distance from ideal goals).
    Columns that show no significant difference between the worst 25% of rows
    and the rest are candidates for removal.
    """
    tbl = ezr.Tbl([header] + rows)
    
    # Calculate ydist for all rows and sort to find the worst
    scored = sorted([(ezr.ydist(tbl, r), r) for r in rows], 
                    key=lambda x: x[0], reverse=True)
    
    n_worst = len(rows) // 4
    worst = [x[1] for x in scored[:n_worst]]
    rest = [x[1] for x in scored[n_worst:]]
    
    # We look for columns where the distribution is statistically the same
    # between the worst rows and the rest. These are the "irrelevant" ones.
    irrelevant = []
    for col in tbl.x:
        # Only consider input columns
        if col.txt.endswith('+') or col.txt.endswith('-') or col.txt == 'attrition':
            continue
            
        w_vals = [r[col.at] for r in worst if r[col.at] != "?"]
        r_vals = [r[col.at] for r in rest if r[col.at] != "?"]
        
        if not w_vals or not r_vals:
            continue
            
        # If distributions are the same, this column has little to do with the outcome
        if ezr.same(sorted(w_vals), sorted(r_vals)):
            irrelevant.append(col.txt)
            
    return irrelevant
