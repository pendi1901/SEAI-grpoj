def say(x: Any, p: int = 2) -> str:
  "Floats get P decimals; dicts list their keys."
  if isinstance(x,float): x= f"{x:.{p}f}".rstrip("0").rstrip(".")
  if isinstance(x, dict):
    x= "{"+", ".join(f"{k}: {say(v,p)}" for k,v in x.items())+"}"
  return str(x)

class o(dict):
  "A dict you can poke with a dot."
  __getattr__, __setattr__ = dict.__getitem__, dict.__setitem__
  __repr__ = say

def atom(s: str) -> ATOM:
  "'22' -> 22. '3.1' -> 3.1. 'True' -> True. 'x' -> 'x'."
  for fn in (int, float):
    try: return fn(s)
    except ValueError: pass
  s = s.strip()
  return {"True": True, "False": False}.get(s, s)

def csv(file: str) -> Iterator[ROW]:
  "ROWS of FILE, each cell coerced."
  with open(os.path.expanduser(file),  # -sig drops any BOM
           encoding="utf-8-sig") as f:
    for line in f:
      if line.strip(): yield [atom(s) for s in line.split(",")]

def Col(txt: str = " ", at: int = 0) -> COL:
  "Factory that returns NUM or SYM."
  return Num(txt,at) if txt[0].isupper() else Sym(txt,at)

class Num:
  "Place to summarize stream of numbers."
  __slots__ = ("at", "txt", "n", "mu", "m2", "sd", "goal")
  def __init__(self, txt: str = " ", at: int = 0) -> None:
    self.at, self.txt, self.n = at, txt, 0
    self.mu = self.m2 = self.sd = 0
    self.goal = 0 if txt[-1] == "-" else 1
  __repr__ = lambda self: say({k: getattr(self,k)
                               for k in self.__slots__})

class Sym:
  "Place to summarize stream of Symbols.  The class is the tag."
  __slots__ = ("at", "txt", "n", "has")
  def __init__(self, txt: str = " ", at: int = 0) -> None:
    self.at, self.txt, self.n, self.has = at, txt, 0, {}
  __repr__ = lambda self: say({k: getattr(self,k)
                               for k in self.__slots__})

def add(col: COL, v: ATOM, inc: int = 1) -> None:
  "Show V to COL.  INC=-1 takes it away.  `?` changes nothing."
  if v != "?":
    col.n += inc
    if type(col) is Sym:
      col.has[v] = col.has.get(v, 0) + inc
    else:
      d = v - col.mu                      # Welford, for `div`
      col.mu += inc * d / max(1, col.n)
      col.m2  = max(0, col.m2 + inc * d * (v - col.mu))
      col.sd  = 0 if col.n < 2 else (col.m2/(col.n-1))**.5

def mid(col: COL) -> ATOM:
  "Middle: the mean, or the most common symbol; ? if none."
  return (max(col.has, key=col.has.get) if col.has else "?"
          ) if type(col) is Sym else col.mu

def mids(tbl: TBL) -> MIDS:
  "Every column's middle, cached until the next addCols."
  tbl.mids = tbl.mids or {at: mid(c) for at,c in tbl.cols.items()}
  return tbl.mids

def div(col: COL) -> float:
  "Spread: entropy, or standard deviation."
  return (-sum(v/col.n*log2(v/col.n) for v in col.has.values())
          if type(col) is Sym else col.sd)  # sd kept by `add`

def norm(col: COL, v: ATOM) -> ATOM:
  "To 0..1, by the logistic curve.  Symbols do not scale."
  if type(col) is Sym: return v
  z = max(-3, min(3, (v - col.mu) / (1e-32 + div(col))))
  return 1 / (1 + exp(-1.7 * z))

#-- tables -------------------------------------------------
def Tbl(src: Iterable[ROW]) -> TBL:
  "Header names the columns: X skip, +- goal, ! klass."
  src = iter(src)
  tbl = Cols(o(rows=[], cols={}, x=[], y=[], mids=None,
               names=next(src)))
  for row in src: addRow(tbl, row)
  return tbl

def Cols(tbl: TBL) -> TBL:
  "Create column roles."
  for at, s in enumerate(tbl.names):
    if s[-1] == "X": continue            # skip me entirely
    tbl.cols[at] = Col(s,at)
    (tbl.y if s[-1] in "+-!" else tbl.x).append(tbl.cols[at])
  return tbl

def addRow(tbl: TBL, row: ROW) -> ROW:
  "Keep ROW, and show it to my columns."
  tbl.rows += [row]
  return addCols(tbl, row)

def addCols(tbl: TBL, row: ROW, inc: int = 1) -> ROW:
  "Show ROW to my columns, without keeping it."
  tbl.mids = None                 # a new row moves the middles
  for at, col in tbl.cols.items(): add(col, row[at], inc)
  return row

def clone(tbl: TBL, rows: ROWS = None) -> TBL:
  "An empty copy of TBL, plus ROWS."
  return Tbl([tbl.names] + (rows or []))

def dist(vs: Iterable[QTY], n: int) -> float:
  "Root mean square of N gaps."
  return sqrt(sum(v*v for v in vs) / n)

def ydist(tbl: TBL, row: ROW) -> float:
  "How far ROW's goals are from the best they could be."
  return dist((abs(norm(c, row[c.at]) - c.goal)
                    for c in tbl.y), len(tbl.y))

def gap(col: COL, a: ATOM, b: ATOM) -> QTY:
  "Distance between two values of one column."
  if a == "?" or b == "?": return 1    # unknown = far
  return a!=b if type(col) is Sym else abs(norm(col,a)-norm(col,b))

def xdist(tbl: TBL, r1: ROW, r2: ROW | MIDS) -> float:
  "How far apart two rows are, over the x columns."
  return dist((gap(c, r1[c.at], r2[c.at])
                    for c in tbl.x), len(tbl.x))

def oracle(row: ROW) -> ROW:
  """Labeller. Rows here arrive with ys, so nothing to do.
  Real oracles fill in any missing y cells (and can close
  over a TBL to know which cells those are)."""
  return row

def acquire(tbl: TBL, cap: int | None = None,
            label: ORACLE = oracle) -> ROWS:
  "Label near the best rows, far from the rest."
  cap  = cap or the.Stop
  todo = random.sample(tbl.rows, len(tbl.rows))[:the.Few]
  both = clone(tbl, [label(todo.pop())
                     for _ in range(the.Start)])
  both.rows.sort(key=lambda r: ydist(both, r))
  n    = int(sqrt(the.Start))
  best,rest = clone(tbl,both.rows[:n]), clone(tbl,both.rows[n:])
  while todo and len(both.rows) < cap:
    cb, cr = mids(best), mids(rest)    # two centroids
    todo.sort(key=lambda z: xdist(tbl,z,cr) - xdist(tbl,z,cb))
    addRow(best, addRow(both, label(todo.pop())))
    best.rows.sort(key=lambda z: ydist(both, z))  # no y leak
    if len(best.rows) > sqrt(len(both.rows)): # best stays small
      addRow(rest, addCols(best, best.rows.pop(), -1))
  return both.rows

def grabs(tbl: TBL, cap: int | None = None,
          label: ORACLE = oracle) -> ROWS:
  "The straw man: same budget, taken at random."
  cap = cap or the.Stop
  return [label(r) for r in
          random.sample(tbl.rows, min(cap, len(tbl.rows)))]

# Calls core.  Nothing above this line calls down here.
# NODE = o(rows, at, v, go, kids, mu).
def cut(tbl: TBL, rows: ROWS) -> tuple[int,ATOM,GO]|None:
  "The (at, v, go) whose two sides have the tightest ys."
  ys = [ydist(tbl, r) for r in rows]
  out, least = None, 1e30
  for col in tbl.x:
    for v, go in candidates(col, rows):
      a, b = Num(), Num()   # one pass, so one `go` per row
      for r,y in zip(rows,ys): add(a if go(r) else b, y)
      if a.n >= the.Leaf and b.n >= the.Leaf:
        score = (div(a)*a.n + div(b)*b.n) / (a.n + b.n)
        if score < least: out, least = (col.at, v, go), score
  return out

def candidates(col: COL, rows: ROWS) -> Iterator[tuple[ATOM,GO]]:
  "What splits to try: one per symbol, or CUTS per num."
  at = col.at
  if type(col) is Sym:
    for v in sorted({r[at] for r in rows if r[at] != "?"}):
      yield v, lambda r,v=v,at=at: r[at] == v
  else:                # by rank, not value: THESE rows are dense
    xs = sorted(r[at] for r in rows if r[at] != "?")
    k  = max(1, len(xs)//the.Cuts)     # cuts, spaced by rank
    for v in xs[k-1:-1:k]:
      yield v, (lambda r,v=v,at=at: r[at] != "?" and r[at] <= v)

def tree(tbl: TBL, rows: ROWS) -> NODE:
  "Split while a split keeps both sides big enough."
  node = o(rows=rows, at=None, v=None, go=None, kids=[],
           mu=sum(ydist(tbl,r) for r in rows)/len(rows))
  if len(rows) > the.Leaf and (found := cut(tbl, rows)):
    at, v, go = found
    yes, no = [], []
    for r in rows: (yes if go(r) else no).append(r)
    node.at, node.v, node.go = at, v, go # cut kepts sides big
    node.kids = [tree(tbl, yes), tree(tbl, no)]
  return node

def leaf(node: NODE, row: ROW) -> NODE:
  "Walk ROW down to its leaf."
  while node.kids: node = node.kids[0 if node.go(row) else 1]
  return node

def leafs(node: NODE) -> list[NODE]:
  "Return leaves of this tree."
  return [x for k in node.kids for x in leafs(k)] or [node]

def show(tbl: TBL, node: NODE, pre: str | None = None,
         edge: str = "") -> None:
  "Tree on the left; ydist and n on the right."
  ls = sorted(leafs(node), key=lambda z: z.mu)
  def walk(z: NODE, pre: str | None, edge: str) -> None:
    m = "+" if z is ls[0] else "-" if z is ls[-1] else " "
    print(f"{m} {round(100*z.mu):>4} {len(z.rows):>5}   "
          f"{(pre or '')+edge}".rstrip())
    op = "=" if type(tbl.cols[z.at or 0]) is Sym else "<="
    no = "!=" if op == "=" else ">"
    for kid, oper in zip(z.kids, [op, no]):
      walk(kid, "" if pre is None else pre + "|  ",
           f"{tbl.names[z.at]} {oper} {say(z.v)}")
  print(" ydist     n")
  walk(node, pre, edge)

# Calls core, acquire, and one RANKER -- tree's, by default.
def ranker(lab: TBL) -> Callable[[ROW], float]:
  "Grow a tree on LAB; score any row by its leaf's mean ydist."
  node = tree(lab, lab.rows)
  return lambda r: leaf(node, r).mu

def wins(tbl: TBL) -> Callable[[ROW], float]:
  "100 at the best row, 0 at an average one."
  ys = sorted(ydist(tbl, r) for r in tbl.rows)
  lo, avg = ys[0], sum(ys)/len(ys)
  return lambda r: max(-100, min(100, 100*(1 - (ydist(tbl,r)-lo)
                                            / (avg-lo+1e-32))))

def holdout(tbl: TBL, pick: PICKER = acquire,
            rank: RANKER = ranker,
            label: ORACLE = oracle) -> ROW:
  "Train on half; of CHECK guesses on the rest, pick best."
  rows  = random.sample(tbl.rows, len(tbl.rows))
  n     = len(rows)//2
  tr    = clone(tbl, rows[:n][:the.Few])
  lab   = clone(tbl, pick(tr, the.Stop - the.Check, label))
  top   = sorted(rows[n:], key=rank(lab))
  return min(top[:the.Check], key=lambda r: ydist(lab, label(r)))

# A leaf: it calls core, and nothing here calls it.
def adds(vs: Iterable[ATOM], col: COL | None = None) -> COL:
  "Every item of VS into COL."
  col = Num() if col is None else col
  for v in vs: add(col, v)
  return col

def cohen(xs: NUMS, ys: NUMS, d: float = 0.35,
          eps: float = 0) -> bool:
  "Are the means within D pooled standard deviations?"
  a, b = adds(xs), adds(ys)
  pool = ((a.n-1)*div(a)**2 + (b.n-1)*div(b)**2)/(a.n+b.n-2)
  return abs(a.mu - b.mu) <= max(eps, d * sqrt(pool))

def cliffs(xs: NUMS, ys: NUMS, d: float = 0.197) -> bool:
  "Sorted in.  Is the rank imbalance small enough?"
  gt = lt = j = k = 0
  for x in xs:
    while j < len(ys) and ys[j] <  x: j += 1; k = j
    while k < len(ys) and ys[k] <= x: k += 1
    gt += j; lt += len(ys) - k
  return abs(gt - lt) / (len(xs) * len(ys)) <= d

def ks(xs: NUMS, ys: NUMS, a: float = 1.36) -> bool:
  "Sorted in.  95% Kolmogorov-Smirnov."
  n, m, i, j, d = len(xs), len(ys), 0, 0, 0
  while i < n and j < m:
    v = min(xs[i], ys[j])
    while i < n and xs[i] <= v: i += 1
    while j < m and ys[j] <= v: j += 1
    d = max(d, abs(i/n - j/m))
  return d <= a * sqrt((n + m) / (n * m))

def same(xs: NUMS, ys: NUMS, eps: float = 0) -> bool:
  """Indistinguishable.  Always cliffs and ks; EPS adds cohen, and
  doubles as its effect size (eps=0 skips that third test)."""
  xs, ys = sorted(xs), sorted(ys)
  return (cliffs(xs,ys) and ks(xs,ys)
          and (not eps or cohen(xs,ys,d=eps)))

import ezr
import random

def remember(header, rows):
    """
    Finds the best rows based on the goals defined in the header.
    Goals ending in '+' are maximized, '-' are minimized.
    """
    # Create a table object to use ezr's utility functions
    # We need to reconstruct the header with the correct format for ezr
    # ezr expects the header to contain the column names
    tbl = ezr.Tbl([header] + rows)
    
    # Calculate ydist for all rows
    # ydist measures how far a row is from the ideal goal values
    scored = []
    for i, row in enumerate(rows):
        d = ezr.ydist(tbl, row)
        scored.append((d, i))
    
    # Sort by distance (lower is better)
    scored.sort()
    
    # Return the top 5% of rows (or at least 1)
    limit = max(1, len(rows) // 20)
    return [i for d, i in scored[:limit]]

import ezr
import random

def guess(header, train, test):
    """
    Estimates the value of the first goal column for test rows.
    Uses a tree-based approach to predict the outcome based on training data.
    """
    # Create a table from the training data
    # ezr.Tbl expects the header as the first row of the input
    train_tbl = ezr.Tbl([header] + train)
    
    # Build a tree on the training data to learn patterns
    # The tree nodes contain the mean ydist, but we need to predict 
    # the specific value of the first goal column.
    # We can use the tree to find the 'leaf' for each test row,
    # then return the average of the first goal column in that leaf.
    
    # Identify the first goal column index
    # Goals are columns ending in '+' or '-'
    goal_indices = [i for i, name in enumerate(header) if name.endswith(("+", "-"))]
    first_goal_idx = goal_indices[0]
    
    # Build the tree
    root = ezr.tree(train_tbl, train_tbl.rows)
    
    # Helper to get the average of the first goal in a leaf
    def get_leaf_prediction(node):
        vals = [r[first_goal_idx] for r in node.rows if isinstance(r[first_goal_idx], (int, float))]
        return sum(vals) / len(vals) if vals else 0.0

    # Map leaves to their average goal value
    leaf_map = {id(leaf): get_leaf_prediction(leaf) for leaf in ezr.leafs(root)}
    
    # Predict for each test row
    predictions = []
    for row in test:
        leaf = ezr.leaf(root, row)
        predictions.append(leaf_map[id(leaf)])
        
    return predictions

def certify(xs, ys):
    """
    Returns True if the two distributions are statistically indistinguishable
    (i.e., the difference is just noise), False if they significantly differ.

    # relying on cliffs and ks to determine if they are 'the same'.
    return ezr.same(xs, ys)

import ezr
import random

def flow(header, row, seen):
    """
    Decide if the current row is a 'keeper' (needs a retention conversation).
    We use a simple heuristic: if the row is significantly better than the 
    average of what we have seen so far, we flag it.
    """
    if not seen:
        return True
    
    # Create a temporary table to calculate ydist
    # We need to handle the fact that 'seen' might contain rows with '?'
    # but ezr.ydist works on the goals.
    tbl = ezr.Tbl([header] + seen)
    
    # Calculate the ydist of the current row
    # If the row has '?' in goal columns, we treat it as neutral
    current_ydist = ezr.ydist(tbl, row)
    
    # Calculate the average ydist of seen rows
    seen_ydists = [ezr.ydist(tbl, r) for r in seen]
    avg_ydist = sum(seen_ydists) / len(seen_ydists)
    
    # If the current row is better (lower ydist) than the average,
    # we flag it for a retention conversation.
    # We use a small buffer to avoid flagging everything.
    return current_ydist < (avg_ydist * 0.9)

import ezr

def blame(header, rows):
    """
    Identifies input columns that correlate with 'bad' outcomes (high ydist).
    We compare the distribution of input values in the worst 25% of rows
    against the rest of the data.
    """
    tbl = ezr.Tbl([header] + rows)
    
    # Calculate ydist for all rows
    scored = []
    for row in rows:
        scored.append((ezr.ydist(tbl, row), row))
    
    # Sort by ydist (higher is worse)
    scored.sort(key=lambda x: x[0], reverse=True)
    
    # Split into worst 25% and the rest
    n = len(scored) // 4
    worst = [x[1] for x in scored[:n]]
    rest = [x[1] for x in scored[n:]]
    
    # For each input column, check if the distribution in 'worst'
    # is significantly different from 'rest'.
    # We use a simple heuristic: compare the 'mid' (mean or mode)
    # of the worst rows vs the rest.
    explanations = []
    for col in tbl.x:
        # Create temporary columns to summarize the two groups
        w_col = ezr.Col(col.txt, col.at)
        r_col = ezr.Col(col.txt, col.at)
        
        for r in worst: ezr.add(w_col, r[col.at])
        for r in rest:  ezr.add(r_col, r[col.at])
        
        # If the middle value is significantly different, it's a candidate
        # For simplicity, we look for columns where the middle value differs
        # or where the spread is notably different.
        # Here we return the column name if the 'mid' is different.
        if ezr.mid(w_col) != ezr.mid(r_col):
            explanations.append(col.txt)
            
    return explanations

import ezr
import random

def justify(header, rows, row):
    """
    Returns (p, reason): p is the fraction of rows that 'row' is better than;
    reason is a sentence explaining the ranking.
    """
    # Create a table to handle the data and calculate ydist
    tbl = ezr.Tbl([header] + rows)
    
    # Calculate ydist for all known rows
    # ydist is the distance from the ideal goal values (lower is better)
    all_ydists = [ezr.ydist(tbl, r) for r in rows]
    
    # To estimate the goal values for the input 'row', we use the tree
    # to find the leaf node that matches the input characteristics.
    # The leaf's mean ydist serves as our estimate for the row's performance.
    root = ezr.tree(tbl, rows)
    leaf_node = ezr.leaf(root, row)
    estimated_ydist = leaf_node.mu
    
    # Calculate p: fraction of rows that are worse than our estimated row
    # (i.e., have a higher ydist)
    worse_count = sum(1 for d in all_ydists if d > estimated_ydist)
    p = worse_count / len(rows)
    
    # Generate a reason based on the ranking
    reason = (f"This arrangement ranks in the top {int((1-p)*100)}% of all observed "
              f"employee records based on our multi-objective performance model.")
    
    return p, reason

import ezr
import random

def choose(header, rows, label):
    """
    Find the best working arrangement using at most 50 labels.
    We use the 'acquire' strategy from ezr to intelligently sample 
    the search space. We must ensure the table is fully initialized
    with numeric data before calling ydist/norm.
    """
    # Create a table object.
    tbl = ezr.Tbl([header] + rows)
    
    # The error occurred because ezr.acquire calls ydist on rows that 
    # still have '?' in goal columns. We must ensure the oracle 
    # returns the full row and the table is updated.
    def oracle_wrapper(row):
        # Find the index of this row in the original list
        idx = rows.index(row)
        true_row = label(idx)
        # Update the row in place with the true values
        for i, val in enumerate(true_row):
            row[i] = val
        # Re-add the row to the table to update column statistics (mu, sd)
        # so that norm() works correctly for subsequent ydist calculations.
        ezr.addCols(tbl, row)
        return row

    # Set the budget
    ezr.the.Stop = 50
    
    # acquire returns the rows that were labeled and explored
    labeled_rows = ezr.acquire(tbl, cap=50, label=oracle_wrapper)
    
    # Find the best row among the labeled ones using ydist
    best_row = min(labeled_rows, key=lambda r: ezr.ydist(tbl, r))
    
    # Return the index of the best row
    return rows.index(best_row)

import ezr

def spot(header, rows):
    """
    Identifies suspicious rows by finding those that are 'outliers' 
    relative to their local neighborhood in the data space.
    
    We use the tree structure to group similar rows. A row is suspicious 
    if its ydist (performance) is significantly worse than the mean 
    ydist of its leaf node, suggesting it might be an anomaly or 
    data entry error.
    """
    # Create a table to calculate ydist and build the tree
    tbl = ezr.Tbl([header] + rows)
    
    # Build a tree to cluster similar rows
    root = ezr.tree(tbl, rows)
    
    # Calculate the deviation of each row from its leaf's mean performance
    # Suspiciousness = (row_ydist - leaf_mean_ydist)
    # Higher deviation means the row is an outlier in its own cluster
    suspects = []
    for i, row in enumerate(rows):
        leaf = ezr.leaf(root, row)
        row_ydist = ezr.ydist(tbl, row)
        # We look for rows that perform much worse than their peers
        deviation = row_ydist - leaf.mu
        suspects.append((deviation, i))
    
    # Sort by deviation descending (most suspicious first)
    suspects.sort(key=lambda x: x[0], reverse=True)
    
    # Return the indices of the top 10 most suspicious rows
    return [i for dev, i in suspects[:10]]

import ezr
import random

def race(header, rows, label):
    """
    Compares three strategies for finding the best employee arrangement.
    Budget: 50 labels total.
    """
    # Shared state for the race
    labeled_indices = set()
    
    def oracle_wrapper(i):
        if i not in labeled_indices:
            if len(labeled_indices) >= 50:
                raise Exception("Label budget exceeded!")
            labeled_indices.add(i)
            # Update the row with true values
            true_row = label(i)
            for col_idx, val in enumerate(true_row):
                rows[i][col_idx] = val
        return rows[i]

    # Initialize table with current (partially labeled) rows
    tbl = ezr.Tbl([header] + rows)

    def get_ydist(row):
        # Ensure row is fully updated in the table before calculating distance
        return ezr.ydist(tbl, row)

    # Strategy 1: Random
    def strategy_random(rows, oracle):
        indices = random.sample(range(len(rows)), 16)
        for i in indices: oracle(i)
        return min(indices, key=lambda i: get_ydist(rows[i]))

    # Strategy 2: Active Learning
    def strategy_active(rows, oracle):
        # Use a fresh table for the active learner to avoid state pollution
        local_tbl = ezr.Tbl([header] + rows)
        def local_oracle(row):
            idx = rows.index(row)
            return oracle(idx)
        
        ezr.the.Stop = 17
        labeled = ezr.acquire(local_tbl, cap=17, label=local_oracle)
        best_row = min(labeled, key=lambda r: get_ydist(r))
        return rows.index(best_row)

    # Strategy 3: Greedy (Pick best based on initial sample)
    def strategy_greedy(rows, oracle):
        indices = random.sample(range(len(rows)), 17)
        for i in indices: oracle(i)
        return min(indices, key=lambda i: get_ydist(rows[i]))

    # Run race
    results = {
        "random": strategy_random(rows, oracle_wrapper),
        "active": strategy_active(rows, oracle_wrapper),
        "greedy": strategy_greedy(rows, oracle_wrapper)
    }
    
    # Find winner (lowest ydist)
    winner = min(results, key=lambda name: get_ydist(rows[results[name]]))
    
    return {
        "winner": winner,
        "row": results[winner],
        "tried": ["random", "active", "greedy"]
    }

def guess(header, train, test):

    """

    # Identify the first goal column index
    goal_indices = [i for i, name in enumerate(header) if name.endswith(("+", "-"))]
    first_goal_idx = goal_indices[0]
    
    # Build the tree
    root = ezr.tree(train_tbl, train_tbl.rows)
    
    # Helper to get the average of the first goal in a leaf
    def get_leaf_prediction(node):
        vals = [r[first_goal_idx] for r in node.rows if isinstance(r[first_goal_idx], (int, float))]
        return sum(vals) / len(vals) if vals else 0.0

    # Map leaves to their average goal value
    leaf_map = {id(leaf): get_leaf_prediction(leaf) for leaf in ezr.leafs(root)}
    
    # Predict for each test row
    predictions = []
    for row in test:
        leaf = ezr.leaf(root, row)
        predictions.append(leaf_map.get(id(leaf), 0.0))
        
    return predictions

import ezr

def blame(header, rows):
    """
    Identifies input columns that have the least impact on performance.
    We compare the distribution of input values in the worst 25% of rows
    against the rest of the data. If the distributions are statistically
    indistinguishable, the column likely has nothing to do with the outcome.
    """
    tbl = ezr.Tbl([header] + rows)
    
    # Calculate ydist for all rows
    scored = []
    for row in rows:
        scored.append((ezr.ydist(tbl, row), row))
    
    # Sort by ydist (higher is worse)
    scored.sort(key=lambda x: x[0], reverse=True)
    
    # Split into worst 25% and the rest
    n = len(scored) // 4
    worst = [x[1] for x in scored[:n]]
    rest = [x[1] for x in scored[n:]]
    
    irrelevant = []
    for col in tbl.x:
        # Extract values for this column in both groups
        w_vals = [r[col.at] for r in worst if r[col.at] != "?"]
        r_vals = [r[col.at] for r in rest if r[col.at] != "?"]
        
        # If we have enough data, check if distributions are the same
        if len(w_vals) > 10 and len(r_vals) > 10:
            # For numeric, use same() (cliffs/ks/cohen)
            # For symbolic, we check if the most common symbol is the same
            if type(col) is ezr.Num:
                if ezr.same(sorted(w_vals), sorted(r_vals)):
                    irrelevant.append(col.txt)
            else:
                # Symbolic: compare frequency distribution
                w_counts = {}
                for v in w_vals: w_counts[v] = w_counts.get(v, 0) + 1
                r_counts = {}
                for v in r_vals: r_counts[v] = r_counts.get(v, 0) + 1
                
                w_mode = max(w_counts, key=w_counts.get)
                r_mode = max(r_counts, key=r_counts.get)
                
                if w_mode == r_mode:
                    irrelevant.append(col.txt)
                    
    return irrelevant

import ezr
import random

def justify(header, rows, row):
    """
    Estimates the performance of a row and compares it to the population.
    Returns (p, reason) where p is the fraction of rows worse than the input,
    and reason explains why simply increasing income is not a universal fix.
    """
    # Create a table to handle column definitions and distance calculations
    tbl = ezr.Tbl([header] + rows)
    
    # Build a tree to estimate performance (ydist) for the input row
    # The tree clusters rows by input characteristics; the leaf mean is our estimate.
    root = ezr.tree(tbl, rows)
    leaf_node = ezr.leaf(root, row)
    estimated_ydist = leaf_node.mu
    
    # Calculate p: fraction of rows that are worse than our estimated row
    # (i.e., have a higher ydist, meaning they are further from the goals)
    all_ydists = [ezr.ydist(tbl, r) for r in rows]
    worse_count = sum(1 for d in all_ydists if d > estimated_ydist)
    p = worse_count / len(rows)
    
    # Reason: Explain that performance is multi-dimensional (WorkLifeBalance, 
    # YearsInCurrentRole, etc.) and income is only one factor.
    reason = (
        "Increasing income does not guarantee retention because our model shows "
        "that performance is driven by a complex mix of factors like work-life "
        "balance and career growth, not just salary."
    )
    
    return p, reason
