import ezr
import random

def choose(header, rows, label):
    """
    Find the best working arrangement using at most 50 labels.
    We use the 'acquire' strategy from ezr to intelligently sample 
    the search space. We must ensure the table is fully initialized
    with numeric data before calling ydist/norm.
    """
    # Create a table object.
    tbl = ezr.Tbl([header] + rows)
    
    # The error occurred because ezr.acquire calls ydist on rows that 
    # still have '?' in goal columns. We must ensure the oracle 
    # returns the full row and the table is updated.
    def oracle_wrapper(row):
        # Find the index of this row in the original list
        idx = rows.index(row)
        true_row = label(idx)
        # Update the row in place with the true values
        for i, val in enumerate(true_row):
            row[i] = val
        # Re-add the row to the table to update column statistics (mu, sd)
        # so that norm() works correctly for subsequent ydist calculations.
        ezr.addCols(tbl, row)
        return row

    # Set the budget
    ezr.the.Stop = 50
    
    # acquire returns the rows that were labeled and explored
    labeled_rows = ezr.acquire(tbl, cap=50, label=oracle_wrapper)
    
    # Find the best row among the labeled ones using ydist
    best_row = min(labeled_rows, key=lambda r: ezr.ydist(tbl, r))
    
    # Return the index of the best row
    return rows.index(best_row)

if __name__ == "__main__":
    filename = "WA_Fn-UseC_-HR-Employee-Attrition.n1000.train.csv"
    data = list(ezr.csv(filename))
    header = data[0]
    rows = data[1:]
    
    # Keep a copy of the original rows to simulate the oracle
    full_data = [list(r) for r in rows]
    
    # Mask the goals in the rows passed to choose()
    goal_indices = [i for i, name in enumerate(header) if name.endswith(("+", "-"))]
    masked_rows = []
    for r in rows:
        new_row = list(r)
        for idx in goal_indices:
            new_row[idx] = "?"
        masked_rows.append(new_row)
            
    def label(i):
        return full_data[i]
    
    best_idx = choose(header, masked_rows, label)
    
    print(f"Active learning complete. Best row found at index: {best_idx}")
    print("The identified arrangement optimizes for the goal columns provided.")
    
    exit(0)
