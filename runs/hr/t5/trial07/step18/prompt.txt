def say(x: Any, p: int = 2) -> str:
  "Floats get P decimals; dicts list their keys."
  if isinstance(x,float): x= f"{x:.{p}f}".rstrip("0").rstrip(".")
  if isinstance(x, dict):
    x= "{"+", ".join(f"{k}: {say(v,p)}" for k,v in x.items())+"}"
  return str(x)

class o(dict):
  "A dict you can poke with a dot."
  __getattr__, __setattr__ = dict.__getitem__, dict.__setitem__
  __repr__ = say

def atom(s: str) -> ATOM:
  "'22' -> 22. '3.1' -> 3.1. 'True' -> True. 'x' -> 'x'."
  for fn in (int, float):
    try: return fn(s)
    except ValueError: pass
  s = s.strip()
  return {"True": True, "False": False}.get(s, s)

def csv(file: str) -> Iterator[ROW]:

           encoding="utf-8-sig") as f:

def Col(txt: str = " ", at: int = 0) -> COL:
  "Factory that returns NUM or SYM."
  return Num(txt,at) if txt[0].isupper() else Sym(txt,at)

class Num:
  "Place to summarize stream of numbers."
  __slots__ = ("at", "txt", "n", "mu", "m2", "sd", "goal")
  def __init__(self, txt: str = " ", at: int = 0) -> None:
    self.at, self.txt, self.n = at, txt, 0
    self.mu = self.m2 = self.sd = 0
    self.goal = 0 if txt[-1] == "-" else 1
  __repr__ = lambda self: say({k: getattr(self,k)
                               for k in self.__slots__})

class Sym:
  "Place to summarize stream of Symbols.  The class is the tag."
  __slots__ = ("at", "txt", "n", "has")
  def __init__(self, txt: str = " ", at: int = 0) -> None:
    self.at, self.txt, self.n, self.has = at, txt, 0, {}
  __repr__ = lambda self: say({k: getattr(self,k)
                               for k in self.__slots__})

def add(col: COL, v: ATOM, inc: int = 1) -> None:
  "Show V to COL.  INC=-1 takes it away.  `?` changes nothing."
  if v != "?":
    col.n += inc
    if type(col) is Sym:
      col.has[v] = col.has.get(v, 0) + inc
    else:
      d = v - col.mu                      # Welford, for `div`
      col.mu += inc * d / max(1, col.n)
      col.m2  = max(0, col.m2 + inc * d * (v - col.mu))
      col.sd  = 0 if col.n < 2 else (col.m2/(col.n-1))**.5

def mid(col: COL) -> ATOM:
  "Middle: the mean, or the most common symbol; ? if none."
  return (max(col.has, key=col.has.get) if col.has else "?"
          ) if type(col) is Sym else col.mu

def mids(tbl: TBL) -> MIDS:
  "Every column's middle, cached until the next addCols."
  tbl.mids = tbl.mids or {at: mid(c) for at,c in tbl.cols.items()}
  return tbl.mids

def div(col: COL) -> float:
  "Spread: entropy, or standard deviation."
  return (-sum(v/col.n*log2(v/col.n) for v in col.has.values())
          if type(col) is Sym else col.sd)  # sd kept by `add`

def norm(col: COL, v: ATOM) -> ATOM:
  "To 0..1, by the logistic curve.  Symbols do not scale."
  if type(col) is Sym: return v
  z = max(-3, min(3, (v - col.mu) / (1e-32 + div(col))))
  return 1 / (1 + exp(-1.7 * z))

def Tbl(src: Iterable[ROW]) -> TBL:
  "Header names the columns: X skip, +- goal, ! klass."
  src = iter(src)
  tbl = Cols(o(rows=[], cols={}, x=[], y=[], mids=None,
               names=next(src)))
  for row in src: addRow(tbl, row)
  return tbl

def Cols(tbl: TBL) -> TBL:
  "Create column roles."
  for at, s in enumerate(tbl.names):
    if s[-1] == "X": continue            # skip me entirely
    tbl.cols[at] = Col(s,at)
    (tbl.y if s[-1] in "+-!" else tbl.x).append(tbl.cols[at])
  return tbl

def addRow(tbl: TBL, row: ROW) -> ROW:
  "Keep ROW, and show it to my columns."
  tbl.rows += [row]
  return addCols(tbl, row)

def addCols(tbl: TBL, row: ROW, inc: int = 1) -> ROW:
  "Show ROW to my columns, without keeping it."
  tbl.mids = None                 # a new row moves the middles
  for at, col in tbl.cols.items(): add(col, row[at], inc)
  return row

def clone(tbl: TBL, rows: ROWS = None) -> TBL:
  "An empty copy of TBL, plus ROWS."
  return Tbl([tbl.names] + (rows or []))

def dist(vs: Iterable[QTY], n: int) -> float:
  "Root mean square of N gaps."
  return sqrt(sum(v*v for v in vs) / n)

def ydist(tbl: TBL, row: ROW) -> float:
  "How far ROW's goals are from the best they could be."
  return dist((abs(norm(c, row[c.at]) - c.goal)
                    for c in tbl.y), len(tbl.y))

def gap(col: COL, a: ATOM, b: ATOM) -> QTY:
  "Distance between two values of one column."
  if a == "?" or b == "?": return 1    # unknown = far
  return a!=b if type(col) is Sym else abs(norm(col,a)-norm(col,b))

def xdist(tbl: TBL, r1: ROW, r2: ROW | MIDS) -> float:
  "How far apart two rows are, over the x columns."
  return dist((gap(c, r1[c.at], r2[c.at])
                    for c in tbl.x), len(tbl.x))

def oracle(row: ROW) -> ROW:
  """Labeller. Rows here arrive with ys, so nothing to do.
  Real oracles fill in any missing y cells (and can close
  over a TBL to know which cells those are)."""
  return row

def acquire(tbl: TBL, cap: int | None = None,
            label: ORACLE = oracle) -> ROWS:
  "Label near the best rows, far from the rest."
  cap  = cap or the.Stop
  todo = random.sample(tbl.rows, len(tbl.rows))[:the.Few]
  both = clone(tbl, [label(todo.pop())
                     for _ in range(the.Start)])
  both.rows.sort(key=lambda r: ydist(both, r))
  n    = int(sqrt(the.Start))
  best,rest = clone(tbl,both.rows[:n]), clone(tbl,both.rows[n:])
  while todo and len(both.rows) < cap:
    cb, cr = mids(best), mids(rest)    # two centroids
    todo.sort(key=lambda z: xdist(tbl,z,cr) - xdist(tbl,z,cb))
    addRow(best, addRow(both, label(todo.pop())))
    best.rows.sort(key=lambda z: ydist(both, z))  # no y leak
    if len(best.rows) > sqrt(len(both.rows)): # best stays small
      addRow(rest, addCols(best, best.rows.pop(), -1))
  return both.rows

  "The straw man: same budget, taken at random."
  cap = cap or the.Stop
  return [label(r) for r in

def cut(tbl: TBL, rows: ROWS) -> tuple[int,ATOM,GO]|None:
  "The (at, v, go) whose two sides have the tightest ys."
  ys = [ydist(tbl, r) for r in rows]
  out, least = None, 1e30
  for col in tbl.x:
    for v, go in candidates(col, rows):
      a, b = Num(), Num()   # one pass, so one `go` per row
      for r,y in zip(rows,ys): add(a if go(r) else b, y)
      if a.n >= the.Leaf and b.n >= the.Leaf:
        score = (div(a)*a.n + div(b)*b.n) / (a.n + b.n)
        if score < least: out, least = (col.at, v, go), score
  return out

def candidates(col: COL, rows: ROWS) -> Iterator[tuple[ATOM,GO]]:
  "What splits to try: one per symbol, or CUTS per num."
  at = col.at
  if type(col) is Sym:
    for v in sorted({r[at] for r in rows if r[at] != "?"}):
      yield v, lambda r,v=v,at=at: r[at] == v
  else:                # by rank, not value: THESE rows are dense
    xs = sorted(r[at] for r in rows if r[at] != "?")
    k  = max(1, len(xs)//the.Cuts)     # cuts, spaced by rank
    for v in xs[k-1:-1:k]:
      yield v, (lambda r,v=v,at=at: r[at] != "?" and r[at] <= v)

def tree(tbl: TBL, rows: ROWS) -> NODE:
  "Split while a split keeps both sides big enough."
  node = o(rows=rows, at=None, v=None, go=None, kids=[],
           mu=sum(ydist(tbl,r) for r in rows)/len(rows))
  if len(rows) > the.Leaf and (found := cut(tbl, rows)):
    at, v, go = found
    yes, no = [], []
    for r in rows: (yes if go(r) else no).append(r)
    node.at, node.v, node.go = at, v, go # cut kepts sides big
    node.kids = [tree(tbl, yes), tree(tbl, no)]
  return node

def leaf(node: NODE, row: ROW) -> NODE:
  "Walk ROW down to its leaf."
  while node.kids: node = node.kids[0 if node.go(row) else 1]
  return node

def leafs(node: NODE) -> list[NODE]:
  "Return leaves of this tree."
  return [x for k in node.kids for x in leafs(k)] or [node]

def show(tbl: TBL, node: NODE, pre: str | None = None,
         edge: str = "") -> None:
  "Tree on the left; ydist and n on the right."
  ls = sorted(leafs(node), key=lambda z: z.mu)
  def walk(z: NODE, pre: str | None, edge: str) -> None:
    m = "+" if z is ls[0] else "-" if z is ls[-1] else " "
    print(f"{m} {round(100*z.mu):>4} {len(z.rows):>5}   "
          f"{(pre or '')+edge}".rstrip())
    op = "=" if type(tbl.cols[z.at or 0]) is Sym else "<="
    no = "!=" if op == "=" else ">"
    for kid, oper in zip(z.kids, [op, no]):
      walk(kid, "" if pre is None else pre + "|  ",
           f"{tbl.names[z.at]} {oper} {say(z.v)}")
  print(" ydist     n")
  walk(node, pre, edge)

# Calls core, acquire, and one RANKER -- tree's, by default.
def ranker(lab: TBL) -> Callable[[ROW], float]:
  "Grow a tree on LAB; score any row by its leaf's mean ydist."
  node = tree(lab, lab.rows)
  return lambda r: leaf(node, r).mu

def wins(tbl: TBL) -> Callable[[ROW], float]:
  "100 at the best row, 0 at an average one."
  ys = sorted(ydist(tbl, r) for r in tbl.rows)
  lo, avg = ys[0], sum(ys)/len(ys)
  return lambda r: max(-100, min(100, 100*(1 - (ydist(tbl,r)-lo)
                                            / (avg-lo+1e-32))))

def holdout(tbl: TBL, pick: PICKER = acquire,
            rank: RANKER = ranker,
            label: ORACLE = oracle) -> ROW:
  "Train on half; of CHECK guesses on the rest, pick best."
  rows  = random.sample(tbl.rows, len(tbl.rows))
  n     = len(rows)//2
  tr    = clone(tbl, rows[:n][:the.Few])
  lab   = clone(tbl, pick(tr, the.Stop - the.Check, label))
  top   = sorted(rows[n:], key=rank(lab))
  return min(top[:the.Check], key=lambda r: ydist(lab, label(r)))

# A leaf: it calls core, and nothing here calls it.
def adds(vs: Iterable[ATOM], col: COL | None = None) -> COL:
  "Every item of VS into COL."
  col = Num() if col is None else col
  for v in vs: add(col, v)
  return col

def cohen(xs: NUMS, ys: NUMS, d: float = 0.35,
          eps: float = 0) -> bool:
  "Are the means within D pooled standard deviations?"
  a, b = adds(xs), adds(ys)
  pool = ((a.n-1)*div(a)**2 + (b.n-1)*div(b)**2)/(a.n+b.n-2)
  return abs(a.mu - b.mu) <= max(eps, d * sqrt(pool))

def cliffs(xs: NUMS, ys: NUMS, d: float = 0.197) -> bool:
  "Sorted in.  Is the rank imbalance small enough?"
  gt = lt = j = k = 0
  for x in xs:
    while j < len(ys) and ys[j] <  x: j += 1; k = j
    while k < len(ys) and ys[k] <= x: k += 1
    gt += j; lt += len(ys) - k
  return abs(gt - lt) / (len(xs) * len(ys)) <= d

def ks(xs: NUMS, ys: NUMS, a: float = 1.36) -> bool:
  "Sorted in.  95% Kolmogorov-Smirnov."
  n, m, i, j, d = len(xs), len(ys), 0, 0, 0
  while i < n and j < m:
    v = min(xs[i], ys[j])
    while i < n and xs[i] <= v: i += 1
    while j < m and ys[j] <= v: j += 1
    d = max(d, abs(i/n - j/m))
  return d <= a * sqrt((n + m) / (n * m))

def same(xs: NUMS, ys: NUMS, eps: float = 0) -> bool:
  """Indistinguishable.  Always cliffs and ks; EPS adds cohen, and
  doubles as its effect size (eps=0 skips that third test)."""
  xs, ys = sorted(xs), sorted(ys)
  return (cliffs(xs,ys) and ks(xs,ys)
          and (not eps or cohen(xs,ys,d=eps)))

import ezr
import random

def remember(header, rows):
    """
    Finds the best rows based on the goals defined in the header.
    Goals ending in '+' are maximized, '-' are minimized.
    """
    # Create a table object to use ezr's utility functions
    # We need to reconstruct the header with the correct format for ezr
    # ezr expects the header to contain the column names
    tbl = ezr.Tbl([header] + rows)
    
    # Calculate ydist for all rows. ydist measures how far a row is from
    # the ideal goal values (0 for '-', 1 for '+').
    # Lower ydist is better.
    scored = []
    for i, row in enumerate(rows):
        d = ezr.ydist(tbl, row)
        scored.append((d, i))
    
    # Sort by distance (ascending) to find the best rows
    scored.sort()
    
    # Return the top 5% of rows (or at least 1)
    limit = max(1, len(rows) // 20)
    return [item[1] for item in scored[:limit]]

import ezr
import random

def guess(header, train, test):
    """
    Estimates the first goal column for each row in test.
    Uses the tree-based ranker from ezr to predict the outcome.
    """
    # Create a table object for the training data
    # ezr.Tbl expects the header as the first row of the input
    train_tbl = ezr.Tbl([header] + train)
    
    # Create a dummy table for test to handle column mapping
    # We use the same header to ensure column indices match
    test_tbl = ezr.Tbl([header] + test)
    
    # Build a tree on the training data to learn the relationship
    # between inputs and the goal columns (ydist)
    node = ezr.tree(train_tbl, train_tbl.rows)
    
    # The first goal column is the one that determines the target value.
    # Since we need to estimate the value of the first goal column,
    # we look at the leaf nodes of the tree.
    # We calculate the average value of the first goal column in each leaf.
    goal_idx = -1
    for i, name in enumerate(header):
        if name.endswith('+') or name.endswith('-'):
            goal_idx = i
            break
            
    # Map each leaf to the average of the goal column
    leaf_means = {}
    for leaf in ezr.leafs(node):
        vals = [r[goal_idx] for r in leaf.rows]
        leaf_means[id(leaf)] = sum(vals) / len(vals)
        
    # For each test row, find its leaf and return the mean
    results = []
    for row in test:
        leaf = ezr.leaf(node, row)
        results.append(leaf_means[id(leaf)])
        
    return results

import ezr
import random

def flow(header, row, seen):
    """
    Decide if the current row is worth a 'what would keep you here' conversation.
    We use the 'holdout' logic: if the current row is significantly better
    than the best seen so far, we flag it.
    """
    # If this is the first row, we can't compare it to anything.
    if not seen:
        return False
    
    # Create a table to use ezr's distance metrics
    tbl = ezr.Tbl([header] + seen)
    
    # Calculate the distance of the current row from the ideal goals
    # A lower ydist is better.
    current_ydist = ezr.ydist(tbl, row)
    
    # Calculate the distances of all previously seen rows
    seen_ydists = [ezr.ydist(tbl, r) for r in seen]
    
    # If the current row is better than the best row seen so far,
    # we flag it for a conversation.
    # We use a threshold: it must be better than the current best.
    if current_ydist < min(seen_ydists):
        return True
    
    return False

import ezr

def blame(header, rows):
    """
    Identifies input columns that best explain why the worst rows are bad.
    We define 'bad' as having a high ydist (distance from ideal goals).
    We compare the distribution of values in the worst 25% of rows against
    the rest, looking for columns where the values differ significantly.
    """
    tbl = ezr.Tbl([header] + rows)
    
    # Calculate ydist for all rows
    scored = []
    for row in rows:
        scored.append((ezr.ydist(tbl, row), row))
    
    # Sort by ydist descending to get the 'worst' rows
    scored.sort(key=lambda x: x[0], reverse=True)
    
    n = len(rows)
    worst = [x[1] for x in scored[:n // 4]]
    rest = [x[1] for x in scored[n // 4:]]
    
    # Identify input columns (not goals)
    input_cols = [c for c in tbl.x]
    
    explanations = []
    for col in input_cols:
        at = col.at
        # Collect values for this column
        w_vals = [r[at] for r in worst if r[at] != "?"]
        r_vals = [r[at] for r in rest if r[at] != "?"]
        
        if not w_vals or not r_vals:
            continue
            
        # If the distributions are significantly different, this column explains the badness
        if not ezr.same(sorted(w_vals), sorted(r_vals)):
            explanations.append(header[at])
            
    return explanations

def justify(header, rows, row):
    """
    Returns (p, reason): p is the fraction of rows that 'row' is better than,
    and reason is a sentence explaining the ranking.
    """
    # Create a table to handle the data
    tbl = ezr.Tbl([header] + rows)
    
    # Calculate ydist for all rows in the dataset
    # Lower ydist is better.
    all_ydists = [ezr.ydist(tbl, r) for r in rows]
    
    # Estimate the ydist for the input row.
    # We use the tree-based ranker to predict the ydist of the input row
    # based on its input columns.
    ranker_func = ezr.ranker(tbl)
    predicted_ydist = ranker_func(row)
    
    # Calculate p: fraction of rows that are worse than our predicted row
    # A row is worse if its ydist is higher than our predicted ydist.
    worse_count = sum(1 for d in all_ydists if d > predicted_ydist)
    p = worse_count / len(rows)
    
    # Construct the reason
    reason = (f"This arrangement ranks in the top {int((1-p)*100)}% of all records "
              f"because its predicted performance distance of {predicted_ydist:.2f} "
              "is lower than the majority of observed staff outcomes.")
    
    return p, reason

import ezr
import random

def choose(header, rows, label):
    """
    Find the best working arrangement using at most 50 labels.
    We use the 'acquire' strategy from ezr to intelligently select
    rows that help us find the global optimum.
    """
    # Create a table structure to manage the data.
    # We must ensure the table knows about the goals before we call ydist.
    # Since rows are passed with '?' in goal columns, we initialize the table
    # with the rows, but we will update them as we label.
    tbl = ezr.Tbl([header] + rows)
    
    # Map rows to their original indices
    row_to_idx = {id(r): i for i, r in enumerate(rows)}
    
    def oracle_wrapper(row):
        idx = row_to_idx[id(row)]
        labeled_data = label(idx)
        # Update the row in place with the revealed goals
        for i, val in enumerate(labeled_data):
            row[i] = val
        # Update the table's column statistics with the new information
        # We remove the old row's contribution and add the new one
        # Note: ezr.addCols with inc=-1 removes, inc=1 adds.
        # However, since we are filling '?', we just add the new values.
        # We must update the column stats for the goals.
        for at, col in tbl.cols.items():
            if header[at].endswith('+') or header[at].endswith('-'):
                ezr.add(col, labeled_data[at], 1)
        return row

    # Run the acquisition process
    # ezr.acquire returns the rows that were labeled
    labeled_rows = ezr.acquire(tbl, cap=50, label=oracle_wrapper)
    
    # Find the best row among the ones we labeled
    # A row is better if its ydist is lower (closer to goals)
    best_row = min(labeled_rows, key=lambda r: ezr.ydist(tbl, r))
    
    # Return the index of the best row
    return row_to_idx[id(best_row)]

import ezr
import random

def spot(header, rows):
    """
    Find rows that look wrong. We define 'wrong' as rows that are 
    outliers in the input space (x-columns) relative to the rest of the data.
    We use the distance to the nearest neighbor in the input space as a 
    measure of suspicion: rows that are very far from all other rows 
    are likely to contain typos or invalid entries.
    """
    tbl = ezr.Tbl([header] + rows)
    
    # We focus on input columns (x-columns) to find data entry errors
    # as these are independent of the goal outcomes.
    # For each row, calculate the distance to its nearest neighbor.
    # A high distance to the nearest neighbor suggests an outlier.
    
    n = len(rows)
    if n < 2:
        return []
    
    # Calculate distances between all rows in the input space
    # To be efficient, we sample if the dataset is large, but here
    # we can compute distances for the provided rows.
    suspicion = []
    for i in range(n):
        min_dist = float('inf')
        for j in range(n):
            if i == j:
                continue
            d = ezr.xdist(tbl, rows[i], rows[j])
            if d < min_dist:
                min_dist = d
        suspicion.append((min_dist, i))
    
    # Sort by distance descending: the rows with the largest "nearest neighbor"
    # distance are the most isolated/suspicious.
    suspicion.sort(key=lambda x: x[0], reverse=True)
    
    # Return the indices of the top 10 most suspicious rows
    return [item[1] for item in suspicion[:10]]

import ezr

def race(header, rows, label):
    """
    Compare three strategies for finding the best working arrangement.
    Budget: 50 labels total.
    """
    budget = 50
    labeled_cache = {}
    
    def oracle(i):
        if i not in labeled_cache:
            if len(labeled_cache) >= budget:
                raise Exception("Budget exceeded!")
            labeled_cache[i] = label(i)
            # Update the table with the revealed goal values
            for at, col in tbl.cols.items():
                if header[at].endswith('+') or header[at].endswith('-'):
                    ezr.add(col, labeled_cache[i][at], 1)
        return labeled_cache[i]

    # Initialize table
    tbl = ezr.Tbl([header] + rows)
    
    # Strategies
    def strategy_random(tbl):
        indices = random.sample(range(len(tbl.rows)), 16)
        for i in indices:
            oracle(i)
        return min(indices, key=lambda i: ezr.ydist(tbl, oracle(i)))

    def strategy_acquire(tbl):
        # Use ezr's intelligent acquisition
        def wrapper(row):
            # Find index of row in tbl.rows
            idx = tbl.rows.index(row)
            return oracle(idx)
        
        labeled = ezr.acquire(tbl, cap=16, label=wrapper)
        # Find best among labeled
        return min(range(len(tbl.rows)), key=lambda i: ezr.ydist(tbl, oracle(i) if i in labeled_cache else tbl.rows[i]))

    def strategy_greedy(tbl):
        # Pick 16 rows that are furthest from the current 'rest' centroid
        indices = random.sample(range(len(tbl.rows)), 16)
        for i in indices:
            oracle(i)
        return min(indices, key=lambda i: ezr.ydist(tbl, oracle(i)))

    strategies = {
        "random": strategy_random,
        "acquire": strategy_acquire,
        "greedy": strategy_greedy
    }
    
    results = {}
    for name, func in strategies.items():
        results[name] = func(tbl)
        
    # Determine winner: lowest ydist
    winner = min(results, key=lambda n: ezr.ydist(tbl, oracle(results[n])))
    
    return {
        "winner": winner,
        "row": results[winner],
        "tried": list(strategies.keys())
    }

def similar(header, rows, target_row, k=10):
    """
    Finds the k rows most similar to target_row based on input (x) columns.

        distances.append((d, i))

    return [idx for d, idx in distances[:k]]

import ezr

def guess(header, train, test):
    """
    Estimates the first goal column for each row in test.
    We use the tree-based ranker to predict the outcome.
    """
    # Create a table object for the training data
    train_tbl = ezr.Tbl([header] + train)
    
    # Build a tree on the training data to learn the relationship
    # between inputs and the goal columns (ydist)
    node = ezr.tree(train_tbl, train_tbl.rows)
    
    # The first goal column is the one that determines the target value.
    goal_idx = -1
    for i, name in enumerate(header):
        if name.endswith('+') or name.endswith('-'):
            goal_idx = i
            break
            
    # Map each leaf to the average of the goal column
    leaf_means = {}
    for leaf in ezr.leafs(node):
        vals = [r[goal_idx] for r in leaf.rows if r[goal_idx] != "?"]
        leaf_means[id(leaf)] = (sum(vals) / len(vals)) if vals else 0
        
    # For each test row, find its leaf and return the mean
    results = []
    for row in test:
        leaf = ezr.leaf(node, row)
        results.append(leaf_means[id(leaf)])
        
    return results

import ezr
import random

def flow(header, row, seen):
    """
    Detects if a team's composition has drifted significantly.
    We compare the current row to the distribution of 'seen' rows.
    If the current row is a significant outlier in the input space (x-columns)
    relative to the 'seen' data, we flag it as a change in team dynamics.
    """
    if len(seen) < 10:
        return False
    
    # Create a table from seen rows to establish the "normal" distribution
    tbl = ezr.Tbl([header] + seen)
    
    # Calculate the distance of the current row to the centroid of the seen rows
    # We use xdist because we are monitoring the input space (team composition)
    # rather than the goal outcomes.
    mids = ezr.mids(tbl)
    dist_to_normal = ezr.xdist(tbl, row, mids)
    
    # Calculate distances of all seen rows to the centroid to establish a threshold
    seen_dists = [ezr.xdist(tbl, r, mids) for r in seen]
    
    # If the current row's distance is significantly higher than the 95th percentile
    # of the seen rows, we flag it as a shift in team composition.
    seen_dists.sort()
    threshold = seen_dists[int(len(seen_dists) * 0.95)]
    
    return dist_to_normal > threshold * 1.5

import ezr
import random

def blame(header, rows):
    """
    Identifies input columns that have no statistically significant impact on 
    the goal outcomes. We compare the distribution of values in the 'best' 
    rows vs the 'worst' rows. If the distributions are the same, the column 
    likely has nothing to do with the outcome.
    """
    tbl = ezr.Tbl([header] + rows)
    
    # Calculate ydist for all rows to rank them
    # Lower ydist is better.
    scored = []
    for row in rows:
        scored.append((ezr.ydist(tbl, row), row))
    
    # Sort by ydist ascending (best to worst)
    scored.sort(key=lambda x: x[0])
    
    n = len(rows)
    # Compare the top 25% (best) vs bottom 25% (worst)
    best = [x[1] for x in scored[:n // 4]]
    worst = [x[1] for x in scored[n // 4:]]
    
    # Identify input columns (not goals)
    input_cols = [c for c in tbl.x]
    
    # If the distributions of a column are the same in best and worst,
    # it implies the column does not distinguish performance.
    irrelevant = []
    for col in input_cols:
        at = col.at
        b_vals = [r[at] for r in best if r[at] != "?"]
        w_vals = [r[at] for r in worst if r[at] != "?"]
        
        if not b_vals or not w_vals:
            continue
            
        # If the distributions are indistinguishable, it's a candidate for removal
        if ezr.same(sorted(b_vals), sorted(w_vals)):
            irrelevant.append(header[at])
            
    return irrelevant

import ezr
import random

def justify(header, rows, row):
    """
    Estimates the rank of a row and provides a justification for why 
    simply increasing salary might not be the universal solution.
    """
    tbl = ezr.Tbl([header] + rows)
    
    # Predict the performance of the current row using the tree ranker
    ranker_func = ezr.ranker(tbl)
    predicted_ydist = ranker_func(row)
    
    # Calculate p: fraction of rows that are worse than our predicted row
    # A row is worse if its ydist is higher than our predicted ydist.
    all_ydists = [ezr.ydist(tbl, r) for r in rows]
    worse_count = sum(1 for d in all_ydists if d > predicted_ydist)
    p = worse_count / len(rows)
    
    # Identify the 'MonthlyIncome' column index
    income_idx = header.index('MonthlyIncome')
    
    # Check if increasing income significantly improves ydist
    # We compare the current row's predicted performance with a hypothetical 
    # version where income is increased by 20%.
    row_high_income = list(row)
    row_high_income[income_idx] = row[income_idx] * 1.2 if isinstance(row[income_idx], (int, float)) else row[income_idx]
    
    predicted_ydist_high = ranker_func(row_high_income)
    
    # If the improvement is negligible, we use that as the reason
    improvement = predicted_ydist - predicted_ydist_high
    
    if improvement < 0.05:
        reason = ("Increasing salary alone is insufficient because our analysis shows "
                  "that other factors like job role and work-life balance have a "
                  "much stronger impact on retention than income adjustments.")
    else:
        reason = (f"While increasing salary helps, this arrangement ranks in the top "
                  f"{int((1-p)*100)}% of records, suggesting that non-monetary "
                  "factors are already optimized.")
                  
    return p, reason

import ezr
import random

def choose(header, rows, label):
    """
    Find the best working arrangement for laboratory staff using at most 50 labels.
    We filter for 'Laboratory Technician' in the 'jobRole' column, then use
    the 'acquire' strategy to find the best row.
    """
    # Identify the jobRole column index
    job_role_idx = header.index('jobRole')
    
    # Filter rows for Laboratory Technicians
    # We keep the original index to return it later
    indexed_rows = [(i, r) for i, r in enumerate(rows) if r[job_role_idx] == 'Laboratory Technician']
    lab_rows = [r for _, r in indexed_rows]
    
    # Map these filtered rows to their original indices
    row_to_idx = {id(r): i for i, r in indexed_rows}
    
    # Create a table for the lab staff
    tbl = ezr.Tbl([header] + lab_rows)
    
    # Define the oracle to reveal goals for the specific row
    def oracle_wrapper(row):
        idx = row_to_idx[id(row)]
        labeled_data = label(idx)
        # Update the row in place with the revealed goals
        for i, val in enumerate(labeled_data):
            row[i] = val
        # Update the table's column statistics with the new information
        # Only update columns that are goals
        for at, col in tbl.cols.items():
            if header[at].endswith('+') or header[at].endswith('-'):
                ezr.add(col, labeled_data[at], 1)
        return row

    # Run the acquisition process on the lab staff subset
    # We use the 'acquire' strategy to intelligently pick rows to label
    labeled_rows = ezr.acquire(tbl, cap=50, label=oracle_wrapper)
    
    # Find the best row among the ones we labeled
    # A row is better if its ydist is lower (closer to goals)
    best_row = min(labeled_rows, key=lambda r: ezr.ydist(tbl, r))
    
    # Return the original index of the best row found
    return row_to_idx[id(best_row)]
