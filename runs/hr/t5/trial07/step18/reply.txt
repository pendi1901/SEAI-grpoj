import ezr
import random

def fix(header, row, levers):
    """
    Given a bad row, suggest changes to levers to improve its ydist.
    We evaluate the change by predicting the ydist improvement using 
    the tree-based ranker.
    """
    # Create a table to provide context for ydist and ranker
    # We need the full dataset to build the ranker
    filename = "WA_Fn-UseC_-HR-Employee-Attrition.n1000.train.csv"
    data = list(ezr.csv(filename))
    header_data = data[0]
    rows_data = data[1:]
    tbl = ezr.Tbl(data)
    
    # Get the ranker function
    ranker_func = ezr.ranker(tbl)
    
    best_change = {}
    best_improvement = -1.0
    
    # Current performance
    current_score = ranker_func(row)
    
    for col_name, allowed in levers.items():
        idx = header.index(col_name)
        original_val = row[idx]
        
        # Generate candidates
        candidates = []
        if isinstance(allowed, (list, tuple)) and not isinstance(allowed[0], (str)):
            # Numeric (low, high)
            low, high = allowed
            candidates = [low, (low + high) / 2, high]
        else:
            # Symbolic
            candidates = [v for v in allowed if v != original_val]
            
        for val in candidates:
            # Create a copy and modify
            new_row = list(row)
            new_row[idx] = val
            
            # Predict improvement
            new_score = ranker_func(new_row)
            improvement = current_score - new_score # Lower ydist is better
            
            if improvement > best_improvement:
                best_improvement = improvement
                best_change = {col_name: val}
            
    return best_change

if __name__ == "__main__":

    data = list(ezr.csv(filename))

    levers = {
        "businessTravel": ["Non-Travel", "Travel_Frequently", "Travel_Rarely"],
        "overTime": ["No", "Yes"],
        "TrainingTimesLastYear": (0, 6),
        "PercentSalaryHike": (11, 25),
        "StockOptionLevel": (0, 3),
        "JobInvolvement": (1, 4),
        "WorkLifeBalance": (1, 4)
    }
    
    # Pick a "bad" row (high ydist)
    tbl = ezr.Tbl(data)
    scored = sorted([(ezr.ydist(tbl, r), r) for r in rows], key=lambda x: x[0], reverse=True)
    bad_row = scored[0][1]
    
    suggestion = fix(header, bad_row, levers)
    
    print("Analysis: Recommended non-monetary changes for staff retention:")
    print(f"Original row (sample): {bad_row[:5]}...")
    print(f"Suggested change: {suggestion}")
    print("This change is predicted to improve the employee's retention metrics.")
