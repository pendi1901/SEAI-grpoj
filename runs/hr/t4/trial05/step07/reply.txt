import csv
from step02 import guess

def choose(header, rows, label):
    """
    Finds the best working arrangement by sampling 50 rows.
    We use a simple strategy: sample 50 rows, label them, and pick the one
    with the highest 'YearsAtCompany+' value.
    """
    # Identify goal column
    goal_idx = next(i for i, name in enumerate(header) if name.endswith('+'))
    
    # We can label up to 50 rows. Let's pick 50 evenly spaced rows.
    n = len(rows)
    indices = [int(i * n / 50) for i in range(50)]
    
    best_idx = -1
    max_val = -1.0
    
    for i in indices:
        # Get true value
        true_vals = label(i)
        val = true_vals[0] # Assuming first goal is the one we care about
        
        if val > max_val:
            max_val = val
            best_idx = i
            
    return best_idx

if __name__ == "__main__":
    filename = 'WA_Fn-UseC_-HR-Employee-Attrition.n1000.train.csv'
    rows = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = []
            for i, val in enumerate(row):
                if header[i][0].isupper():
                    processed_row.append(float(val))
                else:
                    processed_row.append(val)
            rows.append(processed_row)

    # Keep track of labeled rows to satisfy the 50-limit constraint
    labeled_data = {}
    def label(i):
        if i not in labeled_data:
            if len(labeled_data) >= 50:
                raise ValueError("Limit of 50 labels exceeded.")
            # Find goal index
            goal_idx = next(idx for idx, name in enumerate(header) if name.endswith('+') or name.endswith('-'))
            labeled_data[i] = [rows[i][goal_idx]]
        return labeled_data[i]

    best_row_idx = choose(header, rows, label)
    
    print(f"Analysis complete.")
    print(f"The best working arrangement found is at row index {best_row_idx}.")
    print(f"Years at company for this employee: {rows[best_row_idx][header.index('YearsAtCompany+')]}")
