    """
    Identifies staff who stay 'unusually long'.
    'YearsAtCompany+' is a goal to maximize.
    We calculate the 95th percentile of 'YearsAtCompany+' and return
    the indices of rows that meet or exceed this threshold.
    """
    # Identify index for 'YearsAtCompany+'
    idx = header.index('YearsAtCompany+')
    
    # Extract values
    years = [row[idx] for row in rows]
    
    # Calculate threshold for "unusually long" (top 5%)
    sorted_years = sorted(years, reverse=True)
    limit = max(1, len(rows) // 20)
    threshold = sorted_years[limit - 1]
    
    # Find indices of rows meeting the criteria
    best_indices = [i for i, val in enumerate(years) if val >= threshold]
    
    # Ensure we return at most 5% (the logic above naturally fits this)
    return best_indices[:limit]

import csv
from scipy import stats

def certify(xs, ys):
    """
    Returns True if the difference between two groups is not statistically 
    significant (noise), False if they really differ.
    """
    t_stat, p_value = stats.ttest_ind(xs, ys, equal_var=False)
    # If p-value > 0.05, we fail to reject the null hypothesis (they are the same)
    return p_value > 0.05

import csv
from sklearn.ensemble import IsolationForest

def spot(header, rows):
    """
    Identifies rows that look 'wrong' (outliers) using an Isolation Forest.
    Returns the indices of the most suspicious rows, sorted by suspicion level.
    """
    # Prepare data: encode symbolic columns to numeric
    encoded_rows = []
    for row in rows:
        new_row = []
        for i, val in enumerate(row):
            if isinstance(val, str):
                # Simple hash-based encoding for symbolic columns
                new_row.append(float(hash(val) % 1000))
            else:
                new_row.append(float(val))
        encoded_rows.append(new_row)

    # Isolation Forest is effective for anomaly detection
    # contamination='auto' lets the model decide the proportion of outliers
    clf = IsolationForest(contamination=0.05, random_state=42)
    
    # decision_function returns the anomaly score (lower is more abnormal)
    scores = clf.fit_predict(encoded_rows)
    decision_scores = clf.decision_function(encoded_rows)
    
    # Get indices of rows identified as outliers (-1)
    # Sort by decision_scores (most negative = most suspicious)
    suspicious_indices = [i for i, score in enumerate(decision_scores) if score < 0]
    suspicious_indices.sort(key=lambda i: decision_scores[i])
    
    return suspicious_indices

import csv

def race(header, rows, label):
    """
    Compares three strategies for finding the best employee row.
    Budget: 50 labels total.
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith('+') or name.endswith('-')]
    
    def get_score(row_idx):
        # Higher score is better. Normalize goals.
        vals = label(row_idx)
        score = 0
        for i, g_idx in enumerate(goal_indices):
            val = vals[i]
            # If goal ends in +, higher is better. If -, lower is better.
            if header[g_idx].endswith('+'):
                score += val
            else:
                score -= val
        return score

    # Strategy 1: Random
    def strategy_random():
        indices = random.sample(range(len(rows)), 16)
        best_idx = max(indices, key=get_score)
        return best_idx, get_score(best_idx)

    # Strategy 2: Greedy (Sample first 16)
    def strategy_greedy():
        indices = list(range(16))
        best_idx = max(indices, key=get_score)
        return best_idx, get_score(best_idx)

    # Strategy 3: Search (Sample last 16)
    def strategy_search():
        indices = list(range(len(rows) - 16, len(rows)))
        best_idx = max(indices, key=get_score)
        return best_idx, get_score(best_idx)

    strategies = {
        "random": strategy_random,
        "greedy": strategy_greedy,
        "search": strategy_search
    }

    results = {}
    for name, func in strategies.items():
        idx, score = func()
        results[name] = (idx, score)

    winner = max(results, key=lambda n: results[n][1])
    
    return {
        "winner": winner,
        "row": results[winner][0],
        "tried": list(strategies.keys())
    }

import csv
from sklearn.ensemble import RandomForestRegressor

def blame(header, rows):
    """
    Identifies input columns that have the least impact on the goal columns.
    We use Random Forest feature importance to rank all inputs, then return
    the names of the columns that have the lowest importance scores.
    """
    # Identify goal columns (ending in + or -)
    goal_indices = [i for i, name in enumerate(header) if name.endswith('+') or name.endswith('-')]
    input_indices = [i for i in range(len(header)) if i not in goal_indices]
    
    # Prepare data: encode symbolic columns to numeric
    encoded_rows = []
    for row in rows:
        new_row = []
        for i, val in enumerate(row):
            if isinstance(val, str):
                new_row.append(float(hash(val) % 1000))
            else:
                new_row.append(float(val))
        encoded_rows.append(new_row)
    
    # We aggregate importance across all goal columns
    total_importances = [0.0] * len(input_indices)
    
    for g_idx in goal_indices:
        X = [[r[i] for i in input_indices] for r in encoded_rows]
        y = [r[g_idx] for r in encoded_rows]
        
        model = RandomForestRegressor(n_estimators=50, random_state=42)
        model.fit(X, y)
        
        for i, imp in enumerate(model.feature_importances_):
            total_importances[i] += imp
            
    # Normalize and rank
    feature_names = [header[i] for i in input_indices]
    ranked = sorted(zip(feature_names, total_importances), key=lambda x: x[1])
    
    # Return the bottom 5 columns that have the least influence
    return [name for name, score in ranked[:5]]

import csv
from sklearn.ensemble import IsolationForest

def spot(header, rows):
    """
    Identifies rows that are 'unlike anyone we have employed' (outliers)
    using an Isolation Forest. Returns the indices of the most suspicious 
    rows, sorted by suspicion level (most suspicious first).
    """
    # Prepare data: encode symbolic columns to numeric
    encoded_rows = []
    for row in rows:
        new_row = []
        for i, val in enumerate(row):
            if isinstance(val, str):
                # Simple hash-based encoding for symbolic columns
                new_row.append(float(hash(val) % 1000))
            else:
                new_row.append(float(val))
        encoded_rows.append(new_row)

    # Isolation Forest is effective for anomaly detection.
    # We set contamination to a small value to find the most extreme outliers.
    clf = IsolationForest(contamination=0.01, random_state=42)
    
    # decision_function returns the anomaly score (lower is more abnormal)
    decision_scores = clf.fit(encoded_rows).decision_function(encoded_rows)
    
    # Get indices of rows identified as outliers (score < 0)
    # Sort by decision_scores (most negative = most suspicious)
    suspicious_indices = [i for i, score in enumerate(decision_scores) if score < 0]
    suspicious_indices.sort(key=lambda i: decision_scores[i])
    
    return suspicious_indices
