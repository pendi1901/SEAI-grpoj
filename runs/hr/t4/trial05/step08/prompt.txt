import csv

def remember(header, rows):
    """
    Identifies staff who stay 'unusually long'.
    'YearsAtCompany+' is a goal to maximize.
    We calculate the 95th percentile of 'YearsAtCompany+' and return
    the indices of rows that meet or exceed this threshold.
    """
    # Identify index for 'YearsAtCompany+'
    idx = header.index('YearsAtCompany+')
    
    # Extract values
    years = [row[idx] for row in rows]
    
    # Calculate threshold for "unusually long" (top 5%)
    sorted_years = sorted(years, reverse=True)
    limit = max(1, len(rows) // 20)
    threshold = sorted_years[limit - 1]
    
    # Find indices of rows meeting the criteria
    best_indices = [i for i, val in enumerate(years) if val >= threshold]
    
    # Ensure we return at most 5% (the logic above naturally fits this)
    return best_indices[:limit]

def guess(header, train, test):
    """
    Estimates the first goal column ('YearsAtCompany+') for test rows
    using a K-Nearest Neighbors regressor.
    """
    # Identify goal column (first one ending in + or -)
    goal_idx = -1
    for i, name in enumerate(header):
        if name.endswith('+') or name.endswith('-'):
            goal_idx = i
            break
    
    # Prepare data: convert symbolic to numeric
    def encode_rows(data):
        encoded = []
        for row in data:
            new_row = []
            for i, val in enumerate(row):
                if isinstance(val, str):
                    # Simple label encoding for symbolic columns
                    new_row.append(float(hash(val) % 1000))
                else:
                    new_row.append(float(val))
            encoded.append(new_row)
        return encoded

    test_encoded = encode_rows(test)

import csv
from scipy import stats

def certify(xs, ys):
    """
    Returns True if the difference between two groups is not statistically 
    significant (noise), False if they really differ.
    """
    t_stat, p_value = stats.ttest_ind(xs, ys, equal_var=False)
    # If p-value > 0.05, we fail to reject the null hypothesis (they are the same)
    return p_value > 0.05

    """
    Finds the best working arrangement by sampling 50 rows.
    We use a simple strategy: sample 50 rows, label them, and pick the one
    with the highest 'YearsAtCompany+' value.
    """

    goal_idx = next(i for i, name in enumerate(header) if name.endswith('+'))
    
    # We can label up to 50 rows. Let's pick 50 evenly spaced rows.
    n = len(rows)
    indices = [int(i * n / 50) for i in range(50)]

        # Get true value
        true_vals = label(i)
        val = true_vals[0] # Assuming first goal is the one we care about
        
        if val > max_val:
