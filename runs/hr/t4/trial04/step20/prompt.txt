import csv
from scipy import stats

def certify(xs, ys):
    """
    Performs a Welch's t-test to determine if two samples are 
    statistically different. Returns True if they are likely from 
    the same distribution (p > 0.05), False otherwise.
    """
    t_stat, p_value = stats.ttest_ind(xs, ys, equal_var=False)
    return p_value > 0.05

def flow(header, row, seen):
    """
    Decides if the current row is a candidate for a 'retention conversation'.
    We compare the current row's 'YearsAtCompany+' against the distribution 
    of 'seen' rows. If the current row is in the top 5% of the combined 
    history (or if it's the first row), we flag it.
    """
    goal_idx = header.index('YearsAtCompany+')
    
    # If no history, we can't compare, so we take the first one
    if not seen:
        return True
    
    # Combine current row with history to determine the threshold
    all_rows = seen + [row]
    years = [r[goal_idx] for r in all_rows]
    
    # Calculate threshold for top 5%
    sorted_years = sorted(years, reverse=True)
    limit = max(1, len(all_rows) // 20)
    threshold = sorted_years[limit - 1]
    
    # If this row meets or exceeds the threshold, flag it
    return row[goal_idx] >= threshold

def justify(header, rows, row):
    """
    Estimates the performance of a given row against historical data.
    Returns (p, reason) where p is the fraction of rows that 'row' is better than.
    """
    # Identify goal column
    goal_col = next(c for c in header if c.endswith('+') or c.endswith('-'))
    goal_idx = header.index(goal_col)
    
    # Prepare training data
    df = pd.DataFrame(rows, columns=header)
    X = df.drop(columns=[goal_col])
    y = df[goal_col]
    
    # Encode symbolic columns consistently
    encoders = {}
    for col in X.columns:
        if X[col].dtype == 'object':
            le = LabelEncoder()
            X[col] = le.fit_transform(X[col].astype(str))
            encoders[col] = le
        else:
            X[col] = X[col].astype(float)
            
    # Train model
    model = RandomForestRegressor(n_estimators=100, random_state=42)
    model.fit(X, y)
    
    # Prepare input row
    input_df = pd.DataFrame([row], columns=header).drop(columns=[goal_col])
    for col in input_df.columns:
        if col in encoders:
            input_df[col] = encoders[col].transform(input_df[col].astype(str))
        else:
            input_df[col] = input_df[col].astype(float)
            
    pred = model.predict(input_df)[0]
    
    # Calculate fraction of rows that 'row' is better than
    history = [r[goal_idx] for r in rows]
    if goal_col.endswith('+'):
        better_than = [h for h in history if pred > h]
    else:
        better_than = [h for h in history if pred < h]
        
    p = len(better_than) / len(history)
    reason = f"This arrangement is predicted to perform better than {p:.1%} of historical records regarding {goal_col}."
    
    return (p, reason)

def choose(header, rows, label):

    Finds the best working arrangement by sampling 50 rows, 
    labeling them, and using a Random Forest to predict the best row.
    """
    goal_col = next(c for c in header if c.endswith('+') or c.endswith('-'))
    goal_idx = header.index(goal_col)
    is_maximize = goal_col.endswith('+')

    indices_to_label = random.sample(range(len(rows)), min(50, len(rows)))

def fix(header, row, levers):
    """
    Recommends changes to a 'bad' row to improve its goal columns.
    We simulate changes for each lever and pick the one that results
    in the best predicted outcome.
    """
    goal_col = next(c for c in header if c.endswith('+') or c.endswith('-'))
    is_maximize = goal_col.endswith('+')
    
    filename = 'WA_Fn-UseC_-HR-Employee-Attrition.n1000.train.csv'
    rows = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header_full = next(reader)
        for r in reader:
            rows.append([float(val) if header_full[i][0].isupper() else val for i, val in enumerate(r)])
    
    df = pd.DataFrame(rows, columns=header)
    X = df.drop(columns=[goal_col])
    y = df[goal_col]
    
    encoders = {}
    for col in X.columns:
        if X[col].dtype == 'object':
            le = LabelEncoder()
            X[col] = le.fit_transform(X[col].astype(str))
            encoders[col] = le
        else:
            X[col] = X[col].astype(float)
            
    model = RandomForestRegressor(n_estimators=50, random_state=42)
    model.fit(X, y)
    
    best_val = -float('inf') if is_maximize else float('inf')
    best_change = {}
    
    for col, allowed in levers.items():
        idx = header.index(col)
        original_val = row[idx]
        
        test_vals = []
        if isinstance(allowed, tuple):
            test_vals = [allowed[0], allowed[1], (allowed[0] + allowed[1]) / 2]
        else:
            test_vals = allowed
            
        for val in test_vals:
            if val == original_val: continue
            
            temp_row = list(row)
            temp_row[idx] = val
            
            input_df = pd.DataFrame([temp_row], columns=header).drop(columns=[goal_col])
            for c in input_df.columns:
                if c in encoders:
                    input_df[c] = encoders[c].transform(input_df[c].astype(str))
                else:
                    input_df[c] = input_df[c].astype(float)
            
            pred = model.predict(input_df)[0]
            
            if (is_maximize and pred > best_val) or (not is_maximize and pred < best_val):
                best_val = pred
                best_change = {col: val}
                
    return best_change

def spot(header, rows):
    """
    Identifies rows that look wrong (outliers) using an Isolation Forest.
    Returns the indices of the rows, sorted by suspicion (most suspicious first).
    """
    df = pd.DataFrame(rows, columns=header)
    
    # Prepare features: encode symbolic columns and ensure numeric are float
    X = pd.DataFrame()
    for col in header:
        if df[col].dtype == 'object':
            le = LabelEncoder()
            X[col] = le.fit_transform(df[col].astype(str))
        else:
            X[col] = df[col].astype(float).fillna(0)
            
    # Isolation Forest detects anomalies
    # contamination=0.05 assumes roughly 5% of data might be errors
    clf = IsolationForest(n_estimators=100, contamination=0.05, random_state=42)
    
    # decision_function returns anomaly score; lower values are more abnormal
    scores = clf.decision_function(X)
    
    # Get indices sorted by score (ascending, so most negative/abnormal first)
    suspicious_indices = sorted(range(len(scores)), key=lambda i: scores[i])
    
    # Return only those identified as outliers (score < 0)
    return [i for i in suspicious_indices if scores[i] < 0]

import csv
import random
import pandas as pd
from sklearn.ensemble import RandomForestRegressor
from sklearn.preprocessing import LabelEncoder
from step05 import blame

def race(header, rows, label):
    """
    Compares three strategies for finding the best row using a shared budget of 50 labels.
    """
    goal_col = next(c for c in header if c.endswith('+') or c.endswith('-'))
    goal_idx = header.index(goal_col)
    is_maximize = goal_col.endswith('+')
    
    budget = 50
    labeled_indices = {}
    
    def get_label(i):
        if i not in labeled_indices:
            if len(labeled_indices) >= budget:
                return None
            labeled_indices[i] = label(i)[0]
        return labeled_indices[i]

    # Strategy 1: Random
    def strategy_random():
        indices = random.sample(range(len(rows)), 16)
        for i in indices: get_label(i)
        return max(indices, key=lambda i: labeled_indices[i]) if is_maximize else min(indices, key=lambda i: labeled_indices[i])

    # Strategy 2: Importance (using blame on a numeric-only subset)
    def strategy_importance():
        # Filter rows to ensure numeric columns are actually numeric for blame()
        numeric_rows = []
        for r in rows[:200]:
            if isinstance(r[goal_idx], (int, float)):
                numeric_rows.append(r)
        
        top_features = blame(header, numeric_rows)
        # Pick rows with high/low values in top features
        indices = sorted(range(len(rows)), key=lambda i: sum(rows[i][header.index(f)] for f in top_features if isinstance(rows[i][header.index(f)], (int, float))), reverse=is_maximize)[:16]
        for i in indices: get_label(i)
        return max(indices, key=lambda i: labeled_indices[i]) if is_maximize else min(indices, key=lambda i: labeled_indices[i])

    # Strategy 3: Model-based
    def strategy_model():
        indices = random.sample(range(len(rows)), 18)
        for i in indices: get_label(i)
        
        train_idx = list(labeled_indices.keys())
        df_train = pd.DataFrame([rows[i] for i in train_idx], columns=header)
        X = df_train.drop(columns=[goal_col])
        y = pd.Series([labeled_indices[i] for i in train_idx])
        
        encoders = {}
        for col in X.columns:
            if X[col].dtype == 'object':
                le = LabelEncoder()
                X[col] = le.fit_transform(X[col].astype(str))
                encoders[col] = le
            else:
                X[col] = X[col].astype(float)
        
        model = RandomForestRegressor(n_estimators=50).fit(X, y)
        
        X_all = pd.DataFrame(rows, columns=header).drop(columns=[goal_col])
        for col in X_all.columns:
            if col in encoders:
                X_all[col] = encoders[col].transform(X_all[col].astype(str))
            else:
                X_all[col] = X_all[col].astype(float)
        
        preds = model.predict(X_all)
        best_idx = max(range(len(preds)), key=lambda i: preds[i]) if is_maximize else min(range(len(preds)), key=lambda i: preds[i])
        get_label(best_idx)
        return best_idx

    results = {
        "random": strategy_random(),
        "importance": strategy_importance(),
        "model": strategy_model()
    }
    
    best_strategy = max(results, key=lambda s: labeled_indices[results[s]]) if is_maximize else min(results, key=lambda s: labeled_indices[results[s]])
    
    return {"winner": best_strategy, "row": results[best_strategy], "tried": list(results.keys())}

import csv

from sklearn.preprocessing import LabelEncoder

def flow(header, row, seen):
    """
    Detects if the current row is an outlier compared to the 'seen' history.
    We use an Isolation Forest to determine if the new row deviates from 
    the distribution of the previously seen data.
    """
    if len(seen) < 20:
        return False

    # Combine seen rows and current row into a temporary dataset
    all_rows = seen + [row]
    
    # Encode data for Isolation Forest
    X = []
    for r in all_rows:
        encoded_row = []
        for i, val in enumerate(r):
            if isinstance(val, (int, float)):
                encoded_row.append(float(val))
            else:
                # Simple hash or index-based encoding for symbolic data
                encoded_row.append(float(hash(val) % 1000))
        X.append(encoded_row)
    
    # Use Isolation Forest to detect if the last row is an anomaly
    # We train on the 'seen' data and predict for the new row
    clf = IsolationForest(n_estimators=50, contamination=0.05, random_state=42)
    clf.fit(X[:-1])
    
    # If the score is negative, it's considered an anomaly
    score = clf.decision_function([X[-1]])
    return score[0] < 0

from sklearn.preprocessing import LabelEncoder

def blame(header, rows):
    """
    Identifies input columns that have the least impact on the goal columns.
    We use a Random Forest Regressor to calculate feature importances and
    return the columns with the lowest importance scores.
    """
    # Identify goal columns
    goal_cols = [c for c in header if c.endswith('+') or c.endswith('-')]
    input_cols = [c for c in header if c not in goal_cols]
    
    df = pd.DataFrame(rows, columns=header)
    
    # Prepare features: Encode symbolic columns to numeric
    X = pd.DataFrame()
    for col in input_cols:
        if df[col].dtype == 'object':
            le = LabelEncoder()
            X[col] = le.fit_transform(df[col].astype(str))
        else:
            X[col] = df[col].astype(float).fillna(0)
            
    # We aggregate importance across all goal columns
    total_importances = pd.Series(0.0, index=input_cols)
    
    for goal in goal_cols:
        y = df[goal]
        model = RandomForestRegressor(n_estimators=50, random_state=42)
        model.fit(X, y)
        total_importances += pd.Series(model.feature_importances_, index=input_cols)
    
    # Normalize and sort
    avg_importances = total_importances / len(goal_cols)
    
    # Return the columns with the lowest importance (the ones we can stop collecting)
    # We return the bottom 5 features as candidates to stop collecting
    return avg_importances.nsmallest(5).index.tolist()

import csv
import pandas as pd
from sklearn.ensemble import IsolationForest
from sklearn.preprocessing import LabelEncoder

def spot(header, rows):
    """
    Identifies rows that are 'unlike anyone we have employed' (outliers).
    We use an Isolation Forest to detect rows that deviate from the 
    historical distribution. Returns indices of suspicious rows, 
    most suspicious first.
    """
    df = pd.DataFrame(rows, columns=header)
    
    # Prepare features: encode symbolic columns and ensure numeric are float
    X = pd.DataFrame()
    for col in header:
        if df[col].dtype == 'object':
            le = LabelEncoder()
            X[col] = le.fit_transform(df[col].astype(str))
        else:
            # Ensure numeric columns are float, filling NaNs if any
            X[col] = df[col].astype(float).fillna(0.0)
            
    # Isolation Forest is effective for detecting anomalies in high-dimensional data
    # contamination='auto' lets the model decide the threshold
    clf = IsolationForest(n_estimators=100, contamination=0.05, random_state=42)
    
    # decision_function returns anomaly score; lower values are more abnormal
    scores = clf.fit(X).decision_function(X)
    
    # Get indices sorted by score (ascending, so most negative/abnormal first)
    suspicious_indices = sorted(range(len(scores)), key=lambda i: scores[i])
    
    # Return only those identified as outliers (score < 0)
    return [i for i in suspicious_indices if scores[i] < 0]
