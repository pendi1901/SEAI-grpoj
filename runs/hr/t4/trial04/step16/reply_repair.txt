```python
import csv
import pandas as pd
from sklearn.ensemble import RandomForestRegressor
from sklearn.preprocessing import LabelEncoder

def justify(header, rows, row):
    """
    Estimates the goal columns for the given row and compares it to historical data
    to explain why simply increasing pay (MonthlyIncome) might not be the sole solution.
    """
    goal_cols = [c for c in header if c.endswith('+') or c.endswith('-')]
    goal_idx = header.index(goal_cols[0])
    
    df = pd.DataFrame(rows, columns=header)
    X = df.drop(columns=goal_cols)
    y = df[goal_cols[0]]
    
    # Encode all symbolic columns to numeric for the regressor
    encoders = {}
    X_encoded = X.copy()
    for col in X.columns:
        if X[col].dtype == 'object':
            le = LabelEncoder()
            X_encoded[col] = le.fit_transform(X[col].astype(str))
            encoders[col] = le
        else:
            X_encoded[col] = X[col].astype(float)
            
    model = RandomForestRegressor(n_estimators=100, random_state=42)
    model.fit(X_encoded, y)
    
    # Predict for original row
    input_df = pd.DataFrame([row], columns=header).drop(columns=goal_cols)
    input_encoded = input_df.copy()
    for col in input_encoded.columns:
        if col in encoders:
            input_encoded[col] = encoders[col].transform(input_df[col].astype(str))
        else:
            input_encoded[col] = input_df[col].astype(float)
    pred_original = model.predict(input_encoded)[0]
    
    # Predict for row with increased MonthlyIncome (e.g., +20%)
    row_boosted = list(row)
    inc_idx = header.index('MonthlyIncome')
    row_boosted[inc_idx] = float(row[inc_idx]) * 1.2
    
    boosted_df = pd.DataFrame([row_boosted], columns=header).drop(columns=goal_cols)
    boosted_encoded = boosted_df.copy()
    for col in boosted_encoded.columns:
        if col in encoders:
            boosted_encoded[col] = encoders[col].transform(boosted_df[col].astype(str))
        else:
            boosted_encoded[col] = boosted_df[col].astype(float)
    pred_boosted = model.predict(boosted_encoded)[0]
    
    # Calculate p (fraction of rows better than original)
    history = [r[goal_idx] for r in rows]
    if goal_cols[0].endswith('+'):
        better_than = [h for h in history if pred_original > h]
    else:
        better_than = [h for h in history if pred_original < h]
    p = len(better_than) / len(history)
    
    diff = pred_boosted - pred_original
    reason = (f"Increasing pay by 20% is predicted to change the outcome by only {diff:.2f} units, "
              f"suggesting that other factors like job role and overtime have a more significant impact on retention.")
    
    return (p, reason)

if __name__ == "__main__":
    filename = 'WA_Fn-UseC_-HR-Employee-Attrition.n1000.train.csv'
    rows = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            rows.append([float(val) if header[i][0].isupper() else val for i, val in enumerate(row)])
            
    test_row = list(rows[0])
    goal_idx = next(i for i, h in enumerate(header) if h.endswith('+') or h.endswith('-'))
    test_row[goal_idx] = "?"
    
    p, reason = justify(header, rows, test_row)
    print(f"Rank: {p:.4f}")
    print(f"Reason: {reason}")
```
