import csv
import pandas as pd
from sklearn.ensemble import RandomForestRegressor
from sklearn.preprocessing import LabelEncoder
from step02 import guess

def justify(header, rows, row):
    """
    Estimates the performance of a given row against historical data.
    Returns (p, reason) where p is the percentile rank and reason is a summary.
    """
    # Identify goal column (first column ending in + or -)
    goal_col = next(c for c in header if c.endswith('+') or c.endswith('-'))
    goal_idx = header.index(goal_col)
    
    # Extract historical values for the goal
    history = [r[goal_idx] for r in rows]
    
    # Predict the value for the input row
    # We use the guess logic from step02, but need to handle the "?"
    # Create a dummy test set with the single row
    test_row = list(row)
    test_row[goal_idx] = 0.0 # Placeholder
    
    # Get prediction
    pred = guess(header, rows, [test_row])[0]
    
    # Calculate percentile rank (fraction of rows that 'row' is better than)
    # If goal ends in +, higher is better. If -, lower is better.
    if goal_col.endswith('+'):
        better_than = [h for h in history if pred > h]
    else:
        better_than = [h for h in history if pred < h]
        
    p = len(better_than) / len(history)
    
    reason = f"This team arrangement ranks in the top {int((1-p)*100)}% of all historical records based on the predicted {goal_col}."
    
    return (p, reason)

if __name__ == "__main__":
    filename = 'WA_Fn-UseC_-HR-Employee-Attrition.n1000.train.csv'
    rows = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = [float(val) if header[i][0].isupper() else val for i, val in enumerate(row)]
            rows.append(processed_row)
    
    # Example: Test with the first row, masking its goal
    test_row = list(rows[0])
    test_row[next(i for i, h in enumerate(header) if h.endswith('+') or h.endswith('-'))] = "?"
    
    p, reason = justify(header, rows, test_row)
    print(f"Rank: {p:.4f}")
    print(f"Reason: {reason}")
