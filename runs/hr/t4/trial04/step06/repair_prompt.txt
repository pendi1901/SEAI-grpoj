def guess(header, train, test):
    """
    Estimates the first goal column ('YearsAtCompany+') for test rows
    using a Random Forest Regressor after encoding symbolic columns.

    # Prepare DataFrames
    df_train = pd.DataFrame(train, columns=header)
    df_test = pd.DataFrame(test, columns=header)
    
    # Separate features and target
    X = df_train.drop(columns=[goal_col])
    y = df_train[goal_col]
    X_test = df_test.drop(columns=[goal_col])
    
    # Encode symbolic columns
    # We must ensure all columns are numeric for the regressor

import csv
from scipy import stats

def certify(xs, ys):
    """
    Performs a Welch's t-test to determine if two samples are 
    statistically different. Returns True if they are likely from 
    the same distribution (p > 0.05), False otherwise.
    """
    t_stat, p_value = stats.ttest_ind(xs, ys, equal_var=False)
    return p_value > 0.05

def analyze_overtime_impact(header, rows):
    """
    Compares 'YearsInCurrentRole+' for employees who work overtime 
    vs those who do not.
    """
    ot_idx = header.index('overTime')
    role_idx = header.index('YearsInCurrentRole+')
    
    yes_ot = [row[role_idx] for row in rows if row[ot_idx] == 'Yes']
    no_ot = [row[role_idx] for row in rows if row[ot_idx] == 'No']
    
    is_noise = certify(yes_ot, no_ot)
    
    print(f"Average years in role (Overtime): {sum(yes_ot)/len(yes_ot):.2f}")
    print(f"Average years in role (No Overtime): {sum(no_ot)/len(no_ot):.2f}")
    
    if is_noise:
        print("Conclusion: The difference is likely just noise.")
    else:
        print("Conclusion: The difference is statistically significant.")
    
    return is_noise

def flow(header, row, seen):
    """
    Decides if the current row is a candidate for a 'retention conversation'.
    We compare the current row's 'YearsAtCompany+' against the distribution 
    of 'seen' rows. If the current row is in the top 5% of the combined 
    history (or if it's the first row), we flag it.
    """
    goal_idx = header.index('YearsAtCompany+')
    
    # If no history, we can't compare, so we take the first one
    if not seen:
        return True
    
    # Combine current row with history to determine the threshold
    all_rows = seen + [row]
    years = [r[goal_idx] for r in all_rows]
    
    # Calculate threshold for top 5%
    sorted_years = sorted(years, reverse=True)
    limit = max(1, len(all_rows) // 20)
    threshold = sorted_years[limit - 1]
    
    # If this row meets or exceeds the threshold, flag it
    return row[goal_idx] >= threshold

def blame(header, rows):
    """
    Identifies input columns that best explain short stays.
    'YearsAtCompany+' is a goal to maximize. We define 'worst' rows as those
    in the bottom 25% of 'YearsAtCompany+'. We use a Random Forest
    to identify which inputs correlate with these short stays.
    """
    goal_col = 'YearsAtCompany+'
    df = pd.DataFrame(rows, columns=header)
    
    # Define "worst" as the bottom 25% of tenure
    threshold = df[goal_col].quantile(0.25)
    df['is_short_stay'] = (df[goal_col] <= threshold).astype(int)
    
    # Prepare features: exclude goal columns
    input_cols = [c for c in header if not (c.endswith('+') or c.endswith('-'))]
    X = df[input_cols].copy()
    y = df['is_short_stay']
    
    # Encode symbolic columns to numeric
    for col in X.columns:
        if X[col].dtype == 'object':
            le = LabelEncoder()
            X[col] = le.fit_transform(X[col].astype(str))
        else:
            # Ensure numeric columns are float and handle potential NaNs
            X[col] = X[col].astype(float).fillna(0)
            
    # Use Random Forest to find feature importance
    clf = RandomForestClassifier(n_estimators=100, random_state=42)
    clf.fit(X, y)
    
    # Get top 3 features
    importances = pd.Series(clf.feature_importances_, index=input_cols)
    top_features = importances.nlargest(3).index.tolist()
    
    return top_features
