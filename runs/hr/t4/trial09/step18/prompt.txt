    sorted_years = sorted(years, reverse=True)

    best_indices = [i for i, val in enumerate(years) if val >= threshold]

def guess(header, train, test):
    """
    Estimates the first goal column ('YearsAtCompany+') for test rows
    using a Random Forest Regressor.
    """
    # Identify goal columns
    goal_indices = [i for i, name in enumerate(header) if name.endswith('+') or name.endswith('-')]
    target_idx = goal_indices[0]
    
    # Prepare data
    X_train = []
    y_train = []
    
    # Encode symbolic columns
    encoders = {}
    for i, name in enumerate(header):
        if not name[0].isupper():
            le = LabelEncoder()
            # Combine train and test to ensure all categories are seen
            # Filter out '?' from test for fitting
            all_vals = [row[i] for row in train] + [row[i] for row in test if row[i] != '?']
            le.fit(all_vals)
            encoders[i] = le

    def transform(row):
        new_row = []
        for i, val in enumerate(row):
            if i == target_idx: continue
            if i in encoders:
                # If test row has '?', replace with a neutral value (e.g., 0)
                val_to_transform = val if val != '?' else encoders[i].classes_[0]
                new_row.append(encoders[i].transform([val_to_transform])[0])
            else:
                # Handle '?' in numeric columns if any
                new_row.append(float(val) if val != '?' else 0.0)
        return new_row

    for row in train:
        X_train.append(transform(row))
        y_train.append(float(row[target_idx]))
        
    X_test = [transform(row) for row in test]
    
    # Train model
    model = RandomForestRegressor(n_estimators=100, random_state=42)
    model.fit(X_train, y_train)
    
    return model.predict(X_test).tolist()

def certify(xs, ys):
    """
    Returns True if the difference between two groups is statistically 
    insignificant (noise), False if they really differ (p < 0.05).
    """
    t_stat, p_value = stats.ttest_ind(xs, ys, equal_var=False)
    return p_value > 0.05

from sklearn.preprocessing import LabelEncoder

def spot(header, rows):
    """
    Identifies suspicious rows using Isolation Forest.
    Rows that are outliers in the input space are considered 'wrong' or 'suspicious'.
    Returns a list of row indices, most suspicious first.
    """
    # Prepare data for anomaly detection
    X = []
    input_indices = [i for i, name in enumerate(header) if not (name.endswith('+') or name.endswith('-'))]
    
    encoders = {}
    for i in input_indices:
        if not header[i][0].isupper():
            le = LabelEncoder()
            # Fit on all data to handle categories
            le.fit([str(row[i]) for row in rows])
            encoders[i] = le

    for row in rows:
        features = []
        for i in input_indices:
            if i in encoders:
                features.append(encoders[i].transform([str(row[i])])[0])
            else:
                features.append(float(row[i]))
        X.append(features)
    
    # Isolation Forest to find anomalies
    # contamination is set to 0.05 as a heuristic for "a few" rows
    clf = IsolationForest(n_estimators=100, contamination=0.05, random_state=42)
    clf.fit(X)
    scores = clf.decision_function(X)
    
    # Lower scores are more anomalous. Sort indices by score ascending.
    suspicious_indices = np.argsort(scores)
    
    return suspicious_indices.tolist()

    Compares three strategies for finding the best row:

            le.fit([str(row[i]) for row in rows])
            encoders[i] = le

    for row in rows:
        features = []
        for i in input_indices:
            if i in encoders:
                features.append(encoders[i].transform([str(row[i])])[0])
            else:
                features.append(float(row[i]))
        X.append(features)
        y.append(float(row[target_idx]))

    # Sort by importance ascending (least important first)
    feature_importances.sort(key=lambda x: x[1])
    
    # Return the names of the columns with the lowest importance
    # We return the bottom 5 columns as candidates for removal
    return [name for name, imp in feature_importances[:5]]
