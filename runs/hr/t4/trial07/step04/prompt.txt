import csv

def remember(header, rows):
    """
    Identifies staff who stay 'unusually long'.
    We focus on 'YearsAtCompany+' (maximize) and 'YearsWithCurrManager+' (maximize).
    We calculate a score based on these two goals and return the top 5% indices.
    """
    # Identify indices for goal columns
    idx_years_at_company = header.index('YearsAtCompany+')
    idx_years_with_manager = header.index('YearsWithCurrManager+')
    
    # Calculate scores: sum of normalized values (or just raw sum since units are years)
    scores = []
    for i, row in enumerate(rows):
        # Goal: Maximize these values
        score = row[idx_years_at_company] + row[idx_years_with_manager]
        scores.append((score, i))
    
    # Sort by score descending
    scores.sort(key=lambda x: x[0], reverse=True)
    
    # Return top 5% (800 rows * 0.05 = 40 rows)
    limit = max(1, int(len(rows) * 0.05))
    return [item[1] for item in scores[:limit]]

import csv
from sklearn.ensemble import RandomForestRegressor
from sklearn.preprocessing import LabelEncoder

def guess(header, train, test):
    """
    Estimates 'YearsAtCompany+' for test rows using a Random Forest Regressor.
    Handles '?' in test data by excluding the goal column from feature encoding.
    """
    goal_idx = header.index('YearsAtCompany+')
    
    # Identify feature indices (all except goal)
    feature_indices = [i for i in range(len(header)) if i != goal_idx]
    
    # Prepare data: encode symbolic columns to numeric
    def encode_data(train_data, test_data):
        encoded_train = []
        encoded_test = []
        
        for i in feature_indices:
            col_train = [row[i] for row in train_data]
            col_test = [row[i] for row in test_data]
            
            if not header[i][0].isupper():
                le = LabelEncoder()
                le.fit([str(x) for x in (col_train + col_test)])
                encoded_train.append(le.transform([str(x) for x in col_train]))
                encoded_test.append(le.transform([str(x) for x in col_test]))
            else:
                encoded_train.append([float(x) for x in col_train])
                encoded_test.append([float(x) for x in col_test])
                
        return list(zip(*encoded_train)), list(zip(*encoded_test))

    X_train, X_test = encode_data(train, test)
    y_train = [row[goal_idx] for row in train]

    # Train model
    model = RandomForestRegressor(n_estimators=100, random_state=42)
    model.fit(X_train, y_train)
    
    # Predict
    return model.predict(X_test).tolist()

import csv
from scipy import stats

def certify(xs, ys):
    """
    Returns True if the difference between two groups is not statistically 
    significant (p > 0.05), False if they really differ.
    """
    t_stat, p_value = stats.ttest_ind(xs, ys, equal_var=False)
    return p_value > 0.05

def analyze_overtime_impact(header, rows):
    """
    Compares 'YearsInCurrentRole+' for employees who work overtime vs those who don't.
    """
    idx_overtime = header.index('overTime')
    idx_years = header.index('YearsInCurrentRole+')
    
    overtime_yes = [row[idx_years] for row in rows if row[idx_overtime] == 'Yes']
    overtime_no = [row[idx_years] for row in rows if row[idx_overtime] == 'No']
    
    is_noise = certify(overtime_yes, overtime_no)
    
    return is_noise, overtime_yes, overtime_no
