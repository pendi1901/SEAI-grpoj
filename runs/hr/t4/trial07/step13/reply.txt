import csv
from scipy import stats
import step03

def certify(xs, ys):
    """
    Returns True if the difference between two groups is not statistically 
    significant (p > 0.05), False if they really differ.
    """
    return step03.certify(xs, ys)

def analyze_training_impact(header, rows):
    """
    Compares 'YearsWithCurrManager+' for employees with high training 
    vs low training.
    """
    idx_training = header.index('TrainingTimesLastYear')
    idx_years = header.index('YearsWithCurrManager+')
    
    # Split into two groups based on median training
    trainings = [row[idx_training] for row in rows]
    median_training = sum(trainings) / len(trainings)
    
    high_training = [row[idx_years] for row in rows if row[idx_training] >= median_training]
    low_training = [row[idx_years] for row in rows if row[idx_training] < median_training]
    
    is_noise = certify(high_training, low_training)
    return is_noise, high_training, low_training

if __name__ == "__main__":
    filename = 'WA_Fn-UseC_-HR-Employee-Attrition.n1000.train.csv'
    rows = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = []
            for i, val in enumerate(row):
                if header[i][0].isupper():
                    processed_row.append(float(val))
                else:
                    processed_row.append(val)
            rows.append(processed_row)
            
    is_noise, high_group, low_group = analyze_training_impact(header, rows)
    
    print("Analysis: Do people with more training stay longer with their manager?")
    if is_noise:
        print("Result: The observed difference is likely just noise (p > 0.05).")
    else:
        print("Result: There is a statistically significant difference in tenure with manager.")
    
    avg_high = sum(high_group) / len(high_group)
    avg_low = sum(low_group) / len(low_group)
    print(f"Average years with manager (High Training): {avg_high:.2f}")
    print(f"Average years with manager (Low Training): {avg_low:.2f}")
