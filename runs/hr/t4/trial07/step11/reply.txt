```python
import csv
from sklearn.neighbors import NearestNeighbors
from sklearn.preprocessing import LabelEncoder
import numpy as np

def remember(header, rows):
    """
    Finds the 5% of employees most similar to the first employee (rows[0]).
    Similarity is based on input features (non-goal columns).
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith(('-', '+'))]
    input_indices = [i for i in range(len(header)) if i not in goal_indices]

    # Prepare feature matrix
    X = []
    for row in rows:
        features = []
        for i in input_indices:
            val = row[i]
            if isinstance(val, str):
                # Simple encoding for symbolic columns
                features.append(hash(val) % 1000)
            else:
                features.append(val)
        X.append(features)
    
    X = np.array(X)
    
    # Find 5% nearest neighbors to the first row
    n_neighbors = max(1, int(len(rows) * 0.05))
    nn = NearestNeighbors(n_neighbors=n_neighbors + 1)
    nn.fit(X)
    
    # distances, indices (excluding the first one which is the query itself)
    _, indices = nn.kneighbors([X[0]])
    return indices[0][1:].tolist()

if __name__ == "__main__":
    filename = 'WA_Fn-UseC_-HR-Employee-Attrition.n1000.train.csv'
    rows = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = []
            for i, val in enumerate(row):
                if header[i][0].isupper():
                    processed_row.append(float(val))
                else:
                    processed_row.append(val)
            rows.append(processed_row)

    indices = remember(header, rows)
    
    print("Employees most similar to the first record:")
    idx_at = header.index('YearsAtCompany+')
    for i in indices:
        print(f"- Row {i}: Stayed {rows[i][idx_at]} years")
```
