import csv
from sklearn.ensemble import RandomForestRegressor
from sklearn.preprocessing import LabelEncoder

def guess(header, train, test):
    """
    Estimates 'YearsAtCompany+' for test rows using a Random Forest Regressor.
    """
    # Identify goal column index
    goal_idx = header.index('YearsAtCompany+')
    
    # Prepare data: convert symbolic columns to numeric for scikit-learn
    def encode_data(data):
        encoded = []
        encoders = {}
        for i in range(len(header)):
            col_data = [row[i] for row in data]
            if not header[i][0].isupper():
                le = LabelEncoder()
                encoded_col = le.fit_transform([str(x) for x in col_data])
                encoded.append(encoded_col)
                encoders[i] = le
            else:
                encoded.append([float(x) for x in col_data])
        return list(zip(*encoded)), encoders

    # Prepare training features and target
    X_train = []
    y_train = []
    for row in train:
        # Use all columns except the goal column as features
        features = [row[i] for i in range(len(header)) if i != goal_idx]
        X_train.append(features)
        y_train.append(row[goal_idx])

    # Prepare test features
    X_test = []
    for row in test:
        features = [row[i] for i in range(len(header)) if i != goal_idx]
        X_test.append(features)

    # Combine to encode consistently
    all_data = X_train + X_test
    encoded_all, _ = encode_data(all_data)
    
    X_train_enc = encoded_all[:len(train)]
    X_test_enc = encoded_all[len(train):]

    # Train model
    model = RandomForestRegressor(n_estimators=100, random_state=42)
    model.fit(X_train_enc, y_train)
    
    # Predict
    return model.predict(X_test_enc).tolist()

if __name__ == "__main__":
    filename = 'WA_Fn-UseC_-HR-Employee-Attrition.n1000.train.csv'
    rows = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = []
            for i, val in enumerate(row):
                if header[i][0].isupper():
                    processed_row.append(float(val))
                else:
                    processed_row.append(val)
            rows.append(processed_row)

    # Split: 700 train, 100 test
    train_data = rows[:700]
    test_data = rows[700:]
    
    # Mask the goal column in test data
    goal_idx = header.index('YearsAtCompany+')
    masked_test = []
    for row in test_data:
        new_row = list(row)
        new_row[goal_idx] = "?"
        masked_test.append(new_row)
        
    predictions = guess(header, train_data, masked_test)
    
    print("Estimation of 'YearsAtCompany+' for new hires:")
    for i in range(min(5, len(predictions))):
        print(f"Test Row {i}: Predicted {predictions[i]:.2f} years")
