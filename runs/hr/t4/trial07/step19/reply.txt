import csv
import numpy as np
import step09

def spot(header, rows):
    """
    Identifies rows that look 'wrong' by detecting outliers in the data.
    We use the logic from step09 (Z-score analysis) to find rows that 
    deviate significantly from the mean of their columns.
    """
    # Reuse the logic from step09
    return step09.spot(header, rows)

if __name__ == "__main__":
    filename = 'WA_Fn-UseC_-HR-Employee-Attrition.n1000.train.csv'
    rows = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = []
            for i, val in enumerate(row):
                if header[i][0].isupper():
                    processed_row.append(float(val))
                else:
                    processed_row.append(val)
            rows.append(processed_row)
            
    suspicious = spot(header, rows)
    
    print("Analysis: Identifying when estimates cannot be trusted.")
    print("The following row indices represent employees who are statistical outliers.")
    print("Estimates for such unique individuals are less reliable as they deviate")
    print("significantly from our historical data patterns:")
    for idx in suspicious[:5]:
        print(f"- Row index: {idx}")
    
    print("\nLogic: Rows were flagged based on high statistical deviation (Z-score) across numeric input fields.")
