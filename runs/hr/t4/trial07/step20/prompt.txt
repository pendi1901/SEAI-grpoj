    return [item[1] for item in scores[:limit]]

def certify(xs, ys):
    """
    Returns True if the difference between two groups is not statistically 
    significant (p > 0.05), False if they really differ.
    """
    t_stat, p_value = stats.ttest_ind(xs, ys, equal_var=False)
    return p_value > 0.05

def justify(header, rows, row):

    """

                features.append(val)

        # Multi-objective: normalize goals to a single score (higher is better)

        score = 0
        for g in goal_cols:
            val = r[g]
            if header[g].endswith('-'):
                score -= val
            else:
                score += val
        y_train.append(score)

    # Train model to predict the aggregate score
    model = RandomForestRegressor(n_estimators=50, random_state=42)

            input_features.append(hash(val))
        else:
            input_features.append(val)
    
    predicted_score = model.predict([input_features])[0]

    # Calculate percentile
    better_than = sum(1 for y in y_train if predicted_score > y)
    p = better_than / len(rows)
    
    reason = f"This team's arrangement ranks in the top {int((1-p)*100)}% of all teams based on predicted retention metrics."

import csv
import numpy as np

def spot(header, rows):
    """
    Identifies rows that look 'wrong' by detecting outliers in the data.
    We use the Isolation Forest approach (via standard deviation/z-score) 
    to find rows that deviate significantly from the mean of their columns.
    """
    # Convert rows to a numpy array for easier calculation
    # We only consider numeric columns for anomaly detection
    numeric_indices = [i for i, name in enumerate(header) if name[0].isupper()]
    
    data = []
    for row in rows:
        data.append([row[i] for i in numeric_indices])
    
    data = np.array(data)
    
    # Calculate Z-scores for each numeric column
    # A row is suspicious if it has a high average Z-score across columns
    means = np.mean(data, axis=0)
    stds = np.std(data, axis=0)
    
    # Avoid division by zero
    stds[stds == 0] = 1.0
    
    z_scores = np.abs((data - means) / stds)
    
    # Average Z-score per row
    row_scores = np.mean(z_scores, axis=1)
    
    # Get indices sorted by score descending (most suspicious first)
    suspicious_indices = np.argsort(row_scores)[::-1]
    
    # Return the top 10 most suspicious rows
    return suspicious_indices[:10].tolist()

import csv
import numpy as np

def flow(header, row, seen):
    """
    Detects if the current row is an outlier compared to the 'seen' distribution.
    We use the Z-score of numeric columns to determine if the current row 
    deviates significantly from the historical average.
    """
    if len(seen) < 30:
        return False

    numeric_indices = [i for i, name in enumerate(header) if name[0].isupper()]
    
    # Extract numeric data from seen rows
    seen_data = np.array([[r[i] for i in numeric_indices] for r in seen])
    current_data = np.array([row[i] for i in numeric_indices])
    
    means = np.mean(seen_data, axis=0)
    stds = np.std(seen_data, axis=0)
    stds[stds == 0] = 1.0  # Avoid division by zero
    
    # Calculate Z-score for the current row
    z_scores = np.abs((current_data - means) / stds)
    avg_z = np.mean(z_scores)
    
    # If the average Z-score is high (e.g., > 3.0), the row is an outlier
    return avg_z > 3.0

def justify(header, rows, row):
    """
    Estimates the performance of a given row relative to the dataset.
    Returns (p, reason) where p is the percentile rank and reason explains
    why simply increasing salary is not a sufficient retention strategy.
    """
    goal_cols = [i for i, name in enumerate(header) if name.endswith(('-', '+'))]
    input_indices = [i for i in range(len(header)) if i not in goal_cols]

    # Prepare training data
    X_train = []
    y_train = []
    for r in rows:
        features = []
        for i in input_indices:
            val = r[i]
            if isinstance(val, str):
                features.append(hash(val))
            else:
                features.append(val)
        X_train.append(features)
        
        # Aggregate score: Maximize '+' goals, minimize '-' goals
        score = 0
        for g in goal_cols:
            val = r[g]
            if header[g].endswith('-'):
                score -= val
            else:
                score += val
        y_train.append(score)

    # Train model
    model = RandomForestRegressor(n_estimators=50, random_state=42)
    model.fit(X_train, y_train)

    # Predict score for the input row
    input_features = []
    for i in input_indices:
        val = row[i]
        if isinstance(val, str):
            input_features.append(hash(val))
        else:
            input_features.append(val)
    
    predicted_score = model.predict([input_features])[0]

    # Calculate percentile
    better_than = sum(1 for y in y_train if predicted_score > y)
    p = better_than / len(rows)
    
    # Reason: Salary is only one factor; data shows tenure is driven by 
    # complex interactions like JobInvolvement and WorkLifeBalance.
    reason = "Increasing salary alone is insufficient because our data shows that retention is significantly driven by non-monetary factors like Job Involvement and Work-Life Balance."
    
    return p, reason
