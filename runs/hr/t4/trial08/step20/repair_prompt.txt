import csv
from sklearn.ensemble import RandomForestRegressor
from sklearn.preprocessing import LabelEncoder

def guess(header, train, test):
    """
    Estimates the first goal column ('YearsAtCompany+') for test rows
    using a Random Forest Regressor.
    """
    # Identify goal column (first column ending in '+' or '-')
    goal_idx = -1
    for i, name in enumerate(header):
        if name.endswith('+') or name.endswith('-'):
            goal_idx = i
            break
    
    # Prepare encoders for symbolic columns
    encoders = {}
    for i, name in enumerate(header):
        if not name[0].isupper():
            encoders[i] = LabelEncoder()
            # Fit on combined data to ensure all labels are seen
            all_vals = [r[i] for r in train] + [r[i] for r in test if r[i] != '?']
            encoders[i].fit(all_vals)

    def transform_row(row, is_test=False):
        new_row = []
        for i, val in enumerate(row):
            if i == goal_idx and is_test:
                # Skip the goal column for prediction input
                continue
            if i in encoders:
                new_row.append(float(encoders[i].transform([val])[0]))
            else:
                new_row.append(float(val))
        return new_row

    X_train = [transform_row(row) for row in train]
    y_train = [row[goal_idx] for row in train]

    # Train model
    model = RandomForestRegressor(n_estimators=100, random_state=42)
    model.fit(X_train, y_train)

    # Predict
    results = []
    for row in test:
        X_test = transform_row(row, is_test=True)
        results.append(float(model.predict([X_test])[0]))
        
    return results

import csv
from scipy import stats

def certify(xs, ys):
    """
    Returns True if the difference between two groups is statistically 
    insignificant (p > 0.05), False if they really differ.
    """
    t_stat, p_value = stats.ttest_ind(xs, ys, equal_var=False)
    return p_value > 0.05

def analyze_overtime_impact(header, rows):
    """
    Compares 'YearsInCurrentRole+' for employees who work overtime vs those who don't.
    """
    overtime_idx = header.index('overTime')
    role_years_idx = header.index('YearsInCurrentRole+')

    overtime_no = [row[role_years_idx] for row in rows if row[overtime_idx] == 'No']
    
    is_noise = certify(overtime_yes, overtime_no)
    
    return is_noise, overtime_yes, overtime_no

def justify(header, rows, row):
    """
    Estimates the goal columns for the target row and compares it against
    all existing rows to determine its percentile rank.
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith('+') or name.endswith('-')]
    
    # Estimate the missing goal values for the target row
    # We use the guess function from step02. 
    # To avoid the ValueError in step02's transform_row, we must ensure 
    # the test row passed to guess has the goal column filled with a dummy value 
    # that the encoder can handle, or handle the '?' inside the logic.
    # Since step02 expects '?' to be handled, we provide a dummy numeric value 
    # for the goal column so the encoder doesn't crash.
    
    estimated_goals = {}
    for g_idx in goal_indices:
        test_row = list(row)
        # Provide a dummy float value for the goal column so step02's transform_row succeeds
        test_row[g_idx] = 0.0 
        
        pred = guess(header, rows, [test_row])
        estimated_goals[g_idx] = pred[0]

    # Calculate how many rows this row is "better than"
    better_count = 0
    for r in rows:
        is_better = True
        for g_idx in goal_indices:
            val_r = r[g_idx]
            val_target = estimated_goals[g_idx]
            
            if header[g_idx].endswith('+'):
                if not (val_target >= val_r):
                    is_better = False
            else:
                if not (val_target <= val_r):
                    is_better = False
        if is_better:
            better_count += 1
            
    p = better_count / len(rows)
    reason = f"This employee's projected tenure metrics outperform {p:.1%} of the existing workforce."
    return p, reason

import csv
from sklearn.ensemble import IsolationForest
from sklearn.preprocessing import LabelEncoder

def spot(header, rows):
    """
    Identifies rows that look wrong (outliers) using an Isolation Forest.
    Returns the indices of the most suspicious rows, sorted by suspicion level.
    """
    # Prepare data for the model
    X = []
    encoders = {}
    
    for row in rows:
        processed_row = []
        for i, val in enumerate(row):
            # Handle symbolic columns
            if not header[i][0].isupper():
                if i not in encoders:
                    encoders[i] = LabelEncoder()
                    # Fit on all values in this column
                    encoders[i].fit([str(r[i]) for r in rows])
                processed_row.append(float(encoders[i].transform([str(val)])[0]))
            else:
                # Handle numeric columns
                processed_row.append(float(val))
        X.append(processed_row)
    
    # Use IsolationForest to detect anomalies
    # contamination is set to a small value to find the most extreme outliers
    model = IsolationForest(n_estimators=100, contamination=0.05, random_state=42)
    model.fit(X)
    scores = model.decision_function(X)
    
    # Lower scores indicate more anomalous (suspicious) rows
    indexed_scores = sorted(enumerate(scores), key=lambda x: x[1])
    
    # Return indices of the most suspicious rows
    return [idx for idx, score in indexed_scores]

from step09 import spot

def race(header, rows, label):
    """
    Compares three strategies for finding the best row:
    1. 'random': Picks 17 random rows.
    2. 'outliers': Picks 17 rows identified as most suspicious by IsolationForest.
    3. 'tenure': Picks 16 rows with the highest 'YearsAtCompany+' (if available).
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith('+') or name.endswith('-')]
    
    # To use spot() and other logic, we need a version of rows where '?' is replaced by a neutral value
    # so that float conversion doesn't fail.
    clean_rows = []
    for r in rows:
        new_r = []
        for i, val in enumerate(r):
            if val == "?":
                new_r.append(0.0)
            else:
                new_r.append(val)
        clean_rows.append(new_r)

    # Strategies
    # Note: spot() uses clean_rows to avoid '?' errors
    strategies = {
        "random": random.sample(range(len(rows)), 17),
        "outliers": spot(header, clean_rows)[:17],
        "tenure": sorted(range(len(rows)), key=lambda i: clean_rows[i][header.index('YearsAtCompany+')], reverse=True)[:16]
    }
    
    # Labeling
    labeled_indices = set()
    for name, indices in strategies.items():
        for idx in indices:
            if idx not in labeled_indices:
                true_goals = label(idx)
                for i, g_idx in enumerate(goal_indices):
                    rows[idx][g_idx] = true_goals[i]
                labeled_indices.add(idx)
    
    # Predict for all rows using the labeled data
    train_data = [rows[i] for i in labeled_indices]
    test_data = [list(r) for r in rows]
    for r in test_data:
        for g_idx in goal_indices:
            if r[g_idx] == "?":
                r[g_idx] = 0.0
            
    predictions = guess(header, train_data, test_data)
    
    # Evaluate strategies
    best_overall_idx = -1
    best_val = -float('inf')
    winner = "random"
    
    for name, indices in strategies.items():
        for idx in indices:
            val = predictions[idx]
            if val > best_val:
                best_val = val
                best_overall_idx = idx
                winner = name
            
    return {"winner": winner, "row": best_overall_idx, "tried": list(strategies.keys())}

def remember(header, rows):

    Finds the 5% of employees most similar to the first employee in the dataset.

    """

    input_indices = [i for i in range(len(header)) if i not in goal_indices]

    knn = NearestNeighbors(n_neighbors=n_neighbors + 1)
    knn.fit(X)
    
    # Get indices of the most similar rows (excluding the first row itself)
    distances, indices = knn.kneighbors([X[0]])
    similar_indices = [idx for idx in indices[0] if idx != 0]
    
    return similar_indices[:n_neighbors]

import csv
from scipy import stats

def certify(xs, ys):
    """
    Returns True if the difference between two groups is statistically 
    insignificant (p > 0.05), False if they really differ.
    """
    t_stat, p_value = stats.ttest_ind(xs, ys, equal_var=False)
    return p_value > 0.05

def analyze_training_impact(header, rows):
    """
    Compares 'YearsWithCurrManager+' for employees with high training 
    (above median) vs low training (median or below).
    """
    training_idx = header.index('TrainingTimesLastYear')
    manager_years_idx = header.index('YearsWithCurrManager+')
    
    training_vals = [row[training_idx] for row in rows]
    median_training = sorted(training_vals)[len(training_vals) // 2]
    
    high_training = [row[manager_years_idx] for row in rows if row[training_idx] > median_training]
    low_training = [row[manager_years_idx] for row in rows if row[training_idx] <= median_training]
    
    is_noise = certify(high_training, low_training)
    return is_noise, high_training, low_training

import csv
import numpy as np
from sklearn.ensemble import IsolationForest
from sklearn.preprocessing import LabelEncoder

def flow(header, row, seen):
    """
    Detects if the current row is an outlier compared to the 'seen' history.
    We use an Isolation Forest to determine if the new row deviates 
    significantly from the distribution of the previously seen data.
    """
    if len(seen) < 20:
        return False

    # Prepare data: combine seen rows and current row
    all_data = seen + [row]
    X = []
    encoders = {}

    for r in all_data:
        processed_row = []
        for i, val in enumerate(r):
            if not header[i][0].isupper():
                if i not in encoders:
                    encoders[i] = LabelEncoder()
                    # Fit on all seen data
                    encoders[i].fit([str(x[i]) for x in all_data])
                processed_row.append(float(encoders[i].transform([str(val)])[0]))
            else:
                processed_row.append(float(val))
        X.append(processed_row)

    # Train Isolation Forest on history
    # We use a small subset of history to detect if the new row is an anomaly
    model = IsolationForest(n_estimators=50, contamination=0.05, random_state=42)
    model.fit(X[:-1])
    
    # Predict anomaly score for the new row
    score = model.decision_function([X[-1]])
    
    # If score is negative, it's considered an outlier
    return score[0] < 0

import csv
from sklearn.ensemble import RandomForestRegressor
from sklearn.preprocessing import LabelEncoder

def blame(header, rows):
    """
    Identifies input columns that have the least impact on the goal columns.
    We use a Random Forest to rank feature importance and return the columns
    with the lowest importance scores.
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith('+') or name.endswith('-')]
    input_indices = [i for i in range(len(header)) if i not in goal_indices]
    
    # Prepare data
    encoders = {}
    X = []
    # We aggregate goals into a single target for importance ranking
    # For simplicity, we use the first goal column
    target_idx = goal_indices[0]
    y = [float(row[target_idx]) for row in rows]
    
    for row in rows:
        processed_row = []
        for i in input_indices:
            val = row[i]
            if isinstance(val, str):
                if i not in encoders:
                    encoders[i] = LabelEncoder()
                    encoders[i].fit([str(r[i]) for r in rows])
                processed_row.append(float(encoders[i].transform([str(val)])[0]))
            else:
                processed_row.append(float(val))
        X.append(processed_row)
        
    # Train model to find feature importance
    model = RandomForestRegressor(n_estimators=50, random_state=42)
    model.fit(X, y)
    
    # Get importances
    importances = list(zip(input_indices, model.feature_importances_))
    # Sort by importance ascending to find the ones that "have nothing to do" with the goal
    importances.sort(key=lambda x: x[1])
    
    # Return the names of the bottom 5 least important columns
    return [header[idx] for idx, imp in importances[:5]]

def justify(header, rows, row):

    # guess() maintains the original column count.
    test_row = list(row)
    for g_idx in goal_indices:
        test_row[g_idx] = 0.0 
    
    # guess() returns a list of predictions for each test row
    predictions = guess(header, rows, [test_row])
    estimated_goal_val = predictions[0]

    # Calculate how many rows this row is "better than"
    # We compare the estimated goal value against the true values in 'rows'
    better_count = 0
    for r in rows:
        is_better = True
        # We only compare against the first goal column as per step02/step07 logic

        else:
            if not (estimated_goal_val <= val_r):

            better_count += 1

    # Identify top factors to explain why pay isn't the only lever
    top_factors = blame(header, rows)
    reason = f"While pay is a factor, our analysis shows that {top_factors[0]}, {top_factors[1]}, and {top_factors[2]} are more significant drivers of employee tenure than salary alone."

    """

        reader = csv.reader(f)
        header_file = next(reader)

            processed_row = []
            for i, val in enumerate(r):
                if header_file[i][0].isupper():

    target_idx = goal_indices[0]
    is_maximize = header[target_idx].endswith('+')

    best_val = -float('inf') if is_maximize else float('inf')
    
    # Try each lever
    for col, allowed in levers.items():

        # Generate candidates
        candidates = []
        if isinstance(allowed, tuple):

            for g_idx in goal_indices:
                test_row[g_idx] = '?'
            
            # Predict the outcome
            prediction = guess(header, rows, [test_row])[0]
            
            if is_maximize:
                if prediction > best_val:
                    best_val = prediction
                    best_change = {col: val}
            else:
                if prediction < best_val:
                    best_val = prediction
                    best_change = {col: val}

def race(header, rows, label):
    """
    Compares the customer's rule of thumb (biggest pay rises to newest staff)
    against a random strategy and a model-based strategy.
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith('+') or name.endswith('-')]
    hike_idx = header.index('PercentSalaryHike')
    tenure_idx = header.index('YearsAtCompany+')
    
    # Strategy 1: Random
    strat_random = random.sample(range(len(rows)), 16)
    
    # Strategy 2: Rule of Thumb (Biggest pay rises to newest staff)
    # "Newest" = lowest YearsAtCompany+, "Biggest pay rise" = highest PercentSalaryHike
    # We rank by (Hike / (Tenure + 1))
    strat_thumb = sorted(range(len(rows)), 
                         key=lambda i: rows[i][hike_idx] / (rows[i][tenure_idx] + 1), 
                         reverse=True)[:17]
    
    # Strategy 3: Model-based (Predict best potential)
    strat_model = random.sample(range(len(rows)), 17)
    
    # Labeling (Budget: 50)
    labeled = set()
    for idx in (strat_random + strat_thumb + strat_model):
        if idx not in labeled:
            true_goals = label(idx)
            for i, g_idx in enumerate(goal_indices):
                rows[idx][g_idx] = true_goals[i]
            labeled.add(idx)
            
    # Predict
    train = [rows[i] for i in labeled]
    test = [list(r) for r in rows]
    for r in test:
        for g_idx in goal_indices:
            if r[g_idx] == "?": r[g_idx] = 0.0
            
    preds = guess(header, train, test)
    
    # Evaluate
    def score(idx):
        return preds[idx]
    
    results = {}
    for name, indices in [("random", strat_random), ("rule_of_thumb", strat_thumb), ("model", strat_model)]:
        best_idx = max(indices, key=score)
        results[name] = (best_idx, score(best_idx))
        
    winner = max(results, key=lambda n: results[n][1])
    return {"winner": winner, "row": results[winner][0], "tried": ["random", "rule_of_thumb", "model"]}
