```python
import csv
import pandas as pd
from sklearn.ensemble import RandomForestRegressor
from sklearn.preprocessing import LabelEncoder

def guess(header, train, test):
    """
    Estimates the first goal column ('YearsAtCompany+') for test rows
    using a Random Forest Regressor, ensuring all symbolic data is encoded.
    """
    df_train = pd.DataFrame(train, columns=header)
    df_test = pd.DataFrame(test, columns=header)
    
    # Identify the first goal column (ending in + or -)
    goal_col = [c for c in header if c.endswith('+') or c.endswith('-')][0]
    
    # Prepare features: drop goal columns
    features = [c for c in header if not (c.endswith('+') or c.endswith('-'))]
    
    # Encode symbolic columns
    df_combined = pd.concat([df_train[features], df_test[features]], axis=0).copy()
    for col in df_combined.columns:
        if df_combined[col].dtype == 'object':
            le = LabelEncoder()
            # Fit on combined data to ensure all categories are seen
            df_combined[col] = le.fit_transform(df_combined[col].astype(str))
            
    X_train = df_combined.iloc[:len(train)]
    X_test = df_combined.iloc[len(train):]
    y_train = df_train[goal_col]
    
    # Train model
    model = RandomForestRegressor(n_estimators=100, random_state=42)
    model.fit(X_train, y_train)
    
    return model.predict(X_test).tolist()

if __name__ == "__main__":
    filename = 'WA_Fn-UseC_-HR-Employee-Attrition.n1000.train.csv'
    rows = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = []
            for i, val in enumerate(row):
                if header[i][0].isupper():
                    processed_row.append(float(val))
                else:
                    processed_row.append(val)
            rows.append(processed_row)
    
    # Split data: use first 700 for training, next 100 for testing
    train_data = rows[:700]
    test_data = [row[:] for row in rows[700:800]]
    
    # Mask the goal column in test data
    goal_idx = [i for i, h in enumerate(header) if h.endswith('+') or h.endswith('-')][0]
    for row in test_data:
        row[goal_idx] = "?"
        
    predictions = guess(header, train_data, test_data)
    
    print("Estimation for 'YearsAtCompany+' (First 5 test rows):")
    for i in range(min(5, len(predictions))):
        print(f"Row {i+700}: Predicted {predictions[i]:.2f} years")
```
