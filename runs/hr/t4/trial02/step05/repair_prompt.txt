import csv

def remember(header, rows):
    """
    Identifies staff who stay 'unusually long'.
    'YearsAtCompany+' is a goal to maximize.
    We calculate the 95th percentile of 'YearsAtCompany+' and return
    the indices of rows that meet or exceed this threshold.
    """
    # Identify index for 'YearsAtCompany+'
    try:
        idx = header.index('YearsAtCompany+')
    except ValueError:
        return [0]

    # Extract all values for this column
    years = [row[idx] for row in rows]
    
    # Calculate threshold for "unusually long" (top 5%)
    sorted_years = sorted(years, reverse=True)
    limit = max(1, int(len(rows) * 0.05))
    threshold = sorted_years[limit - 1]
    
    # Find indices of rows meeting the threshold
    best_indices = [i for i, val in enumerate(years) if val >= threshold]
    
    # Ensure we return at most 5% of rows (handle ties at the threshold)
    return best_indices[:limit]

import csv
from scipy import stats

def certify(xs, ys):
    """
    Performs a Welch's t-test to determine if the means of two groups
    are statistically different. Returns True if they are likely the same
    (p-value > 0.05), False if they are significantly different.
    """
    t_stat, p_value = stats.ttest_ind(xs, ys, equal_var=False)
    return p_value > 0.05

    """
    idx = header.index('YearsAtCompany+')
    years_val = row[idx]
    
    # If this is the first row, we can't compare, so let it go
    if not seen:
        return False

    sorted_history = sorted(history, reverse=True)

def blame(header, rows):
    """
    Identifies input columns that best explain why employees have short stays.
    'YearsAtCompany+' is the goal to maximize. We define 'worst' as the bottom 25%
    of tenure and use a Random Forest feature importance to explain the variance.
    """
    df = pd.DataFrame(rows, columns=header)
    goal = 'YearsAtCompany+'
    
    # Identify input columns (exclude goals)
    inputs = [c for c in header if not (c.endswith('+') or c.endswith('-'))]
    
    # Encode symbolic data
    df_encoded = df.copy()
    for col in inputs:
        if df_encoded[col].dtype == 'object':
            le = LabelEncoder()
            df_encoded[col] = le.fit_transform(df_encoded[col].astype(str))
            
    # Focus on the worst 25% of rows (short stays)
    threshold = df[goal].quantile(0.25)
    worst_rows = df_encoded[df[goal] <= threshold]
    
    X = worst_rows[inputs]
    y = worst_rows[goal]
    
    # Use Random Forest to find features that correlate with the tenure of short-stay employees
    model = RandomForestRegressor(n_estimators=50, random_state=42)
    model.fit(X, y)
    
    # Get top 3 features
    importances = sorted(zip(inputs, model.feature_importances_), key=lambda x: x[1], reverse=True)
    return [name for name, score in importances[:3]]
