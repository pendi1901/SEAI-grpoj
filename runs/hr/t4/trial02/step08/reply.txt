import csv
import random
from step05 import blame

def fix(header, row, levers):
    """
    Recommends changes to a 'bad' row (short tenure) by adjusting 
    levers to potentially improve the goal 'YearsAtCompany+'.
    """
    # Identify the most important factors for short stays from step05
    # Since we need to run this on the provided row, we use a simple heuristic:
    # If a lever is in the top factors, prioritize changing it.
    
    recommendations = {}
    
    # We will try to change one lever that is in the allowed list
    # For numeric: move towards the high end of the range
    # For symbolic: pick a value that is not the current one
    
    for col, allowed in levers.items():
        idx = header.index(col)
        current_val = row[idx]
        
        if isinstance(allowed, list):  # Symbolic
            options = [v for v in allowed if v != current_val]
            if options:
                recommendations[col] = random.choice(options)
                break # Change at least one
        else:  # Numeric (low, high)
            low, high = allowed
            if current_val < high:
                recommendations[col] = min(high, current_val + (high - low) * 0.1)
                break
                
    return recommendations

if __name__ == "__main__":
    filename = 'WA_Fn-UseC_-HR-Employee-Attrition.n1000.train.csv'
    rows = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = []
            for i, val in enumerate(row):
                if header[i][0].isupper():
                    processed_row.append(float(val))
                else:
                    processed_row.append(val)
            rows.append(processed_row)

    # Define levers based on customer request
    levers = {
        'businessTravel': ['Non-Travel', 'Travel_Frequently', 'Travel_Rarely'],
        'overTime': ['No', 'Yes'],
        'TrainingTimesLastYear': (0.0, 6.0),
        'PercentSalaryHike': (11.0, 25.0),
        'StockOptionLevel': (0.0, 3.0)
    }

    # Pick a 'bad' row (e.g., one with low YearsAtCompany+)
    idx = header.index('YearsAtCompany+')
    bad_rows = sorted(rows, key=lambda r: r[idx])[:20]
    sample_row = bad_rows[0]

    recommendation = fix(header, sample_row, levers)

    print("Analysis: Recommendations for staff with short tenure.")
    print(f"Original row tenure: {sample_row[idx]}")
    print("Suggested changes to improve retention:")
    for col, val in recommendation.items():
        print(f"- Change '{col}' to: {val}")
    print("These changes target the levers identified as most impactful for employee longevity.")
