```python
import csv
import pandas as pd
from sklearn.ensemble import RandomForestRegressor
from sklearn.preprocessing import LabelEncoder

def guess(header, train, test):
    """
    Estimates the first goal column for test rows using a Random Forest Regressor.
    Ensures all symbolic data is encoded consistently across train and test.
    """
    df_train = pd.DataFrame(train, columns=header)
    df_test = pd.DataFrame(test, columns=header)
    
    # Identify the first goal column
    goal_col = [c for c in header if c.endswith('+') or c.endswith('-')][0]
    
    # Features are all columns except the goal
    features = [c for c in header if not (c.endswith('+') or c.endswith('-'))]
    
    # Combine to ensure consistent LabelEncoding
    df_combined = pd.concat([df_train[features], df_test[features]], axis=0).copy()
    
    for col in features:
        if df_combined[col].dtype == 'object':
            le = LabelEncoder()
            df_combined[col] = le.fit_transform(df_combined[col].astype(str))
            
    X_train = df_combined.iloc[:len(train)]
    X_test = df_combined.iloc[len(train):]
    y_train = df_train[goal_col]
    
    model = RandomForestRegressor(n_estimators=100, random_state=42)
    model.fit(X_train, y_train)
    
    return model.predict(X_test).tolist()

if __name__ == "__main__":
    filename = 'WA_Fn-UseC_-HR-Employee-Attrition.n1000.train.csv'
    rows = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = []
            for i, val in enumerate(row):
                if header[i][0].isupper():
                    processed_row.append(float(val))
                else:
                    processed_row.append(val)
            rows.append(processed_row)

    dept_idx = header.index('department')
    ot_idx = header.index('overTime')
    goal_idx = [i for i, h in enumerate(header) if h.endswith('+') or h.endswith('-')][0]

    # Filter for Sales department
    sales_rows = [r for r in rows if r[dept_idx] == 'Sales']
    
    # Create two scenarios for Sales
    test_rows_current = [r[:] for r in sales_rows]
    test_rows_no_ot = [r[:] for r in sales_rows]
    for r in test_rows_no_ot:
        r[ot_idx] = 'No'
        
    # Mask goals
    for r in test_rows_current: r[goal_idx] = "?"
    for r in test_rows_no_ot: r[goal_idx] = "?"
    
    # Estimate
    preds_current = guess(header, rows, test_rows_current)
    preds_no_ot = guess(header, rows, test_rows_no_ot)
    
    avg_current = sum(preds_current) / len(preds_current)
    avg_no_ot = sum(preds_no_ot) / len(preds_no_ot)
    
    print("Analysis: Impact of cutting overtime in Sales on retention (YearsAtCompany+).")
    print(f"Average tenure (Current): {avg_current:.2f} years")
    print(f"Average tenure (If Overtime cut): {avg_no_ot:.2f} years")
    
    if avg_no_ot > avg_current:
        print("Result: Cutting overtime is estimated to increase average tenure in Sales.")
    else:
        print("Result: Cutting overtime is estimated to decrease or not improve average tenure in Sales.")
```
