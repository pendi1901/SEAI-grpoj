import csv
import random
from step07 import choose as base_choose

def choose(header, rows, label):
    """
    Finds the best row for 'Laboratory Technician' staff by sampling 50 rows,
    labeling them to get their true goal values, and returning the index of the best one.
    """
    # Identify jobRole index and goal column index
    role_idx = header.index('jobRole')
    goal_col = [c for c in header if c.endswith('+') or c.endswith('-')][0]
    is_maximize = goal_col.endswith('+')

    # Filter indices for Laboratory Technician
    lab_indices = [i for i, row in enumerate(rows) if row[role_idx] == 'Laboratory Technician']
    
    # If fewer than 50, sample all; otherwise sample 50
    sample_size = min(len(lab_indices), 50)
    indices = random.sample(lab_indices, sample_size)
    
    best_val = float('-inf') if is_maximize else float('inf')
    best_idx = -1
    
    for i in indices:
        # label(i) returns a list of true goal values for row i
        true_values = label(i)
        val = true_values[0]
        
        if is_maximize:
            if val > best_val:
                best_val = val
                best_idx = i
        else:
            if val < best_val:
                best_val = val
                best_idx = i
                
    return best_idx

if __name__ == "__main__":
    filename = 'WA_Fn-UseC_-HR-Employee-Attrition.n1000.train.csv'
    rows = []
    true_goals = []
    
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        goal_idx = [i for i, h in enumerate(header) if h.endswith('+') or h.endswith('-')][0]
        
        for row in reader:
            processed_row = []
            for i, val in enumerate(row):
                if header[i][0].isupper():
                    processed_row.append(float(val))
                else:
                    processed_row.append(val)
            
            # Store true goal and mask it in the working row
            true_goals.append([processed_row[goal_idx]])
            processed_row[goal_idx] = "?"
            rows.append(processed_row)

    def label(i):
        return true_goals[i]

    best_index = choose(header, rows, label)
    
    print(f"Analysis complete for Laboratory Technician staff.")
    print(f"The best working arrangement found is at row index: {best_index}")
    print(f"Goal value for this row: {true_goals[best_index][0]}")
    print("This row was identified as the top performer among the sampled laboratory staff.")
