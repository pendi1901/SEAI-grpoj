```python
import csv
import pandas as pd
from sklearn.ensemble import RandomForestRegressor
from sklearn.preprocessing import LabelEncoder

def blame(header, rows):
    """
    Identifies input columns that best explain short stays (low 'YearsAtCompany+').
    Uses a Random Forest to identify feature importance for the goal column.
    """
    df = pd.DataFrame(rows, columns=header)
    
    # Identify goal column
    goal_col = [c for c in header if c.endswith('+') or c.endswith('-')][0]
    
    # Identify input columns (exclude goals)
    inputs = [c for c in header if not (c.endswith('+') or c.endswith('-'))]
    
    # Encode symbolic columns
    df_encoded = df.copy()
    for col in header:
        if not col[0].isupper():
            le = LabelEncoder()
            df_encoded[col] = le.fit_transform(df[col].astype(str))
            
    X = df_encoded[inputs]
    y = df_encoded[goal_col]
    
    # Train a regressor to find feature importance
    model = RandomForestRegressor(n_estimators=100, random_state=42)
    model.fit(X, y)
    
    # Get feature importances
    importances = pd.Series(model.feature_importances_, index=inputs)
    
    # Return the top 3 most influential input columns
    return importances.nlargest(3).index.tolist()

if __name__ == "__main__":
    filename = 'WA_Fn-UseC_-HR-Employee-Attrition.n1000.train.csv'
    rows = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = []
            for i, val in enumerate(row):
                if header[i][0].isupper():
                    processed_row.append(float(val))
                else:
                    processed_row.append(val)
            rows.append(processed_row)
            
    top_factors = blame(header, rows)
    
    print("Analysis: What factors explain short stays at the company?")
    print("The most significant input factors influencing tenure are:")
    for factor in top_factors:
        print(f"- {factor}")
    print("\nThese columns show the strongest correlation with how long employees remain.")
```
