```python
import csv
from scipy import stats

def certify(xs, ys):
    """
    Returns True if the difference between two groups is statistically 
    insignificant (p > 0.05), False if they are significantly different.
    """
    t_stat, p_value = stats.ttest_ind(xs, ys, equal_var=False)
    return p_value > 0.05

def analyze_attrition_impact(header, rows):
    """
    Compares 'YearsAtCompany+' for employees who work overtime vs those who don't.
    """
    overtime_idx = header.index('overTime')
    years_idx = header.index('YearsAtCompany+')
    
    yes_overtime = [row[years_idx] for row in rows if row[overtime_idx] == 'Yes']
    no_overtime = [row[years_idx] for row in rows if row[overtime_idx] == 'No']
    
    is_noise = certify(yes_overtime, no_overtime)
    
    avg_yes = sum(yes_overtime) / len(yes_overtime)
    avg_no = sum(no_overtime) / len(no_overtime)
    
    return is_noise, avg_yes, avg_no

if __name__ == "__main__":
    filename = 'WA_Fn-UseC_-HR-Employee-Attrition.n1000.train.csv'
    rows = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = []
            for i, val in enumerate(row):
                if header[i][0].isupper():
                    processed_row.append(float(val))
                else:
                    processed_row.append(val)
            rows.append(processed_row)
            
    is_noise, avg_yes, avg_no = analyze_attrition_impact(header, rows)
    
    print("Analysis: Do people who work overtime leave sooner?")
    print(f"Average Years at Company (Overtime): {avg_yes:.2f}")
    print(f"Average Years at Company (No Overtime): {avg_no:.2f}")
    
    if is_noise:
        print("Conclusion: The observed difference is statistically insignificant (noise).")
    else:
        print("Conclusion: There is a statistically significant difference in tenure.")
```
