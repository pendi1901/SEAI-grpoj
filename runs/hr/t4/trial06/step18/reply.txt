import csv
import random
import step02
import step05

def fix(header, row, levers):
    """
    Recommends changes to a 'bad' row to improve its goal metrics.
    We use a simple hill-climbing approach: try changing each lever
    to its best possible value within the allowed range.
    """
    goal_col = [c for c in header if c.endswith('+') or c.endswith('-')][0]
    is_maximize = goal_col.endswith('+')
    
    # We need a model to evaluate the impact of changes.
    # Since we don't have the full dataset passed in, we re-read it.
    filename = 'WA_Fn-UseC_-HR-Employee-Attrition.n1000.train.csv'
    rows = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header_file = next(reader)
        for r in reader:
            processed_row = [float(val) if header_file[i][0].isupper() else val for i, val in enumerate(r)]
            rows.append(processed_row)

    best_change = {}
    best_val = -float('inf') if is_maximize else float('inf')
    
    # Try changing each lever to a random valid value and see if it improves the prediction
    for col, allowed in levers.items():
        idx = header.index(col)
        
        # Try a few random valid values for this lever
        for _ in range(5):
            if isinstance(allowed, list):  # Symbolic
                test_val = random.choice(allowed)
            else:  # Numeric (low, high)
                test_val = random.uniform(allowed[0], allowed[1])
            
            # Create a test row with the change
            test_row = list(row)
            test_row[idx] = test_val
            
            # Predict the goal for this test row
            # We use the full dataset to train the model
            prediction = step02.guess(header, rows, [test_row])[0]
            
            if (is_maximize and prediction > best_val) or (not is_maximize and prediction < best_val):
                best_val = prediction
                best_change = {col: test_val}
                
    return best_change
