import csv
import random
import step02
import step05

def race(header, rows, label):
    """
    Compares three strategies for finding the best row using a shared 50-label budget.
    Strategies:
    1. random: Picks rows at random.
    2. importance: Picks rows based on the most important features (from step05).
    3. model_based: Uses a regressor to predict the best row (from step02).
    """
    goal_col = [c for c in header if c.endswith('+') or c.endswith('-')][0]
    goal_idx = header.index(goal_col)
    is_maximize = goal_col.endswith('+')
    
    budget = 50
    labeled = {}
    
    def get_label(i):
        if i not in labeled:
            if len(labeled) >= budget:
                raise ValueError("Budget exceeded")
            labeled[i] = label(i)[0]
        return labeled[i]

    # Strategy 1: Random
    def strategy_random():
        indices = random.sample(range(len(rows)), 16)
        for i in indices: get_label(i)
        return max(indices, key=lambda i: labeled[i]) if is_maximize else min(indices, key=lambda i: labeled[i])

    # Strategy 2: Importance (Pick rows with high/low values in top features)
    def strategy_importance():
        top_factors = step05.blame(header, [r for r in rows if '?' not in r]) # Simplified
        idx = header.index(top_factors[0])
        # Pick 17 rows with extreme values in the most important feature
        sorted_rows = sorted(range(len(rows)), key=lambda i: rows[i][idx])
        indices = sorted_rows[:8] + sorted_rows[-9:]
        for i in indices: get_label(i)
        return max(indices, key=lambda i: labeled[i]) if is_maximize else min(indices, key=lambda i: labeled[i])

    # Strategy 3: Model-based (Train on 17, predict on rest)
    def strategy_model():
        train_idx = random.sample(range(len(rows)), 17)
        for i in train_idx: get_label(i)
        train_data = [rows[i] for i in train_idx]
        test_data = [rows[i] for i in range(len(rows)) if i not in train_idx]
        preds = step02.guess(header, train_data, test_data)
        best_pred_idx = preds.index(max(preds)) if is_maximize else preds.index(min(preds))
        return [i for i in range(len(rows)) if i not in train_idx][best_pred_idx]

    results = {
        "random": strategy_random(),
        "importance": strategy_importance(),
        "model_based": strategy_model()
    }
    
    # Determine winner
    if is_maximize:
        winner = max(results, key=lambda s: labeled[results[s]])
    else:
        winner = min(results, key=lambda s: labeled[results[s]])
        
    return {"winner": winner, "row": results[winner], "tried": list(results.keys())}

if __name__ == "__main__":
    filename = 'WA_Fn-UseC_-HR-Employee-Attrition.n1000.train.csv'
    rows = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = [float(val) if header[i][0].isupper() else val for i, val in enumerate(row)]
            rows.append(processed_row)

    goal_idx = [i for i, h in enumerate(header) if h.endswith('+') or h.endswith('-')][0]
    
    def label(i):
        return [rows[i][goal_idx]]

    # Mask rows
    masked_rows = [list(r) for r in rows]
    for r in masked_rows: r[goal_idx] = "?"

    result = race(header, masked_rows, label)
    print(f"Strategy Race Results:")
    print(f"Winner: {result['winner']}")
    print(f"Best row found at index: {result['row']}")
    print(f"Strategies compared: {', '.join(result['tried'])}")
