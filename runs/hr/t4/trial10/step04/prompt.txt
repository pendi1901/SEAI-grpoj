import csv

def remember(header, rows):
    """
    Identifies staff who stay 'unusually long'.
    'YearsAtCompany+' is a goal to maximize.
    We calculate the 95th percentile of 'YearsAtCompany+' and return
    the indices of rows that meet or exceed this threshold.
    """
    # Identify index for 'YearsAtCompany+'
    try:
        idx = header.index('YearsAtCompany+')
    except ValueError:
        return [0]

    # Extract all values for this column
    years = [row[idx] for row in rows]
    
    # Calculate threshold for "unusually long" (top 5%)
    sorted_years = sorted(years, reverse=True)
    limit = max(1, int(len(rows) * 0.05))
    threshold = sorted_years[limit - 1]
    
    # Find indices of rows meeting the threshold
    best_indices = [i for i, val in enumerate(years) if val >= threshold]
    
    # Ensure we return at most 5% of rows
    return best_indices[:limit]

import csv
from sklearn.ensemble import RandomForestRegressor
import numpy as np

def guess(header, train, test):
    """
    Estimates the first goal column ('YearsAtCompany+') for test rows
    using a Random Forest Regressor based on input columns.
    """
    # Identify goal columns (ending in + or -)
    goal_indices = [i for i, h in enumerate(header) if h.endswith('+') or h.endswith('-')]
    target_idx = goal_indices[0]
    
    # Identify input columns (those not ending in + or -)
    input_indices = [i for i in range(len(header)) if i not in goal_indices]
    
    def encode(data):
        """Simple encoding: numeric stays, symbolic becomes categorical index."""
        encoded = []
        for row in data:
            new_row = []
            for i, val in enumerate(row):
                if isinstance(val, str):
                    # Convert string to unique integer
                    new_row.append(hash(val) % 1000)
                else:
                    new_row.append(val)
            encoded.append(new_row)
        return np.array(encoded)

    # Prepare training data
    X_train = encode([ [row[i] for i in input_indices] for row in train ])
    y_train = [row[target_idx] for row in train]
    
    # Prepare test data
    X_test = encode([ [row[i] for i in input_indices] for row in test ])
    
    # Train model
    model = RandomForestRegressor(n_estimators=50, random_state=42)
    model.fit(X_train, y_train)
    
    return model.predict(X_test).tolist()

import csv
from scipy import stats

def certify(xs, ys):
    """
    Returns True if the difference between two groups is statistically 
    insignificant (p-value > 0.05), False if they really differ.
    """
    t_stat, p_value = stats.ttest_ind(xs, ys, equal_var=False)
    return p_value > 0.05
