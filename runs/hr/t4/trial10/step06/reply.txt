import csv
import numpy as np
from step02 import guess

def justify(header, rows, row):
    """
    Estimates the goal for the given row and calculates its percentile rank
    against the existing rows.
    """
    # Identify goal columns
    goal_indices = [i for i, h in enumerate(header) if h.endswith('+') or h.endswith('-')]
    
    # Estimate the goal for the target row
    # We use the guess function from step02 which handles the prediction
    prediction = guess(header, rows, [row])[0]
    
    # Determine if the goal is to maximize (+) or minimize (-)
    target_idx = goal_indices[0]
    is_maximize = header[target_idx].endswith('+')
    
    # Get all values for the goal column from the dataset
    all_goals = [r[target_idx] for r in rows]
    
    # Calculate how many rows the target row is "better than"
    if is_maximize:
        better_than = [g for g in all_goals if prediction > g]
    else:
        better_than = [g for g in all_goals if prediction < g]
        
    p = len(better_than) / len(all_goals)
    
    reason = (f"This arrangement ranks in the top {100 - int(p * 100)}% of all "
              f"recorded team arrangements based on its predicted performance.")
    
    return p, reason

if __name__ == "__main__":
    filename = 'WA_Fn-UseC_-HR-Employee-Attrition.n1000.train.csv'
    rows = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = []
            for i, val in enumerate(row):
                if header[i][0].isupper():
                    processed_row.append(float(val))
                else:
                    processed_row.append(val)
            rows.append(processed_row)

    # Pick a sample row to test, mask its goals
    test_row = list(rows[0])
    goal_indices = [i for i, h in enumerate(header) if h.endswith('+') or h.endswith('-')]
    for idx in goal_indices:
        test_row[idx] = "?"

    p, reason = justify(header, rows, test_row)
    
    print(f"Rank: {p:.2f}")
    print(f"Reason: {reason}")
