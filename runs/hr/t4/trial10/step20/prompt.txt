def guess(header, train, test):
    """
    Estimates the first goal column ('YearsAtCompany+') for test rows
    using a Random Forest Regressor based on input columns.
    """
    # Identify goal columns (ending in + or -)
    goal_indices = [i for i, h in enumerate(header) if h.endswith('+') or h.endswith('-')]
    target_idx = goal_indices[0]
    
    # Identify input columns (those not ending in + or -)
    input_indices = [i for i in range(len(header)) if i not in goal_indices]
    
    def encode(data):
        """Simple encoding: numeric stays, symbolic becomes categorical index."""
        encoded = []
        for row in data:
            new_row = []
            for i, val in enumerate(row):
                if isinstance(val, str):
                    # Convert string to unique integer
                    new_row.append(hash(val) % 1000)
                else:
                    new_row.append(val)
            encoded.append(new_row)
        return np.array(encoded)

    # Prepare training data
    X_train = encode([ [row[i] for i in input_indices] for row in train ])
    y_train = [row[target_idx] for row in train]
    
    # Prepare test data
    X_test = encode([ [row[i] for i in input_indices] for row in test ])
    
    # Train model
    model = RandomForestRegressor(n_estimators=50, random_state=42)
    model.fit(X_train, y_train)
    
    return model.predict(X_test).tolist()

def certify(xs, ys):
    """
    Returns True if the difference between two groups is statistically 
    insignificant (p-value > 0.05), False if they really differ.
    """
    t_stat, p_value = stats.ttest_ind(xs, ys, equal_var=False)
    return p_value > 0.05

def justify(header, rows, row):

    Estimates the goal for the given row and calculates its percentile rank
    against the existing rows.
    """

    # We use the guess function from step02 which handles the prediction

    target_idx = goal_indices[0]
    is_maximize = header[target_idx].endswith('+')
    
    # Get all values for the goal column from the dataset
    all_goals = [r[target_idx] for r in rows]
    
    # Calculate how many rows the target row is "better than"
    if is_maximize:
        better_than = [g for g in all_goals if prediction > g]
    else:
        better_than = [g for g in all_goals if prediction < g]

    reason = (f"This arrangement ranks in the top {100 - int(p * 100)}% of all "
              f"recorded team arrangements based on its predicted performance.")
    
    return p, reason

import csv

def spot(header, rows):
    """
    Identifies rows that look 'wrong' by checking for outliers in numeric columns.
    A row is considered suspicious if any of its numeric values are more than 
    3 standard deviations away from the mean of that column.
    """
    suspicious = {}
    
    # Identify numeric columns (starting with uppercase)
    numeric_indices = [i for i, h in enumerate(header) if h[0].isupper()]
    
    for idx in numeric_indices:
        col_values = [row[idx] for row in rows if isinstance(row[idx], (int, float))]
        if not col_values:
            continue
            
        mean = sum(col_values) / len(col_values)
        std = (sum((x - mean) ** 2 for x in col_values) / len(col_values)) ** 0.5
        
        if std == 0:
            continue
            
        # Flag rows where value is > 3 standard deviations from mean
        for i, row in enumerate(rows):
            val = row[idx]
            if isinstance(val, (int, float)):
                z_score = abs(val - mean) / std
                if z_score > 3:
                    suspicious[i] = suspicious.get(i, 0) + z_score
                    
    # Return indices sorted by total suspicious score (most suspicious first)
    return sorted(suspicious, key=suspicious.get, reverse=True)

import csv

def race(header, rows, label):
    """
    Compares three strategies for finding the best employee row within a 50-label budget.
    Strategies:
    1. random: Picks 17 random rows.
    2. age_heuristic: Picks 17 rows with the highest 'Age'.
    3. model_based: Uses a Random Forest to predict 'YearsAtCompany+' and picks the top 16.
    """
    goal_indices = [i for i, h in enumerate(header) if h.endswith('+') or h.endswith('-')]
    target_idx = goal_indices[0]
    
    budget = 50
    labeled = {}

    def get_val(i):
        if i not in labeled:
            if len(labeled) >= budget:
                return -float('inf')
            labeled[i] = label(i)[0]
        return labeled[i]

    # Strategy 1: Random
    def strategy_random():
        indices = random.sample(range(len(rows)), 17)
        best_i = -1
        best_v = -float('inf')
        for i in indices:
            v = get_val(i)
            if v > best_v:
                best_v, best_i = v, i
        return best_i, best_v

    # Strategy 2: Age-based
    def strategy_age():
        age_idx = header.index('Age')
        sorted_indices = sorted(range(len(rows)), key=lambda i: rows[i][age_idx], reverse=True)
        best_i = -1
        best_v = -float('inf')
        for i in sorted_indices[:17]:
            v = get_val(i)
            if v > best_v:
                best_v, best_i = v, i
        return best_i, best_v

    # Strategy 3: Model-based
    def strategy_model():
        train_idx = list(range(100))
        # Prepare training data: replace '?' with dummy values for training
        train_data = []
        for i in train_idx:
            row = list(rows[i])
            row[target_idx] = get_val(i)
            train_data.append(row)
        
        # Predict on remaining
        test_idx = list(range(100, 116))
        test_data = [list(rows[i]) for i in test_idx]
        for row in test_data: row[target_idx] = 0.0 # Dummy
        
        preds = guess(header, train_data, test_data)
        best_pred_idx = test_idx[preds.index(max(preds))]
        return best_pred_idx, get_val(best_pred_idx)

    results = {
        "random": strategy_random(),
        "age_heuristic": strategy_age(),
        "model_based": strategy_model()
    }

    winner = max(results, key=lambda k: results[k][1])
    
    return {
        "winner": winner,
        "row": results[winner][0],
        "tried": ["random", "age_heuristic", "model_based"]
    }

import csv

def flow(header, row, seen):
    """
    Detects if a team's composition has shifted significantly.
    We compare the mean of numeric inputs of the current row against 
    the historical mean of all 'seen' rows. If the current row's 
    average numeric input is more than 2 standard deviations away 
    from the historical mean, we flag it as a shift.
    """
    if not seen:
        return False

    # Identify numeric input columns (excluding goals)
    numeric_indices = [
        i for i, h in enumerate(header) 
        if h[0].isupper() and not (h.endswith('+') or h.endswith('-'))
    ]
    
    if not numeric_indices:
        return False

    # Calculate historical average of numeric inputs
    history_vals = []
    for r in seen:
        history_vals.extend([r[i] for i in numeric_indices])
    
    if not history_vals:
        return False
        
    mean = sum(history_vals) / len(history_vals)
    std = (sum((x - mean) ** 2 for x in history_vals) / len(history_vals)) ** 0.5
    
    if std == 0:
        return False
        
    # Calculate current row's numeric average
    current_vals = [row[i] for i in numeric_indices]
    current_mean = sum(current_vals) / len(current_vals)
    
    # Flag if current row is a significant outlier (Z-score > 2)
    z_score = abs(current_mean - mean) / std
    return z_score > 2

import csv
from sklearn.neighbors import NearestNeighbors
import numpy as np

def spot(header, rows):
    """
    Identifies rows that are 'unlike anyone we have employed'.
    We use Nearest Neighbors to find the distance to the closest existing employee.
    Rows with a large distance to their nearest neighbor are considered 'unlike'
    and thus their estimates cannot be trusted.
    """
    goal_indices = [i for i, h in enumerate(header) if h.endswith('+') or h.endswith('-')]
    input_indices = [i for i in range(len(header)) if i not in goal_indices]
    
    def encode(data):
        encoded = []
        for row in data:
            new_row = []
            for i, val in enumerate(row):
                if isinstance(val, str):
                    new_row.append(hash(val) % 1000)
                else:
                    new_row.append(val)
            encoded.append(new_row)
        return np.array(encoded)

    X = encode([[row[i] for i in input_indices] for row in rows])
    
    # Find the distance to the nearest neighbor for every row
    # We use n_neighbors=2 because the closest neighbor is the row itself (dist=0)
    nn = NearestNeighbors(n_neighbors=2)
    nn.fit(X)
    distances, _ = nn.kneighbors(X)
    
    # The second column of distances is the distance to the nearest *other* row
    neighbor_distances = distances[:, 1]
    
    # Identify indices of rows sorted by distance (most suspicious/unlike first)
    indices = np.argsort(neighbor_distances)[::-1]
    
    return indices.tolist()
