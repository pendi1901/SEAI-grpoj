import csv

def choose(header, rows, label):
    """
    Finds the best working arrangement for 'Laboratory Technician' staff 
    by sampling up to 50 rows from that subset.
    """
    # Identify the jobRole column
    role_idx = header.index('jobRole')
    
    # Filter rows for 'Laboratory Technician'
    lab_indices = [i for i, row in enumerate(rows) if row[role_idx] == 'Laboratory Technician']
    
    if not lab_indices:
        return 0
        
    # We have a budget of 50 labels.
    # Sample up to 50 indices from the laboratory staff
    sample_indices = random.sample(lab_indices, min(50, len(lab_indices)))
    
    best_idx = -1
    best_val = -float('inf')
    
    # Identify goal column (YearsAtCompany+)
    goal_indices = [i for i, h in enumerate(header) if h.endswith('+') or h.endswith('-')]
    target_idx = goal_indices[0]
    
    for i in sample_indices:
        # Label the row to get the true goal values
        true_goals = label(i)
        val = true_goals[0] # YearsAtCompany+
        
        if val > best_val:
            best_val = val
            best_idx = i
            
    return best_idx
