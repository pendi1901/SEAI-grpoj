import csv
import numpy as np
from sklearn.ensemble import RandomForestRegressor
from step02 import guess

def justify(header, rows, row):
    """
    Estimates the goal for the given row and compares it to the dataset.
    Answers the customer's question about why we don't just pay everyone more
    by highlighting that performance is multi-faceted and salary is only one factor.
    """
    # Identify goal columns
    goal_indices = [i for i, h in enumerate(header) if h.endswith('+') or h.endswith('-')]
    target_idx = goal_indices[0]
    
    # Estimate the goal for the target row
    prediction = guess(header, rows, [row])[0]
    
    # Determine if the goal is to maximize (+) or minimize (-)
    is_maximize = header[target_idx].endswith('+')
    
    # Get all values for the goal column from the dataset
    all_goals = [r[target_idx] for r in rows]
    
    # Calculate how many rows the target row is "better than"
    if is_maximize:
        better_than = [g for g in all_goals if prediction > g]
    else:
        better_than = [g for g in all_goals if prediction < g]
        
    p = len(better_than) / len(all_goals)
    
    reason = (f"While increasing salary is a lever, our model shows that retention is driven by a complex "
              f"combination of factors, meaning that simply paying more does not guarantee the same "
              f"performance improvement as optimizing the specific balance of roles and environment.")
    
    return p, reason

if __name__ == "__main__":
    filename = 'WA_Fn-UseC_-HR-Employee-Attrition.n1000.train.csv'
    rows = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = [float(val) if header[i][0].isupper() else val for i, val in enumerate(row)]
            rows.append(processed_row)

    # Pick a sample row to test, mask its goals
    test_row = list(rows[0])
    goal_indices = [i for i, h in enumerate(header) if h.endswith('+') or h.endswith('-')]
    for idx in goal_indices:
        test_row[idx] = "?"

    p, reason = justify(header, rows, test_row)
    
    print("Customer Question: Why do you not simply recommend paying everyone more?")
    print(f"Answer: {reason}")
    print(f"Statistical context: This arrangement ranks better than {p:.2%} of existing records.")
