import csv
import random
from step07 import choose as base_choose

def choose(header, rows, label):
    """
    Finds the best working arrangement for 'Laboratory Technician' staff
    by sampling up to 50 rows from that specific subset.
    """
    idx_role = header.index('jobRole')
    
    # Filter rows to only include 'Laboratory Technician'
    lab_indices = [i for i, row in enumerate(rows) if row[idx_role] == 'Laboratory Technician']
    
    # If no lab techs, fallback to all rows
    search_space = lab_indices if lab_indices else range(len(rows))
    
    # Sample up to 50 indices from the lab tech subset
    sample_size = min(50, len(search_space))
    indices_to_sample = random.sample(search_space, sample_size)
    
    best_idx = -1
    max_tenure = -float('inf')
    
    for i in indices_to_sample:
        # label(i) returns the true goal values for rows[i]
        true_values = label(i)
        tenure = true_values[0] # The goal is the first (and only) goal column
        
        if tenure > max_tenure:
            max_tenure = tenure
            best_idx = i
            
    return best_idx

if __name__ == "__main__":
    filename = 'WA_Fn-UseC_-HR-Employee-Attrition.n1000.train.csv'
    rows = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = []
            for i, val in enumerate(row):
                if header[i][0].isupper():
                    processed_row.append(float(val))
                else:
                    processed_row.append(val)
            rows.append(processed_row)

    # Setup for the harness
    goal_idx = next(i for i, col in enumerate(header) if col.endswith('+'))
    
    # Store true values to simulate the label() function
    true_goals = [row[goal_idx] for row in rows]
    
    # Mask the rows
    for row in rows:
        row[goal_idx] = "?"
        
    def label(i):
        return [true_goals[i]]
    
    best_index = choose(header, rows, label)
    
    print(f"Analysis complete for Laboratory Technicians.")
    print(f"The best row found among the sampled lab staff is at index: {best_index}")
    print(f"True tenure for this record: {true_goals[best_index]} years.")
    exit(0)
