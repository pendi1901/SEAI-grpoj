import csv

def remember(header, rows):
    """
    Identifies rows where staff stay 'unusually long'.
    'YearsAtCompany+' is a goal to maximize.
    We calculate the 95th percentile of 'YearsAtCompany+' and return the indices
    of rows that meet or exceed this threshold, limited to 5% of total rows.
    """
    idx_years = header.index('YearsAtCompany+')
    
    # Extract all values for the goal column
    years = [row[idx_years] for row in rows]
    
    # Determine threshold for "unusually long" (top 5%)
    sorted_years = sorted(years, reverse=True)
    limit = max(1, len(rows) // 20)
    threshold = sorted_years[limit - 1]
    
    # Find indices of rows that meet the threshold
    best_indices = [i for i, val in enumerate(years) if val >= threshold]
    
    # Ensure we return at most 5% of the rows
    return best_indices[:limit]

import csv

from sklearn.ensemble import RandomForestRegressor
from sklearn.preprocessing import LabelEncoder

def guess(header, train, test):
    """
    Estimates the first goal column ('YearsAtCompany+') for test rows
    using a Random Forest Regressor, handling '?' in test data.
    """
    # Identify goal column (first column ending in + or -)
    goal_idx = -1
    for i, col in enumerate(header):
        if col.endswith('+') or col.endswith('-'):
            goal_idx = i
            break
    
    # Prepare data: encode symbolic columns using training data only
    # to ensure consistency and avoid '?' issues in test set
    encoders = {}
    
    def encode_row(row, is_train=True):
        encoded = []
        for i, val in enumerate(row):
            if i == goal_idx and not is_train:
                encoded.append(0.0)  # Placeholder for '?'
            elif isinstance(val, str):
                if i not in encoders:
                    le = LabelEncoder()
                    # Fit on all available string data from train
                    le.fit([r[i] for r in train])
                    encoders[i] = le
                encoded.append(float(encoders[i].transform([val])[0]))
            else:
                encoded.append(float(val))
        return encoded

    X_train = []
    y_train = []
    for row in train:
        encoded = encode_row(row, is_train=True)
        y_train.append(encoded[goal_idx])
        X_train.append(encoded[:goal_idx] + encoded[goal_idx+1:])
    
    X_test = []
    for row in test:
        encoded = encode_row(row, is_train=False)
        X_test.append(encoded[:goal_idx] + encoded[goal_idx+1:])
    
    # Train model
    model = RandomForestRegressor(n_estimators=100, random_state=42)
    model.fit(X_train, y_train)
    
    return model.predict(X_test).tolist()

import csv
from scipy import stats

def certify(xs, ys):
    """
    Returns True if the difference between two groups is not statistically 
    significant (p > 0.05), suggesting the difference is likely noise.
    """
    t_stat, p_value = stats.ttest_ind(xs, ys, equal_var=False)
    return p_value > 0.05

def analyze_attrition_tenure(header, rows):
    """
    Compares 'YearsAtCompany+' for employees who work overtime vs those who don't.
    """
    idx_overtime = header.index('overTime')
    idx_tenure = header.index('YearsAtCompany+')
    
    overtime_yes = [row[idx_tenure] for row in rows if row[idx_overtime] == 'Yes']
    overtime_no = [row[idx_tenure] for row in rows if row[idx_overtime] == 'No']
    
    avg_yes = sum(overtime_yes) / len(overtime_yes)
    avg_no = sum(overtime_no) / len(overtime_no)
    
    is_noise = certify(overtime_yes, overtime_no)
    
    print(f"Average tenure (Overtime=Yes): {avg_yes:.2f} years")
    print(f"Average tenure (Overtime=No): {avg_no:.2f} years")
    if is_noise:
        print("Conclusion: The difference is likely just noise.")
    else:
        print("Conclusion: The difference is statistically significant.")
    return is_noise

import csv
from sklearn.ensemble import RandomForestRegressor
from sklearn.preprocessing import LabelEncoder

def blame(header, rows):
    """
    Identifies input columns that best explain short tenure ('YearsAtCompany+').
    We define 'worst' as the bottom 20% of tenure. We then use a Random Forest
    to find feature importances for predicting tenure, filtering for input columns.
    """
    goal_col = 'YearsAtCompany+'
    goal_idx = header.index(goal_col)
    
    # Filter out goal columns from inputs
    input_indices = [i for i, col in enumerate(header) if not (col.endswith('+') or col.endswith('-'))]
    
    # Encode data
    encoders = {}
    X = []
    y = []
    for row in rows:
        encoded_row = []
        for i, val in enumerate(row):
            if isinstance(val, str):
                if i not in encoders:
                    le = LabelEncoder()
                    le.fit([r[i] for r in rows])
                    encoders[i] = le
                encoded_row.append(float(encoders[i].transform([val])[0]))
            else:
                encoded_row.append(float(val))
        X.append([encoded_row[i] for i in input_indices])
        y.append(encoded_row[goal_idx])
    
    # Train a regressor to find feature importance
    model = RandomForestRegressor(n_estimators=50, random_state=42)
    model.fit(X, y)
    
    # Get importances
    importances = model.feature_importances_
    feature_scores = sorted(zip(input_indices, importances), key=lambda x: x[1], reverse=True)
    
    # Return top 3 input columns that explain tenure
    return [header[idx] for idx, score in feature_scores[:3]]
