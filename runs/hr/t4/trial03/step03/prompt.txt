import csv

def remember(header, rows):
    """
    Identifies rows where staff stay 'unusually long'.
    'YearsAtCompany+' is a goal to maximize.
    We calculate the 95th percentile of 'YearsAtCompany+' and return the indices
    of rows that meet or exceed this threshold, limited to 5% of total rows.
    """
    idx_years = header.index('YearsAtCompany+')
    
    # Extract all values for the goal column
    years = [row[idx_years] for row in rows]
    
    # Determine threshold for "unusually long" (top 5%)
    sorted_years = sorted(years, reverse=True)
    limit = max(1, len(rows) // 20)
    threshold = sorted_years[limit - 1]
    
    # Find indices of rows that meet the threshold
    best_indices = [i for i, val in enumerate(years) if val >= threshold]
    
    # Ensure we return at most 5% of the rows
    return best_indices[:limit]

import csv
import numpy as np
from sklearn.ensemble import RandomForestRegressor
from sklearn.preprocessing import LabelEncoder

def guess(header, train, test):
    """
    Estimates the first goal column ('YearsAtCompany+') for test rows
    using a Random Forest Regressor, handling '?' in test data.
    """
    # Identify goal column (first column ending in + or -)
    goal_idx = -1
    for i, col in enumerate(header):
        if col.endswith('+') or col.endswith('-'):
            goal_idx = i
            break
    
    # Prepare data: encode symbolic columns using training data only
    # to ensure consistency and avoid '?' issues in test set
    encoders = {}
    
    def encode_row(row, is_train=True):
        encoded = []
        for i, val in enumerate(row):
            if i == goal_idx and not is_train:
                encoded.append(0.0)  # Placeholder for '?'
            elif isinstance(val, str):
                if i not in encoders:
                    le = LabelEncoder()
                    # Fit on all available string data from train
                    le.fit([r[i] for r in train])
                    encoders[i] = le
                encoded.append(float(encoders[i].transform([val])[0]))
            else:
                encoded.append(float(val))
        return encoded

    X_train = []
    y_train = []
    for row in train:
        encoded = encode_row(row, is_train=True)
        y_train.append(encoded[goal_idx])
        X_train.append(encoded[:goal_idx] + encoded[goal_idx+1:])
    
    X_test = []
    for row in test:
        encoded = encode_row(row, is_train=False)
        X_test.append(encoded[:goal_idx] + encoded[goal_idx+1:])
    
    # Train model
    model = RandomForestRegressor(n_estimators=100, random_state=42)
    model.fit(X_train, y_train)
    
    return model.predict(X_test).tolist()
