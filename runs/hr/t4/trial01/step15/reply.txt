import csv
from sklearn.ensemble import RandomForestRegressor
from sklearn.preprocessing import LabelEncoder

def blame(header, rows):
    """
    Identifies input columns that have the least impact on the goal columns.
    We use a Random Forest to rank feature importance and return the columns
    with the lowest importance scores.
    """
    # Identify goal columns
    goal_indices = [i for i, col in enumerate(header) if col.endswith('+') or col.endswith('-')]
    # Identify input columns
    input_indices = [i for i, col in enumerate(header) if not (col.endswith('+') or col.endswith('-'))]
    input_names = [header[i] for i in input_indices]
    
    # We focus on the first goal column for simplicity as per previous steps
    goal_idx = goal_indices[0]
    
    # Prepare data
    X = []
    y = []
    encoders = {}
    
    for row in rows:
        processed_row = []
        for i in input_indices:
            val = row[i]
            if isinstance(val, str):
                if i not in encoders:
                    le = LabelEncoder()
                    le.fit([str(r[i]) for r in rows])
                    encoders[i] = le
                processed_row.append(encoders[i].transform([str(val)])[0])
            else:
                processed_row.append(val)
        X.append(processed_row)
        y.append(row[goal_idx])
        
    # Train model to find feature importance
    model = RandomForestRegressor(n_estimators=50, random_state=42)
    model.fit(X, y)
    
    # Get feature importances
    importances = model.feature_importances_
    
    # Sort features by importance (ascending)
    feature_importance = sorted(zip(input_names, importances), key=lambda x: x[1])
    
    # Return the bottom 5 features that have the least impact
    return [f[0] for f in feature_importance[:5]]

if __name__ == "__main__":
    filename = 'WA_Fn-UseC_-HR-Employee-Attrition.n1000.train.csv'
    rows = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = []
            for i, val in enumerate(row):
                if header[i][0].isupper():
                    processed_row.append(float(val))
                else:
                    processed_row.append(val)
            rows.append(processed_row)
            
    unimportant_factors = blame(header, rows)
    
    print("Columns that have little to no impact on employee tenure:")
    for i, factor in enumerate(unimportant_factors, 1):
        print(f"{i}. {factor}")
    print("\nThese columns contribute the least to predicting tenure and could potentially be retired.")
