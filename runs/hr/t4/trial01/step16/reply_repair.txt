import csv
from step02 import guess

def justify(header, rows, row):
    """
    Estimates the goal column for the given row and calculates its rank
    relative to the existing rows. Addresses the 'pay everyone more' question
    by comparing the current row's goal to a version with increased income.
    """
    goal_idx = next(i for i, col in enumerate(header) if col.endswith('+') or col.endswith('-'))
    income_idx = header.index('MonthlyIncome')
    is_maximize = header[goal_idx].endswith('+')

    # Prepare data for prediction: replace '?' with a dummy value for the model
    # We use the mean of the column as a neutral dummy value
    goal_vals = [r[goal_idx] for r in rows]
    mean_goal = sum(goal_vals) / len(goal_vals)

    def get_prediction(r):
        test_row = list(r)
        test_row[goal_idx] = mean_goal
        return guess(header, rows, [test_row])[0]

    # Baseline prediction
    pred_baseline = get_prediction(row)
    
    # Hypothetical prediction (20% pay increase)
    hypothetical_row = list(row)
    hypothetical_row[income_idx] = row[income_idx] * 1.2
    pred_boosted = get_prediction(hypothetical_row)

    # Calculate rank (p)
    if is_maximize:
        better_count = sum(1 for g in goal_vals if pred_baseline > g)
    else:
        better_count = sum(1 for g in goal_vals if pred_baseline < g)
    p = better_count / len(goal_vals)

    # Formulate reason
    diff = abs(pred_boosted - pred_baseline)
    if diff < 0.1:
        reason = "Increasing pay is not recommended as it shows negligible impact on the outcome for this employee."
    else:
        reason = f"Increasing pay by 20% is estimated to change the outcome by {diff:.2f} units, which may not be cost-effective compared to other factors."

    return p, reason

if __name__ == "__main__":
    filename = 'WA_Fn-UseC_-HR-Employee-Attrition.n1000.train.csv'
    rows = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = [float(val) if header[i][0].isupper() else val for i, val in enumerate(row)]
            rows.append(processed_row)
            
    # Test with the last row
    test_row = list(rows[-1])
    goal_idx = next(i for i, col in enumerate(header) if col.endswith('+') or col.endswith('-'))
    test_row[goal_idx] = "?"
    
    p, reason = justify(header, rows[:-1], test_row)
    
    print(f"Rank: {p:.4f}")
    print(f"Reason: {reason}")
