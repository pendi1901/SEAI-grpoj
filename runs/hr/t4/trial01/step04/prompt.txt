import pandas as pd


def remember(header, rows):
    n = len(rows)
    goals_plus = [i for i, h in enumerate(header) if h.endswith('+')]
    goals_minus = [i for i, h in enumerate(header) if h.endswith('-')]

    col_min = {}
    col_max = {}
    for idx in range(n):
        row = rows[idx]
        for c in goals_plus:
            v = float(row[c])
            col_max[c] = max(col_max.get(c, -float('inf')), v)
        for c in goals_minus:
            v = float(row[c])
            col_min[c] = min(col_min.get(c, float('inf')), v)

    scores = []
    for idx in range(n):
        s = 0.0
        rng = {}
        for c in goals_plus:
            lo, hi = col_min.get(c, -float('inf')), col_max.get(c, -float('inf'))
            rng[c] = (hi - lo) or 1.0
            s += (float(rows[idx][c]) - lo) / rng[c]
        for c in goals_minus:
            lo, hi = col_min.get(c, float('inf')), col_max.get(c, float('inf'))
            rng[c] = (hi - lo) or 1.0
            s += (hi - float(rows[idx][c])) / rng[c]
        scores.append(s)

    ranked = sorted(range(n), key=lambda i: scores[i], reverse=True)
    max_ret = max(1, int(n * 0.05))
    return ranked[:max_ret]

import pandas as pd
from sklearn.ensemble import RandomForestRegressor


def guess(header, train, test):
    """Estimate YearsAtCompany+ (the first goal column) for each test row."""
    
    target_col = None
    for col in header:
        if col.endswith('+') or col.endswith('-'):
            target_col = col
            break
    
    df_train = pd.DataFrame(train, columns=header)
    df_test = pd.DataFrame(test, columns=header)
    
    # Replace "?" with NaN so missing values are handled properly
    df_train = df_train.replace('?', float('nan')).infer_objects()
    df_test = df_test.replace('?', float('nan')).infer_objects()
    
    X_train = df_train.drop(columns=[target_col]).values
    y_train = df_train[target_col].values
    
    X_test = df_test.drop(columns=[target_col]).values
    
    model = RandomForestRegressor(n_estimators=100, random_state=42)
    model.fit(X_train, y_train)
    
    predictions = model.predict(X_test).tolist()
    return predictions

def certify(xs, ys):
    """Returns True if xs and ys are the same apart from noise."""
    mean_x = sum(xs) / len(xs) if xs else 0
    mean_y = sum(ys) / len(ys) if ys else 0
    
    diff = abs(mean_x - mean_y)
    scale = max(abs(mean_x), abs(mean_y)) or 1.0
    
    return diff < 0.5 * scale
