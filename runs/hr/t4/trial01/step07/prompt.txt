import csv
from sklearn.ensemble import RandomForestRegressor
from sklearn.preprocessing import LabelEncoder

def guess(header, train, test):
    """
    Estimates the first goal column ('YearsAtCompany+') for test rows
    using a Random Forest Regressor.
    """
    # Identify goal column (first column ending in '+' or '-')
    goal_idx = -1
    for i, col in enumerate(header):
        if col.endswith('+') or col.endswith('-'):
            goal_idx = i
            break
    
    # Prepare data: convert symbolic columns to numeric for scikit-learn
    def encode_data(data, header_subset):
        encoded = []
        encoders = {}
        for i in range(len(header_subset)):
            if isinstance(data[0][i], str):
                le = LabelEncoder()
                col_data = [row[i] for row in data]
                le.fit(col_data)
                encoders[i] = le
                encoded.append(le.transform(col_data))
            else:
                encoded.append([row[i] for row in data])
        return list(zip(*encoded)), encoders

    # Define input columns (exclude goal column)
    input_indices = [i for i in range(len(header)) if i != goal_idx]
    header_subset = [header[i] for i in input_indices]
    
    X_train_raw = [[row[i] for i in input_indices] for row in train]
    y_train = [row[goal_idx] for row in train]
    
    X_train, encoders = encode_data(X_train_raw, header_subset)
    
    # Encode test data using same encoders
    X_test_raw = [[row[i] for i in input_indices] for row in test]
    X_test = []
    for row in X_test_raw:
        new_row = []
        for i, val in enumerate(row):
            if i in encoders:
                try:
                    new_row.append(encoders[i].transform([val])[0])
                except ValueError:
                    new_row.append(0)
            else:
                new_row.append(val)
        X_test.append(new_row)
        
    # Train model
    model = RandomForestRegressor(n_estimators=100, random_state=42)
    model.fit(X_train, y_train)
    
    return model.predict(X_test).tolist()

import csv
from scipy import stats

def certify(xs, ys):
    """
    Returns True if the difference between two groups is not statistically 
    significant (p-value > 0.05), suggesting the difference is likely noise.
    Returns False if the difference is statistically significant.
    """
    t_stat, p_value = stats.ttest_ind(xs, ys, equal_var=False)
    return p_value > 0.05

def analyze_overtime_impact(header, rows):
    """
    Compares 'YearsAtCompany+' for employees who work overtime vs those who don't.
    """
    ot_idx = header.index('overTime')
    yac_idx = header.index('YearsAtCompany+')
    
    overtime_yes = [row[yac_idx] for row in rows if row[ot_idx] == 'Yes']
    overtime_no = [row[yac_idx] for row in rows if row[ot_idx] == 'No']
    
    is_noise = certify(overtime_yes, overtime_no)
    
    avg_yes = sum(overtime_yes) / len(overtime_yes)
    avg_no = sum(overtime_no) / len(overtime_no)
    
    return is_noise, avg_yes, avg_no

import csv
from sklearn.ensemble import RandomForestRegressor
from sklearn.preprocessing import LabelEncoder

def blame(header, rows):
    """
    Identifies input columns that best explain short stays (low 'YearsAtCompany+').
    Uses a Random Forest to rank feature importance for predicting 'YearsAtCompany+'.
    """
    goal_col = "YearsAtCompany+"
    goal_idx = header.index(goal_col)
    
    # Filter out goal columns and identify input columns
    input_indices = [i for i, col in enumerate(header) if not (col.endswith('+') or col.endswith('-'))]
    input_names = [header[i] for i in input_indices]
    
    # Prepare data
    X = []
    y = []
    encoders = {}
    
    for row in rows:
        processed_row = []
        for i in input_indices:
            val = row[i]
            if isinstance(val, str):
                if i not in encoders:
                    le = LabelEncoder()
                    le.fit([r[i] for r in rows])
                    encoders[i] = le
                processed_row.append(encoders[i].transform([val])[0])
            else:
                processed_row.append(val)
        X.append(processed_row)
        y.append(row[goal_idx])
        
    # Train model to find feature importance
    model = RandomForestRegressor(n_estimators=50, random_state=42)
    model.fit(X, y)
    
    # Get feature importances
    importances = model.feature_importances_
    
    # Sort features by importance
    feature_importance = sorted(zip(input_names, importances), key=lambda x: x[1], reverse=True)
    
    # Return the top 3 features that explain the variance in 'YearsAtCompany+'
    return [f[0] for f in feature_importance[:3]]

def justify(header, rows, row):

    Estimates the goal column for the given row and calculates its rank

    goal_idx = next(i for i, col in enumerate(header) if col.endswith('+') or col.endswith('-'))
    is_maximize = header[goal_idx].endswith('+')
    
    # Create a copy of the row to fill in the goal

    # doesn't crash on the '?' string.

    test_row[goal_idx] = 0.0

    prediction = guess(header, rows, [test_row])[0]
    
    # Compare prediction against all rows
    all_goals = [r[goal_idx] for r in rows]
