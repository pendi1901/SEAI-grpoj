    """Compare three strategies for finding the best employee among up to 50 labelled ones."""

    goals_plus = [h for h in header if h.endswith('+')]
    goals_minus = [h for h in header if h.endswith('-')]

    # --- Strategy: random (pick rows uniformly at random) ---
    def _random():

    def _score(row):
        s = 0.0
        for c in goals_plus:
            v = float(row[c]) if row[c] != '?' else 0.0
            lo, hi = -float('inf'), float('inf')
            rng = (hi - lo) or 1.0
            s += (v - lo) / rng
        for c in goals_minus:
            v = float(row[c]) if row[c] != '?' else 0.0
            lo, hi = -float('inf'), float('inf')
            rng = (hi - lo) or 1.0
            s += (hi - v) / rng
        return s

        best_idx = None

                if s > best_score:
                    best_score = s
                    best_idx = i
        return [best_idx] if best_idx is not None else []
