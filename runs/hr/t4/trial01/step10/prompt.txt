import pandas as pd


def remember(header, rows):
    n = len(rows)
    goals_plus = [i for i, h in enumerate(header) if h.endswith('+')]
    goals_minus = [i for i, h in enumerate(header) if h.endswith('-')]

    col_min = {}
    col_max = {}
    for idx in range(n):
        row = rows[idx]
        for c in goals_plus:
            v = float(row[c])
            col_max[c] = max(col_max.get(c, -float('inf')), v)
        for c in goals_minus:
            v = float(row[c])
            col_min[c] = min(col_min.get(c, float('inf')), v)

    scores = []
    for idx in range(n):
        s = 0.0
        rng = {}
        for c in goals_plus:
            lo, hi = col_min.get(c, -float('inf')), col_max.get(c, -float('inf'))
            rng[c] = (hi - lo) or 1.0
            s += (float(rows[idx][c]) - lo) / rng[c]
        for c in goals_minus:
            lo, hi = col_min.get(c, float('inf')), col_max.get(c, float('inf'))
            rng[c] = (hi - lo) or 1.0
            s += (hi - float(rows[idx][c])) / rng[c]
        scores.append(s)

    ranked = sorted(range(n), key=lambda i: scores[i], reverse=True)
    max_ret = max(1, int(n * 0.05))
    return ranked[:max_ret]

def certify(xs, ys):
    """Returns True if xs and ys are the same apart from noise."""
    mean_x = sum(xs) / len(xs) if xs else 0
    mean_y = sum(ys) / len(ys) if ys else 0
    
    diff = abs(mean_x - mean_y)
    scale = max(abs(mean_x), abs(mean_y)) or 1.0
    
    return diff < 0.5 * scale

import pandas as pd


def flow(header, row, seen):
    """Returns True if this person needs a 'what would keep you here' conversation."""
    
    goals_plus = [h for h in header if h.endswith('+')]
    goals_minus = [h for h in header if h.endswith('-')]
    
    # Accumulate running min/max from all previously seen rows
    col_min = {}
    col_max = {}
    for s_row in seen:
        for c in goals_plus:
            v = float(s_row[c])
            col_max[c] = max(col_max.get(c, -float('inf')), v)
        for c in goals_minus:
            v = float(s_row[c])
            col_min[c] = min(col_min.get(c, float('inf')), v)
    
    # Score the current row using accumulated statistics
    s = 0.0
    for c in goals_plus:
        lo = col_min.get(c, -float('inf'))
        hi = col_max.get(c, -float('inf'))
        rng = (hi - lo) or 1.0
        s += (float(row[c]) - lo) / rng
    for c in goals_minus:
        lo = col_min.get(c, float('inf'))
        hi = col_max.get(c, float('inf'))
        rng = (hi - lo) or 1.0
        s += (hi - float(row[c])) / rng
    
    # Threshold: top ~5% of the combined seen+current population
    n_total = len(seen) + 1
    threshold = max(1, int(n_total * 0.05))
    
    return s >= 3.0

def blame(header, rows):

    scores = []

        row = rows[idx]
        for c in goals_plus:
            v = float(row[c])
            col_max[c] = max(col_max.get(c, -float('inf')), v)
        for c in goals_minus:
            v = float(row[c])
            col_min[c] = min(col_min.get(c, float('inf')), v)

    # Compute scores
    for idx in range(len(rows)):
        s = 0.0
        for c in goals_plus:
            lo = col_min.get(c, -float('inf'))
            hi = col_max.get(c, -float('inf'))
            rng = (hi - lo) or 1.0

        for c in goals_minus:
            lo = col_min.get(c, float('inf'))
            hi = col_max.get(c, float('inf'))
            rng = (hi - lo) or 1.0
            s += (hi - float(rows[idx][c])) / rng
        scores.append(s)

    # Worst rows are those with lowest scores
    worst_frac = 0.05
    n_worst = max(1, int(len(scores) * worst_frac))
    worst_indices = sorted(range(len(scores)), key=lambda i: scores[i])[:n_worst]

    # Non-goal columns that could explain the difference
    input_cols = [h for h in header if not h.endswith('+') and not h.endswith('-')]

    # For each input column, check mean difference between worst and rest
    worst_set = set(worst_indices)
    result = []
    for col in input_cols:
        vals = []
        for i in range(len(rows)):
            v = rows[i][col]
            if v == '?':
                vals.append(0.0)
            else:
                try:

                except ValueError:
                    vals.append(0.0)

        worst_mean = sum(vals[i] for i in worst_indices) / len(worst_indices) if worst_indices else 0
        rest_vals = [vals[i] for i in range(len(rows)) if i not in worst_set]
        rest_mean = sum(rest_vals) / len(rest_vals) if rest_vals else 0

        diff = abs(worst_mean - rest_mean)
        scale = max(abs(worst_mean), abs(rest_mean)) or 1.0
        result.append((diff, col))

    # Sort by absolute difference (largest first)
    result.sort(key=lambda x: x[0], reverse=True)
    return [col for _, col in result[:5]]

def justify(header, rows, row):
    """Return (fraction_better, reason) for a single employee."""

    goals_plus = [h for h in header if h.endswith('+')]
    goals_minus = [h for h in header if h.endswith('-')]

    # Compute global min/max from the known population
    col_min = {}
    col_max = {}
    for r in rows:
        for c in goals_plus:
            v = float(r[c])
            col_max[c] = max(col_max.get(c, -float('inf')), v)
        for c in goals_minus:
            v = float(r[c])
            col_min[c] = min(col_min.get(c, float('inf')), v)

    # Score each known row using the same normalization
    scores = []
    for r in rows:
        s = 0.0
        for c in goals_plus:
            lo, hi = col_min.get(c, -float('inf')), col_max.get(c, -float('inf'))
            rng = (hi - lo) or 1.0
            s += (float(r[c]) - lo) / rng
        for c in goals_minus:
            lo, hi = col_min.get(c, float('inf')), col_max.get(c, float('inf'))
            rng = (hi - lo) or 1.0
            s += (hi - float(r[c])) / rng
        scores.append(s)

    # Estimate the query row's score: substitute "?" goals with population means
    est_row = list(row)
    for c in goals_plus:
        vals = [float(r[c]) for r in rows]
        est_row[c] = sum(vals) / len(vals) if vals else 0.0
    for c in goals_minus:
        vals = [float(r[c]) for r in rows]
        est_row[c] = sum(vals) / len(vals) if vals else 0.0

    # Score the estimated row with the same normalization
    s_query = 0.0
    for c in goals_plus:
        lo, hi = col_min.get(c, -float('inf')), col_max.get(c, -float('inf'))
        rng = (hi - lo) or 1.0
        s_query += (est_row[c] - lo) / rng
    for c in goals_minus:
        lo, hi = col_min.get(c, float('inf')), col_max.get(c, float('inf'))
        rng = (hi - lo) or 1.0
        s_query += (hi - est_row[c]) / rng

    # Fraction of population that the query row beats
    better_count = sum(1 for s in scores if s < s_query)
    p = better_count / len(scores)

    reason = f"Better than {p:.0%} of employees on tenure and satisfaction goals."
    return (p, reason)

import pandas as pd


def choose(header, rows, label):
    """Pick up to 50 employees and return the position of the one with the best goals."""
    
    # Goals are '+' columns (YearsAtCompany+, YearsInCurrentRole+)
    goals_plus = [h for h in header if h.endswith('+')]
    
    # We don't know any row's goals initially, so pick 50 rows to interview.
    n_rows = len(rows)
    sample_size = min(50, n_rows)
    
    best_idx = None
    best_score = -float('inf')
    
    for i in range(sample_size):
        true_goals = label(i)
        
        # Score: sum of normalized '+' goals (higher is better)
        s = 0.0
        for c in goals_plus:
            v = float(true_goals[c])
            lo, hi = -float('inf'), float('inf')
            rng = (hi - lo) or 1.0
            s += (v - lo) / rng
        
        if s > best_score:
            best_score = s
            best_idx = i
    
    return best_idx

import pandas as pd


def fix(header, row, levers):
    """Recommend changes to at least one lever for an underperforming employee."""

    goals_plus = [h for h in header if h.endswith('+')]
    goals_minus = [h for h in header if h.endswith('-')]

    # Build a reference population from the training data (loaded once)
    df = pd.read_csv("WA_Fn-UseC_-HR-Employee-Attrition.train.csv")
    ref_rows = [list(r) for r in df.values]
    ref_header = list(df.columns)

    # Compute global min/max across all reference rows for goals
    col_min = {}
    col_max = {}
    for r in ref_rows:
        for c in goals_plus:
            v = float(r[c])
            col_max[c] = max(col_max.get(c, -float('inf')), v)
        for c in goals_minus:
            v = float(r[c])
            col_min[c] = min(col_min.get(c, float('inf')), v)

    # Score each reference row
    ref_scores = []
    for r in ref_rows:
        s = 0.0
        for c in goals_plus:
            lo, hi = col_min.get(c, -float('inf')), col_max.get(c, -float('inf'))
            rng = (hi - lo) or 1.0
            s += (float(r[c]) - lo) / rng
        for c in goals_minus:
            lo, hi = col_min.get(c, float('inf')), col_max.get(c, float('inf'))
            rng = (hi - lo) or 1.0
            s += (hi - float(r[c])) / rng
        ref_scores.append(s)

    # Score the bad row using its actual goal values (no "?" for goals in this scenario)
    s_bad = 0.0
    for c in goals_plus:
        lo, hi = col_min.get(c, -float('inf')), col_max.get(c, -float('inf'))
        rng = (hi - lo) or 1.0
        s_bad += (float(row[c]) - lo) / rng
    for c in goals_minus:
        lo, hi = col_min.get(c, float('inf')), col_max.get(c, float('inf'))
        rng = (hi - lo) or 1.0
        s_bad += (hi - float(row[c])) / rng

    # Determine which lever to change based on the row's weakest goal gap
    # Map lever names to their '+' or '-' goal column
    lever_map = {
        'travel': 'businessTravel',
        'overtime': 'overTime',
        'training': 'TrainingTimesLastYear',
        'pay_rise': 'PercentSalaryHike',
        'stock': 'StockOptionLevel',
    }

    # For each lever, compute how much the row's score would improve if we set it to its best value
    candidate_changes = []
    for lever_name, goal_col in lever_map.items():
        if goal_col not in levers:
            continue
        lo, hi = col_min.get(goal_col, -float('inf')), col_max.get(goal_col, float('inf'))
        rng = (hi - lo) or 1.0

        # Current contribution of this goal to the bad row's score
        current_val = float(row[goal_col])
        if goal_col in goals_plus:
            cur_contrib = (current_val - lo) / rng
        else:
            cur_contrib = (hi - current_val) / rng

        # Best possible contribution
        best_val = levers[lever_name]
        if isinstance(best_val, tuple):
            best_val = float(best_val[1])  # high for '+' goals
        elif goal_col in goals_minus:
            best_val = float(best_val)
        else:
            best_val = float(best_val)

        new_contrib = (best_val - lo) / rng if goal_col in goals_plus else (hi - best_val) / rng
        improvement = new_contrib - cur_contrib

        candidate_changes.append((improvement, lever_name))

    # Pick the lever with the largest positive impact; fall back to first if none improve
    candidate_changes.sort(key=lambda x: x[0], reverse=True)
    chosen_improvement, chosen_lever = candidate_changes[0]

    result = {}
    if chosen_improvement > 0:
        best_val = levers[chosen_lever]
        if isinstance(best_val, tuple):
            result[chosen_lever] = str(best_val[1])
        else:
            result[chosen_lever] = str(best_val)
    else:
        # No lever improves; pick the one with least negative impact (closest to 0)
        chosen_improvement, chosen_lever = candidate_changes[-1]
        best_val = levers[chosen_lever]
        if isinstance(best_val, tuple):
            result[chosen_lever] = str(best_val[1])
        else:
            result[chosen_lever] = str(best_val)

    return result

import pandas as pd


def spot(header, rows):
    """Return row indices whose input values look wrong (most suspicious first)."""

    goals_plus = [h for h in header if h.endswith('+')]
    goals_minus = [h for h in header if h.endswith('-')]

    # Build a reference population from the training data to know what "normal" looks like.
    df_ref = pd.read_csv("WA_Fn-UseC_-HR-Employee-Attrition.train.csv")
    ref_header = list(df_ref.columns)
    ref_rows = [list(r) for r in df_ref.values]

    # Compute global min/max across reference rows for goals (skip '?' cells).
    col_min = {}
    col_max = {}
    for r in ref_rows:
        for c in goals_plus:
            v = float(r[c]) if r[c] != '?' else 0.0
            col_max[c] = max(col_max.get(c, -float('inf')), v)
        for c in goals_minus:
            v = float(r[c]) if r[c] != '?' else 0.0
            col_min[c] = min(col_min.get(c, float('inf')), v)

    # Score each reference row using the same normalization as earlier steps.
    ref_scores = []
    for r in ref_rows:
        s = 0.0
        for c in goals_plus:
            lo, hi = col_min.get(c, -float('inf')), col_max.get(c, -float('inf'))
            rng = (hi - lo) or 1.0
            s += (float(r[c]) if r[c] != '?' else 0.0 - lo) / rng
        for c in goals_minus:
            lo, hi = col_min.get(c, float('inf')), col_max.get(c, float('inf'))
            rng = (hi - lo) or 1.0
            s += (hi - (float(r[c]) if r[c] != '?' else 0.0)) / rng
        ref_scores.append(s)

    # Determine the typical score range for reference rows.
    mean_ref = sum(ref_scores) / len(ref_scores)
    std_ref = (sum((s - mean_ref) ** 2 for s in ref_scores) / len(ref_scores)) ** 0.5 or 1.0

    suspicious = []
    for i, row in enumerate(rows):
        # Compute score for the current row using reference statistics.
        s = 0.0
        for c in goals_plus:
            lo, hi = col_min.get(c, -float('inf')), col_max.get(c, -float('inf'))
            rng = (hi - lo) or 1.0
            v = float(row[c]) if row[c] != '?' else 0.0
            s += (v - lo) / rng
        for c in goals_minus:
            lo, hi = col_min.get(c, float('inf')), col_max.get(c, float('inf'))
            rng = (hi - lo) or 1.0
            v = float(row[c]) if row[c] != '?' else 0.0
            s += (hi - v) / rng

        # A suspicious row has a score far from the reference distribution.
        z = abs(s - mean_ref) / std_ref
        if z > 2.5:
            suspicious.append((z, i))

    suspicious.sort(key=lambda x: x[0], reverse=True)
    return [idx for _, idx in suspicious[:10]]
