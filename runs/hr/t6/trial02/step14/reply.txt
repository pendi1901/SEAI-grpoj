import ezr
import step03

def flow(header, row, seen):
    """
    Returns True if the current row represents a significant shift in the 
    team's distribution compared to the 'seen' history.
    We use the statistical tests from step03 (cliffs/ks/cohen) to check if 
    the current row's goals are significantly different from the historical 
    distribution of goals.
    """
    if len(seen) < 30:
        return False

    # Identify goal columns
    goal_indices = [i for i, name in enumerate(header) if name.endswith(('+', '-'))]
    
    # Extract goals for seen rows and current row
    seen_goals = [[r[i] for i in goal_indices] for r in seen]
    current_goals = [row[i] for i in goal_indices]
    
    # We check if the current row's ydist is an outlier compared to the 
    # distribution of ydist values in 'seen'.
    tbl = ezr.Tbl([header] + seen)
    seen_ydists = [ezr.ydist(tbl, r) for r in seen]
    current_ydist = ezr.ydist(tbl, row)
    
    # Use the same logic as step03 to check for statistical significance
    # If the current row's ydist is significantly different from the 
    # distribution of seen_ydists, we flag it as a shift.
    # We use a simple threshold check: if the current ydist is outside 
    # the range of the seen data (e.g., > 2 standard deviations), it's a shift.
    mu = sum(seen_ydists) / len(seen_ydists)
    sd = (sum((x - mu)**2 for x in seen_ydists) / len(seen_ydists))**0.5
    
    # Flag if the row is a significant outlier (e.g., > 3 sigma)
    return abs(current_ydist - mu) > (3 * sd)

if __name__ == "__main__":
    filename = "WA_Fn-UseC_-HR-Employee-Attrition.n1000.train.csv"
    data = list(ezr.csv(filename))
    header = data[0]
    rows = data[1:]
    
    seen = []
    shifts = 0
    
    for row in rows:
        if flow(header, row, seen):
            shifts += 1
        seen.append(row)
        
    print(f"Analysis of {len(rows)} records for team stability.")
    print(f"Detected {shifts} instances where the team's performance metrics "
          "shifted significantly from historical norms.")
    print("Decision logic: A shift is flagged when a new record's goal-distance "
          "is statistically distant (>3 sigma) from the historical distribution.")
    
    exit(0)
