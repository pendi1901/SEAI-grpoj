import ezr
import random

def flow(header, row, seen):
    """
    Decide if the current row is worth a 'what would keep you here' conversation.
    We use the 'acquire' logic: we want to identify rows that are 'best' 
    (closest to the ideal goals) compared to the 'seen' population.
    """
    # If this is the first row, we have no context, so let it go.
    if not seen:
        return False
    
    # Create a table of what we have seen so far to establish a baseline
    tbl = ezr.Tbl([header] + seen)
    
    # Calculate the distance of the current row from the ideal goals
    # We need to handle the fact that the current row might have missing goals
    # (though the prompt implies we evaluate the row as it arrives).
    # We use ydist to see how 'good' this row is relative to the current table.
    current_ydist = ezr.ydist(tbl, row)
    
    # Calculate the average ydist of the 'seen' rows
    seen_ydists = [ezr.ydist(tbl, r) for r in seen]
    avg_ydist = sum(seen_ydists) / len(seen_ydists)
    
    # If the current row is significantly better (lower ydist) than the average,
    # it's a candidate for a retention conversation.
    # We use a simple threshold: if it's in the top 10% of what we've seen.
    threshold = sorted(seen_ydists)[max(0, int(len(seen_ydists) * 0.1))]
    
    return current_ydist <= threshold

if __name__ == "__main__":
    # Load the data
    filename = "WA_Fn-UseC_-HR-Employee-Attrition.n1000.train.csv"
    data = list(ezr.csv(filename))
    header = data[0]
    rows = data[1:]
    
    # Simulate the arrival of rows
    random.seed(1234567891)
    random.shuffle(rows)
    
    seen = []
    selected_count = 0
    
    # Process rows one by one
    for row in rows:
        if flow(header, row, seen):
            selected_count += 1
        seen.append(row)
    
    print("Retention Analysis: 'What would keep you here' conversation trigger.")
    print(f"Processed {len(rows)} employee records.")
    print(f"Identified {selected_count} employees for retention outreach.")
    print("-" * 60)
    print("The model identifies high-value employees by comparing their")
    print("performance metrics against the historical baseline of the team.")
    
    exit(0)
