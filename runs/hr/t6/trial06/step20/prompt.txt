def say(x: Any, p: int = 2) -> str:
  "Floats get P decimals; dicts list their keys."
  if isinstance(x,float): x= f"{x:.{p}f}".rstrip("0").rstrip(".")
  if isinstance(x, dict):
    x= "{"+", ".join(f"{k}: {say(v,p)}" for k,v in x.items())+"}"
  return str(x)

class o(dict):
  "A dict you can poke with a dot."
  __getattr__, __setattr__ = dict.__getitem__, dict.__setitem__
  __repr__ = say

def atom(s: str) -> ATOM:
  "'22' -> 22. '3.1' -> 3.1. 'True' -> True. 'x' -> 'x'."
  for fn in (int, float):
    try: return fn(s)
    except ValueError: pass
  s = s.strip()
  return {"True": True, "False": False}.get(s, s)

def csv(file: str) -> Iterator[ROW]:

  with open(os.path.expanduser(file),  # -sig drops any BOM
           encoding="utf-8-sig") as f:

def Col(txt: str = " ", at: int = 0) -> COL:
  "Factory that returns NUM or SYM."
  return Num(txt,at) if txt[0].isupper() else Sym(txt,at)

class Num:
  "Place to summarize stream of numbers."
  __slots__ = ("at", "txt", "n", "mu", "m2", "sd", "goal")
  def __init__(self, txt: str = " ", at: int = 0) -> None:
    self.at, self.txt, self.n = at, txt, 0
    self.mu = self.m2 = self.sd = 0
    self.goal = 0 if txt[-1] == "-" else 1
  __repr__ = lambda self: say({k: getattr(self,k)
                               for k in self.__slots__})

class Sym:
  "Place to summarize stream of Symbols.  The class is the tag."
  __slots__ = ("at", "txt", "n", "has")
  def __init__(self, txt: str = " ", at: int = 0) -> None:
    self.at, self.txt, self.n, self.has = at, txt, 0, {}
  __repr__ = lambda self: say({k: getattr(self,k)
                               for k in self.__slots__})

def add(col: COL, v: ATOM, inc: int = 1) -> None:
  "Show V to COL.  INC=-1 takes it away.  `?` changes nothing."
  if v != "?":
    col.n += inc
    if type(col) is Sym:
      col.has[v] = col.has.get(v, 0) + inc
    else:
      d = v - col.mu                      # Welford, for `div`
      col.mu += inc * d / max(1, col.n)
      col.m2  = max(0, col.m2 + inc * d * (v - col.mu))
      col.sd  = 0 if col.n < 2 else (col.m2/(col.n-1))**.5

def mid(col: COL) -> ATOM:
  "Middle: the mean, or the most common symbol; ? if none."
  return (max(col.has, key=col.has.get) if col.has else "?"
          ) if type(col) is Sym else col.mu

def mids(tbl: TBL) -> MIDS:
  "Every column's middle, cached until the next addCols."
  tbl.mids = tbl.mids or {at: mid(c) for at,c in tbl.cols.items()}
  return tbl.mids

def div(col: COL) -> float:
  "Spread: entropy, or standard deviation."
  return (-sum(v/col.n*log2(v/col.n) for v in col.has.values())
          if type(col) is Sym else col.sd)  # sd kept by `add`

def norm(col: COL, v: ATOM) -> ATOM:
  "To 0..1, by the logistic curve.  Symbols do not scale."
  if type(col) is Sym: return v
  z = max(-3, min(3, (v - col.mu) / (1e-32 + div(col))))
  return 1 / (1 + exp(-1.7 * z))

#-- tables -------------------------------------------------
def Tbl(src: Iterable[ROW]) -> TBL:
  "Header names the columns: X skip, +- goal, ! klass."
  src = iter(src)
  tbl = Cols(o(rows=[], cols={}, x=[], y=[], mids=None,
               names=next(src)))
  for row in src: addRow(tbl, row)
  return tbl

def Cols(tbl: TBL) -> TBL:
  "Create column roles."
  for at, s in enumerate(tbl.names):
    if s[-1] == "X": continue            # skip me entirely
    tbl.cols[at] = Col(s,at)
    (tbl.y if s[-1] in "+-!" else tbl.x).append(tbl.cols[at])
  return tbl

def addRow(tbl: TBL, row: ROW) -> ROW:
  "Keep ROW, and show it to my columns."
  tbl.rows += [row]
  return addCols(tbl, row)

def addCols(tbl: TBL, row: ROW, inc: int = 1) -> ROW:
  "Show ROW to my columns, without keeping it."
  tbl.mids = None                 # a new row moves the middles
  for at, col in tbl.cols.items(): add(col, row[at], inc)
  return row

def clone(tbl: TBL, rows: ROWS = None) -> TBL:
  "An empty copy of TBL, plus ROWS."
  return Tbl([tbl.names] + (rows or []))

def dist(vs: Iterable[QTY], n: int) -> float:
  "Root mean square of N gaps."
  return sqrt(sum(v*v for v in vs) / n)

def ydist(tbl: TBL, row: ROW) -> float:
  "How far ROW's goals are from the best they could be."
  return dist((abs(norm(c, row[c.at]) - c.goal)
                    for c in tbl.y), len(tbl.y))

def gap(col: COL, a: ATOM, b: ATOM) -> QTY:
  "Distance between two values of one column."
  if a == "?" or b == "?": return 1    # unknown = far
  return a!=b if type(col) is Sym else abs(norm(col,a)-norm(col,b))

def xdist(tbl: TBL, r1: ROW, r2: ROW | MIDS) -> float:
  "How far apart two rows are, over the x columns."
  return dist((gap(c, r1[c.at], r2[c.at])
                    for c in tbl.x), len(tbl.x))

def oracle(row: ROW) -> ROW:
  """Labeller. Rows here arrive with ys, so nothing to do.
  Real oracles fill in any missing y cells (and can close
  over a TBL to know which cells those are)."""
  return row

def acquire(tbl: TBL, cap: int | None = None,
            label: ORACLE = oracle) -> ROWS:
  "Label near the best rows, far from the rest."
  cap  = cap or the.Stop
  todo = random.sample(tbl.rows, len(tbl.rows))[:the.Few]
  both = clone(tbl, [label(todo.pop())
                     for _ in range(the.Start)])
  both.rows.sort(key=lambda r: ydist(both, r))
  n    = int(sqrt(the.Start))
  best,rest = clone(tbl,both.rows[:n]), clone(tbl,both.rows[n:])
  while todo and len(both.rows) < cap:
    cb, cr = mids(best), mids(rest)    # two centroids
    todo.sort(key=lambda z: xdist(tbl,z,cr) - xdist(tbl,z,cb))
    addRow(best, addRow(both, label(todo.pop())))
    best.rows.sort(key=lambda z: ydist(both, z))  # no y leak
    if len(best.rows) > sqrt(len(both.rows)): # best stays small
      addRow(rest, addCols(best, best.rows.pop(), -1))
  return both.rows

def grabs(tbl: TBL, cap: int | None = None,
          label: ORACLE = oracle) -> ROWS:
  "The straw man: same budget, taken at random."
  cap = cap or the.Stop
  return [label(r) for r in
          random.sample(tbl.rows, min(cap, len(tbl.rows)))]

def cut(tbl: TBL, rows: ROWS) -> tuple[int,ATOM,GO]|None:
  "The (at, v, go) whose two sides have the tightest ys."
  ys = [ydist(tbl, r) for r in rows]
  out, least = None, 1e30
  for col in tbl.x:
    for v, go in candidates(col, rows):
      a, b = Num(), Num()   # one pass, so one `go` per row
      for r,y in zip(rows,ys): add(a if go(r) else b, y)
      if a.n >= the.Leaf and b.n >= the.Leaf:
        score = (div(a)*a.n + div(b)*b.n) / (a.n + b.n)
        if score < least: out, least = (col.at, v, go), score
  return out

def candidates(col: COL, rows: ROWS) -> Iterator[tuple[ATOM,GO]]:
  "What splits to try: one per symbol, or CUTS per num."
  at = col.at
  if type(col) is Sym:
    for v in sorted({r[at] for r in rows if r[at] != "?"}):
      yield v, lambda r,v=v,at=at: r[at] == v
  else:                # by rank, not value: THESE rows are dense
    xs = sorted(r[at] for r in rows if r[at] != "?")
    k  = max(1, len(xs)//the.Cuts)     # cuts, spaced by rank
    for v in xs[k-1:-1:k]:
      yield v, (lambda r,v=v,at=at: r[at] != "?" and r[at] <= v)

def tree(tbl: TBL, rows: ROWS) -> NODE:
  "Split while a split keeps both sides big enough."
  node = o(rows=rows, at=None, v=None, go=None, kids=[],
           mu=sum(ydist(tbl,r) for r in rows)/len(rows))
  if len(rows) > the.Leaf and (found := cut(tbl, rows)):
    at, v, go = found
    yes, no = [], []
    for r in rows: (yes if go(r) else no).append(r)
    node.at, node.v, node.go = at, v, go # cut kepts sides big
    node.kids = [tree(tbl, yes), tree(tbl, no)]
  return node

def leaf(node: NODE, row: ROW) -> NODE:
  "Walk ROW down to its leaf."
  while node.kids: node = node.kids[0 if node.go(row) else 1]
  return node

def leafs(node: NODE) -> list[NODE]:
  "Return leaves of this tree."
  return [x for k in node.kids for x in leafs(k)] or [node]

def show(tbl: TBL, node: NODE, pre: str | None = None,
         edge: str = "") -> None:
  "Tree on the left; ydist and n on the right."
  ls = sorted(leafs(node), key=lambda z: z.mu)
  def walk(z: NODE, pre: str | None, edge: str) -> None:
    m = "+" if z is ls[0] else "-" if z is ls[-1] else " "
    print(f"{m} {round(100*z.mu):>4} {len(z.rows):>5}   "
          f"{(pre or '')+edge}".rstrip())
    op = "=" if type(tbl.cols[z.at or 0]) is Sym else "<="
    no = "!=" if op == "=" else ">"
    for kid, oper in zip(z.kids, [op, no]):
      walk(kid, "" if pre is None else pre + "|  ",
           f"{tbl.names[z.at]} {oper} {say(z.v)}")
  print(" ydist     n")
  walk(node, pre, edge)

# Calls core, acquire, and one RANKER -- tree's, by default.
def ranker(lab: TBL) -> Callable[[ROW], float]:
  "Grow a tree on LAB; score any row by its leaf's mean ydist."
  node = tree(lab, lab.rows)
  return lambda r: leaf(node, r).mu

def wins(tbl: TBL) -> Callable[[ROW], float]:
  "100 at the best row, 0 at an average one."
  ys = sorted(ydist(tbl, r) for r in tbl.rows)
  lo, avg = ys[0], sum(ys)/len(ys)
  return lambda r: max(-100, min(100, 100*(1 - (ydist(tbl,r)-lo)
                                            / (avg-lo+1e-32))))

def holdout(tbl: TBL, pick: PICKER = acquire,
            rank: RANKER = ranker,
            label: ORACLE = oracle) -> ROW:
  "Train on half; of CHECK guesses on the rest, pick best."
  rows  = random.sample(tbl.rows, len(tbl.rows))
  n     = len(rows)//2
  tr    = clone(tbl, rows[:n][:the.Few])
  lab   = clone(tbl, pick(tr, the.Stop - the.Check, label))
  top   = sorted(rows[n:], key=rank(lab))
  return min(top[:the.Check], key=lambda r: ydist(lab, label(r)))

# A leaf: it calls core, and nothing here calls it.
def adds(vs: Iterable[ATOM], col: COL | None = None) -> COL:
  "Every item of VS into COL."
  col = Num() if col is None else col
  for v in vs: add(col, v)
  return col

def cohen(xs: NUMS, ys: NUMS, d: float = 0.35,
          eps: float = 0) -> bool:
  "Are the means within D pooled standard deviations?"
  a, b = adds(xs), adds(ys)
  pool = ((a.n-1)*div(a)**2 + (b.n-1)*div(b)**2)/(a.n+b.n-2)
  return abs(a.mu - b.mu) <= max(eps, d * sqrt(pool))

def cliffs(xs: NUMS, ys: NUMS, d: float = 0.197) -> bool:
  "Sorted in.  Is the rank imbalance small enough?"
  gt = lt = j = k = 0
  for x in xs:
    while j < len(ys) and ys[j] <  x: j += 1; k = j
    while k < len(ys) and ys[k] <= x: k += 1
    gt += j; lt += len(ys) - k
  return abs(gt - lt) / (len(xs) * len(ys)) <= d

def ks(xs: NUMS, ys: NUMS, a: float = 1.36) -> bool:
  "Sorted in.  95% Kolmogorov-Smirnov."
  n, m, i, j, d = len(xs), len(ys), 0, 0, 0
  while i < n and j < m:
    v = min(xs[i], ys[j])
    while i < n and xs[i] <= v: i += 1
    while j < m and ys[j] <= v: j += 1
    d = max(d, abs(i/n - j/m))
  return d <= a * sqrt((n + m) / (n * m))

def same(xs: NUMS, ys: NUMS, eps: float = 0) -> bool:
  """Indistinguishable.  Always cliffs and ks; EPS adds cohen, and
  doubles as its effect size (eps=0 skips that third test)."""
  xs, ys = sorted(xs), sorted(ys)
  return (cliffs(xs,ys) and ks(xs,ys)
          and (not eps or cohen(xs,ys,d=eps)))

def remember(header, rows):
    """
    Finds the best rows based on the goals defined in the header.
    - Columns ending in '-' are minimized.

    return scored[:limit]

import ezr
import random

def guess(header, train, test):
    """
    Estimates the value of the first goal column for test rows.
    Uses a tree-based approach: train a tree on the labeled data,
    then predict the mean of the leaf node for each test row.
    """
    # Create a table object from the training data
    # ezr.Tbl handles the logic of identifying goals (+/-)
    tbl = ezr.Tbl([header] + train)
    
    # Build a decision tree using the training data
    # The tree splits based on x-columns to minimize y-distance
    root = ezr.tree(tbl, tbl.rows)
    
    # Identify the first goal column index
    # In this dataset, the first goal is 'YearsAtCompany+'
    goal_col = tbl.y[0]
    
    results = []
    for row in test:
        # Walk the tree to find the leaf node for this test row
        node = ezr.leaf(root, row)
        
        # Calculate the mean of the goal column for all rows in this leaf
        # We use the column index from the training table
        vals = [r[goal_col.at] for r in node.rows]
        mean_val = sum(vals) / len(vals)
        results.append(mean_val)
        
    return results

import ezr
import random

def flow(header, row, seen):
    """
    Decide if the current row is worth a 'retention' conversation.
    We compare the current row against the 'seen' population.
    If the row is in the top 5% of the current distribution (based on ydist),
    we flag it for a conversation (True).
    """
    if not seen:
        return False
    
    # Build a table from the seen data to calculate the current distribution
    # We use the header to ensure goals (+/-) are correctly identified
    tbl = ezr.Tbl([header] + seen)
    
    # Calculate how far the current row is from the ideal goals
    # ydist returns 0 for perfect, higher for worse
    current_ydist = ezr.ydist(tbl, row)
    
    # Calculate the ydist for all seen rows
    seen_ydists = [ezr.ydist(tbl, r) for r in seen]
    
    # If the current row is better than 95% of what we have seen, 
    # it is a high-retention-risk/high-value candidate.
    # We define "better" as having a lower ydist (closer to ideal).
    threshold = sorted(seen_ydists)[int(len(seen_ydists) * 0.05)]
    
    return current_ydist <= threshold

import ezr
import random

def blame(header, rows):
    """
    Identify input columns that best explain why the worst rows are bad.
    We define 'bad' as having a high ydist (far from goals).
    We compare the distribution of values in the worst 25% of rows 
    against the best 25% of rows for each input column.
    """
    tbl = ezr.Tbl([header] + rows)
    
    # Sort rows by ydist (distance to ideal goals)
    # Higher ydist means worse performance
    sorted_rows = sorted(rows, key=lambda r: ezr.ydist(tbl, r))
    
    n = len(sorted_rows) // 4
    best = sorted_rows[:n]
    worst = sorted_rows[-n:]
    
    scores = []
    for col in tbl.x:
        # Extract values for this column from best and worst groups
        b_vals = [r[col.at] for r in best if r[col.at] != "?"]
        w_vals = [r[col.at] for r in worst if r[col.at] != "?"]
        
        if not b_vals or not w_vals:
            continue
            
        # If distributions are significantly different (not same), 
        # this column is a candidate for explaining the difference.
        if not ezr.same(b_vals, w_vals):
            # Simple heuristic: difference in means
            b_mid = ezr.mid(ezr.adds(b_vals))
            w_mid = ezr.mid(ezr.adds(w_vals))
            
            # For numeric, use absolute difference; for symbolic, use 1 if different
            diff = abs(b_mid - w_mid) if isinstance(b_mid, (int, float)) else (1 if b_mid != w_mid else 0)
            scores.append((header[col.at], diff))
            
    # Return names of columns with the highest impact
    scores.sort(key=lambda x: x[1], reverse=True)
    return [name for name, score in scores[:3]]

    """

    rank = ezr.ranker(tbl)

    predicted_ydist = rank(row)
    
    # Calculate p: fraction of rows that are worse (have higher ydist) than our prediction
    # A row is better if its ydist is lower.
    worse_count = sum(1 for y in all_ydists if y > predicted_ydist)
    p = worse_count / len(rows)

    # We look at the node's path to explain the ranking
    node = ezr.leaf(ezr.tree(tbl, rows), row)
    
    # Create a simple reason string
    reason = (f"This arrangement ranks in the top {int((1-p)*100)}% of all teams "
              f"because it aligns with patterns found in high-retention groups.")

import ezr
import random

def choose(header, rows, label):
    """
    Finds the best row by using the 'acquire' strategy from ezr.
    We label up to 50 rows, then return the index of the best one found.
    """
    # Create a table object to manage the data.
    # We must ensure the rows passed to Tbl have the same length as the header.
    # The harness passes rows with '?' for goals, which is correct.
    tbl = ezr.Tbl([header] + rows)
    
    # Track the best row found during the labeling process
    best_idx = -1
    best_ydist = 1e30
    
    # The oracle needs to return the full row (with goals) for a given index.
    # We map the internal row object back to its original index.
    def oracle_wrapper(row):
        # Find the index of this row in the current table
        # We use the fact that rows are objects in the table
        for i, r in enumerate(tbl.rows):
            if r is row:
                full_row = label(i)
                # Update the table row with the real values so ydist works
                tbl.rows[i] = full_row
                # Update the best found so far
                d = ezr.ydist(tbl, full_row)
                nonlocal best_idx, best_ydist
                if d < best_ydist:
                    best_ydist = d
                    best_idx = i
                return full_row
        return row

    # Use ezr.acquire to select and label the most informative rows.
    # This respects the budget (the.Stop=50).
    ezr.acquire(tbl, cap=50, label=oracle_wrapper)
    
    return best_idx

            d = ezr.ydist(tbl, new_row)
            
            # If this change improves the ydist, record it
            if d < best_ydist:
                best_ydist = d
                best_change = {col_name: val}

import ezr

def spot(header, rows):
    """
    Finds rows that look wrong (outliers).
    We use the distance to the nearest neighbor (xdist) as a measure of 
    suspicion. Rows that are very far from all other rows in the 
    input space (x-columns) are likely to be data entry errors.
    """
    tbl = ezr.Tbl([header] + rows)
    
    # Calculate the average distance of each row to its k-nearest neighbors.
    # A high average distance indicates the row is an outlier.
    k = 5
    scores = []
    for i, r1 in enumerate(rows):
        # Calculate distance to all other rows
        distances = [ezr.xdist(tbl, r1, r2) for j, r2 in enumerate(rows) if i != j]
        # Sort and take the average of the k closest neighbors
        distances.sort()
        avg_dist = sum(distances[:k]) / k
        scores.append((i, avg_dist))
    
    # Sort by suspicion (highest distance first)
    scores.sort(key=lambda x: x[1], reverse=True)
    
    # Return the indices of the most suspicious rows
    return [i for i, score in scores]

import ezr
import random

def race(header, rows, label):
    """
    Compare three strategies: random, active learning (acquire), 
    and a greedy search based on distance to the best known.
    """
    budget = 50
    labeled_indices = set()
    
    # We need a table that knows the true goal values to evaluate ydist.
    # Since we only have access to the label function, we build a 
    # 'knowledge' table as we go.
    full_tbl = ezr.Tbl([header])
    
    def get_label(i):
        if i not in labeled_indices:
            if len(labeled_indices) >= budget:
                raise Exception("Budget exceeded")
            labeled_indices.add(i)
            real_row = label(i)
            ezr.addRow(full_tbl, real_row)
        return label(i)

    # Strategy 1: Random
    def strategy_random(tbl):
        indices = random.sample(range(len(rows)), budget // 3)
        for i in indices: get_label(i)
        # Evaluate based on the knowledge gained
        return min(indices, key=lambda i: ezr.ydist(full_tbl, label(i)))

    # Strategy 2: Active Learning (ezr.acquire)
    def strategy_acquire(tbl):
        def oracle(row):
            # Find index of this row in the input rows
            for i, r in enumerate(rows):
                if r is row: return get_label(i)
            return row
        ezr.acquire(tbl, cap=budget // 3, label=oracle)
        return min(list(labeled_indices), key=lambda i: ezr.ydist(full_tbl, label(i)))

    # Strategy 3: Greedy
    def strategy_greedy(tbl):
        curr = random.choice(range(len(rows)))
        for _ in range(budget // 3):
            get_label(curr)
            best_n, best_d = -1, 1e30
            for i in range(len(rows)):
                if i not in labeled_indices:
                    d = ezr.xdist(tbl, rows[curr], rows[i])
                    if d < best_d: best_d, best_n = d, i
            if best_n == -1: break
            curr = best_n
        return min(list(labeled_indices), key=lambda i: ezr.ydist(full_tbl, label(i)))

    tbl = ezr.Tbl([header] + rows)
    results = {
        "random": strategy_random(tbl),
        "acquire": strategy_acquire(tbl),
        "greedy": strategy_greedy(tbl)
    }
    
    winner = min(results, key=lambda k: ezr.ydist(full_tbl, label(results[k])))
    
    return {
        "winner": winner,
        "row": results[winner],
        "tried": list(results.keys())
    }

def flow(header, row, seen):
    """
    Returns True if the current row is an outlier compared to the 'seen' 
    population, indicating the team no longer looks the way it normally does.
    We use the statistical 'same' test (cliffs and ks) on the ydist 
    distribution to detect a significant shift in team performance.
    """
    if len(seen) < 20:
        return False
    
    # Create a table for the seen population to define the "normal" state
    tbl_seen = ezr.Tbl([header] + seen)
    
    # Calculate ydist for all seen rows
    seen_ydists = [ezr.ydist(tbl_seen, r) for r in seen]
    
    # Create a temporary table for the current row to calculate its ydist
    # We use the same column definitions as the seen table
    current_ydist = ezr.ydist(tbl_seen, row)

    # from the distribution of the seen population, flag it.

import ezr
import step05

def blame(header, rows):
    """
    Identifies input columns that have 'nothing to do' with performance.
    We define 'nothing to do' as columns where the distribution of values
    in the best 25% of rows is statistically indistinguishable from the
    distribution in the worst 25% of rows.
    """
    tbl = ezr.Tbl([header] + rows)
    
    # Sort rows by ydist (distance to ideal goals)
    # Higher ydist means worse performance
    sorted_rows = sorted(rows, key=lambda r: ezr.ydist(tbl, r))
    
    n = len(sorted_rows) // 4
    best = sorted_rows[:n]
    worst = sorted_rows[-n:]
    
    irrelevant = []
    for col in tbl.x:
        # Extract values for this column from best and worst groups
        b_vals = [r[col.at] for r in best if r[col.at] != "?"]
        w_vals = [r[col.at] for r in worst if r[col.at] != "?"]
        
        if not b_vals or not w_vals:
            continue
            
        # If distributions are indistinguishable (same), 
        # this column has little impact on the performance difference.
        if ezr.same(b_vals, w_vals):
            irrelevant.append(header[col.at])
            
    return irrelevant

import ezr
import step05

def justify(header, rows, row):
    """
    Returns (p, reason): p is the fraction of rows that 'row' is better than.
    The reason explains why paying everyone more is not a simple solution
    by highlighting that other factors (like job role or tenure) are 
    statistically significant drivers of retention.
    """
    tbl = ezr.Tbl([header] + rows)
    
    # Predict the ydist for the target row using a tree-based ranker
    rank = ezr.ranker(tbl)
    predicted_ydist = rank(row)
    
    # Calculate p: fraction of rows that are worse (higher ydist) than our prediction
    all_ydists = [ezr.ydist(tbl, r) for r in rows]
    worse_count = sum(1 for y in all_ydists if y > predicted_ydist)
    p = worse_count / len(rows)
    
    # Identify top factors that actually drive performance (blame)
    # If MonthlyIncome is not the primary driver, we use that to justify the answer.
    top_factors = step05.blame(header, rows)
    
    # Construct the reason
    primary = top_factors[0] if top_factors else "various organizational factors"
    reason = (f"While salary is important, our analysis shows that {primary} "
              f"is a more significant driver of retention than income alone.")
    
    return p, reason

import ezr

def choose(header, rows, label):
    """
    Finds the best row for 'Laboratory Technician' staff using the 
    active learning 'acquire' strategy, limited to 50 labels.
    """
    # Filter rows for Laboratory Technician
    job_role_idx = header.index('jobRole')
    
    # Identify indices of lab staff
    lab_indices = [i for i, r in enumerate(rows) if r[job_role_idx] == 'Laboratory Technician']
    lab_rows = [rows[i] for i in lab_indices]
    
    # Create a table for the lab staff
    # We must ensure the table is initialized with the header so that 
    # ezr.ydist can correctly identify the goal columns.
    tbl = ezr.Tbl([header] + lab_rows)
    
    best_idx = -1
    best_ydist = 1e30
    
    # The oracle needs to return the full row for a given index
    def oracle_wrapper(row):
        # Find which original index this row corresponds to
        # We match by object identity as ezr.acquire maintains the row objects
        for i, r in enumerate(tbl.rows):
            if r is row:
                orig_i = lab_indices[i]
                full_row = label(orig_i)
                # Update the table row with real values so ydist works
                tbl.rows[i] = full_row
                # Update best found
                d = ezr.ydist(tbl, full_row)
                nonlocal best_idx, best_ydist
                if d < best_ydist:
                    best_ydist = d
                    best_idx = orig_i
                return full_row
        return row

    # Use ezr.acquire to select and label the most informative lab rows
    # The cap is 50 as per the requirement.
    ezr.acquire(tbl, cap=50, label=oracle_wrapper)
    
    return best_idx

import ezr

def fix(header, row, levers):
    """
    Suggests a change to a 'bad' row to improve its goal performance,
    ensuring we only use the provided levers and stay within bounds.
    """
    # Create a table to establish the goal context (min/max)
    tbl = ezr.Tbl([header])
    
    best_change = {}
    # Current distance to ideal goals
    best_ydist = ezr.ydist(tbl, row)
    
    # Try each lever
    for col_name, allowed in levers.items():
        if col_name not in header:
            continue
        idx = header.index(col_name)
        original_val = row[idx]
        
        # Generate candidates based on lever type
        candidates = []
        if isinstance(allowed, list):  # Symbolic
            candidates = [v for v in allowed if v != original_val]
        elif isinstance(allowed, tuple):  # Numeric (low, high)
            low, high = allowed
            # Try boundaries and midpoint, ensuring we stay within [low, high]
            candidates = [v for v in [low, (low + high) / 2, high] if v != original_val]
            
        for val in candidates:
            # Create a hypothetical row
            new_row = list(row)
            new_row[idx] = val
            
            # Calculate new ydist
            d = ezr.ydist(tbl, new_row)
            
            # If this change improves the ydist, record it
            if d < best_ydist:
                best_ydist = d
                best_change = {col_name: val}
                
    return best_change
