import ezr
import random

def certify(xs, ys):
    """
    Returns True if the two distributions are statistically indistinguishable 
    (noise), False if they are significantly different.
    """
    # ezr.same uses cliffs and ks tests to compare two distributions
    return ezr.same(xs, ys)

def get_tenure_by_overtime(header, rows):
    """
    Splits the data into two groups: those who work overtime and those who don't.
    Returns the 'YearsAtCompany+' values for both groups.
    """
    overtime_idx = header.index('overTime')
    tenure_idx = header.index('YearsAtCompany+')
    
    yes_overtime = []
    no_overtime = []
    
    for row in rows:
        val = row[tenure_idx]
        if isinstance(val, (int, float)):
            if row[overtime_idx] == 'Yes':
                yes_overtime.append(val)
            else:
                no_overtime.append(val)
                
    return yes_overtime, no_overtime

if __name__ == "__main__":
    file_path = "WA_Fn-UseC_-HR-Employee-Attrition.n1000.train.csv"
    data = list(ezr.csv(file_path))
    header = data[0]
    rows = data[1:]
    
    yes_group, no_group = get_tenure_by_overtime(header, rows)
    
    is_noise = certify(yes_group, no_group)
    
    print("Analysis: Does Overtime affect tenure (YearsAtCompany+)?")
    print(f"Average tenure (Overtime=Yes): {sum(yes_group)/len(yes_group):.2f} years")
    print(f"Average tenure (Overtime=No) : {sum(no_group)/len(no_group):.2f} years")
    
    if is_noise:
        print("Result: The difference is statistically insignificant (noise).")
    else:
        print("Result: The difference is statistically significant (real).")
