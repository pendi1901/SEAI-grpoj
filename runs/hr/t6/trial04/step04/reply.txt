import ezr
import random

def flow(header, row, seen):
    """
    Decide if the current row is 'optimal' enough to trigger a retention 
    conversation. We compare the current row's ydist against the 
    distribution of ydist values seen so far. If it is in the top 5% 
    of observed quality, we flag it.
    """
    # If this is the first row, we can't compare yet.
    if not seen:
        return False
    
    # Create a table to calculate ydist
    tbl = ezr.Tbl([header] + seen)
    
    # Calculate ydist for all seen rows
    seen_ydists = [ezr.ydist(tbl, r) for r in seen]
    
    # Calculate ydist for current row
    current_ydist = ezr.ydist(tbl, row)
    
    # We want to minimize ydist (closer to goals).
    # If current row is better than 95% of what we have seen, flag it.
    threshold = sorted(seen_ydists)[int(len(seen_ydists) * 0.05)]
    
    return current_ydist <= threshold

if __name__ == "__main__":
    # Load data
    filename = "WA_Fn-UseC_-HR-Employee-Attrition.n1000.train.csv"
    data = list(ezr.csv(filename))
    header = data[0]
    rows = data[1:]
    
    # Simulate arrival
    random.seed(1234567891)
    random.shuffle(rows)
    
    seen = []
    flags = 0
    for row in rows:
        if flow(header, row, seen):
            flags += 1
        seen.append(row)
    
    print(f"Processed {len(rows)} employee records.")
    print(f"Flagged {flags} employees for 'what would keep you here' conversations.")
    print("Decision logic: Flagged employees whose performance metrics are in the top 5% of observed history.")
