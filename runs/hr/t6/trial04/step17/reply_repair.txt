import ezr
import step07

def choose(header, rows, label):
    """
    Finds the best arrangement for 'Laboratory Technician' staff only,
    using the active learning search within a 50-label budget.
    """
    # Filter rows for Laboratory Technician
    idx_role = header.index("jobRole")
    lab_indices = [i for i, r in enumerate(rows) if r[idx_role] == "Laboratory Technician"]
    
    # If no lab staff, return first row
    if not lab_indices:
        return 0
    
    # Create a subset of rows for the search
    lab_rows = [rows[i] for i in lab_indices]
    
    # We need a wrapper for the label function to map back to global indices.
    # The harness expects us to return the index in the original 'rows' list.
    # step07.choose expects a label function that takes an index and returns a full row.
    def lab_label(i):
        # i is the index in the lab_rows list
        global_idx = lab_indices[i]
        return label(global_idx)
    
    # We cannot use step07.choose directly because it assumes the label function 
    # takes an index into the provided rows list, but our label function 
    # needs to map back to the original global indices.
    # We implement the logic here manually using ezr.acquire.
    
    tbl = ezr.Tbl([header] + lab_rows)
    
    # Oracle that updates the table's internal column statistics
    def oracle(row):
        # Find the index of this row in our lab_rows subset
        # We use the row object identity or content to find the index
        # Since rows are lists, we find the index in lab_rows
        idx = lab_rows.index(row)
        true_row = lab_label(idx)
        
        # Update the row in place with the true values
        for i, val in enumerate(true_row):
            row[i] = val
            
        # Update the table's column statistics with the new information
        ezr.addCols(tbl, true_row, 1)
        return row

    # Use ezr.acquire to perform the active learning search.
    labeled_rows = ezr.acquire(tbl, cap=50, label=oracle)
    
    # Find the best row among the labeled ones based on ydist
    best_row = min(labeled_rows, key=lambda r: ezr.ydist(tbl, r))
    
    # Return the global index
    return lab_indices[lab_rows.index(best_row)]

if __name__ == "__main__":
    filename = "WA_Fn-UseC_-HR-Employee-Attrition.n1000.train.csv"
    data = list(ezr.csv(filename))
    header = data[0]
    rows = data[1:]
    
    # Mask the goal columns
    goal_indices = [i for i, h in enumerate(header) if h.endswith('+') or h.endswith('-')]
    masked_rows = []
    for row in rows:
        new_row = list(row)
        for idx in goal_indices:
            new_row[idx] = "?"
        masked_rows.append(new_row)
        
    # Define the label function for the harness
    def label_func(i):
        return rows[i]
    
    # Run the selection
    best_idx = choose(header, masked_rows, label_func)
    
    print(f"Best laboratory staff arrangement found at index: {best_idx}")
    print("This row represents the optimal balance of goals for laboratory staff "
          "based on targeted interviews.")
