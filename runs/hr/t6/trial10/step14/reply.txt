import ezr
import step09

def flow(header, row, seen):
    """
    Detects if a team's composition has drifted.
    We compare the incoming row to the centroid of the 'seen' population.
    If the row is an outlier (high xdist from the centroid of seen rows),
    it suggests the team no longer looks like it normally does.
    """
    if len(seen) < 10:
        return False
    
    # Create a table of the history to calculate the current norm
    tbl = ezr.Tbl([header] + seen)
    centroid = ezr.mids(tbl)
    
    # Calculate how far the new row is from the current norm
    d = ezr.xdist(tbl, row, centroid)
    
    # Calculate the distribution of distances in the history to find a threshold
    # We flag the row if it is in the top 5% of distance (extreme outlier)
    distances = [ezr.xdist(tbl, r, centroid) for r in seen]
    distances.sort()
    threshold = distances[int(len(distances) * 0.95)]
    
    return d > threshold

if __name__ == "__main__":
    filename = "WA_Fn-UseC_-HR-Employee-Attrition.n1000.train.csv"
    data = list(ezr.csv(filename))
    header = data[0]
    rows = data[1:]
    
    seen = []
    drift_count = 0
    
    for row in rows:
        if flow(header, row, seen):
            drift_count += 1
        seen.append(row)
        
    print("Analysis: Detecting team composition drift.")
    print(f"Processed {len(rows)} records.")
    print(f"Flagged {drift_count} records where the team composition deviated significantly from the norm.")
    print("Logic: A record is flagged if its input features are in the top 5% of distance from the historical centroid.")
