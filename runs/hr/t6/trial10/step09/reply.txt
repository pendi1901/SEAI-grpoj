import ezr
import random

def spot(header, rows):
    """
    Finds rows that look 'wrong' by identifying outliers.
    A row is suspicious if its input values (x-columns) are far from 
    the centroid of the rest of the data.
    """
    tbl = ezr.Tbl([header] + rows)
    
    # Calculate the centroid (mids) of the entire dataset
    # We use the x-columns to define the 'normal' input space
    centroid = ezr.mids(tbl)
    
    # Calculate the distance of each row from the centroid
    # Rows with high xdist are outliers (potentially entered wrongly)
    scores = []
    for i, row in enumerate(rows):
        d = ezr.xdist(tbl, row, centroid)
        scores.append((d, i))
    
    # Sort by distance descending (most suspicious first)
    scores.sort(key=lambda x: x[0], reverse=True)
    
    # Return the indices of the top 10 most suspicious rows
    return [item[1] for item in scores[:10]]

if __name__ == "__main__":
    filename = "WA_Fn-UseC_-HR-Employee-Attrition.n1000.train.csv"
    data = list(ezr.csv(filename))
    header = data[0]
    rows = data[1:]
    
    suspicious_indices = spot(header, rows)
    
    print("Analysis: Identifying potentially erroneous data entries.")
    print("Logic: Flagging rows with input values furthest from the population centroid.")
    print(f"\nFound {len(suspicious_indices)} most suspicious records:")
    
    for rank, idx in enumerate(suspicious_indices):
        print(f"  Rank {rank+1}: Row index {idx}")
        
    print("\nThese rows contain input values that deviate significantly from the norm.")
