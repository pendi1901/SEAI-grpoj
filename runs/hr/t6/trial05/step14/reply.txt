import ezr
import random

def flow(header, row, seen):
    """
    Detects if a team's distribution has shifted significantly.
    We compare the current row to the distribution of 'seen' rows.
    If the current row is an outlier (high xdist from the centroid of 'seen'),
    it suggests the team composition is changing.
    """
    if len(seen) < 10:
        return False
    
    # Create a table from the history to calculate the centroid
    tbl = ezr.Tbl([header] + seen)
    centroid = ezr.mids(tbl)
    
    # Calculate how far the new row is from the current team's norm
    dist = ezr.xdist(tbl, row, centroid)
    
    # If the distance is significantly higher than the average distance 
    # of the seen rows, flag it as a shift in team composition.
    dists = [ezr.xdist(tbl, r, centroid) for r in seen]
    avg_dist = sum(dists) / len(dists)
    
    # Flag if the new row is an outlier (e.g., > 2x the average distance)
    return dist > (2.0 * avg_dist)

if __name__ == "__main__":
    data = list(ezr.csv("WA_Fn-UseC_-HR-Employee-Attrition.n1000.train.csv"))
    header = data[0]
    rows = data[1:]
    
    # Simulate arrival of rows
    random.seed(1234567891)
    random.shuffle(rows)
    
    seen = []
    shifts = 0
    for row in rows:
        if flow(header, row, seen):
            shifts += 1
        seen.append(row)
        
    print("Analysis: Monitoring team composition shifts.")
    print(f"Processed {len(rows)} employee records.")
    print(f"Detected {shifts} instances where a new record significantly deviated")
    print("from the established team norms (potential team composition shift).")
    
    exit(0)
