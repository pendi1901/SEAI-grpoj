import ezr
import random

def flow(header, row, seen):
    """
    Detects if a row is an outlier compared to the 'seen' distribution.
    We use the xdist of the current row from the centroid of the 'seen' rows.
    If the distance is significantly higher than the average distance 
    observed in the 'seen' data, we flag it as a shift in team behavior.
    """
    if len(seen) < 10:
        return False
    
    # Create a table of seen rows to calculate the centroid
    tbl = ezr.Tbl([header] + seen)
    centroid = ezr.mids(tbl)
    
    # Calculate the current row's distance from the centroid
    current_dist = ezr.xdist(tbl, row, centroid)
    
    # Calculate the average distance of seen rows from the centroid
    # to establish a baseline of "normal" behavior
    seen_dists = [ezr.xdist(tbl, r, centroid) for r in seen]
    avg_seen_dist = sum(seen_dists) / len(seen_dists)
    
    # If the current row is more than 2 standard deviations away from the 
    # average distance, we flag it as a shift.
    # We use a simple threshold: 2.0 * average distance.
    return current_dist > (2.0 * avg_seen_dist)

if __name__ == "__main__":
    filename = "WA_Fn-UseC_-HR-Employee-Attrition.n1000.train.csv"
    data = list(ezr.csv(filename))
    header = data[0]
    rows = data[1:]
    
    # Simulate the arrival process
    random.seed(1234567891)
    random.shuffle(rows)
    
    seen = []
    shifts = 0
    
    for row in rows:
        if flow(header, row, seen):
            shifts += 1
        seen.append(row)
    
    print("Team Behavior Shift Analysis")
    print("-" * 40)
    print(f"Total records processed: {len(rows)}")
    print(f"Number of behavioral shifts detected: {shifts}")
    print("-" * 40)
    print("A 'shift' is detected when a new record deviates significantly")
    print("from the established patterns of the preceding team data.")
