import ezr
import step03

def flow(header, row, seen):
    """
    Detects if the current team (represented by 'seen') has drifted 
    significantly from the norm. We compare the distribution of 
    'YearsAtCompany+' (a key goal) in the current batch vs the 
    historical norm.
    """
    # Need a baseline to detect drift
    if len(seen) < 50:
        return False
    
    # Identify index for a key goal
    yac_idx = header.index('YearsAtCompany+')
    
    # Define 'norm' as the first 50 rows, 'current' as the last 50
    norm = [r[yac_idx] for r in seen[:50]]
    current = [r[yac_idx] for r in seen[-50:]]
    
    # If the distributions are statistically different, the team has drifted
    # We use step03's certify (which uses cliffs and ks tests)
    is_same = step03.certify(norm, current)
    
    # Return True if the team no longer looks like it used to (drift detected)
    return not is_same

if __name__ == "__main__":
    filename = "WA_Fn-UseC_-HR-Employee-Attrition.n1000.train.csv"
    data = list(ezr.csv(filename))
    header = data[0]
    rows = data[1:]
    
    seen = []
    drift_count = 0
    
    for row in rows:
        if flow(header, row, seen):
            drift_count += 1
        seen.append(row)
        
    print("Analysis: Detecting team drift over time.")
    print(f"Processed {len(rows)} employee records.")
    print(f"Number of times the team composition drifted significantly: {drift_count}.")
    print("Logic: Drift is detected when the distribution of 'YearsAtCompany+' "
          "in the recent window significantly deviates from the initial baseline.")
    
    exit(0)
