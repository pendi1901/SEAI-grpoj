import ezr
import random

def fix(header, row, levers):
    """
    Given a 'bad' row, recommend changes to levers to improve its ydist.
    We simulate the impact of changing each lever and pick the one that
    results in the lowest ydist.
    """
    # Create a dummy table to use ydist
    # We need a full table to calculate the norm/goal distances
    filename = "WA_Fn-UseC_-HR-Employee-Attrition.n1000.train.csv"
    data = list(ezr.csv(filename))
    tbl = ezr.Tbl(data)
    
    best_row = list(row)
    best_ydist = ezr.ydist(tbl, row)
    recommendation = {}

    for col_name, allowed in levers.items():
        idx = header.index(col_name)
        original_val = row[idx]
        
        # Try all allowed values
        candidates = allowed if isinstance(allowed, list) else [allowed[0], allowed[1]]
        
        for val in candidates:
            test_row = list(row)
            test_row[idx] = val
            current_ydist = ezr.ydist(tbl, test_row)
            
            if current_ydist < best_ydist:
                best_ydist = current_ydist
                recommendation = {col_name: val}
                
    return recommendation

if __name__ == "__main__":
    filename = "WA_Fn-UseC_-HR-Employee-Attrition.n1000.train.csv"
    data = list(ezr.csv(filename))
    header = data[0]
    rows = data[1:]
    
    # Define levers
    levers = {
        "businessTravel": ["Non-Travel", "Travel_Rarely", "Travel_Frequently"],
        "overTime": ["No", "Yes"],
        "TrainingTimesLastYear": (0, 6),
        "PercentSalaryHike": (11, 25),
        "StockOptionLevel": (0, 3)
    }
    
    # Pick a "bad" row (high ydist)
    tbl = ezr.Tbl(data)
    sorted_rows = sorted(rows, key=lambda r: ezr.ydist(tbl, r), reverse=True)
    bad_row = sorted_rows[0]
    
    # Get recommendation
    change = fix(header, bad_row, levers)
    
    print("Analysis: Recommendations for improving employee retention.")
    print(f"Bad row identified with ydist: {ezr.ydist(tbl, bad_row):.2f}")
    if change:
        col, val = list(change.items())[0]
        print(f"Recommended change: Set '{col}' to '{val}'.")
    else:
        print("No single lever change found to improve this specific record.")
    
    exit(0)
