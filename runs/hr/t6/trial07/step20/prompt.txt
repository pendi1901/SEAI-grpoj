def say(x: Any, p: int = 2) -> str:
  "Floats get P decimals; dicts list their keys."
  if isinstance(x,float): x= f"{x:.{p}f}".rstrip("0").rstrip(".")
  if isinstance(x, dict):
    x= "{"+", ".join(f"{k}: {say(v,p)}" for k,v in x.items())+"}"
  return str(x)

class o(dict):
  "A dict you can poke with a dot."
  __getattr__, __setattr__ = dict.__getitem__, dict.__setitem__
  __repr__ = say

def atom(s: str) -> ATOM:
  "'22' -> 22. '3.1' -> 3.1. 'True' -> True. 'x' -> 'x'."
  for fn in (int, float):
    try: return fn(s)
    except ValueError: pass
  s = s.strip()
  return {"True": True, "False": False}.get(s, s)

def csv(file: str) -> Iterator[ROW]:
  "ROWS of FILE, each cell coerced."
  with open(os.path.expanduser(file),  # -sig drops any BOM
           encoding="utf-8-sig") as f:
    for line in f:
      if line.strip(): yield [atom(s) for s in line.split(",")]

def Col(txt: str = " ", at: int = 0) -> COL:
  "Factory that returns NUM or SYM."
  return Num(txt,at) if txt[0].isupper() else Sym(txt,at)

class Num:
  "Place to summarize stream of numbers."
  __slots__ = ("at", "txt", "n", "mu", "m2", "sd", "goal")
  def __init__(self, txt: str = " ", at: int = 0) -> None:
    self.at, self.txt, self.n = at, txt, 0
    self.mu = self.m2 = self.sd = 0
    self.goal = 0 if txt[-1] == "-" else 1
  __repr__ = lambda self: say({k: getattr(self,k)
                               for k in self.__slots__})

class Sym:
  "Place to summarize stream of Symbols.  The class is the tag."
  __slots__ = ("at", "txt", "n", "has")
  def __init__(self, txt: str = " ", at: int = 0) -> None:
    self.at, self.txt, self.n, self.has = at, txt, 0, {}
  __repr__ = lambda self: say({k: getattr(self,k)
                               for k in self.__slots__})

def add(col: COL, v: ATOM, inc: int = 1) -> None:
  "Show V to COL.  INC=-1 takes it away.  `?` changes nothing."
  if v != "?":
    col.n += inc
    if type(col) is Sym:
      col.has[v] = col.has.get(v, 0) + inc
    else:
      d = v - col.mu                      # Welford, for `div`
      col.mu += inc * d / max(1, col.n)
      col.m2  = max(0, col.m2 + inc * d * (v - col.mu))
      col.sd  = 0 if col.n < 2 else (col.m2/(col.n-1))**.5

def mid(col: COL) -> ATOM:
  "Middle: the mean, or the most common symbol; ? if none."
  return (max(col.has, key=col.has.get) if col.has else "?"
          ) if type(col) is Sym else col.mu

def mids(tbl: TBL) -> MIDS:
  "Every column's middle, cached until the next addCols."
  tbl.mids = tbl.mids or {at: mid(c) for at,c in tbl.cols.items()}
  return tbl.mids

def div(col: COL) -> float:
  "Spread: entropy, or standard deviation."
  return (-sum(v/col.n*log2(v/col.n) for v in col.has.values())
          if type(col) is Sym else col.sd)  # sd kept by `add`

def norm(col: COL, v: ATOM) -> ATOM:
  "To 0..1, by the logistic curve.  Symbols do not scale."
  if type(col) is Sym: return v
  z = max(-3, min(3, (v - col.mu) / (1e-32 + div(col))))
  return 1 / (1 + exp(-1.7 * z))

#-- tables -------------------------------------------------
def Tbl(src: Iterable[ROW]) -> TBL:
  "Header names the columns: X skip, +- goal, ! klass."
  src = iter(src)
  tbl = Cols(o(rows=[], cols={}, x=[], y=[], mids=None,
               names=next(src)))
  for row in src: addRow(tbl, row)
  return tbl

def Cols(tbl: TBL) -> TBL:
  "Create column roles."
  for at, s in enumerate(tbl.names):
    if s[-1] == "X": continue            # skip me entirely
    tbl.cols[at] = Col(s,at)
    (tbl.y if s[-1] in "+-!" else tbl.x).append(tbl.cols[at])
  return tbl

def addRow(tbl: TBL, row: ROW) -> ROW:
  "Keep ROW, and show it to my columns."
  tbl.rows += [row]
  return addCols(tbl, row)

def addCols(tbl: TBL, row: ROW, inc: int = 1) -> ROW:
  "Show ROW to my columns, without keeping it."
  tbl.mids = None                 # a new row moves the middles
  for at, col in tbl.cols.items(): add(col, row[at], inc)
  return row

def clone(tbl: TBL, rows: ROWS = None) -> TBL:
  "An empty copy of TBL, plus ROWS."
  return Tbl([tbl.names] + (rows or []))

def dist(vs: Iterable[QTY], n: int) -> float:
  "Root mean square of N gaps."
  return sqrt(sum(v*v for v in vs) / n)

def ydist(tbl: TBL, row: ROW) -> float:
  "How far ROW's goals are from the best they could be."
  return dist((abs(norm(c, row[c.at]) - c.goal)
                    for c in tbl.y), len(tbl.y))

def gap(col: COL, a: ATOM, b: ATOM) -> QTY:
  "Distance between two values of one column."
  if a == "?" or b == "?": return 1    # unknown = far
  return a!=b if type(col) is Sym else abs(norm(col,a)-norm(col,b))

def xdist(tbl: TBL, r1: ROW, r2: ROW | MIDS) -> float:
  "How far apart two rows are, over the x columns."
  return dist((gap(c, r1[c.at], r2[c.at])
                    for c in tbl.x), len(tbl.x))

def oracle(row: ROW) -> ROW:
  """Labeller. Rows here arrive with ys, so nothing to do.
  Real oracles fill in any missing y cells (and can close
  over a TBL to know which cells those are)."""
  return row

def acquire(tbl: TBL, cap: int | None = None,
            label: ORACLE = oracle) -> ROWS:
  "Label near the best rows, far from the rest."
  cap  = cap or the.Stop
  todo = random.sample(tbl.rows, len(tbl.rows))[:the.Few]
  both = clone(tbl, [label(todo.pop())
                     for _ in range(the.Start)])
  both.rows.sort(key=lambda r: ydist(both, r))
  n    = int(sqrt(the.Start))
  best,rest = clone(tbl,both.rows[:n]), clone(tbl,both.rows[n:])
  while todo and len(both.rows) < cap:
    cb, cr = mids(best), mids(rest)    # two centroids
    todo.sort(key=lambda z: xdist(tbl,z,cr) - xdist(tbl,z,cb))
    addRow(best, addRow(both, label(todo.pop())))
    best.rows.sort(key=lambda z: ydist(both, z))  # no y leak
    if len(best.rows) > sqrt(len(both.rows)): # best stays small
      addRow(rest, addCols(best, best.rows.pop(), -1))
  return both.rows

def grabs(tbl: TBL, cap: int | None = None,
          label: ORACLE = oracle) -> ROWS:
  "The straw man: same budget, taken at random."
  cap = cap or the.Stop
  return [label(r) for r in
          random.sample(tbl.rows, min(cap, len(tbl.rows)))]

# Calls core.  Nothing above this line calls down here.
# NODE = o(rows, at, v, go, kids, mu).
def cut(tbl: TBL, rows: ROWS) -> tuple[int,ATOM,GO]|None:
  "The (at, v, go) whose two sides have the tightest ys."
  ys = [ydist(tbl, r) for r in rows]
  out, least = None, 1e30
  for col in tbl.x:
    for v, go in candidates(col, rows):
      a, b = Num(), Num()   # one pass, so one `go` per row
      for r,y in zip(rows,ys): add(a if go(r) else b, y)
      if a.n >= the.Leaf and b.n >= the.Leaf:
        score = (div(a)*a.n + div(b)*b.n) / (a.n + b.n)
        if score < least: out, least = (col.at, v, go), score
  return out

def candidates(col: COL, rows: ROWS) -> Iterator[tuple[ATOM,GO]]:
  "What splits to try: one per symbol, or CUTS per num."
  at = col.at
  if type(col) is Sym:
    for v in sorted({r[at] for r in rows if r[at] != "?"}):
      yield v, lambda r,v=v,at=at: r[at] == v
  else:                # by rank, not value: THESE rows are dense
    xs = sorted(r[at] for r in rows if r[at] != "?")
    k  = max(1, len(xs)//the.Cuts)     # cuts, spaced by rank
    for v in xs[k-1:-1:k]:
      yield v, (lambda r,v=v,at=at: r[at] != "?" and r[at] <= v)

def tree(tbl: TBL, rows: ROWS) -> NODE:
  "Split while a split keeps both sides big enough."
  node = o(rows=rows, at=None, v=None, go=None, kids=[],
           mu=sum(ydist(tbl,r) for r in rows)/len(rows))
  if len(rows) > the.Leaf and (found := cut(tbl, rows)):
    at, v, go = found
    yes, no = [], []
    for r in rows: (yes if go(r) else no).append(r)
    node.at, node.v, node.go = at, v, go # cut kepts sides big
    node.kids = [tree(tbl, yes), tree(tbl, no)]
  return node

def leaf(node: NODE, row: ROW) -> NODE:
  "Walk ROW down to its leaf."
  while node.kids: node = node.kids[0 if node.go(row) else 1]
  return node

def leafs(node: NODE) -> list[NODE]:
  "Return leaves of this tree."
  return [x for k in node.kids for x in leafs(k)] or [node]

def show(tbl: TBL, node: NODE, pre: str | None = None,
         edge: str = "") -> None:
  "Tree on the left; ydist and n on the right."
  ls = sorted(leafs(node), key=lambda z: z.mu)
  def walk(z: NODE, pre: str | None, edge: str) -> None:
    m = "+" if z is ls[0] else "-" if z is ls[-1] else " "
    print(f"{m} {round(100*z.mu):>4} {len(z.rows):>5}   "
          f"{(pre or '')+edge}".rstrip())
    op = "=" if type(tbl.cols[z.at or 0]) is Sym else "<="
    no = "!=" if op == "=" else ">"
    for kid, oper in zip(z.kids, [op, no]):
      walk(kid, "" if pre is None else pre + "|  ",
           f"{tbl.names[z.at]} {oper} {say(z.v)}")
  print(" ydist     n")
  walk(node, pre, edge)

# Calls core, acquire, and one RANKER -- tree's, by default.
def ranker(lab: TBL) -> Callable[[ROW], float]:
  "Grow a tree on LAB; score any row by its leaf's mean ydist."
  node = tree(lab, lab.rows)
  return lambda r: leaf(node, r).mu

def wins(tbl: TBL) -> Callable[[ROW], float]:
  "100 at the best row, 0 at an average one."
  ys = sorted(ydist(tbl, r) for r in tbl.rows)
  lo, avg = ys[0], sum(ys)/len(ys)
  return lambda r: max(-100, min(100, 100*(1 - (ydist(tbl,r)-lo)
                                            / (avg-lo+1e-32))))

def holdout(tbl: TBL, pick: PICKER = acquire,
            rank: RANKER = ranker,
            label: ORACLE = oracle) -> ROW:
  "Train on half; of CHECK guesses on the rest, pick best."
  rows  = random.sample(tbl.rows, len(tbl.rows))
  n     = len(rows)//2
  tr    = clone(tbl, rows[:n][:the.Few])
  lab   = clone(tbl, pick(tr, the.Stop - the.Check, label))
  top   = sorted(rows[n:], key=rank(lab))
  return min(top[:the.Check], key=lambda r: ydist(lab, label(r)))

# A leaf: it calls core, and nothing here calls it.
def adds(vs: Iterable[ATOM], col: COL | None = None) -> COL:
  "Every item of VS into COL."
  col = Num() if col is None else col
  for v in vs: add(col, v)
  return col

def cohen(xs: NUMS, ys: NUMS, d: float = 0.35,
          eps: float = 0) -> bool:
  "Are the means within D pooled standard deviations?"
  a, b = adds(xs), adds(ys)
  pool = ((a.n-1)*div(a)**2 + (b.n-1)*div(b)**2)/(a.n+b.n-2)
  return abs(a.mu - b.mu) <= max(eps, d * sqrt(pool))

def cliffs(xs: NUMS, ys: NUMS, d: float = 0.197) -> bool:
  "Sorted in.  Is the rank imbalance small enough?"
  gt = lt = j = k = 0
  for x in xs:
    while j < len(ys) and ys[j] <  x: j += 1; k = j
    while k < len(ys) and ys[k] <= x: k += 1
    gt += j; lt += len(ys) - k
  return abs(gt - lt) / (len(xs) * len(ys)) <= d

def ks(xs: NUMS, ys: NUMS, a: float = 1.36) -> bool:
  "Sorted in.  95% Kolmogorov-Smirnov."
  n, m, i, j, d = len(xs), len(ys), 0, 0, 0
  while i < n and j < m:
    v = min(xs[i], ys[j])
    while i < n and xs[i] <= v: i += 1
    while j < m and ys[j] <= v: j += 1
    d = max(d, abs(i/n - j/m))
  return d <= a * sqrt((n + m) / (n * m))

def same(xs: NUMS, ys: NUMS, eps: float = 0) -> bool:
  """Indistinguishable.  Always cliffs and ks; EPS adds cohen, and
  doubles as its effect size (eps=0 skips that third test)."""
  xs, ys = sorted(xs), sorted(ys)
  return (cliffs(xs,ys) and ks(xs,ys)
          and (not eps or cohen(xs,ys,d=eps)))

def remember(header, rows):
    """
    Finds the best rows based on the goals defined in the header.
    Goals ending in '+' are maximized, '-' are minimized.
    """
    # Create a table object to use ezr's utility functions
    tbl = ezr.Tbl([header] + rows)
    
    # Sort rows by ydist (distance from ideal goals)
    # ydist calculates how far a row is from the best possible outcome
    # Lower ydist is better.
    indexed_rows = list(enumerate(rows))
    indexed_rows.sort(key=lambda x: ezr.ydist(tbl, x[1]))
    
    # Return the top 5% of rows (or at least 1)
    limit = max(1, len(rows) // 20)
    return [idx for idx, row in indexed_rows[:limit]]

import ezr
import random

def guess(header, train, test):
    """
    Estimates the first goal column for each test row.
    We use the tree-based ranker from ezr to predict the outcome.
    """
    # Create a table for the training data
    tbl = ezr.Tbl([header] + train)
    
    # Grow a tree on the training data to learn the relationship between
    # inputs and the goal columns (ydist).
    node = ezr.tree(tbl, tbl.rows)
    
    # For each test row, walk the tree to find the leaf node.
    # The estimate is the mean ydist of that leaf.
    # Note: The goal is to estimate the first goal column.
    # Since ydist is a normalized distance, we map it back to the 
    # scale of the first goal column using the leaf's centroid.
    
    results = []
    # Identify the first goal column index
    goal_col = [c for c in tbl.y if c.at == [i for i, h in enumerate(header) if h.endswith(("+", "-"))][0]][0]
    
    for row in test:
        leaf = ezr.leaf(node, row)
        # Calculate the average value of the first goal column in this leaf
        vals = [r[goal_col.at] for r in leaf.rows]
        results.append(sum(vals) / len(vals))
        
    return results

if __name__ == "__main__":
    # Load data
    filename = "WA_Fn-UseC_-HR-Employee-Attrition.n1000.train.csv"
    data = list(ezr.csv(filename))

    random.seed(1234567891)

import ezr

def flow(header, row, seen):
    """
    Decide if the current row is worth a 'what would keep you here' conversation.
    We compare the current row against the 'seen' population using the 
    distance-to-ideal (ydist) metric. If the row is in the top 5% of 
    performance/tenure potential, we flag it.
    """
    if not seen:
        return False
    
    # Create a table from seen rows to establish the context of 'good'
    tbl = ezr.Tbl([header] + seen)
    
    # Calculate ydist for the current row
    # Note: row might have '?' for goals, but ydist handles normalization
    # We need to estimate the goals if they are missing.
    # For simplicity, we use the median of the seen data to fill '?'
    current_row = list(row)
    for i, val in enumerate(current_row):
        if val == "?":
            current_row[i] = ezr.mid(tbl.cols[i])
            
    current_ydist = ezr.ydist(tbl, current_row)
    
    # Compare against the distribution of seen rows
    seen_ydists = [ezr.ydist(tbl, r) for r in seen]
    
    # If this row is better (lower ydist) than 95% of what we've seen,
    # it's a priority for a retention conversation.
    threshold = sorted(seen_ydists)[int(len(seen_ydists) * 0.05)]
    
    return current_ydist <= threshold

import ezr
import step01

def blame(header, rows):
    """
    Identifies input columns that best explain why the worst rows are 'bad'.
    We define 'bad' as having a high ydist (distance from ideal goals).
    We compare the distribution of input values in the worst 25% of rows
    against the best 25% of rows to find the most significant differences.
    """
    tbl = ezr.Tbl([header] + rows)
    
    # Sort rows by ydist (distance from ideal)
    # Higher ydist = worse performance/tenure
    sorted_rows = sorted(rows, key=lambda r: ezr.ydist(tbl, r), reverse=True)
    
    n = len(sorted_rows) // 4
    worst = sorted_rows[:n]
    best = sorted_rows[-n:]
    
    # Score columns by how much their distribution differs between worst and best
    # For numeric: difference in means. For symbolic: difference in entropy/frequency.
    scores = []
    for col in tbl.x:
        if isinstance(col, ezr.Num):
            mu_worst = sum(r[col.at] for r in worst) / n
            mu_best = sum(r[col.at] for r in best) / n
            scores.append((abs(mu_worst - mu_best), col.txt))
        else:
            # For symbols, check if the most common value in 'worst' is different
            # or if the distribution is skewed compared to 'best'
            w_vals = [r[col.at] for r in worst]
            b_vals = [r[col.at] for r in best]
            # Simple heuristic: difference in frequency of the most common item
            w_mode = max(set(w_vals), key=w_vals.count)
            b_mode = max(set(b_vals), key=b_vals.count)
            diff = 1.0 if w_mode != b_mode else 0.0
            scores.append((diff, col.txt))
            
    # Return top 3 columns that explain the difference
    scores.sort(key=lambda x: x[0], reverse=True)
    return [s[1] for s in scores[:3]]

import ezr

def choose(header, rows, label):
    """
    Uses the acquire function from ezr to select the best working arrangement
    within a budget of 50 labels.
    """
    # Create a table object from the provided rows.
    # Note: The rows passed to choose have '?' in goal columns.
    tbl = ezr.Tbl([header] + rows)
    
    # Map rows to their original indices to return the correct index later.
    # We use id() to handle potential duplicate row content.
    row_to_idx = {id(r): i for i, r in enumerate(rows)}
    
    # Identify goal column indices
    goal_indices = [i for i, h in enumerate(header) if h.endswith(("+", "-"))]
    
    def oracle(row):
        # Find the index of this row
        idx = row_to_idx[id(row)]
        # Get true goals from the harness
        true_goals = label(idx)
        # Update the row with the true goal values
        for i, goal_val in zip(goal_indices, true_goals):
            row[i] = goal_val
        return row

    # Use ezr.acquire to spend the budget of 50 labels.
    # ezr.acquire returns the rows that were labeled.
    labeled_rows = ezr.acquire(tbl, cap=50, label=oracle)
    
    # Find the best row among the ones we labeled.
    # Lower ydist is better.
    best_row = min(labeled_rows, key=lambda r: ezr.ydist(tbl, r))
    
    # Return the index of the best row found
    return row_to_idx[id(best_row)]

import ezr

def spot(header, rows):
    """
    Identifies suspicious rows by calculating the distance of each row
    from the global centroid of the data. Rows that are extreme outliers
    in the input space (x-columns) are flagged as potentially entered wrongly.
    """
    # Create a table to handle normalization and distance calculations
    tbl = ezr.Tbl([header] + rows)
    
    # Calculate the distance of each row from the global centroid (mids)
    # in the input (x) space.
    centroid = ezr.mids(tbl)
    
    # Calculate xdist for every row
    # xdist measures how far a row's inputs are from the average inputs
    distances = []
    for i, row in enumerate(rows):
        d = ezr.xdist(tbl, row, centroid)
        distances.append((d, i))
    
    # Sort by distance descending (most suspicious = furthest from typical)
    distances.sort(key=lambda x: x[0], reverse=True)
    
    # Return the indices of the top 10 most suspicious rows
    return [idx for d, idx in distances[:10]]

import ezr
import random

def race(header, rows, label):
    """
    Compares three strategies for finding the best employee record:
    1. 'random': Picks rows at random.
    2. 'acquire': Uses the model-based acquisition.
    3. 'grabs': Uses the straw man random grabber.
    """
    budget = 50
    labeled_indices = {}
    
    # We need a stable way to map rows to indices for the oracle
    # Since rows are lists, we use their position in the input list
    row_map = {id(r): i for i, r in enumerate(rows)}
    
    def oracle_wrapper(row):
        idx = row_map[id(row)]
        if idx not in labeled_indices:
            if len(labeled_indices) >= budget:
                raise Exception("Budget exceeded!")
            true_goals = label(idx)
            goal_indices = [i for i, h in enumerate(header) if h.endswith(("+", "-"))]
            for i, val in zip(goal_indices, true_goals):
                row[i] = val
            labeled_indices[idx] = row
        return labeled_indices[idx]

    # Create a base table to handle normalization
    # We use a copy of rows to avoid mutating the input
    tbl = ezr.Tbl([header] + [list(r) for r in rows])
    
    # Strategy 1: Random
    def run_random():
        indices = random.sample(range(len(rows)), 16)
        best_idx, min_ydist = -1, 1e30
        for i in indices:
            r = oracle_wrapper(rows[i])
            d = ezr.ydist(tbl, r)
            if d < min_ydist:
                min_ydist, best_idx = d, i
        return best_idx, min_ydist

    # Strategy 2: Acquire
    def run_acquire():
        # Acquire needs a table with some labeled data
        # We use 17 labels for this strategy
        labeled = ezr.acquire(tbl, cap=17, label=oracle_wrapper)
        best_r = min(labeled, key=lambda r: ezr.ydist(tbl, r))
        return row_map[id(best_r)], ezr.ydist(tbl, best_r)

    # Strategy 3: Grabs
    def run_grabs():
        indices = random.sample(range(len(rows)), 17)
        best_idx, min_ydist = -1, 1e30
        for i in indices:
            r = oracle_wrapper(rows[i])
            d = ezr.ydist(tbl, r)
            if d < min_ydist:
                min_ydist, best_idx = d, i
        return best_idx, min_ydist

    results = {
        "random": run_random(),
        "acquire": run_acquire(),
        "grabs": run_grabs()
    }

    winner = min(results, key=lambda k: results[k][1])
    return {
        "winner": winner,
        "row": results[winner][0],
        "tried": ["random", "acquire", "grabs"]
    }

    """

    """

    return [idx for d, idx in distances[:n]]

    population of rows seen so far. If the row is significantly different 
    from the historical centroid, return True.
    """
    if len(seen) < 20:
        return False
    
    # Create a table from the seen rows to establish the current 'normal'
    tbl = ezr.Tbl([header] + seen)
    
    # Calculate the centroid of the seen data
    centroid = ezr.mids(tbl)
    
    # Calculate the distance of the new row from the current centroid
    # We use xdist to measure deviation in input space
    current_dist = ezr.xdist(tbl, row, centroid)
    
    # Calculate the distribution of distances seen so far
    seen_dists = [ezr.xdist(tbl, r, centroid) for r in seen]
    
    # If the current row's distance is in the top 5% of deviations, 
    # it is an anomaly (the team no longer looks the way it normally does).
    threshold = sorted(seen_dists)[int(len(seen_dists) * 0.95)]
    
    return current_dist > threshold

def choose(header, rows, label):
    """
    Finds the best working arrangement for 'Laboratory Technician' staff
    only, using the acquisition strategy within a budget of 50 labels.
    """
    # Filter rows to only include Laboratory Technicians
    job_role_idx = header.index('jobRole')
    lab_rows = [r for r in rows if r[job_role_idx] == 'Laboratory Technician']
    
    # Map these filtered rows back to their original indices in the full dataset
    # We use the row content to identify them since they are unique enough
    # or we can rely on the fact that the harness expects the index of the row
    # in the original 'rows' list provided to the function.
    original_indices = [i for i, r in enumerate(rows) if r[job_role_idx] == 'Laboratory Technician']
    
    # Create a mapping for the oracle
    row_to_orig_idx = {id(lab_rows[i]): original_indices[i] for i in range(len(lab_rows))}
    
    # Define a local oracle that translates the lab_rows index to the original index
    def local_label(row):
        orig_idx = row_to_orig_idx[id(row)]
        true_goals = label(orig_idx)
        goal_indices = [i for i, h in enumerate(header) if h.endswith(("+", "-"))]
        for i, val in zip(goal_indices, true_goals):
            row[i] = val
        return row

    # Use the logic from step07 to find the best row among the filtered set
    tbl = ezr.Tbl([header] + lab_rows)
    labeled_rows = ezr.acquire(tbl, cap=50, label=local_label)
    
    # Find the best row among the ones we labeled
    best_row = min(labeled_rows, key=lambda r: ezr.ydist(tbl, r))
    
    return row_to_orig_idx[id(best_row)]

import ezr
import step09

def spot(header, rows):
    """
    Identifies rows that are 'unlike anyone we have employed'.
    We use the distance from the global centroid (mids) in the input space.
    Rows that are extreme outliers (high xdist) are those that the model
    has not seen before and thus cannot reliably estimate.
    """
    # Create a table to handle normalization and distance calculations
    tbl = ezr.Tbl([header] + rows)
    
    # Calculate the centroid of the data
    centroid = ezr.mids(tbl)
    
    # Calculate xdist for every row
    # xdist measures how far a row's inputs are from the average inputs
    distances = []
    for i, row in enumerate(rows):
        d = ezr.xdist(tbl, row, centroid)
        distances.append((d, i))
    
    # Sort by distance descending (most suspicious/unlike others = furthest from typical)
    distances.sort(key=lambda x: x[0], reverse=True)
    
    # Return the indices of the rows that are most 'unlike' our historical data
    return [idx for d, idx in distances]
