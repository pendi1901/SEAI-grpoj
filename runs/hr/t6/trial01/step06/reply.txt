import ezr

def justify(header, rows, row):
    """Estimate how good an unknown employee is."""
    tbl = ezr.Tbl(iter([header] + rows))

    # Train a decision tree on all data (splits only on x columns)
    node = ezr.tree(tbl, tbl.rows)

    # Walk the tree with the unknown row using its known x values
    leaf_node = ezr.leaf(node, row)

    # The leaf's mean ydist is our prediction for this employee type
    pred = leaf_node.mu

    # Score all training rows to get actual ydists
    scores = [ezr.ydist(tbl, r) for r in rows]

    # Fraction of training employees that this one beats
    p = sum(1 for s in scores if s < pred) / len(scores)

    reason = f"Better than {p:.0%} of employees."
    return (p, reason)

if __name__ == "__main__":
    import csv
    with open("WA_Fn-UseC_-HR-Employee-Attrition.train.csv", encoding="utf-8-sig") as f:
        reader = csv.reader(f)
        header = next(reader)
        rows = [next(reader) for _ in range(1470)]

    # Unknown row with goals hidden
    unknown = ["?"] * len(header)
    p, reason = justify(header, rows, unknown)
    print(reason)
