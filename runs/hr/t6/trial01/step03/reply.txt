import ezr

def certify(xs, ys):
    """Return True if xs and ys are the same apart from noise."""
    mx = sum(xs) / len(xs)
    my = sum(ys) / len(ys)
    diff = abs(mx - my)
    spread = (ezr.dist((v ** 2 for v in xs), len(xs)) + ezr.dist((v ** 2 for v in ys), len(ys))) / 2
    return diff <= 0.15 * spread

if __name__ == "__main__":
    tbl = ezr.Tbl(open("WA_Fn-UseC_-HR-Employee-Attrition.train.csv", encoding="utf-8-sig"))
    over_yes = [r for r in tbl.rows if r[20] == "Y"]  # overTime column (index 20)
    over_no = [r for r in tbl.rows if r[20] == "N"]
    xs = [ezr.ydist(tbl, r) for r in over_yes]
    ys = [ezr.ydist(tbl, r) for r in over_no]
    print(f"Overtime workers: mean ydist={sum(xs)/len(xs):.3f}, non-overtime: {sum(ys)/len(ys):.3f}")
    print("Real effect?" if not certify(xs, ys) else "Just noise.")
