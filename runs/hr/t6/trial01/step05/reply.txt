import ezr
from step01 import remember


def blame(header, rows):
    """Return input column names that best explain why the worst rows are bad."""
    tbl = ezr.Tbl(iter([header] + rows))
    ys = [ezr.ydist(tbl, r) for r in tbl.rows]

    # Worst rows: highest ydist (farthest from ideal goals)
    worst = sorted(range(len(ys)), key=ys.__getitem__, reverse=True)[:max(1, len(rows)//20)]
    good = [i for i in range(len(ys)) if i not in set(worst)]

    # For each input column (non-goal, non-skip), check if it separates worst from good
    out = []
    for col in tbl.x:
        vals = {r[col.at] for r in tbl.rows}
        if len(vals) < 2: continue
        # Split rows by this column's value and compare mean ydist
        groups = {}
        for i, r in enumerate(tbl.rows):
            v = r[col.at]
            if v == "?": continue
            groups.setdefault(v, []).append(ys[i])

        if len(groups) < 2: continue
        means = [sum(g)/len(g) for g in groups.values()]
        # If worst group has much higher ydist than good group, this column matters
        worst_mean = max(means)
        good_mean = min(means)
        spread = ezr.dist((v**2 for v in ys), len(ys))
        if abs(worst_mean - good_mean) > 0.15 * (spread + 1e-32):
            out.append(col.txt)

    return out


if __name__ == "__main__":
    import csv
    with open("WA_Fn-UseC_-HR-Employee-Attrition.train.csv", encoding="utf-8-sig") as f:
        reader = csv.reader(f)
        header = next(reader)
        rows = [next(reader) for _ in range(1470)]

    result = blame(header, rows)
    print("What explains short stays:")
    for c in result:
        print(f"  {c}")
