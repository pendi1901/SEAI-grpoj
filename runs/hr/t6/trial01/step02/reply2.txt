import ezr

def guess(header, train, test):
    """Estimate how long a newly hired employee is likely to stay."""
    # Identify goal columns (names ending in +)
    goals = [header[i] for i, h in enumerate(header) if h.endswith("+")]

    # Build column summaries from training data
    class _Col:
        def __init__(self, name, idx):
            self.name = name
            self.idx = idx
            self.n = 0
            self.mu = 0.0
            self.m2 = 0.0
            self.sd = 0.0

        def add(self, v):
            if v != "?":
                self.n += 1
                d = v - self.mu
                self.mu += d / max(1, self.n)
                self.m2 = max(0, self.m2 + d * (v - self.mu))
                if self.n >= 2:
                    self.sd = (self.m2 / (self.n - 1)) ** 0.5

    cols = [_Col(h, i) for i, h in enumerate(header)]

    # Process training rows to build column statistics
    ys = []
    for row in train:
        vals = [row[i] if i < len(row) else "?" for i in range(len(cols))]
        for col in cols:
            col.add(vals[col.idx])

        goal_vals = []
        for col in cols:
            if col.name.endswith("+"):
                v = vals[col.idx]
                if v != "?":
                    z = max(-3, min(3, (v - col.mu) / (1e-32 + col.sd)))
                    goal_vals.append(abs(1 / (1 + (-1.7 * z)) - 1))
                else:
                    goal_vals.append(0)

        if len(goal_vals) > 0:
            yd = sum(goal_vals) / len(goal_vals)
        else:
            yd = 0
        ys.append(yd)

    # Sort by ydist (lower is better - closer to best goals)
    indexed = list(enumerate(ys))
    indexed.sort(key=lambda x: x[1])

    n = max(1, int(len(train) * 0.05))
    top_indices = [idx for idx, _ in indexed[:n]]

    # Now process test rows - predict the first goal column value
    # Use the trained columns to estimate ydist for each test row
    test_ys = []
    for row in test:
        vals = [row[i] if i < len(row) else "?" for i in range(len(cols))]

        goal_vals = []
        for col in cols:
            if col.name.endswith("+"):
                v = vals[col.idx]
                # Test rows have "?" for goals, so we need to predict
                # Use the relationship between ydist and the first goal column
                # For now, estimate based on training distribution
                goal_vals.append(0)

        if len(goal_vals) > 0:
            yd = sum(goal_vals) / len(goal_vals)
        else:
            yd = 0
        test_ys.append(yd)

    # Use the top employees from training to calibrate our prediction
    # The first goal column is "YearsAtCompany+" - we want to predict this
    # Based on training data, find the relationship between ydist and YearsAtCompany+
    
    # Calculate the mean of YearsAtCompany+ for top performers (low ydist)
    years_col = cols[header.index("YearsAtCompany+") if "YearsAtCompany+" in header else 0]
    
    # Get the distribution of YearsAtCompany+ from training data
    train_years = []
    for row in train:
        v = row[years_col.idx]
        if v != "?":
            train_years.append(float(v))
    
    if len(train_years) > 0:
        mean_years = sum(train_years) / len(train_years)
        # For new hires, estimate they'll stay around the average or slightly above
        # since we want to be optimistic about retention
        return [mean_years] * len(test)
    else:
        return [0.0] * len(test)


if __name__ == "__main__":
