import ezr
import step08

def fix(header, row, levers):
    """
    Given a 'bad' row, find a change in the allowed levers that 
    improves the row's predicted retention (ydist).
    We exclude 'MonthlyIncome' and 'PercentSalaryHike' to respect the 
    'no money for pay' constraint.
    """
    # Load data to build a model for ydist calculations
    data = list(ezr.csv("WA_Fn-UseC_-HR-Employee-Attrition.n1000.train.csv"))
    tbl = ezr.Tbl(data)
    
    best_dist = ezr.ydist(tbl, row)
    best_change = {}

    # Try changing each lever to improve the row
    for col_name, allowed in levers.items():
        if col_name not in header:
            continue
        idx = header.index(col_name)
        original_val = row[idx]
        
        candidates = []
        if isinstance(allowed, list):  # Symbolic
            candidates = [v for v in allowed if v != original_val]
        else:  # Numeric (low, high)
            low, high = allowed
            # Try discrete steps within the range
            candidates = [low, (low + high) / 2, high]
            
        for val in candidates:
            test_row = list(row)
            test_row[idx] = val
            
            # Check if this change improves the ydist
            dist = ezr.ydist(tbl, test_row)
            if dist < best_dist:
                best_dist = dist
                best_change = {col_name: val}
                
    return best_change

if __name__ == "__main__":
    # Load data
    data = list(ezr.csv("WA_Fn-UseC_-HR-Employee-Attrition.n1000.train.csv"))
    header = data[0]
    rows = data[1:]
    
    # Define levers excluding pay-related columns
    levers = {
        "businessTravel": ["Non-Travel", "Travel_Rarely", "Travel_Frequently"],
        "overTime": ["No", "Yes"],
        "TrainingTimesLastYear": (0, 6),
        "StockOptionLevel": (0, 3),
        "JobInvolvement": (1, 4),
        "WorkLifeBalance": (1, 4)
    }
    
    # Find a 'bad' row (high ydist)
    tbl = ezr.Tbl(data)
    bad_rows = sorted(rows, key=lambda r: ezr.ydist(tbl, r), reverse=True)
    bad_row = bad_rows[0]
    
    # Get recommendation
    recommendation = fix(header, bad_row, levers)
    
    print("Analysis: Recommendations for improving retention without increasing pay.")
    print(f"Targeted employee (Age: {bad_row[header.index('Age')]}) has high attrition risk.")
    if recommendation:
        for col, val in recommendation.items():
            print(f"Recommended change: Set '{col}' to '{val}'.")
    else:
        print("No lever change found that improves the outcome under current constraints.")
