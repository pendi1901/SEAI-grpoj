import ezr
import step08

def fix(header, row, levers):
    """
    Given a 'bad' row, find a change in the allowed levers that 
    improves the row's predicted retention (ydist).
    We exclude 'MonthlyIncome' from the levers to respect the 
    'no money for pay' constraint.
    """
    # Filter out MonthlyIncome from levers if it exists
    if "MonthlyIncome" in levers:
        del levers["MonthlyIncome"]
        
    # Reuse the logic from step08 to find the best lever change
    return step08.fix(header, row, levers)

if __name__ == "__main__":
    # Load data
    data = list(ezr.csv("WA_Fn-UseC_-HR-Employee-Attrition.n1000.train.csv"))
    header = data[0]
    rows = data[1:]
    
    # Define levers excluding pay-related columns
    levers = {
        "businessTravel": ["Non-Travel", "Travel_Rarely", "Travel_Frequently"],
        "overTime": ["No", "Yes"],
        "TrainingTimesLastYear": (0, 6),
        "PercentSalaryHike": (11, 25),
        "StockOptionLevel": (0, 3),
        "JobInvolvement": (1, 4),
        "WorkLifeBalance": (1, 4)
    }
    
    # Find a 'bad' row (high ydist)
    tbl = ezr.Tbl(data)
    bad_rows = sorted(rows, key=lambda r: ezr.ydist(tbl, r), reverse=True)
    bad_row = bad_rows[0]
    
    # Get recommendation
    recommendation = fix(header, bad_row, levers)
    
    print("Analysis: Recommendations for improving retention without increasing pay.")
    print(f"Targeted employee (Age: {bad_row[header.index('Age')]}) has high attrition risk.")
    if recommendation:
        for col, val in recommendation.items():
            print(f"Recommended change: Set '{col}' to '{val}'.")
    else:
        print("No lever change found that improves the outcome under current constraints.")
