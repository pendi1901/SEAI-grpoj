import ezr
import random

def flow(header, row, seen):
    """
    Decide if the current row represents a shift in team behavior.
    We compare the current row's input distribution (x-space) to the 
    distribution of the 'seen' rows. If the current row is an outlier 
    (distance to the centroid of 'seen' is significantly higher than 
    the average distance of 'seen' rows to their centroid), we flag it.
    """
    if len(seen) < 10:
        return False
    
    # Create a table from seen rows to establish a baseline
    tbl = ezr.Tbl([header] + seen)
    
    # Calculate the centroid of the seen data
    centroid = ezr.mids(tbl)
    
    # Calculate distance of the current row to the seen centroid
    current_dist = ezr.xdist(tbl, row, centroid)
    
    # Calculate the average distance of the seen rows to their centroid
    seen_dists = [ezr.xdist(tbl, r, centroid) for r in seen]
    avg_seen_dist = sum(seen_dists) / len(seen_dists)
    
    # If the current row is significantly further away (e.g., > 2x average),
    # it indicates the team no longer looks like it normally does.
    return current_dist > (2.0 * avg_seen_dist)

if __name__ == "__main__":
    # Load data
    data = list(ezr.csv("WA_Fn-UseC_-HR-Employee-Attrition.n1000.train.csv"))
    header = data[0]
    rows = data[1:]
    
    # Simulate arrival of rows
    random.seed(1234567891)
    random.shuffle(rows)
    
    seen = []
    shifts = 0
    for row in rows:
        if flow(header, row, seen):
            shifts += 1
        seen.append(row)
        
    print("Analysis: Detecting shifts in team composition.")
    print(f"Processed {len(rows)} employee records.")
    print(f"Detected {shifts} instances where the team's profile shifted significantly.")
    print("Logic: A shift is flagged when a new record is an outlier relative to the "
          "input feature distribution of all previously seen records.")
