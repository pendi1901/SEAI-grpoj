import csv

def remember(header, rows):
    """
    Identify staff who stay 'unusually long'.
    'YearsAtCompany+' is a goal to maximize.
    We return the indices of the top 5% (max 40 rows for 800 records).
    """
    # Find index of the goal column
    goal_col = -1
    for i, name in enumerate(header):
        if name.endswith('+'):
            goal_col = i
            break
    
    if goal_col == -1:
        return [0]

    # Create list of (value, original_index)
    indexed_rows = []
    for idx, row in enumerate(rows):
        val = row[goal_col]
        indexed_rows.append((val, idx))
    
    # Sort by value descending (maximize)
    indexed_rows.sort(key=lambda x: x[0], reverse=True)
    
    # Return top 5% (40 rows for 800 records)
    limit = max(1, len(rows) // 20)
    return [item[1] for item in indexed_rows[:limit]]

if __name__ == "__main__":
    filename = 'WA_Fn-UseC_-HR-Employee-Attrition.n1000.train.csv'
    rows = []
    header = []
    
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            new_row = []
            for i, val in enumerate(row):
                # Convert numeric columns (Uppercase start)
                if header[i][0].isupper():
                    try:
                        new_row.append(float(val))
                    except ValueError:
                        new_row.append(0.0)
                else:
                    new_row.append(val)
            rows.append(new_row)
            
    indices = remember(header, rows)
    
    print("Staff with unusually long tenure (YearsAtCompany+):")
    # Identify index for YearsAtCompany+
    idx_years = [i for i, h in enumerate(header) if h == 'YearsAtCompany+'][0]
    
    for i in indices:
        row = rows[i]
        print(f"Row {i}: {row[idx_years]} years at company, "
              f"Role: {row[header.index('jobRole')]}, "
              f"Dept: {row[header.index('department')]}")
