import csv
import math

def certify(xs, ys):
    """
    Returns True if the two distributions are statistically similar 
    (null hypothesis not rejected), False if they differ significantly.
    Uses a simple Welch's t-test approach.
    """
    n1, n2 = len(xs), len(ys)
    if n1 < 2 or n2 < 2:
        return True
    
    m1, m2 = sum(xs) / n1, sum(ys) / n2
    v1 = sum((x - m1) ** 2 for x in xs) / (n1 - 1)
    v2 = sum((y - m2) ** 2 for y in ys) / (n2 - 1)
    
    # Standard error
    se = math.sqrt((v1 / n1) + (v2 / n2))
    if se == 0:
        return m1 == m2
    
    t_stat = abs(m1 - m2) / se
    # For a 95% confidence level, t-critical is approx 1.96
    return t_stat < 1.96

def flow(header, row, seen):
    """
    Decide if the current row is a 'keeper' based on the 'YearsAtCompany+' goal.
    We compare the current row's tenure against the historical average of 
    previous rows (seen). If the current row's tenure is significantly 
    higher than the average of what we have seen so far, we flag it.
    """
    # Identify goal column
    goal_idx = -1
    for i, name in enumerate(header):
        if name.endswith('+'):
            goal_idx = i
            break
    
    if goal_idx == -1:
        return False

    current_val = row[goal_idx]
    
    # If this is the first row, keep it
    if not seen:
        return True
    
    # Calculate average of seen rows
    seen_vals = [r[goal_idx] for r in seen]
    avg_seen = sum(seen_vals) / len(seen_vals)
    
    # If current tenure is at least 20% higher than the average of seen,
    # we consider it a 'what would keep you here' candidate.
    if current_val > (avg_seen * 1.2):
        return True
    
    return False

import csv

def blame(header, rows):
    """
    Identify input columns that correlate with low 'YearsAtCompany+' (short stays).
    We define 'worst' as the bottom 20% of tenure.
    We return the names of the top 3 input columns that have the highest 
    difference in mean/frequency between the 'worst' group and the rest.
    """
    goal_idx = -1
    for i, name in enumerate(header):
        if name.endswith('+'):
            goal_idx = i
            break
            
    # Sort rows by tenure to find the bottom 20%
    sorted_rows = sorted(rows, key=lambda r: r[goal_idx])
    limit = len(rows) // 5
    worst = sorted_rows[:limit]
    rest = sorted_rows[limit:]
    
    scores = []
    for i, name in enumerate(header):
        # Skip goals and the goal column itself
        if name.endswith('+') or name.endswith('-'):
            continue
            
        # Calculate a simple "blame" score: difference in means (numeric) 
        # or difference in mode frequency (symbolic)
        if isinstance(worst[0][i], float):
            m_worst = sum(r[i] for r in worst) / len(worst)
            m_rest = sum(r[i] for r in rest) / len(rest)
            score = abs(m_worst - m_rest)
        else:
            # For symbolic, check if the most common value in 'worst' 
            # is more frequent than in 'rest'
            val = max(set(r[i] for r in worst), key=lambda x: [r[i] for r in worst].count(x))
            freq_worst = [r[i] for r in worst].count(val) / len(worst)
            freq_rest = [r[i] for r in rest].count(val) / len(rest)
            score = abs(freq_worst - freq_rest)
            
        scores.append((score, name))
        
    scores.sort(key=lambda x: x[0], reverse=True)
    return [s[1] for s in scores[:3]]

import csv

def spot(header, rows):
    """
    Identifies suspicious rows by checking for outliers in numeric columns.
    A value is considered suspicious if it is more than 3 standard deviations 
    away from the mean of its column.
    """
    suspicious = {}
    
    # Identify numeric columns
    numeric_indices = [i for i, h in enumerate(header) if h[0].isupper()]
    
    for idx in numeric_indices:
        col_data = [r[idx] for r in rows if isinstance(r[idx], (int, float))]
        if not col_data:
            continue
            
        mean = sum(col_data) / len(col_data)
        variance = sum((x - mean) ** 2 for x in col_data) / len(col_data)
        std_dev = variance ** 0.5
        
        if std_dev == 0:
            continue
            
        # Flag rows where value is > 3 std devs from mean
        for row_idx, row in enumerate(rows):
            val = row[idx]
            if isinstance(val, (int, float)):
                z_score = abs(val - mean) / std_dev
                if z_score > 3:
                    suspicious[row_idx] = suspicious.get(row_idx, 0) + z_score
                    
    # Sort by suspicion score descending
    sorted_rows = sorted(suspicious.items(), key=lambda x: x[1], reverse=True)
    return [item[0] for item in sorted_rows]

    for i, name in enumerate(header):
        if name.endswith('+'):
            goal_idx = i
            break

    # Budget: 50 labels total
    # Strategy 1: Random
    random_indices = random.sample(range(len(rows)), 16)

    # We will sample 17 from the top 5% of the dataset as a heuristic.

    high_tenure_indices = random.sample(top_5_percent, min(17, len(top_5_percent)))
    
    # Strategy 3: Low tenure (control)
    sorted_rows = sorted(range(len(rows)), key=lambda i: rows[i][goal_idx])

import csv
import math

def get_distance(row1, row2, header):
    """
    Calculate Euclidean distance between two rows for numeric columns
    and Hamming distance for symbolic columns.
    """
    dist = 0.0
    for i, val in enumerate(row1):
        if header[i].endswith('+') or header[i].endswith('-'):
            continue
        
        if isinstance(val, (int, float)):
            # Normalize numeric distance
            dist += (val - row2[i]) ** 2
        else:
            # Symbolic distance
            if val != row2[i]:
                dist += 1.0
    return math.sqrt(dist)

def remember(header, rows):
    """
    Find the 5% of rows most similar to the first row in the dataset.
    """
    target = rows[0]
    distances = []
    for i, row in enumerate(rows):
        if i == 0:
            continue
        d = get_distance(target, row, header)
        distances.append((d, i))
    
    # Sort by distance ascending (most similar first)
    distances.sort(key=lambda x: x[0])
    
    limit = max(1, len(rows) // 20)
    return [item[1] for item in distances[:limit]]
