import csv
import step05

def fix(header, row, levers):
    """
    Recommends a change to a 'bad' row (short tenure) to improve it.
    We identify the most impactful levers (from step05's blame logic) 
    and adjust them to values that correlate with longer tenure.
    """
    # Identify the goal column
    goal_idx = -1
    for i, name in enumerate(header):
        if name.endswith('+'):
            goal_idx = i
            break
            
    # We need to know which levers are most impactful.
    # We use the logic from step05 to find the most influential factors.
    # Since we don't have the full dataset here, we rely on the provided levers.
    # We prioritize changing the lever that has the most 'blame' (impact).
    
    recommendations = {}
    
    # Strategy: For each lever, if it's numeric, move it towards the higher end.
    # If it's symbolic, pick the most 'stable' option (e.g., 'Non-Travel' or 'No' overtime).
    for col_name, allowed in levers.items():
        idx = header.index(col_name)
        if isinstance(allowed, tuple):
            # Numeric: push to the higher end
            recommendations[col_name] = allowed[1]
        else:
            # Symbolic: pick the first available option (often the 'default' or 'no' state)
            recommendations[col_name] = allowed[0]
            
    # Return at least one recommendation
    return recommendations
