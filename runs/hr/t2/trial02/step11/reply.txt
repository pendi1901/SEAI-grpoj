import csv
import math

def get_distance(row1, row2, header):
    """
    Calculate Euclidean distance between two rows for numeric columns
    and Hamming distance for symbolic columns.
    """
    dist = 0.0
    for i, val in enumerate(row1):
        if header[i].endswith('+') or header[i].endswith('-'):
            continue
        
        if isinstance(val, (int, float)):
            # Normalize numeric distance
            dist += (val - row2[i]) ** 2
        else:
            # Symbolic distance
            if val != row2[i]:
                dist += 1.0
    return math.sqrt(dist)

def remember(header, rows):
    """
    Find the 5% of rows most similar to the first row in the dataset.
    """
    target = rows[0]
    distances = []
    for i, row in enumerate(rows):
        if i == 0:
            continue
        d = get_distance(target, row, header)
        distances.append((d, i))
    
    # Sort by distance ascending (most similar first)
    distances.sort(key=lambda x: x[0])
    
    limit = max(1, len(rows) // 20)
    return [item[1] for item in distances[:limit]]

if __name__ == "__main__":
    filename = 'WA_Fn-UseC_-HR-Employee-Attrition.n1000.train.csv'
    rows = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            new_row = []
            for i, val in enumerate(row):
                if header[i][0].isupper():
                    try:
                        new_row.append(float(val))
                    except ValueError:
                        new_row.append(0.0)
                else:
                    new_row.append(val)
            rows.append(new_row)
            
    indices = remember(header, rows)
    
    goal_idx = -1
    for i, name in enumerate(header):
        if name.endswith('+') or name.endswith('-'):
            goal_idx = i
            break
            
    print(f"Comparing against employee at row 0 (Tenure: {rows[0][goal_idx]} years).")
    print("The most similar employees and their tenure:")
    for i in indices:
        print(f"- Row {i}: {rows[i][goal_idx]} years")
