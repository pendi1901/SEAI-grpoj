import csv
import math

def certify(xs, ys):
    """
    Returns True if the two distributions are statistically similar (noise),
    False if they significantly differ. Uses a simple Welch's t-test approach.
    """
    n1, n2 = len(xs), len(ys)
    if n1 < 2 or n2 < 2:
        return True
    
    m1, m2 = sum(xs) / n1, sum(ys) / n2
    v1 = sum((x - m1) ** 2 for x in xs) / (n1 - 1)
    v2 = sum((y - m2) ** 2 for y in ys) / (n2 - 1)
    
    # Standard error of the difference
    se = math.sqrt((v1 / n1) + (v2 / n2))
    if se == 0:
        return m1 == m2
    
    t_stat = abs(m1 - m2) / se
    # Threshold for 95% confidence (approx 1.96 for large samples)
    return t_stat < 1.96

def justify(header, rows, row):
    """
    Returns (p, reason) where p is the fraction of rows that 'row' is better than,
    and reason is a sentence explaining the ranking.
    """
    # Identify goal columns
    goals = [(i, name) for i, name in enumerate(header) if name.endswith('+') or name.endswith('-')]
    
    # Estimate goal values for the input row using the average from the dataset
    # (Using step02 logic for estimation)
    estimates = {}
    for idx, name in goals:
        vals = [r[idx] for r in rows if isinstance(r[idx], (int, float))]
        estimates[idx] = sum(vals) / len(vals) if vals else 0.0
    
    # Define "better" logic:
    # For '+', higher is better. For '-', lower is better.
    def is_better(r1, r2):
        for idx, name in goals:
            v1 = r1[idx] if r1[idx] != "?" else estimates[idx]
            v2 = r2[idx]
            if name.endswith('+'):
                if v1 < v2: return False
            else:
                if v1 > v2: return False
        return True

    # Calculate fraction of rows that the input row is better than
    better_count = sum(1 for r in rows if is_better(estimates, r))
    p = better_count / len(rows)
    
    reason = f"This team arrangement ranks in the top {int((1-p)*100)}% of all records based on its projected performance across all goal metrics."
    return (p, reason)

import csv

def choose(header, rows, label):
    """
    Finds the best working arrangement by sampling 50 rows, 
    labeling them, and picking the one with the best average goal performance.
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith('+') or name.endswith('-')]
    
    # Randomly sample 50 indices to label
    indices_to_sample = random.sample(range(len(rows)), min(50, len(rows)))
    
    best_idx = -1
    best_score = -float('inf')
    
    for i in indices_to_sample:
        # Get true values for the goals
        true_goals = label(i)
        
        # Calculate a normalized score for this row
        # We want to maximize (+) and minimize (-)
        # Since we don't have global min/max, we use the relative performance 
        # of the sampled set to rank them.
        score = 0
        for j, goal_idx in enumerate(goal_indices):
            val = true_goals[j]
            if header[goal_idx].endswith('+'):
                score += val
            else:
                score -= val
        
        if score > best_score:
            best_score = score
            best_idx = i
            
    return best_idx

import csv

def fix(header, row, levers):
    """
    Recommends changes for a 'bad' row (short tenure) to improve it.
    We pick a lever and move it towards the direction that typically 
    correlates with higher tenure.
    """
    recommendations = {}
    
    # We focus on the levers provided.
    # For numeric, we move towards the higher end of the range.
    # For symbolic, we pick a value that is not the current one.
    for col, allowed in levers.items():
        idx = header.index(col)
        current_val = row[idx]
        
        if isinstance(allowed, (list, tuple)) and not isinstance(allowed[0], (int, float)):
            # Symbolic: pick a value different from current
            options = [v for v in allowed if v != current_val]
            if options:
                recommendations[col] = random.choice(options)
        else:
            # Numeric: move towards the high end of the range
            low, high = allowed
            if current_val < high:
                recommendations[col] = min(high, current_val + (high - low) * 0.1)
        
        # Only need one change to satisfy the requirement
        if recommendations:
            break
            
    return recommendations
