import csv
import step05

def blame(header, rows):
    """
    Identifies input columns that have the least correlation with the goal 
    'YearsAtCompany+'. Columns that do not help explain the variance in 
    tenure are candidates for removal.
    """
    goal_col = "YearsAtCompany+"
    goal_idx = header.index(goal_col)
    
    # Identify input columns (exclude goals and IDs)
    input_cols = []
    for i, name in enumerate(header):
        if not (name.endswith('+') or name.endswith('-') or 'Number' in name):
            input_cols.append((i, name))
            
    # We calculate the variance of the goal column for each unique value 
    # (or bin) of the input column. If the variance remains high, the 
    # input column is not a good predictor of the goal.
    results = []
    for idx, name in input_cols:
        vals = [r[idx] for r in rows]
        
        # Group goal values by input value
        groups = {}
        for i, val in enumerate(vals):
            if val not in groups:
                groups[val] = []
            groups[val].append(rows[i][goal_idx])
            
        # Calculate average variance within groups
        total_var = 0
        for g in groups.values():
            if len(g) > 1:
                avg = sum(g) / len(g)
                var = sum((x - avg) ** 2 for x in g) / (len(g) - 1)
                total_var += var
        
        # Higher average variance means the column is less informative
        results.append((total_var / len(groups), name))
            
    # Sort by variance descending (least informative first)
    results.sort(key=lambda x: x[0], reverse=True)
    
    # Return the top 5 columns that have the least to do with the goal
    return [s[1] for s in results[:5]]
