def remember(header, rows):
    """
    Identify staff who stay 'unusually long'.
    'YearsAtCompany+' is a goal to maximize.
    We return the indices of the top 5% of rows based on this metric.
    """
    # Identify the index of the goal column
    goal_col = "YearsAtCompany+"
    try:
        idx = header.index(goal_col)
    except ValueError:
        return []

    # Create a list of (value, original_index)
    indexed_rows = []
    for i, row in enumerate(rows):
        val = row[idx]
        if isinstance(val, (int, float)):
            indexed_rows.append((val, i))

    # Sort by value descending (maximize)
    indexed_rows.sort(key=lambda x: x[0], reverse=True)

    # Calculate 5% limit (at least 1, max 5% of total rows)
    limit = max(1, len(rows) // 20)
    
    # Return the indices of the top performers
    return [item[1] for item in indexed_rows[:limit]]

import csv
import math

def certify(xs, ys):
    """
    Returns True if the two distributions are statistically similar (noise),
    False if they significantly differ. Uses a simple Welch's t-test approach.
    """
    n1, n2 = len(xs), len(ys)
    if n1 < 2 or n2 < 2:
        return True
    
    m1, m2 = sum(xs) / n1, sum(ys) / n2
    v1 = sum((x - m1) ** 2 for x in xs) / (n1 - 1)
    v2 = sum((y - m2) ** 2 for y in ys) / (n2 - 1)
    
    # Standard error of the difference
    se = math.sqrt((v1 / n1) + (v2 / n2))
    if se == 0:
        return m1 == m2
    
    t_stat = abs(m1 - m2) / se
    # Threshold for 95% confidence (approx 1.96 for large samples)
    return t_stat < 1.96

def analyze_overtime_impact(header, rows):
    """
    Compares 'YearsInCurrentRole+' for employees who work overtime vs those who don't.
    """
    ot_idx = header.index('overTime')
    tenure_idx = header.index('YearsInCurrentRole+')
    
    yes_ot = [r[tenure_idx] for r in rows if r[ot_idx] == 'Yes']
    no_ot = [r[tenure_idx] for r in rows if r[ot_idx] == 'No']
    
    is_noise = certify(yes_ot, no_ot)
    
    avg_yes = sum(yes_ot) / len(yes_ot)
    avg_no = sum(no_ot) / len(no_ot)
    
    return is_noise, avg_yes, avg_no
