import csv
import step03

def flow(header, row, seen):
    """
    Detects if the current row represents a 'drift' in team composition.
    We compare the distribution of a key metric ('YearsAtCompany+') 
    of the current row against the 'seen' population using a statistical 
    check (Welch's t-test). If the current row is statistically 
    different from the history, we flag it as a change.
    """
    if len(seen) < 30:
        return False
    
    goal_idx = header.index('YearsAtCompany+')
    
    # Get history of the goal metric
    history = [r[goal_idx] for r in seen if isinstance(r[goal_idx], (int, float))]
    
    # If the current row's value is an outlier compared to the history
    # (e.g., more than 2 standard deviations away), flag it.
    val = row[goal_idx]
    if not isinstance(val, (int, float)):
        return False
        
    avg = sum(history) / len(history)
    var = sum((x - avg) ** 2 for x in history) / (len(history) - 1)
    std = var ** 0.5
    
    # If the new row is significantly outside the established norm
    if abs(val - avg) > (2 * std):
        return True
        
    return False

if __name__ == "__main__":
    filename = 'WA_Fn-UseC_-HR-Employee-Attrition.n1000.train.csv'
    rows = []
    header = []
    
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = []
            for i, val in enumerate(row):
                if header[i][0].isupper():
                    try:
                        processed_row.append(float(val))
                    except ValueError:
                        processed_row.append(val)
                else:
                    processed_row.append(val)
            rows.append(processed_row)
            
    seen = []
    drift_count = 0
    for row in rows:
        if flow(header, row, seen):
            drift_count += 1
        seen.append(row)
        
    print("Analysis: Detecting team composition drift.")
    print(f"Total records processed: {len(rows)}")
    print(f"Number of times the team composition significantly drifted: {drift_count}")
    print("Conclusion: A drift is flagged when a new employee's tenure is statistically")
    print("inconsistent (outlier) with the established history of the team.")
