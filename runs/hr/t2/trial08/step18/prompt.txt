import csv
import math

def certify(xs, ys):
    """
    Returns True if the two distributions are statistically similar (noise),
    False if they significantly differ. Uses a simple Welch's t-test approach.
    """
    n1, n2 = len(xs), len(ys)
    if n1 < 2 or n2 < 2:
        return True
    
    m1, m2 = sum(xs) / n1, sum(ys) / n2
    v1 = sum((x - m1) ** 2 for x in xs) / (n1 - 1)
    v2 = sum((y - m2) ** 2 for y in ys) / (n2 - 1)
    
    # Standard error of the difference
    se = math.sqrt((v1 / n1) + (v2 / n2))
    if se == 0:
        return m1 == m2
    
    t_stat = abs(m1 - m2) / se
    # Threshold for 95% confidence (approx 1.96 for large samples)
    return t_stat < 1.96

    # Simple heuristic: find columns where the worst rows have a distinct

    # For symbolic, we look for the most frequent value in the worst rows.
    # For numeric, we look for the lowest average.

def justify(header, rows, row):
    """
    Returns (p, reason) where p is the fraction of rows that 'row' is better than,
    and reason is a sentence explaining the ranking.
    """
    # Identify goal columns
    goals = [(i, name) for i, name in enumerate(header) if name.endswith('+') or name.endswith('-')]
    
    # Estimate goal values for the input row using the average from the dataset
    # (Using step02 logic for estimation)
    estimates = {}
    for idx, name in goals:
        vals = [r[idx] for r in rows if isinstance(r[idx], (int, float))]
        estimates[idx] = sum(vals) / len(vals) if vals else 0.0
    
    # Define "better" logic:
    # For '+', higher is better. For '-', lower is better.
    def is_better(r1, r2):
        for idx, name in goals:
            v1 = r1[idx] if r1[idx] != "?" else estimates[idx]
            v2 = r2[idx]
            if name.endswith('+'):
                if v1 < v2: return False
            else:
                if v1 > v2: return False
        return True

    # Calculate fraction of rows that the input row is better than
    better_count = sum(1 for r in rows if is_better(estimates, r))
    p = better_count / len(rows)
    
    reason = f"This team arrangement ranks in the top {int((1-p)*100)}% of all records based on its projected performance across all goal metrics."
    return (p, reason)

import csv

def choose(header, rows, label):
    """
    Finds the best working arrangement by sampling 50 rows, 
    labeling them, and picking the one with the best average goal performance.
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith('+') or name.endswith('-')]
    
    # Randomly sample 50 indices to label
    indices_to_sample = random.sample(range(len(rows)), min(50, len(rows)))
    
    best_idx = -1
    best_score = -float('inf')
    
    for i in indices_to_sample:
        # Get true values for the goals
        true_goals = label(i)
        
        # Calculate a normalized score for this row
        # We want to maximize (+) and minimize (-)
        # Since we don't have global min/max, we use the relative performance 
        # of the sampled set to rank them.
        score = 0
        for j, goal_idx in enumerate(goal_indices):
            val = true_goals[j]
            if header[goal_idx].endswith('+'):
                score += val
            else:
                score -= val
        
        if score > best_score:
            best_score = score
            best_idx = i
            
    return best_idx

import csv

def fix(header, row, levers):
    """
    Recommends changes for a 'bad' row (short tenure) to improve it.
    We pick a lever and move it towards the direction that typically 
    correlates with higher tenure.
    """
    recommendations = {}
    
    # We focus on the levers provided.
    # For numeric, we move towards the higher end of the range.
    # For symbolic, we pick a value that is not the current one.
    for col, allowed in levers.items():
        idx = header.index(col)
        current_val = row[idx]
        
        if isinstance(allowed, (list, tuple)) and not isinstance(allowed[0], (int, float)):
            # Symbolic: pick a value different from current
            options = [v for v in allowed if v != current_val]
            if options:
                recommendations[col] = random.choice(options)
        else:
            # Numeric: move towards the high end of the range
            low, high = allowed
            if current_val < high:
                recommendations[col] = min(high, current_val + (high - low) * 0.1)
        
        # Only need one change to satisfy the requirement
        if recommendations:
            break
            
    return recommendations

    Identifies suspicious rows by checking for logical inconsistencies

    suspicious = []
    
    # Indices for logical checks
    try:
        idx_total = header.index('TotalWorkingYears')
        idx_role = header.index('YearsInCurrentRole+')
        idx_company = header.index('YearsAtCompany+')
        idx_manager = header.index('YearsWithCurrManager+')
    except ValueError:
        return []

    for i, row in enumerate(rows):
        # Logic check: Years in role/company cannot exceed total working years
        # Also check if YearsInCurrentRole > YearsAtCompany
        score = 0
        if row[idx_role] > row[idx_total]:

        if row[idx_company] > row[idx_total]:

            suspicious.append((score, i))
            
    # Sort by score descending (most suspicious first)
    suspicious.sort(key=lambda x: x[0], reverse=True)
    return [item[1] for item in suspicious]

import csv

def race(header, rows, label):
    """
    Compares three strategies for finding the best employee row.
    Budget: 50 labels total.
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith('+') or name.endswith('-')]
    
    def get_score(row_idx):
        vals = label(row_idx)
        score = 0
        for j, goal_idx in enumerate(goal_indices):
            val = vals[j]
            # Normalize: + is better high, - is better low
            if header[goal_idx].endswith('+'):
                score += val
            else:
                score -= val
        return score

    # Strategies
    def strategy_random(budget):
        indices = random.sample(range(len(rows)), budget)
        best_idx = max(indices, key=get_score)
        return best_idx, indices

    def strategy_first_n(budget):
        indices = list(range(budget))
        best_idx = max(indices, key=get_score)
        return best_idx, indices

    def strategy_last_n(budget):
        indices = list(range(len(rows) - budget, len(rows)))
        best_idx = max(indices, key=get_score)
        return best_idx, indices

    # Budget allocation: 3 strategies, 16 labels each (48 total)

import csv
import math

def get_distance(row1, row2, header):
    """
    Calculate Euclidean distance between two rows for numeric columns,
    and Hamming distance for symbolic columns.
    """
    dist = 0.0
    for i, col in enumerate(header):
        if col.endswith('+') or col.endswith('-'):
            continue
        
        v1, v2 = row1[i], row2[i]
        if isinstance(v1, (int, float)) and isinstance(v2, (int, float)):
            # Normalize numeric distance (rough approximation)
            dist += (v1 - v2) ** 2
        else:
            # Symbolic distance
            if v1 != v2:
                dist += 1.0
    return math.sqrt(dist)

def remember(header, rows):
    """
    Finds the 5% of rows most similar to the first row in the dataset.
    """
    if not rows:
        return []
    
    target = rows[0]
    # Calculate distances to the target row
    distances = []
    for i, row in enumerate(rows):
        if i == 0: continue
        d = get_distance(target, row, header)
        distances.append((d, i))
    
    # Sort by distance (ascending)
    distances.sort(key=lambda x: x[0])
    
    # Return top 5%
    limit = max(1, len(rows) // 20)
    return [item[1] for item in distances[:limit]]

import csv

def flow(header, row, seen):
    """
    Detects if the current row represents a 'drift' in team composition.
    We compare the distribution of a key metric ('YearsAtCompany+') 
    of the current row against the 'seen' population using a statistical 
    check (Welch's t-test). If the current row is statistically 
    different from the history, we flag it as a change.
    """
    if len(seen) < 30:
        return False
    
    goal_idx = header.index('YearsAtCompany+')
    
    # Get history of the goal metric
    history = [r[goal_idx] for r in seen if isinstance(r[goal_idx], (int, float))]
    
    # If the current row's value is an outlier compared to the history
    # (e.g., more than 2 standard deviations away), flag it.
    val = row[goal_idx]
    if not isinstance(val, (int, float)):
        return False
        
    avg = sum(history) / len(history)
    var = sum((x - avg) ** 2 for x in history) / (len(history) - 1)
    std = var ** 0.5
    
    # If the new row is significantly outside the established norm
    if abs(val - avg) > (2 * std):
        return True
        
    return False

import csv
import step05

def blame(header, rows):
    """
    Identifies input columns that have the least correlation with the goal 
    'YearsAtCompany+'. Columns that do not help explain the variance in 
    tenure are candidates for removal.
    """
    goal_col = "YearsAtCompany+"
    goal_idx = header.index(goal_col)
    
    # Identify input columns (exclude goals and IDs)
    input_cols = []
    for i, name in enumerate(header):
        if not (name.endswith('+') or name.endswith('-') or 'Number' in name):
            input_cols.append((i, name))
            
    # We calculate the variance of the goal column for each unique value 
    # (or bin) of the input column. If the variance remains high, the 
    # input column is not a good predictor of the goal.
    results = []
    for idx, name in input_cols:
        vals = [r[idx] for r in rows]
        
        # Group goal values by input value
        groups = {}
        for i, val in enumerate(vals):
            if val not in groups:
                groups[val] = []
            groups[val].append(rows[i][goal_idx])
            
        # Calculate average variance within groups
        total_var = 0
        for g in groups.values():
            if len(g) > 1:
                avg = sum(g) / len(g)
                var = sum((x - avg) ** 2 for x in g) / (len(g) - 1)
                total_var += var
        
        # Higher average variance means the column is less informative
        results.append((total_var / len(groups), name))
            
    # Sort by variance descending (least informative first)
    results.sort(key=lambda x: x[0], reverse=True)
    
    # Return the top 5 columns that have the least to do with the goal
    return [s[1] for s in results[:5]]
