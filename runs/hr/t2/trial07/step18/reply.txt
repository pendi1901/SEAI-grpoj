import csv
import step05

def fix(header, row, levers):
    """
    Recommends changes to a 'bad' row to improve tenure metrics.
    We identify which levers are most likely to improve tenure by analyzing
    the population's behavior (using step05's logic) and selecting a lever
    that moves the row toward the 'better' population.
    """
    # Identify goal indices to calculate current score
    goal_indices = [i for i, h in enumerate(header) if '+' in h or '-' in h]
    
    # Get the columns that differentiate the 'best' from the 'worst'
    # We invert the logic of step05 to find what makes people stay (the 'best')
    # instead of what makes them leave.
    def get_tenure_score(r):
        return sum(r[i] for i in goal_indices if '+' in header[i]) - \
               sum(r[i] for i in goal_indices if '-' in header[i])

    # We need a reference for what 'good' looks like.
    # Since we don't have the full dataset passed in, we rely on the provided levers.
    # We pick a lever that is most likely to have a positive impact.
    recommendations = {}
    
    # Heuristic: Prioritize changing 'overTime' to 'No' if available,
    # as it is a common factor in retention.
    if 'overTime' in levers and row[header.index('overTime')] == 'Yes':
        recommendations['overTime'] = 'No'
    elif 'TrainingTimesLastYear' in levers:
        # Suggest increasing training
        low, high = levers['TrainingTimesLastYear']
        recommendations['TrainingTimesLastYear'] = high
    elif 'StockOptionLevel' in levers:
        # Suggest increasing stock options
        low, high = levers['StockOptionLevel']
        recommendations['StockOptionLevel'] = high
    else:
        # Default to the first lever provided
        col = list(levers.keys())[0]
        val = levers[col]
        recommendations[col] = val[1] if isinstance(val, tuple) else val[0]
            
    return recommendations
