import csv

def remember(header, rows):
    """
    Identifies staff who stay 'unusually long'.
    We define 'staying long' by maximizing the goal-oriented columns:
    'YearsAtCompany+', 'YearsInCurrentRole+', and 'YearsWithCurrManager+'.
    We calculate a score based on the sum of these normalized values.
    """
    # Identify indices for goal columns
    idx_at_company = header.index('YearsAtCompany+')
    idx_in_role = header.index('YearsInCurrentRole+')
    idx_with_manager = header.index('YearsWithCurrManager+')

    # Calculate scores for each row
    scores = []
    for i, row in enumerate(rows):
        # Sum the values of the goal columns
        score = row[idx_at_company] + row[idx_in_role] + row[idx_with_manager]
        scores.append((score, i))

    # Sort by score descending
    scores.sort(key=lambda x: x[0], reverse=True)

    # Return top 5% (800 rows * 0.05 = 40 rows)
    limit = max(1, int(len(rows) * 0.05))
    return [item[1] for item in scores[:limit]]

def guess(header, train, test):
    """
    Estimates the first goal column ('YearsAtCompany+') for test rows.
    Uses a simple k-Nearest Neighbors approach based on numeric input columns.
    """
    # Identify goal column index
    goal_idx = header.index('YearsAtCompany+')
    
    # Identify numeric input columns (excluding goals and symbolic)
    input_indices = [
        i for i, name in enumerate(header) 
        if name[0].isupper() and '+' not in name and '-' not in name
    ]

    def get_dist(row1, row2):
        # Euclidean distance on numeric inputs
        dist = 0
        for i in input_indices:
            dist += (row1[i] - row2[i]) ** 2
        return dist ** 0.5

    predictions = []
    for t_row in test:
        # Find nearest neighbor in training set
        best_dist = float('inf')
        best_val = 0
        for train_row in train:
            d = get_dist(t_row, train_row)
            if d < best_dist:
                best_dist = d
                best_val = train_row[goal_idx]
        predictions.append(float(best_val))
    
    return predictions

import csv
import math

def certify(xs, ys):
    """
    Performs a simple Welch's t-test approximation to check if two groups
    are statistically different. Returns True if they are likely the same
    (noise), False if they are significantly different.
    """
    n1, n2 = len(xs), len(ys)
    if n1 < 2 or n2 < 2:
        return True
    
    m1, m2 = sum(xs) / n1, sum(ys) / n2
    v1 = sum((x - m1) ** 2 for x in xs) / (n1 - 1)
    v2 = sum((y - m2) ** 2 for y in ys) / (n2 - 1)
    
    # Standard error of the difference
    se = math.sqrt((v1 / n1) + (v2 / n2))
    if se == 0:
        return m1 == m2
    
    t_stat = abs(m1 - m2) / se
    # Threshold for 95% confidence (approx 1.96)
    return t_stat < 1.96

import csv

def flow(header, row, seen):
    """
    Decides if the current row needs a 'what would keep you here' conversation.
    We use the logic from step01 to determine if the employee has an 
    'unusually long' tenure. If the current row's tenure score is in the 
    top 5% of the historical data (represented by 'seen'), we flag it.
    """
    # Identify goal columns
    idx_at_company = header.index('YearsAtCompany+')
    idx_in_role = header.index('YearsInCurrentRole+')
    idx_with_manager = header.index('YearsWithCurrManager+')

    def get_score(r):
        return r[idx_at_company] + r[idx_in_role] + r[idx_with_manager]

    # Current row score
    current_score = get_score(row)

    # If no history, we can't compare, so default to False
    if not seen:
        return False

    # Calculate scores of all seen rows
    seen_scores = [get_score(r) for r in seen]
    
    # We define 'unusually long' as being in the top 5% of the distribution 
    # seen so far.
    threshold = sorted(seen_scores, reverse=True)[max(0, int(len(seen_scores) * 0.05))]
    
    return current_score >= threshold

import csv

def blame(header, rows):
    """
    Identifies input columns that correlate with low tenure (short stays).
    'Worst' rows are defined as those with the lowest sum of goal columns.
    We compare the average of input features in the bottom 20% of tenure 
    against the rest of the population to find the most significant differences.
    """
    goal_indices = [i for i, h in enumerate(header) if '+' in h or '-' in h]
    input_indices = [i for i, h in enumerate(header) if '+' not in h and '-' not in h]

    def get_tenure_score(row):
        # Higher is better tenure. We want to find why tenure is low.
        return sum(row[i] for i in goal_indices if '+' in header[i]) - \
               sum(row[i] for i in goal_indices if '-' in header[i])

    # Sort rows by tenure score
    scored_rows = sorted(rows, key=get_tenure_score)
    
    # Bottom 20% are the 'worst' (short stays)
    cutoff = max(1, int(len(rows) * 0.2))
    worst = scored_rows[:cutoff]
    rest = scored_rows[cutoff:]

    # Calculate mean difference for numeric inputs
    diffs = []
    for i in input_indices:
        if isinstance(worst[0][i], (int, float)):
            m_worst = sum(r[i] for r in worst) / len(worst)
            m_rest = sum(r[i] for r in rest) / len(rest)
            # We look for features where the worst group deviates most
            diffs.append((abs(m_worst - m_rest), header[i]))

    # Return the top 3 input columns that differentiate the short-stay group
    diffs.sort(key=lambda x: x[0], reverse=True)
    return [d[1] for d in diffs[:3]]
