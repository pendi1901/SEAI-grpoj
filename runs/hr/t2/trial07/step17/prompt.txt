    idx_with_manager = header.index('YearsWithCurrManager+')

    return [item[1] for item in scores[:limit]]

    input_indices = [
        i for i, name in enumerate(header)

    ]

    def get_dist(row1, row2):
        # Euclidean distance on numeric inputs
        dist = 0
        for i in input_indices:
            dist += (row1[i] - row2[i]) ** 2
        return dist ** 0.5

    predictions = []
    for t_row in test:

import csv
import math

def certify(xs, ys):
    """
    Performs a simple Welch's t-test approximation to check if two groups
    are statistically different. Returns True if they are likely the same
    (noise), False if they are significantly different.
    """
    n1, n2 = len(xs), len(ys)
    if n1 < 2 or n2 < 2:
        return True
    
    m1, m2 = sum(xs) / n1, sum(ys) / n2
    v1 = sum((x - m1) ** 2 for x in xs) / (n1 - 1)
    v2 = sum((y - m2) ** 2 for y in ys) / (n2 - 1)
    
    # Standard error of the difference
    se = math.sqrt((v1 / n1) + (v2 / n2))
    if se == 0:
        return m1 == m2
    
    t_stat = abs(m1 - m2) / se
    # Threshold for 95% confidence (approx 1.96)
    return t_stat < 1.96

import csv

def blame(header, rows):
    """
    Identifies input columns that correlate with low tenure (short stays).
    'Worst' rows are defined as those with the lowest sum of goal columns.
    We compare the average of input features in the bottom 20% of tenure 
    against the rest of the population to find the most significant differences.
    """
    goal_indices = [i for i, h in enumerate(header) if '+' in h or '-' in h]
    input_indices = [i for i, h in enumerate(header) if '+' not in h and '-' not in h]

    def get_tenure_score(row):
        # Higher is better tenure. We want to find why tenure is low.
        return sum(row[i] for i in goal_indices if '+' in header[i]) - \
               sum(row[i] for i in goal_indices if '-' in header[i])

    # Sort rows by tenure score
    scored_rows = sorted(rows, key=get_tenure_score)
    
    # Bottom 20% are the 'worst' (short stays)
    cutoff = max(1, int(len(rows) * 0.2))
    worst = scored_rows[:cutoff]
    rest = scored_rows[cutoff:]

    # Calculate mean difference for numeric inputs
    diffs = []
    for i in input_indices:
        if isinstance(worst[0][i], (int, float)):
            m_worst = sum(r[i] for r in worst) / len(worst)
            m_rest = sum(r[i] for r in rest) / len(rest)
            # We look for features where the worst group deviates most
            diffs.append((abs(m_worst - m_rest), header[i]))

    # Return the top 3 input columns that differentiate the short-stay group
    diffs.sort(key=lambda x: x[0], reverse=True)
    return [d[1] for d in diffs[:3]]

        score = 0
        for i in goal_indices:
            val = r[i]

                score += val
            else:
                score -= val
        return score

    # Score the estimated row and the population
    target_score = estimated_goals

import csv

def spot(header, rows):
    """
    Identifies rows that look 'wrong' by checking for outliers in numeric columns.
    A row is considered suspicious if its values are significantly far from the 
    mean (more than 3 standard deviations) for any numeric input column.
    """
    numeric_indices = [
        i for i, name in enumerate(header) 
        if name[0].isupper() and '+' not in name and '-' not in name
    ]
    
    # Calculate means and standard deviations for each numeric column
    stats = {}
    for i in numeric_indices:
        vals = [r[i] for r in rows if isinstance(r[i], (int, float))]
        if not vals:
            continue
        mean = sum(vals) / len(vals)
        std = (sum((x - mean) ** 2 for x in vals) / len(vals)) ** 0.5
        stats[i] = (mean, std)
    
    # Score rows based on how many standard deviations they are from the mean
    # We sum the absolute Z-scores across all numeric columns
    suspicion_scores = []
    for row_idx, row in enumerate(rows):
        score = 0
        for i in numeric_indices:
            if i in stats:
                mean, std = stats[i]
                if std > 0:
                    z_score = abs(row[i] - mean) / std
                    if z_score > 3:  # Threshold for outlier
                        score += z_score
        if score > 0:
            suspicion_scores.append((score, row_idx))
            
    # Sort by suspicion score descending
    suspicion_scores.sort(key=lambda x: x[0], reverse=True)
    
    return [item[1] for item in suspicion_scores]

import csv
import step02

def remember(header, rows):
    """
    Finds the 5% of rows most similar to the first row (the 'target').
    Similarity is defined by the inverse of the Euclidean distance on 
    numeric input columns, as established in step02.
    """
    # Identify numeric input columns (excluding goals and symbolic)
    input_indices = [
        i for i, name in enumerate(header) 
        if name[0].isupper() and '+' not in name and '-' not in name
    ]

    def get_dist(row1, row2):
        dist = 0
        for i in input_indices:
            dist += (row1[i] - row2[i]) ** 2
        return dist ** 0.5

    # Target is the first row
    target = rows[0]
    
    # Calculate distances to all other rows
    distances = []
    for i in range(1, len(rows)):
        d = get_dist(target, rows[i])
        distances.append((d, i))
    
    # Sort by distance ascending (closest first)
    distances.sort(key=lambda x: x[0])
    
    # Return top 5% (800 rows * 0.05 = 40 rows)
    limit = max(1, int(len(rows) * 0.05))
    return [item[1] for item in distances[:limit]]

import csv
import math

def flow(header, row, seen):
    """
    Detects if the current team (represented by the 'seen' history) has shifted 
    significantly. We compare the distribution of the 'YearsAtCompany+' 
    goal column in the 'seen' history against the current row.
    
    If the current row's tenure is more than 2 standard deviations away from 
    the mean of the seen history, we flag it as a shift in team composition.
    """
    if not seen:
        return False

    idx_years = header.index('YearsAtCompany+')
    
    # Extract numeric tenure values from seen rows
    tenures = [r[idx_years] for r in seen if isinstance(r[idx_years], (int, float))]
    
    if len(tenures) < 10:
        return False
    
    mean = sum(tenures) / len(tenures)
    variance = sum((x - mean) ** 2 for x in tenures) / len(tenures)
    std = math.sqrt(variance)
    
    if std == 0:
        return False
    
    # Flag if the new row is an outlier (more than 2 standard deviations)
    current_val = row[idx_years]
    z_score = abs(current_val - mean) / std
    
    return z_score > 2.0

import csv
import step05

def blame(header, rows):
    """
    Identifies input columns that have 'nothing to do' with tenure.
    We define these as columns that show the least difference between 
    the 'worst' 20% of employees and the rest of the population.
    """
    goal_indices = [i for i, h in enumerate(header) if '+' in h or '-' in h]
    input_indices = [i for i, h in enumerate(header) if '+' not in h and '-' not in h]

    def get_tenure_score(row):
        # Higher is better tenure.
        return sum(row[i] for i in goal_indices if '+' in header[i]) - \
               sum(row[i] for i in goal_indices if '-' in header[i])

    # Sort rows by tenure score
    scored_rows = sorted(rows, key=get_tenure_score)
    
    # Bottom 20% are the 'worst' (short stays)
    cutoff = max(1, int(len(rows) * 0.2))
    worst = scored_rows[:cutoff]
    rest = scored_rows[cutoff:]

    # Calculate mean difference for all input columns
    # We look for features where the difference is minimal (near zero)
    diffs = []
    for i in input_indices:
        col_name = header[i]
        if isinstance(worst[0][i], (int, float)):
            m_worst = sum(r[i] for r in worst) / len(worst)
            m_rest = sum(r[i] for r in rest) / len(rest)
            diffs.append((abs(m_worst - m_rest), col_name))
        else:
            # For symbolic, compare frequency distributions
            worst_vals = [r[i] for r in worst]
            rest_vals = [r[i] for r in rest]
            # Simple metric: overlap in distribution
            unique = set(worst_vals + rest_vals)
            overlap = 0
            for val in unique:
                p_worst = worst_vals.count(val) / len(worst_vals)
                p_rest = rest_vals.count(val) / len(rest_vals)
                overlap += min(p_worst, p_rest)
            diffs.append((1.0 - overlap, col_name))

    # Sort by difference ascending (smallest difference = least relevant)
    diffs.sort(key=lambda x: x[0])
    
    # Return the top 5 columns that have the least impact on tenure
    return [d[1] for d in diffs[:5]]

                score += val
            else:
