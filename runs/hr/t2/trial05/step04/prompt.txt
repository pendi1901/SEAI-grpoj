def remember(header, rows):
    """
    Identifies staff who stay 'unusually long'.
    'YearsAtCompany+' is a goal to maximize.
    We return the indices of the top 5% (or at least one) of rows 
    based on 'YearsAtCompany+'.
    """
    # Identify index for 'YearsAtCompany+'
    try:
        idx = header.index('YearsAtCompany+')
    except ValueError:
        return [0]

    # Create list of (value, original_index)
    scored = []
    for i, row in enumerate(rows):
        val = row[idx]
        if isinstance(val, (int, float)):
            scored.append((val, i))
    
    # Sort descending to find those who stay longest
    scored.sort(key=lambda x: x[0], reverse=True)
    
    # Calculate 5% limit
    limit = max(1, len(rows) // 20)
    
    # Return indices of the top performers
    return [item[1] for item in scored[:limit]]

def guess(header, train, test):
    """
    Estimates the first goal column ('YearsAtCompany+') for test rows.
    Uses a simple k-Nearest Neighbors approach based on numeric input columns.
    """
    # Identify the first goal column
    goal_idx = -1
    for i, name in enumerate(header):
        if name.endswith('+') or name.endswith('-'):
            goal_idx = i
            break
    
    # Identify numeric input columns (excluding goals)
    # Ensure we only use columns that are numeric in both train and test
    input_indices = [
        i for i, name in enumerate(header) 
        if name[0].isupper() and i != goal_idx
    ]
    
    predictions = []
    for t_row in test:
        distances = []
        for r_row in train:
            dist = 0
            for i in input_indices:
                # Ensure both values are numeric before subtraction
                val_t = t_row[i]
                val_r = r_row[i]
                if isinstance(val_t, (int, float)) and isinstance(val_r, (int, float)):
                    dist += (val_t - val_r) ** 2
                else:
                    # Penalty for non-numeric mismatch
                    dist += 1e9
            distances.append((dist**0.5, r_row[goal_idx]))
        
        # Sort by distance and take the average of the 5 nearest neighbors
        distances.sort(key=lambda x: x[0])
        k = 5
        neighbors = distances[:k]
        avg_val = sum(n[1] for n in neighbors) / k
        predictions.append(float(avg_val))
        
    return predictions

import csv
import math

def certify(xs, ys):
    """
    Returns True if the two lists of numbers are statistically similar 
    (i.e., they do not differ significantly), False otherwise.
    Uses a simple T-test approach (Welch's T-test logic).
    """
    n1, n2 = len(xs), len(ys)
    if n1 < 2 or n2 < 2:
        return True
    
    m1, m2 = sum(xs) / n1, sum(ys) / n2
    v1 = sum((x - m1) ** 2 for x in xs) / (n1 - 1)
    v2 = sum((y - m2) ** 2 for y in ys) / (n2 - 1)
    
    denom = math.sqrt((v1 / n1) + (v2 / n2))
    if denom == 0:
        return m1 == m2
    
    t_stat = abs(m1 - m2) / denom
    # Threshold for 95% confidence (approx 2.0 for large samples)
    return t_stat < 2.0

def analyze_attrition(header, rows):
    """
    Compares 'YearsAtCompany+' for employees who work overtime vs those who don't.
    """
    ot_idx = header.index('overTime')
    years_idx = header.index('YearsAtCompany+')
    
    yes_ot = [row[years_idx] for row in rows if row[ot_idx] == 'Yes']
    no_ot = [row[years_idx] for row in rows if row[ot_idx] == 'No']
    
    is_similar = certify(yes_ot, no_ot)
    
    return is_similar, len(yes_ot), len(no_ot)
