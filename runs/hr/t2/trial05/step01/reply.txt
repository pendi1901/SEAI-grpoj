import csv

def remember(header, rows):
    """
    Identifies staff who stay 'unusually long'.
    'YearsAtCompany+' is a goal to maximize.
    We return the indices of the top 5% (or at least one) of rows 
    based on 'YearsAtCompany+'.
    """
    # Identify index for 'YearsAtCompany+'
    try:
        idx = header.index('YearsAtCompany+')
    except ValueError:
        return [0]

    # Create list of (value, original_index)
    scored = []
    for i, row in enumerate(rows):
        val = row[idx]
        if isinstance(val, (int, float)):
            scored.append((val, i))
    
    # Sort descending to find those who stay longest
    scored.sort(key=lambda x: x[0], reverse=True)
    
    # Calculate 5% limit
    limit = max(1, len(rows) // 20)
    
    # Return indices of the top performers
    return [item[1] for item in scored[:limit]]

if __name__ == "__main__":
    filename = 'WA_Fn-UseC_-HR-Employee-Attrition.n1000.train.csv'
    rows = []
    header = []
    
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = []
            for i, val in enumerate(row):
                # Convert numeric columns (starting with uppercase)
                if header[i][0].isupper():
                    try:
                        processed_row.append(float(val))
                    except ValueError:
                        processed_row.append(val)
                else:
                    processed_row.append(val)
            rows.append(processed_row)
            
    indices = remember(header, rows)
    
    print("Staff members who stay unusually long (Top 5% by YearsAtCompany+):")
    print(f"{'Index':<10} | {'YearsAtCompany':<15} | {'JobRole'}")
    print("-" * 45)
    
    role_idx = header.index('jobRole')
    years_idx = header.index('YearsAtCompany+')
    
    for i in indices:
        print(f"{i:<10} | {rows[i][years_idx]:<15} | {rows[i][role_idx]}")
