import csv

def remember(header, rows):
    """
    Identifies staff who stay 'unusually long'.
    'YearsAtCompany+' is a goal to maximize.
    We return the indices of the top 5% of rows based on this metric.
    """
    # Identify the index for 'YearsAtCompany+'
    target_col = "YearsAtCompany+"
    try:
        idx = header.index(target_col)
    except ValueError:
        return []

    # Create a list of (value, original_index)
    indexed_rows = []
    for i, row in enumerate(rows):
        val = row[idx]
        if isinstance(val, (int, float)):
            indexed_rows.append((val, i))

    # Sort by value descending (maximize)
    indexed_rows.sort(key=lambda x: x[0], reverse=True)

    # Calculate 5% limit (at least 1, at most 5% of total rows)
    n = len(rows)
    limit = max(1, int(n * 0.05))
    
    # Return the indices of the top rows
    return [item[1] for item in indexed_rows[:limit]]

import csv
import math

def certify(xs, ys):
    """
    Returns True if the two lists of numbers are statistically similar 
    (the difference is likely noise), False if they significantly differ.
    Uses a simple T-test approach (assuming equal variance).
    """
    n1, n2 = len(xs), len(ys)
    if n1 < 2 or n2 < 2:
        return True
    
    mean1, mean2 = sum(xs) / n1, sum(ys) / n2
    var1 = sum((x - mean1) ** 2 for x in xs) / (n1 - 1)
    var2 = sum((y - mean2) ** 2 for y in ys) / (n2 - 1)
    
    pooled_std = math.sqrt(((n1 - 1) * var1 + (n2 - 1) * var2) / (n1 + n2 - 2))
    if pooled_std == 0:
        return mean1 == mean2
    
    t_stat = (mean1 - mean2) / (pooled_std * math.sqrt(1/n1 + 1/n2))
    # For a 95% confidence level, critical t-value is approx 1.96
    return abs(t_stat) < 1.96

import csv

def flow(header, row, seen):
    """
    Decides if the current employee needs a 'what would keep you here' 
    conversation. We target those with low 'YearsAtCompany+' compared 
    to the average of those seen so far, or those who are outliers.
    """
    yac_idx = header.index("YearsAtCompany+")
    
    # If this is the first row, we can't compare, so let it go
    if not seen:
        return False
    
    # Calculate the mean of 'YearsAtCompany+' for all seen rows
    seen_vals = [r[yac_idx] for r in seen if isinstance(r[yac_idx], (int, float))]
    if not seen_vals:
        return False
    
    mean_yac = sum(seen_vals) / len(seen_vals)
    current_yac = row[yac_idx]
    
    # If the current employee has significantly lower tenure than the average
    # of those seen, they are a candidate for a retention conversation.
    # Using a simple threshold: if they are in the bottom 25% of the mean.
    if current_yac < (mean_yac * 0.75):
        return True
        
    return False
