import csv
import math

def certify(xs, ys):
    """
    Returns True if the two lists of numbers are statistically similar 
    (the difference is likely noise), False if they significantly differ.
    Uses a simple T-test approach (assuming equal variance).
    """
    n1, n2 = len(xs), len(ys)
    if n1 < 2 or n2 < 2:
        return True
    
    mean1, mean2 = sum(xs) / n1, sum(ys) / n2
    var1 = sum((x - mean1) ** 2 for x in xs) / (n1 - 1)
    var2 = sum((y - mean2) ** 2 for y in ys) / (n2 - 1)
    
    pooled_std = math.sqrt(((n1 - 1) * var1 + (n2 - 1) * var2) / (n1 + n2 - 2))
    if pooled_std == 0:
        return mean1 == mean2
    
    t_stat = (mean1 - mean2) / (pooled_std * math.sqrt(1/n1 + 1/n2))
    # For a 95% confidence level, critical t-value is approx 1.96
    return abs(t_stat) < 1.96

def spot(header, rows):
    """
    Identifies suspicious rows by checking for outliers in numeric columns.
    A row is suspicious if its values are significantly distant from the 
    median of the column, suggesting a data entry error.
    """
    # Identify numeric columns
    numeric_indices = [i for i, name in enumerate(header) if name[0].isupper()]
    
    # Calculate column medians
    medians = {}
    for i in numeric_indices:
        vals = sorted([r[i] for r in rows if isinstance(r[i], (int, float))])
        medians[i] = vals[len(vals) // 2]
    
    # Calculate suspicion score for each row
    # Score = sum of absolute deviations from median
    scores = []
    for idx, row in enumerate(rows):
        score = 0
        for i in numeric_indices:
            val = row[i]
            if isinstance(val, (int, float)):
                # Simple outlier detection: deviation from median
                # We use a simple threshold or just raw deviation
                score += abs(val - medians[i])
        scores.append((score, idx))
    
    # Sort by score descending (most suspicious first)
    scores.sort(key=lambda x: x[0], reverse=True)
    
    # Return the top 10 most suspicious row indices
    return [item[1] for item in scores[:10]]

def race(header, rows, label):
    """
    Compares three strategies for finding the best employee row.
    Budget: 50 labels total.
    """
    goal_idx = next(i for i, name in enumerate(header) if name.endswith('+') or name.endswith('-'))
    is_maximize = header[goal_idx].endswith('+')
    
    # Track labels used to respect the 50-label budget
    labeled_data = {}
    def get_label(i):
        if i not in labeled_data:
            if len(labeled_data) >= 50:
                return None
            labeled_data[i] = label(i)[0]
        return labeled_data[i]

    # Strategy 1: Random
    def strategy_random():
        indices = random.sample(range(len(rows)), 16)
        best_idx, best_val = -1, None
        for i in indices:
            val = get_label(i)
            if best_val is None or (is_maximize and val > best_val) or (not is_maximize and val < best_val):
                best_val, best_idx = val, i
        return best_idx, best_val

    # Strategy 2: Top 5% (using step01 logic)
    def strategy_top_percent():
        # step01.remember returns indices based on the full dataset
        indices = step01.remember(header, rows)
        # Sample a subset of these to stay within budget
        sample = random.sample(indices, min(len(indices), 17))
        best_idx, best_val = -1, None

def remember(header, rows):
    """
    Finds the 5% of rows most similar to the first row in the dataset.
    Similarity is based on Euclidean distance of numeric input columns.
    """
    # Identify goal column
    goal_idx = next(i for i, name in enumerate(header) if name.endswith('+') or name.endswith('-'))
    
    # Identify numeric input columns
    input_indices = [i for i, name in enumerate(header) if i != goal_idx and name[0].isupper()]
    
    # Target row is the first one
    target = rows[0]
    
    def get_dist(row1, row2):
        d = 0
        for i in input_indices:
            # Handle potential missing values or non-numeric data
            if isinstance(row1[i], (int, float)) and isinstance(row2[i], (int, float)):
                d += (row1[i] - row2[i])**2
        return d**0.5

    # Calculate distances for all rows (excluding the target itself)
    distances = []
    for i in range(1, len(rows)):
        dist = get_dist(target, rows[i])
        distances.append((dist, i))
    
    # Sort by distance ascending (most similar first)
    distances.sort(key=lambda x: x[0])
    
    # Return the indices of the top 5%
    n = len(rows)
    limit = max(1, int(n * 0.05))
    return [item[1] for item in distances[:limit]]
