import csv
import step02

def justify(header, rows, row):
    """
    Estimates the goal for the given row and calculates its rank 
    relative to the existing rows.
    """
    # Identify goal column
    goal_idx = next(i for i, name in enumerate(header) if name.endswith('+') or name.endswith('-'))
    is_maximize = header[goal_idx].endswith('+')
    
    # Filter rows to ensure we only use rows where the goal is a number
    valid_rows = [r for r in rows if isinstance(r[goal_idx], (int, float))]
    
    # Estimate the goal value for the input row using the KNN logic from step02
    # We must ensure the input row passed to guess() has numeric values for input columns
    # The harness provides '?' for the goal, but we need to ensure inputs are clean.
    prediction = step02.guess(header, valid_rows, [row])[0]
    
    # Compare prediction against all valid rows
    better_count = 0
    total = len(valid_rows)
    
    for r in valid_rows:
        val = r[goal_idx]
        if is_maximize:
            if prediction > val:
                better_count += 1
        else:
            if prediction < val:
                better_count += 1
                
    p = better_count / total
    reason = f"This team arrangement ranks in the top {int((1-p)*100)}% of all records because its estimated performance is superior to {int(p*100)}% of the existing staff data."
    
    return (p, reason)

if __name__ == "__main__":
    filename = 'WA_Fn-UseC_-HR-Employee-Attrition.n1000.train.csv'
    rows = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = []
            for i, val in enumerate(row):
                if header[i][0].isupper():
                    try:
                        processed_row.append(float(val))
                    except ValueError:
                        processed_row.append(val)
                else:
                    processed_row.append(val)
            rows.append(processed_row)

    # Test with the first row, masking its goal
    goal_idx = next(i for i, name in enumerate(header) if name.endswith('+') or name.endswith('-'))
    test_row = list(rows[0])
    test_row[goal_idx] = "?"
    
    p, reason = justify(header, rows, test_row)
    print(f"Rank: {p:.4f}")
    print(f"Reason: {reason}")
