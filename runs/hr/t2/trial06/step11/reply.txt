import csv
import step01

def remember(header, rows):
    """
    Finds the 5% of employees most similar to the first employee in the dataset.
    Similarity is calculated using Euclidean distance on numeric input columns.
    """
    # Identify numeric input columns (exclude goals)
    input_indices = [
        i for i, name in enumerate(header) 
        if name[0].isupper() and not name.endswith('+') and not name.endswith('-')
    ]
    
    target = rows[0]
    
    def distance(row1, row2):
        d2 = 0
        for i in input_indices:
            # Normalize distance by range if possible, or simple squared diff
            d2 += (row1[i] - row2[i]) ** 2
        return d2 ** 0.5

    # Calculate distances to the first row
    distances = []
    for idx, row in enumerate(rows):
        if idx == 0: continue
        distances.append((distance(target, row), idx))
    
    # Sort by distance (ascending)
    distances.sort(key=lambda x: x[0])
    
    # Return top 5% (max 40 rows)
    limit = max(1, int(len(rows) * 0.05))
    return [item[1] for item in distances[:limit]]

if __name__ == "__main__":
    filename = 'WA_Fn-UseC_-HR-Employee-Attrition.n1000.train.csv'
    rows = []
    header = []
    
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = []
            for i, val in enumerate(row):
                if header[i][0].isupper():
                    try:
                        processed_row.append(float(val))
                    except ValueError:
                        processed_row.append(0.0)
                else:
                    processed_row.append(val)
            rows.append(processed_row)
            
    best_indices = remember(header, rows)
    tenure_indices = [i for i, name in enumerate(header) if name.endswith('+')]
    
    print("Employees most similar to the first record and their tenure:")
    for idx in best_indices:
        row = rows[idx]
        tenure_vals = [row[i] for i in tenure_indices]
        print(f"Row {idx}: Tenure metrics {tenure_vals}")
