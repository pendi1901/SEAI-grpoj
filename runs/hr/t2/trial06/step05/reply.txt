import csv
import step01

def blame(header, rows):
    """
    Identifies input columns that correlate with low tenure.
    'Worst' rows are those with the lowest sum of '+' goal columns.
    We compare the distribution of input values in the worst 10% 
    against the rest of the population.
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith('+')]
    input_indices = [i for i, name in enumerate(header) 
                     if not name.endswith('+') and not name.endswith('-')]
    
    # Score rows: lower sum of '+' goals is 'worse'
    scored = []
    for idx, row in enumerate(rows):
        score = sum(row[i] for i in goal_indices)
        scored.append((score, row))
    
    scored.sort(key=lambda x: x[0])
    limit = max(1, int(len(rows) * 0.10))
    worst = [x[1] for x in scored[:limit]]
    rest = [x[1] for x in scored[limit:]]
    
    # Simple heuristic: find input columns where the mean/mode 
    # in 'worst' differs significantly from 'rest'
    candidates = []
    for i in input_indices:
        col_name = header[i]
        # Only check symbolic or numeric inputs
        if isinstance(worst[0][i], (int, float)):
            mean_w = sum(r[i] for r in worst) / len(worst)
            mean_r = sum(r[i] for r in rest) / len(rest)
            if abs(mean_w - mean_r) > 0.5: # Significant shift
                candidates.append(col_name)
        else:
            # For symbolic, check if the most frequent value in worst 
            # is different from the most frequent in rest
            mode_w = max(set(r[i] for r in worst), key=lambda x: [r[i] for r in worst].count(x))
            mode_r = max(set(r[i] for r in rest), key=lambda x: [r[i] for r in rest].count(x))
            if mode_w != mode_r:
                candidates.append(col_name)
                
    return candidates[:3] # Return top 3 explanatory factors

if __name__ == "__main__":
    filename = 'WA_Fn-UseC_-HR-Employee-Attrition.n1000.train.csv'
    rows = []
    header = []
    
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = []
            for i, val in enumerate(row):
                if header[i][0].isupper():
                    try:
                        processed_row.append(float(val))
                    except ValueError:
                        processed_row.append(0.0)
                else:
                    processed_row.append(val)
            rows.append(processed_row)
            
    factors = blame(header, rows)
    print("Factors explaining short employee tenure:")
    for f in factors:
        print(f"- {f}")
