import csv
import step09

def spot(header, rows):
    """
    Identifies rows that are 'unlike anyone we have employed'.
    We define this by calculating the distance of each row to the 
    centroid of the population. Rows that are extreme outliers 
    (high distance) are flagged as suspicious.
    """
    # Identify numeric input columns (exclude goals)
    input_indices = [
        i for i, name in enumerate(header) 
        if name[0].isupper() and not name.endswith('+') and not name.endswith('-')
    ]
    
    # Calculate mean and std for each input column
    stats = {}
    for i in input_indices:
        vals = [row[i] for row in rows if isinstance(row[i], (int, float))]
        if not vals: continue
        mean = sum(vals) / len(vals)
        std = (sum((x - mean) ** 2 for x in vals) / len(vals)) ** 0.5
        stats[i] = (mean, std if std > 0 else 1.0)
        
    # Calculate a 'strangeness' score for each row
    # The score is the sum of squared z-scores of input columns
    scores = []
    for idx, row in enumerate(rows):
        dist = 0
        for i in input_indices:
            if i in stats:
                mean, std = stats[i]
                z = (row[i] - mean) / std
                dist += z**2
        scores.append((dist, idx))
        
    # Sort by strangeness descending
    scores.sort(key=lambda x: x[0], reverse=True)
    
    # Return indices of the top 5% most 'unlike' rows
    limit = max(1, int(len(rows) * 0.05))
    return [item[1] for item in scores[:limit]]

if __name__ == "__main__":
    filename = 'WA_Fn-UseC_-HR-Employee-Attrition.n1000.train.csv'
    rows = []
    header = []
    
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = []
            for i, val in enumerate(row):
                if header[i][0].isupper():
                    try:
                        processed_row.append(float(val))
                    except ValueError:
                        processed_row.append(0.0)
                else:
                    processed_row.append(val)
            rows.append(processed_row)
            
    outliers = spot(header, rows)
    
    print("Analysis: Identifying 'unlike' employees (potential estimation risks).")
    print("The following rows represent employees who are statistically unlike our historical population.")
    print("Estimates for these individuals may be unreliable:")
    for idx in outliers[:5]:
        print(f"- Row {idx} is highly atypical.")
    if len(outliers) > 5:
        print(f"... and {len(outliers) - 5} others.")
