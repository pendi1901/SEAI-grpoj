import csv
import step03

def guess(header, train, test):
    """
    Estimates the first goal column (YearsAtCompany+) for Sales employees
    by comparing the average tenure of Sales employees who work overtime
    vs those who do not.
    """
    dept_idx = header.index('department')
    ot_idx = header.index('overTime')
    tenure_idx = header.index('YearsAtCompany+')
    
    # Calculate average tenure for Sales employees by overtime status
    sales_ot_yes = [r[tenure_idx] for r in train if r[dept_idx] == 'Sales' and r[ot_idx] == 'Yes']
    sales_ot_no = [r[tenure_idx] for r in train if r[dept_idx] == 'Sales' and r[ot_idx] == 'No']
    
    avg_ot_yes = sum(sales_ot_yes) / len(sales_ot_yes) if sales_ot_yes else 0.0
    avg_ot_no = sum(sales_ot_no) / len(sales_ot_no) if sales_ot_no else 0.0
    
    predictions = []
    for row in test:
        # If the test row is Sales, predict based on the 'No' overtime scenario
        if row[dept_idx] == 'Sales':
            predictions.append(avg_ot_no)
        else:
            # Fallback for non-Sales: use global average of first goal column
            predictions.append(sum(r[tenure_idx] for r in train) / len(train))
            
    return predictions

if __name__ == "__main__":
    filename = 'WA_Fn-UseC_-HR-Employee-Attrition.n1000.train.csv'
    rows = []
    header = []
    
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = []
            for i, val in enumerate(row):
                if header[i][0].isupper():
                    try:
                        processed_row.append(float(val))
                    except ValueError:
                        processed_row.append(0.0)
                else:
                    processed_row.append(val)
            rows.append(processed_row)
            
    # Split data
    train_data = rows[:700]
    test_data = rows[700:]
    
    # Mask goals
    goal_indices = [i for i, name in enumerate(header) if name.endswith('+') or name.endswith('-')]
    masked_test = []
    for row in test_data:
        new_row = list(row)
        for idx in goal_indices:
            new_row[idx] = "?"
        masked_test.append(new_row)
        
    predictions = guess(header, train_data, masked_test)
    
    print("Impact Analysis: Cutting overtime in Sales")
    print("Based on historical data, employees in Sales who do not work overtime")
    print(f"have an average tenure of {predictions[0]:.2f} years.")
    print("Removing overtime requirements is projected to stabilize tenure at this level.")
