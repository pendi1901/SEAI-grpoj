import csv

def remember(header, rows):
    """
    Identifies staff who stay 'unusually long'.
    We define 'staying long' by the goal columns ending in '+':
    'YearsAtCompany+', 'YearsInCurrentRole+', 'YearsWithCurrManager+'
    We calculate a score based on the sum of these values and return
    the indices of the top 5% (max 40 rows for 800 records).
    """
    # Identify indices of goal columns ending in '+'
    goal_indices = [i for i, name in enumerate(header) if name.endswith('+')]
    
    scored_rows = []
    for idx, row in enumerate(rows):
        # Calculate score as the sum of the '+' goal columns
        score = sum(row[i] for i in goal_indices)
        scored_rows.append((score, idx))
    
    # Sort by score descending
    scored_rows.sort(key=lambda x: x[0], reverse=True)
    
    # Return top 5% (800 * 0.05 = 40 rows)
    limit = max(1, int(len(rows) * 0.05))
    return [item[1] for item in scored_rows[:limit]]

import csv
import step01

def certify(xs, ys):
    """
    Returns True if the two distributions are statistically similar 
    (no significant difference), False if they differ.
    Uses a simple mean comparison relative to the standard deviation.
    """
    if not xs or not ys:
        return True
    
    mean_x = sum(xs) / len(xs)
    mean_y = sum(ys) / len(ys)
    
    # Combined standard deviation
    combined = xs + ys
    mean_all = sum(combined) / len(combined)
    var = sum((x - mean_all) ** 2 for x in combined) / len(combined)
    std = var ** 0.5
    
    # If the difference in means is less than 0.5 standard deviations, 
    # we consider it noise.
    if std == 0:
        return mean_x == mean_y
    return abs(mean_x - mean_y) < (0.5 * std)

def flow(header, row, seen):
    """
    Decides if the current row needs a 'what would keep you here' conversation.
    We identify employees with low tenure (YearsAtCompany+) compared to the 
    average of the 'seen' population, suggesting they are at risk of leaving.
    """
    tenure_idx = header.index('YearsAtCompany+')
    
    # If this is the first row, we can't compare, so let it go
    if not seen:
        return False
    
    # Calculate average tenure of seen employees
    tenures = [r[tenure_idx] for r in seen]
    avg_tenure = sum(tenures) / len(tenures)
    
    # If current employee's tenure is significantly lower than the average 
    # (e.g., less than 50% of the average), flag for retention conversation.
    # We use a threshold to ensure we don't flag everyone.
    if row[tenure_idx] < (0.5 * avg_tenure):
        return True
    
    return False

import csv
import step01

def blame(header, rows):
    """
    Identifies input columns that correlate with low tenure.
    'Worst' rows are those with the lowest sum of '+' goal columns.
    We compare the distribution of input values in the worst 10% 
    against the rest of the population.
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith('+')]
    input_indices = [i for i, name in enumerate(header) 
                     if not name.endswith('+') and not name.endswith('-')]
    
    # Score rows: lower sum of '+' goals is 'worse'
    scored = []
    for idx, row in enumerate(rows):
        score = sum(row[i] for i in goal_indices)
        scored.append((score, row))
    
    scored.sort(key=lambda x: x[0])
    limit = max(1, int(len(rows) * 0.10))
    worst = [x[1] for x in scored[:limit]]
    rest = [x[1] for x in scored[limit:]]
    
    # Simple heuristic: find input columns where the mean/mode 
    # in 'worst' differs significantly from 'rest'
    candidates = []
    for i in input_indices:
        col_name = header[i]
        # Only check symbolic or numeric inputs
        if isinstance(worst[0][i], (int, float)):
            mean_w = sum(r[i] for r in worst) / len(worst)
            mean_r = sum(r[i] for r in rest) / len(rest)
            if abs(mean_w - mean_r) > 0.5: # Significant shift
                candidates.append(col_name)
        else:
            # For symbolic, check if the most frequent value in worst 
            # is different from the most frequent in rest
            mode_w = max(set(r[i] for r in worst), key=lambda x: [r[i] for r in worst].count(x))
            mode_r = max(set(r[i] for r in rest), key=lambda x: [r[i] for r in rest].count(x))
            if mode_w != mode_r:
                candidates.append(col_name)
                
    return candidates[:3] # Return top 3 explanatory factors
