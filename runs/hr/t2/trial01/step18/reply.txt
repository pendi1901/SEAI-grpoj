import csv
import step06
import step08

def fix(header, row, levers):
    """
    Recommends changes to a 'bad' row to improve its tenure score,
    excluding 'MonthlyIncome' as per the customer's constraint.
    """
    # Filter out MonthlyIncome from the levers if it exists
    if 'MonthlyIncome' in levers:
        levers = {k: v for k, v in levers.items() if k != 'MonthlyIncome'}
    
    # Reuse the logic from step08 to find the best lever adjustment
    return step08.fix(header, row, levers)

if __name__ == "__main__":
    filename = 'WA_Fn-UseC_-HR-Employee-Attrition.n1000.train.csv'
    rows = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = []
            for i, val in enumerate(row):
                if header[i][0].isupper():
                    processed_row.append(float(val))
                else:
                    processed_row.append(val)
            rows.append(processed_row)

    # Define levers excluding MonthlyIncome
    levers = {
        'businessTravel': ['Non-Travel', 'Travel_Frequently', 'Travel_Rarely'],
        'overTime': ['No', 'Yes'],
        'TrainingTimesLastYear': (0.0, 6.0),
        'PercentSalaryHike': (11.0, 25.0),
        'StockOptionLevel': (0.0, 3.0)
    }

    # Identify a 'bad' row (e.g., one with low tenure score)
    scores = [(step06.get_score(header, r), i) for i, r in enumerate(rows)]
    scores.sort()
    bad_row_idx = scores[0][1]
    bad_row = rows[bad_row_idx]

    recommendation = fix(header, bad_row, levers)
    
    print("Retention Strategy Recommendation (No Budget for Pay Increases):")
    print(f"For employee at index {bad_row_idx}, we recommend the following non-monetary change:")
    for col, val in recommendation.items():
        print(f"  - Adjust '{col}' to: {val}")
    print("This change is designed to improve the employee's tenure score using available organizational levers.")
