import csv


def remember(header, rows):
    goal_cols = [c for c in header if c.endswith('+')]

    best_positions = []
    for i, row in enumerate(rows):
        score = 0
        valid = True
        for col in goal_cols:
            val = row[header.index(col)]
            if val == '?':
                valid = False
                break
            try:
                score += float(val)
            except ValueError:
                valid = False
                break
        if not valid:
            continue
        best_positions.append((score, i))

    best_positions.sort(reverse=True)

    max_count = max(1, int(len(rows) * 0.05))
    return [int(pos) for _, pos in best_positions[:max_count]]

        mean_x = [sum(row[j] for row in X) / n for j in range(p)]

        lam = 0.01
        XtX = [[0.0]*p for _ in range(p)]
        for j in range(p):
            for k in range(j, p):
                s = sum(X[i][j] * X[i][k] for i in range(n))
                if j == k:
                    XtX[j][k] += lam
                else:
                    XtX[j][k] *= 2
        Xty = [sum(X[i][j] * Y[i] for i in range(n)) for j in range(p)]
        aug = [list(row) + [Xty[j]] for j, row in enumerate(XtX)]
        for col in range(p):
            if abs(aug[col][col]) < 1e-10:
                continue
            for row in range(col+1, p):
                factor = aug[row][col] / aug[col][col]

        weights = [aug[i][p] / aug[i][i] if abs(aug[i][i]) > 1e-10 else 0.0 for i in range(p)]

            return sum(weights[j] * (x[j] - mean_x[j]) for j in range(p)) + mean_y
        return weights, mean_x, mean_y

def certify(xs, ys):
    """Returns True if xs and ys are the same apart from noise."""
    n_x, n_y = len(xs), len(ys)
    if n_x < 2 or n_y < 2:
        return False

    mean_x = sum(xs) / n_x
    mean_y = sum(ys) / n_y
    var_x = sum((v - mean_x) ** 2 for v in xs) / (n_x - 1) if n_x > 1 else 0.0
    var_y = sum((v - mean_y) ** 2 for v in ys) / (n_y - 1) if n_y > 1 else 0.0

    se = ((var_x / n_x) + (var_y / n_y)) ** 0.5
    t_stat = abs(mean_x - mean_y) / se if se > 0 else 0.0

    return t_stat < 2.0

import csv


def flow(header, row, seen):
    goal_cols = [c for c in header if c.endswith('+')]

    def score(r):
        s = 0.0
        valid = True
        for col in goal_cols:
            val = r[header.index(col)]
            try:
                s += float(val)
            except ValueError:
                return -1e9
        return s

    current_score = score(row)

    if not seen:
        # First row always qualifies.
        return True

    best_seen = max(score(r) for r in seen)
    threshold = best_seen * 0.95
    return current_score >= threshold


if __name__ == "__main__":
    with open("WA_Fn-UseC_-HR-Employee-Attrition.train.csv", "r") as f:
        reader = csv.reader(f)
        header = next(reader)
        rows = list(reader)

    chosen = None
    for row in rows:
        if flow(header, row, []):
            chosen = row
            break

    print(f"Chosen employee #{chosen[header.index('EmployeeNumber')]}, "
          f"At company {rows[chosen[header.index('YearsAtCompany+')]} yrs")
