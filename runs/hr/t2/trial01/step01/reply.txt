import csv

def remember(header, rows):
    """
    Identifies staff who stay 'unusually long'.
    We define 'staying long' by the goal columns ending in '+':
    'YearsAtCompany+', 'YearsInCurrentRole+', 'YearsWithCurrManager+'
    We calculate a score based on the sum of these values and return 
    the indices of the top 5% (max 40 rows for 800 records).
    """
    # Identify indices for goal columns ending in '+'
    goal_indices = [i for i, name in enumerate(header) if name.endswith('+')]
    
    scored_rows = []
    for idx, row in enumerate(rows):
        # Calculate score: sum of values in goal columns
        score = sum(row[i] for i in goal_indices)
        scored_rows.append((score, idx))
    
    # Sort by score descending
    scored_rows.sort(key=lambda x: x[0], reverse=True)
    
    # Return top 5% (0.05 * 800 = 40)
    limit = max(1, int(len(rows) * 0.05))
    return [item[1] for item in scored_rows[:limit]]

if __name__ == "__main__":
    filename = 'WA_Fn-UseC_-HR-Employee-Attrition.n1000.train.csv'
    rows = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = []
            for i, val in enumerate(row):
                # Convert numeric columns (starting with uppercase) to float
                if header[i][0].isupper():
                    processed_row.append(float(val))
                else:
                    processed_row.append(val)
            rows.append(processed_row)
    
    best_indices = remember(header, rows)
    
    print("Staff with unusually long tenure (Top 5%):")
    print(f"{'Index':<10} | {'YearsAtCompany':<15} | {'YearsInCurrentRole':<20} | {'YearsWithManager':<15}")
    print("-" * 70)
    
    # Indices for display
    yac = header.index('YearsAtCompany+')
    yicr = header.index('YearsInCurrentRole+')
    ywcm = header.index('YearsWithCurrManager+')
    
    for idx in best_indices:
        r = rows[idx]
        print(f"{idx:<10} | {r[yac]:<15.0f} | {r[yicr]:<20.0f} | {r[ywcm]:<15.0f}")
