import csv

def remember(header, rows):
    """
    Identifies staff who stay 'unusually long'.
    We define 'staying long' by the goal columns ending in '+':
    'YearsAtCompany+', 'YearsInCurrentRole+', 'YearsWithCurrManager+'
    We calculate a score based on the sum of these values and return 
    the indices of the top 5% (max 40 rows for 800 records).
    """
    # Identify indices for goal columns ending in '+'
    goal_indices = [i for i, name in enumerate(header) if name.endswith('+')]
    
    scored_rows = []
    for idx, row in enumerate(rows):
        # Calculate score: sum of values in goal columns
        score = sum(row[i] for i in goal_indices)
        scored_rows.append((score, idx))
    
    # Sort by score descending
    scored_rows.sort(key=lambda x: x[0], reverse=True)
    
    # Return top 5% (0.05 * 800 = 40)
    limit = max(1, int(len(rows) * 0.05))
    return [item[1] for item in scored_rows[:limit]]

import csv
import math

def certify(xs, ys):
    """
    Returns True if the two lists of numbers are statistically similar 
    (i.e., they do not differ significantly), False otherwise.
    Uses a simple t-test approach for independent samples.
    """
    n1, n2 = len(xs), len(ys)
    if n1 < 2 or n2 < 2:
        return True
    
    m1, m2 = sum(xs) / n1, sum(ys) / n2
    v1 = sum((x - m1) ** 2 for x in xs) / (n1 - 1)
    v2 = sum((y - m2) ** 2 for y in ys) / (n2 - 1)
    
    # Pooled standard error
    se = math.sqrt((v1 / n1) + (v2 / n2))
    if se == 0:
        return m1 == m2
    
    # t-statistic
    t_stat = abs(m1 - m2) / se
    
    # For a 95% confidence level, t-critical is approx 1.96
    return t_stat < 1.96

def blame(header, rows):

    Identifies input columns that correlate with the 'worst' rows.

    # Calculate scores for all rows
    scored = []

        # Normalize: for '+', higher is better; for '-', lower is better.

def get_score(header, row):
    """Calculates a score where higher is better."""
    score = 0
    for i, name in enumerate(header):
        if name.endswith('+'):
            score += row[i]
        elif name.endswith('-'):
            score -= row[i]
    return score

def justify(header, rows, row):
    """
    Estimates the score of the target row by averaging the scores of rows
    with matching symbolic input values, then compares it to all rows.
    """
    # Identify input columns (non-goal, non-numeric)
    symbolic_indices = [i for i, name in enumerate(header) 
                        if not name.endswith('+') and not name.endswith('-') 
                        and not header[i][0].isupper()]
    
    # Find rows that match the symbolic inputs of the target row
    matches = []
    for r in rows:
        if all(r[i] == row[i] for i in symbolic_indices):
            matches.append(get_score(header, r))
    
    # Estimate score: use average of matches, or global average if no matches
    if matches:
        est_score = sum(matches) / len(matches)
    else:
        all_scores = [get_score(header, r) for r in rows]
        est_score = sum(all_scores) / len(all_scores)
    
    # Compare estimated score against all rows
    all_scores = [get_score(header, r) for r in rows]
    better_than = [s for s in all_scores if est_score > s]
    p = len(better_than) / len(all_scores)
    
    reason = f"This arrangement ranks in the top {int((1-p)*100)}% of all teams because its estimated tenure score of {est_score:.1f} outperforms {int(p*100)}% of the historical records."
    
    return p, reason

def fix(header, row, levers):
    """
    Recommends changes to a 'bad' row to improve its tenure score.
    Iterates through each lever, tests the impact of changing the value,
    and returns the change that results in the highest score improvement.
    """
    best_change = {}
    best_score_gain = -1.0
    
    current_score = step06.get_score(header, row)
    
    for col_name, allowed in levers.items():
        idx = header.index(col_name)
        original_val = row[idx]
        
        # Test variations
        variations = []
        if isinstance(allowed, tuple):  # Numeric range (low, high)
            # Test mid-point or boundaries
            variations = [allowed[0], allowed[1], (allowed[0] + allowed[1]) / 2]
        else:  # Symbolic list
            variations = allowed
            
        for val in variations:
            if val == original_val:
                continue
            
            # Create a copy and apply change
            new_row = list(row)
            new_row[idx] = val
            
            new_score = step06.get_score(header, new_row)
            gain = new_score - current_score
            
            if gain > best_score_gain:
                best_score_gain = gain
                best_change = {col_name: val}
                
    return best_change

import csv

def spot(header, rows):
    """
    Identifies suspicious rows by checking for outliers in numeric input columns.
    A row is considered suspicious if its values in numeric input columns are 
    statistically extreme (e.g., > 3 standard deviations from the mean).
    """
    numeric_input_indices = [
        i for i, name in enumerate(header) 
        if header[i][0].isupper() and not name.endswith('+') and not name.endswith('-')
    ]
    
    suspicion_scores = []
    
    for row_idx, row in enumerate(rows):
        score = 0
        for i in numeric_input_indices:
            col_data = [r[i] for r in rows]
            mean = sum(col_data) / len(col_data)
            std = (sum((x - mean) ** 2 for x in col_data) / len(col_data)) ** 0.5
            
            if std > 0:
                z_score = abs(row[i] - mean) / std
                # If a value is more than 3 standard deviations away, it's suspicious
                if z_score > 3:
                    score += z_score
        
        if score > 0:
            suspicion_scores.append((score, row_idx))
            
    # Sort by suspicion score descending
    suspicion_scores.sort(key=lambda x: x[0], reverse=True)
    return [item[1] for item in suspicion_scores]

import csv

import step07

def race(header, rows, label):
    """
    Compares three strategies for finding the best employee row.
    Budget: 50 labels total.
    """
    budget = 50
    used_indices = {}

    def get_true_goals(idx):
        if idx not in used_indices:
            used_indices[idx] = label(idx)
        return used_indices[idx]

    def get_score(idx):
        goals = get_true_goals(idx)
        score = 0
        goal_ptr = 0
        for name in header:
            if name.endswith('+'):
                score += goals[goal_ptr]
                goal_ptr += 1
            elif name.endswith('-'):
                score -= goals[goal_ptr]
                goal_ptr += 1
        return score

    # Strategy 1: Random
    def strategy_random():
        indices = random.sample(range(len(rows)), budget // 3)
        best_idx = -1
        best_score = float('-inf')
        for idx in indices:
            s = get_score(idx)
            if s > best_score:
                best_score, best_idx = s, idx
        return best_idx, best_score

    # Strategy 2: High Salary (Greedy on MonthlyIncome)
    def strategy_salary():
        income_idx = header.index('MonthlyIncome')
        # Sort by income descending
        sorted_rows = sorted(range(len(rows)), key=lambda i: rows[i][income_idx], reverse=True)
        best_idx = -1
        best_score = float('-inf')
        for idx in sorted_rows[:budget // 3]:
            s = get_score(idx)
            if s > best_score:
                best_score, best_idx = s, idx
        return best_idx, best_score

    # Strategy 3: High Tenure (Greedy on TotalWorkingYears)

def find_similar(header, rows, target_row, n=5):
    """
    Finds the top n rows most similar to target_row based on symbolic inputs.
    Similarity is defined by the number of matching symbolic column values.
    """
    symbolic_indices = [
        i for i, name in enumerate(header)
        if not name.endswith('+') and not name.endswith('-') and not header[i][0].isupper()
    ]
    
    scored_rows = []
    for idx, row in enumerate(rows):
        # Skip if it's the target row itself
        if row == target_row:
            continue
            
        # Calculate similarity score (number of matching symbolic fields)
        match_count = sum(1 for i in symbolic_indices if row[i] == target_row[i])
        scored_rows.append((match_count, idx))
    
    # Sort by match count descending
    scored_rows.sort(key=lambda x: x[0], reverse=True)
    
    # Return indices of top n matches
    return [item[1] for item in scored_rows[:n]]

import csv
import step06

def flow(header, row, seen):
    """
    Detects if a team's composition has shifted significantly.
    We compare the current row's tenure score to the distribution of 
    tenure scores in the 'seen' history. If the current row is an 
    outlier (e.g., > 2 standard deviations from the mean of seen rows),
    we flag it as a shift in team composition.
    """
    if len(seen) < 30:
        return False
    
    # Calculate scores for all seen rows
    seen_scores = [step06.get_score(header, r) for r in seen]
    
    # Calculate mean and standard deviation of seen scores
    mean = sum(seen_scores) / len(seen_scores)
    variance = sum((s - mean) ** 2 for s in seen_scores) / len(seen_scores)
    std = variance ** 0.5
    
    # Current row score
    current_score = step06.get_score(header, row)
    
    # If std is 0, we can't detect a shift
    if std == 0:
        return False
    
    # Flag if the current row is more than 2 standard deviations away
    z_score = abs(current_score - mean) / std
    return z_score > 2.0

def justify(header, rows, row):
    """
    Estimates the score of the target row by checking if increasing 
    MonthlyIncome improves the score, then compares it to all rows.
    """
    # Identify goal indices
    goal_indices = [i for i, name in enumerate(header) if name.endswith('+') or name.endswith('-')]
    income_idx = header.index('MonthlyIncome')
    
    # Estimate the score of the provided row (with "?" goals)
    # We use the average score of rows with the same department/role as a baseline
    dept_idx = header.index('department')
    role_idx = header.index('jobRole')
    
    matches = [step06.get_score(header, r) for r in rows 
               if r[dept_idx] == row[dept_idx] and r[role_idx] == row[role_idx]]
    
    if not matches:
        matches = [step06.get_score(header, r) for r in rows]
    
    base_score = sum(matches) / len(matches)
    
    # Test if increasing income significantly changes the score
    # We look at the correlation between MonthlyIncome and score in the dataset
    incomes = [r[income_idx] for r in rows]
    scores = [step06.get_score(header, r) for r in rows]
    
    # Simple correlation check: if income is not a strong driver, paying more won't help
    # We calculate the average score gain per unit of income
    avg_income = sum(incomes) / len(incomes)
    high_income_rows = [s for i, s in zip(incomes, scores) if i > avg_income]
    low_income_rows = [s for i, s in zip(incomes, scores) if i <= avg_income]
    
    gain = (sum(high_income_rows)/len(high_income_rows)) - (sum(low_income_rows)/len(low_income_rows))
    
    # If the gain is negligible (e.g., < 1% of total score range), money isn't the primary lever
    all_scores = [step06.get_score(header, r) for r in rows]
    score_range = max(all_scores) - min(all_scores)
    
    if gain < (0.01 * score_range):
        reason = "Increasing pay has a negligible impact on tenure scores compared to other organizational factors."
    else:
        reason = "Increasing pay shows a positive correlation with tenure, but is not the sole driver of retention."
        
    # Calculate p: fraction of rows this row is better than
    better_than = [s for s in all_scores if base_score > s]
    p = len(better_than) / len(all_scores)
    
    return p, reason

    search_pool = lab_indices if lab_indices else range(len(rows))
    
    # Sample up to 50 indices
    sample_size = min(50, len(search_pool))
    indices = random.sample(search_pool, sample_size)
    
    best_idx = -1
    best_score = float('-inf')
    
    for idx in indices:
        # Get true values via label()
        true_goals = label(idx)
        
        # Calculate score: sum of '+' goals minus sum of '-' goals
        score = 0
        goal_ptr = 0
        for i, name in enumerate(header):

            elif name.endswith('-'):
                score -= true_goals[goal_ptr]
                goal_ptr += 1
        
        if score > best_score:
            best_score = score
            best_idx = idx
            
    return best_idx
