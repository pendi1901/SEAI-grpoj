def remember(header, rows):
    """
    Identifies staff who stay 'unusually long'.
    'YearsAtCompany+' is a goal to maximize.
    We return the indices of the top 5% of rows based on this metric.
    """
    # Identify index for 'YearsAtCompany+'
    try:
        idx = header.index('YearsAtCompany+')
    except ValueError:
        return []

    # Create list of (index, value)
    indexed_rows = []
    for i, row in enumerate(rows):
        val = row[idx]
        if isinstance(val, (int, float)):
            indexed_rows.append((i, val))

    # Sort by years descending
    indexed_rows.sort(key=lambda x: x[1], reverse=True)

    # Calculate 5% limit (at least 1, max 5% of total rows)
    limit = max(1, len(rows) // 20)
    
    # Return indices of the top performers
    return [item[0] for item in indexed_rows[:limit]]

import csv
import step01

def certify(xs, ys):
    """
    Performs a simple two-sample t-test approximation.
    Returns True if the means are close (noise), False if they differ significantly.
    """
    n_x, n_y = len(xs), len(ys)
    if n_x < 2 or n_y < 2:
        return True
    
    mean_x, mean_y = sum(xs) / n_x, sum(ys) / n_y
    var_x = sum((x - mean_x) ** 2 for x in xs) / (n_x - 1)
    var_y = sum((y - mean_y) ** 2 for y in ys) / (n_y - 1)
    
    pooled_se = ((var_x / n_x) + (var_y / n_y)) ** 0.5
    if pooled_se == 0:
        return mean_x == mean_y
    
    t_stat = abs(mean_x - mean_y) / pooled_se
    # Threshold for 95% confidence (approx 2.0 for large samples)
    return t_stat < 2.0

def flow(header, row, seen):
    """
    Decides if the current row is a 'top performer' based on 'YearsAtCompany+'
    compared to the history of seen rows. We use a simple threshold:
    if the current row's tenure is significantly better than the mean of 
    the seen rows (using the logic from step03), we flag it.
    """
    tenure_idx = header.index('YearsAtCompany+')
    current_val = row[tenure_idx]
    
    if not seen:
        return True
    
    seen_vals = [r[tenure_idx] for r in seen if isinstance(r[tenure_idx], (int, float))]
