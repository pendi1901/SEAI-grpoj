import csv
import step01

def certify(xs, ys):
    """
    Performs a simple two-sample t-test approximation.
    Returns True if the means are close (noise), False if they differ significantly.
    """
    n_x, n_y = len(xs), len(ys)
    if n_x < 2 or n_y < 2:
        return True
    
    mean_x, mean_y = sum(xs) / n_x, sum(ys) / n_y
    var_x = sum((x - mean_x) ** 2 for x in xs) / (n_x - 1)
    var_y = sum((y - mean_y) ** 2 for y in ys) / (n_y - 1)
    
    pooled_se = ((var_x / n_x) + (var_y / n_y)) ** 0.5
    if pooled_se == 0:
        return mean_x == mean_y
    
    t_stat = abs(mean_x - mean_y) / pooled_se
    # Threshold for 95% confidence (approx 2.0 for large samples)
    return t_stat < 2.0

    """
    Finds the best working arrangement by sampling 50 rows.
    'YearsAtCompany+' is a goal to maximize.
    label(i) returns the full row, so we extract the goal value.
    """
    # Identify the goal column index
    goal_idx = -1
    for i, name in enumerate(header):
        if name.endswith('+') or name.endswith('-'):
            goal_idx = i
            break
    
    # We can afford 50 interviews. Sample 50 indices.
    # Using a step to sample the 800 rows.
    step = len(rows) // 50
    indices_to_sample = [i * step for i in range(50)]
    
    best_idx = -1

        row_data = label(i)
