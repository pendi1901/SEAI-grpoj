            goal_idx = i

import csv
import math

def certify(xs, ys):
    """
    Returns True if the two distributions are statistically similar (noise),
    False if they differ significantly (real effect).
    Uses Cohen's d effect size: if d < 0.5, consider it noise.
    """
    n1, n2 = len(xs), len(ys)
    if n1 < 2 or n2 < 2:
        return True
    
    m1, m2 = sum(xs) / n1, sum(ys) / n2
    v1 = sum((x - m1) ** 2 for x in xs) / (n1 - 1)
    v2 = sum((y - m2) ** 2 for y in ys) / (n2 - 1)
    
    pooled_std = math.sqrt(((n1 - 1) * v1 + (n2 - 1) * v2) / (n1 + n2 - 2))
    if pooled_std == 0:
        return m1 == m2
    
    d = abs(m1 - m2) / pooled_std
    return d < 0.5

import csv

def flow(header, row, seen):
    """
    Decides if the current row is a 'top performer' based on tenure metrics.
    We use the logic from step01 to determine if the current row is in the 
    top 5% of the cumulative data seen so far.
    """
    # Identify goal indices
    goals = [i for i, name in enumerate(header) if name.endswith('+') or name.endswith('-')]
    
    # Calculate score for current row
    # For +, higher is better. For -, lower is better.
    # Since step01 focused on +, we adapt:
    def get_score(r):
        score = 0
        for i in goals:
            val = r[i]
            if header[i].endswith('+'):
                score += val
            else:
                score -= val
        return score

    current_score = get_score(row)
    
    # Compare against all previously seen rows
    # If this row is better than at least 95% of seen rows, we flag it.
    if not seen:
        return True
    
    seen_scores = [get_score(r) for r in seen]
    better_than = sum(1 for s in seen_scores if current_score > s)
    
    # If it's in the top 5% of what we've seen so far
    return better_than >= (len(seen_scores) * 0.95)

def blame(header, rows):
    """
    Identifies input columns that correlate with 'bad' outcomes (low tenure).
    'Bad' is defined as the bottom 20% of rows based on the sum of goal columns.
    We return the names of the input columns that have the highest correlation
    with being in the 'bad' group.
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith('+') or name.endswith('-')]
    input_indices = [i for i, name in enumerate(header) if not (name.endswith('+') or name.endswith('-'))]

    # Calculate score for each row (higher is better)
    def get_score(r):
        score = 0
        for i in goal_indices:
            val = r[i]
            if header[i].endswith('+'):
                score += val
            else:
                score -= val
        return score

    # Sort rows by score and pick bottom 20%
    scored_rows = sorted([(get_score(r), r) for r in rows], key=lambda x: x[0])
    limit = max(1, int(len(rows) * 0.2))
    bad_rows = [x[1] for x in scored_rows[:limit]]
    
    # For each input column, calculate the frequency of values in bad_rows
    # vs the whole dataset to find 'blame' (over-representation in bad group)
    blame_scores = []
    for i in input_indices:
        col_name = header[i]
        # Count values in bad rows
        bad_counts = {}
        for r in bad_rows:
            val = r[i]
            bad_counts[val] = bad_counts.get(val, 0) + 1
        
        # Find the value most associated with bad rows
        max_freq = 0
        for val, count in bad_counts.items():
            # Simple heuristic: frequency in bad group
            max_freq = max(max_freq, count)
        
        blame_scores.append((max_freq, col_name))
    
    # Return top 3 columns that correlate with short stays
    blame_scores.sort(key=lambda x: x[0], reverse=True)
    return [item[1] for item in blame_scores[:3]]

def justify(header, rows, row):
    """
    Estimates goal values for the row, then calculates the percentile
    rank against the provided rows.
    """
    # 1. Estimate missing goals using step02 logic (average by jobRole)
    goal_indices = [i for i, name in enumerate(header) if name.endswith('+') or name.endswith('-')]

    p = better_than / len(rows)

def choose(header, rows, label):
    """
    Finds the best working arrangement by sampling 50 rows.
    We use a simple strategy: sample 50 rows, label them to get true values,
    and return the index of the one with the highest score.
    """
    # We can label at most 50 rows.
    # We pick 50 random indices to evaluate.
    indices = list(range(len(rows)))
    sample_size = min(50, len(rows))
    sampled_indices = random.sample(indices, sample_size)
    
    best_idx = -1
    best_score = -float('inf')
    
    for idx in sampled_indices:
        # Get true values for the goals
        true_goals = label(idx)
        
        # Create a temporary row with true values to calculate score
        temp_row = list(rows[idx])
        goal_indices = [i for i, name in enumerate(header) if name.endswith('+') or name.endswith('-')]
        for i, g_idx in enumerate(goal_indices):
            temp_row[g_idx] = true_goals[i]
            
        score = get_score(header, temp_row)
        
        if score > best_score:
            best_score = score
            best_idx = idx
            
    return best_idx

import csv

def spot(header, rows):
    """
    Identifies rows that look 'wrong' by detecting outliers in numeric columns.
    We calculate the mean and standard deviation for each numeric column and
    flag rows that have values significantly outside the expected range (e.g., 
    more than 3 standard deviations from the mean).
    """
    numeric_indices = [i for i, name in enumerate(header) if name[0].isupper()]
    
    # Calculate stats for each numeric column
    stats = {}
    for i in numeric_indices:
        vals = [r[i] for r in rows if isinstance(r[i], (int, float))]
        if not vals:
            continue
        mean = sum(vals) / len(vals)
        std = (sum((x - mean) ** 2 for x in vals) / len(vals)) ** 0.5
        stats[i] = (mean, std)
    
    # Score rows by how many outliers they contain
    # A higher count of extreme outliers makes a row more suspicious
    suspicion_scores = []
    for idx, row in enumerate(rows):
        score = 0
        for i in numeric_indices:
            if i in stats:
                mean, std = stats[i]
                val = row[i]
                if std > 0:
                    z_score = abs(val - mean) / std
                    if z_score > 3:  # Threshold for "wrong" entry
                        score += z_score
        suspicion_scores.append((score, idx))
    
    # Sort by suspicion score descending
    suspicion_scores.sort(key=lambda x: x[0], reverse=True)
    
    # Return indices of rows with non-zero suspicion scores
    return [item[1] for item in suspicion_scores if item[0] > 0]

def race(header, rows, label):

    Compares three strategies: random, tenure-based, and role-based.
    Budget: 50 labels total.
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith('+') or name.endswith('-')]
    
    # Strategy 1: Random

        # Sort by sum of tenure columns (proxy for 'best')
        # We only know tenure for labeled rows, so we use a heuristic on available data
        return sorted(indices, key=lambda i: rows[i][header.index('TotalWorkingYears')], reverse=True)[:17]

    # Strategy 3: Role-based (using step02 logic)

        # Prioritize roles with high average tenure
        return sorted(indices, key=lambda i: rows[i][header.index('JobLevel')], reverse=True)[:17]

    strategies = {
        "random": strategy_random,
        "tenure": strategy_tenure,
        "role": strategy_role
    }
    
    all_indices = list(range(len(rows)))
    best_overall_score = -float('inf')
    best_row_idx = -1
    winner = ""
    
    budget = 50

        sampled = func(all_indices)

    return {"winner": winner, "row": best_row_idx, "tried": list(strategies.keys())}

def get_similarity(row1, row2, header):
    """
    Calculates a simple similarity score between two rows.
    Higher score means more similar.
    """
    score = 0
    for i, name in enumerate(header):
        # Skip goal columns as they are the target of the query
        if name.endswith('+') or name.endswith('-'):
            continue
        
        # Compare values
        if row1[i] == row2[i]:
            score += 1
        elif isinstance(row1[i], float) and isinstance(row2[i], float):
            # For numeric inputs, inverse of normalized difference
            diff = abs(row1[i] - row2[i])
            score += 1 / (1 + diff)
    return score

def remember(header, rows):
    """
    Finds the 5% of rows most similar to the first row (the 'target' employee).

    similarities = []

    similarities.sort(key=lambda x: x[0], reverse=True)
    
    # Return top 5%
    limit = max(1, int(len(rows) * 0.05))
    return [item[1] for item in similarities[:limit]]

import csv
import step04

def flow(header, row, seen):
    """
    Detects if a team's composition has shifted significantly.
    We compare the current row's tenure metrics against the distribution
    of the 'seen' rows. If the current row is an outlier (e.g., significantly
    higher or lower than the established norm), we flag it as a change.
    """
    if not seen:
        return False

    goal_indices = [i for i, name in enumerate(header) if name.endswith('+') or name.endswith('-')]
    
    def get_score(r):
        score = 0
        for i in goal_indices:
            val = r[i]
            if isinstance(val, (int, float)):
                if header[i].endswith('+'):
                    score += val
                else:
                    score -= val
        return score

    current_score = get_score(row)
    seen_scores = [get_score(r) for r in seen]
    
    # Calculate mean and standard deviation of seen scores
    n = len(seen_scores)
    mean = sum(seen_scores) / n
    std = (sum((s - mean) ** 2 for s in seen_scores) / n) ** 0.5
    
    # If the current row is more than 2 standard deviations away from the mean,
    # the team composition has shifted significantly.
    if std > 0:
        z_score = abs(current_score - mean) / std
        return z_score > 2.0
    
    return False

import csv

def blame(header, rows):
    """
    Identifies input columns that have the least correlation with tenure.
    We reuse the logic from step05 to identify 'bad' rows (bottom 20% tenure).
    Then, we identify columns that show no significant difference in distribution
    between the 'bad' rows and the 'good' rows (top 20% tenure).
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith('+') or name.endswith('-')]
    input_indices = [i for i, name in enumerate(header) if not (name.endswith('+') or name.endswith('-'))]

    def get_score(r):
        score = 0
        for i in goal_indices:
            val = r[i]
            if header[i].endswith('+'):
                score += val
            else:
                score -= val
        return score

    # Sort rows by score
    scored_rows = sorted([(get_score(r), r) for r in rows], key=lambda x: x[0])
    limit = max(1, int(len(rows) * 0.2))
    bad_rows = scored_rows[:limit]
    good_rows = scored_rows[-limit:]
    
    # Identify columns where the distribution is similar between bad and good rows
    # (i.e., the column has nothing to do with tenure)
    useless_columns = []
    for i in input_indices:
        col_name = header[i]
        
        # Collect values for this column in both groups
        bad_vals = [r[1][i] for r in bad_rows]
        good_vals = [r[1][i] for r in good_rows]
        
        # If numeric, check if means are very close (low effect size)
        if isinstance(bad_vals[0], (int, float)):
            m_bad = sum(bad_vals) / len(bad_vals)
            m_good = sum(good_vals) / len(good_vals)
            # If the difference is negligible, it's a candidate
            if abs(m_bad - m_good) < 0.1:
                useless_columns.append(col_name)
        else:
            # If symbolic, check if frequency distribution is similar
            # For simplicity, we check if the most common value is the same
            bad_mode = max(set(bad_vals), key=bad_vals.count)
            good_mode = max(set(good_vals), key=good_vals.count)
            if bad_mode == good_mode:
                useless_columns.append(col_name)
                
    return useless_columns
