import csv

def remember(header, rows):
    """
    Identifies staff who stay 'unusually long'.
    We define 'staying long' by the goal columns:
    YearsAtCompany+, YearsInCurrentRole+, YearsWithCurrManager+
    We calculate a score for each row based on these maximization goals.
    """
    # Identify indices for goal columns
    goals = []
    for i, name in enumerate(header):
        if name.endswith('+'):
            goals.append(i)
    
    # Calculate scores: sum of normalized values or raw sums
    # Since we want the best (highest) values, we rank by the sum of these columns
    scores = []
    for idx, row in enumerate(rows):
        score = sum(row[i] for i in goals)
        scores.append((score, idx))
    
    # Sort by score descending
    scores.sort(key=lambda x: x[0], reverse=True)
    
    # Return top 5% (800 rows * 0.05 = 40 rows)
    limit = max(1, int(len(rows) * 0.05))
    return [item[1] for item in scores[:limit]]

import csv
import step01

def guess(header, train, test):
    """
    Estimates the first goal column (YearsAtCompany+) for test rows
    by finding the average value of that column for rows in the training
    set that share the same 'jobRole'.
    """
    goal_idx = -1
    for i, name in enumerate(header):
        if name.endswith('+'):
            goal_idx = i
            break
    
    role_idx = header.index('jobRole')
    
    # Build a map of jobRole -> list of goal values
    role_map = {}
    for row in train:
        role = row[role_idx]
        val = row[goal_idx]
        if role not in role_map:
            role_map[role] = []
        role_map[role].append(val)
    
    # Calculate averages per role
    averages = {role: sum(vals) / len(vals) for role, vals in role_map.items()}
    
    # Default to global average if role is unknown
    global_avg = sum(averages.values()) / len(averages)
    
    predictions = []
    for row in test:
        role = row[role_idx]
        predictions.append(averages.get(role, global_avg))
        
    return predictions
