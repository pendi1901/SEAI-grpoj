def remember(header, rows):
    """
    Identifies staff who stay 'unusually long'.
    We define 'staying long' by the goal columns:
    YearsAtCompany+, YearsInCurrentRole+, YearsWithCurrManager+
    We calculate a score for each row based on these maximization goals.
    """
    # Identify indices for goal columns
    goals = []
    for i, name in enumerate(header):
        if name.endswith('+'):
            goals.append(i)
    
    # Calculate scores: sum of normalized values or raw sums
    # Since we want the best (highest) values, we rank by the sum of these columns
    scores = []
    for idx, row in enumerate(rows):
        score = sum(row[i] for i in goals)
        scores.append((score, idx))
    
    # Sort by score descending
    scores.sort(key=lambda x: x[0], reverse=True)
    
    # Return top 5% (800 rows * 0.05 = 40 rows)
    limit = max(1, int(len(rows) * 0.05))
    return [item[1] for item in scores[:limit]]

def guess(header, train, test):

    Estimates the first goal column (YearsAtCompany+) for test rows

    """
    goal_idx = -1
    for i, name in enumerate(header):
        if name.endswith('+'):
            goal_idx = i
            break
    
    role_idx = header.index('jobRole')
    
    # Build a map of jobRole -> list of goal values
    role_map = {}
    for row in train:
        role = row[role_idx]
        val = row[goal_idx]
        if role not in role_map:
            role_map[role] = []
        role_map[role].append(val)
    
    # Calculate averages per role
    averages = {role: sum(vals) / len(vals) for role, vals in role_map.items()}

    predictions = []
    for row in test:
        role = row[role_idx]
        predictions.append(averages.get(role, global_avg))
        
    return predictions

import csv
import math

def certify(xs, ys):
    """
    Returns True if the two distributions are statistically similar (noise),
    False if they differ significantly (real effect).
    Uses Cohen's d effect size: if d < 0.5, consider it noise.
    """
    n1, n2 = len(xs), len(ys)
    if n1 < 2 or n2 < 2:
        return True
    
    m1, m2 = sum(xs) / n1, sum(ys) / n2
    v1 = sum((x - m1) ** 2 for x in xs) / (n1 - 1)
    v2 = sum((y - m2) ** 2 for y in ys) / (n2 - 1)
    
    pooled_std = math.sqrt(((n1 - 1) * v1 + (n2 - 1) * v2) / (n1 + n2 - 2))
    if pooled_std == 0:
        return m1 == m2
    
    d = abs(m1 - m2) / pooled_std
    return d < 0.5

def flow(header, row, seen):
    """
    Decides if the current row is a 'top performer' based on tenure metrics.
    We use the logic from step01 to determine if the current row is in the 
    top 5% of the cumulative data seen so far.
    """
    # Identify goal indices
    goals = [i for i, name in enumerate(header) if name.endswith('+') or name.endswith('-')]
    
    # Calculate score for current row
    # For +, higher is better. For -, lower is better.
    # Since step01 focused on +, we adapt:
    def get_score(r):
        score = 0
        for i in goals:
            val = r[i]
            if header[i].endswith('+'):
                score += val
            else:
                score -= val
        return score

    current_score = get_score(row)
    
    # Compare against all previously seen rows
    # If this row is better than at least 95% of seen rows, we flag it.
    if not seen:
        return True
    
    seen_scores = [get_score(r) for r in seen]
    better_than = sum(1 for s in seen_scores if current_score > s)
    
    # If it's in the top 5% of what we've seen so far
    return better_than >= (len(seen_scores) * 0.95)

import csv

def blame(header, rows):
    """
    Identifies input columns that correlate with 'bad' outcomes (low tenure).
    'Bad' is defined as the bottom 20% of rows based on the sum of goal columns.
    We return the names of the input columns that have the highest correlation
    with being in the 'bad' group.
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith('+') or name.endswith('-')]
    input_indices = [i for i, name in enumerate(header) if not (name.endswith('+') or name.endswith('-'))]

    # Calculate score for each row (higher is better)
    def get_score(r):
        score = 0
        for i in goal_indices:
            val = r[i]
            if header[i].endswith('+'):
                score += val
            else:
                score -= val
        return score

    # Sort rows by score and pick bottom 20%
    scored_rows = sorted([(get_score(r), r) for r in rows], key=lambda x: x[0])
    limit = max(1, int(len(rows) * 0.2))
    bad_rows = [x[1] for x in scored_rows[:limit]]
    
    # For each input column, calculate the frequency of values in bad_rows
    # vs the whole dataset to find 'blame' (over-representation in bad group)
    blame_scores = []
    for i in input_indices:
        col_name = header[i]
        # Count values in bad rows
        bad_counts = {}
        for r in bad_rows:
            val = r[i]
            bad_counts[val] = bad_counts.get(val, 0) + 1
        
        # Find the value most associated with bad rows
        max_freq = 0
        for val, count in bad_counts.items():
            # Simple heuristic: frequency in bad group
            max_freq = max(max_freq, count)
        
        blame_scores.append((max_freq, col_name))
    
    # Return top 3 columns that correlate with short stays
    blame_scores.sort(key=lambda x: x[0], reverse=True)
    return [item[1] for item in blame_scores[:3]]

def justify(header, rows, row):
    """
    Estimates goal values for the row, then calculates the percentile
    rank against the provided rows.
    """
    # 1. Estimate missing goals using step02 logic (average by jobRole)
    goal_indices = [i for i, name in enumerate(header) if name.endswith('+') or name.endswith('-')]
    role_idx = header.index('jobRole')
    
    # Build averages from full rows
    role_map = {}
    for r in rows:
        role = r[role_idx]
        if role not in role_map:
            role_map[role] = []
        role_map[role].append([r[i] for i in goal_indices])
    
    averages = {role: [sum(vals)/len(vals) for vals in zip(*data)] for role, data in role_map.items()}
    
    # Fill in the "?" values
    target_role = row[role_idx]
    estimates = averages.get(target_role, [0]*len(goal_indices))

    target_score = get_score(header, filled_row)
    all_scores = [get_score(header, r) for r in rows]
    better_than = sum(1 for s in all_scores if target_score > s)
    p = better_than / len(rows)
    
    reason = f"This employee's estimated tenure metrics rank them in the top {int((1-p)*100)}% of the workforce based on their role as a {target_role}."
    
    return (p, reason)

def get_score(header, row):
    """Calculates a score where higher is better."""
    score = 0
    for i, name in enumerate(header):
        if name.endswith('+'):
            score += row[i]
        elif name.endswith('-'):
            score -= row[i]
    return score

def choose(header, rows, label):
    """
    Finds the best working arrangement by sampling 50 rows.
    We use a simple strategy: sample 50 rows, label them to get true values,
    and return the index of the one with the highest score.
    """
    # We can label at most 50 rows.
    # We pick 50 random indices to evaluate.
    indices = list(range(len(rows)))
    sample_size = min(50, len(rows))
    sampled_indices = random.sample(indices, sample_size)
    
    best_idx = -1
    best_score = -float('inf')
    
    for idx in sampled_indices:
        # Get true values for the goals
        true_goals = label(idx)
        
        # Create a temporary row with true values to calculate score
        temp_row = list(rows[idx])
        goal_indices = [i for i, name in enumerate(header) if name.endswith('+') or name.endswith('-')]
        for i, g_idx in enumerate(goal_indices):
            temp_row[g_idx] = true_goals[i]
            
        score = get_score(header, temp_row)
        
        if score > best_score:
            best_score = score
            best_idx = idx
            
    return best_idx
