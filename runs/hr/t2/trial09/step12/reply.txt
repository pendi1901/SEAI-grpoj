import csv
import step02

def guess(header, train, test):
    """
    Estimates the first goal column (YearsAtCompany+) by considering 
    the effect of 'overTime' within the 'Sales' department.
    """
    goal_idx = -1
    for i, name in enumerate(header):
        if name.endswith('+'):
            goal_idx = i
            break
    
    ot_idx = header.index('overTime')
    dept_idx = header.index('department')
    
    # Calculate average tenure for Sales employees based on Overtime status
    # Filter: department == 'Sales'
    sales_rows = [r for r in train if r[dept_idx] == 'Sales']
    
    # Map: (overTime) -> list of goal values
    ot_map = {}
    for row in sales_rows:
        ot = row[ot_idx]
        val = row[goal_idx]
        if ot not in ot_map:
            ot_map[ot] = []
        ot_map[ot].append(val)
    
    # Calculate averages
    averages = {ot: sum(vals) / len(vals) for ot, vals in ot_map.items() if vals}
    
    # Fallback to global average if data is missing
    all_goals = [r[goal_idx] for r in train]
    global_avg = sum(all_goals) / len(all_goals)
    
    predictions = []
    for row in test:
        # If Sales, we assume the 'No' overtime condition (the cut)
        if row[dept_idx] == 'Sales':
            predictions.append(averages.get('No', global_avg))
        else:
            # For non-sales, use standard role-based estimation from step02
            predictions.append(step02.guess(header, train, [row])[0])
            
    return predictions
