import csv

def remember(header, rows):
    """
    Identifies staff who stay 'unusually long'.
    'YearsAtCompany+' is a goal to maximize.
    We return the indices of the top 5% of rows based on this metric.
    """
    # Identify index for 'YearsAtCompany+'
    try:
        idx = header.index('YearsAtCompany+')
    except ValueError:
        return []

    # Create list of (value, original_index)
    indexed_rows = []
    for i, row in enumerate(rows):
        val = row[idx]
        if isinstance(val, (int, float)):
            indexed_rows.append((val, i))

    # Sort by YearsAtCompany descending
    indexed_rows.sort(key=lambda x: x[0], reverse=True)

    # Calculate 5% limit (max 40 rows for 800 records)
    limit = max(1, len(rows) // 20)
    
    # Return indices of the top performers
    return [item[1] for item in indexed_rows[:limit]]

if __name__ == "__main__":
    filename = 'WA_Fn-UseC_-HR-Employee-Attrition.n1000.train.csv'
    rows = []
    header = []
    
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = []
            for i, val in enumerate(row):
                # Convert numeric columns (Uppercase start)
                if header[i][0].isupper():
                    try:
                        processed_row.append(float(val))
                    except ValueError:
                        processed_row.append(0.0)
                else:
                    processed_row.append(val)
            rows.append(processed_row)

    best_indices = remember(header, rows)
    
    print("Staff who stay unusually long (Top 5% by YearsAtCompany+):")
    print(f"{'Index':<10} | {'YearsAtCompany':<15} | {'JobRole'}")
    print("-" * 45)
    
    role_idx = header.index('jobRole')
    years_idx = header.index('YearsAtCompany+')
    
    for idx in best_indices:
        print(f"{idx:<10} | {rows[idx][years_idx]:<15} | {rows[idx][role_idx]}")
