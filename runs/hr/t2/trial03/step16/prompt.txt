import csv
import math

def certify(xs, ys):
    """
    Performs a simple two-sample t-test (Welch's) to determine if the means
    of two groups (Overtime vs No Overtime) are significantly different.
    Returns True if the difference is likely noise (p > 0.05), False otherwise.
    """
    n1, n2 = len(xs), len(ys)
    if n1 < 2 or n2 < 2:
        return True
    
    m1, m2 = sum(xs) / n1, sum(ys) / n2
    v1 = sum((x - m1) ** 2 for x in xs) / (n1 - 1)
    v2 = sum((y - m2) ** 2 for y in ys) / (n2 - 1)
    
    # Standard error of difference
    se = math.sqrt((v1 / n1) + (v2 / n2))
    if se == 0:
        return m1 == m2
    
    t_stat = abs(m1 - m2) / se
    # For large samples, t > 1.96 corresponds to p < 0.05
    return t_stat < 1.96

    seen_years = [r[years_idx] for r in seen]

    threshold = seen_years[threshold_idx]

    # they are a high-tenure asset worth checking in on.
    return current_years >= threshold

def blame(header, rows):
    """
    Identifies input columns that correlate with short tenure (low YearsAtCompany+).
    We define 'worst' as the bottom 25% of employees by tenure.

        i for i, h in enumerate(header)

    sorted_rows = sorted(rows, key=lambda r: r[years_idx])
    worst_rows = sorted_rows[:len(rows) // 4]

def justify(header, rows, row):
    """
    Estimates the performance of a row and returns its percentile rank 
    relative to the provided dataset.
    """
    # Identify goal columns
    goals = [i for i, h in enumerate(header) if h.endswith('+') or h.endswith('-')]
    
    # Estimate the goal values for the input row using KNN from step02
    # We use the first 700 rows as training data for the estimation
    train_data = rows[:700]
    
    # Simple estimation: average of the goal values of the 5 nearest neighbors
    input_indices = [i for i, h in enumerate(header) if h[0].isupper() and i not in goals]
    
    distances = []
    for r in train_data:
        dist = sum((row[i] - r[i])**2 for i in input_indices)**0.5
        distances.append((dist, [r[g] for g in goals]))
    
    distances.sort(key=lambda x: x[0])
    top_k = distances[:5]
    
    # Estimated goal values
    estimates = [sum(d[1][i] for d in top_k) / len(top_k) for i in range(len(goals))]
    
    # Calculate how many rows this row is "better than"
    better_count = 0
    for r in rows:
        is_better = True
        for i, g_idx in enumerate(goals):
            val = r[g_idx]
            est = estimates[i]
            # If ends in +, higher is better. If -, lower is better.
            if header[g_idx].endswith('+'):
                if not (est >= val): is_better = False
            else:
                if not (est <= val): is_better = False
        if is_better:
            better_count += 1
            
    p = better_count / len(rows)
    reason = f"This arrangement ranks in the top {int((1-p)*100)}% of all teams based on its projected retention metrics."
    
    return (p, reason)

import csv

def spot(header, rows):
    """
    Identifies rows that look wrong by checking for outliers in numeric columns.
    We calculate the Z-score for each numeric input and flag rows where any 
    value is more than 3 standard deviations from the mean.
    """
    numeric_indices = [i for i, h in enumerate(header) if h[0].isupper()]
    
    # Calculate means and std devs for numeric columns
    stats = {}
    for i in numeric_indices:
        vals = [r[i] for r in rows if isinstance(r[i], (int, float))]
        if not vals:
            continue
        mean = sum(vals) / len(vals)
        std = (sum((x - mean) ** 2 for x in vals) / len(vals)) ** 0.5
        stats[i] = (mean, std)
    
    # Score rows based on how many standard deviations they are from the mean
    # A higher score indicates more "suspicious" values
    suspicious = []
    for idx, row in enumerate(rows):
        row_score = 0
        for i in numeric_indices:
            if i in stats:
                mean, std = stats[i]
                if std > 0:
                    z = abs(row[i] - mean) / std
                    if z > 3:
                        row_score += z
        if row_score > 0:
            suspicious.append((row_score, idx))
            
    # Sort by score descending
    suspicious.sort(key=lambda x: x[0], reverse=True)
    return [item[1] for item in suspicious]

import csv
import math

def flow(header, row, seen):
    """
    Detects if a team's composition has drifted significantly from the 
    historical norm. We monitor the 'MonthlyIncome' and 'Age' as proxies 
    for team stability. If the current row's values are more than 2 
    standard deviations away from the mean of the 'seen' population, 
    we flag the team as having changed.
    """
    if len(seen) < 30:
        return False

    # Focus on key numeric indicators of team composition
    indicators = ['MonthlyIncome', 'Age']
    indices = [header.index(col) for col in indicators]
    
    for idx in indices:
        vals = [r[idx] for r in seen]
        mean = sum(vals) / len(vals)
        variance = sum((x - mean) ** 2 for x in vals) / len(vals)
        std = math.sqrt(variance)
        
        if std > 0:
            z_score = abs(row[idx] - mean) / std
            # If the new arrival is a significant outlier, flag the team
            if z_score > 2.0:
                return True
                
    return False

import csv

def blame(header, rows):
    """
    Identifies input columns that have the least correlation with tenure.
    We identify the 'worst' rows (bottom 25% by tenure) and compare the 
    distribution of input columns against the population. Columns with 
    the smallest deviation from the population mean are considered 
    'irrelevant' to the outcome.
    """
    years_idx = header.index('YearsAtCompany+')
    
    # Identify input columns (excluding goals)
    input_indices = [
        i for i, h in enumerate(header) 
        if not (h.endswith('+') or h.endswith('-'))
    ]
    
    # Sort rows by tenure to find the bottom 25%
    sorted_rows = sorted(rows, key=lambda r: r[years_idx])
    worst_rows = sorted_rows[:len(rows) // 4]
    
    # Calculate deviation scores for each input column
    scores = []
    for i in input_indices:
        col_data = [r[i] for r in rows if isinstance(r[i], (int, float))]
        if not col_data:
            continue
            
        pop_mean = sum(col_data) / len(col_data)
        worst_mean = sum(r[i] for r in worst_rows if isinstance(r[i], (int, float))) / len(worst_rows)
        
        # Score based on absolute difference from population mean
        # Smaller difference = less influence on the 'worst' group
        if pop_mean != 0:
            diff = abs(worst_mean - pop_mean) / pop_mean
            scores.append((diff, header[i]))
        else:
            scores.append((0.0, header[i]))
            
    # Return columns with the lowest deviation scores (least explanatory)
    scores.sort(key=lambda x: x[0])
    return [s[1] for s in scores[:5]]
