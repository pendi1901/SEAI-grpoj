import csv

def fix(header, row, levers):
    """
    Recommends changes to a 'bad' row (short tenure) to improve retention.
    We prioritize increasing pay (MonthlyIncome), increasing stock options,
    and reducing overtime.
    """
    recommendations = {}
    
    # Strategy:
    # 1. If overtime is 'Yes', change to 'No'
    # 2. If MonthlyIncome is low, increase it
    # 3. If StockOptionLevel is low, increase it
    
    if 'overTime' in levers and row[header.index('overTime')] == 'Yes':
        recommendations['overTime'] = 'No'
    
    if 'MonthlyIncome' in levers:
        low, high = levers['MonthlyIncome']
        current = row[header.index('MonthlyIncome')]
        if current < high:
            recommendations['MonthlyIncome'] = high
            
    if 'StockOptionLevel' in levers:
        low, high = levers['StockOptionLevel']
        current = row[header.index('StockOptionLevel')]
        if current < high:
            recommendations['StockOptionLevel'] = high

    # Ensure we return at least one change if possible
    if not recommendations and levers:
        for col, val in levers.items():
            if isinstance(val, list):
                recommendations[col] = val[0]
            else:
                recommendations[col] = val[1]
                
    return recommendations

if __name__ == "__main__":
    filename = 'WA_Fn-UseC_-HR-Employee-Attrition.n1000.train.csv'
    rows = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = []
            for i, val in enumerate(row):
                if header[i][0].isupper():
                    try:
                        processed_row.append(float(val))
                    except ValueError:
                        processed_row.append(0.0)
                else:
                    processed_row.append(val)
            rows.append(processed_row)

    # Define levers based on customer request
    levers = {
        'businessTravel': ['Non-Travel', 'Travel_Rarely', 'Travel_Frequently'],
        'overTime': ['No', 'Yes'],
        'TrainingTimesLastYear': (0.0, 6.0),
        'MonthlyIncome': (1000.0, 20000.0),
        'StockOptionLevel': (0.0, 3.0)
    }

    # Identify a 'bad' row (e.g., lowest tenure)
    years_idx = header.index('YearsAtCompany+')
    bad_row = min(rows, key=lambda r: r[years_idx])
    
    recommendations = fix(header, bad_row, levers)
    
    print("Analysis: Recommended changes for employees with short tenure:")
    for col, val in recommendations.items():
        print(f"- Change '{col}' to {val}")
    
    print("\nLogic: We prioritize reducing overtime and increasing compensation/benefits to improve retention.")
