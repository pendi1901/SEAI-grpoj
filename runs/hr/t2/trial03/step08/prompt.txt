    Estimates the first goal column ('YearsAtCompany+') for test rows.

    goal_idx = header.index(goal_col)

    input_indices = [
        i for i, h in enumerate(header) 
        if h[0].isupper() and not (h.endswith('+') or h.endswith('-'))
    ]
    
    predictions = []
    
    for t_row in test:
        # Calculate Euclidean distance to all training rows
        distances = []
        for r_row in train:
            dist = 0
            for i in input_indices:
                # Normalize distance calculation
                dist += (t_row[i] - r_row[i]) ** 2

        top_k = distances[:5]
        avg_val = sum(d[1] for d in top_k) / len(top_k)
        predictions.append(avg_val)

import csv
import math

def certify(xs, ys):
    """
    Performs a simple two-sample t-test (Welch's) to determine if the means
    of two groups (Overtime vs No Overtime) are significantly different.
    Returns True if the difference is likely noise (p > 0.05), False otherwise.
    """
    n1, n2 = len(xs), len(ys)
    if n1 < 2 or n2 < 2:
        return True
    
    m1, m2 = sum(xs) / n1, sum(ys) / n2
    v1 = sum((x - m1) ** 2 for x in xs) / (n1 - 1)
    v2 = sum((y - m2) ** 2 for y in ys) / (n2 - 1)
    
    # Standard error of difference
    se = math.sqrt((v1 / n1) + (v2 / n2))
    if se == 0:
        return m1 == m2
    
    t_stat = abs(m1 - m2) / se
    # For large samples, t > 1.96 corresponds to p < 0.05
    return t_stat < 1.96

import csv

def flow(header, row, seen):
    """
    Decides if the current employee needs a 'what would keep you here' conversation.
    We target employees who are high performers (high YearsAtCompany+) but 
    might be at risk. We use a simple threshold: if the employee's tenure 
    is in the top 10% of the 'seen' population, we flag them for a conversation.
    """
    years_idx = header.index('YearsAtCompany+')
    current_years = row[years_idx]
    
    if not seen:
        return False
    
    # Get all previous years of tenure
    seen_years = [r[years_idx] for r in seen]
    
    # Calculate the 90th percentile threshold of seen data
    seen_years.sort()
    threshold_idx = int(len(seen_years) * 0.9)
    threshold = seen_years[threshold_idx]
    
    # If this employee has been here longer than 90% of the people we've seen,
    # they are a high-tenure asset worth checking in on.
    return current_years >= threshold

def blame(header, rows):
    """
    Identifies input columns that correlate with short tenure (low YearsAtCompany+).
    We define 'worst' as the bottom 25% of employees by tenure.
    We then compare the average values of input columns for these 'worst' 
    employees against the population average to find the strongest drivers.
    """
    years_idx = header.index('YearsAtCompany+')
    
    # Filter out goal columns and identify input columns
    input_indices = [
        i for i, h in enumerate(header) 
        if not (h.endswith('+') or h.endswith('-'))
    ]
    
    # Sort rows by tenure to find the bottom 25%
    sorted_rows = sorted(rows, key=lambda r: r[years_idx])
    worst_rows = sorted_rows[:len(rows) // 4]
    
    # Calculate average of input columns for the worst group
    # We use a simple heuristic: columns where the worst group deviates 
    # most from the population mean are likely explanatory.
    scores = []
    for i in input_indices:
        col_data = [r[i] for r in rows if isinstance(r[i], (int, float))]
        if not col_data:
            continue
            
        pop_mean = sum(col_data) / len(col_data)
        worst_mean = sum(r[i] for r in worst_rows if isinstance(r[i], (int, float))) / len(worst_rows)
        
        # Score based on absolute difference from population mean
        if pop_mean != 0:
            diff = abs(worst_mean - pop_mean) / pop_mean
            scores.append((diff, header[i]))
            
    # Return top 3 columns that explain the short stays
    scores.sort(key=lambda x: x[0], reverse=True)
    return [s[1] for s in scores[:3]]

def justify(header, rows, row):
    """
    Estimates the performance of a row and returns its percentile rank 
    relative to the provided dataset.
    """
    # Identify goal columns
    goals = [i for i, h in enumerate(header) if h.endswith('+') or h.endswith('-')]
    
    # Estimate the goal values for the input row using KNN from step02
    # We use the first 700 rows as training data for the estimation
    train_data = rows[:700]
    
    # Simple estimation: average of the goal values of the 5 nearest neighbors
    input_indices = [i for i, h in enumerate(header) if h[0].isupper() and i not in goals]
    
    distances = []
    for r in train_data:
        dist = sum((row[i] - r[i])**2 for i in input_indices)**0.5
        distances.append((dist, [r[g] for g in goals]))
    
    distances.sort(key=lambda x: x[0])
    top_k = distances[:5]
    
    # Estimated goal values
    estimates = [sum(d[1][i] for d in top_k) / len(top_k) for i in range(len(goals))]
    
    # Calculate how many rows this row is "better than"
    better_count = 0
    for r in rows:
        is_better = True
        for i, g_idx in enumerate(goals):
            val = r[g_idx]
            est = estimates[i]
            # If ends in +, higher is better. If -, lower is better.
            if header[g_idx].endswith('+'):
                if not (est >= val): is_better = False
            else:
                if not (est <= val): is_better = False
        if is_better:
            better_count += 1
            
    p = better_count / len(rows)
    reason = f"This arrangement ranks in the top {int((1-p)*100)}% of all teams based on its projected retention metrics."
    
    return (p, reason)

def choose(header, rows, label):
    """
    Finds the best working arrangement by sampling 50 rows.
    We use a simple strategy: sample 50 rows, label them, 
    and pick the one with the highest 'YearsAtCompany+'.
    """
    # Identify goal column
    goal_col = [h for h in header if h.endswith('+') or h.endswith('-')][0]
    goal_idx = header.index(goal_col)
    
    # We can label up to 50 rows. Let's pick 50 indices spread across the dataset.
    n = len(rows)
    step = max(1, n // 50)
    indices_to_sample = [i for i in range(0, n, step)][:50]
    
    best_idx = -1
    best_val = -float('inf')
    
    for i in indices_to_sample:
        # label(i) returns the true goal values for row i
        true_goals = label(i)
        val = true_goals[0] # Assuming single goal for this specific task
        
        if val > best_val:
            best_val = val
            best_idx = i
            
    return best_idx
