import csv
import step07

def choose(header, rows, label):
    """
    Finds the best working arrangement for 'Laboratory Technician' staff
    by sampling up to 50 rows from that specific subset.
    """
    goal_col = [h for h in header if h.endswith('+') or h.endswith('-')][0]
    role_idx = header.index('jobRole')
    
    # Filter rows to only include Laboratory Technicians
    lab_indices = [i for i, r in enumerate(rows) if r[role_idx] == 'Laboratory Technician']
    
    # If no lab staff found, fallback to all rows
    if not lab_indices:
        lab_indices = range(len(rows))
        
    # Sample up to 50 indices from the lab staff
    n = len(lab_indices)
    step = max(1, n // 50)
    indices_to_sample = lab_indices[::step][:50]
    
    best_idx = -1
    best_val = -float('inf')
    
    for i in indices_to_sample:
        # label(i) returns the true goal values for row i
        true_goals = label(i)
        val = true_goals[0] 
        
        if val > best_val:
            best_val = val
            best_idx = i
            
    return best_idx

    filename = 'WA_Fn-UseC_-HR-Employee-Attrition.n1000.train.csv'

        header = next(reader)

                if header[i][0].isupper():

                        processed_row.append(float(val))
                    except ValueError:
                        processed_row.append(0.0)

                    processed_row.append(val)
            rows.append(processed_row)

    goal_idx = [i for i, h in enumerate(header) if h.endswith('+') or h.endswith('-')][0]
    
    # Store true values to simulate the label function
    true_data = {i: [rows[i][goal_idx]] for i in range(len(rows))}
    
    # Mask the rows
    for r in rows:
        r[goal_idx] = "?"
        
    def label(i):
        return true_data[i]
    
    best_row_idx = choose(header, rows, label)
