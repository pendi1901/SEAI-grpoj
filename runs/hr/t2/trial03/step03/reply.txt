import csv
import math

def certify(xs, ys):
    """
    Performs a simple two-sample t-test (Welch's) to determine if the means
    of two groups (Overtime vs No Overtime) are significantly different.
    Returns True if the difference is likely noise (p > 0.05), False otherwise.
    """
    n1, n2 = len(xs), len(ys)
    if n1 < 2 or n2 < 2:
        return True
    
    m1, m2 = sum(xs) / n1, sum(ys) / n2
    v1 = sum((x - m1) ** 2 for x in xs) / (n1 - 1)
    v2 = sum((y - m2) ** 2 for y in ys) / (n2 - 1)
    
    # Standard error of difference
    se = math.sqrt((v1 / n1) + (v2 / n2))
    if se == 0:
        return m1 == m2
    
    t_stat = abs(m1 - m2) / se
    # For large samples, t > 1.96 corresponds to p < 0.05
    return t_stat < 1.96

if __name__ == "__main__":
    filename = 'WA_Fn-UseC_-HR-Employee-Attrition.n1000.train.csv'
    rows = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = []
            for i, val in enumerate(row):
                if header[i][0].isupper():
                    processed_row.append(float(val))
                else:
                    processed_row.append(val)
            rows.append(processed_row)

    ot_idx = header.index('overTime')
    years_idx = header.index('YearsAtCompany+')
    
    yes_ot = [r[years_idx] for r in rows if r[ot_idx] == 'Yes']
    no_ot = [r[years_idx] for r in rows if r[ot_idx] == 'No']
    
    is_noise = certify(yes_ot, no_ot)
    
    print("Analysis: Do people who work overtime leave their role sooner?")
    if is_noise:
        print("Result: The observed difference is likely just noise.")
    else:
        print("Result: There is a statistically significant difference in tenure.")
