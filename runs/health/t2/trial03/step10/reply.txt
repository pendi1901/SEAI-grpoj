import csv
import random
from step01 import remember
from step07 import choose

def race(header, rows, label):
    """
    Compares three strategies: random, greedy (first 50), and suspicious (step09).
    Budget: 50 labels total.
    """
    budget = 50
    tried = ["random", "greedy", "suspicious"]
    
    # Helper to score a row based on goals
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')
    
    # We need global min/max for scoring. Since we only have 50 labels,
    # we use the full dataset's range if available, or estimate from labels.
    # To be safe, we use the known ranges from the full file.
    mres = [r[mre_idx] for r in rows if isinstance(r[mre_idx], float)]
    accs = [r[acc_idx] for r in rows if isinstance(r[acc_idx], float)]
    preds = [r[pred_idx] for r in rows if isinstance(r[pred_idx], float)]
    
    def get_score(row):
        def norm(v, mi, ma): return (v - mi) / (ma - mi) if ma > mi else 0
        return (1 - norm(row[mre_idx], min(mres), max(mres))) + \
               norm(row[acc_idx], min(accs), max(accs)) + \
               norm(row[pred_idx], min(preds), max(preds))

    # Strategy 1: Random
    rand_indices = random.sample(range(len(rows)), 16)
    best_rand = {"score": -1, "idx": -1}
    for i in rand_indices:
        goals = label(i)
        row = [goals[0], goals[1], goals[2]] + rows[i][3:]
        s = get_score(row)
        if s > best_rand["score"]: best_rand = {"score": s, "idx": i}

    # Strategy 2: Greedy (first 17)
    best_greedy = {"score": -1, "idx": -1}
    for i in range(17):
        goals = label(i)
        row = [goals[0], goals[1], goals[2]] + rows[i][3:]
        s = get_score(row)
        if s > best_greedy["score"]: best_greedy = {"score": s, "idx": i}

    # Strategy 3: Suspicious (last 17)
    from step09 import spot
    suspicious = spot(header, rows)
    best_susp = {"score": -1, "idx": -1}
    for i in suspicious[:17]:
        goals = label(i)
        row = [goals[0], goals[1], goals[2]] + rows[i][3:]
        s = get_score(row)
        if s > best_susp["score"]: best_susp = {"score": s, "idx": i}

    # Determine winner
    results = [("random", best_rand), ("greedy", best_greedy), ("suspicious", best_susp)]
    winner = max(results, key=lambda x: x[1]["score"])
    
    return {"winner": winner[0], "row": winner[1]["idx"], "tried": tried}

if __name__ == "__main__":
    filename = 'Health-ClosedIssues0000.n1000.train.csv'
    rows = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            rows.append([val if header[i] == 'criterion' else float(val) for i, val in enumerate(row)])

    def label(i):
        return [rows[i][0], rows[i][1], rows[i][2]]

    masked_rows = [r[:] for r in rows]
    for r in masked_rows:
        r[0] = r[1] = r[2] = "?"

    result = race(header, masked_rows, label)
    print(f"Winner: {result['winner']}")
    print(f"Best row index found: {result['row']}")
    print(f"Strategies tried: {', '.join(result['tried'])}")
