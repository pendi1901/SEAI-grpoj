import csv
from step01 import remember

def choose(header, rows, label):
    """
    Finds the best setup by sampling 50 rows.
    We use a simple strategy: evaluate the first 50 rows, 
    calculate their scores, and return the index of the best one.
    """
    # We are allowed 50 calls to label().
    # We will sample the first 50 rows to find the best among them.
    num_samples = 50
    best_score = -1.0
    best_idx = 0
    
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    # Collect all rows to normalize properly
    # Since we only have 50 labels, we use the labels to get the true values
    # for the first 50 rows and compare them.
    sampled_data = []
    for i in range(num_samples):
        goals = label(i)
        # Reconstruct row with true goals
        row = [goals[0], goals[1], goals[2]] + rows[i][3:]
        sampled_data.append(row)

    # Calculate scores for the sampled rows
    mres = [r[mre_idx] for r in sampled_data]
    accs = [r[acc_idx] for r in sampled_data]
    preds = [r[pred_idx] for r in sampled_data]

    min_mre, max_mre = min(mres), max(mres)
    min_acc, max_acc = min(accs), max(accs)
    min_pred, max_pred = min(preds), max(preds)

    def norm(v, mi, ma): return (v - mi) / (ma - mi) if ma > mi else 0

    for i in range(num_samples):
        r = sampled_data[i]
        score = (1 - norm(r[mre_idx], min_mre, max_mre)) + \
                norm(r[acc_idx], min_acc, max_acc) + \
                norm(r[pred_idx], min_pred, max_pred)
        
        if score > best_score:
            best_score = score
            best_idx = i
            
    return best_idx

if __name__ == "__main__":
    filename = 'Health-ClosedIssues0000.n1000.train.csv'
    rows = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = [val if header[i] == 'criterion' else float(val) 
                             for i, val in enumerate(row)]
            rows.append(processed_row)

    # Define label function for the harness
    def label(i):
        return [rows[i][0], rows[i][1], rows[i][2]]

    # Mask the rows for the choose function
    masked_rows = [r[:] for r in rows]
    for r in masked_rows:
        r[0] = r[1] = r[2] = "?"

    best_idx = choose(header, masked_rows, label)
    print(f"The best setup found within 50 trials is at index {best_idx}.")
