import csv

def remember(header, rows):
    """
    Identifies the top 5% of rows based on multi-objective optimization.
    Goals: MRE- (minimize), ACC+ (maximize), PRED40+ (maximize).
    We normalize the values and calculate a score for each row.
    """
    # Identify column indices
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    # Extract values for normalization
    mres = [r[mre_idx] for r in rows]
    accs = [r[acc_idx] for r in rows]
    preds = [r[pred_idx] for r in rows]

    min_mre, max_mre = min(mres), max(mres)
    min_acc, max_acc = min(accs), max(accs)
    min_pred, max_pred = min(preds), max(preds)

    def normalize(val, min_v, max_v):
        if max_v == min_v: return 0.0
        return (val - min_v) / (max_v - min_v)

    scores = []
    for i, row in enumerate(rows):
        # Normalize: MRE (lower is better -> 1 - norm), ACC/PRED (higher is better -> norm)
        n_mre = 1.0 - normalize(row[mre_idx], min_mre, max_mre)
        n_acc = normalize(row[acc_idx], min_acc, max_acc)
        n_pred = normalize(row[pred_idx], min_pred, max_pred)
        
        # Simple aggregate score
        score = n_mre + n_acc + n_pred
        scores.append((score, i))

    # Sort by score descending
    scores.sort(key=lambda x: x[0], reverse=True)
    
    # Return top 5% (max 5% of 800 = 40)
    limit = max(1, len(rows) // 20)
    return [item[1] for item in scores[:limit]]

import csv
from step01 import remember

def guess(header, train, test):
    """
    Estimates the first goal column (MRE-) for test rows.
    Uses a simple k-nearest neighbors approach based on input features.
    """
    # Identify indices
    goal_idx = 0  # MRE- is the first column
    input_indices = [i for i, name in enumerate(header) if name not in ['MRE-', 'ACC+', 'PRED40+']]
    
    # Pre-calculate numeric ranges for normalization
    mins = []
    maxs = []
    for i in input_indices:
        col_vals = [r[i] for r in train if isinstance(r[i], (int, float))]
        if col_vals:
            mins.append(min(col_vals))
            maxs.append(max(col_vals))
        else:
            mins.append(0)
            maxs.append(1)

    def dist(r1, r2):
        d = 0
        for idx, i in enumerate(input_indices):
            v1, v2 = r1[i], r2[i]
            if isinstance(v1, str):
                d += 0 if v1 == v2 else 1
            else:
                d += ((v1 - v2) / (maxs[idx] - mins[idx] + 1e-9))**2
        return d**0.5

    predictions = []
    for t_row in test:
        # Find 5 nearest neighbors
        neighbors = sorted(train, key=lambda x: dist(x, t_row))[:5]
        avg_mre = sum(n[goal_idx] for n in neighbors) / len(neighbors)
        predictions.append(avg_mre)
    
    return predictions

import csv

def certify(xs, ys):
    """
    Returns True if the difference between two distributions is likely noise,
    False if they are statistically different. Uses a simple T-test approach.
    """
    if not xs or not ys:
        return True
    
    n1, n2 = len(xs), len(ys)
    m1, m2 = sum(xs) / n1, sum(ys) / n2
    
    v1 = sum((x - m1) ** 2 for x in xs) / (n1 - 1) if n1 > 1 else 0
    v2 = sum((y - m2) ** 2 for y in ys) / (n2 - 1) if n2 > 1 else 0
    
    # Standard error of the difference
    se = math.sqrt((v1 / n1) + (v2 / n2))
    if se == 0:
        return m1 == m2
    
    t_stat = abs(m1 - m2) / se
    # Threshold for 95% confidence (approx 1.96 for large samples)
    return t_stat < 1.96

import csv

def blame(header, rows):
    """
    Identifies input columns that correlate most strongly with 'bad' performance.
    'Bad' is defined as the bottom 25% of rows based on the remember() score.
    We calculate the correlation between each input and the score.
    """
    # Identify goal indices
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')
    
    # Calculate scores for all rows to identify the 'bad' ones
    # remember() returns indices of the top 5%. We want the bottom 25%.
    # Re-implementing score logic locally for efficiency
    mres = [r[mre_idx] for r in rows]
    accs = [r[acc_idx] for r in rows]
    preds = [r[pred_idx] for r in rows]
    
    min_mre, max_mre = min(mres), max(mres)
    min_acc, max_acc = min(accs), max(accs)
    min_pred, max_pred = min(preds), max(preds)

    def norm(v, mi, ma): return (v - mi) / (ma - mi) if ma > mi else 0
    
    scores = []
    for r in rows:
        s = (1 - norm(r[mre_idx], min_mre, max_mre)) + \
            norm(r[acc_idx], min_acc, max_acc) + \
            norm(r[pred_idx], min_pred, max_pred)
        scores.append(s)
    
    # Identify input columns
    input_cols = [i for i, h in enumerate(header) if h not in ['MRE-', 'ACC+', 'PRED40+']]
    
    # Find correlation: how much does a feature value deviate from the mean
    # in the worst 25% of rows?
    threshold = sorted(scores)[len(scores) // 4]
    bad_rows = [rows[i] for i, s in enumerate(scores) if s <= threshold]
    
    # Simple heuristic: feature importance = abs(mean(bad) - mean(all))
    impacts = []
    for i in input_cols:
        col_vals = [r[i] for r in rows if not isinstance(r[i], str)]
        if not col_vals: continue
        
        bad_vals = [r[i] for r in bad_rows if not isinstance(r[i], str)]
        if not bad_vals: continue
        
        diff = abs((sum(bad_vals)/len(bad_vals)) - (sum(col_vals)/len(col_vals)))
        impacts.append((diff, header[i]))
    
    impacts.sort(reverse=True)
    return [name for _, name in impacts[:2]]

def justify(header, rows, row):
    """
    Estimates the performance of the given row and compares it to the existing dataset.
    Returns (p, reason) where p is the percentile rank and reason is a summary.
    """
    # 1. Estimate the goals for the target row
    # Use guess() logic: find neighbors to estimate MRE-, ACC+, PRED40+
    # Since guess() only estimates MRE-, we extend the logic for all goals
    input_indices = [i for i, name in enumerate(header) if name not in ['MRE-', 'ACC+', 'PRED40+']]
    
    def get_dist(r1, r2):
        d = 0
        for i in input_indices:
            if isinstance(r1[i], str):
                d += 0 if r1[i] == r2[i] else 1
            else:
                d += ((r1[i] - r2[i]) / 100.0)**2 # Simplified normalization
        return d**0.5

    # Find 5 nearest neighbors to estimate all 3 goals
    neighbors = sorted(rows, key=lambda x: get_dist(x, row))[:5]
    est_mre = sum(n[0] for n in neighbors) / 5
    est_acc = sum(n[1] for n in neighbors) / 5
    est_pred = sum(n[2] for n in neighbors) / 5
    
    # 2. Calculate score for the target row
    # Re-using normalization logic from step01
    mres = [r[0] for r in rows]
    accs = [r[1] for r in rows]
    preds = [r[2] for r in rows]
    
    def norm(v, mi, ma): return (v - mi) / (ma - mi) if ma > mi else 0
    
    target_score = (1 - norm(est_mre, min(mres), max(mres))) + \
                   norm(est_acc, min(accs), max(accs)) + \
                   norm(est_pred, min(preds), max(preds))
    
    # 3. Compare against all rows
    scores = []
    for r in rows:
        s = (1 - norm(r[0], min(mres), max(mres))) + \
            norm(r[1], min(accs), max(accs)) + \
            norm(r[2], min(preds), max(preds))
        scores.append(s)
    
    better_than = [s for s in scores if s < target_score]
    p = len(better_than) / len(rows)
    
    reason = f"This setup is estimated to outperform {p:.1%} of all known configurations based on its predicted multi-objective score."
    return p, reason
