import csv
from step03 import certify as base_certify

def certify(xs, ys):
    """
    Returns True if the difference between two distributions is likely noise,
    False if they are statistically different.
    """
    return base_certify(xs, ys)

if __name__ == "__main__":
    filename = 'Health-ClosedIssues0000.n1000.train.csv'
    rows = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = [val if header[i] == 'criterion' else float(val) 
                             for i, val in enumerate(row)]
            rows.append(processed_row)

    # The customer asks about the choice of 'criterion' (the tool's built-in rule).
    # We compare the performance (ACC+) of 'absolute_error' vs 'squared_error'.
    crit_idx = header.index('criterion')
    acc_idx = header.index('ACC+')
    
    abs_err_accs = [r[acc_idx] for r in rows if r[crit_idx] == 'absolute_error']
    sq_err_accs = [r[acc_idx] for r in rows if r[crit_idx] == 'squared_error']
    
    is_noise = certify(abs_err_accs, sq_err_accs)
    
    print(f"Mean ACC+ for 'absolute_error': {sum(abs_err_accs)/len(abs_err_accs):.4f}")
    print(f"Mean ACC+ for 'squared_error': {sum(sq_err_accs)/len(sq_err_accs):.4f}")
    
    if is_noise:
        print("Conclusion: The difference between these rules is likely noise.")
        print("You can stop arguing about it; the choice does not make a significant difference.")
    else:
        print("Conclusion: The difference is statistically significant.")
        print("The choice of rule does matter for your performance.")
