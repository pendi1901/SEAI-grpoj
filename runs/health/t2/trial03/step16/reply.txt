import csv
from step06 import justify as base_justify

def justify(header, rows, row):
    """
    Justifies why 'biggest and deepest' is not always best by comparing
    the target row's performance against the distribution of all rows.
    """
    # The justify function from step06 already calculates the percentile rank
    # of a row based on its estimated multi-objective score.
    return base_justify(header, rows, row)

if __name__ == "__main__":
    filename = 'Health-ClosedIssues0000.n1000.train.csv'
    rows = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = [val if header[i] == 'criterion' else float(val) 
                             for i, val in enumerate(row)]
            rows.append(processed_row)

    # Define a "biggest, deepest" setup:
    # N_estimators: 200, Max_depth: 20, Min_sample_leaves: 1
    # We use a common criterion like 'squared_error'
    big_setup = [
        "?", "?", "?", 
        200.0, 'squared_error', 1.0, 0.0, 20.0
    ]
    
    p, reason = justify(header, rows, big_setup)
    
    print("Question: Why not simply use the biggest, deepest setup the machine can hold?")
    print(f"Answer: {reason}")
    print(f"Specifically, this configuration is estimated to be better than {p:.1%} of all known configurations.")
    print("Conclusion: Bigger is not always better; excessive complexity often leads to diminishing returns or overfitting.")
