import csv
from step01 import remember

def guess(header, train, test):
    """
    Estimates the first goal (MRE-) for each test row using a 
    k-nearest neighbors approach based on input columns.
    """
    # Identify input columns (those not ending in - or +)
    input_indices = [
        i for i, name in enumerate(header) 
        if not (name.endswith('-') or name.endswith('+'))
    ]
    
    # Identify the target column (first goal)
    target_idx = 0 
    
    def distance(row1, row2):
        d = 0
        for i in input_indices:
            val1, val2 = row1[i], row2[i]
            if isinstance(val1, str):  # Symbolic
                d += 0 if val1 == val2 else 1
            else:  # Numeric
                d += abs(val1 - val2)
        return d

    results = []
    for t_row in test:
        # Find the 5 nearest neighbors in the training set
        neighbors = sorted(train, key=lambda r: distance(r, t_row))[:5]
        # Return the average MRE of the neighbors
        avg_mre = sum(n[target_idx] for n in neighbors) / len(neighbors)
        results.append(avg_mre)
    
    return results

if __name__ == "__main__":
    filename = 'Health-ClosedIssues0000.n1000.train.csv'
    rows = []
    try:
        with open(filename, 'r') as f:
            reader = csv.reader(f)
            header = next(reader)
            for row in reader:
                processed_row = []
                for i, val in enumerate(row):
                    if header[i] == 'criterion':
                        processed_row.append(val)
                    else:
                        processed_row.append(float(val))
                rows.append(processed_row)
        
        # Use the first 700 as train, last 100 as test (simulated)
        train_data = rows[:700]
        test_data = [row[:] for row in rows[700:]]
        
        # Mask the goals in test data
        for row in test_data:
            for i, name in enumerate(header):
                if name.endswith('-') or name.endswith('+'):
                    row[i] = "?"
        
        predictions = guess(header, train_data, test_data)
        
        print("Forecast estimation complete.")
        print(f"Estimated MRE for the first 5 test setups: {[round(p, 4) for p in predictions[:5]]}")
        
    except FileNotFoundError:
        print(f"Error: {filename} not found.")
        exit(1)
