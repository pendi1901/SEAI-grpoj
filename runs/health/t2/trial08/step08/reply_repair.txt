import csv
from step01 import remember

def fix(header, row, levers):
    """
    Finds the smallest change to a 'bad' row's settings that improves its score.
    We use the remember logic from step01 to evaluate the impact of changes.
    """
    # Identify goal indices
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    def get_score(r):
        # Normalize goals based on the row's values.
        # Since we need to compare, we use the row's own metrics.
        # Higher is better: (1-MRE) + ACC + PRED
        s1 = 1.0 - r[mre_idx]
        s2 = r[acc_idx]
        s3 = r[pred_idx]
        return (s1 + s2 + s3) / 3.0

    best_change = {}
    best_score = get_score(row)
    
    for col_name, allowed in levers.items():
        if col_name not in header:
            continue
        idx = header.index(col_name)
        original_val = row[idx]
        
        candidates = []
        if isinstance(allowed, list):  # Symbolic
            candidates = [v for v in allowed if v != original_val]
        else:  # Numeric (low, high)
            # Try boundaries and midpoint if they differ from original
            low, high = allowed
            for val in [low, (low + high) / 2.0, high]:
                if val != original_val:
                    candidates.append(val)
            
        for val in candidates:
            new_row = list(row)
            new_row[idx] = val
            new_score = get_score(new_row)
            
            if new_score > best_score:
                best_score = new_score
                best_change = {col_name: val}
                
    return best_change

if __name__ == "__main__":
    filename = 'Health-ClosedIssues0000.n1000.train.csv'
    try:
        with open(filename, 'r') as f:
            reader = csv.reader(f)
            header = next(reader)
            rows = []
            for row in reader:
                processed = []
                for i, val in enumerate(row):
                    processed.append(val if header[i] == 'criterion' else float(val))
                rows.append(processed)
        
        # Define levers based on the dataset structure
        levers = {
            'N_estimators': (10.0, 200.0),
            'criterion': ['absolute_error', 'poisson', 'squared_error'],
            'Min_sample_leaves': (1.0, 20.0),
            'Min_impurity_decrease': (0.0, 10.0),
            'Max_depth': (1.0, 20.0)
        }
        
        # Pick a row that is not in the top 5% (a 'bad' row)
        best_indices = remember(header, rows)
        bad_row = None
        for i, row in enumerate(rows):
            if i not in best_indices:
                bad_row = row
                break
        
        if bad_row:
            recommendation = fix(header, bad_row, levers)
            print("Recommendation for improvement:")
            if recommendation:
                for col, val in recommendation.items():
                    print(f"Change '{col}' to {val}")
            else:
                print("No single lever change found to improve performance.")
        else:
            print("No bad rows found to improve.")
            
    except FileNotFoundError:
        print(f"Error: {filename} not found.")
        exit(1)
