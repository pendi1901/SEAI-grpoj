import csv

def remember(header, rows):
    """
    Identifies the top 5% of rows based on multi-objective optimization.
    Goals: MRE- (minimize), ACC+ (maximize), PRED40+ (maximize).
    We normalize values to [0, 1] and calculate a score for each row.
    """
    n = len(rows)
    if n == 0:
        return []

    # Identify column indices
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    # Extract columns for normalization
    mres = [row[mre_idx] for row in rows]
    accs = [row[acc_idx] for row in rows]
    preds = [row[pred_idx] for row in rows]

    min_mre, max_mre = min(mres), max(mres)
    min_acc, max_acc = min(accs), max(accs)
    min_pred, max_pred = min(preds), max(preds)

    def normalize(val, min_v, max_v, reverse=False):
        if max_v == min_v: return 0.0
        norm = (val - min_v) / (max_v - min_v)
        return (1.0 - norm) if reverse else norm

    scores = []
    for i in range(n):
        # Score = average of normalized goals (higher is better)
        s1 = normalize(rows[i][mre_idx], min_mre, max_mre, reverse=True)
        s2 = normalize(rows[i][acc_idx], min_acc, max_acc)
        s3 = normalize(rows[i][pred_idx], min_pred, max_pred)
        scores.append((s1 + s2 + s3) / 3.0)

    # Select top 5% (or at least 1)
    limit = max(1, n // 20)
    indexed_scores = sorted(enumerate(scores), key=lambda x: x[1], reverse=True)
    return [idx for idx, score in indexed_scores[:limit]]

def guess(header, train, test):
    """
    Estimates the first goal (MRE-) for each test row using a 
    k-nearest neighbors approach based on input columns.
    """
    # Identify input columns (those not ending in - or +)
    input_indices = [
        i for i, name in enumerate(header) 
        if not (name.endswith('-') or name.endswith('+'))
    ]
    
    # Identify the target column (first goal)
    target_idx = 0 
    
    def distance(row1, row2):
        d = 0
        for i in input_indices:
            val1, val2 = row1[i], row2[i]
            if isinstance(val1, str):  # Symbolic
                d += 0 if val1 == val2 else 1
            else:  # Numeric
                d += abs(val1 - val2)
        return d

    results = []
    for t_row in test:
        # Find the 5 nearest neighbors in the training set
        neighbors = sorted(train, key=lambda r: distance(r, t_row))[:5]
        # Return the average MRE of the neighbors
        avg_mre = sum(n[target_idx] for n in neighbors) / len(neighbors)
        results.append(avg_mre)
    
    return results

import csv
import math

def certify(xs, ys):
    """
    Returns True if the distributions of xs and ys are statistically similar 
    (i.e., the difference is likely noise), False if they are significantly different.
    Uses a simple T-test approach for independent samples.
    """
    n1, n2 = len(xs), len(ys)
    if n1 < 2 or n2 < 2:
        return True
    
    mean1, mean2 = sum(xs) / n1, sum(ys) / n2
    var1 = sum((x - mean1) ** 2 for x in xs) / (n1 - 1)
    var2 = sum((y - mean2) ** 2 for y in ys) / (n2 - 1)
    
    # Pooled standard error
    se = math.sqrt((var1 / n1) + (var2 / n2))
    if se == 0:
        return mean1 == mean2
    
    t_stat = abs(mean1 - mean2) / se
    # For a 95% confidence level, t-critical is approx 1.96
    return t_stat < 1.96

import csv

def blame(header, rows):
    """
    Identifies input columns that correlate most strongly with 'bad' rows.
    A row is 'bad' if it is in the bottom 25% of performance.
    We compare the distribution of each input column in the 'bad' set
    against the 'good' set (top 25%) to find the most significant difference.
    """
    # Identify goal indices and input indices
    goal_indices = [i for i, h in enumerate(header) if h.endswith('-') or h.endswith('+')]
    input_indices = [i for i, h in enumerate(header) if not (h.endswith('-') or h.endswith('+'))]

    # Calculate scores for all rows to rank them
    best_indices = remember(header, rows)
    n = len(rows)
    
    # Sort rows by score (descending)
    indexed_rows = sorted(enumerate(rows), key=lambda x: x[0] in best_indices, reverse=True)
    
    # Define bad rows as the bottom 25%
    split = n // 4
    bad_rows = [r for i, r in indexed_rows[-split:]]
    good_rows = [r for i, r in indexed_rows[:split]]

    def get_diff(idx):
        # Calculate mean difference between bad and good sets for a column
        vals_bad = [r[idx] for r in bad_rows]
        vals_good = [r[idx] for r in good_rows]
        
        if isinstance(vals_bad[0], str):
            # For symbolic, compare frequency of the most common value in bad rows
            mode_bad = max(set(vals_bad), key=vals_bad.count)
            return abs(vals_bad.count(mode_bad) / len(vals_bad) - vals_good.count(mode_bad) / len(vals_good))
        else:
            # For numeric, compare normalized mean difference
            mean_bad = sum(vals_bad) / len(vals_bad)
            mean_good = sum(vals_good) / len(vals_good)
            return abs(mean_bad - mean_good)

    # Rank inputs by their difference score
    diffs = [(header[i], get_diff(i)) for i in input_indices]
    diffs.sort(key=lambda x: x[1], reverse=True)
    
    # Return the top 2 inputs that explain the bad performance
    return [d[0] for d in diffs[:2]]

def justify(header, rows, row):
    """
    Estimates the performance of 'row' and compares it against all 'rows'.
    Returns (p, reason): p is the fraction of rows that 'row' is better than,
    and reason is a human-readable explanation.
    """
    # 1. Estimate the goals for the target row
    # We use the guess function from step02, but we need to handle all goals.
    # Since guess only does MRE-, we extend the logic to estimate all goals.
    
    def estimate_row(target_row):
        # Use k-NN (k=5) to estimate all goal columns
        input_indices = [i for i, h in enumerate(header) if not (h.endswith('-') or h.endswith('+'))]
        goal_indices = [i for i, h in enumerate(header) if h.endswith('-') or h.endswith('+')]
        
        def dist(r1, r2):
            d = 0
            for i in input_indices:
                if isinstance(r1[i], str):
                    d += 0 if r1[i] == r2[i] else 1
                else:
                    d += abs(r1[i] - r2[i])
            return d
        
        neighbors = sorted(rows, key=lambda r: dist(r, target_row))[:5]
        estimated = list(target_row)
        for g_idx in goal_indices:
            estimated[g_idx] = sum(n[g_idx] for n in neighbors) / len(neighbors)
        return estimated

    est_row = estimate_row(row)
    
    # 2. Compare against all rows
    # We use the scoring logic from step01 to compare
    all_data = rows + [est_row]
    best_indices = remember(header, all_data)
    
    # Calculate scores for all
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')
    
    def get_score(r):
        mres = [x[mre_idx] for x in all_data]
        accs = [x[acc_idx] for x in all_data]
        preds = [x[pred_idx] for x in all_data]
        
        def norm(v, vals, rev=False):
            mn, mx = min(vals), max(vals)
            if mx == mn: return 0.0
            n = (v - mn) / (mx - mn)
            return (1.0 - n) if rev else n
            
        return (norm(r[mre_idx], mres, True) + norm(r[acc_idx], accs) + norm(r[pred_idx], preds)) / 3.0

    target_score = get_score(est_row)
    better_count = sum(1 for r in rows if get_score(r) < target_score)
    p = better_count / len(rows)
    
    reason = f"This setup outperforms {p:.1%} of all historical configurations based on its estimated goal metrics."
    return p, reason
