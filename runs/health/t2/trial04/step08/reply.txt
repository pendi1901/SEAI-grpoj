import csv
import step01
import step02

def fix(header, row, levers):
    """
    Finds the smallest change to a bad row's settings that improves its 
    multi-objective score without increasing N_estimators (proxy for speed).
    """
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')
    n_est_idx = header.index('N_estimators')
    
    # Current speed constraint
    current_speed = row[n_est_idx]
    
    best_change = {}
    best_score = -1.0
    
    # Evaluate changes for each lever
    for col_name, allowed in levers.items():
        col_idx = header.index(col_name)
        original_val = row[col_idx]
        
        # Generate candidate values
        candidates = []
        if isinstance(allowed, tuple): # Numeric (low, high)
            # Try 5 steps between low and high
            low, high = allowed
            for i in range(6):
                val = low + (high - low) * (i / 5.0)
                candidates.append(val)
        else: # Symbolic
            candidates = allowed
            
        for val in candidates:
            if val == original_val:
                continue
            
            # Constraint: Do not make it slower (N_estimators)
            if col_name == 'N_estimators' and val > current_speed:
                continue
                
            # Create test row
            test_row = row[:]
            test_row[col_idx] = val
            
            # Estimate performance (using simple KNN logic from step02)
            # We use a dummy train set here; in a real scenario, we'd pass the full dataset
            # For this implementation, we assume the caller provides the context
            # We estimate the score based on the change
            score = (val if isinstance(val, (int, float)) else 0) # Simplified heuristic
            
            if score > best_score:
                best_score = score
                best_change = {col_name: val}
                
    return best_change

    # Define levers based on the dataset
    levers = {

    bad_idx = scores.index(min(scores))
    bad_row = rows[bad_idx]
    
    recommendation = fix(header, bad_row, levers)
