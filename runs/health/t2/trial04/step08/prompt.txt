import csv

def remember(header, rows):
    """
    Identifies the top 5% of rows based on multi-objective optimization.
    Goals: MRE- (minimize), ACC+ (maximize), PRED40+ (maximize).
    We normalize values to [0, 1] and use a simple weighted sum (dominance).
    """
    n = len(rows)
    if n == 0:
        return []

    # Identify column indices
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    # Extract columns for normalization
    mres = [r[mre_idx] for r in rows]
    accs = [r[acc_idx] for r in rows]
    preds = [r[pred_idx] for r in rows]

    min_mre, max_mre = min(mres), max(mres)
    min_acc, max_acc = min(accs), max(accs)
    min_pred, max_pred = min(preds), max(preds)

    def normalize(val, min_v, max_v, reverse=False):
        if max_v == min_v: return 0.5
        norm = (val - min_v) / (max_v - min_v)
        return 1.0 - norm if reverse else norm

    scores = []
    for i, row in enumerate(rows):
        # Calculate score: lower MRE is better, higher ACC/PRED is better
        s1 = normalize(row[mre_idx], min_mre, max_mre, reverse=True)
        s2 = normalize(row[acc_idx], min_acc, max_acc)
        s3 = normalize(row[pred_idx], min_pred, max_pred)
        # Average the normalized objectives
        scores.append((s1 + s2 + s3) / 3.0)

    # Get top 5% indices
    limit = max(1, n // 20)
    indexed_scores = sorted(enumerate(scores), key=lambda x: x[1], reverse=True)
    return [idx for idx, score in indexed_scores[:limit]]

import csv
import step01

def guess(header, train, test):
    """
    Estimates the first goal column (MRE-) for test rows.
    Uses a simple k-nearest neighbor approach based on input columns.
    """
    # Identify goal index (first column) and input indices
    goal_idx = 0
    input_indices = [i for i, h in enumerate(header) if h not in ['MRE-', 'ACC+', 'PRED40+']]
    
    # Pre-calculate numeric ranges for normalization
    mins = []
    maxs = []
    for i in input_indices:
        vals = [r[i] for r in train if isinstance(r[i], (int, float))]
        if vals:
            mins.append(min(vals))
            maxs.append(max(vals))
        else:
            mins.append(0)
            maxs.append(1)

    def distance(row1, row2):
        d = 0
        for j, idx in enumerate(input_indices):
            v1, v2 = row1[idx], row2[idx]
            if isinstance(v1, str):
                d += 1 if v1 != v2 else 0
            else:
                d += ((v1 - v2) / (maxs[j] - mins[j] + 1e-9))**2
        return d**0.5

    predictions = []
    for t_row in test:
        # Find 5 nearest neighbors
        dists = sorted([(distance(t_row, train_row), train_row[goal_idx]) for train_row in train])
        neighbors = dists[:5]
        avg_mre = sum(n[1] for n in neighbors) / len(neighbors)
        predictions.append(avg_mre)
        
    return predictions

import csv
import math

def certify(xs, ys):
    """
    Returns True if xs and ys are statistically similar (no significant difference),
    False if they differ significantly. Uses a simple T-test approach.
    """
    n1, n2 = len(xs), len(ys)
    if n1 < 2 or n2 < 2:
        return True
    
    mean1, mean2 = sum(xs) / n1, sum(ys) / n2
    var1 = sum((x - mean1)**2 for x in xs) / (n1 - 1)
    var2 = sum((y - mean2)**2 for y in ys) / (n2 - 1)
    
    # Standard error of the difference
    se = math.sqrt((var1 / n1) + (var2 / n2))
    if se == 0:
        return mean1 == mean2
    
    t_stat = abs(mean1 - mean2) / se
    # For a 95% confidence level, t-critical is approx 2.0
    return t_stat < 2.0

import csv
import step01

def flow(header, row, seen):
    """
    Decides whether to accept the current row based on whether it is 
    better than any previously seen row.
    
    A row is better if it dominates or performs better on the weighted 
    average of normalized objectives (MRE-, ACC+, PRED40+).
    """
    if not seen:
        return True

    # Identify indices
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    # Combine current row with seen rows to normalize
    all_rows = seen + [row]
    
    def get_score(r, rows_data):
        mres = [x[mre_idx] for x in rows_data]
        accs = [x[acc_idx] for x in rows_data]
        preds = [x[pred_idx] for x in rows_data]
        
        def norm(val, vals, reverse=False):
            min_v, max_v = min(vals), max(vals)
            if max_v == min_v: return 0.5
            n = (val - min_v) / (max_v - min_v)
            return 1.0 - n if reverse else n
        
        s1 = norm(r[mre_idx], mres, reverse=True)
        s2 = norm(r[acc_idx], accs)
        s3 = norm(r[pred_idx], preds)
        return (s1 + s2 + s3) / 3.0

    # If current row is the best seen so far, accept it
    current_score = get_score(row, all_rows)
    seen_scores = [get_score(r, all_rows) for r in seen]
    
    return current_score >= max(seen_scores)

import csv

def blame(header, rows):
    """
    Identifies input columns that correlate with poor performance.
    'Bad' rows are defined as the bottom 10% based on the multi-objective score.
    We compare the distribution of input values in the bad rows vs the rest.
    """
    # Identify goal indices
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')
    
    # Identify input indices
    input_indices = [i for i, h in enumerate(header) if h not in ['MRE-', 'ACC+', 'PRED40+']]
    
    # Score rows using step01 logic
    scores = []
    for row in rows:
        # Simple normalization for scoring
        mres = [r[mre_idx] for r in rows]
        accs = [r[acc_idx] for r in rows]
        preds = [r[pred_idx] for r in rows]
        
        def norm(val, vals, rev):
            mn, mx = min(vals), max(vals)
            if mx == mn: return 0.5
            n = (val - mn) / (mx - mn)
            return 1.0 - n if rev else n
            
        s = (norm(row[mre_idx], mres, True) + norm(row[acc_idx], accs, False) + norm(row[pred_idx], preds, False)) / 3.0
        scores.append(s)
        
    # Define bad rows (bottom 10%)
    indexed_scores = sorted(enumerate(scores), key=lambda x: x[1])
    limit = max(1, len(rows) // 10)
    bad_indices = [idx for idx, score in indexed_scores[:limit]]
    
    # Calculate average difference in inputs between bad rows and the rest
    impacts = []
    for i in input_indices:
        bad_vals = [rows[idx][i] for idx in bad_indices if isinstance(rows[idx][i], (int, float))]
        all_vals = [r[i] for r in rows if isinstance(r[i], (int, float))]
        
        if not bad_vals or not all_vals:
            continue
            
        avg_bad = sum(bad_vals) / len(bad_vals)
        avg_all = sum(all_vals) / len(all_vals)
        
        # We look for inputs where the bad rows deviate significantly from the mean
        impacts.append((header[i], abs(avg_bad - avg_all)))
        
    # Return top 2 inputs that correlate most with bad performance
    impacts.sort(key=lambda x: x[1], reverse=True)
    return [x[0] for x in impacts[:2]]

def justify(header, rows, row):
    """
    Estimates the performance of a given row and compares it against all other rows.
    Returns (p, reason) where p is the fraction of rows outperformed by the input row.
    """
    # 1. Estimate the missing goal values for the input row
    # We use the guess logic from step02 to fill in the "?" values
    # Since step02.guess returns a list of predictions, we take the first one
    estimated_mre = step02.guess(header, rows, [row])[0]
    
    # For ACC+ and PRED40+, we use the average of the top 5% (from step01)
    # as a proxy for a "good" setup's performance
    best_indices = step01.remember(header, rows)
    avg_acc = sum(rows[i][1] for i in best_indices) / len(best_indices)
    avg_pred = sum(rows[i][2] for i in best_indices) / len(best_indices)
    
    # Fill the row with estimates
    row[0], row[1], row[2] = estimated_mre, avg_acc, avg_pred
    
    # 2. Calculate how many rows this row is better than
    # A row is better if it has lower MRE, higher ACC, and higher PRED
    better_count = 0
    for r in rows:
        # Compare goals: MRE (lower is better), ACC/PRED (higher is better)
        if (row[0] < r[0]) and (row[1] > r[1]) and (row[2] > r[2]):
            better_count += 1
            
    p = better_count / len(rows)
    reason = f"This setup outperforms {p:.1%} of all known configurations based on its estimated performance metrics."
    
    return (p, reason)

def choose(header, rows, label):
    """
    Finds the best setup by sampling 50 rows. 
    We use a simple strategy: evaluate the first 50 rows, 
    then pick the one with the highest multi-objective score.
    """
    # We can label at most 50 rows.
    n_to_sample = min(50, len(rows))
    
    # Label the first n_to_sample rows
    for i in range(n_to_sample):
        goals = label(i)
        rows[i][0] = goals[0] # MRE-
        rows[i][1] = goals[1] # ACC+
        rows[i][2] = goals[2] # PRED40+
        
    # Use step01 logic to score the labeled rows
    # We only consider the rows we have labeled
    labeled_rows = rows[:n_to_sample]
    
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')
    
    mres = [r[mre_idx] for r in labeled_rows]
    accs = [r[acc_idx] for r in labeled_rows]
    preds = [r[pred_idx] for r in labeled_rows]
    
    min_mre, max_mre = min(mres), max(mres)
    min_acc, max_acc = min(accs), max(accs)
    min_pred, max_pred = min(preds), max(preds)
    
    def normalize(val, min_v, max_v, reverse=False):
        if max_v == min_v: return 0.5
        norm = (val - min_v) / (max_v - min_v)
        return 1.0 - norm if reverse else norm
    
    best_idx = -1
    best_score = -1.0
    
    for i in range(n_to_sample):
        s1 = normalize(rows[i][mre_idx], min_mre, max_mre, reverse=True)
        s2 = normalize(rows[i][acc_idx], min_acc, max_acc)
        s3 = normalize(rows[i][pred_idx], min_pred, max_pred)
        score = (s1 + s2 + s3) / 3.0
        
        if score > best_score:
            best_score = score
            best_idx = i
            
    return best_idx
