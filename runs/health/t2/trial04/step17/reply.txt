import csv

def choose(header, rows, label):
    """
    Finds a setup that forecasts well (high multi-objective score) 
    and is small (low N_estimators).
    
    We sample 50 rows, label them, and pick the one that maximizes:
    Score = (Multi-objective score) - (Penalty for N_estimators).
    """
    n_to_sample = min(50, len(rows))
    
    # Label the first 50 rows
    for i in range(n_to_sample):
        goals = label(i)
        rows[i][0], rows[i][1], rows[i][2] = goals
        
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')
    n_est_idx = header.index('N_estimators')
    
    # Normalize objectives for scoring
    labeled_rows = rows[:n_to_sample]
    mres = [r[mre_idx] for r in labeled_rows]
    accs = [r[acc_idx] for r in labeled_rows]
    preds = [r[pred_idx] for r in labeled_rows]
    n_ests = [r[n_est_idx] for r in labeled_rows]
    
    min_mre, max_mre = min(mres), max(mres)
    min_acc, max_acc = min(accs), max(accs)
    min_pred, max_pred = min(preds), max(preds)
    min_n, max_n = min(n_ests), max(n_ests)
    
    def norm(val, mn, mx, rev=False):
        if mx == mn: return 0.5
        n = (val - mn) / (mx - mn)
        return 1.0 - n if rev else n
    
    best_idx = -1
    best_score = -float('inf')
    
    for i in range(n_to_sample):
        # Multi-objective performance
        s1 = norm(rows[i][mre_idx], min_mre, max_mre, True)
        s2 = norm(rows[i][acc_idx], min_acc, max_acc)
        s3 = norm(rows[i][pred_idx], min_pred, max_pred)
        perf = (s1 + s2 + s3) / 3.0
        
        # Penalty for size (N_estimators)
        size_penalty = norm(rows[i][n_est_idx], min_n, max_n, False)
        
        # Combined score: maximize performance, minimize size
        total_score = perf - (0.3 * size_penalty)
        
        if total_score > best_score:
            best_score = total_score
            best_idx = i
            
    return best_idx
