import csv
import random

def race(header, rows, label):
    """
    Compares three strategies: random, greedy (first-improvement), and 
    top-k (sampling then picking best). Total budget: 50 labels.
    """
    budget = 50
    labeled = {}

    def get_score(row):
        # Normalize goals: MRE- (min), ACC+ (max), PRED40+ (max)
        # We use the full dataset range to normalize
        mre_idx, acc_idx, pred_idx = 0, 1, 2
        mres = [r[mre_idx] for r in rows if r[mre_idx] != "?"]
        accs = [r[acc_idx] for r in rows if r[acc_idx] != "?"]
        preds = [r[pred_idx] for r in rows if r[pred_idx] != "?"]
        
        def norm(val, vals, rev):
            mn, mx = min(vals), max(vals)
            if mx == mn: return 0.5
            n = (val - mn) / (mx - mn)
            return 1.0 - n if rev else n
        
        return (norm(row[mre_idx], mres, True) + 
                norm(row[acc_idx], accs, False) + 
                norm(row[pred_idx], preds, False)) / 3.0

    def get_label(i):
        if i not in labeled:
            labeled[i] = label(i)
            rows[i][0], rows[i][1], rows[i][2] = labeled[i]
        return labeled[i]

    # Strategy 1: Random
    def strategy_random():
        indices = random.sample(range(len(rows)), 16)
        for i in indices: get_label(i)
        return max(indices, key=lambda i: get_score(rows[i]))

    # Strategy 2: Greedy (First improvement)
    def strategy_greedy():
        best_idx = 0
        get_label(0)
        for i in range(1, 17):
            get_label(i)
            if get_score(rows[i]) > get_score(rows[best_idx]):
                best_idx = i
        return best_idx

    # Strategy 3: Top-k (Sample 18, pick best)
    def strategy_topk():
        indices = list(range(17, 35))
        for i in indices: get_label(i)
        return max(indices, key=lambda i: get_score(rows[i]))

    results = {
        "random": strategy_random(),
        "greedy": strategy_greedy(),
        "topk": strategy_topk()
    }

    # Determine winner
    winner = max(results, key=lambda s: get_score(rows[results[s]]))
    return {"winner": winner, "row": results[winner], "tried": ["random", "greedy", "topk"]}
