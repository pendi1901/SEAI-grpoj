import csv

def remember(header, rows):
    """
    Identifies the top 5% of rows based on multi-objective optimization.
    Goals: MRE- (minimize), ACC+ (maximize), PRED40+ (maximize).
    We normalize the goals to [0, 1] and calculate a composite score.
    """
    n = len(rows)
    if n == 0:
        return []

    # Identify indices for goals
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    # Extract values to find min/max for normalization
    mres = [r[mre_idx] for r in rows]
    accs = [r[acc_idx] for r in rows]
    preds = [r[pred_idx] for r in rows]

    min_mre, max_mre = min(mres), max(mres)
    min_acc, max_acc = min(accs), max(accs)
    min_pred, max_pred = min(preds), max(preds)

    def normalize(val, min_v, max_v, reverse=False):
        if max_v == min_v: return 0.5
        norm = (val - min_v) / (max_v - min_v)
        return 1 - norm if reverse else norm

    scores = []
    for i, row in enumerate(rows):
        # Score = average of normalized goals
        s1 = normalize(row[mre_idx], min_mre, max_mre, reverse=True)
        s2 = normalize(row[acc_idx], min_acc, max_acc)
        s3 = normalize(row[pred_idx], min_pred, max_pred)
        scores.append(( (s1 + s2 + s3) / 3, i))

    # Sort by score descending
    scores.sort(key=lambda x: x[0], reverse=True)
    
    # Return top 5% (or at least 1)
    limit = max(1, int(n * 0.05))
    return [item[1] for item in scores[:limit]]

def guess(header, train, test):
    """
    Estimates the first goal (MRE-) for test rows using a K-Nearest Neighbors 
    approach based on the input features (non-goal columns).
    """
    # Identify indices
    goal_idx = 0  # MRE-
    input_indices = [i for i, h in enumerate(header) if not (h.endswith('-') or h.endswith('+'))]
    
    # Pre-calculate numeric ranges for normalization to handle distance
    input_data = []
    for row in train:
        input_data.append([row[i] for i in input_indices])
    
    # Simple distance function: Euclidean for numeric, equality for symbolic
    def dist(row1, row2):
        d = 0
        for i, idx in enumerate(input_indices):
            val1, val2 = row1[idx], row2[idx]
            if isinstance(val1, float):
                # Normalize by range if possible, else simple diff
                d += (val1 - val2) ** 2
            else:
                d += 0 if val1 == val2 else 1
        return d ** 0.5

    predictions = []
    for t_row in test:
        # Find nearest neighbor in training set
        best_d = float('inf')
        best_val = 0
        for train_row in train:
            d = dist(t_row, train_row)
            if d < best_d:
                best_d = d
                best_val = train_row[goal_idx]
        predictions.append(best_val)
    
    return predictions

import csv
import math

def certify(xs, ys):
    """
    Returns True if the difference between two distributions is likely noise,
    False if they are statistically distinct. Uses a simple A12 effect size test.
    """
    def a12(lst1, lst2):
        m, n = len(lst1), len(lst2)
        count = 0
        for x in lst1:
            for y in lst2:
                if x > y: count += 1
                elif x == y: count += 0.5
        return count / (m * n)

    # A12 threshold: 0.56 is a small effect size, 0.64 medium, 0.71 large.
    # If A12 is close to 0.5, they are likely the same (noise).
    score = a12(xs, ys)
    return 0.44 <= score <= 0.56

def analyze_size_vs_accuracy(header, rows):
    """
    Compares the accuracy (ACC+) of 'big' setups vs 'small' setups.
    'Big' is defined as N_estimators > median.
    """
    acc_idx = header.index('ACC+')
    n_idx = header.index('N_estimators')
    
    vals = [r[n_idx] for r in rows]
    median = sorted(vals)[len(vals) // 2]
    
    big = [r[acc_idx] for r in rows if r[n_idx] > median]
    small = [r[acc_idx] for r in rows if r[n_idx] <= median]
    
    is_noise = certify(big, small)
    
    print(f"Analysis: Comparing ACC+ for {len(big)} big setups vs {len(small)} small setups.")
    if is_noise:
        print("Result: The difference is likely just noise.")
    else:
        print("Result: The difference is statistically significant.")

import csv

def flow(header, row, seen):
    """
    Decides whether to accept the current row based on whether it is 
    better than all previously seen rows across all goals.
    """
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    # If this is the first row, we have no baseline, so keep looking
    if not seen:
        return False

    # A row is "good enough" if it dominates or is better than the best 
    # seen so far. We define "better" as having a better composite score 
    # than any previous row.
    def get_score(r):
        # We don't have global min/max here, so we compare relative to 
        # the current row's performance against the seen set.
        # Simple heuristic: is this row better than the best seen?
        return (-r[mre_idx], r[acc_idx], r[pred_idx])

    current_score = get_score(row)
    best_seen_score = max([get_score(r) for r in seen])

    # If current row is better than the best seen, we stop.
    return current_score > best_seen_score

if __name__ == "__main__":
    filename = 'Health-ClosedIssues0000.n1000.train.csv'
    try:
        with open(filename, 'r') as f:
            reader = csv.reader(f)

            rows = []
            for row in reader:
                processed = [float(val) if header[i] not in ['criterion'] else val 
                             for i, val in enumerate(row)]

        seen = []
        found = False
        for i, row in enumerate(rows):

import csv
import step01

def blame(header, rows):
    """
    Identifies input columns that correlate with poor performance.
    We define 'bad' as the bottom 25% of rows based on the composite score.
    We then compare the distribution of input values in 'bad' rows 
    vs 'good' rows to find features with the largest divergence.
    """
    # Use step01 logic to score rows
    n = len(rows)
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')
    
    mres = [r[mre_idx] for r in rows]
    accs = [r[acc_idx] for r in rows]
    preds = [r[pred_idx] for r in rows]
    
    min_mre, max_mre = min(mres), max(mres)
    min_acc, max_acc = min(accs), max(accs)
    min_pred, max_pred = min(preds), max(preds)

    def normalize(val, min_v, max_v, reverse=False):
        if max_v == min_v: return 0.5
        norm = (val - min_v) / (max_v - min_v)
        return 1 - norm if reverse else norm

    scored = []
    for row in rows:
        s = (normalize(row[mre_idx], min_mre, max_mre, True) +
             normalize(row[acc_idx], min_acc, max_acc) +
             normalize(row[pred_idx], min_pred, max_pred)) / 3
        scored.append((s, row))
    
    scored.sort(key=lambda x: x[0])
    
    # Bottom 25% are 'bad', top 25% are 'good'
    q = n // 4
    bad = [x[1] for x in scored[:q]]
    good = [x[1] for x in scored[-q:]]
    
    input_cols = [i for i, h in enumerate(header) if not (h.endswith('-') or h.endswith('+'))]
    
    # Simple heuristic: find input columns where the mean/mode differs most
    diffs = []
    for i in input_cols:
        if isinstance(bad[0][i], float):
            bad_val = sum(r[i] for r in bad) / len(bad)
            good_val = sum(r[i] for r in good) / len(good)
            diffs.append((abs(bad_val - good_val), header[i]))
        else:
            # For symbolic, compare frequency of the most common value
            bad_mode = max(set(r[i] for r in bad), key=[r[i] for r in bad].count)
            good_mode = max(set(r[i] for r in good), key=[r[i] for r in good].count)
            diffs.append((1 if bad_mode != good_mode else 0, header[i]))
            
    diffs.sort(key=lambda x: x[0], reverse=True)
    return [d[1] for d in diffs[:2]]

def justify(header, rows, row):
    """
    Estimates the performance of 'row' using the nearest neighbor from 'rows',
    then calculates the percentile rank of this estimate against all 'rows'.
    """
    # 1. Estimate the goals for the '?' row
    # We use the guess logic from step02 to predict the goals
    # Since step02 only predicts MRE-, we extend the logic to all goals
    goal_indices = [i for i, h in enumerate(header) if h.endswith('-') or h.endswith('+')]
    input_indices = [i for i, h in enumerate(header) if not (h.endswith('-') or h.endswith('+'))]
    
    def dist(r1, r2):
        d = 0
        for i in input_indices:
            v1, v2 = r1[i], r2[i]
            if isinstance(v1, float):
                d += (v1 - v2) ** 2
            else:
                d += 0 if v1 == v2 else 1
        return d ** 0.5

    # Find nearest neighbor to estimate goals
    best_d = float('inf')
    neighbor = None
    for r in rows:
        d = dist(row, r)
        if d < best_d:
            best_d = d
            neighbor = r
    
    # Estimated goals for the input row
    est_goals = [neighbor[i] for i in goal_indices]
    
    # 2. Calculate how many rows are worse than this estimate
    # A row is better if it has a higher composite score
    # We use the scoring logic from step01
    def get_score(r_goals):
        scores = []
        for i, g_idx in enumerate(goal_indices):
            col_vals = [row[g_idx] for row in rows]
            min_v, max_v = min(col_vals), max(col_vals)
            val = r_goals[i]
            norm = (val - min_v) / (max_v - min_v) if max_v != min_v else 0.5
            # If ends in -, minimize (1-norm), if +, maximize (norm)
            scores.append(1 - norm if header[g_idx].endswith('-') else norm)
        return sum(scores) / len(scores)

    est_score = get_score(est_goals)
    all_scores = [get_score([r[i] for i in goal_indices]) for r in rows]
    
    better_than_count = sum(1 for s in all_scores if est_score > s)
    p = better_than_count / len(rows)
    
    reason = f"This setup is estimated to outperform {p:.1%} of all configurations based on its similarity to high-performing historical data."
    return (p, reason)
