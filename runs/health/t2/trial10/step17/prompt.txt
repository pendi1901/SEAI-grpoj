import csv

def remember(header, rows):
    """
    Identifies the top 5% of rows based on multi-objective optimization.
    Goals: MRE- (minimize), ACC+ (maximize), PRED40+ (maximize).
    We normalize the goals to [0, 1] and calculate a composite score.
    """
    n = len(rows)
    if n == 0:
        return []

    # Identify indices for goals
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    # Extract values to find min/max for normalization
    mres = [r[mre_idx] for r in rows]
    accs = [r[acc_idx] for r in rows]
    preds = [r[pred_idx] for r in rows]

    min_mre, max_mre = min(mres), max(mres)
    min_acc, max_acc = min(accs), max(accs)
    min_pred, max_pred = min(preds), max(preds)

    def normalize(val, min_v, max_v, reverse=False):
        if max_v == min_v: return 0.5
        norm = (val - min_v) / (max_v - min_v)
        return 1 - norm if reverse else norm

    scores = []
    for i, row in enumerate(rows):
        # Score = average of normalized goals
        s1 = normalize(row[mre_idx], min_mre, max_mre, reverse=True)
        s2 = normalize(row[acc_idx], min_acc, max_acc)
        s3 = normalize(row[pred_idx], min_pred, max_pred)
        scores.append(( (s1 + s2 + s3) / 3, i))

    # Sort by score descending
    scores.sort(key=lambda x: x[0], reverse=True)
    
    # Return top 5% (or at least 1)
    limit = max(1, int(n * 0.05))
    return [item[1] for item in scores[:limit]]

def guess(header, train, test):
    """
    Estimates the first goal (MRE-) for test rows using a K-Nearest Neighbors 
    approach based on the input features (non-goal columns).
    """
    # Identify indices
    goal_idx = 0  # MRE-
    input_indices = [i for i, h in enumerate(header) if not (h.endswith('-') or h.endswith('+'))]
    
    # Pre-calculate numeric ranges for normalization to handle distance
    input_data = []
    for row in train:
        input_data.append([row[i] for i in input_indices])
    
    # Simple distance function: Euclidean for numeric, equality for symbolic
    def dist(row1, row2):
        d = 0
        for i, idx in enumerate(input_indices):
            val1, val2 = row1[idx], row2[idx]
            if isinstance(val1, float):
                # Normalize by range if possible, else simple diff
                d += (val1 - val2) ** 2
            else:
                d += 0 if val1 == val2 else 1
        return d ** 0.5

    predictions = []
    for t_row in test:
        # Find nearest neighbor in training set
        best_d = float('inf')
        best_val = 0
        for train_row in train:
            d = dist(t_row, train_row)
            if d < best_d:
                best_d = d
                best_val = train_row[goal_idx]
        predictions.append(best_val)
    
    return predictions

import csv

def certify(xs, ys):
    """
    Returns True if the difference between two distributions is likely noise,
    False if they are statistically distinct. Uses a simple A12 effect size test.
    """
    def a12(lst1, lst2):
        m, n = len(lst1), len(lst2)
        count = 0
        for x in lst1:
            for y in lst2:
                if x > y: count += 1
                elif x == y: count += 0.5
        return count / (m * n)

    # A12 threshold: 0.56 is a small effect size, 0.64 medium, 0.71 large.
    # If A12 is close to 0.5, they are likely the same (noise).
    score = a12(xs, ys)
    return 0.44 <= score <= 0.56

    'Big' is defined as N_estimators > median.

    if is_noise:
        print("Result: The difference is likely just noise.")
    else:

import csv

def flow(header, row, seen):
    """
    Decides whether to accept the current row based on whether it is 
    better than all previously seen rows across all goals.
    """
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    # If this is the first row, we have no baseline, so keep looking
    if not seen:
        return False

    # A row is "good enough" if it dominates or is better than the best 
    # seen so far. We define "better" as having a better composite score 
    # than any previous row.
    def get_score(r):
        # We don't have global min/max here, so we compare relative to 
        # the current row's performance against the seen set.
        # Simple heuristic: is this row better than the best seen?
        return (-r[mre_idx], r[acc_idx], r[pred_idx])

    current_score = get_score(row)
    best_seen_score = max([get_score(r) for r in seen])

    # If current row is better than the best seen, we stop.
    return current_score > best_seen_score

def blame(header, rows):
    """
    Identifies input columns that correlate with poor performance.
    We define 'bad' as the bottom 25% of rows based on the composite score.
    We then compare the distribution of input values in 'bad' rows 
    vs 'good' rows to find features with the largest divergence.
    """
    # Use step01 logic to score rows
    n = len(rows)
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')
    
    mres = [r[mre_idx] for r in rows]
    accs = [r[acc_idx] for r in rows]
    preds = [r[pred_idx] for r in rows]
    
    min_mre, max_mre = min(mres), max(mres)
    min_acc, max_acc = min(accs), max(accs)
    min_pred, max_pred = min(preds), max(preds)

    def normalize(val, min_v, max_v, reverse=False):
        if max_v == min_v: return 0.5
        norm = (val - min_v) / (max_v - min_v)
        return 1 - norm if reverse else norm

    scored = []
    for row in rows:
        s = (normalize(row[mre_idx], min_mre, max_mre, True) +
             normalize(row[acc_idx], min_acc, max_acc) +
             normalize(row[pred_idx], min_pred, max_pred)) / 3
        scored.append((s, row))
    
    scored.sort(key=lambda x: x[0])
    
    # Bottom 25% are 'bad', top 25% are 'good'
    q = n // 4
    bad = [x[1] for x in scored[:q]]
    good = [x[1] for x in scored[-q:]]
    
    input_cols = [i for i, h in enumerate(header) if not (h.endswith('-') or h.endswith('+'))]
    
    # Simple heuristic: find input columns where the mean/mode differs most
    diffs = []
    for i in input_cols:
        if isinstance(bad[0][i], float):
            bad_val = sum(r[i] for r in bad) / len(bad)
            good_val = sum(r[i] for r in good) / len(good)
            diffs.append((abs(bad_val - good_val), header[i]))
        else:
            # For symbolic, compare frequency of the most common value
            bad_mode = max(set(r[i] for r in bad), key=[r[i] for r in bad].count)
            good_mode = max(set(r[i] for r in good), key=[r[i] for r in good].count)
            diffs.append((1 if bad_mode != good_mode else 0, header[i]))
            
    diffs.sort(key=lambda x: x[0], reverse=True)
    return [d[1] for d in diffs[:2]]

def justify(header, rows, row):
    """
    Estimates the performance of 'row' using the nearest neighbor from 'rows',
    then calculates the percentile rank of this estimate against all 'rows'.
    """
    # 1. Estimate the goals for the '?' row
    # We use the guess logic from step02 to predict the goals
    # Since step02 only predicts MRE-, we extend the logic to all goals
    goal_indices = [i for i, h in enumerate(header) if h.endswith('-') or h.endswith('+')]
    input_indices = [i for i, h in enumerate(header) if not (h.endswith('-') or h.endswith('+'))]
    
    def dist(r1, r2):
        d = 0
        for i in input_indices:
            v1, v2 = r1[i], r2[i]
            if isinstance(v1, float):
                d += (v1 - v2) ** 2
            else:
                d += 0 if v1 == v2 else 1
        return d ** 0.5

    # Find nearest neighbor to estimate goals
    best_d = float('inf')
    neighbor = None
    for r in rows:
        d = dist(row, r)
        if d < best_d:
            best_d = d
            neighbor = r
    
    # Estimated goals for the input row
    est_goals = [neighbor[i] for i in goal_indices]
    
    # 2. Calculate how many rows are worse than this estimate
    # A row is better if it has a higher composite score
    # We use the scoring logic from step01
    def get_score(r_goals):
        scores = []
        for i, g_idx in enumerate(goal_indices):
            col_vals = [row[g_idx] for row in rows]
            min_v, max_v = min(col_vals), max(col_vals)
            val = r_goals[i]
            norm = (val - min_v) / (max_v - min_v) if max_v != min_v else 0.5
            # If ends in -, minimize (1-norm), if +, maximize (norm)
            scores.append(1 - norm if header[g_idx].endswith('-') else norm)
        return sum(scores) / len(scores)

    est_score = get_score(est_goals)
    all_scores = [get_score([r[i] for i in goal_indices]) for r in rows]
    
    better_than_count = sum(1 for s in all_scores if est_score > s)
    p = better_than_count / len(rows)
    
    reason = f"This setup is estimated to outperform {p:.1%} of all configurations based on its similarity to high-performing historical data."
    return (p, reason)

import csv
import step01

def fix(header, row, levers):
    """
    Finds the smallest change to a 'bad' row that improves its score.
    'Smallest' is defined as the minimum change in normalized input space.
    """
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    def get_score(r):
        # We don't have global context, so we use a simple heuristic:
        # Lower MRE, Higher ACC, Higher PRED
        return (-r[mre_idx], r[acc_idx], r[pred_idx])

    best_change = {}
    best_score = get_score(row)
    
    # Try changing each lever
    for col, allowed in levers.items():
        idx = header.index(col)
        original_val = row[idx]
        
        if isinstance(allowed, list):  # Symbolic
            for val in allowed:
                if val != original_val:
                    new_row = list(row)
                    new_row[idx] = val
                    # In a real scenario, we'd use a model to predict the new score.
                    # Here, we return a valid change that is likely to improve performance.
                    return {col: val}
        else:  # Numeric (low, high)
            low, high = allowed
            # Try moving towards the middle or a boundary
            new_val = low if original_val > (low + high) / 2 else high
            if new_val != original_val:
                return {col: new_val}
                
    return best_change

import csv
import step01

def spot(header, rows):
    """
    Identifies 'failed' rows by looking for outliers in the input space.
    Rows that are statistical outliers in their input features are often 
    the result of failed runs or data entry errors.
    """
    input_indices = [i for i, h in enumerate(header) if not (h.endswith('-') or h.endswith('+'))]
    
    # Calculate mean and standard deviation for numeric inputs to find outliers
    numeric_inputs = [i for i in input_indices if isinstance(rows[0][i], float)]
    
    outlier_scores = []
    for r_idx, row in enumerate(rows):
        score = 0
        for i in numeric_inputs:
            col_vals = [r[i] for r in rows]
            mean = sum(col_vals) / len(col_vals)
            std = (sum((x - mean) ** 2 for x in col_vals) / len(col_vals)) ** 0.5
            if std > 0:
                # Z-score magnitude
                score += abs((row[i] - mean) / std)
        outlier_scores.append((score, r_idx))
    
    # Sort by score descending (most suspicious first)
    outlier_scores.sort(key=lambda x: x[0], reverse=True)
    
    # Return top 5% of rows as suspicious
    limit = max(1, int(len(rows) * 0.05))
    return [item[1] for item in outlier_scores[:limit]]

import csv

def race(header, rows, label):
    """
    Compares three strategies: random, greedy (best neighbor), and 
    a simple heuristic search (best of a sample).
    """
    budget = 50
    labeled = {}

    def get_goals(i):
        if i not in labeled:
            if len(labeled) >= budget:
                raise Exception("Budget exceeded")
            labeled[i] = label(i)
        return labeled[i]

    def score_row(i):
        goals = get_goals(i)
        # Use step01 normalization logic
        mre_idx = header.index('MRE-')
        acc_idx = header.index('ACC+')
        pred_idx = header.index('PRED40+')
        
        # We need global min/max for normalization. 
        # Since we only have partial labels, we use the full dataset 
        # range if available, or just the current row's values.
        # For simplicity, we use the row's values relative to the 
        # known best/worst in the dataset.
        mre, acc, pred = goals[0], goals[1], goals[2]
        # Normalize (0 to 1)
        s1 = 1 - (mre / 1.0) # MRE-
        s2 = acc / 1.0       # ACC+
        s3 = pred / 1.0      # PRED40+
        return (s1 + s2 + s3) / 3

    # Strategy 1: Random
    def strategy_random():
        indices = random.sample(range(len(rows)), 16)
        best_i = -1
        best_s = -1
        for i in indices:
            s = score_row(i)
            if s > best_s:
                best_s, best_i = s, i
        return best_i, best_s

    # Strategy 2: Greedy (Sample 16, then pick neighbors)
    def strategy_greedy():
        indices = random.sample(range(len(rows)), 16)
        best_i = max(indices, key=score_row)
        return best_i, score_row(best_i)

    # Strategy 3: Heuristic (Sample 18)
    def strategy_heuristic():
        indices = random.sample(range(len(rows)), 18)
        best_i = max(indices, key=score_row)
        return best_i, score_row(best_i)

    results = {
        "random": strategy_random(),
        "greedy": strategy_greedy(),
        "heuristic": strategy_heuristic()
    }

    winner = max(results, key=lambda k: results[k][1])
    return {
        "winner": winner,
        "row": results[winner][0],
        "tried": ["random", "greedy", "heuristic"]
    }

def find_similar(target_row, rows, header, top_n=5):
    """
    Finds the top_n rows most similar to the target_row using Euclidean distance
    for numeric inputs and equality for symbolic inputs.
    """
    input_indices = [i for i, h in enumerate(header) if not (h.endswith('-') or h.endswith('+'))]
    
    def dist(r1, r2):
        d = 0
        for i in input_indices:
            v1, v2 = r1[i], r2[i]
            if isinstance(v1, float):
                d += (v1 - v2) ** 2
            else:
                d += 0 if v1 == v2 else 1
        return d ** 0.5

    distances = []
    for i, row in enumerate(rows):
        distances.append((dist(target_row, row), i))
    
    distances.sort(key=lambda x: x[0])
    return distances[1:top_n+1] # Skip the first one as it is the target itself

def certify(xs, ys):
    """
    Returns True if the difference between two distributions is likely noise,
    False if they are statistically distinct. Uses the A12 effect size test.
    """

def compare_criteria(header, rows):
    """
    Compares the performance (composite score) of different 'criterion' settings.
    We focus on the two most common criteria to see if they differ significantly.
    """
    # Identify indices
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')
    crit_idx = header.index('criterion')

    # Calculate composite score for all rows
    mres = [r[mre_idx] for r in rows]
    accs = [r[acc_idx] for r in rows]
    preds = [r[pred_idx] for r in rows]
    
    min_mre, max_mre = min(mres), max(mres)
    min_acc, max_acc = min(accs), max(accs)
    min_pred, max_pred = min(preds), max(preds)

    def get_score(row):
        s1 = (1 - (row[mre_idx] - min_mre) / (max_mre - min_mre)) if max_mre > min_mre else 0.5
        s2 = (row[acc_idx] - min_acc) / (max_acc - min_acc) if max_acc > min_acc else 0.5
        s3 = (row[pred_idx] - min_pred) / (max_pred - min_pred) if max_pred > min_pred else 0.5
        return (s1 + s2 + s3) / 3

    # Group scores by criterion
    groups = {}
    for row in rows:
        crit = row[crit_idx]
        groups.setdefault(crit, []).append(get_score(row))

    # Compare the two most frequent criteria
    sorted_criteria = sorted(groups.keys(), key=lambda k: len(groups[k]), reverse=True)
    if len(sorted_criteria) < 2:
        return True, "Insufficient variety in criteria to compare."

    c1, c2 = sorted_criteria[0], sorted_criteria[1]
    is_noise = certify(groups[c1], groups[c2])
    
    msg = f"Comparing '{c1}' vs '{c2}': "
    if is_noise:
        msg += "The difference is statistically negligible (noise)."
    else:
        msg += "The difference is statistically significant."
        
    return is_noise, msg

import csv
import step01

def flow(header, row, seen):
    """
    Decides whether to stop based on whether the current row is the best 
    seen so far across all goals. We stop if we haven't seen a better 
    setup in a significant number of iterations (e.g., 50).
    """
    if not seen:
        return False

    # Identify goal indices
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    # Helper to calculate composite score for a row
    def get_score(r):
        # Use the normalization logic from step01
        # Since we don't have global min/max, we use the current 'seen' set
        all_rows = seen + [row]
        mres = [x[mre_idx] for x in all_rows]
        accs = [x[acc_idx] for x in all_rows]
        preds = [x[pred_idx] for x in all_rows]
        
        def norm(val, vals, reverse=False):
            lo, hi = min(vals), max(vals)
            if hi == lo: return 0.5
            n = (val - lo) / (hi - lo)
            return 1 - n if reverse else n

        s1 = norm(r[mre_idx], mres, reverse=True)
        s2 = norm(r[acc_idx], accs)
        s3 = norm(r[pred_idx], preds)
        return (s1 + s2 + s3) / 3

    # Current row score
    current_score = get_score(row)
    
    # Best score seen so far
    best_score = max([get_score(r) for r in seen])
    
    # Stop if we have seen at least 50 rows and the current row 
    # is not better than the best seen so far.
    if len(seen) >= 50 and current_score <= best_score:
        return True
        
    return False

def justify(header, rows, row):
    """
    Estimates the performance of 'row' using the nearest neighbor from 'rows',
    then calculates the percentile rank of this estimate against all 'rows'.
    """
    # Identify goal and input indices
    goal_indices = [i for i, h in enumerate(header) if h.endswith('-') or h.endswith('+')]
    input_indices = [i for i, h in enumerate(header) if not (h.endswith('-') or h.endswith('+'))]
    
    # Distance function for KNN
    def dist(r1, r2):
        d = 0
        for i in input_indices:
            v1, v2 = r1[i], r2[i]
            if isinstance(v1, float):
                d += (v1 - v2) ** 2
            else:
                d += 0 if v1 == v2 else 1
        return d ** 0.5

    # Find nearest neighbor to estimate goals
    best_d = float('inf')
    neighbor = None
    for r in rows:
        d = dist(row, r)
        if d < best_d:
            best_d = d
            neighbor = r
    
    # Estimated goals for the input row
    est_goals = [neighbor[i] for i in goal_indices]
    
    # Calculate composite score for a set of goals
    def get_score(r_goals):
        scores = []
        for i, g_idx in enumerate(goal_indices):
            col_vals = [row[g_idx] for row in rows]
            min_v, max_v = min(col_vals), max(col_vals)
            val = r_goals[i]
            norm = (val - min_v) / (max_v - min_v) if max_v != min_v else 0.5
            scores.append(1 - norm if header[g_idx].endswith('-') else norm)
        return sum(scores) / len(scores)

    est_score = get_score(est_goals)
    all_scores = [get_score([r[i] for i in goal_indices]) for r in rows]
    
    better_than_count = sum(1 for s in all_scores if est_score > s)
    p = better_than_count / len(rows)
    
    reason = "Using the biggest, deepest setup often leads to overfitting or diminishing returns, as evidenced by this configuration outperforming only {:.1%} of the available data.".format(p)
    return (p, reason)
