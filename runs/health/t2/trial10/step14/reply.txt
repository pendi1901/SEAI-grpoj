import csv
import step01

def flow(header, row, seen):
    """
    Decides whether to stop based on whether the current row is the best 
    seen so far across all goals. We stop if we haven't seen a better 
    setup in a significant number of iterations (e.g., 50).
    """
    if not seen:
        return False

    # Identify goal indices
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    # Helper to calculate composite score for a row
    def get_score(r):
        # Use the normalization logic from step01
        # Since we don't have global min/max, we use the current 'seen' set
        all_rows = seen + [row]
        mres = [x[mre_idx] for x in all_rows]
        accs = [x[acc_idx] for x in all_rows]
        preds = [x[pred_idx] for x in all_rows]
        
        def norm(val, vals, reverse=False):
            lo, hi = min(vals), max(vals)
            if hi == lo: return 0.5
            n = (val - lo) / (hi - lo)
            return 1 - n if reverse else n

        s1 = norm(r[mre_idx], mres, reverse=True)
        s2 = norm(r[acc_idx], accs)
        s3 = norm(r[pred_idx], preds)
        return (s1 + s2 + s3) / 3

    # Current row score
    current_score = get_score(row)
    
    # Best score seen so far
    best_score = max([get_score(r) for r in seen])
    
    # Stop if we have seen at least 50 rows and the current row 
    # is not better than the best seen so far.
    if len(seen) >= 50 and current_score <= best_score:
        return True
        
    return False

if __name__ == "__main__":
    filename = 'Health-ClosedIssues0000.n1000.train.csv'
    try:
        with open(filename, 'r') as f:
            reader = csv.reader(f)
            header = next(reader)
            rows = []
            for row in reader:
                processed = [float(val) if header[i] not in ['criterion'] else val 
                             for i, val in enumerate(row)]
                rows.append(processed)

        seen = []
        stopped_at = -1
        for i, row in enumerate(rows):
            if flow(header, row, seen):
                stopped_at = i + 1
                break
            seen.append(row)
        
        if stopped_at != -1:
            print(f"Optimization stopped at row {stopped_at} as no further improvement was detected.")
        else:
            print("Optimization completed without early stopping.")
            
    except Exception as e:
        print(f"Error: {e}")
        exit(1)
