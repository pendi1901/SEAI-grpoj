import csv
import step05

def blame(header, rows):
    """
    Identifies input columns that correlate with poor performance.
    We reuse the logic from step05, which identifies the top 2 culprits 
    responsible for the difference between 'good' and 'bad' rows.
    """
    return step05.blame(header, rows)

if __name__ == "__main__":
    filename = 'Health-ClosedIssues0000.n1000.train.csv'
    rows = []
    try:
        with open(filename, 'r') as f:
            reader = csv.reader(f)
            header = next(reader)
            for row in reader:
                # Convert numeric columns to float, keep symbolic as strings
                processed = []
                for i, val in enumerate(row):
                    if header[i] in ['N_estimators', 'Min_sample_leaves', 
                                     'Min_impurity_decrease', 'Max_depth']:
                        processed.append(float(val))
                    else:
                        processed.append(val)
                rows.append(processed)
        
        # Get the culprits (the settings that most differentiate bad performance)
        culprits = blame(header, rows)
        
        # Identify all input columns
        all_inputs = [h for h in header if not (h.endswith('-') or h.endswith('+'))]
        
        # Settings that are NOT culprits are the ones that can likely be left at defaults
        safe_settings = [h for h in all_inputs if h not in culprits]
        
        print("Analysis: Which settings can be left at factory defaults?")
        print("The following settings have the least impact on performance degradation:")
        for s in safe_settings:
            print(f"- {s}")
        print("\nNote: The settings most responsible for poor performance (and thus should be tuned) are:")
        for c in culprits:
            print(f"- {c}")
            
    except Exception as e:
        print(f"Error: {e}")
        exit(1)
