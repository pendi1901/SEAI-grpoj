import csv

def remember(header, rows):
    """
    Identifies the top 5% of rows based on multi-objective optimization.
    Goals: MRE- (minimize), ACC+ (maximize), PRED40+ (maximize).
    We normalize the goals to [0, 1] and calculate a composite score.
    """
    n = len(rows)
    if n == 0:
        return []

    # Identify indices for goals
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    # Extract values to find min/max for normalization
    mres = [r[mre_idx] for r in rows]
    accs = [r[acc_idx] for r in rows]
    preds = [r[pred_idx] for r in rows]

    min_mre, max_mre = min(mres), max(mres)
    min_acc, max_acc = min(accs), max(accs)
    min_pred, max_pred = min(preds), max(preds)

    def normalize(val, min_v, max_v, reverse=False):
        if max_v == min_v: return 0.5
        norm = (val - min_v) / (max_v - min_v)
        return 1 - norm if reverse else norm

    scores = []
    for i, row in enumerate(rows):
        # Score = average of normalized goals
        s1 = normalize(row[mre_idx], min_mre, max_mre, reverse=True)
        s2 = normalize(row[acc_idx], min_acc, max_acc)
        s3 = normalize(row[pred_idx], min_pred, max_pred)
        scores.append(( (s1 + s2 + s3) / 3, i))

    # Sort by score descending
    scores.sort(key=lambda x: x[0], reverse=True)
    
    # Return top 5% (or at least 1)
    limit = max(1, int(n * 0.05))
    return [item[1] for item in scores[:limit]]

import csv
import step01

def guess(header, train, test):
    """
    Estimates the first goal (MRE-) for test rows using a K-Nearest Neighbors 
    approach based on the input features (non-goal columns).
    """
    # Identify indices
    goal_idx = 0  # MRE-
    input_indices = [i for i, h in enumerate(header) if not (h.endswith('-') or h.endswith('+'))]
    
    # Pre-calculate numeric ranges for normalization to handle distance
    input_data = []
    for row in train:
        input_data.append([row[i] for i in input_indices])
    
    # Simple distance function: Euclidean for numeric, equality for symbolic
    def dist(row1, row2):
        d = 0
        for i, idx in enumerate(input_indices):
            val1, val2 = row1[idx], row2[idx]
            if isinstance(val1, float):
                # Normalize by range if possible, else simple diff
                d += (val1 - val2) ** 2
            else:
                d += 0 if val1 == val2 else 1
        return d ** 0.5

    predictions = []
    for t_row in test:
        # Find nearest neighbor in training set
        best_d = float('inf')
        best_val = 0
        for train_row in train:
            d = dist(t_row, train_row)
            if d < best_d:
                best_d = d
                best_val = train_row[goal_idx]
        predictions.append(best_val)
    
    return predictions
