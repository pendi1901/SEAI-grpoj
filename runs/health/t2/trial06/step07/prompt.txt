import csv

def remember(header, rows):
    """
    Identifies the top 5% of rows based on multi-objective optimization.
    Goals: MRE- (minimize), ACC+ (maximize), PRED40+ (maximize).
    We normalize values to [0, 1] and use a simple weighted sum (or Pareto-like approach)
    to rank the rows.
    """
    n = len(rows)
    limit = max(1, n // 20)  # 5% limit
    
    # Identify column indices
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')
    
    # Collect values to normalize
    mres = [r[mre_idx] for r in rows]
    accs = [r[acc_idx] for r in rows]
    preds = [r[pred_idx] for r in rows]
    
    min_mre, max_mre = min(mres), max(mres)
    min_acc, max_acc = min(accs), max(accs)
    min_pred, max_pred = min(preds), max(preds)
    
    def normalize(val, min_v, max_v):
        if max_v == min_v: return 0
        return (val - min_v) / (max_v - min_v)
    
    scores = []
    for i, row in enumerate(rows):
        # Score = (1 - norm_mre) + norm_acc + norm_pred
        s = (1 - normalize(row[mre_idx], min_mre, max_mre)) + \
            normalize(row[acc_idx], min_acc, max_acc) + \
            normalize(row[pred_idx], min_pred, max_pred)
        scores.append((s, i))
    
    # Sort by score descending
    scores.sort(key=lambda x: x[0], reverse=True)
    
    return [item[1] for item in scores[:limit]]

def guess(header, train, test):
    """
    Estimates the first goal column (MRE-) for test rows.
    Uses a simple K-Nearest Neighbors approach based on input columns.
    """
    # Identify input columns (those not ending in - or +)
    input_indices = [
        i for i, h in enumerate(header) 
        if not (h.endswith('-') or h.endswith('+'))
    ]
    
    def distance(row1, row2):
        d = 0
        for i in input_indices:
            v1, v2 = row1[i], row2[i]
            if isinstance(v1, str):
                d += 0 if v1 == v2 else 1
            else:
                d += abs(v1 - v2)
        return d

    predictions = []
    for t_row in test:
        # Find the closest row in training data
        best_dist = float('inf')
        best_val = 0.0
        for train_row in train:
            dist = distance(t_row, train_row)
            if dist < best_dist:
                best_dist = dist
                best_val = train_row[0]  # MRE- is at index 0
        predictions.append(best_val)
    
    return predictions

import csv

def certify(xs, ys):
    """
    Returns True if the distributions of xs and ys are statistically similar 
    (i.e., the difference is likely noise), False if they differ significantly.
    Uses a simple mean comparison relative to the standard deviation.
    """
    if not xs or not ys:
        return True
    
    mean_x = sum(xs) / len(xs)
    mean_y = sum(ys) / len(ys)
    
    # Calculate pooled standard deviation
    combined = xs + ys
    mean_all = sum(combined) / len(combined)
    var = sum((x - mean_all) ** 2 for x in combined) / len(combined)
    std = var ** 0.5
    
    if std == 0:
        return mean_x == mean_y
        
    # Cohen's d effect size: if difference is less than 0.5 std, treat as noise
    return abs(mean_x - mean_y) < (0.5 * std)

import csv

def flow(header, row, seen):
    """
    Decides if the current row is 'good enough' to ship.
    We define 'good enough' as being in the top 5% of the expected distribution
    based on the historical data provided in 'seen'.
    """
    if not seen:
        return False

    # Identify goal indices
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    # Helper to calculate score for a row
    def get_score(r, all_rows):
        mres = [x[mre_idx] for x in all_rows]
        accs = [x[acc_idx] for x in all_rows]
        preds = [x[pred_idx] for x in all_rows]
        
        def norm(val, vals):
            mn, mx = min(vals), max(vals)
            return (val - mn) / (mx - mn) if mx > mn else 0
        
        return (1 - norm(r[mre_idx], mres)) + norm(r[acc_idx], accs) + norm(r[pred_idx], preds)

    # Calculate score of current row relative to history
    current_score = get_score(row, seen + [row])
    
    # Calculate scores of all seen rows
    seen_scores = [get_score(r, seen + [row]) for r in seen]
    
    # Threshold: if current row is better than 95% of what we've seen
    threshold = sorted(seen_scores, reverse=True)[max(0, len(seen_scores) // 20)]
    
    return current_score >= threshold

def blame(header, rows):
    """
    Identifies input columns that correlate most strongly with 'bad' performance.
    'Bad' is defined as the bottom 25% of rows based on the remember() logic.
    We compare the average value of inputs in the worst 25% vs the best 25%.
    """
    # Identify input columns
    input_indices = [i for i, h in enumerate(header) if not (h.endswith('-') or h.endswith('+'))]
    
    # Use remember logic to rank rows (remember returns indices of top 5%)
    # We need a full ranking to find the bottom 25%
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')
    
    def get_score(row):
        # Simple score: (1-norm_mre) + norm_acc + norm_pred
        # Using fixed min/max from the whole dataset for consistency
        mres = [r[mre_idx] for r in rows]
        accs = [r[acc_idx] for r in rows]
        preds = [r[pred_idx] for r in rows]
        
        def norm(v, vals):
            mn, mx = min(vals), max(vals)
            return (v - mn) / (mx - mn) if mx > mn else 0
        
        return (1 - norm(row[mre_idx], mres)) + norm(row[acc_idx], accs) + norm(row[pred_idx], preds)

    scored_rows = sorted([(get_score(r), r) for r in rows], key=lambda x: x[0])
    
    n = len(scored_rows)
    worst = [r for _, r in scored_rows[:n // 4]]
    best = [r for _, r in scored_rows[-n // 4:]]
    
    # Calculate "damage" score: difference in means between worst and best
    damages = []
    for i in input_indices:
        if isinstance(worst[0][i], str):
            # For symbolic, check frequency difference
            w_val = max(set(r[i] for r in worst), key=lambda x: [r[i] for r in worst].count(x))
            b_val = max(set(r[i] for r in best), key=lambda x: [r[i] for r in best].count(x))
            diff = 1.0 if w_val != b_val else 0.0
        else:
            w_avg = sum(r[i] for r in worst) / len(worst)
            b_avg = sum(r[i] for r in best) / len(best)
            diff = abs(w_avg - b_avg)
        damages.append((diff, header[i]))
    
    # Return top 3 columns that contribute most to the difference
    damages.sort(key=lambda x: x[0], reverse=True)
    return [name for _, name in damages[:3]]

def justify(header, rows, row):
    """
    Estimates the performance of the given row and compares it against the 
    historical dataset to justify its quality.
    """
    # 1. Estimate the missing goal values for the target row
    # We use the guess function from step02. 
    # Note: guess returns a list of estimates for the first goal (MRE-).
    # We need to estimate all goals. For simplicity, we use the logic 
    # that the row's performance is determined by its input similarity.
    
    # Create a temporary list containing the target row to use with guess
    test_rows = [row]
    # guess() in step02 only estimates the first goal (MRE-). 
    # We will use the average of the 5 nearest neighbors for all goals.
    input_indices = [i for i, h in enumerate(header) if not (h.endswith('-') or h.endswith('+'))]
    
    def get_dist(r1, r2):
        d = 0
        for i in input_indices:
            if isinstance(r1[i], str):
                d += 0 if r1[i] == r2[i] else 1
            else:
                d += abs(r1[i] - r2[i])
        return d

    # Find 5 nearest neighbors
    neighbors = sorted(rows, key=lambda r: get_dist(row, r))[:5]
    
    # Estimate goals (indices 0, 1, 2)
    estimated_goals = []
    for g_idx in range(3):
        estimated_goals.append(sum(n[g_idx] for n in neighbors) / 5)
    
    # 2. Calculate score for the target row
    def get_score(r):
        # Goals: MRE- (min), ACC+ (max), PRED40+ (max)
        # Normalize using global min/max
        scores = []
        for i in range(3):
            vals = [x[i] for x in rows]
            mn, mx = min(vals), max(vals)
            norm = (r[i] - mn) / (mx - mn) if mx > mn else 0
            scores.append(1 - norm if i == 0 else norm)
        return sum(scores)

    target_row_with_goals = row[:]
    for i in range(3):
        target_row_with_goals[i] = estimated_goals[i]
    
    target_score = get_score(target_row_with_goals)
    
    # 3. Compare against all rows
    all_scores = [get_score(r) for r in rows]
    better_than_count = sum(1 for s in all_scores if target_score > s)
    p = better_than_count / len(rows)
    
    reason = f"This setup outperforms {p:.1%} of historical configurations based on its estimated performance metrics."
