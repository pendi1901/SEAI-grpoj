import csv
from step03 import certify

def get_score(row):
    """
    Calculates a performance score for a row.
    Goals: MRE- (index 0, minimize), ACC+ (index 1, maximize), PRED40+ (index 2, maximize).
    """
    # Note: This is a simplified score based on the provided data structure.
    # Since we need to compare distributions, we use the raw goal values.
    # MRE- is minimized, others maximized.
    return -row[0] + row[1] + row[2]

if __name__ == "__main__":
    filename = 'Health-ClosedIssues0000.n1000.train.csv'
    rows = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = [val if header[i] == 'criterion' else float(val) 
                             for i, val in enumerate(row)]
            rows.append(processed_row)

    # The customer asks about the "two built-in rules". 
    # In the context of this dataset, 'criterion' is the primary categorical choice.
    # We compare 'absolute_error' vs 'squared_error' (the two most common).
    crit_idx = header.index('criterion')
    
    abs_error_scores = [get_score(r) for r in rows if r[crit_idx] == 'absolute_error']
    sq_error_scores = [get_score(r) for r in rows if r[crit_idx] == 'squared_error']
    
    is_noise = certify(abs_error_scores, sq_error_scores)
    
    print("Analysis: Does the choice of 'criterion' make a real difference?")
    if is_noise:
        print("Result: The difference between 'absolute_error' and 'squared_error' is likely noise.")
        print("Advice: You can stop arguing about it; the choice does not significantly impact performance.")
    else:
        print("Result: There is a statistically significant difference between the criteria.")
        print("Advice: The choice of criterion matters; continue your investigation.")
