import csv

def remember(header, rows):
    """
    Identifies the top 5% of rows based on multi-objective optimization.
    Goals: MRE- (minimize), ACC+ (maximize), PRED40+ (maximize).
    We normalize values to [0, 1] and use a simple weighted sum (or Pareto-like approach)
    to rank the rows.
    """
    n = len(rows)
    limit = max(1, n // 20)  # 5% limit
    
    # Identify column indices
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')
    
    # Collect values to normalize
    mres = [r[mre_idx] for r in rows]
    accs = [r[acc_idx] for r in rows]
    preds = [r[pred_idx] for r in rows]
    
    min_mre, max_mre = min(mres), max(mres)
    min_acc, max_acc = min(accs), max(accs)
    min_pred, max_pred = min(preds), max(preds)
    
    def normalize(val, min_v, max_v):
        if max_v == min_v: return 0
        return (val - min_v) / (max_v - min_v)
    
    scores = []
    for i, row in enumerate(rows):
        # Score = (1 - norm_mre) + norm_acc + norm_pred
        s = (1 - normalize(row[mre_idx], min_mre, max_mre)) + \
            normalize(row[acc_idx], min_acc, max_acc) + \
            normalize(row[pred_idx], min_pred, max_pred)
        scores.append((s, i))
    
    # Sort by score descending
    scores.sort(key=lambda x: x[0], reverse=True)
    
    return [item[1] for item in scores[:limit]]

    ]
    
    def distance(row1, row2):
        d = 0
        for i in input_indices:
            v1, v2 = row1[i], row2[i]

def certify(xs, ys):
    """
    Returns True if the distributions of xs and ys are statistically similar 
    (i.e., the difference is likely noise), False if they differ significantly.
    Uses a simple mean comparison relative to the standard deviation.
    """
    if not xs or not ys:
        return True
    
    mean_x = sum(xs) / len(xs)
    mean_y = sum(ys) / len(ys)
    
    # Calculate pooled standard deviation
    combined = xs + ys
    mean_all = sum(combined) / len(combined)
    var = sum((x - mean_all) ** 2 for x in combined) / len(combined)
    std = var ** 0.5
    
    if std == 0:
        return mean_x == mean_y
        
    # Cohen's d effect size: if difference is less than 0.5 std, treat as noise
    return abs(mean_x - mean_y) < (0.5 * std)

import csv

def flow(header, row, seen):
    """
    Decides if the current row is 'good enough' to ship.
    We define 'good enough' as being in the top 5% of the expected distribution
    based on the historical data provided in 'seen'.
    """
    if not seen:
        return False

    # Identify goal indices
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    # Helper to calculate score for a row
    def get_score(r, all_rows):
        mres = [x[mre_idx] for x in all_rows]
        accs = [x[acc_idx] for x in all_rows]
        preds = [x[pred_idx] for x in all_rows]
        
        def norm(val, vals):
            mn, mx = min(vals), max(vals)
            return (val - mn) / (mx - mn) if mx > mn else 0
        
        return (1 - norm(r[mre_idx], mres)) + norm(r[acc_idx], accs) + norm(r[pred_idx], preds)

    # Calculate score of current row relative to history
    current_score = get_score(row, seen + [row])
    
    # Calculate scores of all seen rows
    seen_scores = [get_score(r, seen + [row]) for r in seen]
    
    # Threshold: if current row is better than 95% of what we've seen
    threshold = sorted(seen_scores, reverse=True)[max(0, len(seen_scores) // 20)]
    
    return current_score >= threshold

import csv

def spot(header, rows):
    """
    Identifies suspicious rows by checking for outliers in numeric input columns.
    Rows with values significantly outside the range (e.g., > 3 standard deviations)
    are flagged as potentially erroneous entries.
    """
    input_indices = [i for i, h in enumerate(header) 
                     if not (h.endswith('-') or h.endswith('+')) 
                     and isinstance(rows[0][i], (int, float))]
    
    suspicion_scores = [0.0] * len(rows)
    
    for i in input_indices:
        col_vals = [r[i] for r in rows]
        mean = sum(col_vals) / len(col_vals)
        variance = sum((x - mean) ** 2 for x in col_vals) / len(col_vals)
        std = variance ** 0.5
        
        if std > 0:
            for row_idx, row in enumerate(rows):
                z_score = abs(row[i] - mean) / std
                # If a value is more than 3 standard deviations away, it's suspicious
                if z_score > 3:
                    suspicion_scores[row_idx] += z_score
                    
    # Return indices of rows with suspicion scores > 0, sorted by score descending
    indexed_scores = [(suspicion_scores[i], i) for i in range(len(rows)) if suspicion_scores[i] > 0]
    indexed_scores.sort(key=lambda x: x[0], reverse=True)
    
    return [idx for score, idx in indexed_scores]

import csv

from step01 import remember

def race(header, rows, label):
    """
    Compares three strategies: random, greedy, and smart.
    Budget: 50 labels total shared across all strategies.
    """
    budget = 50
    labeled = {}

    def get_goals(idx):
        if idx not in labeled:
            if len(labeled) >= budget:
                return None
            labeled[idx] = label(idx)
        return labeled[idx]

    def get_score(idx):
        goals = get_goals(idx)
        if goals is None: return -float('inf')
        # Goals: MRE- (0), ACC+ (1), PRED40+ (2)
        # Score = (1 - MRE) + ACC + PRED
        return (1.0 - goals[0]) + goals[1] + goals[2]

    # 1. Random Strategy: Sample 16
    def strategy_random():
        indices = random.sample(range(len(rows)), 16)
        best_idx = indices[0]
        best_s = -float('inf')
        for idx in indices:
            s = get_score(idx)
            if s > best_s:
                best_s, best_idx = s, idx
        return best_idx

    # 2. Greedy Strategy: Start at random, look at 16 neighbors
    def strategy_greedy():
        curr = random.randint(0, len(rows) - 1)
        best_s = get_score(curr)
        best_idx = curr
        for _ in range(16):
            n = random.randint(0, len(rows) - 1)
            s = get_score(n)
            if s > best_s:
                best_s, s = s, best_s
                best_idx = n
        return best_idx

    # 3. Smart Strategy: Sample 18, use remember to pick best
    def strategy_smart():
        indices = random.sample(range(len(rows)), 18)
        data = []

    """

    signaling that we have reached a local optimum and can stop.

    # Identify goal indices

    pred_idx = header.index('PRED40+')

    def get_score(r):
        # Goals: MRE- (min), ACC+ (max), PRED40+ (max)

        return -r[mre_idx] + r[acc_idx] + r[pred_idx]

    # To answer "when it stopped improving", we return True if it's the new best.
    return current_score > best_seen_score

def blame(header, rows):
    """
    Identifies input columns that correlate most strongly with 'bad' performance.
    The customer wants to know which settings can be left at factory defaults 
    without hurting anything; these are the settings that have the least 
    impact on performance (i.e., the lowest 'damage' scores).
    """
    # We reuse the logic from step05 which calculates the 'damage' (impact)
    # of each input column on performance.
    # The columns with the lowest damage are the ones that can be left at defaults.
    
    # Identify input columns
    input_indices = [i for i, h in enumerate(header) if not (h.endswith('-') or h.endswith('+'))]
    
    # Calculate damage for all input columns
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')
    
    def get_score(row):
        mres = [r[mre_idx] for r in rows]
        accs = [r[acc_idx] for r in rows]
        preds = [r[pred_idx] for r in rows]
        def norm(v, vals):
            mn, mx = min(vals), max(vals)
            return (v - mn) / (mx - mn) if mx > mn else 0
        return (1 - norm(row[mre_idx], mres)) + norm(row[acc_idx], accs) + norm(row[pred_idx], preds)

    scored_rows = sorted([(get_score(r), r) for r in rows], key=lambda x: x[0])
    n = len(scored_rows)
    worst = [r for _, r in scored_rows[:n // 4]]
    best = [r for _, r in scored_rows[-n // 4:]]
    
    damages = []
    for i in input_indices:
        if isinstance(worst[0][i], str):
            w_val = max(set(r[i] for r in worst), key=lambda x: [r[i] for r in worst].count(x))
            b_val = max(set(r[i] for r in best), key=lambda x: [r[i] for r in best].count(x))
            diff = 1.0 if w_val != b_val else 0.0
        else:
            w_avg = sum(r[i] for r in worst) / len(worst)
            b_avg = sum(r[i] for r in best) / len(best)
            diff = abs(w_avg - b_avg)
        damages.append((diff, header[i]))
    
    # Sort by damage ascending: lowest damage = safest to leave at default
    damages.sort(key=lambda x: x[0])
    
    return [name for _, name in damages]

import csv

def spot(header, rows):
    """
    Identifies rows that contain input values far outside the range of 
    what has been seen in the training data.
    
    We define 'suspicious' as any input value that is more than 3 standard 
    deviations away from the mean of that column.
    """
    # Identify input columns (those not ending in - or +)
    input_indices = [
        i for i, h in enumerate(header) 
        if not (h.endswith('-') or h.endswith('+')) 
        and isinstance(rows[0][i], (int, float))
    ]
    
    suspicion_scores = [0.0] * len(rows)
    
    for i in input_indices:
        col_vals = [r[i] for r in rows]
        mean = sum(col_vals) / len(col_vals)
        variance = sum((x - mean) ** 2 for x in col_vals) / len(col_vals)
        std = variance ** 0.5
        
        # If std is 0, all values are the same; no outliers possible
        if std > 0:
            for row_idx, row in enumerate(rows):
                # Calculate Z-score
                z_score = abs(row[i] - mean) / std
                # If a value is more than 3 standard deviations away, flag it
                if z_score > 3:
                    suspicion_scores[row_idx] += z_score
                    
    # Collect indices of rows with suspicion scores > 0
    results = [(suspicion_scores[i], i) for i in range(len(rows)) if suspicion_scores[i] > 0]
    
    # Sort by score descending (most suspicious first)
    results.sort(key=lambda x: x[0], reverse=True)
    
    return [idx for score, idx in results]
