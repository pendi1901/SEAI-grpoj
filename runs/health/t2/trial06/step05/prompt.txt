import csv

def remember(header, rows):
    """
    Identifies the top 5% of rows based on multi-objective optimization.
    Goals: MRE- (minimize), ACC+ (maximize), PRED40+ (maximize).
    We normalize values to [0, 1] and use a simple weighted sum (or Pareto-like approach)
    to rank the rows.
    """
    n = len(rows)
    limit = max(1, n // 20)  # 5% limit
    
    # Identify column indices
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')
    
    # Collect values to normalize
    mres = [r[mre_idx] for r in rows]
    accs = [r[acc_idx] for r in rows]
    preds = [r[pred_idx] for r in rows]
    
    min_mre, max_mre = min(mres), max(mres)
    min_acc, max_acc = min(accs), max(accs)
    min_pred, max_pred = min(preds), max(preds)
    
    def normalize(val, min_v, max_v):
        if max_v == min_v: return 0
        return (val - min_v) / (max_v - min_v)
    
    scores = []
    for i, row in enumerate(rows):
        # Score = (1 - norm_mre) + norm_acc + norm_pred
        s = (1 - normalize(row[mre_idx], min_mre, max_mre)) + \
            normalize(row[acc_idx], min_acc, max_acc) + \
            normalize(row[pred_idx], min_pred, max_pred)
        scores.append((s, i))
    
    # Sort by score descending
    scores.sort(key=lambda x: x[0], reverse=True)
    
    return [item[1] for item in scores[:limit]]

    """

    ]
    
    def distance(row1, row2):
        d = 0
        for i in input_indices:
            v1, v2 = row1[i], row2[i]

                d += 0 if v1 == v2 else 1
            else:
                d += abs(v1 - v2)
        return d

        # Find the closest row in training data

import csv

def certify(xs, ys):
    """
    Returns True if the distributions of xs and ys are statistically similar 
    (i.e., the difference is likely noise), False if they differ significantly.
    Uses a simple mean comparison relative to the standard deviation.
    """
    if not xs or not ys:
        return True
    
    mean_x = sum(xs) / len(xs)
    mean_y = sum(ys) / len(ys)
    
    # Calculate pooled standard deviation
    combined = xs + ys
    mean_all = sum(combined) / len(combined)
    var = sum((x - mean_all) ** 2 for x in combined) / len(combined)
    std = var ** 0.5
    
    if std == 0:
        return mean_x == mean_y
        
    # Cohen's d effect size: if difference is less than 0.5 std, treat as noise
    return abs(mean_x - mean_y) < (0.5 * std)

import csv

def flow(header, row, seen):
    """
    Decides if the current row is 'good enough' to ship.
    We define 'good enough' as being in the top 5% of the expected distribution
    based on the historical data provided in 'seen'.
    """
    if not seen:
        return False

    # Identify goal indices
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    # Helper to calculate score for a row
    def get_score(r, all_rows):
        mres = [x[mre_idx] for x in all_rows]
        accs = [x[acc_idx] for x in all_rows]
        preds = [x[pred_idx] for x in all_rows]
        
        def norm(val, vals):
            mn, mx = min(vals), max(vals)
            return (val - mn) / (mx - mn) if mx > mn else 0
        
        return (1 - norm(r[mre_idx], mres)) + norm(r[acc_idx], accs) + norm(r[pred_idx], preds)

    # Calculate score of current row relative to history
    current_score = get_score(row, seen + [row])
    
    # Calculate scores of all seen rows
    seen_scores = [get_score(r, seen + [row]) for r in seen]
    
    # Threshold: if current row is better than 95% of what we've seen
    threshold = sorted(seen_scores, reverse=True)[max(0, len(seen_scores) // 20)]
    
    return current_score >= threshold

if __name__ == "__main__":

            rows.append([val if header[i] == 'criterion' else float(val) for i, val in enumerate(row)])

            print(f"Shipped at row {i+1} (Index {i})")
            shipped = True
            break
        seen.append(row)
    
    if not shipped:
        print("No setup met the criteria for shipping.")
