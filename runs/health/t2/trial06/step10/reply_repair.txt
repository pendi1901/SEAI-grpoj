import csv
import random
from step01 import remember

def race(header, rows, label):
    """
    Compares three strategies: random, greedy, and smart.
    Budget: 50 labels total shared across all strategies.
    """
    budget = 50
    labeled = {}

    def get_goals(idx):
        if idx not in labeled:
            if len(labeled) >= budget:
                return None
            labeled[idx] = label(idx)
        return labeled[idx]

    def get_score(idx):
        goals = get_goals(idx)
        if goals is None: return -float('inf')
        # Goals: MRE- (0), ACC+ (1), PRED40+ (2)
        # Score = (1 - MRE) + ACC + PRED
        return (1.0 - goals[0]) + goals[1] + goals[2]

    # 1. Random Strategy: Sample 16
    def strategy_random():
        indices = random.sample(range(len(rows)), 16)
        best_idx = indices[0]
        best_s = -float('inf')
        for idx in indices:
            s = get_score(idx)
            if s > best_s:
                best_s, best_idx = s, idx
        return best_idx

    # 2. Greedy Strategy: Start at random, look at 16 neighbors
    def strategy_greedy():
        curr = random.randint(0, len(rows) - 1)
        best_s = get_score(curr)
        best_idx = curr
        for _ in range(16):
            n = random.randint(0, len(rows) - 1)
            s = get_score(n)
            if s > best_s:
                best_s, s = s, best_s
                best_idx = n
        return best_idx

    # 3. Smart Strategy: Sample 18, use remember to pick best
    def strategy_smart():
        indices = random.sample(range(len(rows)), 18)
        data = []
        for idx in indices:
            row_with_goals = list(rows[idx])
            goals = get_goals(idx)
            row_with_goals[0:3] = goals
            data.append(row_with_goals)
        
        best_local = remember(header, data)[0]
        return indices[best_local]

    strategies = {"random": strategy_random, "greedy": strategy_greedy, "smart": strategy_smart}
    results = {}
    for name, func in strategies.items():
        idx = func()
        results[name] = {"idx": idx, "score": get_score(idx)}

    winner = max(results, key=lambda n: results[n]["score"])
    return {"winner": winner, "row": results[winner]["idx"], "tried": list(strategies.keys())}

if __name__ == "__main__":
    filename = 'Health-ClosedIssues0000.n1000.train.csv'
    rows = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            rows.append([val if header[i] == 'criterion' else float(val) for i, val in enumerate(row)])

    goal_indices = [i for i, h in enumerate(header) if h.endswith('-') or h.endswith('+')]
    true_goals = [[r[i] for i in goal_indices] for r in rows]
    
    masked_rows = []
    for r in rows:
        masked = list(r)
        for i in goal_indices: masked[i] = "?"
        masked_rows.append(masked)

    result = race(header, masked_rows, lambda i: true_goals[i])
    print(f"Winner: {result['winner']}")
    print(f"Best row found at index: {result['row']}")
    print(f"Strategies tried: {', '.join(result['tried'])}")
