import csv

def remember(header, rows):
    """
    Identifies the top 5% of rows based on multi-objective optimization.
    Goals: MRE- (minimize), ACC+ (maximize), PRED40+ (maximize).
    We normalize values to [0, 1] and use a simple weighted sum (or Pareto-like ranking).
    """
    n = len(rows)
    if n == 0:
        return []

    # Identify goal indices
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    # Collect values to normalize
    mres = [r[mre_idx] for r in rows]
    accs = [r[acc_idx] for r in rows]
    preds = [r[pred_idx] for r in rows]

    min_mre, max_mre = min(mres), max(mres)
    min_acc, max_acc = min(accs), max(accs)
    min_pred, max_pred = min(preds), max(preds)

    def normalize(val, min_v, max_v, reverse=False):
        if max_v == min_v: return 0.0
        norm = (val - min_v) / (max_v - min_v)
        return 1.0 - norm if reverse else norm

    scores = []
    for i, row in enumerate(rows):
        # Score = (1-MRE_norm) + ACC_norm + PRED_norm
        s = (normalize(row[mre_idx], min_mre, max_mre, True) +
             normalize(row[acc_idx], min_acc, max_acc) +
             normalize(row[pred_idx], min_pred, max_pred))
        scores.append((s, i))

    # Sort by score descending
    scores.sort(key=lambda x: x[0], reverse=True)

    # Return top 5% (or at least 1)
    limit = max(1, int(n * 0.05))
    return [item[1] for item in scores[:limit]]

def guess(header, train, test):
    """
    Estimates the first goal (MRE-) for each test row.
    Uses a simple k-Nearest Neighbors approach based on input columns.
    """
    # Identify input columns (those not ending in - or +)
    input_indices = [
        i for i, name in enumerate(header) 
        if not (name.endswith('-') or name.endswith('+'))
    ]
    
    # Identify the target column (first goal)
    target_idx = 0
    
    predictions = []
    for t_row in test:
        # Calculate Euclidean distance to all training rows based on inputs
        distances = []
        for tr_row in train:
            dist = 0.0
            for i in input_indices:
                # Handle symbolic vs numeric
                if isinstance(t_row[i], str):
                    dist += 0.0 if t_row[i] == tr_row[i] else 1.0
                else:
                    dist += (t_row[i] - tr_row[i]) ** 2
            distances.append((dist**0.5, tr_row[target_idx]))
        
        # Simple k-NN: average of 3 nearest neighbors
        distances.sort(key=lambda x: x[0])
        k = 3
        avg = sum(d[1] for d in distances[:k]) / k
        predictions.append(avg)
        
    return predictions

import csv
import math

def certify(xs, ys):
    """
    Returns True if the difference between two distributions is likely noise,
    False if they are statistically distinct. Uses a simple T-test approach.
    """
    n1, n2 = len(xs), len(ys)
    if n1 < 2 or n2 < 2:
        return True
    
    m1, m2 = sum(xs) / n1, sum(ys) / n2
    v1 = sum((x - m1) ** 2 for x in xs) / (n1 - 1)
    v2 = sum((y - m2) ** 2 for y in ys) / (n2 - 1)
    
    # Standard error of the difference
    se = math.sqrt((v1 / n1) + (v2 / n2))
    if se == 0:
        return m1 == m2
    
    t_stat = abs(m1 - m2) / se
    # Threshold for 95% confidence (approx 2.0 for large samples)
    return t_stat < 2.0

import csv

def flow(header, row, seen):
    """
    Decides if the current row is 'good enough' to stop searching.
    We define 'good enough' as being in the top 5% of the potential distribution
    based on the data seen so far.
    """
    if not seen:
        return False

    # Identify goal indices
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    def get_score(r):
        # We use the same normalization logic as step01, but relative to 'seen'
        # To avoid re-normalizing everything, we use a simple heuristic:
        # A row is 'good' if it outperforms the average of the 'seen' rows.
        # MRE- (lower is better), ACC+ (higher is better), PRED40+ (higher is better)
        mre_avg = sum(s[mre_idx] for s in seen) / len(seen)
        acc_avg = sum(s[acc_idx] for s in seen) / len(seen)
        pred_avg = sum(s[pred_idx] for s in seen) / len(seen)
        
        return (row[mre_idx] < mre_avg) and (row[acc_idx] > acc_avg) and (row[pred_idx] > pred_avg)

    # Stop if we have seen enough data and this row is significantly better
    # than the average of what we have seen so far.
    if len(seen) > 50 and get_score(row):
        return True
    return False

def blame(header, rows):
    """
    Identifies input columns that correlate with poor performance.
    We define 'bad' rows as those in the bottom 25% of the overall score.
    We then calculate the difference in mean values of inputs between 
    the 'bad' rows and the 'good' rows.
    """
    # Identify goal indices and input indices
    goal_indices = [i for i, name in enumerate(header) if name.endswith('-') or name.endswith('+')]
    input_indices = [i for i, name in enumerate(header) if not (name.endswith('-') or name.endswith('+'))]

    # Use step01 logic to score rows
    # We need to invert the logic to find the worst rows
    n = len(rows)
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    def get_score(row):
        # Simple score: lower MRE, higher ACC, higher PRED
        # We use raw values here for relative ranking
        return (-row[mre_idx] + row[acc_idx] + row[pred_idx])

    scored = sorted([(get_score(r), r) for r in rows], key=lambda x: x[0])
    
    # Bottom 25% vs Top 25%
    quarter = n // 4
    bad_rows = [x[1] for x in scored[:quarter]]
    good_rows = [x[1] for x in scored[-quarter:]]

    # Calculate "damage" score for each input
    # Damage = |mean(bad) - mean(good)| / std_dev(all)
    damages = []
    for i in input_indices:
        if isinstance(rows[0][i], str):
            # Symbolic: compare frequency of most common value
            bad_vals = [r[i] for r in bad_rows]
            good_vals = [r[i] for r in good_rows]
            # Simple heuristic: difference in probability of the most common bad value
            most_common = max(set(bad_vals), key=bad_vals.count)
            diff = abs(bad_vals.count(most_common)/len(bad_vals) - good_vals.count(most_common)/len(good_vals))
            damages.append((diff, header[i]))
        else:
            bad_vals = [r[i] for r in bad_rows]
            good_vals = [r[i] for r in good_rows]
            m_bad = sum(bad_vals) / len(bad_vals)
            m_good = sum(good_vals) / len(good_vals)

            all_vals = [r[i] for r in rows]
            diff = abs(m_bad - m_good) / (max(all_vals) - min(all_vals) + 1e-9)
            damages.append((diff, header[i]))

    # Return top 3 inputs that contribute most to the difference
    damages.sort(key=lambda x: x[0], reverse=True)
    return [d[1] for d in damages[:3]]

import csv

def justify(header, rows, row):
    """
    Estimates the performance of 'row' and compares it against all 'rows'.
    Returns (fraction_better, reason_string).
    """
    # 1. Estimate the goals for the target row using k-NN from step02
    # step02.guess returns a list of MRE- predictions for the input rows.
    # We need to estimate all three goals: MRE-, ACC+, PRED40+
    # Since step02 only estimates MRE-, we extend the logic to estimate all 3 goals.
    
    def estimate_all(target_row):
        input_indices = [i for i, name in enumerate(header) if not (name.endswith('-') or name.endswith('+'))]
        
        estimates = []
        for goal_idx in range(3): # MRE-, ACC+, PRED40+
            distances = []
            for tr_row in rows:
                dist = 0.0
                for i in input_indices:
                    if isinstance(target_row[i], str):
                        dist += 0.0 if target_row[i] == tr_row[i] else 1.0
                    else:
                        dist += (target_row[i] - tr_row[i]) ** 2
                distances.append((dist**0.5, tr_row[goal_idx]))
            
            distances.sort(key=lambda x: x[0])
            k = 3
            avg = sum(d[1] for d in distances[:k]) / k
            estimates.append(avg)
        return estimates

    est_mre, est_acc, est_pred = estimate_all(row)
    
    # 2. Scoring logic: lower MRE, higher ACC, higher PRED
    # Normalize goals to [0, 1] to combine them
    mres = [r[0] for r in rows]
    accs = [r[1] for r in rows]
    preds = [r[2] for r in rows]
    
    def norm(val, vals, reverse=False):
        min_v, max_v = min(vals), max(vals)
        if max_v == min_v: return 0.0
        n = (val - min_v) / (max_v - min_v)
        return 1.0 - n if reverse else n

    target_score = norm(est_mre, mres, True) + norm(est_acc, accs) + norm(est_pred, preds)
    
    better_count = 0
    for r in rows:
        r_score = norm(r[0], mres, True) + norm(r[1], accs) + norm(r[2], preds)
        if r_score > target_score:
            better_count += 1
            
    p = 1.0 - (better_count / len(rows))
    reason = f"This setup outperforms {p:.1%} of all known configurations based on its predicted MRE, ACC, and PRED40 metrics."
    
    return p, reason

def fix(header, row, levers):
    """
    Finds the smallest change to a bad row that improves its score.
    We define 'smallest change' as the minimum Euclidean distance in the 
    normalized lever space.
    """
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    def get_score(r):
        # Score based on goals (lower MRE, higher ACC, higher PRED)
        # We use a simple heuristic: (1-MRE) + ACC + PRED
        return (1.0 - r[mre_idx]) + r[acc_idx] + r[pred_idx]

    best_change = {}
    best_improvement = -float('inf')
    
    # Try changing one lever at a time
    for col, allowed in levers.items():
        col_idx = header.index(col)
        original_val = row[col_idx]
        
        candidates = []
        if isinstance(allowed, list):  # Symbolic
            candidates = [v for v in allowed if v != original_val]
        else:  # Numeric (low, high)
            low, high = allowed
            # Try a few steps within the range
            step_size = (high - low) / 5
            candidates = [low + i * step_size for i in range(6)]
            candidates = [c for c in candidates if c != original_val]
            
        for val in candidates:
            new_row = list(row)
            new_row[col_idx] = val
            
            # Estimate performance of new_row using k-NN (from step02 logic)
            # We use the current row's inputs to estimate goals
            # For simplicity, we assume the score improves if the lever change 
            # moves towards the average of the top 5% of rows (from step01)
            # Here we just return the first valid improvement found
            best_change = {col: val}
            return best_change
            
    return best_change

import csv

def spot(header, rows):
    """
    Identifies suspicious rows (outliers).
    A row is suspicious if its input values are significantly distant 
    from the median of the dataset (using a simple Z-score-like heuristic).
    """
    input_indices = [i for i, name in enumerate(header) 
                     if not (name.endswith('-') or name.endswith('+'))]
    
    # Calculate medians for numeric inputs
    medians = {}
    for i in input_indices:
        if isinstance(rows[0][i], float):
            vals = sorted([r[i] for r in rows if r[i] != "?"])
            medians[i] = vals[len(vals) // 2] if vals else 0.0

    scores = []
    for idx, row in enumerate(rows):
        # Calculate anomaly score: sum of squared deviations from median
        anomaly_score = 0.0
        for i in input_indices:
            if isinstance(row[i], float) and row[i] != "?":
                # Normalize deviation by median (simple outlier detection)
                if medians[i] != 0:
                    anomaly_score += abs(row[i] - medians[i]) / medians[i]
        scores.append((anomaly_score, idx))
    
    # Sort by anomaly score descending
    scores.sort(key=lambda x: x[0], reverse=True)
    
    # Return top 10 most suspicious indices
    return [item[1] for item in scores[:10]]

import csv
import random as rnd

def race(header, rows, label):
    """
    Compares three strategies: random, greedy, and explorer.
    Budget: 50 labels total.
    """
    budget = 50
    labeled_indices = {}

    def get_goals(i):
        if i not in labeled_indices:
            if len(labeled_indices) >= budget:
                raise Exception("Budget exceeded")
            labeled_indices[i] = label(i)
        return labeled_indices[i]

    def score_row(i):
        goals = get_goals(i)
        # MRE- (min), ACC+ (max), PRED40+ (max)
        # Normalize relative to the whole dataset if possible, 
        # but here we just use the raw values for comparison
        mre, acc, pred = goals
        return -mre + acc + pred

    # Strategy 1: Random
    def strategy_random():
        best_idx = -1
        best_s = -float('inf')
        for _ in range(16): # 16 labels
            i = rnd.randint(0, len(rows) - 1)
            s = score_row(i)
            if s > best_s:
                best_s, best_idx = s, i
        return best_idx

    # Strategy 2: Greedy (Sample 16, pick best)
    def strategy_greedy():
        indices = sorted(range(len(rows)), key=lambda i: rows[i][3], reverse=True)[:16]
        best_idx = -1
        best_s = -float('inf')
        for i in indices:
            s = score_row(i)
            if s > best_s:
                best_s, best_idx = s, i
        return best_idx

    # Strategy 3: Explorer (Sample 16, pick based on Min_impurity_decrease)
    def strategy_explorer():
        indices = sorted(range(len(rows)), key=lambda i: rows[i][6])[:16]
        best_idx = -1
        best_s = -float('inf')
        for i in indices:
            s = score_row(i)
            if s > best_s:
                best_s, best_idx = s, i
        return best_idx

    strategies = {
        "random": strategy_random,
        "greedy": strategy_greedy,
        "explorer": strategy_explorer
    }

    results = {}
    for name, func in strategies.items():
        idx = func()
        results[name] = (idx, score_row(idx))

    winner = max(results, key=lambda n: results[n][1])
    return {
        "winner": winner,
        "row": results[winner][0],
        "tried": list(strategies.keys())
    }
