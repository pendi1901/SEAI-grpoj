import csv
import step01

def guess(header, train, test):
    """
    Estimates the first goal (MRE-) for each test row.
    Uses a simple k-Nearest Neighbors approach based on input columns.
    """
    # Identify input columns (those not ending in - or +)
    input_indices = [
        i for i, name in enumerate(header) 
        if not (name.endswith('-') or name.endswith('+'))
    ]
    
    # Identify the target column (first goal)
    target_idx = 0
    
    predictions = []
    for t_row in test:
        # Calculate Euclidean distance to all training rows based on inputs
        distances = []
        for tr_row in train:
            dist = 0.0
            for i in input_indices:
                # Handle symbolic vs numeric
                if isinstance(t_row[i], str):
                    dist += 0.0 if t_row[i] == tr_row[i] else 1.0
                else:
                    dist += (t_row[i] - tr_row[i]) ** 2
            distances.append((dist**0.5, tr_row[target_idx]))
        
        # Simple k-NN: average of 3 nearest neighbors
        distances.sort(key=lambda x: x[0])
        k = 3
        avg = sum(d[1] for d in distances[:k]) / k
        predictions.append(avg)
        
    return predictions

if __name__ == "__main__":
    filename = "Health-ClosedIssues0000.n1000.train.csv"
    rows = []
    try:
        with open(filename, 'r') as f:
            reader = csv.reader(f)
            header = next(reader)
            for row in reader:
                processed_row = []
                for i, val in enumerate(row):
                    if header[i] in ['criterion']:
                        processed_row.append(val)
                    else:
                        processed_row.append(float(val))
                rows.append(processed_row)
        
        # Split: use first 700 for training, next 100 for testing
        train_data = rows[:700]
        test_data = [[(r[i] if i >= 3 else "?") for i in range(len(r))] for r in rows[700:]]
        
        results = guess(header, train_data, test_data)
        
        print("Estimated MRE for the first 5 test cases:")
        for i in range(min(5, len(results))):
            print(f"Test row {i+1}: Estimated MRE = {results[i]:.4f}")
            
    except FileNotFoundError:
        print(f"Error: {filename} not found.")
        exit(1)
