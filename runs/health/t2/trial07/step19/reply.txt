import csv
import step09

def spot(header, rows):
    """
    Identifies rows that contain input values far outside the observed range 
    of the training data. This leverages the outlier detection logic from step09.
    """
    # The requirement is to identify rows with values "far outside anything we have tried".
    # We reuse the logic from step09 which calculates Z-scores for numeric inputs
    # and flags rows with values significantly outside the distribution.
    return step09.spot(header, rows)

if __name__ == "__main__":
    filename = "Health-ClosedIssues0000.n1000.train.csv"
    try:
        with open(filename, 'r') as f:
            reader = csv.reader(f)
            header = next(reader)
            rows = []
            for row in reader:
                # Convert numeric columns to float, keep symbolic as strings
                processed_row = []
                for i, val in enumerate(row):
                    if header[i] == 'criterion':
                        processed_row.append(val)
                    else:
                        processed_row.append(float(val))
                rows.append(processed_row)
        
        suspicious_indices = spot(header, rows)
        
        if not suspicious_indices:
            print("No suspicious settings found; all inputs are within expected ranges.")
        else:
            print(f"Found {len(suspicious_indices)} rows with settings far outside observed ranges.")
            print("The most suspicious rows (by index) are:")
            for idx in suspicious_indices[:5]:
                print(f"- Row {idx}")
            
    except FileNotFoundError:
        print(f"Error: {filename} not found.")
        exit(1)
