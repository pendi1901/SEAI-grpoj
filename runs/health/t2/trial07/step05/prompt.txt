import csv

def remember(header, rows):
    """
    Identifies the top 5% of rows based on multi-objective optimization.
    Goals: MRE- (minimize), ACC+ (maximize), PRED40+ (maximize).
    We normalize values to [0, 1] and calculate a score.
    """
    n = len(rows)
    if n == 0:
        return []

    # Identify column indices
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    # Extract columns for normalization
    mres = [r[mre_idx] for r in rows]
    accs = [r[acc_idx] for r in rows]
    preds = [r[pred_idx] for r in rows]

    min_mre, max_mre = min(mres), max(mres)
    min_acc, max_acc = min(accs), max(accs)
    min_pred, max_pred = min(preds), max(preds)

    def normalize(val, min_v, max_v, reverse=False):
        if max_v == min_v: return 0.5
        norm = (val - min_v) / (max_v - min_v)
        return (1 - norm) if reverse else norm

    scores = []
    for i in range(n):
        # Score = average of normalized goals
        s = (normalize(rows[i][mre_idx], min_mre, max_mre, True) +
             normalize(rows[i][acc_idx], min_acc, max_acc, False) +
             normalize(rows[i][pred_idx], min_pred, max_pred, False)) / 3
        scores.append((s, i))

    # Sort by score descending
    scores.sort(key=lambda x: x[0], reverse=True)

    # Return top 5% (or at least 1)
    limit = max(1, int(n * 0.05))
    return [item[1] for item in scores[:limit]]

def guess(header, train, test):

    Estimates the first goal (MRE-) for test rows using a K-Nearest Neighbors 
    approach based on the input features (non-goal columns).
    """

    # Goals are columns ending in '-' or '+'
    goal_indices = [i for i, h in enumerate(header) if h.endswith(('-', '+'))]
    target_idx = goal_indices[0]
    input_indices = [i for i in range(len(header)) if i not in goal_indices]

    def dist(row1, row2):
        d = 0
        for i in input_indices:
            v1, v2 = row1[i], row2[i]
            if isinstance(v1, str):  # Symbolic column
                d += 0 if v1 == v2 else 1
            else:  # Numeric column
                d += (v1 - v2)**2
        return d**0.5

    predictions = []
    for t_row in test:
        # Find nearest neighbor in training set
        best_d = float('inf')
        best_val = 0
        for train_row in train:
            d = dist(t_row, train_row)
            if d < best_d:
                best_d = d
                best_val = train_row[target_idx]
        predictions.append(best_val)
    
    return predictions

import csv
import math

def certify(xs, ys):
    """
    Returns True if xs and ys are statistically indistinguishable (noise),
    False if they differ significantly (real effect).
    Uses a simple A12 effect size test (non-parametric).
    """
    def a12(lst1, lst2):
        m, n = len(lst1), len(lst2)
        if m == 0 or n == 0: return 0.5
        greater = 0
        for x in lst1:
            for y in lst2:
                if x > y: greater += 1
                elif x == y: greater += 0.5
        return greater / (m * n)

    # A12 threshold: 0.56 is a small effect, 0.64 medium, 0.71 large.
    # If A12 is close to 0.5, they are likely the same.
    score = a12(xs, ys)
    return 0.44 <= score <= 0.56

import csv
import step01

def flow(header, row, seen):
    """
    Decides if the current row is 'good enough' to stop the build process.
    We define 'good enough' as being in the top 5% of the expected distribution 
    based on the historical data provided in step01.
    """
    if not seen:
        return False

    # Combine seen rows with the current row to establish a context
    all_rows = seen + [row]
    
    # Use step01 logic to identify the top 5% threshold
    # We calculate the score of the current row relative to all seen rows
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    def get_score(r, rows):
        mres = [x[mre_idx] for x in rows]
        accs = [x[acc_idx] for x in rows]
        preds = [x[pred_idx] for x in rows]
        
        def norm(val, vals, rev=False):
            lo, hi = min(vals), max(vals)
            if hi == lo: return 0.5
            n = (val - lo) / (hi - lo)
            return (1 - n) if rev else n
        
        return (norm(r[mre_idx], mres, True) + 
                norm(r[acc_idx], accs, False) + 
                norm(r[pred_idx], preds, False)) / 3

    # If the current row is in the top 5% of the currently seen population, stop.
    current_score = get_score(row, all_rows)
    scores = [get_score(r, all_rows) for r in all_rows]
    scores.sort(reverse=True)
    
    limit = max(1, int(len(all_rows) * 0.05))
    threshold = scores[limit - 1]
    
    return current_score >= threshold

if __name__ == "__main__":
