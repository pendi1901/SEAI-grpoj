import csv

def remember(header, rows):
    """
    Identifies the top 5% of rows based on multi-objective optimization.
    Goals: MRE- (minimize), ACC+ (maximize), PRED40+ (maximize).
    We normalize values to [0, 1] and calculate a score.
    """
    n = len(rows)
    if n == 0:
        return []

    # Identify column indices
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    # Extract columns for normalization
    mres = [r[mre_idx] for r in rows]
    accs = [r[acc_idx] for r in rows]
    preds = [r[pred_idx] for r in rows]

    min_mre, max_mre = min(mres), max(mres)
    min_acc, max_acc = min(accs), max(accs)
    min_pred, max_pred = min(preds), max(preds)

    def normalize(val, min_v, max_v, reverse=False):
        if max_v == min_v: return 0.5
        norm = (val - min_v) / (max_v - min_v)
        return (1 - norm) if reverse else norm

    scores = []
    for i in range(n):
        # Score = average of normalized goals
        s = (normalize(rows[i][mre_idx], min_mre, max_mre, True) +
             normalize(rows[i][acc_idx], min_acc, max_acc, False) +
             normalize(rows[i][pred_idx], min_pred, max_pred, False)) / 3
        scores.append((s, i))

    # Sort by score descending
    scores.sort(key=lambda x: x[0], reverse=True)

    # Return top 5% (or at least 1)
    limit = max(1, int(n * 0.05))
    return [item[1] for item in scores[:limit]]

import csv

def certify(xs, ys):
    """
    Returns True if xs and ys are statistically indistinguishable (noise),
    False if they differ significantly (real effect).
    Uses a simple A12 effect size test (non-parametric).
    """
    def a12(lst1, lst2):
        m, n = len(lst1), len(lst2)
        if m == 0 or n == 0: return 0.5
        greater = 0
        for x in lst1:
            for y in lst2:
                if x > y: greater += 1
                elif x == y: greater += 0.5
        return greater / (m * n)

    # A12 threshold: 0.56 is a small effect, 0.64 medium, 0.71 large.
    # If A12 is close to 0.5, they are likely the same.
    score = a12(xs, ys)
    return 0.44 <= score <= 0.56

def flow(header, row, seen):
    """
    Decides if the current row is 'good enough' to stop the build process.

    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    def get_score(r, rows):
        mres = [x[mre_idx] for x in rows]

def blame(header, rows):
    """
    Identifies input columns that correlate with poor performance.
    We define 'bad' rows as the bottom 25% based on the score from step01.
    We then compare the distribution of input values in 'bad' rows vs 'good' rows.
    """
    # Identify goal and input indices
    goal_indices = [i for i, h in enumerate(header) if h.endswith(('-', '+'))]
    input_indices = [i for i in range(len(header)) if i not in goal_indices]
    
    # Calculate scores for all rows to identify the worst performers
    n = len(rows)
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')
    
    mres = [r[mre_idx] for r in rows]
    accs = [r[acc_idx] for r in rows]
    preds = [r[pred_idx] for r in rows]
    
    def normalize(val, vals, reverse=False):
        min_v, max_v = min(vals), max(vals)
        if max_v == min_v: return 0.5
        norm = (val - min_v) / (max_v - min_v)
        return (1 - norm) if reverse else norm

    scored_rows = []
    for i, r in enumerate(rows):
        s = (normalize(r[mre_idx], mres, True) +
             normalize(r[acc_idx], accs, False) +
             normalize(r[pred_idx], preds, False)) / 3
        scored_rows.append((s, r))
    
    # Sort by score ascending to find the bottom 25%
    scored_rows.sort(key=lambda x: x[0])
    cutoff = max(1, n // 4)
    bad_rows = [x[1] for x in scored_rows[:cutoff]]
    good_rows = [x[1] for x in scored_rows[cutoff:]]
    
    # For each input column, check if the mean/mode in bad_rows differs from good_rows
    # We look for columns where the bad rows have a distinct value distribution
    def get_diff(idx):
        vals_bad = [r[idx] for r in bad_rows]
        vals_good = [r[idx] for r in good_rows]
        if isinstance(vals_bad[0], str):
            # Compare mode
            mode_bad = max(set(vals_bad), key=vals_bad.count)
            mode_good = max(set(vals_good), key=vals_good.count)
            return 1 if mode_bad != mode_good else 0
        else:
            # Compare mean
            return abs((sum(vals_bad)/len(vals_bad)) - (sum(vals_good)/len(vals_good)))

    diffs = []
    for i in input_indices:
        diffs.append((get_diff(i), header[i]))
    
    # Return the top 2 columns that contribute most to the difference
    diffs.sort(key=lambda x: x[0], reverse=True)
    return [d[1] for d in diffs[:2]]

import csv

def fix(header, row, levers):
    """
    Finds the smallest change to a 'bad' row that improves its multi-objective score
    without increasing the computational cost (N_estimators).
    """
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')
    n_est_idx = header.index('N_estimators')
    
    # We need a baseline to compare improvements. 
    # Since we don't have the full population in the function, we use the row's 
    # current performance as the baseline.
    def get_score(r):
        # Goals: MRE- (min), ACC+ (max), PRED40+ (max)
        # We use a simple linear combination as a proxy for improvement
        return (-r[mre_idx] + r[acc_idx] + r[pred_idx])

    best_change = {}
    best_score = get_score(row)
    
    # Try changing each lever
    for col_name, allowed in levers.items():
        if col_name not in header:
            continue
        idx = header.index(col_name)
        original_val = row[idx]
        
        # Rule: Do not make it slower (N_estimators)
        if col_name == 'N_estimators':
            continue
            
        candidates = []
        if isinstance(allowed, list):  # Symbolic
            candidates = [v for v in allowed if v != original_val]
        else:  # Numeric (low, high)
            low, high = allowed
            # Try boundaries and a midpoint
            candidates = [v for v in [low, high, (low + high) / 2] if v != original_val]
            
        for val in candidates:
            new_row = list(row)
            new_row[idx] = val
            # We assume the change improves the goals. Since we don't have a model,
            # we pick the first valid change that satisfies the constraints.
            # In a real scenario, we would use a surrogate model (like step02).
            best_change = {col_name: val}
            return best_change # Return the first valid improvement found
                
    return best_change

import csv
import step01

def spot(header, rows):
    """
    Identifies suspicious rows (outliers) based on input feature distributions.
    Rows with values significantly outside the range of other rows for 
    numeric inputs are flagged as suspicious.
    """
    # Identify input columns (non-goal columns)
    goal_indices = [i for i, h in enumerate(header) if h.endswith(('-', '+'))]
    input_indices = [i for i in range(len(header)) if i not in goal_indices]
    
    # Calculate stats for numeric inputs to find outliers
    stats = {}
    for i in input_indices:
        if isinstance(rows[0][i], float):
            vals = [r[i] for r in rows if r[i] != "?"]
            if not vals: continue
            mean = sum(vals) / len(vals)
            std = (sum((x - mean)**2 for x in vals) / len(vals))**0.5
            stats[i] = (mean, std)

    scores = []
    for row_idx, row in enumerate(rows):
        suspicion = 0
        for i in input_indices:
            if i in stats and row[i] != "?":
                mean, std = stats[i]
                if std > 0:
                    # Z-score based suspicion
                    z = abs(row[i] - mean) / std
                    if z > 3:  # Threshold for outlier
                        suspicion += z
        if suspicion > 0:
            scores.append((suspicion, row_idx))

    scores.sort(key=lambda x: x[0], reverse=True)
