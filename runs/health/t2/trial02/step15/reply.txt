import csv
import step05

def blame(header, rows):
    """
    Identifies input columns that correlate most strongly with 'bad' performance.
    The customer asks which settings can be left at factory defaults without hurting anything.
    We interpret this as: identify the inputs that have the LEAST impact on performance
    (i.e., those that are NOT in the top list of 'blame' for bad performance).
    
    However, the function signature requires returning the names of input columns 
    that best explain why the worst rows are bad. We reuse step05.blame to 
    identify the most impactful settings, and then return the inverse (the settings
    that are NOT the most impactful) as the ones safe to leave at defaults.
    """
    # Get the most impactful settings from step05
    impactful = step05.blame(header, rows)
    
    # Identify all input columns
    input_indices = [i for i, name in enumerate(header) 
                     if not (name.endswith('-') or name.endswith('+'))]
    all_inputs = [header[i] for i in input_indices]
    
    # Return inputs that are not in the 'impactful' list
    safe_to_default = [name for name in all_inputs if name not in impactful]
    
    return safe_to_default

if __name__ == "__main__":
    filename = 'Health-ClosedIssues0000.n1000.train.csv'
    rows = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = [val if header[i] == 'criterion' else float(val) 
                             for i, val in enumerate(row)]
            rows.append(processed_row)

    safe_settings = blame(header, rows)
    
    print("Customer Inquiry: 'Which settings can I leave at the factory default without hurting anything?'")
    print(f"Answer: You can safely leave these settings at their defaults: {', '.join(safe_settings)}.")
