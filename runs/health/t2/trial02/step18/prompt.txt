import csv

def remember(header, rows):
    """
    Identifies the top 5% of rows based on multi-objective optimization.
    Goals: MRE- (minimize), ACC+ (maximize), PRED40+ (maximize).
    We normalize the values and use a simple weighted sum (Pareto-like approach).
    """
    n = len(rows)
    if n == 0:
        return []

    # Identify column indices
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    # Extract values for normalization
    mres = [r[mre_idx] for r in rows]
    accs = [r[acc_idx] for r in rows]
    preds = [r[pred_idx] for r in rows]

    min_mre, max_mre = min(mres), max(mres)
    min_acc, max_acc = min(accs), max(accs)
    min_pred, max_pred = min(preds), max(preds)

    def normalize(val, min_v, max_v):
        if max_v == min_v: return 0
        return (val - min_v) / (max_v - min_v)

    scores = []
    for i, row in enumerate(rows):
        # Normalize: MRE is minimized (1 - norm), others maximized (norm)
        s_mre = 1 - normalize(row[mre_idx], min_mre, max_mre)
        s_acc = normalize(row[acc_idx], min_acc, max_acc)
        s_pred = normalize(row[pred_idx], min_pred, max_pred)
        
        # Aggregate score
        total = s_mre + s_acc + s_pred
        scores.append((total, i))

    # Sort by score descending
    scores.sort(key=lambda x: x[0], reverse=True)
    
    # Return top 5% (or at least 1)
    limit = max(1, int(n * 0.05))
    return [item[1] for item in scores[:limit]]

import csv

def guess(header, train, test):
    """
    Estimates the MRE- (first goal column) for test rows using a 
    k-Nearest Neighbors approach based on input columns.
    """
    # Identify input columns (those not ending in - or +)
    input_indices = [
        i for i, name in enumerate(header) 
        if not (name.endswith('-') or name.endswith('+'))
    ]
    
    # Identify the target column (first goal column)
    target_idx = 0 
    
    predictions = []
    
    for t_row in test:
        distances = []
        for train_row in train:
            dist = 0
            for i in input_indices:
                # Handle symbolic vs numeric inputs
                if isinstance(t_row[i], str):
                    if t_row[i] != train_row[i]:
                        dist += 1
                else:
                    # Simple Euclidean distance for numeric inputs
                    dist += (t_row[i] - train_row[i])**2
            distances.append((dist, train_row[target_idx]))
        
        # Sort by distance and take the average of the 5 nearest neighbors
        distances.sort(key=lambda x: x[0])
        k = 5
        neighbors = distances[:k]
        avg_mre = sum(n[1] for n in neighbors) / k
        predictions.append(avg_mre)
        
    return predictions

import csv
import math

def certify(xs, ys):
    """
    Returns True if the difference between two distributions is likely noise,
    False if they really differ. Uses a simple T-test approach.
    """
    n1, n2 = len(xs), len(ys)
    if n1 < 2 or n2 < 2:
        return True
    
    m1, m2 = sum(xs) / n1, sum(ys) / n2
    v1 = sum((x - m1)**2 for x in xs) / (n1 - 1)
    v2 = sum((y - m2)**2 for y in ys) / (n2 - 1)
    
    # Pooled standard deviation
    pooled_std = math.sqrt(((n1 - 1) * v1 + (n2 - 1) * v2) / (n1 + n2 - 2))
    if pooled_std == 0:
        return m1 == m2
    
    # T-statistic
    t_stat = (m1 - m2) / (pooled_std * math.sqrt(1/n1 + 1/n2))
    
    # For a 95% confidence interval, critical value is approx 1.96
    return abs(t_stat) < 1.96

import csv

def flow(header, row, seen):
    """
    Decides if the current row is 'good enough' to stop searching.
    We define 'good enough' as being in the top 5% of the potential 
    distribution based on the historical data (using the logic from step01).
    """
    # If this is the first row, we can't compare, so keep looking
    if not seen:
        return False

    # Identify goal indices
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    # Combine seen rows with current row to establish the current context
    all_rows = seen + [row]
    
    # Calculate normalization bounds from all rows seen so far
    mres = [r[mre_idx] for r in all_rows]
    accs = [r[acc_idx] for r in all_rows]
    preds = [r[pred_idx] for r in all_rows]

    min_mre, max_mre = min(mres), max(mres)
    min_acc, max_acc = min(accs), max(accs)
    min_pred, max_pred = min(preds), max(preds)

    def get_score(r):
        def norm(val, min_v, max_v):
            return (val - min_v) / (max_v - min_v) if max_v > min_v else 0
        
        s_mre = 1 - norm(r[mre_idx], min_mre, max_mre)
        s_acc = norm(r[acc_idx], min_acc, max_acc)
        s_pred = norm(r[pred_idx], min_pred, max_pred)
        return s_mre + s_acc + s_pred

    # If the current row is in the top 5% of what we've seen, stop
    scores = [get_score(r) for r in all_rows]
    current_score = scores[-1]
    
    # Threshold: is current score better than 95% of the seen distribution?
    threshold = sorted(scores)[int(len(scores) * 0.95)]
    
    return current_score >= threshold

import csv
import step01

def blame(header, rows):
    """
    Identifies input columns that correlate most strongly with 'bad' performance.
    'Bad' is defined as the bottom 10% of rows based on the multi-objective score.
    We compare the distribution of each input in the 'bad' set vs the 'good' set.
    """
    # Identify goal indices
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')
    
    # Identify input indices
    input_indices = [i for i, name in enumerate(header) 
                     if not (name.endswith('-') or name.endswith('+'))]

    # Calculate scores for all rows
    mres = [r[mre_idx] for r in rows]
    accs = [r[acc_idx] for r in rows]
    preds = [r[pred_idx] for r in rows]
    min_mre, max_mre = min(mres), max(mres)
    min_acc, max_acc = min(accs), max(accs)
    min_pred, max_pred = min(preds), max(preds)

    def get_score(r):
        def norm(val, min_v, max_v):
            return (val - min_v) / (max_v - min_v) if max_v > min_v else 0
        return (1 - norm(r[mre_idx], min_mre, max_mre)) + \
               norm(r[acc_idx], min_acc, max_acc) + \
               norm(r[pred_idx], min_pred, max_pred)

    scored_rows = sorted([(get_score(r), r) for r in rows], key=lambda x: x[0])
    
    # Define bad (bottom 10%) and good (top 10%)
    n = len(scored_rows)
    bad = [x[1] for x in scored_rows[:n // 10]]
    good = [x[1] for x in scored_rows[-n // 10:]]

    # Calculate "blame" score: difference in mean (numeric) or frequency (symbolic)
    blame_scores = []
    for i in input_indices:
        if isinstance(rows[0][i], (int, float)):
            mean_bad = sum(r[i] for r in bad) / len(bad)
            mean_good = sum(r[i] for r in good) / len(good)
            diff = abs(mean_bad - mean_good)
        else:
            # For symbolic, check frequency of the most common value in bad set
            vals = [r[i] for r in bad]
            most_common = max(set(vals), key=vals.count)
            freq_bad = vals.count(most_common) / len(bad)
            freq_good = [r[i] for r in good].count(most_common) / len(good)
            diff = freq_bad - freq_good
        blame_scores.append((diff, header[i]))

    # Return top 3 inputs that contribute to bad performance
    blame_scores.sort(key=lambda x: x[0], reverse=True)
    return [name for diff, name in blame_scores[:3]]

import csv

def choose(header, rows, label):
    """
    Finds the best setup among 50 trials.
    Strategy: 
    1. Sample 50 rows randomly (or first 50).
    2. Label them to get their true goal values.
    3. Use the labeled data to train a k-NN model.
    4. Predict goals for all remaining rows.
    5. Return the index of the best performing row.
    """
    n = len(rows)
    # We have a budget of 50 labels.
    # Let's pick 50 indices to sample.
    sample_indices = list(range(0, min(n, 50)))
    
    labeled_rows = []
    for i in sample_indices:
        goals = label(i)
        # Update the row with the true goal values
        rows[i][0], rows[i][1], rows[i][2] = goals
        labeled_rows.append(rows[i])
    
    # Predict for all rows using the 50 labeled ones as training data
    # We use step02.guess to estimate goals for the rest
    # Note: step02.guess expects test rows to have '?' in goal columns
    all_estimates = step02.guess(header, labeled_rows, rows)
    
    # Calculate scores for all rows using the estimates
    mre_idx, acc_idx, pred_idx = 0, 1, 2
    
    # We need to normalize based on the estimates we just generated
    all_mres = [r[mre_idx] if i in sample_indices else all_estimates[i] for i, r in enumerate(rows)]
    all_accs = [r[acc_idx] if i in sample_indices else 0.5 for i, r in enumerate(rows)] # Simplified
    all_preds = [r[pred_idx] if i in sample_indices else 0.5 for i, r in enumerate(rows)] # Simplified
    
    def get_score(i, row):
        mre = row[mre_idx] if i in sample_indices else all_estimates[i]
        # For simplicity in this constrained environment, we use the MRE estimate
        # as the primary driver for the "best" choice.
        return -mre # Minimize MRE

    best_idx = 0
    best_score = -float('inf')
    
    for i in range(n):
        score = get_score(i, rows[i])
        if score > best_score:
            best_score = score
            best_idx = i
            
    return best_idx

import csv
import step02

def fix(header, row, levers):
    """
    Finds the smallest change to settings that improves the row's score.
    We estimate the performance of potential changes using k-NN (step02).
    """
    # Load the training data to provide context for k-NN
    filename = 'Health-ClosedIssues0000.n1000.train.csv'
    rows = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header_file = next(reader)
        for r in reader:
            rows.append([val if header_file[i] == 'criterion' else float(val) for i, val in enumerate(r)])

    def get_score(r):
        # Goals: MRE- (min), ACC+ (max), PRED40+ (max)
        # We use a simple weighted sum of goals
        return -r[0] + r[1] + r[2]

    best_change = {}
    best_score = -float('inf')
    
    # Try changing one lever at a time
    for col_name, allowed in levers.items():
        if col_name not in header:
            continue
        idx = header.index(col_name)
        original_val = row[idx]
        
        candidates = []
        if isinstance(allowed, list):  # Symbolic
            candidates = [v for v in allowed if v != original_val]
        else:  # Numeric (low, high)
            low, high = allowed
            # Try a few discrete steps within the range
            step = (high - low) / 5
            candidates = [low + i * step for i in range(6)]
            candidates = [c for c in candidates if low <= c <= high and abs(c - original_val) > 1e-5]
            
        for val in candidates:
            new_row = row[:]
            new_row[idx] = val
            
            # Estimate goals using k-NN
            # We need to pass a test row with '?' for goals
            test_row = new_row[:]
            test_row[0], test_row[1], test_row[2] = "?", "?", "?"
            
            est_mre = step02.guess(header, rows, [test_row])[0]
            # For simplicity, assume ACC and PRED40 are stable or use average
            est_acc = sum(r[1] for r in rows) / len(rows)
            est_pred = sum(r[2] for r in rows) / len(rows)
            
            new_score = -est_mre + est_acc + est_pred
            
            if new_score > best_score:
                best_score = new_score
                best_change = {col_name: val}
                
    return best_change

import csv

def spot(header, rows):
    """
    Identifies suspicious rows (outliers) based on the assumption that 
    input values should generally fall within a reasonable range 
    relative to the rest of the dataset.
    """
    input_indices = [
        i for i, name in enumerate(header) 
        if not (name.endswith('-') or name.endswith('+'))
    ]
    
    # Calculate statistics for numeric inputs to find outliers
    stats = {}
    for i in input_indices:
        if isinstance(rows[0][i], (int, float)):
            vals = [r[i] for r in rows if r[i] is not None]
            if not vals: continue
            mean = sum(vals) / len(vals)
            std = (sum((x - mean)**2 for x in vals) / len(vals))**0.5
            stats[i] = (mean, std)

    scores = []
    for row_idx, row in enumerate(rows):
        anomaly_score = 0
        for i in input_indices:
            if i in stats:
                mean, std = stats[i]
                if std > 0:
                    # Z-score based anomaly detection
                    z = abs(row[i] - mean) / std
                    if z > 3:  # Threshold for statistical outlier
                        anomaly_score += z
        
        if anomaly_score > 0:

import csv
import random

def race(header, rows, label):
    """
    Compares three strategies for finding the best setup with a 50-label budget.
    Strategies: 'random', 'first_n', 'last_n'.
    """
    budget = 50
    per_strategy = budget // 3
    
    # Identify goal indices
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')
    
    # Helper to evaluate a row based on currently known labeled data
    def get_score(row, labeled_rows):
        mres = [r[mre_idx] for r in labeled_rows]
        accs = [r[acc_idx] for r in labeled_rows]
        preds = [r[pred_idx] for r in labeled_rows]
        
        def norm(val, vals):
            min_v, max_v = min(vals), max(vals)
            return (val - min_v) / (max_v - min_v) if max_v > min_v else 0
        
        s_mre = 1 - norm(row[mre_idx], mres)
        s_acc = norm(row[acc_idx], accs)
        s_pred = norm(row[pred_idx], preds)
        return s_mre + s_acc + s_pred

    results = {}
    labeled_data = []

    def run_strategy(indices, name):
        best_r = -1
        best_s = -float('inf')
        for i in indices:
            goals = label(i)
            rows[i][0], rows[i][1], rows[i][2] = goals
            labeled_data.append(rows[i])
            score = get_score(rows[i], labeled_data)
            if score > best_s:
                best_s, best_r = score, i
        results[name] = (best_s, best_r)

    # 1. Random Strategy
    run_strategy(random.sample(range(len(rows)), per_strategy), "random")
    # 2. First-N Strategy
    run_strategy(list(range(per_strategy)), "first_n")
    # 3. Last-N Strategy
    run_strategy(list(range(len(rows) - per_strategy, len(rows))), "last_n")

    winner = max(results, key=lambda k: results[k][0])
    return {"winner": winner, "row": results[winner][1], "tried": ["random", "first_n", "last_n"]}

import csv

def flow(header, row, seen):
    """
    Decides whether to stop based on whether the current row is the best 
    seen so far. We use the multi-objective scoring logic from step01.
    If the current row's score is higher than any previous row, we stop.
    """
    if not seen:
        return True

    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    # Combine all seen rows to establish the current normalization context
    all_rows = seen + [row]
    mres = [r[mre_idx] for r in all_rows]
    accs = [r[acc_idx] for r in all_rows]
    preds = [r[pred_idx] for r in all_rows]

    min_mre, max_mre = min(mres), max(mres)
    min_acc, max_acc = min(accs), max(accs)
    min_pred, max_pred = min(preds), max(preds)

    def get_score(r):
        def norm(val, min_v, max_v):
            return (val - min_v) / (max_v - min_v) if max_v > min_v else 0
        
        s_mre = 1 - norm(r[mre_idx], min_mre, max_mre)
        s_acc = norm(r[acc_idx], min_acc, max_acc)
        s_pred = norm(r[pred_idx], min_pred, max_pred)
        return s_mre + s_acc + s_pred

    # Calculate scores for all seen rows
    scores = [get_score(r) for r in all_rows]
    
    # If the current row (the last one) has the highest score, we stop
    return scores[-1] >= max(scores[:-1])
