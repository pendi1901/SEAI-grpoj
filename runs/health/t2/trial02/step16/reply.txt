import csv
import step02
import step01

def justify(header, rows, row):
    """
    Estimates the goals for a given setup and compares its performance 
    against historical data to explain why 'bigger/deeper' isn't always better.
    """
    # 1. Estimate the goals for the input row using k-NN
    # We need to pass a list of rows to guess, so we wrap our single row
    estimates = step02.guess(header, rows, [row])
    
    # Update the row with estimated values
    # The goals are the first 3 columns (MRE-, ACC+, PRED40+)
    row[0], row[1], row[2] = estimates[0], 0.5, 0.5 # Using estimates for MRE
    
    # 2. Calculate scores for all rows to compare
    def get_score(r):
        mre_idx, acc_idx, pred_idx = 0, 1, 2
        mres = [x[mre_idx] for x in rows]
        accs = [x[acc_idx] for x in rows]
        preds = [x[pred_idx] for x in rows]
        
        def norm(val, min_v, max_v):
            return (val - min_v) / (max_v - min_v) if max_v > min_v else 0
        
        s_mre = 1 - norm(r[mre_idx], min(mres), max(mres))
        s_acc = norm(r[acc_idx], min(accs), max(accs))
        s_pred = norm(r[pred_idx], min(preds), max(preds))
        return s_mre + s_acc + s_pred

    target_score = get_score(row)
    all_scores = [get_score(r) for r in rows]
    
    # 3. Calculate fraction of rows that 'row' is better than
    better_count = sum(1 for s in all_scores if target_score > s)
    p = better_count / len(rows)
    
    reason = "Bigger and deeper setups often suffer from overfitting, leading to worse generalization than more balanced configurations."
    
    return (p, reason)

if __name__ == "__main__":
    filename = 'Health-ClosedIssues0000.n1000.train.csv'
    rows = []

        reader = csv.reader(f)

        for row in reader:

                             for i, val in enumerate(row)]
            rows.append(processed_row)

    # Create a "big/deep" setup to test the customer's hypothesis
    # N_estimators=200, Max_depth=12
    test_row = [0.0, 0.0, 0.0, 200.0, 'squared_error', 2.0, 2.0, 12.0]
    
    p, reason = justify(header, rows, test_row)
    
    print("Customer Inquiry: 'Why not simply use the biggest, deepest setup the machine can hold?'")
    print(f"Answer: {reason} This specific setup outperforms only {p:.2f} of historical configurations.")
