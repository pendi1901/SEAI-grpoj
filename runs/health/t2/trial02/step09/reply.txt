import csv

def spot(header, rows):
    """
    Identifies suspicious rows (outliers) based on the assumption that 
    input values should generally fall within a reasonable range 
    relative to the rest of the dataset.
    """
    input_indices = [
        i for i, name in enumerate(header) 
        if not (name.endswith('-') or name.endswith('+'))
    ]
    
    # Calculate statistics for numeric inputs to find outliers
    stats = {}
    for i in input_indices:
        if isinstance(rows[0][i], (int, float)):
            vals = [r[i] for r in rows if r[i] is not None]
            if not vals: continue
            mean = sum(vals) / len(vals)
            std = (sum((x - mean)**2 for x in vals) / len(vals))**0.5
            stats[i] = (mean, std)

    scores = []
    for row_idx, row in enumerate(rows):
        anomaly_score = 0
        for i in input_indices:
            if i in stats:
                mean, std = stats[i]
                if std > 0:
                    # Z-score based anomaly detection
                    z = abs(row[i] - mean) / std
                    if z > 3:  # Threshold for statistical outlier
                        anomaly_score += z
        
        if anomaly_score > 0:
            scores.append((anomaly_score, row_idx))
    
    # Sort by anomaly score descending
    scores.sort(key=lambda x: x[0], reverse=True)
    return [item[1] for item in scores]

if __name__ == "__main__":
    filename = 'Health-ClosedIssues0000.n1000.train.csv'
    rows = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = [val if header[i] == 'criterion' else float(val) 
                             for i, val in enumerate(row)]
            rows.append(processed_row)

    suspicious = spot(header, rows)
    
    print("Customer Inquiry: 'Some result rows look like a run that failed and got written down anyway. Find them.'")
    if suspicious:
        print(f"Answer: Found {len(suspicious)} suspicious rows.")
        print("The top 5 most suspicious row indices are:")
        print(suspicious[:5])
    else:
        print("Answer: No suspicious rows detected.")
