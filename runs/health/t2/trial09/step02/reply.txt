import csv
from step01 import remember

def guess(header, train, test):
    """
    Estimates the first goal (MRE-) for test rows by finding the 
    average MRE of the 5% most similar training rows (based on 
    the input columns).
    """
    mre_idx = 0
    # Identify input columns (those not ending in - or +)
    input_indices = [i for i, h in enumerate(header) if not (h.endswith('-') or h.endswith('+'))]
    
    def distance(row1, row2):
        d = 0
        for i in input_indices:
            v1, v2 = row1[i], row2[i]
            if isinstance(v1, str):
                d += 0 if v1 == v2 else 1
            else:
                d += abs(v1 - v2)
        return d

    results = []
    for t_row in test:
        # Find the 5% closest rows in training data
        distances = [(distance(t_row, train_row), train_row[mre_idx]) for train_row in train]
        distances.sort(key=lambda x: x[0])
        
        k = max(1, len(train) // 20)
        neighbors = distances[:k]
        avg_mre = sum(n[1] for n in neighbors) / k
        results.append(avg_mre)
        
    return results

if __name__ == "__main__":
    filename = 'Health-ClosedIssues0000.n1000.train.csv'
    rows = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = []
            for i, val in enumerate(row):
                if header[i] in ['criterion']:
                    processed_row.append(val)
                else:
                    processed_row.append(float(val))
            rows.append(processed_row)

    # Simulate a test set by taking the last 5 rows and masking goals
    test_rows = [r[:] for r in rows[-5:]]
    for r in test_rows:
        for i in range(len(header)):
            if header[i].endswith('-') or header[i].endswith('+'):
                r[i] = "?"
    
    predictions = guess(header, rows[:-5], test_rows)
    
    print("Forecast estimation for new setups:")
    for i, pred in enumerate(predictions):
        print(f"Setup {i+1}: Estimated MRE = {pred:.4f}")
