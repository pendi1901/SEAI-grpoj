import csv
from step01 import remember

def choose(header, rows, label):
    """
    Finds the best setup among 50 trials.
    We use a simple strategy: sample 50 rows, label them, and pick the best.
    """
    # Identify indices
    goal_indices = [i for i, h in enumerate(header) if h.endswith('-') or h.endswith('+')]
    
    # We have 800 rows, we can sample 50. 
    # To get a good spread, we pick indices at regular intervals.
    step = len(rows) // 50
    indices_to_test = [i * step for i in range(50)]
    
    labeled_data = []
    for i in indices_to_test:
        # Get true goals
        goals = label(i)
        # Create a temporary row with goals filled in
        row = list(rows[i])
        for idx, goal_val in zip(goal_indices, goals):
            row[idx] = goal_val
        labeled_data.append((row, i))
    
    # Use remember logic to find the best among the 50 labeled
    # We need to pass a list of rows to remember()
    best_indices_in_labeled = remember(header, [x[0] for x in labeled_data])
    
    # Return the original index of the top-ranked row
    best_idx = labeled_data[best_indices_in_labeled[0]][1]
    return best_idx

if __name__ == "__main__":
    filename = 'Health-ClosedIssues0000.n1000.train.csv'
    rows = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = [val if header[i] == 'criterion' else float(val) 
                             for i, val in enumerate(row)]
            rows.append(processed_row)

    # Store true goals to simulate the label function
    goal_indices = [i for i, h in enumerate(header) if h.endswith('-') or h.endswith('+')]
    true_goals = [[r[i] for i in goal_indices] for r in rows]
    
    called_count = 0
    def label(i):
        global called_count
        called_count += 1
        if called_count > 50:
            raise Exception("Too many labels!")
        return true_goals[i]

    # Mask rows for the choose function
    masked_rows = []
    for r in rows:
        new_r = list(r)
        for i in goal_indices:
            new_r[i] = "?"
        masked_rows.append(new_r)

    best_idx = choose(header, masked_rows, label)
    
    print(f"Search complete. Best setup found at index {best_idx}.")
    print(f"Setup details: {rows[best_idx]}")
