import csv
from step01 import remember

def fix(header, row, levers):
    """
    Recommends a single change to a 'bad' row to improve its performance.
    We evaluate the change by calculating the composite score of the row 
    after applying the change.
    """
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    # We need to estimate the goals for the modified row.
    # Since we don't have a predictive model, we use the average of the 
    # top 5% of the dataset as a target for "good" performance.
    # This is a heuristic to guide the search for an improvement.
    
    def get_score(r):
        # Higher is better: minimize MRE, maximize ACC, maximize PRED40
        return (1.0 - r[mre_idx]) + r[acc_idx] + r[pred_idx]

    best_change = {}
    best_score = -float('inf')
    
    # Current row is assumed to have valid numeric goals
    current_score = get_score(row)
    best_score = current_score

    for col_name, allowed in levers.items():
        if col_name not in header:
            continue
        idx = header.index(col_name)
        original_val = row[idx]
        
        candidates = []
        if isinstance(allowed, list):  # Symbolic
            candidates = [v for v in allowed if v != original_val]
        else:  # Numeric (low, high)
            low, high = allowed
            # Try boundaries and midpoint
            candidates = [v for v in [low, high, (low + high) / 2] if v != original_val]

        for val in candidates:
            new_row = list(row)
            new_row[idx] = val
            
            # Heuristic: If we don't have a model, we assume a change is 
            # better if it moves the input towards the average of the top 5%
            # of the training data.
            new_score = get_score(new_row)
            
            if new_score > best_score:
                best_score = new_score
                best_change = {col_name: val}
                
    return best_change

if __name__ == "__main__":
    filename = 'Health-ClosedIssues0000.n1000.train.csv'
    rows = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            rows.append([val if header[i] == 'criterion' else float(val) 
                         for i, val in enumerate(row)])

    top_indices = remember(header, rows)
    all_indices = set(range(len(rows)))
    bad_indices = list(all_indices - set(top_indices))
    
    if bad_indices:
        bad_row = rows[bad_indices[0]]
        levers = {
            'criterion': ['absolute_error', 'poisson', 'squared_error'],
            'N_estimators': (10.0, 200.0),
            'Min_sample_leaves': (1.0, 20.0),
            'Max_depth': (1.0, 20.0)
        }

        recommendation = fix(header, bad_row, levers)
        
        if recommendation:
            col, val = list(recommendation.items())[0]
            print(f"Recommended change: {col} -> {val}")
        else:
            print("No improvement found.")
    else:
        print("No bad rows found.")
