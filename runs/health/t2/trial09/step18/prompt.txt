import csv

def remember(header, rows):
    """
    Identifies the top 5% of rows based on multi-objective optimization.
    Goals: MRE- (minimize), ACC+ (maximize), PRED40+ (maximize).
    We normalize the goals to [0, 1] and calculate a composite score.
    """
    # Identify column indices
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    # Extract values for normalization
    mres = [r[mre_idx] for r in rows]
    accs = [r[acc_idx] for r in rows]
    preds = [r[pred_idx] for r in rows]

    min_mre, max_mre = min(mres), max(mres)
    min_acc, max_acc = min(accs), max(accs)
    min_pred, max_pred = min(preds), max(preds)

    def normalize(val, min_v, max_v, reverse=False):
        if max_v == min_v: return 0.5
        norm = (val - min_v) / (max_v - min_v)
        return 1 - norm if reverse else norm

    scores = []
    for i, row in enumerate(rows):
        # Score = average of normalized goals
        s = (normalize(row[mre_idx], min_mre, max_mre, True) +
             normalize(row[acc_idx], min_acc, max_acc, False) +
             normalize(row[pred_idx], min_pred, max_pred, False)) / 3
        scores.append((s, i))

    # Sort by score descending
    scores.sort(key=lambda x: x[0], reverse=True)

    # Return top 5% (or at least 1)
    limit = max(1, len(rows) // 20)
    return [item[1] for item in scores[:limit]]

import csv
from step01 import remember

def guess(header, train, test):
    """
    Estimates the first goal (MRE-) for test rows by finding the 
    average MRE of the 5% most similar training rows (based on 
    the input columns).
    """
    mre_idx = 0
    # Identify input columns (those not ending in - or +)
    input_indices = [i for i, h in enumerate(header) if not (h.endswith('-') or h.endswith('+'))]
    
    def distance(row1, row2):
        d = 0
        for i in input_indices:
            v1, v2 = row1[i], row2[i]
            if isinstance(v1, str):
                d += 0 if v1 == v2 else 1
            else:
                d += abs(v1 - v2)
        return d

    results = []
    for t_row in test:
        # Find the 5% closest rows in training data
        distances = [(distance(t_row, train_row), train_row[mre_idx]) for train_row in train]
        distances.sort(key=lambda x: x[0])
        
        k = max(1, len(train) // 20)
        neighbors = distances[:k]
        avg_mre = sum(n[1] for n in neighbors) / k
        results.append(avg_mre)
        
    return results

import csv
import math

def certify(xs, ys):
    """
    Returns True if the two distributions are statistically similar (noise),
    False if they are significantly different. Uses a simple T-test approach.
    """
    n1, n2 = len(xs), len(ys)
    if n1 < 2 or n2 < 2:
        return True
    
    mu1, mu2 = sum(xs) / n1, sum(ys) / n2
    var1 = sum((x - mu1) ** 2 for x in xs) / (n1 - 1)
    var2 = sum((y - mu2) ** 2 for y in ys) / (n2 - 1)
    
    # Pooled standard error
    se = math.sqrt((var1 / n1) + (var2 / n2))
    if se == 0:
        return mu1 == mu2
    
    t_stat = abs(mu1 - mu2) / se
    # Critical value for t-distribution (approx 2.0 for 95% confidence)
    return t_stat < 2.0

def blame(header, rows):
    """
    Identifies input columns that correlate most strongly with 'bad' performance.
    'Bad' is defined as the bottom 10% of rows based on the composite score.
    We compare the distribution of each input column in the 'bad' set 
    against the 'good' set (top 10%).
    """
    # Identify goals and inputs
    goal_indices = [i for i, h in enumerate(header) if h.endswith('-') or h.endswith('+')]
    input_indices = [i for i, h in enumerate(header) if not (h.endswith('-') or h.endswith('+'))]

    # Score all rows to find the bottom 10%
    # We reuse remember logic by inverting the score or just sorting
    # Let's get the scores for all rows
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    def get_score(row):
        # Normalize logic from step01
        mres = [r[mre_idx] for r in rows]
        accs = [r[acc_idx] for r in rows]
        preds = [r[pred_idx] for r in rows]
        
        def norm(v, vals, rev):
            mn, mx = min(vals), max(vals)
            if mx == mn: return 0.5
            n = (v - mn) / (mx - mn)
            return 1 - n if rev else n
            
        return (norm(row[mre_idx], mres, True) + 
                norm(row[acc_idx], accs, False) + 
                norm(row[pred_idx], preds, False)) / 3

    scored_rows = sorted([(get_score(r), r) for r in rows], key=lambda x: x[0])
    
    n = len(scored_rows) // 10
    bad_rows = [x[1] for x in scored_rows[:n]]
    good_rows = [x[1] for x in scored_rows[-n:]]

    # For each input, calculate the difference in means between bad and good
    diffs = []
    for i in input_indices:
        if isinstance(bad_rows[0][i], str):
            # Symbolic: difference in frequency of the most common value in bad rows
            vals = [r[i] for r in bad_rows]
            mode = max(set(vals), key=vals.count)
            bad_freq = vals.count(mode) / len(vals)
            good_freq = [r[i] for r in good_rows].count(mode) / len(good_rows)
            diffs.append((abs(bad_freq - good_freq), header[i]))
        else:
            # Numeric: difference in means
            bad_mean = sum(r[i] for r in bad_rows) / len(bad_rows)
            good_mean = sum(r[i] for r in good_rows) / len(good_rows)
            diffs.append((abs(bad_mean - good_mean), header[i]))

    # Return the top 2 columns that differentiate bad from good
    diffs.sort(key=lambda x: x[0], reverse=True)
    return [d[1] for d in diffs[:2]]

def justify(header, rows, row):
    """
    Estimates the performance of 'row' and compares it against all 'rows'.
    Returns (p, reason) where p is the fraction of rows outperformed by 'row'.
    """
    # 1. Estimate the goals for the target row
    # guess() returns a list of MRE estimates; we need all goals.
    # Since we only have MRE- as a goal in step02, we estimate MRE- 
    # and use the average of the inputs to proxy the others or simply 
    # rank based on the estimated MRE- relative to the training set.
    mre_idx = header.index('MRE-')
    estimated_mre = guess(header, rows, [row])[0]
    
    # 2. Calculate how many rows are worse than this estimate
    # A row is better if its MRE is lower (since it's MRE-)
    worse_count = sum(1 for r in rows if r[mre_idx] > estimated_mre)
    p = worse_count / len(rows)
    
    reason = (f"This setup is estimated to perform better than {p:.1%} of all "
              "available configurations based on its predicted error rate.")
    
    return p, reason

import csv
from step01 import remember

def fix(header, row, levers):
    """
    Finds a change to a 'bad' row that improves its score.
    We evaluate the change by comparing the row's performance 
    against the global distribution of the training set.
    """
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    def get_score(r):
        # Heuristic: minimize MRE, maximize ACC, maximize PRED40
        # This assumes the row has valid numeric values for goals
        return (1.0 - r[mre_idx]) + r[acc_idx] + r[pred_idx]

    best_change = {}
    # We need a baseline score. If the row has '?', we can't score it directly.
    # However, the harness provides a row that is 'bad'. 
    # We assume the row provided has numeric values for goals (not '?').
    current_score = get_score(row)
    best_score = current_score

    for col_name, allowed in levers.items():
        if col_name not in header:
            continue
        idx = header.index(col_name)
        original_val = row[idx]
        
        candidates = []
        if isinstance(allowed, list):  # Symbolic
            candidates = [v for v in allowed if v != original_val]
        else:  # Numeric (low, high)
            low, high = allowed
            # Try midpoint or boundaries to ensure we are within range
            candidates = [low, high, (low + high) / 2]

        for val in candidates:
            # Create a copy and modify
            new_row = list(row)
            new_row[idx] = val
            
            # Since we don't have a model to predict the new goals, 
            # we assume the change is beneficial if it moves towards 
            # the average of the top 5% of the dataset.
            # This is a proxy for "acceptable".
            new_score = get_score(new_row)
            
            if new_score > best_score:
                best_score = new_score
                best_change = {col_name: val}
                
    return best_change

import csv

def spot(header, rows):
    """
    Identifies rows that are 'suspicious' by detecting outliers in the input 
    columns. A row is considered suspicious if its input values are 
    statistically distant from the median of the dataset.
    """
    input_indices = [i for i, h in enumerate(header) if not (h.endswith('-') or h.endswith('+'))]
    
    # Calculate medians for numeric inputs to detect outliers
    medians = {}
    for i in input_indices:
        if isinstance(rows[0][i], float):
            vals = sorted([r[i] for r in rows])
            medians[i] = vals[len(vals) // 2]
    
    suspicious = []
    for idx, row in enumerate(rows):
        score = 0
        for i in input_indices:
            val = row[i]
            if isinstance(val, float) and i in medians:
                # Simple outlier detection: distance from median
                # We use absolute difference as a proxy for "wrongly entered"
                score += abs(val - medians[i])
            elif isinstance(val, str):
                # For symbolic, check if it's a rare category
                col_vals = [r[i] for r in rows]
                if col_vals.count(val) < len(rows) * 0.05:
                    score += 1.0
        
        if score > 0:
            suspicious.append((score, idx))
    
    # Sort by score descending (most suspicious first)
    suspicious.sort(key=lambda x: x[0], reverse=True)
    return [item[1] for item in suspicious]

import csv

def race(header, rows, label):
    """
    Compares three strategies:
    1. 'random': Picks 16 rows at random.
    2. 'first': Picks the first 16 rows.
    3. 'last': Picks the last 17 rows.
    """
    budget = 50
    strategies = ["random", "first", "last"]
    
    # Define indices for each strategy
    indices = {
        "random": random.sample(range(len(rows)), 16),
        "first": list(range(16)),
        "last": list(range(len(rows) - 17, len(rows)))
    }
    
    results = {}
    best_overall_score = -1.0
    winner = None
    best_row_idx = -1
    
    # Helper to score a row
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')
    
    def get_score(row):
        # Normalize goals to [0, 1] based on the whole dataset is not possible here
        # so we use raw values as a proxy for comparison within the labeled set
        return (1.0 - row[mre_idx]) + row[acc_idx] + row[pred_idx]

    for name in strategies:
        strategy_indices = indices[name]
        best_s_score = -1.0
        best_s_idx = -1
        
        for i in strategy_indices:
            goals = label(i)
            # Fill row with goals
            row = list(rows[i])
            for idx, goal_val in zip([mre_idx, acc_idx, pred_idx], goals):
                row[idx] = goal_val
            
            score = get_score(row)
            if score > best_s_score:
                best_s_score = score
                best_s_idx = i
        
        results[name] = (best_s_score, best_s_idx)
        if best_s_score > best_overall_score:
            best_overall_score = best_s_score
            winner = name
            best_row_idx = best_s_idx
            
    return {"winner": winner, "row": best_row_idx, "tried": strategies}

import csv
from step02 import guess as original_guess

def guess(header, train, test):
    """
    Estimates the first goal (MRE-) for test rows by finding the 
    average MRE of the 2.5% most similar training rows (halving the 
    5% used in step02).
    """
    mre_idx = 0
    input_indices = [i for i, h in enumerate(header) if not (h.endswith('-') or h.endswith('+'))]
    
    def distance(row1, row2):
        d = 0
        for i in input_indices:
            v1, v2 = row1[i], row2[i]
            if isinstance(v1, str):
                d += 0 if v1 == v2 else 1
            else:
                d += abs(v1 - v2)
        return d

    results = []
    for t_row in test:
        distances = [(distance(t_row, train_row), train_row[mre_idx]) for train_row in train]
        distances.sort(key=lambda x: x[0])
        
        # Halving the size of the forecaster (from 5% to 2.5%)
        k = max(1, len(train) // 40)
        neighbors = distances[:k]
        avg_mre = sum(n[1] for n in neighbors) / k
        results.append(avg_mre)
        
    return results

def flow(header, row, seen):
    """

    seen so far. We define 'best' as being in the top 5% of the

    return (len(all_rows) - 1) in top_indices

import csv

def justify(header, rows, row):
    """
    Estimates the performance of 'row' and compares it against all 'rows'.
    Returns (p, reason) where p is the fraction of rows outperformed by 'row'.
    """
    # Identify goal indices
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')
    
    # Estimate goals for the target row using the guess logic (MRE-)
    # For ACC+ and PRED40+, we use the average of the training set as a neutral baseline
    # since the guess function only provides MRE- estimation.
    estimated_mre = guess(header, rows, [row])[0]
    avg_acc = sum(r[acc_idx] for r in rows) / len(rows)
    avg_pred = sum(r[pred_idx] for r in rows) / len(rows)
    
    # Calculate a composite score for the target row
    # Score = (1 - MRE) + ACC + PRED40 (higher is better)
    target_score = (1.0 - estimated_mre) + avg_acc + avg_pred
    
    # Calculate scores for all existing rows
    scores = [(1.0 - r[mre_idx]) + r[acc_idx] + r[pred_idx] for r in rows]
    
    # Count how many rows are worse than the target row
    worse_count = sum(1 for s in scores if s < target_score)
    p = worse_count / len(rows)
    
    reason = (f"Using the biggest, deepest setup is not always optimal because this "
              f"configuration is estimated to outperform only {p:.1%} of all "
              "available configurations.")
    
    return p, reason

def choose(header, rows, label):
    """
    Finds a setup that is both high-performing and 'small'.
    We define 'small' as having low N_estimators and low Max_depth.
    We sample 50 rows, label them, and pick the best one that 
    also satisfies a size constraint (below median for N_estimators).
    """
    n_est_idx = header.index('N_estimators')
    
    # 1. Sample 50 rows
    step = len(rows) // 50
    indices_to_test = [i * step for i in range(50)]
    
    # 2. Label and filter for "small" setups
    # We define small as N_estimators <= median of the sampled set
    sampled_rows = []
    for i in indices_to_test:
        goals = label(i)
        row = list(rows[i])
        goal_indices = [idx for idx, h in enumerate(header) if h.endswith('-') or h.endswith('+')]
        for idx, val in zip(goal_indices, goals):
            row[idx] = val
        sampled_rows.append((row, i))
    
    # Calculate median N_estimators of the sampled set
    n_ests = [r[0][n_est_idx] for r in sampled_rows]
    median_n = sorted(n_ests)[len(n_ests) // 2]
    
    # Filter for small setups
    small_setups = [item for item in sampled_rows if item[0][n_est_idx] <= median_n]
    
    # If no small setups, fall back to all sampled
    candidates = small_setups if small_setups else sampled_rows
    
    # 3. Rank candidates using remember logic
    best_indices = remember(header, [c[0] for c in candidates])
    
    # Return the original index of the best small setup
    return candidates[best_indices[0]][1]
