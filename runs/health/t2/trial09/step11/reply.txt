import csv
from step01 import remember as base_remember

def distance(row1, row2, input_indices):
    """Calculates Manhattan distance between two rows based on input columns."""
    d = 0
    for i in input_indices:
        v1, v2 = row1[i], row2[i]
        if isinstance(v1, str):
            d += 0 if v1 == v2 else 1
        else:
            d += abs(v1 - v2)
    return d

def remember(header, rows):
    """
    Returns the top 5% of rows based on the multi-objective optimization 
    defined in step01.
    """
    return base_remember(header, rows)

def find_similar_setups(header, rows, target_row, n=5):
    """
    Finds the 'n' rows in 'rows' most similar to 'target_row' 
    based on input columns.
    """
    input_indices = [i for i, h in enumerate(header) if not (h.endswith('-') or h.endswith('+'))]
    
    # Calculate distances to target
    scored = []
    for i, row in enumerate(rows):
        d = distance(row, target_row, input_indices)
        scored.append((d, i))
    
    # Sort by distance ascending
    scored.sort(key=lambda x: x[0])
    return scored[:n]

if __name__ == "__main__":
    filename = 'Health-ClosedIssues0000.n1000.train.csv'
    rows = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = [val if header[i] == 'criterion' else float(val) 
                             for i, val in enumerate(row)]
            rows.append(processed_row)

    # Assume the team's current setup is the first row
    current_setup = rows[0]
    
    # Find 5 most similar setups
    similar = find_similar_setups(header, rows[1:], current_setup)
    
    print("Analysis of setups similar to the current configuration:")
    print(f"Current Setup: {current_setup}")
    print("-" * 30)
    
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')
    
    for dist, idx in similar:
        row = rows[idx + 1]
        print(f"Row {idx + 1} (Dist: {dist:.2f}): "
              f"MRE={row[mre_idx]:.4f}, ACC={row[acc_idx]:.4f}, PRED40={row[pred_idx]:.4f}")
