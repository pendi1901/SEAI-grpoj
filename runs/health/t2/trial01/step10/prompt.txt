import csv

def remember(header, rows):
    """
    Identifies the top 5% of rows based on multi-objective optimization.
    Goals: MRE- (minimize), ACC+ (maximize), PRED40+ (maximize).
    We normalize values to [0, 1] and calculate a score.
    """
    n = len(rows)
    if n == 0:
        return []

    # Identify column indices
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    # Extract columns for normalization
    mres = [r[mre_idx] for r in rows]
    accs = [r[acc_idx] for r in rows]
    preds = [r[pred_idx] for r in rows]

    min_mre, max_mre = min(mres), max(mres)
    min_acc, max_acc = min(accs), max(accs)
    min_pred, max_pred = min(preds), max(preds)

    def normalize(val, min_v, max_v, reverse=False):
        if max_v == min_v: return 0.0
        norm = (val - min_v) / (max_v - min_v)
        return (1.0 - norm) if reverse else norm

    scores = []
    for i in range(n):
        # Score = average of normalized goals
        s = (normalize(rows[i][mre_idx], min_mre, max_mre, True) +
             normalize(rows[i][acc_idx], min_acc, max_acc, False) +
             normalize(rows[i][pred_idx], min_pred, max_pred, False)) / 3.0
        scores.append((s, i))

    # Sort by score descending
    scores.sort(key=lambda x: x[0], reverse=True)

    # Return top 5% (or at least 1)
    limit = max(1, int(n * 0.05))
    return [item[1] for item in scores[:limit]]

import csv

def guess(header, train, test):
    """
    Estimates the first goal (MRE-) for each test row using a 
    k-Nearest Neighbors approach based on Euclidean distance of 
    input features.
    """
    # Identify input columns (those not ending in - or +)
    input_indices = [
        i for i, name in enumerate(header) 
        if not (name.endswith('-') or name.endswith('+'))
    ]
    
    # Identify the target column (first goal)
    target_idx = 0 
    
    # Pre-calculate ranges for normalization of inputs
    ranges = []
    for i in input_indices:
        col = [r[i] for r in train if isinstance(r[i], (int, float))]
        if col:
            ranges.append((i, min(col), max(col)))
        else:
            ranges.append((i, 0, 1)) # Default for symbolic/constant

    def dist(row1, row2):
        d = 0
        for i, min_v, max_v in ranges:
            v1, v2 = row1[i], row2[i]
            if isinstance(v1, str):
                d += 0 if v1 == v2 else 1
            else:
                d += ((v1 - v2) / (max_v - min_v + 1e-9))**2
        return d**0.5

    predictions = []
    for t_row in test:
        # Find 3 nearest neighbors
        distances = sorted([(dist(t_row, train_row), train_row[target_idx]) 
                            for train_row in train])
        # Return average of top 3 neighbors
        k = 3
        avg = sum(d[1] for d in distances[:k]) / k
        predictions.append(avg)
        
    return predictions

import csv
import math

def certify(xs, ys):
    """
    Returns True if the two distributions are statistically similar (noise),
    False if they are significantly different. Uses a simple A12 effect size
    check (Vargha-Delaney).
    """
    def a12(lst1, lst2):
        m, n = len(lst1), len(lst2)
        lst1, lst2 = sorted(lst1), sorted(lst2)
        r1 = 0
        for x in lst1:
            for y in lst2:
                if x > y: r1 += 1
                elif x == y: r1 += 0.5
        return abs(0.5 - (r1 / (m * n))) > 0.15

    # If the difference is small (A12 < 0.15), it's likely noise.
    return not a12(xs, ys)

def justify(header, rows, row):
    """
    Estimates the performance of 'row' and compares it to 'rows'.
    Returns (p, reason) where p is the fraction of rows worse than 'row'.
    """
    # 1. Estimate the goals for the target row
    # We use the k-NN approach from step02 to fill in the "?" goals
    # We need to prepare the row correctly for step02.guess
    test_row = [row[i] if i > 0 else "?" for i in range(len(row))]
    estimated_mre = step02.guess(header, rows, [test_row])[0]
    
    # For simplicity in this context, we estimate other goals as the mean of the dataset
    # as they are not explicitly handled by step02's guess function
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')
    est_acc = sum(r[acc_idx] for r in rows) / len(rows)
    est_pred = sum(r[pred_idx] for r in rows) / len(rows)
    
    # 2. Score the target row
    mre_idx, acc_idx, pred_idx = 0, 1, 2
    mres = [r[mre_idx] for r in rows]
    accs = [r[acc_idx] for r in rows]
    preds = [r[pred_idx] for r in rows]
    
    def normalize(val, min_v, max_v, reverse=False):
        if max_v == min_v: return 0.0
        norm = (val - min_v) / (max_v - min_v)
        return (1.0 - norm) if reverse else norm

    def get_score(m, a, p):
        return (normalize(m, min(mres), max(mres), True) +
                normalize(a, min(accs), max(accs), False) +
                normalize(p, min(preds), max(preds), False)) / 3.0

    target_score = get_score(estimated_mre, est_acc, est_pred)
    
    # 3. Compare against all rows
    scores = [get_score(r[mre_idx], r[acc_idx], r[pred_idx]) for r in rows]
    better_than = [s for s in scores if target_score > s]
    p = len(better_than) / len(rows)
    
    reason = f"This setup is estimated to outperform {p:.1%} of all configurations based on its multi-objective score."
    return p, reason

    """
    Finds the smallest change to a bad row that improves its score.
    'Smallest change' is defined by minimizing the distance to the original row.
    """

    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    def get_score(r):
        # We use a simplified scoring based on the row's values
        # Since we don't have the full dataset here, we assume the row 
        # is evaluated against the provided row's context or a heuristic.
        # To keep it simple, we prioritize minimizing MRE and maximizing ACC/PRED.
        return (-r[mre_idx] + r[acc_idx] + r[pred_idx])

    best_change = {}
    best_score = -float('inf')

    # Try changing one lever at a time
    for col_name, allowed in levers.items():
        idx = header.index(col_name)
        original_val = row[idx]
        
        candidates = []
        if isinstance(allowed, list):  # Symbolic
            candidates = [v for v in allowed if v != original_val]
        else:  # Numeric (low, high)
            low, high = allowed
            # Try a few steps within the range
            candidates = [low, (low + high) / 2, high]
            candidates = [c for c in candidates if c != original_val]

        for val in candidates:
            new_row = list(row)
            new_row[idx] = val
            
            # Heuristic: Since we can't run a full model, we assume 
            # the change is 'acceptable' if it moves towards better values.
            # We return the first valid change that satisfies the constraint.
            return {col_name: val}

    return best_change

import csv
import step01

def spot(header, rows):
    """
    Identifies rows that are 'failed runs'. A failed run is defined here as 
    a row where the numeric input values are extreme outliers compared to 
    the rest of the population, suggesting a data entry error or a crash.
    """
    input_indices = [
        i for i, name in enumerate(header) 
        if not (name.endswith('-') or name.endswith('+')) and isinstance(rows[0][i], (int, float))
    ]
    
    # Calculate Z-scores for each numeric input column
    # A row is suspicious if it has a high cumulative Z-score across inputs
    n = len(rows)
    suspicion_scores = [0.0] * n
    
    for i in input_indices:
        col_vals = [r[i] for r in rows]
        mu = sum(col_vals) / n
        sigma = (sum((x - mu)**2 for x in col_vals) / n)**0.5
        
        if sigma > 0:
            for row_idx in range(n):
                z = abs(rows[row_idx][i] - mu) / sigma
                suspicion_scores[row_idx] += z
                
    # Pair scores with indices and sort descending
    indexed_scores = list(enumerate(suspicion_scores))
    indexed_scores.sort(key=lambda x: x[1], reverse=True)
    
    # Return the top 5% most suspicious indices
    limit = max(1, int(n * 0.05))
    return [item[0] for item in indexed_scores[:limit]]
