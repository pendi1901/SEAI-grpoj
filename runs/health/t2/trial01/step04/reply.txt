import csv
import step01

def flow(header, row, seen):
    """
    Decides if the current row is 'good enough' to ship.
    We use the logic from step01 to determine the quality of the current row
    relative to the history of seen rows. If the current row's score is 
    in the top 5% of all seen rows (including itself), we ship.
    """
    # Combine seen rows with current row to evaluate relative rank
    all_rows = seen + [row]
    
    # Use step01's remember logic to find indices of top 5%
    # Note: step01.remember returns indices of the best rows
    best_indices = step01.remember(header, all_rows)
    
    # If the current row (index len(seen)) is in the top 5%, ship it
    return len(seen) in best_indices

if __name__ == "__main__":
    filename = "Health-ClosedIssues0000.n1000.train.csv"
    try:
        with open(filename, 'r') as f:
            reader = csv.reader(f)
            header = next(reader)
            rows = []
            for row in reader:
                processed_row = []
                for i, val in enumerate(row):
                    if header[i] in ['MRE-', 'ACC+', 'PRED40+', 'N_estimators', 
                                     'Min_sample_leaves', 'Min_impurity_decrease', 'Max_depth']:
                        processed_row.append(float(val))
                    else:
                        processed_row.append(val)
                rows.append(processed_row)
        
        shipped = False
        seen = []
        for i, row in enumerate(rows):
            if flow(header, row, seen):
                print(f"Shipped setup at index {i} (MRE-: {row[0]}, ACC+: {row[1]})")
                shipped = True
                break
            seen.append(row)
        
        if not shipped:
            print("No setup met the criteria for shipping.")
            
    except Exception as e:
        print(f"Error: {e}")
        exit(1)
