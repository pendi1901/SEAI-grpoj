import csv
import step05

def blame(header, rows):
    """
    Identifies input columns that correlate with poor performance.
    We reuse the logic from step05 to identify the culprits.
    """
    return step05.blame(header, rows)

if __name__ == "__main__":
    filename = "Health-ClosedIssues0000.n1000.train.csv"
    try:
        rows = []
        with open(filename, 'r') as f:
            reader = csv.reader(f)
            header = next(reader)
            for row in reader:
                processed = []
                for i, val in enumerate(row):
                    if header[i] in ['MRE-', 'ACC+', 'PRED40+', 'N_estimators', 
                                     'Min_sample_leaves', 'Min_impurity_decrease', 'Max_depth']:
                        processed.append(float(val))
                    else:
                        processed.append(val)
                rows.append(processed)
        
        # The customer asks which settings can be left at default without hurting anything.
        # Settings that are NOT identified as 'culprits' (those most responsible for 
        # poor performance) are safe to leave at default.
        
        all_inputs = [name for name in header if not (name.endswith('-') or name.endswith('+'))]
        culprits = blame(header, rows)
        safe_settings = [name for name in all_inputs if name not in culprits]
        
        print("Analysis: Which settings can I leave at factory default?")
        print("The following settings are not strongly associated with poor performance:")
        print(", ".join(safe_settings))
        print("\nNote: The settings most responsible for poor performance (and thus worth tuning) are:")
        print(", ".join(culprits))
        
        exit(0)
    except Exception as e:
        print(f"Error: {e}")
        exit(1)
