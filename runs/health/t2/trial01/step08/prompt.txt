def remember(header, rows):

    goal_cols = [c for c in header if c.endswith('-') or c.endswith('+')]

            val_str = row[header.index(col)]
            if val_str == '?' or val_str is None:
                row_valid[i] = False
            else:
                try:
                    val = float(val_str)
                except ValueError:
                    row_valid[i] = False
                    break
                col_vals.setdefault(col, []).append((i, val))

    # Pass 2 – normalise each goal to [0,1] and sum; higher is better

            continue
        score = 0.0

            else:                          # maximise → high is best
                score += (val - mn) / (mx - mn + 1e-9)
        scored.append((i, score))

    n = len(scored)
    k = max(1, int(n * 0.05))           # at most 5% of rows
    return [pos for pos, _ in scored[:k]]

                d += diff * diff
            dists.append((d ** 0.5, train_mre))

        # Sort by distance and take the closest k
        dists.sort(key=lambda x: x[0])
        k = max(1, min(len(dists), len(train_data) // 2))

def certify(xs, ys):
    """Return True if xs and ys agree up to noise; False if they really differ."""
    n = len(xs)
    if n != len(ys):
        return False
    # Use a simple threshold: differences must be small in most positions.
    diffs = [abs(a - b) for a, b in zip(xs, ys)]
    # If more than half the absolute differences exceed 0.1, they really differ.
    bad = sum(1 for d in diffs if d > 0.1)
    return bad <= n // 2

def blame(header, rows):
    """Return input column names that explain why the worst rows are bad."""

    goal_cols = [c for c in header if c.endswith('-') or c.endswith('+')]

    # Score every row with the same normalisation as remember()
    scored = remember(header, rows)
    n = len(rows)

    # The bottom ~5% are the worst setups
    bad_count = max(1, int(n * 0.05))
    bad_indices = set(pos for pos in scored[-bad_count:])

    # Numeric input columns (uppercase, not a goal)
    input_cols = []
    for col in header:
        if col[0].isupper() and col not in goal_cols:
            input_cols.append(col)

    explanations = []
    for col in input_cols:
        bad_vals = [float(rows[i][header.index(col)]) for i in bad_indices]
        good_indices = [i for i in range(n) if i not in bad_indices]
        good_vals = [float(rows[i][header.index(col)]) for i in good_indices]

        bad_mean = sum(bad_vals) / len(bad_vals) if bad_vals else 0.0
        good_mean = sum(good_vals) / len(good_vals) if good_vals else 0.0
        diff = abs(bad_mean - good_mean)

        explanations.append((col, diff))

    # Largest mean-shift first — these are the settings doing the damage
    explanations.sort(key=lambda x: -x[1])
    return [explanations[i][0] for i in range(len(explanations))]

def justify(header, rows, row):
    """Return (p, reason) where p is fraction of rows that `row` beats."""

    goal_cols = [c for c in header if c.endswith('-') or c.endswith('+')]

    # Score all complete rows to understand the landscape
    scored_rows = remember(header, rows)
    n_complete = len(scored_rows)

    # For the new row with "?", predict MRE using KNN on input columns
    target_col = goal_cols[0]  # "MRE-"

    input_cols = []
    for col in header:
        if col == target_col:
            continue
        if col[0].isupper():
            input_cols.append(col)

    # Build training data from complete rows
    train_data = []
    for r in rows:
        inputs = [float(r[col]) for col in input_cols]
        mre_str = r[target_col]
        try:
            mre = float(mre_str)
        except ValueError:
            continue
        train_data.append((inputs, mre))

    # Predict MRE for the new row using KNN
    inputs_new = [float(row[col]) for col in input_cols]

    dists = []
    for train_inputs, train_mre in train_data:
        d = 0.0
        for a, b in zip(inputs_new, train_inputs):
            diff = a - b
            d += diff * diff
        dists.append((d ** 0.5, train_mre))

    dists.sort(key=lambda x: x[0])
    k = max(1, min(len(dists), len(train_data) // 2))
    predicted_mre = sum(m for _, m in dists[:k]) / k

    # Count how many complete rows have MRE- <= predicted_mre (lower is better)
    count_better = sum(1 for r in rows if float(r[target_col]) <= predicted_mre)

    n_total = len(rows) + 1  # including the new row
    p = count_better / n_total

    reason = f"This setup is better than {p:.1%} of all configurations."

    return (p, reason)

import csv


def choose(header, rows, label):
    """Find best row among up to 50 labeled evaluations."""

    goal_cols = [c for c in header if c.endswith('-') or c.endswith('+')]

    # Identify numeric input columns (uppercase, not goals)
    input_cols = []
    for col in header:
        if col[0].isupper() and col not in goal_cols:
            input_cols.append(col)

    target_col = goal_cols[0]  # "MRE-"

    # Build training data from rows with known goals (non-? cells)
    train_data = []
    for row in rows:
        inputs = [float(row[col]) for col in input_cols]
        mre_str = row[target_col]
        try:
            mre = float(mre_str)
        except ValueError:
            continue
        train_data.append((inputs, mre))

    # Predict MRE- for each row using KNN on inputs
    predicted_scores = []
    for i, row in enumerate(rows):
        inputs_new = [float(row[col]) for col in input_cols]

        dists = []
        for train_inputs, train_mre in train_data:
            d = 0.0
            for a, b in zip(inputs_new, train_inputs):
                diff = a - b
                d += diff * diff
            dists.append((d ** 0.5, train_mre))

        if not dists:
            predicted_scores.append(float('inf'))
            continue

        dists.sort(key=lambda x: x[0])
        k = max(1, min(len(dists), len(train_data) // 2))
        avg_mre = sum(m for _, m in dists[:k]) / k
        predicted_scores.append(avg_mre)

    # Rank rows by predicted score (lower MRE- is better)
    ranked = sorted(range(len(predicted_scores)), key=lambda i: predicted_scores[i])

    best_pos = None
    best_score = float('inf')

    # Try labeling top candidates until we have enough or hit 50 calls
    for pos in ranked:
        if label(pos):
            true_mre = rows[pos][header.index(target_col)]
            try:
                mre_val = float(true_mre)
            except ValueError:
                continue

            # Compare with current best
            if mre_val < best_score:
                best_score = mre_val
                best_pos = pos

        if best_pos is not None and len([p for p in ranked if p <= pos]) >= 50:
            break

    return best_pos
