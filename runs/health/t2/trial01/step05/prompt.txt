import csv

def remember(header, rows):
    """
    Identifies the top 5% of rows based on multi-objective optimization.
    Goals: MRE- (minimize), ACC+ (maximize), PRED40+ (maximize).
    We normalize values to [0, 1] and calculate a score.
    """
    n = len(rows)
    if n == 0:
        return []

    # Identify column indices
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    # Extract columns for normalization
    mres = [r[mre_idx] for r in rows]
    accs = [r[acc_idx] for r in rows]
    preds = [r[pred_idx] for r in rows]

    min_mre, max_mre = min(mres), max(mres)
    min_acc, max_acc = min(accs), max(accs)
    min_pred, max_pred = min(preds), max(preds)

    def normalize(val, min_v, max_v, reverse=False):
        if max_v == min_v: return 0.0
        norm = (val - min_v) / (max_v - min_v)
        return (1.0 - norm) if reverse else norm

    scores = []
    for i in range(n):
        # Score = average of normalized goals
        s = (normalize(rows[i][mre_idx], min_mre, max_mre, True) +
             normalize(rows[i][acc_idx], min_acc, max_acc, False) +
             normalize(rows[i][pred_idx], min_pred, max_pred, False)) / 3.0
        scores.append((s, i))

    # Sort by score descending
    scores.sort(key=lambda x: x[0], reverse=True)

    # Return top 5% (or at least 1)
    limit = max(1, int(n * 0.05))
    return [item[1] for item in scores[:limit]]

import csv
import step01

def guess(header, train, test):
    """
    Estimates the first goal (MRE-) for each test row using a 
    k-Nearest Neighbors approach based on Euclidean distance of 
    input features.
    """
    # Identify input columns (those not ending in - or +)
    input_indices = [
        i for i, name in enumerate(header) 
        if not (name.endswith('-') or name.endswith('+'))
    ]
    
    # Identify the target column (first goal)
    target_idx = 0 
    
    # Pre-calculate ranges for normalization of inputs
    ranges = []
    for i in input_indices:
        col = [r[i] for r in train if isinstance(r[i], (int, float))]
        if col:
            ranges.append((i, min(col), max(col)))
        else:
            ranges.append((i, 0, 1)) # Default for symbolic/constant

    def dist(row1, row2):
        d = 0
        for i, min_v, max_v in ranges:
            v1, v2 = row1[i], row2[i]
            if isinstance(v1, str):
                d += 0 if v1 == v2 else 1
            else:
                d += ((v1 - v2) / (max_v - min_v + 1e-9))**2
        return d**0.5

    predictions = []
    for t_row in test:
        # Find 3 nearest neighbors
        distances = sorted([(dist(t_row, train_row), train_row[target_idx]) 
                            for train_row in train])
        # Return average of top 3 neighbors
        k = 3
        avg = sum(d[1] for d in distances[:k]) / k
        predictions.append(avg)
        
    return predictions

import csv
import math

def certify(xs, ys):
    """
    Returns True if the two distributions are statistically similar (noise),
    False if they are significantly different. Uses a simple A12 effect size
    check (Vargha-Delaney).
    """
    def a12(lst1, lst2):
        m, n = len(lst1), len(lst2)
        lst1, lst2 = sorted(lst1), sorted(lst2)
        r1 = 0
        for x in lst1:
            for y in lst2:
                if x > y: r1 += 1
                elif x == y: r1 += 0.5
        return abs(0.5 - (r1 / (m * n))) > 0.15

    # If the difference is small (A12 < 0.15), it's likely noise.
    return not a12(xs, ys)

def flow(header, row, seen):
    """
    Decides if the current row is 'good enough' to ship.
    We use the logic from step01 to determine the quality of the current row
    relative to the history of seen rows. If the current row's score is 
    in the top 5% of all seen rows (including itself), we ship.
    """

    all_rows = seen + [row]
