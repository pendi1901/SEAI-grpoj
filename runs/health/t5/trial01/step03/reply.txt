import ezr


def certify(xs, ys):
    """Return True if xs and ys are the same apart from noise."""
    if len(xs) != len(ys): return False
    diffs = [abs(a - b) for a, b in zip(xs, ys)]
    avg_diff = sum(diffs) / len(diffs)
    scale = max(max(xs), max(ys)) if xs and ys else 1.0
    return avg_diff < 0.1 * scale


if __name__ == "__main__":
    import os
    path = os.path.join(os.path.dirname(__file__) or ".", "Health-ClosedIssues0000.train.csv")
    with open(path, encoding="utf-8-sig") as f:
        lines = [l.strip() for l in f if l.strip()]
    header = lines[0].split(",")
    rows = [[ezr.atom(s) for s in line.split(",")] for line in lines[1:]]

    n = len(rows)
    split = int(n * 0.8)
    train_rows, test_rows = rows[:split], rows[split:]

    # Big setup: tree on all training rows
    tbl_big = ezr.Tbl(zip(header, train_rows))
    node_big = ezr.tree(tbl_big, tbl_big.rows)

    # Small setup: subsample half the training rows
    import random
    random.seed(42)
    small_idx = random.sample(range(len(train_rows)), len(train_rows) // 2)
    small_rows = [train_rows[i] for i in sorted(small_idx)]
    tbl_small = ezr.Tbl(zip(header, small_rows))
    node_small = ezr.tree(tbl_small, tbl_small.rows)

    preds_big = [ezr.leaf(node_big, r).mu for r in test_rows]
    preds_small = [ezr.leaf(node_small, r).mu for r in test_rows]

    agree = certify(preds_big, preds_small)
    print(f"Big vs small MRE predictions agree: {agree}")
