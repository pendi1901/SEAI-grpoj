import ezr


def remember(header, rows):
    """Return positions (0-based) of rows whose forecasts are unusually trustworthy."""
    tbl = ezr.Tbl(zip(header, rows))
    # Build a ranker from the existing tree on all labelled rows.
    rank = ezr.ranker(tbl)

    # Identify goal columns and compute each row's ydist.
    goals = [c for c in tbl.y if c.goal != 0]
    ys = [(ezr.ydist(tbl, r), i) for i, r in enumerate(rows)]
    ys.sort()

    # Pick the top rows whose ydist is well below average and above the best.
    n = len(ys)
    avg_y = sum(y for y, _ in ys) / n
    lo_y = ys[0][0]
    threshold = (avg_y + lo_y) / 2

    # Filter: rows with ydist below threshold and at least one goal column.
    candidates = [i for i, y in zip(*zip(*ys), range(n)) if y < threshold]
    # If no candidates pass the threshold, take the top 5% by ydist.
    if not candidates:
        idx = max(1, int(0.05 * n))
        return [i for i, _ in ys[:idx]]

    # Return up to 5% of rows (at least one).
    limit = max(1, int(0.05 * n))
    best = sorted(candidates)[:limit]
    return list(best)


if __name__ == "__main__":
    import os
    path = os.path.join(os.path.dirname(__file__) or ".", "Health-ClosedIssues0000.train.csv")
    with open(path, encoding="utf-8-sig") as f:
        lines = [l.strip() for l in f if l.strip()]
    header = lines[0].split(",")
    rows = [[ezr.atom(s) for s in line.split(",")] for line in lines[1:]]
    positions = remember(header, rows)
    print(f"Trustworthy row positions ({len(positions)}): {positions}")
