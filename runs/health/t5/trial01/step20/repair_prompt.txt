ezr.py: multi-objective XAI using minimal labels.   
Faster, simpler AI. Better maps, not bigger boots.    
(c) 2026 Tim Menzies <timm@ieee.org> MIT license.    

Options:    
      -Start=4         acquire: initial random labels   
      -Stop=50         acquire: total labelling budget   
      -Few=128         holdout: max training rows   
      -Cuts=8          tree: splits to try, per num   
      -Leaf=4          tree: min rows in any leaf

knows nothing of the ones below.  Cut on the banners to get
five files back:

      acquire  spend a label budget.

type QTY    = int | float             # any number
type ATOM   = QTY | str               # one cell; "?" = unknown
type ROW    = list[ATOM]
type MIDS   = dict[int, ATOM]         # a centroid

#-- misc ---------------------------------------------------
def say(x: Any, p: int = 2) -> str:
  "Floats get P decimals; dicts list their keys."
  if isinstance(x,float): x= f"{x:.{p}f}".rstrip("0").rstrip(".")
  if isinstance(x, dict):
    x= "{"+", ".join(f"{k}: {say(v,p)}" for k,v in x.items())+"}"
  return str(x)

class o(dict):
  "A dict you can poke with a dot."
  __getattr__, __setattr__ = dict.__getitem__, dict.__setitem__
  __repr__ = say

def atom(s: str) -> ATOM:
  "'22' -> 22. '3.1' -> 3.1. 'True' -> True. 'x' -> 'x'."
  for fn in (int, float):
    try: return fn(s)
    except ValueError: pass
  s = s.strip()
  return {"True": True, "False": False}.get(s, s)

def csv(file: str) -> Iterator[ROW]:
  "ROWS of FILE, each cell coerced."
  with open(os.path.expanduser(file),  # -sig drops any BOM
           encoding="utf-8-sig") as f:
    for line in f:
      if line.strip(): yield [atom(s) for s in line.split(",")]

def Col(txt: str = " ", at: int = 0) -> COL:
  "Factory that returns NUM or SYM."
  return Num(txt,at) if txt[0].isupper() else Sym(txt,at)

class Num:
  "Place to summarize stream of numbers."
  __slots__ = ("at", "txt", "n", "mu", "m2", "sd", "goal")
  def __init__(self, txt: str = " ", at: int = 0) -> None:
    self.at, self.txt, self.n = at, txt, 0
    self.mu = self.m2 = self.sd = 0
    self.goal = 0 if txt[-1] == "-" else 1
  __repr__ = lambda self: say({k: getattr(self,k)
                               for k in self.__slots__})

class Sym:
  "Place to summarize stream of Symbols.  The class is the tag."
  __slots__ = ("at", "txt", "n", "has")
  def __init__(self, txt: str = " ", at: int = 0) -> None:
    self.at, self.txt, self.n, self.has = at, txt, 0, {}
  __repr__ = lambda self: say({k: getattr(self,k)
                               for k in self.__slots__})

def add(col: COL, v: ATOM, inc: int = 1) -> None:
  "Show V to COL.  INC=-1 takes it away.  `?` changes nothing."
  if v != "?":
    col.n += inc
    if type(col) is Sym:
      col.has[v] = col.has.get(v, 0) + inc
    else:
      d = v - col.mu                      # Welford, for `div`
      col.mu += inc * d / max(1, col.n)
      col.m2  = max(0, col.m2 + inc * d * (v - col.mu))
      col.sd  = 0 if col.n < 2 else (col.m2/(col.n-1))**.5

def mid(col: COL) -> ATOM:
  "Middle: the mean, or the most common symbol; ? if none."
  return (max(col.has, key=col.has.get) if col.has else "?"
          ) if type(col) is Sym else col.mu

def mids(tbl: TBL) -> MIDS:
  "Every column's middle, cached until the next addCols."
  tbl.mids = tbl.mids or {at: mid(c) for at,c in tbl.cols.items()}
  return tbl.mids

def div(col: COL) -> float:
  "Spread: entropy, or standard deviation."
  return (-sum(v/col.n*log2(v/col.n) for v in col.has.values())
          if type(col) is Sym else col.sd)  # sd kept by `add`

def norm(col: COL, v: ATOM) -> ATOM:
  "To 0..1, by the logistic curve.  Symbols do not scale."
  if type(col) is Sym: return v
  z = max(-3, min(3, (v - col.mu) / (1e-32 + div(col))))
  return 1 / (1 + exp(-1.7 * z))

#-- tables -------------------------------------------------
def Tbl(src: Iterable[ROW]) -> TBL:
  "Header names the columns: X skip, +- goal, ! klass."
  src = iter(src)
  tbl = Cols(o(rows=[], cols={}, x=[], y=[], mids=None,
               names=next(src)))
  for row in src: addRow(tbl, row)
  return tbl

def Cols(tbl: TBL) -> TBL:
  "Create column roles."
  for at, s in enumerate(tbl.names):
    if s[-1] == "X": continue            # skip me entirely
    tbl.cols[at] = Col(s,at)
    (tbl.y if s[-1] in "+-!" else tbl.x).append(tbl.cols[at])
  return tbl

def addRow(tbl: TBL, row: ROW) -> ROW:
  "Keep ROW, and show it to my columns."
  tbl.rows += [row]
  return addCols(tbl, row)

def addCols(tbl: TBL, row: ROW, inc: int = 1) -> ROW:
  "Show ROW to my columns, without keeping it."
  tbl.mids = None                 # a new row moves the middles
  for at, col in tbl.cols.items(): add(col, row[at], inc)
  return row

def clone(tbl: TBL, rows: ROWS = None) -> TBL:
  "An empty copy of TBL, plus ROWS."
  return Tbl([tbl.names] + (rows or []))

#-- distance -----------------------------------------------
def dist(vs: Iterable[QTY], n: int) -> float:
  "Root mean square of N gaps."
  return sqrt(sum(v*v for v in vs) / n)

def ydist(tbl: TBL, row: ROW) -> float:
  "How far ROW's goals are from the best they could be."
  return dist((abs(norm(c, row[c.at]) - c.goal)
                    for c in tbl.y), len(tbl.y))

def gap(col: COL, a: ATOM, b: ATOM) -> QTY:
  "Distance between two values of one column."
  if a == "?" or b == "?": return 1    # unknown = far
  return a!=b if type(col) is Sym else abs(norm(col,a)-norm(col,b))

def xdist(tbl: TBL, r1: ROW, r2: ROW | MIDS) -> float:
  "How far apart two rows are, over the x columns."
  return dist((gap(c, r1[c.at], r2[c.at])
                    for c in tbl.x), len(tbl.x))

def oracle(row: ROW) -> ROW:
  """Labeller. Rows here arrive with ys, so nothing to do.
  Real oracles fill in any missing y cells (and can close
  over a TBL to know which cells those are)."""
  return row

def acquire(tbl: TBL, cap: int | None = None,
            label: ORACLE = oracle) -> ROWS:
  "Label near the best rows, far from the rest."
  cap  = cap or the.Stop
  todo = random.sample(tbl.rows, len(tbl.rows))[:the.Few]
  both = clone(tbl, [label(todo.pop())
                     for _ in range(the.Start)])
  both.rows.sort(key=lambda r: ydist(both, r))
  n    = int(sqrt(the.Start))
  best,rest = clone(tbl,both.rows[:n]), clone(tbl,both.rows[n:])
  while todo and len(both.rows) < cap:
    cb, cr = mids(best), mids(rest)    # two centroids
    todo.sort(key=lambda z: xdist(tbl,z,cr) - xdist(tbl,z,cb))
    addRow(best, addRow(both, label(todo.pop())))
    best.rows.sort(key=lambda z: ydist(both, z))  # no y leak
    if len(best.rows) > sqrt(len(both.rows)): # best stays small
      addRow(rest, addCols(best, best.rows.pop(), -1))
  return both.rows

def grabs(tbl: TBL, cap: int | None = None,
          label: ORACLE = oracle) -> ROWS:
  "The straw man: same budget, taken at random."
  cap = cap or the.Stop
  return [label(r) for r in
          random.sample(tbl.rows, min(cap, len(tbl.rows)))]

# Calls core.  Nothing above this line calls down here.
# NODE = o(rows, at, v, go, kids, mu).
def cut(tbl: TBL, rows: ROWS) -> tuple[int,ATOM,GO]|None:
  "The (at, v, go) whose two sides have the tightest ys."
  ys = [ydist(tbl, r) for r in rows]
  out, least = None, 1e30
  for col in tbl.x:
    for v, go in candidates(col, rows):
      a, b = Num(), Num()   # one pass, so one `go` per row
      for r,y in zip(rows,ys): add(a if go(r) else b, y)
      if a.n >= the.Leaf and b.n >= the.Leaf:
        score = (div(a)*a.n + div(b)*b.n) / (a.n + b.n)
        if score < least: out, least = (col.at, v, go), score
  return out

def candidates(col: COL, rows: ROWS) -> Iterator[tuple[ATOM,GO]]:
  "What splits to try: one per symbol, or CUTS per num."
  at = col.at
  if type(col) is Sym:
    for v in sorted({r[at] for r in rows if r[at] != "?"}):
      yield v, lambda r,v=v,at=at: r[at] == v
  else:                # by rank, not value: THESE rows are dense
    xs = sorted(r[at] for r in rows if r[at] != "?")
    k  = max(1, len(xs)//the.Cuts)     # cuts, spaced by rank
    for v in xs[k-1:-1:k]:
      yield v, (lambda r,v=v,at=at: r[at] != "?" and r[at] <= v)

def tree(tbl: TBL, rows: ROWS) -> NODE:
  "Split while a split keeps both sides big enough."
  node = o(rows=rows, at=None, v=None, go=None, kids=[],
           mu=sum(ydist(tbl,r) for r in rows)/len(rows))
  if len(rows) > the.Leaf and (found := cut(tbl, rows)):
    at, v, go = found
    yes, no = [], []
    for r in rows: (yes if go(r) else no).append(r)
    node.at, node.v, node.go = at, v, go # cut kepts sides big
    node.kids = [tree(tbl, yes), tree(tbl, no)]
  return node

def leaf(node: NODE, row: ROW) -> NODE:
  "Walk ROW down to its leaf."
  while node.kids: node = node.kids[0 if node.go(row) else 1]
  return node

def leafs(node: NODE) -> list[NODE]:
  "Return leaves of this tree."
  return [x for k in node.kids for x in leafs(k)] or [node]

def show(tbl: TBL, node: NODE, pre: str | None = None,
         edge: str = "") -> None:
  "Tree on the left; ydist and n on the right."
  ls = sorted(leafs(node), key=lambda z: z.mu)
  def walk(z: NODE, pre: str | None, edge: str) -> None:
    m = "+" if z is ls[0] else "-" if z is ls[-1] else " "
    print(f"{m} {round(100*z.mu):>4} {len(z.rows):>5}   "
          f"{(pre or '')+edge}".rstrip())
    op = "=" if type(tbl.cols[z.at or 0]) is Sym else "<="
    no = "!=" if op == "=" else ">"
    for kid, oper in zip(z.kids, [op, no]):
      walk(kid, "" if pre is None else pre + "|  ",
           f"{tbl.names[z.at]} {oper} {say(z.v)}")
  print(" ydist     n")
  walk(node, pre, edge)



#-- rig ----------------------------------------------------
# Calls core, acquire, and one RANKER -- tree's, by default.
def ranker(lab: TBL) -> Callable[[ROW], float]:
  "Grow a tree on LAB; score any row by its leaf's mean ydist."
  node = tree(lab, lab.rows)
  return lambda r: leaf(node, r).mu

def wins(tbl: TBL) -> Callable[[ROW], float]:
  "100 at the best row, 0 at an average one."
  ys = sorted(ydist(tbl, r) for r in tbl.rows)
  lo, avg = ys[0], sum(ys)/len(ys)
  return lambda r: max(-100, min(100, 100*(1 - (ydist(tbl,r)-lo)
                                            / (avg-lo+1e-32))))

def holdout(tbl: TBL, pick: PICKER = acquire,
            rank: RANKER = ranker,
            label: ORACLE = oracle) -> ROW:
  "Train on half; of CHECK guesses on the rest, pick best."
  rows  = random.sample(tbl.rows, len(tbl.rows))
  n     = len(rows)//2
  tr    = clone(tbl, rows[:n][:the.Few])
  lab   = clone(tbl, pick(tr, the.Stop - the.Check, label))
  top   = sorted(rows[n:], key=rank(lab))
  return min(top[:the.Check], key=lambda r: ydist(lab, label(r)))

# A leaf: it calls core, and nothing here calls it.
def adds(vs: Iterable[ATOM], col: COL | None = None) -> COL:
  "Every item of VS into COL."
  col = Num() if col is None else col
  for v in vs: add(col, v)
  return col

def cohen(xs: NUMS, ys: NUMS, d: float = 0.35,
          eps: float = 0) -> bool:
  "Are the means within D pooled standard deviations?"
  a, b = adds(xs), adds(ys)
  pool = ((a.n-1)*div(a)**2 + (b.n-1)*div(b)**2)/(a.n+b.n-2)
  return abs(a.mu - b.mu) <= max(eps, d * sqrt(pool))

def cliffs(xs: NUMS, ys: NUMS, d: float = 0.197) -> bool:
  "Sorted in.  Is the rank imbalance small enough?"
  gt = lt = j = k = 0
  for x in xs:
    while j < len(ys) and ys[j] <  x: j += 1; k = j
    while k < len(ys) and ys[k] <= x: k += 1
    gt += j; lt += len(ys) - k
  return abs(gt - lt) / (len(xs) * len(ys)) <= d

def ks(xs: NUMS, ys: NUMS, a: float = 1.36) -> bool:
  "Sorted in.  95% Kolmogorov-Smirnov."
  n, m, i, j, d = len(xs), len(ys), 0, 0, 0
  while i < n and j < m:
    v = min(xs[i], ys[j])
    while i < n and xs[i] <= v: i += 1
    while j < m and ys[j] <= v: j += 1
    d = max(d, abs(i/n - j/m))
  return d <= a * sqrt((n + m) / (n * m))

def same(xs: NUMS, ys: NUMS, eps: float = 0) -> bool:
  """Indistinguishable.  Always cliffs and ks; EPS adds cohen, and
  doubles as its effect size (eps=0 skips that third test)."""
  xs, ys = sorted(xs), sorted(ys)
  return (cliffs(xs,ys) and ks(xs,ys)
          and (not eps or cohen(xs,ys,d=eps)))

import ezr

def remember(header, rows):
    """
    Identify the top 5% of rows based on multi-objective optimization.
    Uses ydist to measure distance from the ideal goal values.
    """
    # Create a table object using the ezr library
    # The first row of the data is the header, so we combine it with the rows
    data = [header] + rows
    tbl = ezr.Tbl(data)
    
    # Calculate ydist for all rows
    # ydist measures how far a row is from the ideal (0.0 for all goals)
    scored = []
    for i, row in enumerate(rows):
        d = ezr.ydist(tbl, row)
        scored.append((d, i))
    
    # Sort by distance (ascending, as lower ydist is better)
    scored.sort(key=lambda x: x[0])
    
    # Return the indices of the top 5%
    limit = max(1, len(rows) // 20)
    return [item[1] for item in scored[:limit]]

import ezr
import random

def guess(header, train, test):
    """
    Estimates the first goal column for each test row by finding the 
    nearest neighbor in the training set based on x-column distance.
    """
    # Create a table object for the training data
    data = [header] + train
    tbl = ezr.Tbl(data)
    
    # Identify the index of the first goal column
    # The first goal column is the first one ending in '-' or '+'
    goal_idx = -1
    for i, name in enumerate(header):
        if name[-1] in "+-":
            goal_idx = i
            break
            
    results = []
    for t_row in test:
        # Find the row in training data that is closest in x-space
        # We use xdist from ezr which compares only input columns
        best_row = min(train, key=lambda r: ezr.xdist(tbl, t_row, r))
        results.append(best_row[goal_idx])
        
    return results

if __name__ == "__main__":
    filename = "Health-ClosedIssues0000.n1000.train.csv"
    data_gen = ezr.csv(filename)
    
    try:
        header = next(data_gen)
        all_rows = list(data_gen)
    except StopIteration:
        exit(0)
        
    # Split data: 80% train, 20% test
    random.seed(1234567891)
    random.shuffle(all_rows)
    split = int(len(all_rows) * 0.8)
    train = all_rows[:split]
    test_raw = all_rows[split:]
    
    # Mask the goals in the test set to simulate "unknown"
    test = []
    for row in test_raw:
        new_row = list(row)
        for i, name in enumerate(header):
            if name[-1] in "+-":
                new_row[i] = "?"
        test.append(new_row)
        
    # Get estimates
    estimates = guess(header, train, test)
    
    # Calculate average error for the test set (first goal column)
    goal_idx = next(i for i, name in enumerate(header) if name[-1] in "+-")
    actuals = [row[goal_idx] for row in test_raw]
    
    errors = [abs(est - act) for est, act in zip(estimates, actuals)]
    avg_error = sum(errors) / len(errors)
    
    print(f"Forecast analysis complete.")
    print(f"Estimated the first goal column for {len(test)} test cases.")
    print(f"Average absolute error of forecasts: {avg_error:.4f}")
    
    exit(0)

import ezr

def certify(xs, ys):
    """
    Returns True if the two distributions are statistically indistinguishable,
    False if they are significantly different.
    """
    # We use the same logic as the stats layer in ezr.py
    # same() checks cliffs delta and KS test, which are robust for this purpose.
    return ezr.same(xs, ys)

def run_analysis():
    filename = "Health-ClosedIssues0000.n1000.train.csv"
    data_gen = ezr.csv(filename)
    
    try:
        header = next(data_gen)
        rows = list(data_gen)
    except StopIteration:
        return

    # Define "big" vs "small" based on N_estimators (index 3)
    # We split by the median of N_estimators
    n_est_idx = 3
    n_estimators = [row[n_est_idx] for row in rows]
    median = sorted(n_estimators)[len(n_estimators) // 2]
    
    # We use ACC+ (index 1) as the measure of accuracy
    acc_idx = 1
    small_setups = [row[acc_idx] for row in rows if row[n_est_idx] <= median]
    big_setups = [row[acc_idx] for row in rows if row[n_est_idx] > median]
    
    # Certify if they are the same
    is_same = certify(small_setups, big_setups)
    
    print("Analysis: Comparing accuracy (ACC+) of small vs big setups.")
    if is_same:
        print("Result: The difference in accuracy is statistically insignificant (noise).")
    else:
        print("Result: There is a statistically significant difference in accuracy.")
    
    print(f"Small setups mean: {sum(small_setups)/len(small_setups):.4f}")

import ezr

def blame(header, rows):
    """
    Identify input columns that best explain why the worst rows are bad.
    We compare the distribution of input values in the worst 25% of rows
    against the rest of the data. Columns with the largest difference
    in distribution (using entropy for symbols or mean difference for numbers)
    are identified as the culprits.
    """
    # 1. Setup table to use ezr's ydist
    data = [header] + rows
    tbl = ezr.Tbl(data)
    
    # 2. Score all rows and find the worst 25%
    scored = sorted([(ezr.ydist(tbl, r), r) for r in rows], key=lambda x: x[0], reverse=True)
    worst_n = max(1, len(rows) // 4)
    worst_rows = [x[1] for x in scored[:worst_n]]
    other_rows = [x[1] for x in scored[worst_n:]]
    
    # 3. Evaluate each input column
    # We look for columns where the distribution in 'worst' is significantly different from 'others'
    scores = []
    for col in tbl.x:
        at = col.at
        if isinstance(col, ezr.Sym):
            # For symbols, compare entropy or frequency shift
            # Simple heuristic: difference in mode frequency
            w_vals = [r[at] for r in worst_rows]
            o_vals = [r[at] for r in other_rows]
            w_mode = max(set(w_vals), key=w_vals.count)
            o_mode = max(set(o_vals), key=o_vals.count)
            diff = 1.0 if w_mode != o_mode else 0.0
        else:
            # For numbers, compare normalized mean difference
            w_vals = [r[at] for r in worst_rows if r[at] != "?"]
            o_vals = [r[at] for r in other_rows if r[at] != "?"]
            if not w_vals or not o_vals: continue
            w_mu = sum(w_vals) / len(w_vals)
            o_mu = sum(o_vals) / len(o_vals)
            # Normalize by range to make columns comparable
            diff = abs(w_mu - o_mu)
            
        scores.append((header[at], diff))
    
    # 4. Return top 3 columns that contribute most to the 'badness'
    scores.sort(key=lambda x: x[1], reverse=True)
    return [name for name, score in scores[:3]]

import ezr

def justify(header, rows, row):
    """
    Returns (p, reason): p is the fraction of rows that `row` is better than.
    The reason explains the performance relative to the dataset.
    """
    # Create a table to handle column roles and distance calculations
    data = [header] + rows
    tbl = ezr.Tbl(data)
    
    # To estimate the goals of the target row, we find its nearest neighbor
    # in the training data (x-space distance)
    best_neighbor = min(rows, key=lambda r: ezr.xdist(tbl, row, r))
    
    # Extract the goal values from the neighbor as an estimate for the target row
    # A row is better if its ydist is lower (closer to ideal)
    # We use the neighbor's ydist as a proxy for the target row's performance
    target_ydist = ezr.ydist(tbl, best_neighbor)
    
    # Calculate ydist for all rows in the dataset
    all_ydists = [ezr.ydist(tbl, r) for r in rows]
    
    # Calculate p: fraction of rows that have a higher (worse) ydist than our target
    better_than_count = sum(1 for d in all_ydists if d > target_ydist)
    p = better_than_count / len(rows)
    
    reason = (f"This configuration is estimated to outperform {p:.1%} of all "
              "possible setups based on its similarity to high-performing historical data.")
    
    return p, reason

import ezr

def choose(header, rows, label):
    """
    Find the best setup by using the acquire strategy from ezr.
    We have a budget of 50 labels.
    """
    # Create a table object to manage column roles and distance metrics
    # We initialize with the header and rows
    data = [header] + rows
    tbl = ezr.Tbl(data)
    
    # The acquire function in ezr.py implements a smart sampling strategy
    # that balances exploration and exploitation to find the best rows
    # within a given budget (the.Stop).
    # We pass a wrapper to the label function to handle the indexing.
    def oracle_wrapper(row):
        # Find the index of the row in the original list to call label()
        # Note: This assumes rows are unique or we just need any index
        idx = rows.index(row)
        labeled_row = label(idx)
        # Update the row in-place with the true goal values
        for i, val in enumerate(labeled_row):
            row[i] = val
        return row

    # Use the acquire strategy to find the best rows
    # ezr.the.Stop is set to 50 by default in ezr.py
    best_rows = ezr.acquire(tbl, cap=50, label=oracle_wrapper)
    
    # Sort the labeled rows by their ydist (distance to ideal)
    # Lower ydist is better
    best_rows.sort(key=lambda r: ezr.ydist(tbl, r))
    
    # Find the original index of the best row found
    best_row = best_rows[0]
    return rows.index(best_row)

import ezr
import random

def fix(header, row, levers):
    """
    Find the smallest change to a 'bad' row that improves its ydist 
    without violating the constraints defined in 'levers'.
    """
    # Build a table to calculate ydist
    # We need the full dataset to establish the context for ydist normalization
    # Since we don't have the full dataset passed in, we assume the caller
    # expects us to use the global context or a minimal table.
    # To ensure ydist works, we need the column definitions.
    tbl = ezr.Tbl([header])
    
    # Re-populate the table with the row to ensure column stats are initialized
    # (Though for a single row, ydist will be relative to the column's own stats)
    ezr.addRow(tbl, row)
    
    current_ydist = ezr.ydist(tbl, row)
    best_change = {}
    min_dist = float('inf')
    
    for col_name, allowed in levers.items():
        if col_name not in header:
            continue
        col_idx = header.index(col_name)
        original_val = row[col_idx]
        
        candidates = []
        if isinstance(allowed, list):  # Symbolic
            candidates = [v for v in allowed if v != original_val]
        else:  # Numeric (low, high)
            low, high = allowed
            # Try a few points in the range
            candidates = [low, (low + high) / 2, high]
            candidates = [c for c in candidates if c != original_val]
            
        for val in candidates:
            new_row = list(row)
            new_row[col_idx] = val
            
            # Check if this change improves ydist
            new_ydist = ezr.ydist(tbl, new_row)
            
            if new_ydist < current_ydist:
                # We want the smallest change. 
                # For numeric, distance is abs(new-old). For symbolic, 1.
                dist = abs(val - original_val) if isinstance(val, (int, float)) else 1
                if dist < min_dist:
                    min_dist = dist
                    best_change = {col_name: val}
                    
    return best_change

import ezr

def spot(header, rows):
    """
    Identify suspicious rows (outliers) that might represent failed runs.
    We use the xdist (input distance) to find rows that are far from the 
    centroid of the input space, as these are statistically anomalous.
    """
    # Create a table to handle column roles and distance metrics
    data = [header] + rows
    tbl = ezr.Tbl(data)
    
    # Calculate the centroid (mids) of the input space
    # The mids function caches the median/mean of each column
    centroid = ezr.mids(tbl)
    
    # Calculate the distance of each row from the centroid
    # A high xdist indicates the row has unusual input settings
    scores = []
    for i, row in enumerate(rows):
        d = ezr.xdist(tbl, row, centroid)
        scores.append((d, i))
    
    # Sort by distance descending (most suspicious first)
    scores.sort(key=lambda x: x[0], reverse=True)
    
    # Return the indices of the top 5% most suspicious rows
    limit = max(1, len(rows) // 20)
    return [item[1] for item in scores[:limit]]

import ezr
import random

def race(header, rows, label):
    """
    Compare three strategies for finding the best setup within a 50-label budget.
    Strategies:
    1. random: Randomly samples rows.
    2. acquire: Uses ezr's smart sampling (exploration/exploitation).
    3. greedy: Picks the next row based on the best-performing leaf in a tree.
    """
    budget = 50
    labeled_indices = set()
    
    # We need a reference table that has the full data to calculate ydist correctly
    # The harness passes rows with '?' in goal columns. We must label them to evaluate.
    # We create a local cache for labeled rows to avoid re-labelling.
    cache = {}

    def oracle_wrapper(idx):
        if idx not in labeled_indices:
            if len(labeled_indices) >= budget:
                raise Exception("Budget exceeded")
            labeled_indices.add(idx)
            cache[idx] = label(idx)
        return cache[idx]

    # Create a base table for distance calculations
    # We use the provided rows, but we must ensure the table knows the goals
    # are numeric for ydist to work.
    tbl = ezr.Tbl([header] + [r for r in rows if "?" not in r])
    
    # 1. Random Strategy
    def strategy_random():
        indices = random.sample(range(len(rows)), budget)
        for i in indices:
            oracle_wrapper(i)
        return min(indices, key=lambda i: ezr.ydist(tbl, cache[i]))

    # 2. Acquire Strategy
    def strategy_acquire():
        # Acquire needs a TBL with data. We use a subset of the budget to start.
        # We create a temporary table for the acquire logic.
        work_tbl = ezr.Tbl([header] + [rows[i] for i in random.sample(range(len(rows)), ezr.the.Start)])
        for i in work_tbl.rows:
            idx = rows.index(i)
            oracle_wrapper(idx)
            
        def wrapper(row):
            idx = rows.index(row)
            return oracle_wrapper(idx)
        
        best_rows = ezr.acquire(work_tbl, cap=budget, label=wrapper)
        return min(range(len(rows)), key=lambda i: ezr.ydist(tbl, cache.get(i, rows[i])))

    # 3. Greedy Strategy
    def strategy_greedy():
        start_indices = random.sample(range(len(rows)), ezr.the.Start)
        for i in start_indices: oracle_wrapper(i)
        
        while len(labeled_indices) < budget:
            labeled_rows = [cache[i] for i in labeled_indices]
            lab_tbl = ezr.Tbl([header] + labeled_rows)
            rank = ezr.ranker(lab_tbl)
            
            unlabeled = [i for i in range(len(rows)) if i not in labeled_indices]
            best_idx = min(unlabeled, key=lambda i: rank(rows[i]))
            oracle_wrapper(best_idx)
            
        return min(labeled_indices, key=lambda i: ezr.ydist(tbl, cache[i]))

    results = {}
    for name, func in [("random", strategy_random), 
                       ("acquire", strategy_acquire), 
                       ("greedy", strategy_greedy)]:
        labeled_indices.clear()
        cache.clear()
        results[name] = func()

    winner = min(results, key=lambda n: ezr.ydist(tbl, cache[results[n]]))
    
    return {
        "winner": winner,
        "row": results[winner],
        "tried": ["random", "acquire", "greedy"]
    }

import ezr

def certify(xs, ys):
    """
    Returns True if the two distributions are statistically indistinguishable,
    False if they are significantly different.
    """
    # Using the same logic as the stats layer in ezr.py
    # same() checks cliffs delta and KS test, which are robust for this purpose.
    return ezr.same(xs, ys)

def compare_rules(header, rows):
    """
    Compares two rules for selecting the 'best' setup:
    Rule 1: Minimize MRE- (the first goal column)
    Rule 2: Maximize ACC+ (the second goal column)
    """
    # Identify goal indices
    # MRE- is index 0, ACC+ is index 1
    mre_idx = 0
    acc_idx = 1
    
    # We evaluate the performance of the top 10% of rows selected by each rule
    # Rule 1: Sort by MRE- (ascending)
    rows_by_mre = sorted(rows, key=lambda r: r[mre_idx])
    # Rule 2: Sort by ACC+ (descending)
    rows_by_acc = sorted(rows, key=lambda r: r[acc_idx], reverse=True)

    # We compare the ydist (overall multi-objective performance) of these two sets
    tbl = ezr.Tbl([header] + rows)
    
    xs = [ezr.ydist(tbl, r) for r in rows_by_mre[:top_n]]
    ys = [ezr.ydist(tbl, r) for r in rows_by_acc[:top_n]]
    
    return certify(xs, ys), xs, ys

import ezr

def blame(header, rows):
    """
    Identify input columns that have the least impact on performance.
    We compare the distribution of input values in the worst 25% of rows
    against the rest of the data. Columns with the smallest difference
    in distribution are identified as 'safe' to leave at default.
    """
    # 1. Setup table to use ezr's ydist
    data = [header] + rows
    tbl = ezr.Tbl(data)
    
    # 2. Score all rows and find the worst 25%
    scored = sorted([(ezr.ydist(tbl, r), r) for r in rows], key=lambda x: x[0], reverse=True)
    worst_n = max(1, len(rows) // 4)
    worst_rows = [x[1] for x in scored[:worst_n]]
    other_rows = [x[1] for x in scored[worst_n:]]
    
    # 3. Evaluate each input column
    # We look for columns where the distribution in 'worst' is most similar to 'others'
    scores = []
    for col in tbl.x:
        at = col.at
        if isinstance(col, ezr.Sym):
            # For symbols, compare frequency distribution
            w_vals = [r[at] for r in worst_rows]
            o_vals = [r[at] for r in other_rows]
            # Use simple overlap of frequencies
            diff = 0
            for val in set(w_vals + o_vals):
                w_freq = w_vals.count(val) / len(w_vals)
                o_freq = o_vals.count(val) / len(o_vals)
                diff += abs(w_freq - o_freq)
        else:
            # For numbers, compare normalized mean difference
            w_vals = [r[at] for r in worst_rows if r[at] != "?"]
            o_vals = [r[at] for r in other_rows if r[at] != "?"]
            if not w_vals or not o_vals: 
                diff = 1.0 # Assume high impact if data missing
            else:
                w_mu = sum(w_vals) / len(w_vals)
                o_mu = sum(o_vals) / len(o_vals)
                # Normalize by range to make columns comparable
                diff = abs(w_mu - o_mu)
            
        scores.append((header[at], diff))
    
    # 4. Return columns with the smallest impact (lowest diff)
    scores.sort(key=lambda x: x[1])
    return [name for name, score in scores]

import ezr
import random

def justify(header, rows, row):
    """
    Returns (p, reason): p is the fraction of rows that `row` is better than.
    The reason explains why 'bigger/deeper' is not always better by 
    comparing the target row's performance to the overall distribution.
    """
    # Create a table to handle column roles and distance calculations
    data = [header] + rows
    tbl = ezr.Tbl(data)
    
    # Estimate the target row's performance by finding its nearest neighbor
    # in the training data (x-space distance).
    # We use xdist to find the most similar configuration.
    best_neighbor = min(rows, key=lambda r: ezr.xdist(tbl, row, r))
    
    # The target row's performance is estimated by its neighbor's ydist
    target_ydist = ezr.ydist(tbl, best_neighbor)
    
    # Calculate ydist for all rows in the dataset to build a distribution
    all_ydists = [ezr.ydist(tbl, r) for r in rows]
    
    # Calculate p: fraction of rows that have a higher (worse) ydist than our target
    # A row is better if its ydist is lower.
    better_than_count = sum(1 for d in all_ydists if d > target_ydist)
    p = better_than_count / len(rows)
    
    # Construct a reason that addresses the "bigger/deeper" question
    reason = (f"This configuration is estimated to outperform {p:.2f} of all "
              "possible setups; simply choosing the largest or deepest model "
              "often leads to overfitting or unnecessary complexity without "
              "guaranteeing better performance.")
    
    return p, reason

import ezr
import random

def choose(header, rows, label):
    """
    Finds a setup that is both accurate (low ydist) and small (low N_estimators).
    We use a multi-objective approach: we want to minimize ydist AND 
    minimize N_estimators (index 3).
    """
    # 1. Setup
    # We use a budget of 50 labels.
    budget = 50
    labeled = {}
    
    def get_label(i):
        if i not in labeled:
            if len(labeled) >= budget:
                return None
            labeled[i] = label(i)
        return labeled[i]

    # 2. Strategy: Active Learning
    # Initial random sample to seed the model
    indices = random.sample(range(len(rows)), min(len(rows), ezr.the.Start))
    for i in indices:
        get_label(i)
        
    # Iterative search
    while len(labeled) < budget:
        # Build a table using only the rows we have labeled
        # We must ensure the rows passed to Tbl match the header length
        data = [header] + [labeled[i] for i in labeled]
        tbl = ezr.Tbl(data)
        
        # Ranker uses the tree to predict ydist
        rank = ezr.ranker(tbl)
        
        # Score unlabeled rows: 
        # We want low ydist (rank) and low N_estimators (index 3)
        # N_estimators is at index 3.
        def score(i):
            r = rows[i]
            # Normalize N_estimators (approx range 50-200)
            n_est_norm = (r[3] - 50) / 150
            return rank(r) + n_est_norm
            
        unlabeled = [i for i in range(len(rows)) if i not in labeled]
        if not unlabeled:
            break
        best_i = min(unlabeled, key=score)
        get_label(best_i)
        
    # 3. Final Selection
    # Return the index of the best row found based on the combined score
    final_data = [header] + [labeled[i] for i in labeled]
    final_tbl = ezr.Tbl(final_data)
    
    best_idx = min(labeled.keys(), key=lambda i: ezr.ydist(final_tbl, labeled[i]) + (rows[i][3]/200))
    return best_idx

import ezr

def fix(header, row, levers):
    """
    Find the single best change to a 'bad' row that improves its ydist 
    the most, staying within the allowed levers.
    """
    # Create a table to establish the context for ydist normalization
    # We use the header to define the column roles
    tbl = ezr.Tbl([header])
    
    # We need to ensure the column stats are initialized for ydist to work.
    # Since we don't have the full dataset, we rely on the fact that 
    # ezr.ydist uses column goals and normalization.
    
    current_ydist = ezr.ydist(tbl, row)
    best_change = {}
    max_improvement = -float('inf')
    
    for col_name, allowed in levers.items():
        if col_name not in header:
            continue
        col_idx = header.index(col_name)
        original_val = row[col_idx]
        
        candidates = []
        if isinstance(allowed, list):  # Symbolic
            candidates = [v for v in allowed if v != original_val]
        else:  # Numeric (low, high)
            low, high = allowed
            # Try a few points in the range to find the best improvement
            candidates = [low, (low + high) / 2, high]
            candidates = [c for c in candidates if c != original_val]
            
        for val in candidates:
            new_row = list(row)
            new_row[col_idx] = val
            
            # Check how much this change improves ydist
            new_ydist = ezr.ydist(tbl, new_row)
            improvement = current_ydist - new_ydist
            
            if improvement > max_improvement:
                max_improvement = improvement
                best_change = {col_name: val}
                    
    return best_change

def spot(header, rows):
    """
    Identify rows containing input values that are 'suspicious' because they
    are far from the observed distributions in the training data.
    We use the xdist (input distance) to find rows that are outliers in the 
    input space, as these represent settings the model has never seen.
    """

    data = [header] + rows
    tbl = ezr.Tbl(data)
    
    # Calculate the centroid (mids) of the input space
    # The mids function caches the median/mean of each column
    centroid = ezr.mids(tbl)

    # A high xdist indicates the row has unusual input settings
    scores = []

        # Calculate distance from the centroid of the input space

    scores.sort(key=lambda x: x[0], reverse=True)
    
    # Return the indices of the top 5% most suspicious rows
