import ezr
import random

def race(header, rows, label):
    """
    Compare three strategies for finding the best setup within a 50-label budget.
    Strategies:
    1. random: Randomly samples rows.
    2. acquire: Uses ezr's smart sampling.
    3. greedy: Picks the next row based on the best-performing leaf in a tree.
    """
    budget = 50
    labeled_indices = {}
    
    def oracle_wrapper(idx):
        if idx not in labeled_indices:
            if len(labeled_indices) >= budget:
                raise Exception("Budget exceeded")
            labeled_indices[idx] = label(idx)
        return labeled_indices[idx]

    # We need a reference table to evaluate ydist. 
    # Since we don't have all labels, we build a table from the initial random sample.
    # We use the full header to ensure column indices are correct.
    initial_indices = random.sample(range(len(rows)), min(len(rows), 10))
    for i in initial_indices:
        oracle_wrapper(i)
    
    tbl = ezr.Tbl([header] + [labeled_indices[i] for i in labeled_indices])

    def strategy_random():
        # Randomly pick from remaining budget
        indices = random.sample([i for i in range(len(rows)) if i not in labeled_indices], 10)
        for i in indices:
            oracle_wrapper(i)
        return min(labeled_indices.keys(), key=lambda i: ezr.ydist(tbl, labeled_indices[i]))

    def strategy_acquire():
        # Use ezr's acquire logic
        # We need a table of the current labeled rows
        work_tbl = ezr.Tbl([header] + [labeled_indices[i] for i in labeled_indices])
        
        def wrapper(row):
            # Find index of this row in the original list
            idx = rows.index(row)
            return oracle_wrapper(idx)
        
        # Acquire returns rows, we need to label them
        new_rows = ezr.acquire(work_tbl, cap=len(labeled_indices) + 10, label=wrapper)
        return min(labeled_indices.keys(), key=lambda i: ezr.ydist(tbl, labeled_indices[i]))

    def strategy_greedy():
        # Use ranker to pick next best
        for _ in range(10):
            lab_tbl = ezr.Tbl([header] + [labeled_indices[i] for i in labeled_indices])
            rank = ezr.ranker(lab_tbl)
            unlabeled = [i for i in range(len(rows)) if i not in labeled_indices]
            if not unlabeled: break
            best_idx = min(unlabeled, key=lambda i: rank(rows[i]))
            oracle_wrapper(best_idx)
        return min(labeled_indices.keys(), key=lambda i: ezr.ydist(tbl, labeled_indices[i]))

    results = {}
    strategies = [("random", strategy_random), ("acquire", strategy_acquire), ("greedy", strategy_greedy)]
    
    for name, func in strategies:
        results[name] = func()

    winner = min(results, key=lambda n: ezr.ydist(tbl, labeled_indices[results[n]]))
    
    return {
        "winner": winner,
        "row": results[winner],
        "tried": [s[0] for s in strategies]
    }

if __name__ == "__main__":
    filename = "Health-ClosedIssues0000.n1000.train.csv"
    data_gen = ezr.csv(filename)
    try:
        header = next(data_gen)
        all_rows = list(data_gen)
    except StopIteration:
        exit(0)

    masked_rows = []
    for row in all_rows:
        new_row = list(row)
        for i, name in enumerate(header):
            if name[-1] in "+-":
                new_row[i] = "?"
        masked_rows.append(new_row)

    def label(i):
        return all_rows[i]

    result = race(header, masked_rows, label)
    print(f"Optimization Race Results:")
    print(f"Winner: {result['winner']}")
    print(f"Best row found at index: {result['row']}")
    print(f"Strategies tried: {', '.join(result['tried'])}")
    exit(0)
