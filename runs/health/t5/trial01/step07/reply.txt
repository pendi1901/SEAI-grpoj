import ezr
import random

def choose(header, rows, label):
    """
    Find the best setup by using the acquire strategy from ezr.
    We have a budget of 50 labels.
    """
    # Create a table object to manage column roles and distance metrics
    # We initialize with the header and rows
    data = [header] + rows
    tbl = ezr.Tbl(data)
    
    # The acquire function in ezr.py implements a smart sampling strategy
    # that balances exploration and exploitation to find the best rows
    # within a given budget (the.Stop).
    # We pass a wrapper to the label function to handle the indexing.
    def oracle_wrapper(row):
        # Find the index of the row in the original list to call label()
        # Note: This assumes rows are unique or we just need any index
        idx = rows.index(row)
        labeled_row = label(idx)
        # Update the row in-place with the true goal values
        for i, val in enumerate(labeled_row):
            row[i] = val
        return row

    # Use the acquire strategy to find the best rows
    # ezr.the.Stop is set to 50 by default in ezr.py
    best_rows = ezr.acquire(tbl, cap=50, label=oracle_wrapper)
    
    # Sort the labeled rows by their ydist (distance to ideal)
    # Lower ydist is better
    best_rows.sort(key=lambda r: ezr.ydist(tbl, r))
    
    # Find the original index of the best row found
    best_row = best_rows[0]
    return rows.index(best_row)

if __name__ == "__main__":
    filename = "Health-ClosedIssues0000.n1000.train.csv"
    data_gen = ezr.csv(filename)
    
    try:
        header = next(data_gen)
        all_rows = list(data_gen)
    except StopIteration:
        exit(0)
        
    # Create a copy of rows with goals masked as "?"
    masked_rows = []
    for row in all_rows:
        new_row = list(row)
        for i, name in enumerate(header):
            if name[-1] in "+-":
                new_row[i] = "?"
        masked_rows.append(new_row)
        
    # Define the label function for the harness
    def label(i):
        return all_rows[i]
        
    # Run the selection
    best_idx = choose(header, masked_rows, label)
    
    # Print the result
    print(f"Optimization complete.")
    print(f"The best setup found is at index {best_idx}.")
    print(f"Setup details: {all_rows[best_idx]}")
    
    exit(0)
