import ezr


def blame(header, rows):
    """Return input column names that best explain why worst rows are bad."""
    tbl = ezr.Tbl(zip(header, rows))

    # Build a tree on all labelled rows; leaf mu is average ydist.
    node = ezr.tree(tbl, tbl.rows)

    # Predictions (ydist) for every row.
    preds = [ezr.leaf(node, r).mu for r in rows]

    # Worst rows: highest ydist (farthest from the best possible goals).
    n = len(rows)
    worst_idx = sorted(range(n), key=lambda i: preds[i], reverse=True)[:max(1, n // 5)]

    # For each input column, see how its values differ between worst and rest.
    inputs = tbl.x
    result = []

    for col in inputs:
        if isinstance(col, ezr.Num):
            vals = [float(rows[i][col.at]) for i in range(n)]
            avg_worst = sum(vals[i] for i in worst_idx) / len(worst_idx)
            rest_vals = [vals[i] for i in range(n) if i not in worst_idx]
            avg_rest = sum(rest_vals) / len(rest_vals) if rest_vals else 0
            diff = avg_worst - avg_rest
            result.append(f"{col.txt}: {diff:+.2f}")

        elif isinstance(col, ezr.Sym):
            # Most common value in worst rows vs the rest.
            worst_val = max(
                set(rows[i][col.at] for i in worst_idx),
                key=lambda v: sum(1 for j in worst_idx if rows[j][col.at] == v)
            )
            rest_val = max(
                set(rows[i][col.at] for i in range(n) if i not in worst_idx),
                key=lambda v: sum(1 for j in range(n) if j not in worst_idx and rows[j][col.at] == v)
            )
            result.append(f"{col.txt}: {worst_val} (worst) vs {rest_val} (rest)")

    return result


if __name__ == "__main__":
    import os
    path = os.path.join(os.path.dirname(__file__) or ".", "Health-ClosedIssues0000.train.csv")
    with open(path, encoding="utf-8-sig") as f:
        lines = [l.strip() for l in f if l.strip()]
    header = lines[0].split(",")
    rows = [[ezr.atom(s) for s in line.split(",")] for line in lines[1:]]

    result = blame(header, rows)
    print(f"Bad rows are explained by: {result}")
