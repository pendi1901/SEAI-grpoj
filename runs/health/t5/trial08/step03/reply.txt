import ezr

def certify(xs, ys):
    """
    Returns True if the two distributions are statistically indistinguishable 
    (noise), False if they are significantly different.
    """
    return ezr.same(xs, ys, eps=0.35)

if __name__ == "__main__":
    # Load the data
    filename = "Health-ClosedIssues0000.n1000.train.csv"
    data_iter = ezr.csv(filename)
    header = next(data_iter)
    rows = list(data_iter)
    
    # Create table to access column metadata
    tbl = ezr.Tbl([header] + rows)
    
    # Identify 'N_estimators' column index (the proxy for "big" vs "small" setup)
    # Based on the header: ['MRE-', 'ACC+', 'PRED40+', 'N_estimators', ...]
    # N_estimators is at index 3
    at = 3
    
    # Split data into "small" (below median) and "big" (above median)
    vals = [r[at] for r in rows if r[at] != "?"]
    median = sorted(vals)[len(vals) // 2]
    
    small = [r[0] for r in rows if r[at] != "?" and r[at] <= median]
    big = [r[0] for r in rows if r[at] != "?" and r[at] > median]
    
    # Certify if the difference in MRE- (index 0) is real
    is_same = certify(small, big)
    
    print("Customer Query: Are big setups more accurate than small ones?")
    if is_same:
        print("Result: The difference is likely just noise.")
    else:
        print("Result: There is a statistically significant difference between small and big setups.")
    
    print(f"Small setup MRE- mean: {sum(small)/len(small):.4f}")
    print(f"Big setup MRE- mean:   {sum(big)/len(big):.4f}")
    
    exit(0)
