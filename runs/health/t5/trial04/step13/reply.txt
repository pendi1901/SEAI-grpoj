import ezr
import random

def certify(xs, ys):
    """
    Returns True if the two distributions are statistically indistinguishable,
    False if they are significantly different.
    """
    # Use ezr's stats module to compare the two distributions.
    # same() checks for statistical difference using cliffs and ks tests.
    return ezr.same(xs, ys)

if __name__ == "__main__":
    # Load data
    filename = "Health-ClosedIssues0000.n1000.train.csv"
    all_data = list(ezr.csv(filename))
    header = all_data[0]
    rows = all_data[1:]

    # The customer asks about the "two built-in rules". 
    # In the context of the provided ezr.py, the primary decision-making 
    # mechanism is the tree-based ranker vs random sampling, or 
    # the split criteria used in the tree.
    # We will test if the choice of 'criterion' (a symbolic input) 
    # significantly impacts the primary goal 'ACC+'.
    
    criterion_idx = header.index("criterion")
    acc_idx = header.index("ACC+")
    
    # Get the two most common criteria
    criteria = {}
    for row in rows:
        c = row[criterion_idx]
        criteria[c] = criteria.get(c, 0) + 1
    
    sorted_criteria = sorted(criteria.keys(), key=lambda k: criteria[k], reverse=True)
    
    if len(sorted_criteria) >= 2:
        c1, c2 = sorted_criteria[0], sorted_criteria[1]
        
        xs = [row[acc_idx] for row in rows if row[criterion_idx] == c1]
        ys = [row[acc_idx] for row in rows if row[criterion_idx] == c2]
        
        is_same = certify(xs, ys)
        
        print(f"Analysis: Does the choice of criterion ('{c1}' vs '{c2}') make a difference?")
        if is_same:
            print("Result: The difference in accuracy is statistically insignificant.")
            print("Conclusion: You can stop arguing about it; the choice makes no real difference.")
        else:
            print("Result: The difference in accuracy is statistically significant.")
            print("Conclusion: The choice of criterion matters.")
    else:
        print("Not enough data to compare two different criteria.")
    
    exit(0)
