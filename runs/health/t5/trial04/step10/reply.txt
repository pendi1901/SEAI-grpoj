import ezr
import random

def race(header, rows, label):
    """
    Compare three strategies for finding the best row with a 50-label budget.
    Strategies:
    1. random: Randomly samples rows.
    2. acquire: Uses ezr's active learning (near best, far from rest).
    3. greedy: Always picks the next row that is closest to the current best.
    """
    budget = 50
    labeled_indices = set()

    def oracle_wrapper(idx):
        if idx not in labeled_indices:
            if len(labeled_indices) >= budget:
                raise Exception("Budget exceeded")
            labeled_indices.add(idx)
        return label(idx)

    # Strategy 1: Random
    def strategy_random(tbl, rows_to_search):
        indices = random.sample(range(len(rows_to_search)), min(budget // 3, len(rows_to_search)))
        for i in indices:
            oracle_wrapper(i)
        return min(indices, key=lambda i: ezr.ydist(tbl, rows_to_search[i]))

    # Strategy 2: Acquire (Active Learning)
    def strategy_acquire(tbl, rows_to_search):
        # We use a subset of the budget for this strategy
        subset_budget = budget // 3
        # Create a temporary table for the acquire function
        # We need to pass the oracle that updates the rows
        def local_oracle(row):
            idx = int(row[-1])
            true_vals = oracle_wrapper(idx)
            for i, name in enumerate(header):
                if name[-1] in "+-": row[i] = true_vals[i]
            return row
        
        # Prepare rows with index
        indexed = [r + [i] for i, r in enumerate(rows_to_search)]
        temp_tbl = ezr.Tbl([header + ["idx"]] + indexed)
        labeled = ezr.acquire(temp_tbl, cap=subset_budget, label=local_oracle)
        best = min(labeled, key=lambda r: ezr.ydist(temp_tbl, r))
        return int(best[-1])

    # Strategy 3: Greedy
    def strategy_greedy(tbl, rows_to_search):
        best_idx = random.choice(range(len(rows_to_search)))
        oracle_wrapper(best_idx)
        for _ in range((budget // 3) - 1):
            # Find row closest to current best
            best_row = rows_to_search[best_idx]
            next_idx = min([i for i in range(len(rows_to_search)) if i not in labeled_indices],
                           key=lambda i: ezr.xdist(tbl, rows_to_search[i], best_row))
            oracle_wrapper(next_idx)
            if ezr.ydist(tbl, rows_to_search[next_idx]) < ezr.ydist(tbl, best_row):
                best_idx = next_idx
        return best_idx

    # Setup
    tbl = ezr.Tbl([header] + rows)
    strategies = {"random": strategy_random, "acquire": strategy_acquire, "greedy": strategy_greedy}
    results = {}
    
    for name, func in strategies.items():
        try:
            best_idx = func(tbl, rows)
            results[name] = (best_idx, ezr.ydist(tbl, rows[best_idx]))
        except:
            continue

    winner = min(results, key=lambda n: results[n][1])
    return {"winner": winner, "row": results[winner][0], "tried": list(strategies.keys())}

if __name__ == "__main__":
    filename = "Health-ClosedIssues0000.n1000.train.csv"
    all_data = list(ezr.csv(filename))
    header = all_data[0]
    rows = all_data[1:]
    
    def label(i):
        return all_data[i+1]

    result = race(header, rows, label)
    print(f"Race Results:")
    print(f"Winner: {result['winner']}")
    print(f"Best row index found: {result['row']}")
    print(f"Strategies tried: {', '.join(result['tried'])}")
