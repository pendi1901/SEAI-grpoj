#-- misc ---------------------------------------------------
def say(x: Any, p: int = 2) -> str:
  "Floats get P decimals; dicts list their keys."
  if isinstance(x,float): x= f"{x:.{p}f}".rstrip("0").rstrip(".")
  if isinstance(x, dict):
    x= "{"+", ".join(f"{k}: {say(v,p)}" for k,v in x.items())+"}"
  return str(x)

class o(dict):
  "A dict you can poke with a dot."
  __getattr__, __setattr__ = dict.__getitem__, dict.__setitem__
  __repr__ = say

def atom(s: str) -> ATOM:
  "'22' -> 22. '3.1' -> 3.1. 'True' -> True. 'x' -> 'x'."
  for fn in (int, float):
    try: return fn(s)
    except ValueError: pass
  s = s.strip()
  return {"True": True, "False": False}.get(s, s)

def csv(file: str) -> Iterator[ROW]:
  "ROWS of FILE, each cell coerced."
  with open(os.path.expanduser(file),  # -sig drops any BOM
           encoding="utf-8-sig") as f:
    for line in f:
      if line.strip(): yield [atom(s) for s in line.split(",")]

def Col(txt: str = " ", at: int = 0) -> COL:
  "Factory that returns NUM or SYM."
  return Num(txt,at) if txt[0].isupper() else Sym(txt,at)

class Num:
  "Place to summarize stream of numbers."
  __slots__ = ("at", "txt", "n", "mu", "m2", "sd", "goal")
  def __init__(self, txt: str = " ", at: int = 0) -> None:
    self.at, self.txt, self.n = at, txt, 0
    self.mu = self.m2 = self.sd = 0
    self.goal = 0 if txt[-1] == "-" else 1
  __repr__ = lambda self: say({k: getattr(self,k)
                               for k in self.__slots__})

class Sym:
  "Place to summarize stream of Symbols.  The class is the tag."
  __slots__ = ("at", "txt", "n", "has")
  def __init__(self, txt: str = " ", at: int = 0) -> None:
    self.at, self.txt, self.n, self.has = at, txt, 0, {}
  __repr__ = lambda self: say({k: getattr(self,k)
                               for k in self.__slots__})

def add(col: COL, v: ATOM, inc: int = 1) -> None:
  "Show V to COL.  INC=-1 takes it away.  `?` changes nothing."
  if v != "?":
    col.n += inc
    if type(col) is Sym:
      col.has[v] = col.has.get(v, 0) + inc
    else:
      d = v - col.mu                      # Welford, for `div`
      col.mu += inc * d / max(1, col.n)
      col.m2  = max(0, col.m2 + inc * d * (v - col.mu))
      col.sd  = 0 if col.n < 2 else (col.m2/(col.n-1))**.5

def mid(col: COL) -> ATOM:
  "Middle: the mean, or the most common symbol; ? if none."
  return (max(col.has, key=col.has.get) if col.has else "?"
          ) if type(col) is Sym else col.mu

def mids(tbl: TBL) -> MIDS:
  "Every column's middle, cached until the next addCols."
  tbl.mids = tbl.mids or {at: mid(c) for at,c in tbl.cols.items()}
  return tbl.mids

def div(col: COL) -> float:
  "Spread: entropy, or standard deviation."
  return (-sum(v/col.n*log2(v/col.n) for v in col.has.values())
          if type(col) is Sym else col.sd)  # sd kept by `add`

def norm(col: COL, v: ATOM) -> ATOM:
  "To 0..1, by the logistic curve.  Symbols do not scale."
  if type(col) is Sym: return v
  z = max(-3, min(3, (v - col.mu) / (1e-32 + div(col))))
  return 1 / (1 + exp(-1.7 * z))

#-- tables -------------------------------------------------
def Tbl(src: Iterable[ROW]) -> TBL:
  "Header names the columns: X skip, +- goal, ! klass."
  src = iter(src)
  tbl = Cols(o(rows=[], cols={}, x=[], y=[], mids=None,
               names=next(src)))
  for row in src: addRow(tbl, row)
  return tbl

def Cols(tbl: TBL) -> TBL:
  "Create column roles."
  for at, s in enumerate(tbl.names):
    if s[-1] == "X": continue            # skip me entirely
    tbl.cols[at] = Col(s,at)
    (tbl.y if s[-1] in "+-!" else tbl.x).append(tbl.cols[at])
  return tbl

def addRow(tbl: TBL, row: ROW) -> ROW:
  "Keep ROW, and show it to my columns."
  tbl.rows += [row]
  return addCols(tbl, row)

def addCols(tbl: TBL, row: ROW, inc: int = 1) -> ROW:
  "Show ROW to my columns, without keeping it."
  tbl.mids = None                 # a new row moves the middles
  for at, col in tbl.cols.items(): add(col, row[at], inc)
  return row

def clone(tbl: TBL, rows: ROWS = None) -> TBL:
  "An empty copy of TBL, plus ROWS."
  return Tbl([tbl.names] + (rows or []))

def dist(vs: Iterable[QTY], n: int) -> float:
  "Root mean square of N gaps."
  return sqrt(sum(v*v for v in vs) / n)

def ydist(tbl: TBL, row: ROW) -> float:
  "How far ROW's goals are from the best they could be."
  return dist((abs(norm(c, row[c.at]) - c.goal)
                    for c in tbl.y), len(tbl.y))

def gap(col: COL, a: ATOM, b: ATOM) -> QTY:
  "Distance between two values of one column."
  if a == "?" or b == "?": return 1    # unknown = far
  return a!=b if type(col) is Sym else abs(norm(col,a)-norm(col,b))

def xdist(tbl: TBL, r1: ROW, r2: ROW | MIDS) -> float:
  "How far apart two rows are, over the x columns."
  return dist((gap(c, r1[c.at], r2[c.at])
                    for c in tbl.x), len(tbl.x))

def oracle(row: ROW) -> ROW:
  """Labeller. Rows here arrive with ys, so nothing to do.
  Real oracles fill in any missing y cells (and can close
  over a TBL to know which cells those are)."""
  return row

def acquire(tbl: TBL, cap: int | None = None,
            label: ORACLE = oracle) -> ROWS:
  "Label near the best rows, far from the rest."
  cap  = cap or the.Stop
  todo = random.sample(tbl.rows, len(tbl.rows))[:the.Few]
  both = clone(tbl, [label(todo.pop())
                     for _ in range(the.Start)])
  both.rows.sort(key=lambda r: ydist(both, r))
  n    = int(sqrt(the.Start))
  best,rest = clone(tbl,both.rows[:n]), clone(tbl,both.rows[n:])
  while todo and len(both.rows) < cap:
    cb, cr = mids(best), mids(rest)    # two centroids
    todo.sort(key=lambda z: xdist(tbl,z,cr) - xdist(tbl,z,cb))
    addRow(best, addRow(both, label(todo.pop())))
    best.rows.sort(key=lambda z: ydist(both, z))  # no y leak
    if len(best.rows) > sqrt(len(both.rows)): # best stays small
      addRow(rest, addCols(best, best.rows.pop(), -1))
  return both.rows

def grabs(tbl: TBL, cap: int | None = None,
          label: ORACLE = oracle) -> ROWS:
  "The straw man: same budget, taken at random."
  cap = cap or the.Stop
  return [label(r) for r in
          random.sample(tbl.rows, min(cap, len(tbl.rows)))]

# Calls core.  Nothing above this line calls down here.
# NODE = o(rows, at, v, go, kids, mu).
def cut(tbl: TBL, rows: ROWS) -> tuple[int,ATOM,GO]|None:
  "The (at, v, go) whose two sides have the tightest ys."
  ys = [ydist(tbl, r) for r in rows]
  out, least = None, 1e30
  for col in tbl.x:
    for v, go in candidates(col, rows):
      a, b = Num(), Num()   # one pass, so one `go` per row
      for r,y in zip(rows,ys): add(a if go(r) else b, y)
      if a.n >= the.Leaf and b.n >= the.Leaf:
        score = (div(a)*a.n + div(b)*b.n) / (a.n + b.n)
        if score < least: out, least = (col.at, v, go), score
  return out

def candidates(col: COL, rows: ROWS) -> Iterator[tuple[ATOM,GO]]:
  "What splits to try: one per symbol, or CUTS per num."
  at = col.at
  if type(col) is Sym:
    for v in sorted({r[at] for r in rows if r[at] != "?"}):
      yield v, lambda r,v=v,at=at: r[at] == v
  else:                # by rank, not value: THESE rows are dense
    xs = sorted(r[at] for r in rows if r[at] != "?")
    k  = max(1, len(xs)//the.Cuts)     # cuts, spaced by rank
    for v in xs[k-1:-1:k]:
      yield v, (lambda r,v=v,at=at: r[at] != "?" and r[at] <= v)

def tree(tbl: TBL, rows: ROWS) -> NODE:
  "Split while a split keeps both sides big enough."
  node = o(rows=rows, at=None, v=None, go=None, kids=[],
           mu=sum(ydist(tbl,r) for r in rows)/len(rows))
  if len(rows) > the.Leaf and (found := cut(tbl, rows)):
    at, v, go = found
    yes, no = [], []
    for r in rows: (yes if go(r) else no).append(r)
    node.at, node.v, node.go = at, v, go # cut kepts sides big
    node.kids = [tree(tbl, yes), tree(tbl, no)]
  return node

def leaf(node: NODE, row: ROW) -> NODE:
  "Walk ROW down to its leaf."
  while node.kids: node = node.kids[0 if node.go(row) else 1]
  return node

def leafs(node: NODE) -> list[NODE]:
  "Return leaves of this tree."
  return [x for k in node.kids for x in leafs(k)] or [node]

def show(tbl: TBL, node: NODE, pre: str | None = None,
         edge: str = "") -> None:
  "Tree on the left; ydist and n on the right."
  ls = sorted(leafs(node), key=lambda z: z.mu)
  def walk(z: NODE, pre: str | None, edge: str) -> None:
    m = "+" if z is ls[0] else "-" if z is ls[-1] else " "
    print(f"{m} {round(100*z.mu):>4} {len(z.rows):>5}   "
          f"{(pre or '')+edge}".rstrip())
    op = "=" if type(tbl.cols[z.at or 0]) is Sym else "<="
    no = "!=" if op == "=" else ">"
    for kid, oper in zip(z.kids, [op, no]):
      walk(kid, "" if pre is None else pre + "|  ",
           f"{tbl.names[z.at]} {oper} {say(z.v)}")
  print(" ydist     n")
  walk(node, pre, edge)



#-- rig ----------------------------------------------------
# Calls core, acquire, and one RANKER -- tree's, by default.
def ranker(lab: TBL) -> Callable[[ROW], float]:
  "Grow a tree on LAB; score any row by its leaf's mean ydist."
  node = tree(lab, lab.rows)
  return lambda r: leaf(node, r).mu

def wins(tbl: TBL) -> Callable[[ROW], float]:
  "100 at the best row, 0 at an average one."
  ys = sorted(ydist(tbl, r) for r in tbl.rows)
  lo, avg = ys[0], sum(ys)/len(ys)
  return lambda r: max(-100, min(100, 100*(1 - (ydist(tbl,r)-lo)
                                            / (avg-lo+1e-32))))

def holdout(tbl: TBL, pick: PICKER = acquire,
            rank: RANKER = ranker,
            label: ORACLE = oracle) -> ROW:
  "Train on half; of CHECK guesses on the rest, pick best."
  rows  = random.sample(tbl.rows, len(tbl.rows))
  n     = len(rows)//2
  tr    = clone(tbl, rows[:n][:the.Few])
  lab   = clone(tbl, pick(tr, the.Stop - the.Check, label))
  top   = sorted(rows[n:], key=rank(lab))
  return min(top[:the.Check], key=lambda r: ydist(lab, label(r)))

# A leaf: it calls core, and nothing here calls it.
def adds(vs: Iterable[ATOM], col: COL | None = None) -> COL:
  "Every item of VS into COL."
  col = Num() if col is None else col
  for v in vs: add(col, v)
  return col

def cohen(xs: NUMS, ys: NUMS, d: float = 0.35,
          eps: float = 0) -> bool:
  "Are the means within D pooled standard deviations?"
  a, b = adds(xs), adds(ys)
  pool = ((a.n-1)*div(a)**2 + (b.n-1)*div(b)**2)/(a.n+b.n-2)
  return abs(a.mu - b.mu) <= max(eps, d * sqrt(pool))

def cliffs(xs: NUMS, ys: NUMS, d: float = 0.197) -> bool:
  "Sorted in.  Is the rank imbalance small enough?"
  gt = lt = j = k = 0
  for x in xs:
    while j < len(ys) and ys[j] <  x: j += 1; k = j
    while k < len(ys) and ys[k] <= x: k += 1
    gt += j; lt += len(ys) - k
  return abs(gt - lt) / (len(xs) * len(ys)) <= d

def ks(xs: NUMS, ys: NUMS, a: float = 1.36) -> bool:
  "Sorted in.  95% Kolmogorov-Smirnov."
  n, m, i, j, d = len(xs), len(ys), 0, 0, 0
  while i < n and j < m:
    v = min(xs[i], ys[j])
    while i < n and xs[i] <= v: i += 1
    while j < m and ys[j] <= v: j += 1
    d = max(d, abs(i/n - j/m))
  return d <= a * sqrt((n + m) / (n * m))

def same(xs: NUMS, ys: NUMS, eps: float = 0) -> bool:
  """Indistinguishable.  Always cliffs and ks; EPS adds cohen, and
  doubles as its effect size (eps=0 skips that third test)."""
  xs, ys = sorted(xs), sorted(ys)
  return (cliffs(xs,ys) and ks(xs,ys)
          and (not eps or cohen(xs,ys,d=eps)))

import ezr

def remember(header, rows):
    """
    Identify the top 5% of rows based on multi-objective optimization.
    Uses the ydist metric from ezr to rank rows by their proximity to the 
    ideal goal values.
    """
    # Create a table object to leverage ezr's column handling and ydist
    # We construct the table using the provided header and rows
    data = [header] + rows
    tbl = ezr.Tbl(data)
    
    # Calculate ydist for every row. Lower ydist is better.
    # ydist measures how far a row's goals are from the best possible.
    indexed_rows = []
    for i, row in enumerate(rows):
        indexed_rows.append((i, ezr.ydist(tbl, row)))
    
    # Sort by ydist ascending (best first)
    indexed_rows.sort(key=lambda x: x[1])
    
    # Return top 5% of rows (800 rows * 0.05 = 40 rows)
    limit = max(1, int(len(rows) * 0.05))
    return [item[0] for item in indexed_rows[:limit]]

import ezr
import random

def guess(header, train, test):
    """
    Estimates the first goal column for each test row.
    Uses the tree-based ranker from ezr to find similar rows in the training set
    and returns the mean of the first goal column from those similar rows.
    """
    # Create a table from the training data
    train_tbl = ezr.Tbl([header] + train)
    
    # Build a tree to partition the space
    node = ezr.tree(train_tbl, train_tbl.rows)
    
    # Identify the first goal column index
    # The first goal is the first column in the 'y' list
    first_goal_at = train_tbl.y[0].at
    
    results = []
    for row in test:
        # Walk the tree to find the leaf containing similar rows
        leaf_node = ezr.leaf(node, row)
        
        # Calculate the mean of the first goal in this leaf
        # If the leaf has rows, average the first goal column
        if leaf_node.rows:
            vals = [r[first_goal_at] for r in leaf_node.rows]
            estimate = sum(vals) / len(vals)
        else:
            # Fallback to global mean if leaf is empty (should not happen with tree logic)
            all_vals = [r[first_goal_at] for r in train_tbl.rows]
            estimate = sum(all_vals) / len(all_vals)
            
        results.append(float(estimate))
        
    return results

if __name__ == "__main__":
    # Load data
    filename = "Health-ClosedIssues0000.n1000.train.csv"
    data_gen = ezr.csv(filename)
    
    try:
        header = next(data_gen)
        all_rows = list(data_gen)
    except StopIteration:
        exit(0)
        
    # Split data: 80% train, 20% test
    random.seed(1234567891)
    random.shuffle(all_rows)
    split_idx = int(len(all_rows) * 0.8)
    train = all_rows[:split_idx]
    test_raw = all_rows[split_idx:]
    
    # Mask the goals in the test set to simulate "unknown"
    # The goals are the columns in the header that end in + or -
    goal_indices = [i for i, name in enumerate(header) if name[-1] in "+-"]
    
    test = []
    for row in test_raw:
        new_row = list(row)
        for idx in goal_indices:
            new_row[idx] = "?"
        test.append(new_row)
        
    # Get estimates
    estimates = guess(header, train, test)
    
    # Calculate average error for demonstration
    actuals = [row[goal_indices[0]] for row in test_raw]
    errors = [abs(e - a) for e, a in zip(estimates, actuals)]
    avg_error = sum(errors) / len(errors)
    
    print(f"Analysis complete on {len(all_rows)} records.")
    print(f"Estimated the first goal ('{header[goal_indices[0]]}') for {len(test)} test cases.")
    print(f"Average absolute error of estimates: {avg_error:.4f}")

import ezr
import random

def certify(xs, ys):
    """
    Returns True if the distributions of xs and ys are statistically 
    indistinguishable (same apart from noise), False otherwise.
    Uses the statistical tests provided in ezr.
    """
    # Sort inputs as required by the statistical tests in ezr
    xs_sorted = sorted(xs)
    ys_sorted = sorted(ys)
    
    # Use ezr.same which combines cliffs, ks, and cohen tests
    # eps=0.35 is a standard medium effect size for Cohen's d
    return ezr.same(xs_sorted, ys_sorted, eps=0.35)

if __name__ == "__main__":
    filename = "Health-ClosedIssues0000.n1000.train.csv"
    data_gen = ezr.csv(filename)
    
    try:
        header = next(data_gen)
        rows = list(data_gen)
    except StopIteration:
        exit(0)

    # Identify "big" vs "small" setups. 
    # 'N_estimators' is a numeric column representing setup size.
    # We find its index in the header.
    n_est_idx = header.index('N_estimators')
    
    # We use 'MRE-' as the accuracy metric (lower is better).
    mre_idx = header.index('MRE-')
    
    # Split data based on the median of N_estimators
    n_vals = [row[n_est_idx] for row in rows]

import ezr
import random

def blame(header, rows):
    """
    Identify input columns that best explain why the worst rows are bad.
    We compare the distribution of input values in the worst 25% of rows
    against the rest of the data. Columns where the distributions differ
    significantly (using the statistical tests in ezr) are flagged as 'blameworthy'.
    """
    tbl = ezr.Tbl([header] + rows)
    
    # Calculate ydist for all rows to identify the "worst" ones
    # Higher ydist means further from the ideal goals
    indexed_rows = []
    for i, row in enumerate(rows):
        indexed_rows.append((row, ezr.ydist(tbl, row)))
    
    # Sort by ydist descending (worst first)
    indexed_rows.sort(key=lambda x: x[1], reverse=True)
    
    # Define worst 25%
    split = len(indexed_rows) // 4
    worst = [x[0] for x in indexed_rows[:split]]
    rest = [x[0] for x in indexed_rows[split:]]
    
    blame_list = []
    
    # Check each input column
    for col in tbl.x:
        at = col.at
        # Extract values for this column
        worst_vals = [r[at] for r in worst if r[at] != "?"]
        rest_vals = [r[at] for r in rest if r[at] != "?"]
        
        if not worst_vals or not rest_vals:
            continue
            
        # If the column is numeric, use statistical tests to see if distributions differ
        if isinstance(col, ezr.Num):
            # If they are NOT the same, the column distribution in the worst rows
            # is significantly different from the rest, implying it contributes to the badness.
            if not ezr.same(sorted(worst_vals), sorted(rest_vals), eps=0.35):
                blame_list.append(header[at])
        else:
            # For symbolic columns, check if the most frequent value differs significantly
            # or if the distribution of symbols is skewed in the worst set.
            # A simple heuristic: if the top symbol in 'worst' is different from 'rest'
            # or the counts are heavily imbalanced.
            w_counts = {v: worst_vals.count(v) for v in set(worst_vals)}
            r_counts = {v: rest_vals.count(v) for v in set(rest_vals)}
            
            w_top = max(w_counts, key=w_counts.get)
            r_top = max(r_counts, key=r_counts.get)
            
            if w_top != r_top:
                blame_list.append(header[at])
                
    return blame_list

import ezr
import random

def justify(header, rows, row):
    """
    Returns (p, reason): p is the fraction of rows that 'row' is better than,
    and a sentence explaining why.
    """
    # Create a table to handle column roles and ydist calculations
    tbl = ezr.Tbl([header] + rows)
    
    # To compare 'row' against 'rows', we need to estimate the goals of 'row'.
    # We use a simple tree-based ranker to estimate the ydist of 'row'.
    # We build the tree on all available data.
    node = ezr.tree(tbl, rows)
    
    # The ranker returns the mean ydist of the leaf where the row lands.
    # Lower ydist is better.
    ranker = lambda r: ezr.leaf(node, r).mu
    
    # Calculate the estimated ydist for the target row
    target_ydist = ranker(row)
    
    # Calculate ydist for all existing rows
    all_ydists = [ezr.ydist(tbl, r) for r in rows]
    
    # p is the fraction of rows that are worse than the target row.
    # A row is better if its ydist is smaller.
    # We count how many rows have a ydist greater than our target.
    better_than_count = sum(1 for y in all_ydists if y > target_ydist)
    p = better_than_count / len(rows)
    
    # Construct a human-readable reason
    reason = (f"This setup is better than {int(p * 100)}% of all known configurations "
              "because its input parameters lead to a predicted performance "
              "closer to the ideal goal values.")
    
    return p, reason

import ezr
import random

def choose(header, rows, label):
    """
    Find the best setup by using the acquire logic from ezr.
    We have a budget of 50 labels. We use the acquire function
    which intelligently samples the space to find the best rows.
    """
    # Create a table object. The rows currently have '?' for goals.
    # We need to wrap the label function to handle the row indices.
    # The acquire function expects an oracle that takes a row and returns a row.
    
    # Map rows to their original indices to return the correct position
    indexed_rows = []
    for i, row in enumerate(rows):
        # We store the index in the row temporarily or use a closure
        indexed_rows.append((i, list(row)))
    
    # Create a lookup for the original index
    row_map = {id(r): i for i, r in indexed_rows}
    
    def oracle_wrapper(row):
        # Find the original index of this row
        idx = row_map[id(row)]
        # Get the true values from the provided label function
        true_row = label(idx)
        # Update the row with true values
        for i, val in enumerate(true_row):
            row[i] = val
        return row

    # Initialize the table with the header
    tbl = ezr.Tbl([header] + [r[1] for r in indexed_rows])
    
    # Use ezr.acquire to spend the budget of 50 labels
    # ezr.acquire returns the rows that were labeled
    labeled_rows = ezr.acquire(tbl, cap=50, label=oracle_wrapper)
    
    # Find the best row among the labeled ones
    # A row is better if its ydist is smaller
    best_row = min(labeled_rows, key=lambda r: ezr.ydist(tbl, r))
    
    # Find the index of the best row
    return row_map[id(best_row)]

        if col_name not in header:
            continue
        at = header.index(col_name)
        
        candidates = []
        if isinstance(allowed, list):  # Symbolic
            candidates = [v for v in allowed if v != row[at]]
        else:  # Numeric (low, high)

            if current_ydist < min_ydist:
                min_ydist = current_ydist
                best_change = {col_name: val}
                
    return best_change

def spot(header, rows):

    We look for rows that are outliers in the input space (x-columns)
    relative to the rest of the data. Rows that are far from the 
    centroid of the input space are flagged as suspicious.
    """

    tbl = ezr.Tbl([header] + rows)
    
    # Calculate the centroid (mids) of the input space
    # We use the cached mids from the table
    centroid = ezr.mids(tbl)

    scores = []
    for i, row in enumerate(rows):
        dist = ezr.xdist(tbl, row, centroid)
        scores.append((i, dist))
    
    # Sort by distance descending (most suspicious/outlier first)
    scores.sort(key=lambda x: x[1], reverse=True)
    
    # Return the indices of the rows
    return [item[0] for item in scores]

import ezr
import random

def race(header, rows, label):
    """
    Compare three strategies for finding the best row within a 50-label budget.
    Strategies:
    1. random: Randomly samples rows.
    2. greedy: Picks the row closest to the current best.
    3. acquire: Uses the intelligent acquisition function from ezr.
    """
    budget = 50
    # We must ensure we don't label more than 50 rows total across all strategies.
    # The harness enforces this globally.
    
    # To calculate ydist, we need a table with populated goals.
    # Since we only have 50 labels, we build a TBL from the labeled rows.
    
    def get_best_idx(labeled_indices, tbl_full):
        # Find the best row among those labeled
        return min(labeled_indices, key=lambda i: ezr.ydist(tbl_full, rows[i]))

    # 1. Random Strategy
    def strategy_random(budget_left):
        indices = random.sample(range(len(rows)), budget_left)
        for i in indices: label(i)
        return indices

    # 2. Greedy Strategy
    def strategy_greedy(budget_left):
        indices = random.sample(range(len(rows)), min(5, budget_left))
        for i in indices: label(i)
        
        # Build a temporary table to calculate ydist for the labeled set
        while len(indices) < budget_left:
            # We need a table of currently labeled rows to define the 'best'
            labeled_rows = [rows[i] for i in indices]
            tbl_lab = ezr.Tbl([header] + labeled_rows)
            
            best_i = min(indices, key=lambda i: ezr.ydist(tbl_lab, rows[i]))
            
            # Find closest unlabelled to best
            unlabelled = [i for i in range(len(rows)) if i not in indices]
            next_i = min(unlabelled, key=lambda i: ezr.xdist(tbl_lab, rows[i], rows[best_i]))
            label(next_i)
            indices.append(next_i)
        return indices

    # 3. Acquire Strategy
    def strategy_acquire(budget_left):
        # Acquire needs a TBL with all rows (even if goals are '?')
        tbl = ezr.Tbl([header] + rows)
        
        # We need to wrap the label function to update the rows in the table
        def oracle_wrapper(i):
            true_vals = label(i)
            # Update the row in the table
            for col_idx, val in enumerate(true_vals):
                rows[i][col_idx] = val
            return rows[i]
            
        # ezr.acquire returns the rows that were labeled
        labeled = ezr.acquire(tbl, cap=budget_left, label=oracle_wrapper)
        # Return indices of labeled rows
        return [i for i, r in enumerate(rows) if r in labeled]

    # Run race
    strategies = [
        ("random", strategy_random),
        ("greedy", strategy_greedy),
        ("acquire", strategy_acquire)
    ]
    
    results = {}
    # We split the 50 budget equally among 3 strategies
    per_budget = budget // 3
    
    for name, func in strategies:
        indices = func(per_budget)
        # Build a table from the labeled rows to evaluate them
        labeled_rows = [rows[i] for i in indices]
        tbl_lab = ezr.Tbl([header] + labeled_rows)
        best_i = min(indices, key=lambda i: ezr.ydist(tbl_lab, rows[i]))
        results[name] = (best_i, ezr.ydist(tbl_lab, rows[best_i]))

    winner_name = min(results, key=lambda n: results[n][1])
    
    return {
        "winner": winner_name,
        "row": results[winner_name][0],
        "tried": [s[0] for s in strategies]
    }
