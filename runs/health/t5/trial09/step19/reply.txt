import ezr
import random

def spot(header, rows):
    """
    Identify rows containing input values that are 'suspicious'.
    We define suspicious as being an outlier in the input space (x-columns).
    We use the xdist metric from ezr to measure how far each row is from 
    the centroid of the input space. Rows with high xdist are considered 
    to contain values 'never seen' or far outside the norm.
    """
    # Create a table to handle column roles and identify x-columns
    tbl = ezr.Tbl([header] + rows)
    
    # Calculate the centroid (mids) of the input space
    centroid = ezr.mids(tbl)
    
    # Calculate the distance of each row from the centroid in the x-space
    # A high xdist indicates the row is an outlier in input parameters
    scores = []
    for i, row in enumerate(rows):
        # xdist measures distance over x columns
        dist = ezr.xdist(tbl, row, centroid)
        scores.append((i, dist))
    
    # Sort by distance descending (most suspicious/outlier first)
    scores.sort(key=lambda x: x[1], reverse=True)
    
    # Return the indices of the rows
    return [item[0] for item in scores]

if __name__ == "__main__":
    filename = "Health-ClosedIssues0000.n1000.train.csv"
    data_gen = ezr.csv(filename)
    
    try:
        header = next(data_gen)
        rows = list(data_gen)
    except StopIteration:
        exit(0)
        
    # Identify suspicious rows
    suspicious_indices = spot(header, rows)
    
    print("Analysis complete.")
    print("The following rows contain input settings that are far outside the norm:")
    
    # Print the top 5 most suspicious rows
    for i in range(min(5, len(suspicious_indices))):
        idx = suspicious_indices[i]
        print(f" - Row {idx} is highly suspicious (outlier in input space).")
        
    exit(0)
