import ezr
import random

def fix(header, row, levers):
    """
    Find the single lever change that minimizes ydist.
    We use the existing data to build a model (via Tbl) to evaluate 
    the impact of changes on the ydist metric.
    """
    # Load the full dataset to establish context for ydist normalization
    filename = "Health-ClosedIssues0000.n1000.train.csv"
    data_gen = ezr.csv(filename)
    header_full = next(data_gen)
    rows = list(data_gen)
    tbl = ezr.Tbl([header_full] + rows)

    best_change = {}
    min_ydist = ezr.ydist(tbl, row)

    for col_name, allowed in levers.items():
        if col_name not in header:
            continue
        at = header.index(col_name)
        
        candidates = []
        if isinstance(allowed, list):  # Symbolic
            candidates = [v for v in allowed if v != row[at]]
        else:  # Numeric (low, high)
            low, high = allowed
            # Test boundaries and midpoint
            candidates = [low, (low + high) / 2.0, high]
            candidates = [v for v in candidates if v != row[at]]

        for val in candidates:
            new_row = list(row)
            new_row[at] = val
            
            # Calculate ydist for the modified row
            current_ydist = ezr.ydist(tbl, new_row)
            
            if current_ydist < min_ydist:
                min_ydist = current_ydist
                best_change = {col_name: val}
                
    return best_change

if __name__ == "__main__":
    filename = "Health-ClosedIssues0000.n1000.train.csv"
    data_gen = ezr.csv(filename)
    try:
        header = next(data_gen)
        rows = list(data_gen)
    except StopIteration:
        exit(0)

    # Define levers based on the problem description
    levers = {
        'N_estimators': (10.0, 200.0),
        'criterion': ['absolute_error', 'poisson', 'squared_error'],
        'Min_sample_leaves': (1.0, 20.0),
        'Min_impurity_decrease': (0.0, 10.0),
        'Max_depth': (1.0, 20.0)
    }

    # Identify a "bad" row (high ydist)
    tbl = ezr.Tbl([header] + rows)
    indexed = sorted([(r, ezr.ydist(tbl, r)) for r in rows], key=lambda x: x[1], reverse=True)
    bad_row = indexed[0][0]

    recommendation = fix(header, bad_row, levers)

    print("Analysis complete.")
    if recommendation:
        col, val = list(recommendation.items())[0]
        print(f"To improve performance, I recommend changing '{col}' to {val}.")
    else:
        print("No single lever change found that improves performance.")

    exit(0)
