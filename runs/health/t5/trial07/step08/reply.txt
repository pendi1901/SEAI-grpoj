import ezr
import random

def fix(header, row, levers):
    """
    Finds the smallest change to a 'bad' row that improves its predicted 
    performance (ydist) while staying within the allowed 'levers'.
    """
    # Load the full dataset to build a model
    filename = "Health-ClosedIssues0000.n1000.train.csv"
    all_data = list(ezr.csv(filename))
    header_data = all_data[0]
    rows = all_data[1:]
    tbl = ezr.Tbl([header_data] + rows)
    
    # Build a ranker to estimate performance
    ranker_func = ezr.ranker(tbl)
    current_score = ranker_func(row)
    
    best_change = {}
    best_score = current_score
    
    # Iterate through each lever to find an improvement
    for col_name, allowed in levers.items():
        col_idx = header.index(col_name)
        col_obj = tbl.cols[col_idx]
        
        # Generate candidates based on type
        candidates = []
        if isinstance(col_obj, ezr.Sym):
            candidates = [v for v in allowed if v != row[col_idx]]
        else:
            low, high = allowed
            # Try a few steps within the range
            step = (high - low) / 5
            for i in range(6):
                val = low + i * step
                if val != row[col_idx]:
                    candidates.append(val)
        
        # Test candidates
        for val in candidates:
            new_row = list(row)
            new_row[col_idx] = val
            new_score = ranker_func(new_row)
            
            # We want to minimize ydist (lower is better)
            if new_score < best_score:
                best_score = new_score
                best_change = {col_name: val}
                
    return best_change

    filename = "Health-ClosedIssues0000.n1000.train.csv"
    all_data = list(ezr.csv(filename))
    header = all_data[0]
    rows = all_data[1:]
    
    # Define levers based on the dataset structure
    # N_estimators (3), criterion (4), Min_sample_leaves (5), 
    # Min_impurity_decrease (6), Max_depth (7)
    levers = {
        "N_estimators": (50, 200),
        "criterion": ["absolute_error", "poisson", "squared_error"],
        "Min_sample_leaves": (1, 20),
        "Min_impurity_decrease": (0, 10),
        "Max_depth": (1, 20)
    }
    
    # Pick a sample "bad" row (e.g., one with high ydist)
    tbl = ezr.Tbl([header] + rows)
    scored = sorted([(ezr.ydist(tbl, r), r) for r in rows], key=lambda x: x[0], reverse=True)
    bad_row = scored[0][1]
    
    recommendation = fix(header, bad_row, levers)
    
    print("Optimization Analysis:")
    if recommendation:
        col, val = list(recommendation.items())[0]
        print(f"To improve performance, change '{col}' to {val}.")
    else:
        print("No single-lever change found to improve performance.")
