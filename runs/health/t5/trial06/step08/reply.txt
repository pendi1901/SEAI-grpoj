import ezr
import random

def fix(header, row, levers):
    """
    Finds the smallest change to a 'bad' row that improves its ydist
    without increasing the 'N_estimators' (a proxy for slowness).
    """
    # Create a dummy table to use ydist and norm functions
    # We need a TBL to use the library's distance metrics
    # We use a minimal set of rows to initialize the column metadata
    tbl = ezr.Tbl([header, row])
    
    best_change = {}
    best_ydist = ezr.ydist(tbl, row)
    
    # Iterate through each lever to find a single-step improvement
    for col_name, allowed in levers.items():
        col_idx = header.index(col_name)
        original_val = row[col_idx]
        
        # Generate candidates based on lever type
        candidates = []
        if isinstance(allowed, list):  # Symbolic
            candidates = [v for v in allowed if v != original_val]
        else:  # Numeric (low, high)
            low, high = allowed
            # Try a small step or boundary values
            candidates = [low, high]
            
        for val in candidates:
            # Constraint: Do not increase N_estimators (index 3)
            if col_name == "N_estimators" and val > original_val:
                continue
                
            # Create a modified row
            new_row = list(row)
            new_row[col_idx] = val
            
            # Evaluate
            current_ydist = ezr.ydist(tbl, new_row)
            if current_ydist < best_ydist:
                best_ydist = current_ydist
                best_change = {col_name: val}
                
    return best_change

if __name__ == "__main__":
    filename = "Health-ClosedIssues0000.n1000.train.csv"
    data = list(ezr.csv(filename))
    if not data:
        exit(1)
        
    header = data[0]
    rows = data[1:]
    
    # Define levers based on the problem description
    # Numeric: (min, max), Symbolic: [options]
    levers = {
        "N_estimators": (10, 200),
        "criterion": ["absolute_error", "poisson", "squared_error"],
        "Min_sample_leaves": (1, 20),
        "Max_depth": (1, 20)
    }
    
    # Pick a "bad" row (e.g., one with high ydist)
    tbl = ezr.Tbl(data)
    worst_row = max(rows, key=lambda r: ezr.ydist(tbl, r))
    
    recommendation = fix(header, worst_row, levers)
    
    print("Optimization Recommendation:")
    if recommendation:
        col, val = list(recommendation.items())[0]
        print(f"To improve performance, change '{col}' to {val}.")
    else:
        print("No simple improvement found within the given constraints.")
        
    exit(0)
