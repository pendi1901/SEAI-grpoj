import ezr
import random

def fix(header, row, levers):
    """
    Finds the single-lever change that results in the best improvement 
    in ydist for the provided row.
    """
    # Create a table to handle column roles and ydist calculations.
    # We use the full dataset to ensure normalization (norm) is consistent
    # with the population, which is required for accurate ydist comparison.
    # Since we don't have the full dataset passed in, we use the header
    # to reconstruct the column roles.
    tbl = ezr.Tbl([header])
    
    best_change = {}
    best_ydist = ezr.ydist(tbl, row)
    
    # Iterate through each lever to find the single-step improvement
    for col_name, allowed in levers.items():
        if col_name not in header:
            continue
        col_idx = header.index(col_name)
        
        # Generate candidates based on lever type
        candidates = []
        if isinstance(allowed, list):  # Symbolic
            candidates = [v for v in allowed if v != row[col_idx]]
        else:  # Numeric (low, high)
            low, high = allowed
            # Try a few points within the range to find a better setting
            # Ensure we stay within the bounds provided
            candidates = [float(low), float(high), float((low + high) / 2)]
            candidates = [v for v in candidates if v != row[col_idx]]
            
        for val in candidates:
            # Create a modified row
            new_row = list(row)
            new_row[col_idx] = val
            
            # Evaluate improvement
            current_ydist = ezr.ydist(tbl, new_row)
            if current_ydist < best_ydist:
                best_ydist = current_ydist
                best_change = {col_name: val}
                
    return best_change

if __name__ == "__main__":
    filename = "Health-ClosedIssues0000.n1000.train.csv"
    data = list(ezr.csv(filename))
    if not data:
        exit(1)
        
    header = data[0]
    rows = data[1:]
    
    # Define levers based on the problem description
    levers = {
        "N_estimators": (10, 200),
        "criterion": ["absolute_error", "poisson", "squared_error"],
        "Min_sample_leaves": (1, 20),
        "Min_impurity_decrease": (0.0, 10.0),
        "Max_depth": (1, 20)
    }
    
    # Pick a "bad" row (e.g., one with high ydist)
    tbl = ezr.Tbl(data)
    worst_row = max(rows, key=lambda r: ezr.ydist(tbl, r))
    
    recommendation = fix(header, worst_row, levers)
    
    print("Optimization Recommendation:")
    if recommendation:
        col, val = list(recommendation.items())[0]
        print(f"To improve performance, change '{col}' to {val}.")
    else:
        print("No single-lever improvement found that reduces ydist.")
        
    exit(0)
