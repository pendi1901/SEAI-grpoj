import ezr
import random

def guess(header, train, test):
    """
    Estimates the first goal column for each test row by building a tree
    on the training data and predicting the mean of the leaf node.
    """
    # Combine header and train to create a TBL object
    train_tbl = ezr.Tbl([header] + train)
    
    # Build a tree on the training data
    # The tree splits based on x-columns to minimize ydist
    node = ezr.tree(train_tbl, train_tbl.rows)
    
    # For each test row, find the leaf node and return the mean of the 
    # first goal column (index 0 of the y-columns)
    results = []
    for row in test:
        leaf_node = ezr.leaf(node, row)
        # Calculate the mean of the first goal column in this leaf
        # The first goal column is train_tbl.y[0]
        goal_col = train_tbl.y[0]
        at = goal_col.at
        
        # Calculate mean of the goal column for rows in this leaf
        vals = [r[at] for r in leaf_node.rows if r[at] != "?"]
        if vals:
            results.append(sum(vals) / len(vals))
        else:
            # Fallback to global mean if leaf is empty or has no data
            results.append(goal_col.mu)
            
    return results

if __name__ == "__main__":
    # Load the data
    filename = "Health-ClosedIssues0000.n1000.train.csv"
    data = list(ezr.csv(filename))
    
    if not data:
        print("No data found.")
        exit(1)
        
    header = data[0]
    all_rows = data[1:]
    
    # Split data: 80% train, 20% test
    random.seed(1234567891)
    random.shuffle(all_rows)
    split = int(len(all_rows) * 0.8)
    train = all_rows[:split]
    test_raw = all_rows[split:]
    
    # Mask the goals in the test set to simulate "unknown"
    # The first goal column is at index 0
    test = []
    for row in test_raw:
        new_row = list(row)
        new_row[0] = "?"
        test.append(new_row)
        
    # Get predictions
    predictions = guess(header, train, test)
    
    # Print a short summary for the customer
    print(f"Estimated the first goal column for {len(predictions)} test cases.")
    print(f"First 5 predictions: {[round(x, 4) for x in predictions[:5]]}")
    
    # Calculate simple error for validation
    actuals = [row[0] for row in test_raw]
    errors = [abs(p - a) for p, a in zip(predictions, actuals)]
    print(f"Average absolute error: {sum(errors)/len(errors):.4f}")
    
    exit(0)
