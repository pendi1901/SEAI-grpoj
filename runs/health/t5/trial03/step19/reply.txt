import ezr
import random

def spot(header, rows):
    """
    Identifies rows containing input values that are 'out of distribution'
    compared to the rest of the dataset. We use the distance from the 
    median (centroid) of the input space to flag suspicious rows.
    """
    # Create a table to leverage ezr's column and distance logic
    tbl = ezr.Tbl([header] + rows)
    
    # Calculate the median/mode of the input space (the 'centroid')
    centroid = ezr.mids(tbl)
    
    # Calculate the distance of each row from this centroid.
    # Rows with high xdist are far from the central tendency of the 
    # input parameters, suggesting they might be 'out of distribution' 
    # or contain values never seen before.
    scores = []
    for i, row in enumerate(rows):
        dist = ezr.xdist(tbl, row, centroid)
        scores.append((dist, i))
    
    # Sort by distance in descending order (most suspicious first)
    scores.sort(key=lambda x: x[0], reverse=True)
    
    # Return the indices of the rows
    return [item[1] for item in scores]

if __name__ == "__main__":
    # Load the data
    filename = "Health-ClosedIssues0000.n1000.train.csv"
    data = list(ezr.csv(filename))
    header = data[0]
    rows = data[1:]
    
    # Identify suspicious rows
    suspicious_indices = spot(header, rows)
    
    # Print a summary for the customer
    print("Analysis complete. Identified configurations that are far outside the norm.")
    print(f"Top 10 most suspicious row indices: {suspicious_indices[:10]}")
    print("These rows contain input settings that deviate significantly from the")
    print("typical configurations observed in the dataset.")
    
    exit(0)
