import ezr

def certify(xs, ys):
    """
    Returns True if xs and ys are statistically indistinguishable,
    False if they are significantly different.
    """
    return ezr.same(xs, ys)

if __name__ == "__main__":
    # Load data
    filename = "Health-ClosedIssues0000.n1000.train.csv"
    data = list(ezr.csv(filename))
    header = data[0]
    rows = data[1:]

    # Identify "big" vs "small" setups based on N_estimators
    # N_estimators is at index 3
    n_est_idx = 3
    acc_idx = 1  # ACC+ is at index 1

    small_setups = []
    big_setups = []

    # Calculate median to split into two groups
    n_ests = [r[n_est_idx] for r in rows]
    median = sorted(n_ests)[len(n_ests) // 2]

    for r in rows:
        if r[n_est_idx] < median:
            small_setups.append(r[acc_idx])
        else:
            big_setups.append(r[acc_idx])

    # Compare the two groups
    is_same = certify(small_setups, big_setups)

    print("Analysis: Are big setups significantly more accurate than small ones?")
    if is_same:
        print("Result: The difference is statistically insignificant (it is likely noise).")
    else:
        print("Result: There is a statistically significant difference between the groups.")
    
    print(f"Small setups (n={len(small_setups)}) mean ACC: {sum(small_setups)/len(small_setups):.4f}")
    print(f"Big setups (n={len(big_setups)}) mean ACC: {sum(big_setups)/len(big_setups):.4f}")
