import ezr

def spot(header, rows):
    """
    Identifies suspicious rows (outliers) by calculating the distance 
    of each row from the median of the input space. Rows with high 
    xdist from the median are considered 'suspicious' as they may 
    represent erroneous entries or extreme outliers.
    """
    # Create a table to leverage ezr's column and distance logic
    tbl = ezr.Tbl([header] + rows)
    
    # Calculate the median of the input space (the 'centroid' of the data)
    # We use the mids function from ezr which provides the median/mode for each column
    centroid = ezr.mids(tbl)
    
    # Calculate the distance of each row from this centroid
    # Rows that are very far from the centroid are likely outliers
    scores = []
    for i, row in enumerate(rows):
        # xdist calculates the distance between a row and the centroid
        dist = ezr.xdist(tbl, row, centroid)
        scores.append((dist, i))
    
    # Sort by distance in descending order (most suspicious first)
    scores.sort(key=lambda x: x[0], reverse=True)
    
    # Return the indices of the rows
    return [item[1] for item in scores]

if __name__ == "__main__":
    # Load the data
    filename = "Health-ClosedIssues0000.n1000.train.csv"
    data = list(ezr.csv(filename))
    header = data[0]
    rows = data[1:]
    
    # Identify suspicious rows
    suspicious_indices = spot(header, rows)
    
    # Print a summary for the customer
    print("Analysis complete. Identified potentially erroneous configurations.")
    print(f"Top 10 most suspicious row indices: {suspicious_indices[:10]}")
    print("These rows are the furthest from the central tendency of the input parameters.")
