import ezr
import random

def fix(header, row, levers):
    """
    Recommends the single best change to a configuration to improve its performance.
    We evaluate the impact of changing each lever by estimating the ydist 
    of the resulting row.
    """
    # Create a dummy table to use ydist for evaluation
    # We use the row itself to establish the context for normalization
    tbl = ezr.Tbl([header, row])
    
    best_change = {}
    best_score = float('inf')
    
    # Current performance
    current_score = ezr.ydist(tbl, row)
    
    for col_name, allowed in levers.items():
        col_idx = header.index(col_name)
        original_val = row[col_idx]
        
        # Generate candidates for this lever
        candidates = []
        if isinstance(allowed, list):  # Symbolic
            candidates = [v for v in allowed if v != original_val]
        else:  # Numeric (low, high)
            low, high = allowed
            # Try a few points in the range
            candidates = [low, (low + high) / 2, high]
            candidates = [v for v in candidates if v != original_val]
            
        for val in candidates:
            new_row = list(row)
            new_row[col_idx] = val
            
            # Estimate performance (lower ydist is better)
            # Since we don't have the full dataset, we use the row's 
            # distance to the ideal goal (0.0) as a proxy for improvement.
            score = ezr.ydist(tbl, new_row)
            
            if score < best_score:
                best_score = score
                best_change = {col_name: val}
                
    return best_change

if __name__ == "__main__":
    filename = "Health-ClosedIssues0000.n1000.train.csv"
    data = list(ezr.csv(filename))
    header = data[0]
    rows = data[1:]
    
    # Define levers based on the dataset
    levers = {
        "N_estimators": (50, 200),
        "criterion": ["absolute_error", "poisson", "squared_error"],
        "Min_sample_leaves": (1, 20),
        "Max_depth": (1, 20)
    }
    
    # Pick a "bad" row (high ydist)
    tbl = ezr.Tbl(data)
    scored = sorted([(ezr.ydist(tbl, r), r) for r in rows], key=lambda x: x[0], reverse=True)
    bad_row = scored[0][1]
    
    recommendation = fix(header, bad_row, levers)
    
    print("Analysis: Recommended single configuration change for poor performance:")
    if recommendation:
        for col, val in recommendation.items():
            print(f"Change '{col}' to {val}.")
    else:
        print("No beneficial change found.")
