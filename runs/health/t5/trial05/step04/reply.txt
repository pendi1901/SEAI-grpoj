import ezr
import random

def flow(header, row, seen):
    """
    Decide if the current row is 'good enough' to stop searching.
    We define 'good enough' as being in the top 5% of the potential 
    distribution based on the multi-objective ydist metric.
    """
    # If this is the first row, we can't compare it yet
    if not seen:
        return False
    
    # Create a table to use ezr's ydist logic
    # We use the current row and all previously seen rows to define the space
    data = [header] + seen + [row]
    tbl = ezr.Tbl(data)
    
    # Calculate ydist for the current row
    current_ydist = ezr.ydist(tbl, row)
    
    # Calculate ydist for all seen rows
    seen_ydists = [ezr.ydist(tbl, r) for r in seen]
    
    # We define "good enough" as being better than 95% of the rows seen so far.
    # This is a heuristic for early stopping in a stream.
    threshold = sorted(seen_ydists)[int(len(seen_ydists) * 0.05)]
    
    return current_ydist <= threshold

if __name__ == "__main__":
    filename = "Health-ClosedIssues0000.n1000.train.csv"
    data_iter = ezr.csv(filename)
    header = next(data_iter)
    rows = list(data_iter)
    
    # Shuffle to simulate random arrival order
    random.seed(1234567891)
    random.shuffle(rows)
    
    seen = []
    found = False
    for i, row in enumerate(rows):
        if flow(header, row, seen):
            print(f"Found a high-quality setup at index {i}!")
            print(f"Row data: {row}")
            found = True
            break
        seen.append(row)
    
    if not found:
        print("No setup met the criteria within the provided data.")
