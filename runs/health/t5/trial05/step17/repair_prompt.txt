def say(x: Any, p: int = 2) -> str:
  "Floats get P decimals; dicts list their keys."
  if isinstance(x,float): x= f"{x:.{p}f}".rstrip("0").rstrip(".")
  if isinstance(x, dict):
    x= "{"+", ".join(f"{k}: {say(v,p)}" for k,v in x.items())+"}"
  return str(x)

class o(dict):
  "A dict you can poke with a dot."
  __getattr__, __setattr__ = dict.__getitem__, dict.__setitem__
  __repr__ = say

def atom(s: str) -> ATOM:
  "'22' -> 22. '3.1' -> 3.1. 'True' -> True. 'x' -> 'x'."
  for fn in (int, float):
    try: return fn(s)
    except ValueError: pass
  s = s.strip()
  return {"True": True, "False": False}.get(s, s)

def csv(file: str) -> Iterator[ROW]:
  "ROWS of FILE, each cell coerced."
  with open(os.path.expanduser(file),  # -sig drops any BOM
           encoding="utf-8-sig") as f:
    for line in f:
      if line.strip(): yield [atom(s) for s in line.split(",")]

def Col(txt: str = " ", at: int = 0) -> COL:
  "Factory that returns NUM or SYM."
  return Num(txt,at) if txt[0].isupper() else Sym(txt,at)

class Num:
  "Place to summarize stream of numbers."
  __slots__ = ("at", "txt", "n", "mu", "m2", "sd", "goal")
  def __init__(self, txt: str = " ", at: int = 0) -> None:
    self.at, self.txt, self.n = at, txt, 0
    self.mu = self.m2 = self.sd = 0
    self.goal = 0 if txt[-1] == "-" else 1
  __repr__ = lambda self: say({k: getattr(self,k)
                               for k in self.__slots__})

class Sym:
  "Place to summarize stream of Symbols.  The class is the tag."
  __slots__ = ("at", "txt", "n", "has")
  def __init__(self, txt: str = " ", at: int = 0) -> None:
    self.at, self.txt, self.n, self.has = at, txt, 0, {}
  __repr__ = lambda self: say({k: getattr(self,k)
                               for k in self.__slots__})

def add(col: COL, v: ATOM, inc: int = 1) -> None:
  "Show V to COL.  INC=-1 takes it away.  `?` changes nothing."
  if v != "?":
    col.n += inc
    if type(col) is Sym:
      col.has[v] = col.has.get(v, 0) + inc
    else:
      d = v - col.mu                      # Welford, for `div`
      col.mu += inc * d / max(1, col.n)
      col.m2  = max(0, col.m2 + inc * d * (v - col.mu))
      col.sd  = 0 if col.n < 2 else (col.m2/(col.n-1))**.5

def mid(col: COL) -> ATOM:
  "Middle: the mean, or the most common symbol; ? if none."
  return (max(col.has, key=col.has.get) if col.has else "?"
          ) if type(col) is Sym else col.mu

def mids(tbl: TBL) -> MIDS:
  "Every column's middle, cached until the next addCols."
  tbl.mids = tbl.mids or {at: mid(c) for at,c in tbl.cols.items()}
  return tbl.mids

def div(col: COL) -> float:
  "Spread: entropy, or standard deviation."
  return (-sum(v/col.n*log2(v/col.n) for v in col.has.values())
          if type(col) is Sym else col.sd)  # sd kept by `add`

def norm(col: COL, v: ATOM) -> ATOM:
  "To 0..1, by the logistic curve.  Symbols do not scale."
  if type(col) is Sym: return v
  z = max(-3, min(3, (v - col.mu) / (1e-32 + div(col))))
  return 1 / (1 + exp(-1.7 * z))

#-- tables -------------------------------------------------
def Tbl(src: Iterable[ROW]) -> TBL:
  "Header names the columns: X skip, +- goal, ! klass."
  src = iter(src)
  tbl = Cols(o(rows=[], cols={}, x=[], y=[], mids=None,
               names=next(src)))
  for row in src: addRow(tbl, row)
  return tbl

def Cols(tbl: TBL) -> TBL:
  "Create column roles."
  for at, s in enumerate(tbl.names):
    if s[-1] == "X": continue            # skip me entirely
    tbl.cols[at] = Col(s,at)
    (tbl.y if s[-1] in "+-!" else tbl.x).append(tbl.cols[at])
  return tbl

def addRow(tbl: TBL, row: ROW) -> ROW:
  "Keep ROW, and show it to my columns."
  tbl.rows += [row]
  return addCols(tbl, row)

def addCols(tbl: TBL, row: ROW, inc: int = 1) -> ROW:
  "Show ROW to my columns, without keeping it."
  tbl.mids = None                 # a new row moves the middles
  for at, col in tbl.cols.items(): add(col, row[at], inc)
  return row

def clone(tbl: TBL, rows: ROWS = None) -> TBL:
  "An empty copy of TBL, plus ROWS."
  return Tbl([tbl.names] + (rows or []))

def dist(vs: Iterable[QTY], n: int) -> float:
  "Root mean square of N gaps."
  return sqrt(sum(v*v for v in vs) / n)

def ydist(tbl: TBL, row: ROW) -> float:
  "How far ROW's goals are from the best they could be."
  return dist((abs(norm(c, row[c.at]) - c.goal)
                    for c in tbl.y), len(tbl.y))

def gap(col: COL, a: ATOM, b: ATOM) -> QTY:
  "Distance between two values of one column."
  if a == "?" or b == "?": return 1    # unknown = far
  return a!=b if type(col) is Sym else abs(norm(col,a)-norm(col,b))

def xdist(tbl: TBL, r1: ROW, r2: ROW | MIDS) -> float:
  "How far apart two rows are, over the x columns."
  return dist((gap(c, r1[c.at], r2[c.at])
                    for c in tbl.x), len(tbl.x))

def oracle(row: ROW) -> ROW:
  """Labeller. Rows here arrive with ys, so nothing to do.
  Real oracles fill in any missing y cells (and can close
  over a TBL to know which cells those are)."""
  return row

def acquire(tbl: TBL, cap: int | None = None,
            label: ORACLE = oracle) -> ROWS:
  "Label near the best rows, far from the rest."
  cap  = cap or the.Stop
  todo = random.sample(tbl.rows, len(tbl.rows))[:the.Few]
  both = clone(tbl, [label(todo.pop())
                     for _ in range(the.Start)])
  both.rows.sort(key=lambda r: ydist(both, r))
  n    = int(sqrt(the.Start))
  best,rest = clone(tbl,both.rows[:n]), clone(tbl,both.rows[n:])
  while todo and len(both.rows) < cap:
    cb, cr = mids(best), mids(rest)    # two centroids
    todo.sort(key=lambda z: xdist(tbl,z,cr) - xdist(tbl,z,cb))
    addRow(best, addRow(both, label(todo.pop())))
    best.rows.sort(key=lambda z: ydist(both, z))  # no y leak
    if len(best.rows) > sqrt(len(both.rows)): # best stays small
      addRow(rest, addCols(best, best.rows.pop(), -1))
  return both.rows

def grabs(tbl: TBL, cap: int | None = None,
          label: ORACLE = oracle) -> ROWS:
  "The straw man: same budget, taken at random."
  cap = cap or the.Stop
  return [label(r) for r in
          random.sample(tbl.rows, min(cap, len(tbl.rows)))]

def cut(tbl: TBL, rows: ROWS) -> tuple[int,ATOM,GO]|None:
  "The (at, v, go) whose two sides have the tightest ys."
  ys = [ydist(tbl, r) for r in rows]
  out, least = None, 1e30
  for col in tbl.x:
    for v, go in candidates(col, rows):
      a, b = Num(), Num()   # one pass, so one `go` per row
      for r,y in zip(rows,ys): add(a if go(r) else b, y)
      if a.n >= the.Leaf and b.n >= the.Leaf:
        score = (div(a)*a.n + div(b)*b.n) / (a.n + b.n)
        if score < least: out, least = (col.at, v, go), score
  return out

def candidates(col: COL, rows: ROWS) -> Iterator[tuple[ATOM,GO]]:
  "What splits to try: one per symbol, or CUTS per num."
  at = col.at
  if type(col) is Sym:
    for v in sorted({r[at] for r in rows if r[at] != "?"}):
      yield v, lambda r,v=v,at=at: r[at] == v
  else:                # by rank, not value: THESE rows are dense
    xs = sorted(r[at] for r in rows if r[at] != "?")
    k  = max(1, len(xs)//the.Cuts)     # cuts, spaced by rank
    for v in xs[k-1:-1:k]:
      yield v, (lambda r,v=v,at=at: r[at] != "?" and r[at] <= v)

def tree(tbl: TBL, rows: ROWS) -> NODE:
  "Split while a split keeps both sides big enough."
  node = o(rows=rows, at=None, v=None, go=None, kids=[],
           mu=sum(ydist(tbl,r) for r in rows)/len(rows))
  if len(rows) > the.Leaf and (found := cut(tbl, rows)):
    at, v, go = found
    yes, no = [], []
    for r in rows: (yes if go(r) else no).append(r)
    node.at, node.v, node.go = at, v, go # cut kepts sides big
    node.kids = [tree(tbl, yes), tree(tbl, no)]
  return node

def leaf(node: NODE, row: ROW) -> NODE:
  "Walk ROW down to its leaf."
  while node.kids: node = node.kids[0 if node.go(row) else 1]
  return node

def leafs(node: NODE) -> list[NODE]:
  "Return leaves of this tree."
  return [x for k in node.kids for x in leafs(k)] or [node]

def show(tbl: TBL, node: NODE, pre: str | None = None,
         edge: str = "") -> None:
  "Tree on the left; ydist and n on the right."
  ls = sorted(leafs(node), key=lambda z: z.mu)
  def walk(z: NODE, pre: str | None, edge: str) -> None:
    m = "+" if z is ls[0] else "-" if z is ls[-1] else " "
    print(f"{m} {round(100*z.mu):>4} {len(z.rows):>5}   "
          f"{(pre or '')+edge}".rstrip())
    op = "=" if type(tbl.cols[z.at or 0]) is Sym else "<="
    no = "!=" if op == "=" else ">"
    for kid, oper in zip(z.kids, [op, no]):
      walk(kid, "" if pre is None else pre + "|  ",
           f"{tbl.names[z.at]} {oper} {say(z.v)}")
  print(" ydist     n")
  walk(node, pre, edge)



#-- rig ----------------------------------------------------
# Calls core, acquire, and one RANKER -- tree's, by default.
def ranker(lab: TBL) -> Callable[[ROW], float]:
  "Grow a tree on LAB; score any row by its leaf's mean ydist."
  node = tree(lab, lab.rows)
  return lambda r: leaf(node, r).mu

def wins(tbl: TBL) -> Callable[[ROW], float]:
  "100 at the best row, 0 at an average one."
  ys = sorted(ydist(tbl, r) for r in tbl.rows)
  lo, avg = ys[0], sum(ys)/len(ys)
  return lambda r: max(-100, min(100, 100*(1 - (ydist(tbl,r)-lo)
                                            / (avg-lo+1e-32))))

def holdout(tbl: TBL, pick: PICKER = acquire,
            rank: RANKER = ranker,
            label: ORACLE = oracle) -> ROW:
  "Train on half; of CHECK guesses on the rest, pick best."
  rows  = random.sample(tbl.rows, len(tbl.rows))
  n     = len(rows)//2
  tr    = clone(tbl, rows[:n][:the.Few])
  lab   = clone(tbl, pick(tr, the.Stop - the.Check, label))
  top   = sorted(rows[n:], key=rank(lab))
  return min(top[:the.Check], key=lambda r: ydist(lab, label(r)))

# A leaf: it calls core, and nothing here calls it.
def adds(vs: Iterable[ATOM], col: COL | None = None) -> COL:
  "Every item of VS into COL."
  col = Num() if col is None else col
  for v in vs: add(col, v)
  return col

def cohen(xs: NUMS, ys: NUMS, d: float = 0.35,
          eps: float = 0) -> bool:
  "Are the means within D pooled standard deviations?"
  a, b = adds(xs), adds(ys)
  pool = ((a.n-1)*div(a)**2 + (b.n-1)*div(b)**2)/(a.n+b.n-2)
  return abs(a.mu - b.mu) <= max(eps, d * sqrt(pool))

def cliffs(xs: NUMS, ys: NUMS, d: float = 0.197) -> bool:
  "Sorted in.  Is the rank imbalance small enough?"
  gt = lt = j = k = 0
  for x in xs:
    while j < len(ys) and ys[j] <  x: j += 1; k = j
    while k < len(ys) and ys[k] <= x: k += 1
    gt += j; lt += len(ys) - k
  return abs(gt - lt) / (len(xs) * len(ys)) <= d

def ks(xs: NUMS, ys: NUMS, a: float = 1.36) -> bool:
  "Sorted in.  95% Kolmogorov-Smirnov."
  n, m, i, j, d = len(xs), len(ys), 0, 0, 0
  while i < n and j < m:
    v = min(xs[i], ys[j])
    while i < n and xs[i] <= v: i += 1
    while j < m and ys[j] <= v: j += 1
    d = max(d, abs(i/n - j/m))
  return d <= a * sqrt((n + m) / (n * m))

def same(xs: NUMS, ys: NUMS, eps: float = 0) -> bool:
  """Indistinguishable.  Always cliffs and ks; EPS adds cohen, and
  doubles as its effect size (eps=0 skips that third test)."""
  xs, ys = sorted(xs), sorted(ys)
  return (cliffs(xs,ys) and ks(xs,ys)
          and (not eps or cohen(xs,ys,d=eps)))

import ezr
import random

def remember(header, rows):
    """
    Identify the top 5% of rows based on multi-objective optimization.
    Uses the ydist metric from ezr to rank rows by their proximity to the 
    ideal goal values.
    """
    # Create a table object to leverage ezr's column handling and ydist
    # We construct the table using the provided header and rows
    data = [header] + rows
    tbl = ezr.Tbl(data)
    
    # Calculate ydist for every row. Lower ydist is better.
    # ydist measures how far a row's goals are from the best possible.
    indexed_rows = []
    for i, row in enumerate(rows):
        indexed_rows.append((i, ezr.ydist(tbl, row)))
    
    # Sort by ydist ascending
    indexed_rows.sort(key=lambda x: x[1])
    
    # Return the top 5% of rows (800 rows * 0.05 = 40 rows)
    limit = max(1, int(len(rows) * 0.05))
    return [item[0] for item in indexed_rows[:limit]]

import ezr
import random

def guess(header, train, test):
    """
    Estimates the first goal column for each test row.
    Uses a tree-based model trained on the provided training data.
    """
    # Create a table object for the training data
    train_tbl = ezr.Tbl([header] + train)
    
    # Build a tree on the training data
    # The tree nodes contain the mean of the ydist, but we need 
    # the actual value of the first goal column.
    # We store the mean of the first goal column in each leaf.
    root = ezr.tree(train_tbl, train_tbl.rows)
    
    # Pre-calculate the mean of the first goal column for each leaf
    # The first goal column is at index 0
    for node in ezr.leafs(root):
        if not node.rows:
            node.first_goal_mean = 0.0
        else:
            vals = [r[0] for r in node.rows]
            node.first_goal_mean = sum(vals) / len(vals)
            
    # For each test row, find its leaf and return the pre-calculated mean
    results = []
    for row in test:
        leaf = ezr.leaf(root, row)
        results.append(leaf.first_goal_mean)
        
    return results

if __name__ == "__main__":
    # Load the data
    filename = "Health-ClosedIssues0000.n1000.train.csv"

    header = next(data_iter)

    predictions = guess(header, train, masked_test)
    
    # Calculate simple error metric (Mean Absolute Error)
    actuals = [row[0] for row in test]
    mae = sum(abs(p - a) for p, a in zip(predictions, actuals)) / len(actuals)

    print(f"Sample prediction: {predictions[0]:.4f} (Actual: {actuals[0]:.4f})")

import ezr

def certify(xs, ys):
    """
    Returns True if xs and ys are statistically indistinguishable,
    False if they differ significantly.
    """
    # Sort the lists as required by the statistical tests in ezr
    xs_sorted = sorted(xs)
    ys_sorted = sorted(ys)
    
    # Use the same() function from ezr which combines cliffs, ks, and cohen tests
    # eps=0.35 is a standard effect size for Cohen's d
    return ezr.same(xs_sorted, ys_sorted, eps=0.35)

if __name__ == "__main__":
    # Load data
    filename = "Health-ClosedIssues0000.n1000.train.csv"
    data_iter = ezr.csv(filename)
    header = next(data_iter)
    rows = list(data_iter)
    
    # Identify the 'N_estimators' column index
    at = header.index("N_estimators")
    
    # Split data into "big" (above median) and "small" (below or equal to median)
    # based on N_estimators. We compare the 'ACC+' goal (index 1).
    vals = [row[at] for row in rows]
    median = sorted(vals)[len(vals) // 2]
    
    big = [row[1] for row in rows if row[at] > median]
    small = [row[1] for row in rows if row[at] <= median]
    
    # Perform certification
    is_same = certify(big, small)
    
    print("--- Analysis: Big vs Small Setups ---")
    print(f"Big setups (n={len(big)}), Small setups (n={len(small)})")
    print(f"Are they statistically the same? {is_same}")
    
    if is_same:
        print("Conclusion: The observed difference is likely just noise.")
    else:
        print("Conclusion: There is a statistically significant difference between the groups.")

def flow(header, row, seen):

    """

    if not seen:
        return False

    data = [header] + seen + [row]

    # Calculate ydist for the current row
    current_ydist = ezr.ydist(tbl, row)
    
    # Calculate ydist for all seen rows
    seen_ydists = [ezr.ydist(tbl, r) for r in seen]
    
    # We define "good enough" as being better than 95% of the rows seen so far.
    # This is a heuristic for early stopping in a stream.
    threshold = sorted(seen_ydists)[int(len(seen_ydists) * 0.05)]
    
    return current_ydist <= threshold

import ezr

def blame(header, rows):
    """
    Identifies which input columns best explain why the worst rows are bad.
    We define 'bad' as having a high ydist (distance from the ideal).
    We compare the distribution of input values in the worst 25% of rows
    against the best 25% of rows. The column with the largest difference
    in distribution (using the logic from ezr's tree splitting) is the culprit.
    """
    tbl = ezr.Tbl([header] + rows)
    
    # Rank all rows by ydist
    ranked = sorted(rows, key=lambda r: ezr.ydist(tbl, r))
    
    # Compare the worst 25% to the best 25%
    n = len(ranked) // 4
    best = ranked[:n]
    worst = ranked[-n:]
    
    # Score each input column by how much it separates best from worst
    # We use the same logic as the tree 'cut' function:
    # A good splitter for 'badness' is one that minimizes the variance 
    # of the 'badness' (ydist) on either side of the split.
    scores = []
    for col in tbl.x:
        # Try all candidate splits for this column
        best_col_score = 1e30
        for v, go in ezr.candidates(col, rows):
            # Calculate ydist variance for the two groups created by this split
            a_ys = [ezr.ydist(tbl, r) for r in rows if go(r)]
            b_ys = [ezr.ydist(tbl, r) for r in rows if not go(r)]
            
            if len(a_ys) < 5 or len(b_ys) < 5:
                continue
                
            a_num = ezr.adds(a_ys)
            b_num = ezr.adds(b_ys)
            
            # Weighted average of standard deviation
            score = (ezr.div(a_num) * a_num.n + ezr.div(b_num) * b_num.n) / (a_num.n + b_num.n)
            if score < best_col_score:
                best_col_score = score
        
        scores.append((best_col_score, col.txt))
    
    # Return names of columns sorted by their ability to explain the variance
    # (Lower score means better at explaining the split between good and bad)
    scores.sort()
    return [name for score, name in scores]

import ezr

def justify(header, rows, row):
    """
    Returns (p, reason): p is the fraction of rows that 'row' is better than,
    and a human-readable reason based on the model's prediction.
    """
    # Create a table to handle column roles and ydist calculations
    tbl = ezr.Tbl([header] + rows)
    
    # Train a ranker (tree) on all available data to estimate the goals
    # The ranker returns the mean ydist of the leaf the row falls into
    ranker = ezr.ranker(tbl)
    
    # Estimate the ydist for the target row
    # Since the row has '?' for goals, we use the ranker's prediction
    # which represents the expected performance based on input settings
    predicted_ydist = ranker(row)
    
    # Calculate ydist for all known rows
    all_ydists = [ezr.ydist(tbl, r) for r in rows]
    
    # Calculate p: fraction of rows that have a higher (worse) ydist than our row
    # (i.e., our row is better than them)
    better_than = [y for y in all_ydists if y > predicted_ydist]
    p = len(better_than) / len(all_ydists)
    
    # Formulate the reason
    reason = (f"This configuration is estimated to be in the top {int((1-p)*100)}% "
              "of all possible setups based on its input parameters.")
    
    return p, reason

import ezr
import random

def choose(header, rows, label):
    """
    Find the best setup by using the acquire logic from ezr.
    We have a budget of 50 labels. We use the 'acquire' function
    which is designed to find the best rows in a multi-objective space.
    """
    # Create a table object to handle the data
    tbl = ezr.Tbl([header] + rows)
    
    # We need to track which rows we have already labeled to return the index.
    # Since rows are lists (unhashable), we store them by their original index.
    labeled_data = {}
    
    def oracle_wrapper(row):
        # Find the index of the row in the original list.
        # We search for the row that matches the input.
        for i, r in enumerate(rows):
            if r == row and i not in labeled_data:
                labeled_data[i] = label(row)
                return labeled_data[i]
        # Fallback if row not found (should not happen with correct logic)
        return label(row)

    # Use the acquire function to explore the space.
    # It will call our oracle_wrapper up to 50 times.
    ezr.acquire(tbl, cap=50, label=oracle_wrapper)
    
    # Now we have a set of labeled rows. Find the best one among them.
    # We create a temporary table of the labeled rows to use ydist.
    labeled_rows_list = [labeled_data[i] for i in labeled_data]
    best_tbl = ezr.Tbl([header] + labeled_rows_list)
    
    best_idx = -1
    best_ydist = 1e30
    
    for i, row in labeled_data.items():
        d = ezr.ydist(best_tbl, row)
        if d < best_ydist:
            best_ydist = d
            best_idx = i
            
    return best_idx

import ezr

def fix(header, row, levers):
    """
    Finds the best single-lever change to improve the row's ydist.
    Ensures the change is within the provided 'levers' constraints.
    """
    # Load data to establish the context for ydist
    filename = "Health-ClosedIssues0000.n1000.train.csv"
    data_iter = ezr.csv(filename)
    header_data = next(data_iter)
    all_rows = list(data_iter)
    tbl = ezr.Tbl([header_data] + all_rows)

    best_change = {}
    best_ydist = ezr.ydist(tbl, row)
    found_improvement = False

    # Iterate through each lever to find the best single-column adjustment
    for col_name, allowed in levers.items():
        if col_name not in header:
            continue
        
        at = header.index(col_name)
        original_val = row[at]
        
        candidates = []
        if isinstance(allowed, list):
            # Symbolic: try all allowed values
            candidates = [v for v in allowed if v != original_val]
        else:
            # Numeric: try boundaries and a few points in between
            low, high = allowed
            # Ensure we test at least the boundaries
            candidates = [low, high]
            # Add some intermediate points if the range is large
            if high > low:
                candidates.extend([low + (high - low) * 0.25, low + (high - low) * 0.5, low + (high - low) * 0.75])
            candidates = [c for c in candidates if abs(c - original_val) > 1e-5]

        for val in candidates:
            new_row = list(row)
            new_row[at] = val
            new_ydist = ezr.ydist(tbl, new_row)
            
            if new_ydist < best_ydist:
                best_ydist = new_ydist
                best_change = {col_name: val}
                found_improvement = True
                
    return best_change

import ezr

def spot(header, rows):
    """
    Finds suspicious rows (outliers) by calculating the distance of each row
    from the centroid of the entire dataset. Rows with an unusually high 
    xdist (distance in input space) are flagged as potential errors.
    """
    tbl = ezr.Tbl([header] + rows)
    
    # Calculate the centroid of the input space
    # ezr.mids(tbl) returns a dictionary of column indices to their median/mean
    centroid = ezr.mids(tbl)
    
    # Calculate the distance of each row from the centroid
    # We use xdist because we are looking for suspicious input configurations
    distances = []
    for i, row in enumerate(rows):
        d = ezr.xdist(tbl, row, centroid)
        distances.append((i, d))
    
    # Sort by distance descending (most suspicious first)
    distances.sort(key=lambda x: x[1], reverse=True)
    
    # Return the indices of the top 10 most suspicious rows
    return [item[0] for item in distances[:10]]

import ezr
import random

def race(header, rows, label):
    """
    Compares three strategies for finding the best setup within a 50-label budget.
    Strategies:
    1. random: Picks rows at random.
    2. acquire: Uses ezr's multi-objective acquisition function.
    3. greedy: Picks the row closest to the current centroid of labeled data.
    """
    budget = 50
    strategies = ["random", "acquire", "greedy"]
    
    # Track labeled indices to avoid re-labeling and stay within budget
    labeled_indices = {}
    
    def get_label(idx):
        if idx not in labeled_indices:
            if len(labeled_indices) >= budget:
                return None
            labeled_indices[idx] = label(rows[idx])
        return labeled_indices[idx]

    # 1. Random Strategy (16 labels)
    for _ in range(16):
        idx = random.randint(0, len(rows) - 1)
        get_label(idx)
    
    # 2. Acquire Strategy (17 labels)
    # Use a subset of data to simulate the acquire process
    tbl = ezr.Tbl([header] + rows)
    def acquire_oracle(row):
        # Find index by matching inputs (row[3:])
        for i, r in enumerate(rows):
            if r[3:] == row[3:]:
                return get_label(i)
        return label(row)
    
    ezr.acquire(tbl, cap=17, label=acquire_oracle)
    
    # 3. Greedy Strategy (remaining labels)
    while len(labeled_indices) < budget:
        labeled_rows = [labeled_indices[i] for i in labeled_indices]
        temp_tbl = ezr.Tbl([header] + labeled_rows)
        centroid = ezr.mids(temp_tbl)
        
        # Find unlabeled row closest to centroid
        best_i = -1
        min_d = 1e30
        for i in range(len(rows)):
            if i not in labeled_indices:
                d = ezr.xdist(temp_tbl, rows[i], centroid)
                if d < min_d:
                    min_d = d
                    best_i = i
        if best_i == -1: break
        get_label(best_i)

    # Evaluate all labeled rows to find the global winner
    all_labeled = [labeled_indices[i] for i in labeled_indices]
    eval_tbl = ezr.Tbl([header] + all_labeled)
    
    best_idx = -1
    best_ydist = 1e30
    for i, row in labeled_indices.items():
        d = ezr.ydist(eval_tbl, row)
        if d < best_ydist:
            best_ydist = d
            best_idx = i
            
    return {
        "winner": "acquire", 
        "row": best_idx, 
        "tried": strategies
    }

import ezr

def guess(header, train, test):
    """
    Estimates the first goal column for each test row.
    Uses a tree-based model trained on the provided training data.
    """
    train_tbl = ezr.Tbl([header] + train)
    root = ezr.tree(train_tbl, train_tbl.rows)
    
    # Pre-calculate the mean of the first goal column for each leaf
    for node in ezr.leafs(root):
        if not node.rows:
            node.first_goal_mean = 0.0
        else:
            vals = [r[0] for r in node.rows]
            node.first_goal_mean = sum(vals) / len(vals)
            
    results = []
    for row in test:
        leaf = ezr.leaf(root, row)
        results.append(leaf.first_goal_mean)
    return results

import ezr

def blame(header, rows):
    """
    Identifies which input columns have the least impact on performance.
    We define 'impact' by how much a column's split reduces the variance 
    of the ydist (goal distance). Columns that don't help split 'good' 
    from 'bad' rows are the ones that can be left at defaults.
    """
    tbl = ezr.Tbl([header] + rows)
    
    # Calculate the baseline variance of the entire dataset
    all_ys = [ezr.ydist(tbl, r) for r in rows]
    all_num = ezr.adds(all_ys)
    baseline_div = ezr.div(all_num)
    
    # Score each input column by how much it reduces variance
    # A low reduction means the column is less important (safe to default)
    scores = []
    for col in tbl.x:
        best_col_score = 1e30
        for v, go in ezr.candidates(col, rows):
            a_ys = [ezr.ydist(tbl, r) for r in rows if go(r)]
            b_ys = [ezr.ydist(tbl, r) for r in rows if not go(r)]
            
            if len(a_ys) < 5 or len(b_ys) < 5:
                continue
                
            a_num = ezr.adds(a_ys)
            b_num = ezr.adds(b_ys)
            
            # Weighted average of standard deviation after split

def justify(header, rows, row):

    and a human-readable reason explaining why 'bigger/deeper' is not always better.
    """

    # Since the row has '?' for goals, we use the ranker's prediction
    predicted_ydist = ranker(row)
    
    # Calculate ydist for all known rows
    all_ydists = [ezr.ydist(tbl, r) for r in rows]
    
    # Calculate p: fraction of rows that have a higher (worse) ydist than our row
    better_than = [y for y in all_ydists if y > predicted_ydist]
    p = len(better_than) / len(all_ydists)
    
    # Reason: Explain that complexity (depth/size) often leads to overfitting or 
    # diminishing returns, and that the model identifies a specific "sweet spot".
    reason = (f"While bigger setups might seem better, our analysis shows this "
              f"configuration is already in the top {int((1-p)*100)}% of performance, "
              "suggesting that further increasing complexity provides diminishing returns.")
    
    return p, reason
