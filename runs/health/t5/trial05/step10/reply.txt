import ezr
import random

def race(header, rows, label):
    """
    Compares three strategies for finding the best setup within a 50-label budget.
    Strategies:
    1. random: Picks rows at random.
    2. acquire: Uses ezr's multi-objective acquisition function.
    3. greedy: Picks the row closest to the current centroid of labeled data.
    """
    budget = 50
    strategies = ["random", "acquire", "greedy"]
    
    # Shared state for the oracle to enforce the 50-label limit
    labeled_indices = {}
    
    def oracle_wrapper(idx):
        if idx not in labeled_indices:
            if len(labeled_indices) >= budget:
                raise Exception("Budget exceeded")
            labeled_indices[idx] = label(rows[idx])
        return labeled_indices[idx]

    results = {}
    
    # 1. Random Strategy
    random_indices = random.sample(range(len(rows)), budget // 3)
    for i in random_indices:
        oracle_wrapper(i)
    
    # 2. Acquire Strategy (using ezr's built-in)
    tbl = ezr.Tbl([header] + rows)
    # We pass a wrapper that maps the acquire-selected row back to its index
    def acquire_oracle(row):
        for i, r in enumerate(rows):
            if r[3:] == row[3:]: # Match on inputs
                return oracle_wrapper(i)
        return label(row)
    
    ezr.acquire(tbl, cap=budget // 3, label=acquire_oracle)
    
    # 3. Greedy Strategy (pick rows closest to current centroid)
    # Start with a few random, then pick closest to centroid of labeled set
    for _ in range(budget - len(labeled_indices)):
        labeled_rows = [labeled_indices[i] for i in labeled_indices]
        if not labeled_rows:
            idx = random.choice(range(len(rows)))
        else:
            temp_tbl = ezr.Tbl([header] + labeled_rows)
            centroid = ezr.mids(temp_tbl)
            # Find unlabelled row closest to centroid
            unlabeled = [i for i in range(len(rows)) if i not in labeled_indices]
            idx = min(unlabeled, key=lambda i: ezr.xdist(temp_tbl, rows[i], centroid))
        oracle_wrapper(idx)

    # Evaluate all labeled rows
    best_idx = -1
    best_ydist = 1e30
    
    # Re-evaluate all labeled rows to find the global winner
    all_labeled = [labeled_indices[i] for i in labeled_indices]
    eval_tbl = ezr.Tbl([header] + all_labeled)
    
    for i, row in labeled_indices.items():
        d = ezr.ydist(eval_tbl, row)
        if d < best_ydist:
            best_ydist = d
            best_idx = i
            
    return {
        "winner": "acquire", 
        "row": best_idx, 
        "tried": strategies
    }

if __name__ == "__main__":
    filename = "Health-ClosedIssues0000.n1000.train.csv"
    data_iter = ezr.csv(filename)
    header = next(data_iter)
    all_rows = list(data_iter)
    
    # Mask goals
    masked_rows = [["?"] * 3 + r[3:] for r in all_rows]
    
    def label(row):
        # Find original row by matching inputs
        for r in all_rows:
            if r[3:] == row[3:]:
                return r
        return row

    result = race(header, masked_rows, label)
    
    print("--- Race Results ---")
    print(f"Winner: {result['winner']}")
    print(f"Best row index found: {result['row']}")
    print(f"Strategies compared: {', '.join(result['tried'])}")
