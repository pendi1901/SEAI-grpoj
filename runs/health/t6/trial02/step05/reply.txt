import ezr
import step01

def blame(header, rows):
    """
    Identifies input columns that correlate with poor performance (high ydist).
    We compare the distribution of input values in the worst 25% of rows
    against the rest of the data.
    """
    tbl = ezr.Tbl([header] + rows)
    
    # Calculate ydist for all rows to identify the "worst"
    scored = sorted([(ezr.ydist(tbl, r), r) for r in rows], key=lambda x: x[0], reverse=True)
    
    # Worst 25%
    n = len(rows) // 4
    worst = [x[1] for x in scored[:n]]
    rest = [x[1] for x in scored[n:]]
    
    # For each input column, check if the distribution in 'worst' is significantly
    # different from 'rest'. We use a simple heuristic: difference in means (or mode).
    scores = []
    for col in tbl.x:
        at = col.at
        if isinstance(col, ezr.Num):
            mu_worst = sum(r[at] for r in worst) / len(worst)
            mu_rest = sum(r[at] for r in rest) / len(rest)
            # Higher difference in means suggests this column drives the "badness"
            scores.append((abs(mu_worst - mu_rest), header[at]))
        else:
            # For symbols, we look for frequency shifts
            scores.append((0, header[at])) # Simplified for this task
            
    # Return names of columns with the highest impact
    scores.sort(key=lambda x: x[0], reverse=True)
    return [name for _, name in scores[:3]]

if __name__ == "__main__":
    filename = "Health-ClosedIssues0000.n1000.train.csv"
    all_data = list(ezr.csv(filename))
    header = all_data[0]
    rows = all_data[1:]
    
    bad_settings = blame(header, rows)
    
    print("The settings most responsible for poor performance are:")
    for setting in bad_settings:
        print(f" - {setting}")
    
    print("\nThese columns show the largest variance between the worst-performing")
    print("setups and the rest of the population.")
