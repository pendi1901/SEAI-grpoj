import ezr
import step01

def guess(header, train, test):
    """
    Estimates the first goal column for each test row by finding the 
    nearest neighbor in the training set based on x-columns.
    """
    # Create a table from training data to use ezr's distance functions
    train_tbl = ezr.Tbl([header] + train)
    
    # The first goal column is the one at index 0 (MRE-)
    # We want to estimate this value for each row in test
    results = []
    for t_row in test:
        # Find the row in training data that is closest in x-space
        # xdist calculates distance based on non-goal columns
        best_row = min(train, key=lambda r: ezr.xdist(train_tbl, t_row, r))
        
        # Return the value of the first goal column from that nearest neighbor
        results.append(best_row[0])
        
    return results

if __name__ == "__main__":
    # Load data
    filename = "Health-ClosedIssues0000.n1000.train.csv"
    all_data = list(ezr.csv(filename))
    header = all_data[0]
    rows = all_data[1:]
    
    # Split: use 80% for training, 20% for testing
    split_idx = int(len(rows) * 0.8)
    train = rows[:split_idx]
    test_full = rows[split_idx:]
    
    # Mask the goals in the test set to simulate "unknown"
    # Goals are columns ending in + or -
    goal_indices = [i for i, h in enumerate(header) if h[-1] in "+-"]
    test = []
    for row in test_full:
        new_row = list(row)
        for idx in goal_indices:
            new_row[idx] = "?"
        test.append(new_row)
    
    # Get estimates
    estimates = guess(header, train, test)
    
    # Calculate average error for the first goal (MRE-)
    actuals = [r[0] for r in test_full]
    errors = [abs(e - a) for e, a in zip(estimates, actuals)]
    avg_error = sum(errors) / len(errors)
    
    print(f"Estimated MRE- for {len(test)} test rows.")
    print(f"Average absolute error in MRE- forecast: {avg_error:.4f}")
