import ezr
import random

def race(header, rows, label):
    """
    Compares three strategies for finding the best setup:
    1. random: Randomly samples rows.
    2. acquire: Uses the ezr.acquire logic (active learning).
    3. greedy: Picks rows based on proximity to the current best-known configuration.
    """
    budget = 50
    per_strategy = budget // 3
    
    # Track labeled rows to ensure we don't exceed the global budget
    labeled_indices = set()
    
    def oracle_wrapper(idx):
        if idx not in labeled_indices:
            if len(labeled_indices) >= budget:
                return rows[idx] # Return masked if budget exceeded
            labeled_indices.add(idx)
            true_vals = label(idx)
            for i, val in enumerate(true_vals):
                rows[idx][i] = val
        return rows[idx]

    # We need a table that knows the true values for ydist calculations.
    # Since we are updating rows in-place, we create a fresh Tbl 
    # only when we need to evaluate performance.
    
    results = {}
    
    # 1. Random Strategy
    random_indices = random.sample(range(len(rows)), per_strategy)
    for i in random_indices:
        oracle_wrapper(i)
    
    # 2. Acquire Strategy
    # We use a temporary table for the acquire logic
    tbl = ezr.Tbl([header] + rows)
    acq_rows = ezr.acquire(tbl, cap=per_strategy, label=oracle_wrapper)
    
    # 3. Greedy Strategy
    greedy_indices = []
    curr = random.choice(range(len(rows)))
    for _ in range(per_strategy):
        greedy_indices.append(curr)
        oracle_wrapper(curr)
        # Find next best by looking for the closest neighbor in x-space
        curr = min(range(len(rows)), key=lambda i: ezr.xdist(tbl, rows[curr], rows[i]))
    
    # Final evaluation table
    final_tbl = ezr.Tbl([header] + rows)
    
    results["random"] = min(random_indices, key=lambda i: ezr.ydist(final_tbl, rows[i]))
    results["acquire"] = min([rows.index(r) for r in acq_rows], key=lambda i: ezr.ydist(final_tbl, rows[i]))
    results["greedy"] = min(greedy_indices, key=lambda i: ezr.ydist(final_tbl, rows[i]))
    
    # Determine winner
    winner = min(results, key=lambda name: ezr.ydist(final_tbl, rows[results[name]]))
    
    return {
        "winner": winner,
        "row": results[winner],
        "tried": ["random", "acquire", "greedy"]
    }

if __name__ == "__main__":
    filename = "Health-ClosedIssues0000.n1000.train.csv"
    all_data = list(ezr.csv(filename))
    header = all_data[0]
    rows = all_data[1:]
    
    # Keep a copy of original rows to look up true values
    original_rows = [list(r) for r in rows]
    
    # Mask goals
    goal_indices = [i for i, h in enumerate(header) if h[-1] in "+-"]
    for row in rows:
        for idx in goal_indices:
            row[idx] = "?"
                
    def label(i):
        return original_rows[i]
    
    outcome = race(header, rows, label)
    print(f"Race Results: {outcome}")
    print(f"The winning strategy was '{outcome['winner']}' which found a top setup at index {outcome['row']}.")
