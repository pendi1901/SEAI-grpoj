```python
import ezr
import step02

def guess(header, train, test):
    """
    Estimates the first goal column for each test row by finding the 
    nearest neighbor in the training set based on x-columns.
    """
    return step02.guess(header, train, test)

def evaluate_impact(header, rows):
    """
    Compares the forecast error of a full-sized training set vs a halved one.
    """
    # Split data
    split_idx = int(len(rows) * 0.8)
    train_full = rows[:split_idx]
    test_full = rows[split_idx:]
    
    # Create test set with masked goals
    goal_indices = [i for i, h in enumerate(header) if h[-1] in "+-"]
    test = []
    for row in test_full:
        new_row = list(row)
        for idx in goal_indices:
            new_row[idx] = "?"
        test.append(new_row)
    
    # Full size forecast
    est_full = guess(header, train_full, test)
    
    # Half size forecast
    train_half = train_full[:len(train_full) // 2]
    est_half = guess(header, train_half, test)
    
    # Calculate errors
    actuals = [r[0] for r in test_full]
    err_full = sum(abs(e - a) for e, a in zip(est_full, actuals)) / len(actuals)
    err_half = sum(abs(e - a) for e, a in zip(est_half, actuals)) / len(actuals)
    
    return err_full, err_half

if __name__ == "__main__":
    filename = "Health-ClosedIssues0000.n1000.train.csv"
    all_data = list(ezr.csv(filename))
    header = all_data[0]
    rows = all_data[1:]
    
    err_full, err_half = evaluate_impact(header, rows)
    
    print(f"Full training set average error: {err_full:.4f}")
    print(f"Half training set average error: {err_half:.4f}")
    print(f"Performance degradation: {((err_half - err_full) / err_full) * 100:.2f}%")
```
