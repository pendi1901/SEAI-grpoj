import ezr
import step02
import step06

def justify(header, rows, row):
    """
    Answers why we shouldn't just use the biggest, deepest setup.
    We compare the performance of the 'biggest' possible configuration
    (max depth, max estimators) against the distribution of all known rows.
    """
    # Create a table to calculate ydist
    tbl = ezr.Tbl([header] + rows)
    
    # Estimate the goals for the target row using the tree-based guesser
    # We use the guesser from step02 to predict performance
    estimates = step02.guess(header, rows, [row])
    
    # Create a copy of the row with estimated goals
    row_with_goals = list(row)
    # Goals are columns 0, 1, 2
    for i in range(3):
        row_with_goals[i] = estimates[0]
        
    # Calculate ydist for the target row and all other rows
    target_ydist = ezr.ydist(tbl, row_with_goals)
    all_ydists = [ezr.ydist(tbl, r) for r in rows]
    
    # Calculate fraction of rows that are worse (have higher ydist)
    better_than = [d for d in all_ydists if d > target_ydist]
    p = len(better_than) / len(all_ydists)
    
    reason = (f"Using the biggest, deepest setup is often counterproductive because "
              f"it is only better than {p:.1%} of all known configurations, "
              f"suggesting that simpler models often achieve better performance "
              f"without the overhead of excessive complexity.")
    
    return p, reason

if __name__ == "__main__":
    filename = "Health-ClosedIssues0000.n1000.train.csv"
    data = list(ezr.csv(filename))
    header = data[0]
    rows = data[1:]
    
    # Define a "biggest, deepest" setup
    # Based on the dataset: N_estimators (max 180), Max_depth (max 11)
    # We construct a row that represents the "biggest" possible configuration
    # [MRE, ACC, PRED40, N_estimators, criterion, Min_sample_leaves, Min_impurity_decrease, Max_depth]
    big_row = ["?", "?", "?", 200.0, "squared_error", 1.0, 0.0, 20.0]
    
    p, reason = justify(header, rows, big_row)
    
    print("Analysis: Why not use the biggest, deepest setup?")
    print(f"Result: p={p:.4f}")
    print(f"Reason: {reason}")
