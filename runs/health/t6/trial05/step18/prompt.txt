#-- misc ---------------------------------------------------
def say(x: Any, p: int = 2) -> str:
  "Floats get P decimals; dicts list their keys."
  if isinstance(x,float): x= f"{x:.{p}f}".rstrip("0").rstrip(".")
  if isinstance(x, dict):
    x= "{"+", ".join(f"{k}: {say(v,p)}" for k,v in x.items())+"}"
  return str(x)

class o(dict):
  "A dict you can poke with a dot."
  __getattr__, __setattr__ = dict.__getitem__, dict.__setitem__
  __repr__ = say

def atom(s: str) -> ATOM:
  "'22' -> 22. '3.1' -> 3.1. 'True' -> True. 'x' -> 'x'."
  for fn in (int, float):
    try: return fn(s)
    except ValueError: pass
  s = s.strip()
  return {"True": True, "False": False}.get(s, s)

def csv(file: str) -> Iterator[ROW]:
  "ROWS of FILE, each cell coerced."
  with open(os.path.expanduser(file),  # -sig drops any BOM
           encoding="utf-8-sig") as f:
    for line in f:
      if line.strip(): yield [atom(s) for s in line.split(",")]

def Col(txt: str = " ", at: int = 0) -> COL:
  "Factory that returns NUM or SYM."
  return Num(txt,at) if txt[0].isupper() else Sym(txt,at)

class Num:
  "Place to summarize stream of numbers."
  __slots__ = ("at", "txt", "n", "mu", "m2", "sd", "goal")
  def __init__(self, txt: str = " ", at: int = 0) -> None:
    self.at, self.txt, self.n = at, txt, 0
    self.mu = self.m2 = self.sd = 0
    self.goal = 0 if txt[-1] == "-" else 1
  __repr__ = lambda self: say({k: getattr(self,k)
                               for k in self.__slots__})

class Sym:
  "Place to summarize stream of Symbols.  The class is the tag."
  __slots__ = ("at", "txt", "n", "has")
  def __init__(self, txt: str = " ", at: int = 0) -> None:
    self.at, self.txt, self.n, self.has = at, txt, 0, {}
  __repr__ = lambda self: say({k: getattr(self,k)
                               for k in self.__slots__})

def add(col: COL, v: ATOM, inc: int = 1) -> None:
  "Show V to COL.  INC=-1 takes it away.  `?` changes nothing."
  if v != "?":
    col.n += inc
    if type(col) is Sym:
      col.has[v] = col.has.get(v, 0) + inc
    else:
      d = v - col.mu                      # Welford, for `div`
      col.mu += inc * d / max(1, col.n)
      col.m2  = max(0, col.m2 + inc * d * (v - col.mu))
      col.sd  = 0 if col.n < 2 else (col.m2/(col.n-1))**.5

def mid(col: COL) -> ATOM:
  "Middle: the mean, or the most common symbol; ? if none."
  return (max(col.has, key=col.has.get) if col.has else "?"
          ) if type(col) is Sym else col.mu

def mids(tbl: TBL) -> MIDS:
  "Every column's middle, cached until the next addCols."
  tbl.mids = tbl.mids or {at: mid(c) for at,c in tbl.cols.items()}
  return tbl.mids

def div(col: COL) -> float:
  "Spread: entropy, or standard deviation."
  return (-sum(v/col.n*log2(v/col.n) for v in col.has.values())
          if type(col) is Sym else col.sd)  # sd kept by `add`

def norm(col: COL, v: ATOM) -> ATOM:
  "To 0..1, by the logistic curve.  Symbols do not scale."
  if type(col) is Sym: return v
  z = max(-3, min(3, (v - col.mu) / (1e-32 + div(col))))
  return 1 / (1 + exp(-1.7 * z))

#-- tables -------------------------------------------------
def Tbl(src: Iterable[ROW]) -> TBL:
  "Header names the columns: X skip, +- goal, ! klass."
  src = iter(src)
  tbl = Cols(o(rows=[], cols={}, x=[], y=[], mids=None,
               names=next(src)))
  for row in src: addRow(tbl, row)
  return tbl

def Cols(tbl: TBL) -> TBL:
  "Create column roles."
  for at, s in enumerate(tbl.names):
    if s[-1] == "X": continue            # skip me entirely
    tbl.cols[at] = Col(s,at)
    (tbl.y if s[-1] in "+-!" else tbl.x).append(tbl.cols[at])
  return tbl

def addRow(tbl: TBL, row: ROW) -> ROW:
  "Keep ROW, and show it to my columns."
  tbl.rows += [row]
  return addCols(tbl, row)

def addCols(tbl: TBL, row: ROW, inc: int = 1) -> ROW:
  "Show ROW to my columns, without keeping it."
  tbl.mids = None                 # a new row moves the middles
  for at, col in tbl.cols.items(): add(col, row[at], inc)
  return row

def clone(tbl: TBL, rows: ROWS = None) -> TBL:
  "An empty copy of TBL, plus ROWS."
  return Tbl([tbl.names] + (rows or []))

def dist(vs: Iterable[QTY], n: int) -> float:
  "Root mean square of N gaps."
  return sqrt(sum(v*v for v in vs) / n)

def ydist(tbl: TBL, row: ROW) -> float:
  "How far ROW's goals are from the best they could be."
  return dist((abs(norm(c, row[c.at]) - c.goal)
                    for c in tbl.y), len(tbl.y))

def gap(col: COL, a: ATOM, b: ATOM) -> QTY:
  "Distance between two values of one column."
  if a == "?" or b == "?": return 1    # unknown = far
  return a!=b if type(col) is Sym else abs(norm(col,a)-norm(col,b))

def xdist(tbl: TBL, r1: ROW, r2: ROW | MIDS) -> float:
  "How far apart two rows are, over the x columns."
  return dist((gap(c, r1[c.at], r2[c.at])
                    for c in tbl.x), len(tbl.x))

def oracle(row: ROW) -> ROW:
  """Labeller. Rows here arrive with ys, so nothing to do.
  Real oracles fill in any missing y cells (and can close
  over a TBL to know which cells those are)."""
  return row

def acquire(tbl: TBL, cap: int | None = None,
            label: ORACLE = oracle) -> ROWS:
  "Label near the best rows, far from the rest."
  cap  = cap or the.Stop
  todo = random.sample(tbl.rows, len(tbl.rows))[:the.Few]
  both = clone(tbl, [label(todo.pop())
                     for _ in range(the.Start)])
  both.rows.sort(key=lambda r: ydist(both, r))
  n    = int(sqrt(the.Start))
  best,rest = clone(tbl,both.rows[:n]), clone(tbl,both.rows[n:])
  while todo and len(both.rows) < cap:
    cb, cr = mids(best), mids(rest)    # two centroids
    todo.sort(key=lambda z: xdist(tbl,z,cr) - xdist(tbl,z,cb))
    addRow(best, addRow(both, label(todo.pop())))
    best.rows.sort(key=lambda z: ydist(both, z))  # no y leak
    if len(best.rows) > sqrt(len(both.rows)): # best stays small
      addRow(rest, addCols(best, best.rows.pop(), -1))
  return both.rows

def grabs(tbl: TBL, cap: int | None = None,
          label: ORACLE = oracle) -> ROWS:
  "The straw man: same budget, taken at random."
  cap = cap or the.Stop
  return [label(r) for r in
          random.sample(tbl.rows, min(cap, len(tbl.rows)))]

# Calls core.  Nothing above this line calls down here.
# NODE = o(rows, at, v, go, kids, mu).
def cut(tbl: TBL, rows: ROWS) -> tuple[int,ATOM,GO]|None:
  "The (at, v, go) whose two sides have the tightest ys."
  ys = [ydist(tbl, r) for r in rows]
  out, least = None, 1e30
  for col in tbl.x:
    for v, go in candidates(col, rows):
      a, b = Num(), Num()   # one pass, so one `go` per row
      for r,y in zip(rows,ys): add(a if go(r) else b, y)
      if a.n >= the.Leaf and b.n >= the.Leaf:
        score = (div(a)*a.n + div(b)*b.n) / (a.n + b.n)
        if score < least: out, least = (col.at, v, go), score
  return out

def candidates(col: COL, rows: ROWS) -> Iterator[tuple[ATOM,GO]]:
  "What splits to try: one per symbol, or CUTS per num."
  at = col.at
  if type(col) is Sym:
    for v in sorted({r[at] for r in rows if r[at] != "?"}):
      yield v, lambda r,v=v,at=at: r[at] == v
  else:                # by rank, not value: THESE rows are dense
    xs = sorted(r[at] for r in rows if r[at] != "?")
    k  = max(1, len(xs)//the.Cuts)     # cuts, spaced by rank
    for v in xs[k-1:-1:k]:
      yield v, (lambda r,v=v,at=at: r[at] != "?" and r[at] <= v)

def tree(tbl: TBL, rows: ROWS) -> NODE:
  "Split while a split keeps both sides big enough."
  node = o(rows=rows, at=None, v=None, go=None, kids=[],
           mu=sum(ydist(tbl,r) for r in rows)/len(rows))
  if len(rows) > the.Leaf and (found := cut(tbl, rows)):
    at, v, go = found
    yes, no = [], []
    for r in rows: (yes if go(r) else no).append(r)
    node.at, node.v, node.go = at, v, go # cut kepts sides big
    node.kids = [tree(tbl, yes), tree(tbl, no)]
  return node

def leaf(node: NODE, row: ROW) -> NODE:
  "Walk ROW down to its leaf."
  while node.kids: node = node.kids[0 if node.go(row) else 1]
  return node

def leafs(node: NODE) -> list[NODE]:
  "Return leaves of this tree."
  return [x for k in node.kids for x in leafs(k)] or [node]

def show(tbl: TBL, node: NODE, pre: str | None = None,
         edge: str = "") -> None:
  "Tree on the left; ydist and n on the right."
  ls = sorted(leafs(node), key=lambda z: z.mu)
  def walk(z: NODE, pre: str | None, edge: str) -> None:
    m = "+" if z is ls[0] else "-" if z is ls[-1] else " "
    print(f"{m} {round(100*z.mu):>4} {len(z.rows):>5}   "
          f"{(pre or '')+edge}".rstrip())
    op = "=" if type(tbl.cols[z.at or 0]) is Sym else "<="
    no = "!=" if op == "=" else ">"
    for kid, oper in zip(z.kids, [op, no]):
      walk(kid, "" if pre is None else pre + "|  ",
           f"{tbl.names[z.at]} {oper} {say(z.v)}")
  print(" ydist     n")
  walk(node, pre, edge)



#-- rig ----------------------------------------------------
# Calls core, acquire, and one RANKER -- tree's, by default.
def ranker(lab: TBL) -> Callable[[ROW], float]:
  "Grow a tree on LAB; score any row by its leaf's mean ydist."
  node = tree(lab, lab.rows)
  return lambda r: leaf(node, r).mu

def wins(tbl: TBL) -> Callable[[ROW], float]:
  "100 at the best row, 0 at an average one."
  ys = sorted(ydist(tbl, r) for r in tbl.rows)
  lo, avg = ys[0], sum(ys)/len(ys)
  return lambda r: max(-100, min(100, 100*(1 - (ydist(tbl,r)-lo)
                                            / (avg-lo+1e-32))))

def holdout(tbl: TBL, pick: PICKER = acquire,
            rank: RANKER = ranker,
            label: ORACLE = oracle) -> ROW:
  "Train on half; of CHECK guesses on the rest, pick best."
  rows  = random.sample(tbl.rows, len(tbl.rows))
  n     = len(rows)//2
  tr    = clone(tbl, rows[:n][:the.Few])
  lab   = clone(tbl, pick(tr, the.Stop - the.Check, label))
  top   = sorted(rows[n:], key=rank(lab))
  return min(top[:the.Check], key=lambda r: ydist(lab, label(r)))

# A leaf: it calls core, and nothing here calls it.
def adds(vs: Iterable[ATOM], col: COL | None = None) -> COL:
  "Every item of VS into COL."
  col = Num() if col is None else col
  for v in vs: add(col, v)
  return col

def cohen(xs: NUMS, ys: NUMS, d: float = 0.35,
          eps: float = 0) -> bool:
  "Are the means within D pooled standard deviations?"
  a, b = adds(xs), adds(ys)
  pool = ((a.n-1)*div(a)**2 + (b.n-1)*div(b)**2)/(a.n+b.n-2)
  return abs(a.mu - b.mu) <= max(eps, d * sqrt(pool))

def cliffs(xs: NUMS, ys: NUMS, d: float = 0.197) -> bool:
  "Sorted in.  Is the rank imbalance small enough?"
  gt = lt = j = k = 0
  for x in xs:
    while j < len(ys) and ys[j] <  x: j += 1; k = j
    while k < len(ys) and ys[k] <= x: k += 1
    gt += j; lt += len(ys) - k
  return abs(gt - lt) / (len(xs) * len(ys)) <= d

def ks(xs: NUMS, ys: NUMS, a: float = 1.36) -> bool:
  "Sorted in.  95% Kolmogorov-Smirnov."
  n, m, i, j, d = len(xs), len(ys), 0, 0, 0
  while i < n and j < m:
    v = min(xs[i], ys[j])
    while i < n and xs[i] <= v: i += 1
    while j < m and ys[j] <= v: j += 1
    d = max(d, abs(i/n - j/m))
  return d <= a * sqrt((n + m) / (n * m))

def same(xs: NUMS, ys: NUMS, eps: float = 0) -> bool:
  """Indistinguishable.  Always cliffs and ks; EPS adds cohen, and
  doubles as its effect size (eps=0 skips that third test)."""
  xs, ys = sorted(xs), sorted(ys)
  return (cliffs(xs,ys) and ks(xs,ys)
          and (not eps or cohen(xs,ys,d=eps)))

import ezr

def remember(header, rows):
    """
    Identify the top 5% of rows based on multi-objective optimization.
    Uses the ydist metric from ezr to find rows closest to the ideal.
    """
    # Create a table object to leverage ezr's column and distance logic
    data = [header] + rows
    tbl = ezr.Tbl(data)
    
    # Calculate ydist for all rows
    # ydist measures how far a row is from the ideal goal values
    scored = []
    for i, row in enumerate(rows):
        d = ezr.ydist(tbl, row)
        scored.append((d, i))
    
    # Sort by distance (ascending, as lower ydist is better)
    scored.sort(key=lambda x: x[0])
    
    # Return the top 5% of rows (800 rows * 0.05 = 40 rows)
    limit = max(1, int(len(rows) * 0.05))
    return [item[1] for item in scored[:limit]]

import ezr

def guess(header, train, test):
    """
    Estimates the first goal column (MRE-) for test rows.
    Uses a tree-based ranker to find the leaf node for a test row
    and returns the mean of the first goal column in that leaf.
    """
    # Create a table for training data
    train_tbl = ezr.Tbl([header] + train)
    
    # Build a tree on the training data
    # The tree splits based on x-columns to group similar rows
    root = ezr.tree(train_tbl, train_tbl.rows)
    
    # Pre-calculate the mean of the first goal (index 0) for each leaf
    # We store this in the node object for quick lookup
    def cache_leaf_means(node):
        if not node.kids:
            # Calculate mean of the first goal (index 0) for this leaf
            vals = [r[0] for r in node.rows if r[0] != "?"]
            node.first_goal_mean = sum(vals) / len(vals) if vals else 0
        else:
            for kid in node.kids:
                cache_leaf_means(kid)
    
    cache_leaf_means(root)
    
    # For each test row, find its leaf and return the cached mean
    results = []
    for row in test:
        leaf_node = ezr.leaf(root, row)
        results.append(leaf_node.first_goal_mean)
        
    return results

if __name__ == "__main__":
    # Load data
    filename = "Health-ClosedIssues0000.n1000.train.csv"
    data = list(ezr.csv(filename))
    header = data[0]
    all_rows = data[1:]
    
    # Split into train (80%) and test (20%)
    split_idx = int(len(all_rows) * 0.8)
    train = all_rows[:split_idx]
    test_raw = all_rows[split_idx:]
    
    # Mask the goals in the test set to simulate unknown forecasts
    test = []
    for row in test_raw:
        masked = [row[i] if i >= len(header) - len(header) + 3 else "?" for i in range(len(row))]
        # The prompt specifies goals are columns 0, 1, 2. Mask them.
        masked[0] = masked[1] = masked[2] = "?"
        test.append(masked)
    
    # Get estimates
    estimates = guess(header, train, test)
    
    # Print a summary
    print(f"Forecast analysis for {len(estimates)} setups:")
    print(f"Average estimated MRE: {sum(estimates)/len(estimates):.4f}")
    print(f"First 5 estimates: {[round(x, 4) for x in estimates[:5]]}")

import ezr
import step01

def certify(xs, ys):
    """
    Returns True if the two distributions are statistically indistinguishable
    (same apart from noise), False if they are significantly different.
    Uses the statistical tests provided in ezr.
    """
    # Use the same logic as ezr.same, which checks cliffs and ks tests.
    # We sort the inputs as required by the statistical functions.
    xs_sorted = sorted(xs)
    ys_sorted = sorted(ys)
    return ezr.cliffs(xs_sorted, ys_sorted) and ezr.ks(xs_sorted, ys_sorted)

if __name__ == "__main__":
    # Load the data
    filename = "Health-ClosedIssues0000.n1000.train.csv"
    all_data = list(ezr.csv(filename))
    header = all_data[0]
    rows = all_data[1:]
    
    # Define "big" vs "small" based on N_estimators (index 3)
    # We use the median of N_estimators to split the data
    n_est_idx = 3
    n_estimators = [row[n_est_idx] for row in rows]
    median_n = sorted(n_estimators)[len(n_estimators) // 2]
    
    # We want to compare the accuracy (ACC+, index 1) of big vs small setups
    acc_idx = 1
    small_setups = [row[acc_idx] for row in rows if row[n_est_idx] <= median_n]
    big_setups = [row[acc_idx] for row in rows if row[n_est_idx] > median_n]
    
    # Run the certification
    is_same = certify(small_setups, big_setups)
    
    print(f"Comparing accuracy of {len(small_setups)} small setups vs {len(big_setups)} big setups.")

    if is_same:
        print("Result: The difference in accuracy is statistically insignificant (noise).")
    else:
        print("Result: The difference in accuracy is statistically significant (real).")

def flow(header, row, seen):
    """
    Decide if the current row is 'good enough' to ship.
    We define 'good enough' as being in the top 5% of the expected 
    distribution of quality (ydist) based on the history seen so far.
    """
    # If this is the first row, we can't compare it yet.
    if not seen:
        return False
    
    # Create a table to calculate ydist. 
    # We use 'seen + [row]' to establish the context of the current data.
    data = [header] + seen + [row]
    tbl = ezr.Tbl(data)
    
    # Calculate ydist for all seen rows and the current row
    # A lower ydist is better.
    current_ydist = ezr.ydist(tbl, row)
    seen_ydists = [ezr.ydist(tbl, r) for r in seen]
    
    # We define "good enough" as being better than 95% of what we've seen.
    # If the current row's ydist is lower than the 5th percentile of 
    # observed ydist values, we stop.
    threshold = sorted(seen_ydists)[max(0, int(len(seen_ydists) * 0.05))]
    
    return current_ydist <= threshold

import ezr
import step01

def blame(header, rows):
    """
    Identify input columns that best explain why the worst rows are bad.
    We compare the distribution of input values in the 'worst' rows 
    against the 'best' rows. Columns where the distributions differ 
    significantly are identified as the culprits.
    """
    # Create a table to use ezr's distance and column logic
    tbl = ezr.Tbl([header] + rows)
    
    # Score rows by ydist (lower is better)
    scored = []
    for row in rows:
        scored.append((ezr.ydist(tbl, row), row))
    
    # Sort by ydist
    scored.sort(key=lambda x: x[0])
    
    # Define worst 25% and best 25%
    n = len(scored) // 4
    worst = [x[1] for x in scored[-n:]]
    best = [x[1] for x in scored[:n]]
    
    culprits = []
    # Check each input column (x-columns)
    for col in tbl.x:
        at = col.at
        # Extract values for this column
        vals_best = [r[at] for r in best if r[at] != "?"]
        vals_worst = [r[at] for r in worst if r[at] != "?"]
        
        if not vals_best or not vals_worst:
            continue
            
        # If distributions are significantly different, it's a culprit
        # We use ezr.same (cliffs + ks) to check for significant difference
        if not ezr.same(sorted(vals_best), sorted(vals_worst)):
            culprits.append(header[at])
            
    return culprits

    # The guesser returns a list of estimates for the goals
    # We map these back to the row structure

    for i in range(3):

import ezr

def choose(header, rows, label):
    """
    Find the best row by using a budget of 50 labels.
    We use the acquire logic from ezr to intelligently sample the space.
    """
    # We need to work with a TBL that has the full data structure.
    # Since we don't have the goals yet, we create a TBL with the header.
    # We will populate it as we label.
    
    # Create a TBL with the header and all rows (goals are '?')
    # This allows ezr to understand the columns.
    tbl = ezr.Tbl([header] + rows)
    
    # The acquire function expects rows to have goals.
    # We define a wrapper that calls the provided label() function
    # and updates the row in place.
    def oracle_wrapper(row):
        # Find the index of this row in the original list
        # We use id() to ensure we match the exact object
        for i, r in enumerate(rows):
            if r is row:
                goals = label(i)
                row[0], row[1], row[2] = goals[0], goals[1], goals[2]
                # Update the TBL's column statistics with the new goals
                ezr.addCols(tbl, row)
                return row
        return row

    # We need to ensure the TBL has the correct column roles
    # The TBL was initialized with '?' in goals, so we need to re-add
    # the rows once they are labeled.
    # We reset the rows in the TBL to be empty so we can add them via acquire.
    tbl.rows = []
    
    # Configure the acquisition budget
    ezr.the.Stop = 50
    ezr.the.Start = 4
    
    # acquire() will call our oracle_wrapper, which updates the TBL
    labeled_rows = ezr.acquire(tbl, cap=50, label=oracle_wrapper)
    
    # Find the best row among the labeled ones
    best_row = min(labeled_rows, key=lambda r: ezr.ydist(tbl, r))
    
    # Return the index of the best row
    for i, r in enumerate(rows):
        if r is best_row:
            return i
    return 0

def fix(header, row, levers):
    """
    Find the smallest change to a bad row that improves its predicted performance.
    We iterate through each lever, try all possible values, and pick the one
    that results in the best predicted ydist.
    """
    # Build a model to predict performance
    filename = "Health-ClosedIssues0000.n1000.train.csv"
    all_data = list(ezr.csv(filename))
    header_data = all_data[0]
    rows = all_data[1:]
    tbl = ezr.Tbl(all_data)
    
    # Use the tree-based ranker from step02 to estimate quality
    # We need a ranker that returns a score (lower is better)
    ranker = step02.guess
    
    best_change = {}
    best_score = float('inf')
    
    # Try changing each lever
    for col_name, allowed in levers.items():
        col_idx = header.index(col_name)
        original_val = row[col_idx]
        
        # Generate candidates based on type
        candidates = []
        if isinstance(allowed, list): # Symbolic
            candidates = [v for v in allowed if v != original_val]
        else: # Numeric (low, high)
            low, high = allowed
            # Try a few points in the range
            candidates = [low, (low + high) / 2, high]
            candidates = [v for v in candidates if v != original_val]
            
        for val in candidates:
            new_row = list(row)
            new_row[col_idx] = val
            
            # Predict quality
            # We use the guesser logic: build tree on all rows, predict for new_row
            # Note: guess() returns a list of estimates for the goals
            # We need to calculate ydist for the new_row with these estimates
            est_goals = step02.guess(header_data, rows, [new_row])
            row_with_goals = list(new_row)
            row_with_goals[0], row_with_goals[1], row_with_goals[2] = est_goals[0], est_goals[0], est_goals[0]
            
            score = ezr.ydist(tbl, row_with_goals)
            
            if score < best_score:
                best_score = score
                best_change = {col_name: val}
                
    return best_change

import ezr
import step01

def spot(header, rows):
    """
    Identify rows that look like 'failed runs'.
    A failed run is defined as a row that is an outlier in the input space
    (x-columns) but performs poorly in the objective space (y-columns).
    We use the distance from the centroid of the input space as a measure
    of 'unusualness' and correlate it with ydist.
    """
    tbl = ezr.Tbl([header] + rows)
    
    # Calculate the centroid of the input space (x-columns)
    # We use the cached mids from the table
    mids = ezr.mids(tbl)
    
    # Calculate two metrics for each row:
    # 1. xdist: How far the input configuration is from the average configuration
    # 2. ydist: How far the performance is from the ideal
    # A 'failed run' is a row that is very far from the average input (xdist)
    # but still has a poor performance (high ydist).
    
    scores = []
    for i, row in enumerate(rows):
        xd = ezr.xdist(tbl, row, mids)
        yd = ezr.ydist(tbl, row)
        # Suspicion score: high xdist (unusual) * high ydist (poor performance)
        scores.append((xd * yd, i))
    
    # Sort by suspicion score descending
    scores.sort(key=lambda x: x[0], reverse=True)
    
    return [item[1] for item in scores]

import ezr
import random
import step07

def race(header, rows, label):
    """
    Compares three strategies: random, acquire, and heuristic.
    Total budget of 50 labels shared.
    """
    budget = 50
    labeled_indices = {}

    def oracle_wrapper(i):
        if i not in labeled_indices:
            if len(labeled_indices) >= budget:
                return None
            labeled_indices[i] = label(i)
            # Update row with true goals
            goals = labeled_indices[i]
            rows[i][0], rows[i][1], rows[i][2] = goals[0], goals[1], goals[2]
        return labeled_indices[i]

    # Create a master table to provide context for ydist/xdist
    # We use the rows as they are (with goals masked as '?')
    # and update them as we label.
    tbl = ezr.Tbl([header] + rows)

    # 1. Random Strategy
    def strategy_random():
        indices = random.sample(range(len(rows)), budget // 3)
        for i in indices: oracle_wrapper(i)
        # Find best among labeled
        return min(indices, key=lambda i: ezr.ydist(tbl, rows[i]))

    # 2. Acquire Strategy (from step07)
    def strategy_acquire():
        # step07.choose uses its own internal budget logic, 
        # we pass a wrapper that respects our shared budget.
        return step07.choose(header, rows, oracle_wrapper)

    # 3. Heuristic Strategy (Sort by xdist from centroid, label top)
    def strategy_heuristic():
        mids = ezr.mids(tbl)
        # Sort by xdist (unusual configurations)
        candidates = sorted(range(len(rows)), key=lambda i: ezr.xdist(tbl, rows[i], mids), reverse=True)
        for i in candidates[:budget // 3]:
            oracle_wrapper(i)
        return min(candidates[:budget // 3], key=lambda i: ezr.ydist(tbl, rows[i]))

    # Execute strategies
    results = {
        "random": strategy_random(),
        "acquire": strategy_acquire(),
        "heuristic": strategy_heuristic()
    }

    # Determine winner based on ydist of the best row found by each
    winner = min(results, key=lambda name: ezr.ydist(tbl, rows[results[name]]))
    
    return {
        "winner": winner,
        "row": results[winner],
        "tried": list(results.keys())
    }

def compare_forecasters(header, all_rows):
    """
    Compares a full-sized forecaster vs a halved-size forecaster.
    Halving the size is simulated by reducing the training data size.
    """
    # Split data: 80% train, 20% test
    split_idx = int(len(all_rows) * 0.8)
    train_full = all_rows[:split_idx]
    train_half = train_full[:len(train_full) // 2]
    test = all_rows[split_idx:]
    
    # Mask goals in test
    test_masked = []
    for row in test:
        masked = list(row)
        masked[0] = masked[1] = masked[2] = "?"
        test_masked.append(masked)
        
    # Get ground truth for comparison
    actuals = [row[0] for row in test]
    
    # Get estimates
    est_full = guess(header, train_full, test_masked)
    est_half = guess(header, train_half, test_masked)
    
    # Calculate Mean Absolute Error for both
    mae_full = sum(abs(a - e) for a, e in zip(actuals, est_full)) / len(actuals)
    mae_half = sum(abs(a - e) for a, e in zip(actuals, est_half)) / len(actuals)
    
    return mae_full, mae_half

import ezr
import random

def choose(header, rows, label):
    """
    Find a setup that forecasts well (low ydist) and is small.
    We use a budget of 50 labels to explore the space, then pick the best
    performing row that also meets our 'small' criteria (N_estimators <= 100, Max_depth <= 5).
    """
    # Initialize table
    tbl = ezr.Tbl([header] + rows)
    
    # We need to track which rows we have labeled to avoid exceeding the budget
    labeled_indices = {}
    
    def oracle_wrapper(row):
        # Find the index of this row in the original list
        for i, r in enumerate(rows):
            if r is row:
                if i not in labeled_indices:
                    if len(labeled_indices) >= 50:
                        return row # Return as is if budget exhausted
                    goals = label(i)
                    row[0], row[1], row[2] = goals[0], goals[1], goals[2]
                    labeled_indices[i] = True
                    ezr.addCols(tbl, row)
                return row
        return row

    # Use the acquisition logic to explore the space
    # We must ensure we don't pass an empty list to acquire
    if not rows:
        return 0
        
    # Run acquisition
    ezr.the.Stop = 50
    ezr.the.Start = 4
    labeled_rows = ezr.acquire(tbl, cap=50, label=oracle_wrapper)
    
    # Define 'small' criteria
    def is_small(row):
        return row[3] <= 100 and row[7] <= 5

    # Filter labeled rows for those that are 'small'
    small_labeled = [r for r in labeled_rows if is_small(r)]
    
    # If we found small rows, pick the best one; otherwise pick the best overall
    candidates = small_labeled if small_labeled else labeled_rows
    best_row = min(candidates, key=lambda r: ezr.ydist(tbl, r))
    
    # Return the index of the best row
    for i, r in enumerate(rows):
        if r is best_row:
            return i
    return 0
