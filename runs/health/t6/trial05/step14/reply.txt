import ezr
import step04

def flow(header, row, seen):
    """
    Decide if the current row is 'good enough' to stop the process.
    We use the logic from step04, which compares the current row's
    performance (ydist) against the 5th percentile of all previously
    seen rows. If the current row is in the top 5% of performance,
    we consider it a stopping point.
    """
    return step04.flow(header, row, seen)

if __name__ == "__main__":
    filename = "Health-ClosedIssues0000.n1000.train.csv"
    data = list(ezr.csv(filename))
    header = data[0]
    rows = data[1:]
    
    seen = []
    found = False
    print("Monitoring incoming nightly results...")
    for i, row in enumerate(rows):
        if flow(header, row, seen):
            print(f"Stopping criteria met at row {i+1}.")
            print(f"The best configuration found: {row}")
            found = True
            break
        seen.append(row)
    
    if not found:
        print("No row met the stopping criteria after processing all records.")
