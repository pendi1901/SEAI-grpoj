import ezr
import step01

def guess(header, train, test):
    """
    Estimates the first goal column for each test row.
    Uses the tree-based ranker from ezr to find the leaf node 
    corresponding to the test row and returns the mean of the 
    first goal column in that leaf.
    """
    # Build a table from training data to define column roles
    data = [header] + train
    tbl = ezr.Tbl(data)
    
    # Build a tree on the training data
    node = ezr.tree(tbl, tbl.rows)
    
    # For each test row, find its leaf and return the mean of the first goal
    # The first goal is the first column in tbl.y
    first_goal_col = tbl.y[0]
    
    results = []
    for row in test:
        leaf_node = ezr.leaf(node, row)
        # Calculate mean of the first goal in this leaf
        vals = [r[first_goal_col.at] for r in leaf_node.rows]
        results.append(sum(vals) / len(vals))
        
    return results

if __name__ == "__main__":
    # Load the data
    filename = "Health-ClosedIssues0000.n1000.train.csv"
    data_iter = ezr.csv(filename)
    header = next(data_iter)
    all_rows = list(data_iter)
    
    # Split into train and test (e.g., 80/20 split)
    split_idx = int(len(all_rows) * 0.8)
    train = all_rows[:split_idx]
    test = all_rows[split_idx:]
    
    # Mask the goals in the test set to simulate "unknown"
    # Goals are columns that end in + or -
    goal_indices = [i for i, name in enumerate(header) if name[-1] in "+-"]
    masked_test = []
    for row in test:
        new_row = list(row)
        for idx in goal_indices:
            new_row[idx] = "?"
        masked_test.append(new_row)
        
    # Get estimates
    estimates = guess(header, train, masked_test)
    
    # Print a summary
    print(f"Analysis complete on {len(all_rows)} records.")
    print(f"Estimated the first goal for {len(estimates)} test records.")
    print(f"Sample estimate: {estimates[0]:.4f}")
    
    exit(0)
