import ezr
import step03

def certify(xs, ys):
    """
    Returns True if xs and ys are statistically indistinguishable,
    False if they differ significantly.
    """
    return ezr.same(xs, ys)

def compare_rules(header, rows):
    """
    Compares the performance of two different rules:
    Rule 1: The default tree-based ranker.
    Rule 2: A random selection (straw man).
    """
    tbl = ezr.Tbl([header] + rows)
    
    # Rule 1: Tree-based ranker performance (ydist)
    ranker = ezr.ranker(tbl)
    scores_tree = [ranker(r) for r in rows]
    
    # Rule 2: Random selection performance (ydist)
    # We simulate random by shuffling the ydist values
    import random
    random.seed(1234567891)
    scores_random = list(scores_tree)
    random.shuffle(scores_random)
    
    return certify(scores_tree, scores_random)

if __name__ == "__main__":
    filename = "Health-ClosedIssues0000.n1000.train.csv"
    data_iter = ezr.csv(filename)
    header = next(data_iter)
    rows = list(data_iter)
    
    is_same = compare_rules(header, rows)
    
    print("Analysis: Does the choice of rule make a real difference?")
    if is_same:
        print("Result: The difference between the rules is statistically insignificant (noise).")
        print("You can stop arguing about it.")
    else:
        print("Result: There is a statistically significant difference between the rules.")
        print("The choice of rule matters.")
    
    exit(0)
