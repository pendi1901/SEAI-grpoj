import ezr

def fix(header, row, levers):
    """
    Find the single change to one setting that results in the best improvement
    in ydist. Levers for numeric columns are (low, high) ranges.
    """
    # Load the full dataset to ensure normalization is consistent
    filename = "Health-ClosedIssues0000.n1000.train.csv"
    data_iter = ezr.csv(filename)
    header_file = next(data_iter)
    rows = list(data_iter)
    tbl = ezr.Tbl([header_file] + rows)
    
    best_change = {}
    best_ydist = ezr.ydist(tbl, row)
    
    for col_name, allowed in levers.items():
        if col_name not in header:
            continue
        idx = header.index(col_name)
        
        # Determine candidate values
        candidates = []
        if isinstance(allowed, (list, tuple)) and len(allowed) == 2 and isinstance(allowed[0], (int, float)):
            # Numeric range: try a few points in the range
            low, high = allowed
            candidates = [low, (low + high) / 2, high]
        else:
            # Symbolic list
            candidates = allowed
            
        for val in candidates:
            # Skip if no change
            if val == row[idx]:
                continue
            
            # Create a candidate row
            candidate = list(row)
            candidate[idx] = val
            
            # Evaluate
            cand_ydist = ezr.ydist(tbl, candidate)
            
            # We want to improve ydist (lower is better)
            if cand_ydist < best_ydist:
                best_ydist = cand_ydist
                best_change = {col_name: val}
                
    return best_change

if __name__ == "__main__":
    filename = "Health-ClosedIssues0000.n1000.train.csv"
    data_iter = ezr.csv(filename)
    header = next(data_iter)
    rows = list(data_iter)
    
    # Identify a "bad" row (highest ydist)
    tbl = ezr.Tbl([header] + rows)
    scored = sorted([(ezr.ydist(tbl, r), r) for r in rows], key=lambda x: x[0])
    bad_row = scored[-1][1]
    
    # Define levers based on column types
    # Ensure these match the data types and ranges
    levers = {
        'N_estimators': (10, 200),
        'criterion': ['absolute_error', 'poisson', 'squared_error'],
        'Min_sample_leaves': (1, 20),
        'Min_impurity_decrease': (0, 10),
        'Max_depth': (1, 20)
    }
    
    recommendation = fix(header, bad_row, levers)
    
    print("Analysis complete.")
    if recommendation:
        col, val = list(recommendation.items())[0]
        print(f"To improve performance, change '{col}' to {val}.")
    else:
        print("No single change found to improve performance.")
    
    exit(0)
