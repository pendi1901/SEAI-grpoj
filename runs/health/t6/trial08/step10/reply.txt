import ezr
import random
import step01
import step07

def race(header, rows, label):
    """
    Compare three strategies: random, greedy (acquire), and a simple heuristic.
    Budget: 50 labels total.
    """
    budget = 50
    labeled_indices = set()
    
    def get_label(idx):
        if idx not in labeled_indices:
            if len(labeled_indices) >= budget:
                raise Exception("Budget exceeded")
            labeled_indices.add(idx)
        return label(idx)

    # Strategy 1: Random
    def strategy_random():
        indices = random.sample(range(len(rows)), budget // 3)
        for i in indices: get_label(i)
        return min(indices, key=lambda i: ezr.ydist(tbl, get_label(i)))

    # Strategy 2: Acquire (from step07)
    def strategy_acquire():
        # We pass a wrapper that uses our shared label budget
        def oracle_wrapper(idx): return get_label(idx)
        # Use a subset of budget
        return step07.choose(header, rows, oracle_wrapper)

    # Strategy 3: Heuristic (Top 5% search)
    def strategy_heuristic():
        # Sample some, then pick from the best
        indices = random.sample(range(len(rows)), 10)
        for i in indices: get_label(i)
        best_so_far = min(indices, key=lambda i: ezr.ydist(tbl, get_label(i)))
        return best_so_far

    tbl = ezr.Tbl([header] + rows)
    
    results = {
        "random": strategy_random(),
        "acquire": strategy_acquire(),
        "heuristic": strategy_heuristic()
    }
    
    # Find the winner: lowest ydist
    winner = min(results, key=lambda name: ezr.ydist(tbl, get_label(results[name])))
    
    return {
        "winner": winner,
        "row": results[winner],
        "tried": list(results.keys())
    }

if __name__ == "__main__":
    filename = "Health-ClosedIssues0000.n1000.train.csv"
    data_iter = ezr.csv(filename)
    header = next(data_iter)
    rows = list(data_iter)
    
    def label(idx):
        return rows[idx]
    
    result = race(header, rows, label)
    
    print(f"Race complete. Winner: {result['winner']}")
    print(f"Best row found at index: {result['row']}")
    print(f"Strategies compared: {', '.join(result['tried'])}")
    
    exit(0)
