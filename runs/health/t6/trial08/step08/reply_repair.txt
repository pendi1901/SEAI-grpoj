import ezr
import step01

def fix(header, row, levers):
    """
    Find the smallest change to a bad row that improves its ydist 
    without increasing the setup size (N_estimators).
    """
    # Create a table to define column roles and calculate ydist
    # We use the full dataset to ensure normalization is consistent
    filename = "Health-ClosedIssues0000.n1000.train.csv"
    data_iter = ezr.csv(filename)
    header_file = next(data_iter)
    rows = list(data_iter)
    tbl = ezr.Tbl([header_file] + rows)
    
    best_change = {}
    best_ydist = ezr.ydist(tbl, row)
    
    n_est_idx = header.index('N_estimators')
    current_n_est = row[n_est_idx]

    for col_name, allowed in levers.items():
        if col_name not in header:
            continue
        idx = header.index(col_name)
        
        for val in allowed:
            # Skip if no change
            if val == row[idx]:
                continue
            
            # Constraint: Do not increase N_estimators
            if idx == n_est_idx and val > current_n_est:
                continue
                
            # Create a candidate row
            candidate = list(row)
            candidate[idx] = val
            
            # Evaluate
            cand_ydist = ezr.ydist(tbl, candidate)
            
            # We want to improve ydist (lower is better)
            if cand_ydist < best_ydist:
                best_ydist = cand_ydist
                best_change = {col_name: val}
                
    return best_change

if __name__ == "__main__":
    filename = "Health-ClosedIssues0000.n1000.train.csv"
    data_iter = ezr.csv(filename)
    header = next(data_iter)
    rows = list(data_iter)
    
    # Identify a "bad" row (high ydist)
    tbl = ezr.Tbl([header] + rows)
    scored = sorted([(ezr.ydist(tbl, r), r) for r in rows], key=lambda x: x[0])
    bad_row = scored[-1][1]
    
    # Define levers based on column types
    # Ensure values are within reasonable bounds found in the data
    levers = {
        'N_estimators': [10, 50, 100, 150, 180],
        'criterion': ['absolute_error', 'poisson', 'squared_error'],
        'Min_sample_leaves': [1, 2, 5, 10, 11],
        'Min_impurity_decrease': [0, 2, 5, 7, 9],
        'Max_depth': [1, 2, 5, 7, 11]
    }
    
    recommendation = fix(header, bad_row, levers)
    
    print("Analysis complete.")
    if recommendation:
        print(f"To improve performance, change: {recommendation}")
    else:
        print("No simple change found to improve performance within constraints.")
    
    exit(0)
