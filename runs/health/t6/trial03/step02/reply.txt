import ezr
import step01

def guess(header, train, test):
    """
    Estimates the first goal column for each test row.
    Uses the tree-based ranker from ezr to predict the mean ydist 
    at the leaf, then maps that back to the first goal column.
    """
    # Create table from training data
    data = [header] + train
    tbl = ezr.Tbl(data)
    
    # Build a tree on the training data
    node = ezr.tree(tbl, tbl.rows)
    
    # The first goal column is the first one in tbl.y
    first_goal_col = tbl.y[0]
    
    # Calculate the mean value of the first goal in each leaf
    # to use as our prediction for that leaf
    leaf_means = {}
    for leaf in ezr.leafs(node):
        vals = [r[first_goal_col.at] for r in leaf.rows]
        leaf_means[id(leaf)] = sum(vals) / len(vals)
        
    results = []
    for row in test:
        # Find the leaf for this test row
        l = ezr.leaf(node, row)
        results.append(leaf_means[id(l)])
        
    return results

if __name__ == "__main__":
    # Load the data
    filename = "Health-ClosedIssues0000.n1000.train.csv"
    data = list(ezr.csv(filename))
    header = data[0]
    rows = data[1:]
    
    # Split: 80% train, 20% test
    n = int(len(rows) * 0.8)
    train = rows[:n]
    test = rows[n:]
    
    # Mask the goals in the test set
    test_masked = []
    for r in test:
        new_r = list(r)
        for col in range(len(header)):
            if header[col][-1] in "+-":
                new_r[col] = "?"
        test_masked.append(new_r)
        
    # Get predictions
    predictions = guess(header, train, test_masked)
    
    # Print a summary
    print(f"Forecast analysis complete for {len(predictions)} test configurations.")
    print(f"Sample prediction for the first goal: {predictions[0]:.4f}")
    print("These estimates are derived from the average performance of similar historical setups.")
    
    exit(0)
