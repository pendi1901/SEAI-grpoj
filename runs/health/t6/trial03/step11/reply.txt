import ezr
import step01

def remember(header, rows):
    """
    Identify the top 5% of rows based on multi-objective optimization.
    Uses the ydist metric from ezr to rank rows by proximity to the ideal.
    """
    # Create a table object to leverage ezr's column handling and ydist
    data = [header] + rows
    tbl = ezr.Tbl(data)
    
    # Calculate ydist for all rows
    # ydist measures how far a row's goals are from the best possible
    scored = []
    for i, row in enumerate(rows):
        d = ezr.ydist(tbl, row)
        scored.append((d, i))
    
    # Sort by distance (ascending: lower is better)
    scored.sort(key=lambda x: x[0])
    
    # Return the top 5% of rows
    limit = max(1, len(rows) // 20)
    return [idx for _, idx in scored[:limit]]

if __name__ == "__main__":
    # Load the data
    filename = "Health-ClosedIssues0000.n1000.train.csv"
    data = list(ezr.csv(filename))
    header = data[0]
    rows = data[1:]
    
    # Get the best indices
    best_indices = remember(header, rows)
    
    # Print a summary for the customer
    print(f"Analysis complete. Found {len(best_indices)} highly trustworthy setups.")
    print("Top 5 indices (0-based):", best_indices[:5])
    print("\nThese setups represent the configurations closest to the ideal objectives.")
    
    exit(0)
