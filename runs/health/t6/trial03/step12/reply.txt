```python
import ezr
import step02

def guess(header, train, test):
    """
    Estimates the first goal column for each test row.
    Uses the tree-based ranker from ezr to predict the mean ydist 
    at the leaf, then maps that back to the first goal column.
    """
    return step02.guess(header, train, test)

def compare_forecaster_sizes():
    filename = "Health-ClosedIssues0000.n1000.train.csv"
    data = list(ezr.csv(filename))
    header = data[0]
    rows = data[1:]
    
    # Split: 80% train, 20% test
    n = int(len(rows) * 0.8)
    train = rows[:n]
    test = rows[n:]
    
    # Mask the goals in the test set
    test_masked = []
    for r in test:
        new_r = list(r)
        for col in range(len(header)):
            if header[col][-1] in "+-":
                new_r[col] = "?"
        test_masked.append(new_r)
        
    # Full size forecast
    full_preds = guess(header, train, test_masked)
    
    # Halved size forecast (using half the training data)
    half_train = train[:len(train)//2]
    half_preds = guess(header, half_train, test_masked)
    
    # Calculate average absolute difference
    diffs = [abs(f - h) for f, h in zip(full_preds, half_preds)]
    avg_diff = sum(diffs) / len(diffs)
    
    print("Impact of halving the forecaster training size:")
    print(f"Average forecast deviation: {avg_diff:.4f}")
    print("This represents the average shift in predicted goal values when using 50% less training data.")

if __name__ == "__main__":
    compare_forecaster_sizes()
    exit(0)
```
