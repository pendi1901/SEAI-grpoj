import ezr
import step01

def guess(header, train, test):
    """
    Estimates the first goal column for each test row.
    Uses the tree-based ranker from ezr to find the leaf node 
    corresponding to the test row and returns the mean of the 
    first goal column in that leaf.
    """
    # Create a table for training data
    train_tbl = ezr.Tbl([header] + train)
    
    # Build a tree on the training data
    node = ezr.tree(train_tbl, train_tbl.rows)
    
    # Identify the index of the first goal column
    # The header names ending in + or - are goals
    goal_indices = [i for i, name in enumerate(header) if name[-1] in "+-"]
    first_goal_idx = goal_indices[0]
    
    results = []
    for row in test:
        # Walk the tree to find the leaf for this test row
        leaf_node = ezr.leaf(node, row)
        
        # Calculate the mean of the first goal column in this leaf
        # We use the values from the rows stored in the leaf
        vals = [r[first_goal_idx] for r in leaf_node.rows]
        mean_val = sum(vals) / len(vals)
        results.append(mean_val)
        
    return results

if __name__ == "__main__":
    # Load the data
    file_path = "Health-ClosedIssues0000.n1000.train.csv"
    all_data = list(ezr.csv(file_path))
    header = all_data[0]
    rows = all_data[1:]
    
    # Split data: 80% train, 20% test
    split_idx = int(len(rows) * 0.8)
    train = rows[:split_idx]
    test_full = rows[split_idx:]
    
    # Mask the goals in the test set to simulate the requirement
    test = []
    goal_indices = [i for i, name in enumerate(header) if name[-1] in "+-"]
    for row in test_full:
        new_row = list(row)
        for idx in goal_indices:
            new_row[idx] = "?"
        test.append(new_row)
        
    # Get estimates
    estimates = guess(header, train, test)
    
    # Print a summary for the customer
    print("Forecast estimation complete.")
    print(f"Estimated first goal for the first 5 test rows:")
    for i in range(min(5, len(estimates))):
        print(f"  Row {i}: {estimates[i]:.4f}")
