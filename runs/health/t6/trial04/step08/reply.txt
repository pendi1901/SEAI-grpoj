import ezr
import step01
import random

def fix(header, row, levers):
    """
    Finds the smallest change to a bad row that improves its ydist 
    without increasing the 'N_estimators' (to keep it from getting slower).
    """
    # Identify indices of levers
    lever_indices = {name: header.index(name) for name in levers}
    
    # Create a table to evaluate ydist
    # We use the full dataset to establish the goal space
    file_path = "Health-ClosedIssues0000.n1000.train.csv"
    all_data = list(ezr.csv(file_path))
    tbl = ezr.Tbl(all_data)
    
    best_change = {}
    min_ydist = ezr.ydist(tbl, row)
    
    # Try small perturbations for each lever
    for name, at in lever_indices.items():
        allowed = levers[name]
        original_val = row[at]
        
        # Generate candidates
        candidates = []
        if isinstance(allowed, list): # Symbolic
            candidates = [v for v in allowed if v != original_val]
        else: # Numeric (low, high)
            low, high = allowed
            # Try a few steps within range
            step = (high - low) / 5
            candidates = [low + i * step for i in range(6)]
            candidates = [c for c in candidates if c != original_val]
            
        for val in candidates:
            new_row = list(row)
            new_row[at] = val
            
            # Constraint: Do not increase N_estimators (index 3)
            # If we are changing N_estimators, ensure it's not higher than original
            if name == "N_estimators" and val > original_val:
                continue
                
            new_ydist = ezr.ydist(tbl, new_row)
            if new_ydist < min_ydist:
                min_ydist = new_ydist
                best_change = {name: val}
                
    return best_change

if __name__ == "__main__":
    file_path = "Health-ClosedIssues0000.n1000.train.csv"
    all_data = list(ezr.csv(file_path))
    header = all_data[0]
    rows = all_data[1:]
    
    # Define levers based on column types
    levers = {
        "N_estimators": (50, 200),
        "criterion": ["absolute_error", "poisson", "squared_error"],
        "Min_sample_leaves": (1, 20),
        "Max_depth": (1, 20)
    }
    
    # Pick a "bad" row (e.g., one with high ydist)
    tbl = ezr.Tbl(all_data)
    scored = sorted([(ezr.ydist(tbl, r), r) for r in rows], key=lambda x: x[0], reverse=True)
    bad_row = scored[0][1]
    
    recommendation = fix(header, bad_row, levers)
    
    print("Optimization recommendation for the poor-performing configuration:")
    if recommendation:
        for col, val in recommendation.items():
            print(f"Change '{col}' to {val} to improve performance without increasing latency.")
    else:
        print("No simple change found to improve performance within constraints.")
