import ezr
import random

def remember(header, rows):
    """
    Identify the top 5% of rows based on multi-objective optimization.
    Uses ydist to measure distance from the ideal goal values.
    """
    # Create a table object using the ezr library
    # The first row of data is the header, so we combine them
    data = [header] + rows
    tbl = ezr.Tbl(data)
    
    # Calculate ydist for every row
    # A lower ydist means the row is closer to the ideal goals
    scored = []
    for i, row in enumerate(rows):
        d = ezr.ydist(tbl, row)
        scored.append((d, i))
    
    # Sort by distance (ascending)
    scored.sort(key=lambda x: x[0])
    
    # Return the indices of the top 5%
    limit = max(1, len(rows) // 20)
    return [x[1] for x in scored[:limit]]

if __name__ == "__main__":
    # Load data
    filename = "Health-ClosedIssues0000.n1000.train.csv"
    data_iter = ezr.csv(filename)
    
    header = next(data_iter)
    rows = list(data_iter)
    
    # Get best indices
    best_indices = remember(header, rows)
    
    # Print results
    print(f"Found {len(best_indices)} top-performing setups out of {len(rows)}.")
    print("Indices of the most trustworthy setups:")
    print(best_indices)
    
    # Show details for the top 3
    print("\nTop 3 setups details:")
    for idx in best_indices[:3]:
        print(f"Row {idx}: {rows[idx]}")
