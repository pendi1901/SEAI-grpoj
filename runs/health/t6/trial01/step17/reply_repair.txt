import ezr

def choose(header, rows, label):
    """
    Finds a setup that is both accurate (low ydist) and small (low N_estimators).
    We use the acquire strategy to find high-performing rows, then filter
    for those that are also computationally efficient.
    """
    # Create a table structure using the header
    tbl = ezr.Tbl([header])
    
    # We need to track indices for the label function.
    # Since rows are passed as lists, we use their position in the list.
    # We must ensure we don't pass '?' to the table's internal logic.
    # We create a mapping to retrieve the original row data for the oracle.
    
    # The error in the previous version was due to passing rows with '?' 
    # to ezr.acquire, which then tries to add them to the table.
    # We will use a wrapper that labels the row before it hits the table.
    
    def wrapped_label(row):
        # Find the index of this row in the original list
        # We use the row content to find the index
        for i, r in enumerate(rows):
            if all(r[j] == row[j] or row[j] == "?" for j in range(len(row))):
                return label(i)
        return row

    # Initialize table with the header
    tbl = ezr.Tbl([header])
    
    # We need to populate the table with the rows, but we only have 50 labels.
    # We use the acquire function which handles the labeling budget.
    # We must ensure the table knows about the rows without violating the budget.
    for r in rows:
        tbl.rows.append(r)
    
    # Update columns based on the rows
    for r in rows:
        ezr.addCols(tbl, r)

    # Set budget
    ezr.the.Stop = 50
    
    # Acquire best rows
    labeled_rows = ezr.acquire(tbl, cap=50, label=wrapped_label)
    
    # Score the labeled rows based on Accuracy (ydist) AND Efficiency (N_estimators).
    # We want low ydist and low N_estimators.
    n_est_idx = header.index("N_estimators")
    
    def score(row):
        # Normalize ydist (0 to 1)
        y_d = ezr.ydist(tbl, row)
        # Normalize N_estimators (10 to 200)
        n_val = row[n_est_idx]
        n_norm = (n_val - 10) / (200 - 10)
        # Combined score: lower is better
        return y_d + (0.5 * n_norm)
    
    best_row = min(labeled_rows, key=score)
    
    # Return the index of the best row
    for i, r in enumerate(rows):
        if r == best_row:
            return i
    return 0

if __name__ == "__main__":
    filename = "Health-ClosedIssues0000.n1000.train.csv"
    data_iter = ezr.csv(filename)
    header = next(data_iter)
    all_rows = list(data_iter)
    
    # Mask the goals for the input to the function
    rows = []
    for row in all_rows:
        new_row = list(row)
        for i, name in enumerate(header):
            if name[-1] in "+-!":
                new_row[i] = "?"
        rows.append(new_row)
        
    def label(i):
        return all_rows[i]
        
    best_idx = choose(header, rows, label)
    print(f"Recommended setup found at index: {best_idx}")
    print(f"Row details: {all_rows[best_idx]}")
    print("This setup balances high forecast accuracy with low computational overhead.")
