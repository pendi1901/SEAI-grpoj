import ezr
import step08

def fix(header, row, levers):
    """
    Recommends the best single lever change to improve performance (ydist).
    Delegates the core logic to step08.fix, which evaluates potential changes
    against the dataset's performance landscape.
    """
    return step08.fix(header, row, levers)

if __name__ == "__main__":
    filename = "Health-ClosedIssues0000.n1000.train.csv"
    data_iter = ezr.csv(filename)
    header = next(data_iter)
    rows = list(data_iter)
    
    # Build table to identify a 'bad' row (high ydist)
    tbl = ezr.Tbl([header] + rows)
    scored = sorted([(ezr.ydist(tbl, r), r) for r in rows], key=lambda x: x[0], reverse=True)
    bad_row = scored[0][1]
    
    # Define levers based on the dataset structure
    levers = {
        "N_estimators": (10.0, 200.0),
        "criterion": ["absolute_error", "poisson", "squared_error"],
        "Min_sample_leaves": (1.0, 20.0),
        "Min_impurity_decrease": (0.0, 10.0),
        "Max_depth": (1.0, 20.0)
    }
    
    recommendation = fix(header, bad_row, levers)
    
    if recommendation:
        col, val = list(recommendation.items())[0]
        print(f"To improve performance for the identified bad setup, change '{col}' to {val}.")
    else:
        print("No improvement found within the given lever constraints.")
