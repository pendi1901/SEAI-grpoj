import ezr


def justify(header, rows, row):
    """Return (p, reason) where p is fraction of rows that row is better than."""
    
    # Build tree on complete rows to learn what makes good setups
    tbl = ezr.Tbl([header] + rows)
    node = ezr.tree(tbl, tbl.rows)
    
    # Predict ydist for unknown row by walking down the tree
    est_ydist = ezr.leaf(node, row).mu
    
    # Compute actual ydists for complete rows (ground truth)
    actuals = [ezr.ydist(tbl, r) for r in rows]
    
    # Fraction of rows that our estimate is better than (lower ydist = better)
    p = sum(1 for a in actuals if est_ydist < a) / len(actuals)
    
    reason = f"This setup is better than {p:.0%} of the others."
    return (p, reason)


if __name__ == "__main__":
    import os
    path = os.path.join(os.path.dirname(__file__), "Health-ClosedIssues0000.train.csv")
    rows = list(ezr.csv(path))
    header = rows[0]
    data_rows = rows[1:]

    # Find a row with "?" goals (the candidate we're justifying)
    unknown_row = None
    for r in data_rows:
        if any(c == "?" for c in r):
            unknown_row = r
            break

    if unknown_row is None:
        print("No unknown row found.")
    else:
        p, reason = justify(header, data_rows, unknown_row)
        print(f"{reason} (p={p:.4f})")
