import ezr
from step01 import remember


def choose(header, rows, label):
    """Find the best setup among up to 50 labelled ones."""
    # Start with a few random labels (like acquire does)
    start = min(4, len(rows))
    todo = list(range(len(rows)))
    random.shuffle(todo)

    # Label the first `start` rows
    known = {i: label(i) for i in todo[:start]}
    labelled = [known[i] for i in sorted(known.keys())]

    # Build a tree on the labelled rows to guide further labelling
    tbl = ezr.Tbl([header] + labelled)
    node = ezr.tree(tbl, tbl.rows)

    def predict_ydist(r):
        """Estimate ydist for an unlabelled row using the tree."""
        return ezr.leaf(node, r).mu

    # Now iteratively label more rows and pick the best so far
    budget = 50
    cap = len(labelled) + budget

    while todo and len(known) < cap:
        # Score each unlabelled row by how close it is to good setups
        scored = []
        for i in todo:
            if i in known:
                continue
            r = rows[i]
            est = predict_ydist(r)
            # Lower ydist = better; we want rows that look promising
            scored.append((est, i))

        if not scored:
            break

        scored.sort(key=lambda x: x[0])  # best estimates first
        pick = scored[0][1]
        known[pick] = label(pick)
        labelled.append(rows[pick])

    # Build final tree on all labelled rows and find the best
    tbl_final = ezr.Tbl([header] + labelled)
    node_final = ezr.tree(tbl_final, tbl_final.rows)

    def score_row(r):
        return ezr.leaf(node_final, r).mu

    scored_rows = [(score_row(rows[i]), i) for i in known.keys()]
    scored_rows.sort(key=lambda x: x[0])

    if not scored_rows:
        return 0

    best_idx = scored_rows[0][1]
    return int(best_idx)
