import ezr
import step05

def blame(header, rows):
    """
    Identify input columns that have the least impact on performance.
    We define 'safe to leave at default' as columns that do not appear
    in the top splits of a tree built to distinguish 'worst' from 'rest'.
    """
    # Use the logic from step05 to find the most influential column
    # We can extend this to find a ranking of all input columns
    data = [header] + rows
    tbl = ezr.Tbl(data)
    
    # Sort rows by ydist (ascending: lower is better)
    # The worst rows are those with the highest ydist
    sorted_rows = sorted(rows, key=lambda r: ezr.ydist(tbl, r), reverse=True)
    
    # Focus on the worst 25%
    n_worst = len(rows) // 4
    worst = sorted_rows[:n_worst]
    
    # Identify all input column names
    input_names = [name for name in header if name[-1] not in "+-"]
    
    # We find the most influential column using the cut logic
    # To find "safe" ones, we identify those NOT used in the top splits
    # of a tree built on the worst rows.
    influential = []
    
    # Check the top 3 levels of splits to see which columns matter
    curr_rows = worst
    for _ in range(3):
        found = ezr.cut(tbl, curr_rows)
        if found:
            at, _, _ = found
            name = header[at]
            if name not in influential:
                influential.append(name)
            # Split the rows to go deeper
            _, _, go = found
            curr_rows = [r for r in curr_rows if go(r)]
            if len(curr_rows) < ezr.the.Leaf: break
        else:
            break
            
    # Safe columns are those not in the influential list
    safe = [name for name in input_names if name not in influential]
    return safe

if __name__ == "__main__":
    filename = "Health-ClosedIssues0000.n1000.train.csv"
    data_gen = ezr.csv(filename)
    
    try:
        header = next(data_gen)
        rows = list(data_gen)
        
        safe_settings = blame(header, rows)
        
        print("Analysis of factory default settings:")
        if safe_settings:
            print("You can safely leave these settings at their factory defaults:")
            for setting in safe_settings:
                print(f"- {setting}")
        else:
            print("All settings appear to have a significant impact on performance.")
            
    except Exception as e:
        print(f"Error: {e}")
        exit(1)
        
    exit(0)
