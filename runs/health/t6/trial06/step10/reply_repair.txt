import ezr
import random

def race(header, rows, label):
    """
    Compare three strategies for finding the best configuration:
    1. random: Randomly samples rows.
    2. acquire: Uses the active learning strategy from ezr.
    3. greedy: Picks rows based on the best known ydist from a small initial sample.
    """
    budget = 50
    labeled_cache = {}

    def oracle_wrapper(idx):
        if idx not in labeled_cache:
            if len(labeled_cache) >= budget:
                raise Exception("Budget exceeded")
            labeled_cache[idx] = label(idx)
        return labeled_cache[idx]

    # To calculate ydist, we need a TBL with valid goal values.
    # We must label rows before calling ydist.
    def get_ydist(tbl, row):
        # If goals are '?', we cannot calculate ydist.
        # This function assumes the row has been labeled.
        return ezr.ydist(tbl, row)

    # Strategy 1: Random
    def strategy_random():
        indices = random.sample(range(len(rows)), budget // 3)
        for i in indices: oracle_wrapper(i)
        # Use the full dataset to build the TBL for evaluation
        tbl = ezr.Tbl([header] + [labeled_cache.get(i, rows[i]) for i in range(len(rows)) if i in labeled_cache])
        return min(indices, key=lambda i: get_ydist(tbl, labeled_cache[i]))

    # Strategy 2: Acquire
    def strategy_acquire():
        # Acquire needs a TBL with goals. We label a small set to start.
        initial_indices = random.sample(range(len(rows)), 10)
        for i in initial_indices: oracle_wrapper(i)
        
        data = [header] + [labeled_cache[i] for i in initial_indices]
        tbl = ezr.Tbl(data)
        
        def label_proxy(row):
            idx = rows.index(row)
            return oracle_wrapper(idx)
        
        results = ezr.acquire(tbl, cap=budget - len(labeled_cache), label=label_proxy)
        best_row = min(results, key=lambda r: get_ydist(tbl, r))
        return rows.index(best_row)

    # Strategy 3: Greedy
    def strategy_greedy():
        indices = random.sample(range(len(rows)), budget // 3)
        for i in indices: oracle_wrapper(i)
        tbl = ezr.Tbl([header] + [labeled_cache[i] for i in indices])
        return min(indices, key=lambda i: get_ydist(tbl, labeled_cache[i]))

    # Run strategies
    results = {
        "random": strategy_random(),
        "acquire": strategy_acquire(),
        "greedy": strategy_greedy()
    }

    # Determine winner
    tbl = ezr.Tbl([header] + [labeled_cache[i] for i in labeled_cache])
    winner = min(results, key=lambda name: get_ydist(tbl, labeled_cache[results[name]]))
    
    return {
        "winner": winner,
        "row": results[winner],
        "tried": list(results.keys())
    }

if __name__ == "__main__":
    filename = "Health-ClosedIssues0000.n1000.train.csv"
    data_gen = ezr.csv(filename)
    header = next(data_gen)
    all_rows = list(data_gen)
    
    masked_rows = []
    for r in all_rows:
        new_row = list(r)
        for i, name in enumerate(header):
            if name[-1] in "+-":
                new_row[i] = "?"
        masked_rows.append(new_row)
        
    def label(i):
        return all_rows[i]
        
    result = race(header, masked_rows, label)
    print(f"Race Results: {result}")
