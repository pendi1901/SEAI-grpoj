import ezr
import step01
import step02

def justify(header, rows, row):
    """
    Returns (p, reason): p is the fraction of rows that 'row' is better than,
    and a sentence explaining the performance.
    """
    # Create a table to handle the distance calculations
    tbl = ezr.Tbl([header] + rows)
    
    # Estimate the goals for the provided row using the tree-based ranker
    # We use the full dataset to train the ranker for the best possible estimate
    rank = step02.guess(header, rows, [row])
    
    # Create a copy of the row with the estimated goals filled in
    # The goal columns are those ending in '+' or '-'
    estimated_row = list(row)
    for i, name in enumerate(header):
        if name[-1] in "+-":
            # In this specific dataset, MRE- is at index 0, ACC+ at 1, PRED40+ at 2
            # We use the ranker's estimate for the primary goal (MRE-)
            # For simplicity, we assume the row's performance is dominated by its MRE-
            if name == "MRE-":
                estimated_row[i] = rank[0]
            else:
                # For other goals, we use the global mean of that column
                col = tbl.cols[i]
                estimated_row[i] = ezr.mid(col)
    
    # Calculate the ydist of our estimated row
    target_ydist = ezr.ydist(tbl, estimated_row)
    
    # Compare against all rows in the dataset
    better_count = 0
    for r in rows:
        if ezr.ydist(tbl, r) > target_ydist:
            better_count += 1
            
    p = better_count / len(rows)
    reason = f"This configuration is estimated to outperform {int(p * 100)}% of all known setups."
    
    return p, reason

if __name__ == "__main__":
    filename = "Health-ClosedIssues0000.n1000.train.csv"
    data_gen = ezr.csv(filename)
    
    try:
        header = next(data_gen)
        rows = list(data_gen)
        
        # Pick a sample row and mask its goals to test the justification
        sample_row = list(rows[0])
        for i, name in enumerate(header):
            if name[-1] in "+-":
                sample_row[i] = "?"
        
        p, reason = justify(header, rows, sample_row)
        print(f"Performance: {p:.4f}")
        print(f"Justification: {reason}")
        
    except Exception as e:
        print(f"Error: {e}")
        exit(1)
        
    exit(0)
