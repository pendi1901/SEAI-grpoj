import ezr
import step05

def blame(header, rows):
    """
    Identify input columns that have the least impact on performance,
    meaning they can be left at factory defaults.
    We reuse the logic from step05 to find the most impactful columns,
    then return the input columns that are NOT in that list.
    """
    # Get the most impactful columns using the logic from step05
    impactful = step05.blame(header, rows)
    
    # Identify all input columns (X-columns)
    # We use a temporary table to distinguish X from Y
    tbl = ezr.Tbl([header] + rows)
    input_col_names = [col.txt for col in tbl.x]
    
    # The columns that can be left at default are those that are NOT
    # in the 'impactful' list.
    defaults = [name for name in input_col_names if name not in impactful]
    
    return defaults

if __name__ == "__main__":
    filename = "Health-ClosedIssues0000.n1000.train.csv"
    all_data = list(ezr.csv(filename))
    header = all_data[0]
    rows = all_data[1:]
    
    safe_defaults = blame(header, rows)
    
    print("--- Analysis: Factory Defaults ---")
    print("The following input settings have minimal impact on performance")
    print("and can be left at factory defaults without hurting outcomes:")
    for col in safe_defaults:
        print(f"- {col}")
    
    exit(0)
