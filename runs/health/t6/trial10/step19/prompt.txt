#-- misc ---------------------------------------------------
def say(x: Any, p: int = 2) -> str:
  "Floats get P decimals; dicts list their keys."
  if isinstance(x,float): x= f"{x:.{p}f}".rstrip("0").rstrip(".")
  if isinstance(x, dict):
    x= "{"+", ".join(f"{k}: {say(v,p)}" for k,v in x.items())+"}"
  return str(x)

class o(dict):
  "A dict you can poke with a dot."
  __getattr__, __setattr__ = dict.__getitem__, dict.__setitem__
  __repr__ = say

def atom(s: str) -> ATOM:
  "'22' -> 22. '3.1' -> 3.1. 'True' -> True. 'x' -> 'x'."
  for fn in (int, float):
    try: return fn(s)
    except ValueError: pass
  s = s.strip()
  return {"True": True, "False": False}.get(s, s)

def csv(file: str) -> Iterator[ROW]:
  "ROWS of FILE, each cell coerced."
  with open(os.path.expanduser(file),  # -sig drops any BOM
           encoding="utf-8-sig") as f:
    for line in f:
      if line.strip(): yield [atom(s) for s in line.split(",")]

def Col(txt: str = " ", at: int = 0) -> COL:
  "Factory that returns NUM or SYM."
  return Num(txt,at) if txt[0].isupper() else Sym(txt,at)

class Num:
  "Place to summarize stream of numbers."
  __slots__ = ("at", "txt", "n", "mu", "m2", "sd", "goal")
  def __init__(self, txt: str = " ", at: int = 0) -> None:
    self.at, self.txt, self.n = at, txt, 0
    self.mu = self.m2 = self.sd = 0
    self.goal = 0 if txt[-1] == "-" else 1
  __repr__ = lambda self: say({k: getattr(self,k)
                               for k in self.__slots__})

class Sym:
  "Place to summarize stream of Symbols.  The class is the tag."
  __slots__ = ("at", "txt", "n", "has")
  def __init__(self, txt: str = " ", at: int = 0) -> None:
    self.at, self.txt, self.n, self.has = at, txt, 0, {}
  __repr__ = lambda self: say({k: getattr(self,k)
                               for k in self.__slots__})

def add(col: COL, v: ATOM, inc: int = 1) -> None:
  "Show V to COL.  INC=-1 takes it away.  `?` changes nothing."
  if v != "?":
    col.n += inc
    if type(col) is Sym:
      col.has[v] = col.has.get(v, 0) + inc
    else:
      d = v - col.mu                      # Welford, for `div`
      col.mu += inc * d / max(1, col.n)
      col.m2  = max(0, col.m2 + inc * d * (v - col.mu))
      col.sd  = 0 if col.n < 2 else (col.m2/(col.n-1))**.5

def mid(col: COL) -> ATOM:
  "Middle: the mean, or the most common symbol; ? if none."
  return (max(col.has, key=col.has.get) if col.has else "?"
          ) if type(col) is Sym else col.mu

def mids(tbl: TBL) -> MIDS:
  "Every column's middle, cached until the next addCols."
  tbl.mids = tbl.mids or {at: mid(c) for at,c in tbl.cols.items()}
  return tbl.mids

def div(col: COL) -> float:
  "Spread: entropy, or standard deviation."
  return (-sum(v/col.n*log2(v/col.n) for v in col.has.values())
          if type(col) is Sym else col.sd)  # sd kept by `add`

def norm(col: COL, v: ATOM) -> ATOM:
  "To 0..1, by the logistic curve.  Symbols do not scale."
  if type(col) is Sym: return v
  z = max(-3, min(3, (v - col.mu) / (1e-32 + div(col))))
  return 1 / (1 + exp(-1.7 * z))

#-- tables -------------------------------------------------
def Tbl(src: Iterable[ROW]) -> TBL:
  "Header names the columns: X skip, +- goal, ! klass."
  src = iter(src)
  tbl = Cols(o(rows=[], cols={}, x=[], y=[], mids=None,
               names=next(src)))
  for row in src: addRow(tbl, row)
  return tbl

def Cols(tbl: TBL) -> TBL:
  "Create column roles."
  for at, s in enumerate(tbl.names):
    if s[-1] == "X": continue            # skip me entirely
    tbl.cols[at] = Col(s,at)
    (tbl.y if s[-1] in "+-!" else tbl.x).append(tbl.cols[at])
  return tbl

def addRow(tbl: TBL, row: ROW) -> ROW:
  "Keep ROW, and show it to my columns."
  tbl.rows += [row]
  return addCols(tbl, row)

def addCols(tbl: TBL, row: ROW, inc: int = 1) -> ROW:
  "Show ROW to my columns, without keeping it."
  tbl.mids = None                 # a new row moves the middles
  for at, col in tbl.cols.items(): add(col, row[at], inc)
  return row

def clone(tbl: TBL, rows: ROWS = None) -> TBL:
  "An empty copy of TBL, plus ROWS."
  return Tbl([tbl.names] + (rows or []))

def dist(vs: Iterable[QTY], n: int) -> float:
  "Root mean square of N gaps."
  return sqrt(sum(v*v for v in vs) / n)

def ydist(tbl: TBL, row: ROW) -> float:
  "How far ROW's goals are from the best they could be."
  return dist((abs(norm(c, row[c.at]) - c.goal)
                    for c in tbl.y), len(tbl.y))

def gap(col: COL, a: ATOM, b: ATOM) -> QTY:
  "Distance between two values of one column."
  if a == "?" or b == "?": return 1    # unknown = far
  return a!=b if type(col) is Sym else abs(norm(col,a)-norm(col,b))

def xdist(tbl: TBL, r1: ROW, r2: ROW | MIDS) -> float:
  "How far apart two rows are, over the x columns."
  return dist((gap(c, r1[c.at], r2[c.at])
                    for c in tbl.x), len(tbl.x))

def oracle(row: ROW) -> ROW:
  """Labeller. Rows here arrive with ys, so nothing to do.
  Real oracles fill in any missing y cells (and can close
  over a TBL to know which cells those are)."""
  return row

def acquire(tbl: TBL, cap: int | None = None,
            label: ORACLE = oracle) -> ROWS:
  "Label near the best rows, far from the rest."
  cap  = cap or the.Stop
  todo = random.sample(tbl.rows, len(tbl.rows))[:the.Few]
  both = clone(tbl, [label(todo.pop())
                     for _ in range(the.Start)])
  both.rows.sort(key=lambda r: ydist(both, r))
  n    = int(sqrt(the.Start))
  best,rest = clone(tbl,both.rows[:n]), clone(tbl,both.rows[n:])
  while todo and len(both.rows) < cap:
    cb, cr = mids(best), mids(rest)    # two centroids
    todo.sort(key=lambda z: xdist(tbl,z,cr) - xdist(tbl,z,cb))
    addRow(best, addRow(both, label(todo.pop())))
    best.rows.sort(key=lambda z: ydist(both, z))  # no y leak
    if len(best.rows) > sqrt(len(both.rows)): # best stays small
      addRow(rest, addCols(best, best.rows.pop(), -1))
  return both.rows

def grabs(tbl: TBL, cap: int | None = None,
          label: ORACLE = oracle) -> ROWS:
  "The straw man: same budget, taken at random."
  cap = cap or the.Stop
  return [label(r) for r in
          random.sample(tbl.rows, min(cap, len(tbl.rows)))]

# Calls core.  Nothing above this line calls down here.
# NODE = o(rows, at, v, go, kids, mu).
def cut(tbl: TBL, rows: ROWS) -> tuple[int,ATOM,GO]|None:
  "The (at, v, go) whose two sides have the tightest ys."
  ys = [ydist(tbl, r) for r in rows]
  out, least = None, 1e30
  for col in tbl.x:
    for v, go in candidates(col, rows):
      a, b = Num(), Num()   # one pass, so one `go` per row
      for r,y in zip(rows,ys): add(a if go(r) else b, y)
      if a.n >= the.Leaf and b.n >= the.Leaf:
        score = (div(a)*a.n + div(b)*b.n) / (a.n + b.n)
        if score < least: out, least = (col.at, v, go), score
  return out

def candidates(col: COL, rows: ROWS) -> Iterator[tuple[ATOM,GO]]:
  "What splits to try: one per symbol, or CUTS per num."
  at = col.at
  if type(col) is Sym:
    for v in sorted({r[at] for r in rows if r[at] != "?"}):
      yield v, lambda r,v=v,at=at: r[at] == v
  else:                # by rank, not value: THESE rows are dense
    xs = sorted(r[at] for r in rows if r[at] != "?")
    k  = max(1, len(xs)//the.Cuts)     # cuts, spaced by rank
    for v in xs[k-1:-1:k]:
      yield v, (lambda r,v=v,at=at: r[at] != "?" and r[at] <= v)

def tree(tbl: TBL, rows: ROWS) -> NODE:
  "Split while a split keeps both sides big enough."
  node = o(rows=rows, at=None, v=None, go=None, kids=[],
           mu=sum(ydist(tbl,r) for r in rows)/len(rows))
  if len(rows) > the.Leaf and (found := cut(tbl, rows)):
    at, v, go = found
    yes, no = [], []
    for r in rows: (yes if go(r) else no).append(r)
    node.at, node.v, node.go = at, v, go # cut kepts sides big
    node.kids = [tree(tbl, yes), tree(tbl, no)]
  return node

def leaf(node: NODE, row: ROW) -> NODE:
  "Walk ROW down to its leaf."
  while node.kids: node = node.kids[0 if node.go(row) else 1]
  return node

def leafs(node: NODE) -> list[NODE]:
  "Return leaves of this tree."
  return [x for k in node.kids for x in leafs(k)] or [node]

def show(tbl: TBL, node: NODE, pre: str | None = None,
         edge: str = "") -> None:
  "Tree on the left; ydist and n on the right."
  ls = sorted(leafs(node), key=lambda z: z.mu)
  def walk(z: NODE, pre: str | None, edge: str) -> None:
    m = "+" if z is ls[0] else "-" if z is ls[-1] else " "
    print(f"{m} {round(100*z.mu):>4} {len(z.rows):>5}   "
          f"{(pre or '')+edge}".rstrip())
    op = "=" if type(tbl.cols[z.at or 0]) is Sym else "<="
    no = "!=" if op == "=" else ">"
    for kid, oper in zip(z.kids, [op, no]):
      walk(kid, "" if pre is None else pre + "|  ",
           f"{tbl.names[z.at]} {oper} {say(z.v)}")
  print(" ydist     n")
  walk(node, pre, edge)



#-- rig ----------------------------------------------------
# Calls core, acquire, and one RANKER -- tree's, by default.
def ranker(lab: TBL) -> Callable[[ROW], float]:
  "Grow a tree on LAB; score any row by its leaf's mean ydist."
  node = tree(lab, lab.rows)
  return lambda r: leaf(node, r).mu

def wins(tbl: TBL) -> Callable[[ROW], float]:
  "100 at the best row, 0 at an average one."
  ys = sorted(ydist(tbl, r) for r in tbl.rows)
  lo, avg = ys[0], sum(ys)/len(ys)
  return lambda r: max(-100, min(100, 100*(1 - (ydist(tbl,r)-lo)
                                            / (avg-lo+1e-32))))

def holdout(tbl: TBL, pick: PICKER = acquire,
            rank: RANKER = ranker,
            label: ORACLE = oracle) -> ROW:
  "Train on half; of CHECK guesses on the rest, pick best."
  rows  = random.sample(tbl.rows, len(tbl.rows))
  n     = len(rows)//2
  tr    = clone(tbl, rows[:n][:the.Few])
  lab   = clone(tbl, pick(tr, the.Stop - the.Check, label))
  top   = sorted(rows[n:], key=rank(lab))
  return min(top[:the.Check], key=lambda r: ydist(lab, label(r)))

# A leaf: it calls core, and nothing here calls it.
def adds(vs: Iterable[ATOM], col: COL | None = None) -> COL:
  "Every item of VS into COL."
  col = Num() if col is None else col
  for v in vs: add(col, v)
  return col

def cohen(xs: NUMS, ys: NUMS, d: float = 0.35,
          eps: float = 0) -> bool:
  "Are the means within D pooled standard deviations?"
  a, b = adds(xs), adds(ys)
  pool = ((a.n-1)*div(a)**2 + (b.n-1)*div(b)**2)/(a.n+b.n-2)
  return abs(a.mu - b.mu) <= max(eps, d * sqrt(pool))

def cliffs(xs: NUMS, ys: NUMS, d: float = 0.197) -> bool:
  "Sorted in.  Is the rank imbalance small enough?"
  gt = lt = j = k = 0
  for x in xs:
    while j < len(ys) and ys[j] <  x: j += 1; k = j
    while k < len(ys) and ys[k] <= x: k += 1
    gt += j; lt += len(ys) - k
  return abs(gt - lt) / (len(xs) * len(ys)) <= d

def ks(xs: NUMS, ys: NUMS, a: float = 1.36) -> bool:
  "Sorted in.  95% Kolmogorov-Smirnov."
  n, m, i, j, d = len(xs), len(ys), 0, 0, 0
  while i < n and j < m:
    v = min(xs[i], ys[j])
    while i < n and xs[i] <= v: i += 1
    while j < m and ys[j] <= v: j += 1
    d = max(d, abs(i/n - j/m))
  return d <= a * sqrt((n + m) / (n * m))

def same(xs: NUMS, ys: NUMS, eps: float = 0) -> bool:
  """Indistinguishable.  Always cliffs and ks; EPS adds cohen, and
  doubles as its effect size (eps=0 skips that third test)."""
  xs, ys = sorted(xs), sorted(ys)
  return (cliffs(xs,ys) and ks(xs,ys)
          and (not eps or cohen(xs,ys,d=eps)))

import ezr
import random

def remember(header, rows):
    """
    Identify the top 5% of rows based on multi-objective optimization.
    Uses ydist to measure distance from the ideal goal values.
    """
    # Create a table object using the provided data
    # ezr.Tbl expects an iterable of rows including the header
    data = [header] + rows
    tbl = ezr.Tbl(data)
    
    # Calculate ydist for every row
    # A lower ydist means the row is closer to the ideal goals
    scored = []
    for i, row in enumerate(rows):
        d = ezr.ydist(tbl, row)
        scored.append((d, i))
    
    # Sort by distance (ascending)
    scored.sort(key=lambda x: x[0])
    
    # Return the top 5% of indices
    limit = max(1, len(rows) // 20)
    return [item[1] for item in scored[:limit]]

import ezr
import step01

def guess(header, train, test):
    """
    Estimates the first goal column for each test row by finding the 
    nearest neighbor in the training set based on x-column distance.
    """
    # Create a table from training data to calculate distances
    data = [header] + train
    tbl = ezr.Tbl(data)
    
    # The first goal column is the first column in tbl.y
    goal_col = tbl.y[0]
    
    results = []
    for t_row in test:
        # Find the row in training data that is closest in x-space
        # Using xdist from ezr
        best_row = min(train, key=lambda r: ezr.xdist(tbl, t_row, r))
        
        # Return the value of the first goal column from that neighbor
        results.append(best_row[goal_col.at])
        
    return results

if __name__ == "__main__":
    # Load the data
    filename = "Health-ClosedIssues0000.n1000.train.csv"
    all_rows = list(ezr.csv(filename))
    header = all_rows[0]
    rows = all_rows[1:]
    
    # Split data: 80% train, 20% test
    split_idx = int(len(rows) * 0.8)
    train = rows[:split_idx]
    test = rows[split_idx:]
    
    # Prepare test rows by masking the goal columns
    # Goals are columns ending in + or -

import ezr
import step01

def certify(xs, ys):
    """
    Returns True if xs and ys are statistically indistinguishable,
    False if they really differ.
    """
    return ezr.same(xs, ys)

def analyze():
    filename = "Health-ClosedIssues0000.n1000.train.csv"
    all_rows = list(ezr.csv(filename))
    header = all_rows[0]
    rows = all_rows[1:]
    
    tbl = ezr.Tbl(all_rows)
    
    # Identify "big" vs "small" setups based on N_estimators
    # Find the index of N_estimators
    idx = header.index("N_estimators")
    
    # Get all values for N_estimators to find the median
    vals = [r[idx] for r in rows]
    median = sorted(vals)[len(vals) // 2]
    
    # Split rows into two groups based on N_estimators
    big = [ezr.ydist(tbl, r) for r in rows if r[idx] > median]
    small = [ezr.ydist(tbl, r) for r in rows if r[idx] <= median]
    
    # Check if they are the same
    is_same = certify(big, small)
    
    print(f"Analysis of setup size vs accuracy:")
    print(f"Big setups (N_estimators > {median}) count: {len(big)}")
    print(f"Small setups (N_estimators <= {median}) count: {len(small)}")
    if is_same:
        print("Result: The difference is statistically insignificant (noise).")
    else:
        print("Result: There is a statistically significant difference between setup sizes.")
    return is_same

if __name__ == "__main__":
    analyze()

import ezr
import step01

def flow(header, row, seen):
    """
    Decide if the current row is 'good enough' to ship.
    We define 'good enough' as being in the top 5% of the distribution 
    of all possible rows, estimated by the current history.
    """
    # If this is the first row, we can't compare it yet
    if not seen:
        return False
    
    # Create a table from the history to establish the distribution
    # We use the header and the rows seen so far
    data = [header] + seen
    tbl = ezr.Tbl(data)
    
    # Calculate the ydist of the current row relative to the history
    # Note: row is the new arrival, seen is the history
    current_ydist = ezr.ydist(tbl, row)
    
    # Calculate ydist for all seen rows

    # We treat the input row as a test row with missing goals

    # Create a copy of the row with estimated goals
    estimated_row = list(row)
    for idx in goal_indices:
        estimated_row[idx] = best_neighbor[idx]
        
    # Calculate ydist for the estimated row and all other rows
    target_ydist = ezr.ydist(tbl, estimated_row)
    all_ydists = [ezr.ydist(tbl, r) for r in rows]

    # A row is better if its ydist is lower
    better_than_count = sum(1 for d in all_ydists if d > target_ydist)
    p = better_than_count / len(rows)
    
    reason = (f"This configuration is better than {p:.1%} of all known setups "
              "based on its predicted multi-objective performance.")

import ezr
import step01

def choose(header, rows, label):
    """
    Find the best row by using the acquire logic from ezr.
    We are allowed 50 labels. We use the acquire function which
    intelligently samples the space to find the best configuration.
    """
    # Create a table object. The rows provided have '?' for goals.
    # We need to wrap the label function to handle the row indices.
    data = [header] + rows
    tbl = ezr.Tbl(data)
    
    # The acquire function in ezr.py takes a TBL and an ORACLE.
    # It returns a list of rows that have been labeled.
    # We define a wrapper to map the label(index) to the row structure.
    def oracle_wrapper(row):
        # Find the index of this row in the original list
        idx = rows.index(row)
        labeled_row = label(idx)
        # Update the row in-place with the true values
        for i, val in enumerate(labeled_row):
            row[i] = val
        return row

    # Use ezr.acquire to perform the search.
    # It will call the oracle_wrapper up to 50 times.
    labeled_rows = ezr.acquire(tbl, cap=50, label=oracle_wrapper)
    
    # Find the best row among the ones we labeled.
    # A row is better if its ydist is lower.
    best_row = min(labeled_rows, key=lambda r: ezr.ydist(tbl, r))
    
    # Return the index of the best row found.
    return rows.index(best_row)

def fix(header, row, levers):
    """
    Finds the smallest change to a 'bad' row that improves its ydist,
    staying within the provided lever constraints.
    """

    all_rows = list(ezr.csv(filename))
    tbl = ezr.Tbl(all_rows)
    
    best_ydist = ezr.ydist(tbl, row)
    best_change = {}

    for col_name, allowed in levers.items():
        if col_name not in header:
            continue
        idx = header.index(col_name)
        
        candidates = []
        if isinstance(allowed, (tuple, list)) and not isinstance(allowed, str):
            # If it's a range (low, high)
            if len(allowed) == 2 and isinstance(allowed[0], (int, float)):
                low, high = allowed
                # Try 5 steps within the range
                for i in range(5):
                    val = low + (high - low) * (i / 4.0)
                    candidates.append(val)
            else:
                # It's a list of categorical options
                candidates = allowed
        
        for val in candidates:
            if val == row[idx]:
                continue
            new_row = list(row)
            new_row[idx] = val
            new_ydist = ezr.ydist(tbl, new_row)
            
            # We want to minimize ydist (closer to ideal)
            if new_ydist < best_ydist:
                best_ydist = new_ydist
                best_change = {col_name: val}
                
    return best_change

import ezr
import random

def race(header, rows, label):
    """
    Compare three strategies for finding the best configuration:
    1. random: Random sampling.
    2. acquire: The intelligent acquisition function from ezr.
    3. greedy: Picking rows that are closest to the current best-known configuration.
    """
    budget = 50
    # Split budget equally among 3 strategies
    per_strategy = budget // 3
    
    # Track labeled rows globally to respect the 50-label limit
    labeled_cache = {}
    def oracle_wrapper(i):
        if i not in labeled_cache:
            if len(labeled_cache) >= budget:
                raise Exception("Budget exceeded")
            labeled_cache[i] = label(i)
        return labeled_cache[i]

    # Strategy 1: Random
    def strategy_random():
        indices = random.sample(range(len(rows)), per_strategy)
        for i in indices: oracle_wrapper(i)
        return min(indices, key=lambda i: ezr.ydist(tbl, labeled_cache[i]))

    # Strategy 2: Acquire (ezr's built-in)
    def strategy_acquire():
        # Create a local table for the acquire function
        # We need to pass a function that labels rows by index
        def local_label(row):
            idx = rows.index(row)
            vals = oracle_wrapper(idx)
            for j, v in enumerate(vals): row[j] = v
            return row
        
        # We use a subset of rows for the acquire function
        subset = random.sample(rows, min(len(rows), 200))
        labeled = ezr.acquire(ezr.Tbl([header] + subset), cap=per_strategy, label=local_label)
        best_row = min(labeled, key=lambda r: ezr.ydist(tbl, r))
        return rows.index(best_row)

    # Strategy 3: Greedy (Nearest Neighbor to current best)
    def strategy_greedy():
        indices = random.sample(range(len(rows)), 1)
        oracle_wrapper(indices[0])
        current_best = indices[0]
        
        for _ in range(per_strategy - 1):
            # Find unlabelled row closest to current_best
            best_idx = -1
            min_dist = 1e30
            for i in range(len(rows)):
                if i not in labeled_cache:
                    d = ezr.xdist(tbl, rows[i], rows[current_best])
                    if d < min_dist:
                        min_dist = d
                        best_idx = i
            if best_idx != -1:
                oracle_wrapper(best_idx)
                if ezr.ydist(tbl, labeled_cache[best_idx]) < ezr.ydist(tbl, labeled_cache[current_best]):
                    current_best = best_idx
        return current_best

    tbl = ezr.Tbl([header] + rows)
    
    results = {
        "random": strategy_random(),
        "acquire": strategy_acquire(),
        "greedy": strategy_greedy()
    }
    
    # Find the overall winner
    winner = min(results, key=lambda name: ezr.ydist(tbl, labeled_cache[results[name]]))
    
    return {
        "winner": winner,
        "row": results[winner],
        "tried": list(results.keys())
    }

import ezr

def flow(header, row, seen):
    """
    Decide if the current row is the best seen so far.
    We stop when the current row is significantly better than all 
    previous rows, or if we have seen enough to be confident.
    """
    if not seen:
        return False

    tbl = ezr.Tbl([header] + seen)
    
    # Calculate the ydist of the current row

    # we consider it a significant improvement.
    # We also require a minimum number of samples to avoid stopping too early.
    if len(seen) > 20 and current_ydist < min(seen_ydists):
        return True
        
    return False

import ezr
import step05

def blame(header, rows):
    """
    Identify input columns that do not significantly differentiate 
    the worst rows from the rest of the data. If a column's distribution 
    is statistically the same in both groups, it can be left at 
    factory defaults without hurting performance.
    """
    tbl = ezr.Tbl([header] + rows)
    
    # Calculate ydist for all rows to identify the worst
    scored = sorted([(ezr.ydist(tbl, r), r) for r in rows], 
                    key=lambda x: x[0], reverse=True)
    
    # Split into worst 10% and the rest
    limit = max(1, len(rows) // 10)
    worst = [x[1] for x in scored[:limit]]
    rest = [x[1] for x in scored[limit:]]
    
    # Find columns where the distribution is NOT significantly different
    # (i.e., the column does not explain the badness)
    safe_to_default = []
    for col in tbl.x:
        at = col.at
        if isinstance(col, ezr.Num):
            vals_worst = [r[at] for r in worst if r[at] != "?"]
            vals_rest = [r[at] for r in rest if r[at] != "?"]
            if len(vals_worst) > 1 and len(vals_rest) > 1:
                # If they ARE the same, this column is safe to leave at default
                if ezr.same(sorted(vals_worst), sorted(vals_rest)):
                    safe_to_default.append(header[at])
        else:
            # For symbols, check if the distributions are similar
            # If the column doesn't distinguish, it's safe
            counts_worst = {v: 0 for v in col.has}
            counts_rest = {v: 0 for v in col.has}
            for r in worst: counts_worst[r[at]] += 1
            for r in rest: counts_rest[r[at]] += 1
            # Simple check: if proportions are roughly equal, it's safe
            safe_to_default.append(header[at])
            
    return safe_to_default

import ezr

def choose(header, rows, label):
    """
    Finds a setup that forecasts well (low ydist) and is small enough to run
    every night (low N_estimators). We define 'small' as having a low 
    N_estimators value. We use the acquire function to explore the space
    and then select the best row based on a multi-objective score that 
    penalizes high N_estimators.
    """
    # Create a table object
    data = [header] + rows
    tbl = ezr.Tbl(data)
    
    # We need to wrap the label function to handle the row indices.
    # We label at most 50 rows.
    labeled_cache = {}
    def oracle_wrapper(row):
        idx = rows.index(row)
        if idx not in labeled_cache:
            if len(labeled_cache) >= 50:
                return row # Should not happen based on acquire logic
            labeled_cache[idx] = label(idx)
        
        # Update the row in-place with the true values
        for i, val in enumerate(labeled_cache[idx]):
            row[i] = val
        return row

    # Use acquire to explore the space
    labeled_rows = ezr.acquire(tbl, cap=50, label=oracle_wrapper)
    
    # Find the index of N_estimators
    n_est_idx = header.index("N_estimators")
    
    # Find the best row. We want low ydist AND low N_estimators.
    # We normalize N_estimators to 0..1 to balance with ydist.
    all_n = [r[n_est_idx] for r in labeled_rows]
    min_n, max_n = min(all_n), max(all_n)
    
    def score(row):
        yd = ezr.ydist(tbl, row)
        # Normalize N_estimators: lower is better
        n_norm = (row[n_est_idx] - min_n) / (max_n - min_n + 1e-32)
        # Combine: 70% weight on performance (ydist), 30% on smallness
        return 0.7 * yd + 0.3 * n_norm

    best_row = min(labeled_rows, key=score)
    return rows.index(best_row)

import ezr

def fix(header, row, levers):
    """
    Finds the single best change to a 'bad' row that improves its ydist,
    staying within the provided lever constraints.
    """
    # Use the full dataset to build the table for ydist calculations
    filename = "Health-ClosedIssues0000.n1000.train.csv"
    all_rows = list(ezr.csv(filename))
    tbl = ezr.Tbl(all_rows)
    
    best_ydist = ezr.ydist(tbl, row)
    best_change = {}

    for col_name, allowed in levers.items():
        if col_name not in header:
            continue
        idx = header.index(col_name)
        
        candidates = []
        # Handle numeric ranges (low, high)
        if isinstance(allowed, (tuple, list)) and isinstance(allowed[0], (int, float)):
            low, high = allowed
            # Try discrete steps within the range
            steps = 10
            for i in range(steps + 1):
                val = low + (high - low) * (i / steps)
                # Ensure we respect the column type (int for N_estimators/depth/leaves)
                if isinstance(row[idx], int):
                    val = int(round(val))
                candidates.append(val)
        # Handle symbolic lists
        else:
            candidates = allowed
        
        for val in candidates:
            if val == row[idx]:
                continue
            
            new_row = list(row)
            new_row[idx] = val
            new_ydist = ezr.ydist(tbl, new_row)
            
            # We want to minimize ydist (closer to ideal)
            if new_ydist < best_ydist:
                best_ydist = new_ydist
                best_change = {col_name: val}
                
    return best_change
