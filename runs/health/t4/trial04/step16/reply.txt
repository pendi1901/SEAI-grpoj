```python
import csv
import numpy as np
from sklearn.neighbors import KNeighborsRegressor
from sklearn.preprocessing import LabelEncoder

def get_score(header, row):
    """Calculates a composite score: (1 - MRE) + ACC + PRED40."""
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')
    return (1.0 - float(row[mre_idx])) + float(row[acc_idx]) + float(row[pred_idx])

def justify(header, rows, row):
    """
    Estimates performance of a 'deep/big' setup and compares it to others.
    Returns (fraction_better, reason).
    """
    # Identify indices
    input_indices = [i for i, h in enumerate(header) if '-' not in h and '+' not in h]
    
    # Prepare training data
    X = []
    y = []
    for r in rows:
        features = []
        for i in input_indices:
            val = r[i]
            features.append(hash(val) if isinstance(val, str) else float(val))
        X.append(features)
        y.append(get_score(header, r))
    
    # Prepare target row features
    target_features = []
    for i in input_indices:
        val = row[i]
        target_features.append(hash(val) if isinstance(val, str) else float(val))
            
    # Estimate score using KNN
    model = KNeighborsRegressor(n_neighbors=10)
    model.fit(X, y)
    pred_score = model.predict([target_features])[0]
    
    # Compare against all known rows
    scores = [get_score(header, r) for r in rows]
    better_than_count = sum(1 for s in scores if pred_score > s)
    p = better_than_count / len(rows)
    
    reason = f"Using the biggest, deepest setup is not always optimal because it only outperforms {p:.1%} of configurations, suggesting that simpler models often achieve better results."
    
    return p, reason

if __name__ == "__main__":
    filename = 'Health-ClosedIssues0000.n1000.train.csv'
    rows = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = []
            for i, val in enumerate(row):
                if header[i] == 'criterion':
                    processed_row.append(val)
                else:
                    processed_row.append(float(val))
            rows.append(processed_row)
    
    # Define a "big/deep" setup: Max depth 20, N_estimators 200
    # Row indices: 3: N_estimators, 4: criterion, 5: Min_sample_leaves, 6: Min_impurity_decrease, 7: Max_depth
    deep_row = ["?", "?", "?", 200.0, 'squared_error', 1.0, 0.0, 20.0]
    
    p, reason = justify(header, rows, deep_row)
    print(reason)
```
