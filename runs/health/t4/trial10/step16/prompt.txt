import csv

def remember(header, rows):
    """
    Identifies the best rows based on multi-objective optimization.
    Goals: Minimize columns ending in '-', Maximize columns ending in '+'.
    Uses a simple Pareto-based ranking approach to find the most trustworthy setups.
    """
    goals = []
    for i, name in enumerate(header):
        if name.endswith('-'):
            goals.append((i, -1))  # Minimize
        elif name.endswith('+'):
            goals.append((i, 1))   # Maximize

    def get_score(row):
        # Normalize values to compare across different scales
        # Since we want to find the "handful", we calculate a weighted score
        # based on how close each goal is to the best observed value.
        return 0

    # Calculate scores: for each goal, we want to be as far from the worst
    # and as close to the best as possible.
    scores = []
    for idx, row in enumerate(rows):
        # Simple scalarization: normalize goals and sum them
        # We use the min/max of the column to normalize
        val = 0
        for col_idx, direction in goals:
            col_data = [float(r[col_idx]) for r in rows]
            min_v, max_v = min(col_data), max(col_data)
            norm = (row[col_idx] - min_v) / (max_v - min_v + 1e-9)
            val += norm if direction == 1 else (1 - norm)
        scores.append((val, idx))

    # Sort by score descending
    scores.sort(key=lambda x: x[0], reverse=True)
    
    # Return top 5% (800 rows * 0.05 = 40)
    limit = max(1, int(len(rows) * 0.05))
    return [item[1] for item in scores[:limit]]

def guess(header, train, test):
    """
    Estimates the first goal column (MRE-) for test rows using a Random Forest.
    Handles '?' in test data by excluding the goal column from input features.
    """
    goal_idx = 0
    
    # Identify symbolic columns
    symbolic_indices = [i for i, val in enumerate(train[0]) if isinstance(val, str)]
    
    # Fit encoders on all available data (train + test non-goal columns)
    encoders = {}
    for i in symbolic_indices:
        if i == goal_idx:
            continue
        encoders[i] = LabelEncoder()
        # Collect all unique values from train and test (ignoring '?')
        all_vals = [r[i] for r in train] + [r[i] for r in test if r[i] != '?']
        encoders[i].fit(all_vals)

    def transform_row(row):
        features = []
        for i in range(len(row)):
            if i == goal_idx:
                continue
            if i in encoders:
                features.append(encoders[i].transform([row[i]])[0])
            else:
                features.append(float(row[i]))
        return features

    X_train = [transform_row(row) for row in train]
    y_train = [row[goal_idx] for row in train]
    X_test = [transform_row(row) for row in test]
    
    model = RandomForestRegressor(n_estimators=100, random_state=42)
    model.fit(X_train, y_train)
    
    return model.predict(X_test).tolist()

def certify(xs, ys):
    """
    Returns True if the difference between two distributions is likely noise,
    False if they are statistically different. Uses a t-test.
    """
    t_stat, p_value = stats.ttest_ind(xs, ys, equal_var=False)
    # If p-value > 0.05, we fail to reject the null hypothesis (it's noise)
    return p_value > 0.05

import csv

def flow(header, row, seen):
    """
    Decides if the current row is 'good enough' to stop searching.
    A row is good if it performs better than the Pareto-optimal 
    frontier of all previously seen rows.
    """
    goals = []
    for i, name in enumerate(header):
        if name.endswith('-'):
            goals.append((i, -1))  # Minimize
        elif name.endswith('+'):
            goals.append((i, 1))   # Maximize

    # If it's the first row, we can't compare, so keep looking
    if not seen:
        return False

    # Calculate a simple score for the current row relative to the seen data
    # We define 'good enough' as being better than the best observed in all goals
    def get_score(r):
        score = 0
        for col_idx, direction in goals:
            col_data = [float(s[col_idx]) for s in seen]
            best_val = max(col_data) if direction == 1 else min(col_data)
            # If we are better than the best seen in this goal, add to score
            if direction == 1:
                if float(r[col_idx]) >= best_val: score += 1
            else:
                if float(r[col_idx]) <= best_val: score += 1
        return score

    # If the row beats the best seen in all goal categories, stop
    return get_score(row) == len(goals)

def blame(header, rows):
    """
    Identifies input columns that best explain poor performance.
    We define 'bad' as the bottom 25% of rows based on a simple aggregate score.
    We then use feature importance from a Random Forest to identify the culprits.
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith(('-', '+'))]
    input_indices = [i for i, name in enumerate(header) if not name.endswith(('-', '+'))]
    
    # Calculate a simple aggregate score for each row
    scores = []
    for row in rows:
        score = 0
        for i in goal_indices:
            col_data = [float(r[i]) for r in rows]
            norm = (row[i] - min(col_data)) / (max(col_data) - min(col_data) + 1e-9)
            score += norm if header[i].endswith('+') else (1 - norm)
        scores.append(score)
    
    # Label rows as 'bad' (bottom 25%)
    threshold = sorted(scores)[len(scores) // 4]
    y = [1 if s <= threshold else 0 for s in scores]
    
    # Prepare features
    X = []
    encoders = {}
    for i in input_indices:
        if isinstance(rows[0][i], str):
            le = LabelEncoder()
            col_vals = [r[i] for r in rows]
            le.fit(col_vals)
            encoders[i] = le
    
    for row in rows:
        features = []
        for i in input_indices:
            if i in encoders:
                features.append(encoders[i].transform([row[i]])[0])
            else:
                features.append(float(row[i]))
        X.append(features)
    
    # Train a model to predict 'badness'
    model = RandomForestRegressor(n_estimators=50, random_state=42)
    model.fit(X, y)
    
    # Get feature importances
    importances = model.feature_importances_
    indexed_importances = sorted(zip(importances, input_indices), reverse=True)
    
    # Return names of top 2 most important input columns
    return [header[idx] for _, idx in indexed_importances[:2]]

                score += val

import csv
from sklearn.ensemble import IsolationForest
from sklearn.preprocessing import LabelEncoder

def spot(header, rows):
    """
    Identifies rows that look like failed runs using an Isolation Forest.
    Failed runs are treated as outliers in the input space.
    """
    # Identify input columns (those not ending in - or +)
    input_indices = [i for i, name in enumerate(header) if not (name.endswith('-') or name.endswith('+'))]
    
    # Prepare feature matrix
    X = []
    encoders = {}
    
    # Fit encoders for symbolic columns
    for i in input_indices:
        if isinstance(rows[0][i], str):
            le = LabelEncoder()
            col_vals = [r[i] for r in rows]
            le.fit(col_vals)
            encoders[i] = le
            
    for row in rows:
        features = []
        for i in input_indices:
            if i in encoders:
                features.append(encoders[i].transform([row[i]])[0])
            else:
                features.append(float(row[i]))
        X.append(features)
    
    # Use Isolation Forest to detect anomalies (failed runs)
    # contamination is set to a small value as we expect few failed runs
    clf = IsolationForest(contamination=0.05, random_state=42)
    clf.fit(X)
    scores = clf.decision_function(X)
    
    # Sort indices by score (lower score = more anomalous/suspicious)
    indexed_scores = sorted(enumerate(scores), key=lambda x: x[1])
    
    return [idx for idx, score in indexed_scores]

def certify(xs, ys):
    """
    Returns True if the difference between two distributions is likely noise,
    False if they are statistically different. Uses a t-test.
    """
    t_stat, p_value = stats.ttest_ind(xs, ys, equal_var=False)
    # If p-value > 0.05, we fail to reject the null hypothesis (it's noise)
    return p_value > 0.05

import csv

def flow(header, row, seen):
    """
    Decides whether to stop the search based on whether the current row
    is a new Pareto-optimal point. If the current row is not dominated by
    any previously seen row, it is considered an improvement.
    """
    if not seen:
        return True

    goals = []
    for i, name in enumerate(header):
        if name.endswith('-'):
            goals.append((i, -1))  # Minimize
        elif name.endswith('+'):
            goals.append((i, 1))   # Maximize

    def dominates(r1, r2):
        """Returns True if r1 dominates r2."""
        better_in_at_least_one = False
        for i, direction in goals:
            v1, v2 = float(r1[i]), float(r2[i])
            if direction == 1:
                if v1 < v2: return False
                if v1 > v2: better_in_at_least_one = True
            else:
                if v1 > v2: return False
                if v1 < v2: better_in_at_least_one = True
        return better_in_at_least_one

    # If the current row is not dominated by any row in 'seen', it's a new best
    for s in seen:
        if dominates(s, row):
            return False
    return True
