import csv

def remember(header, rows):
    """
    Identifies the best rows based on multi-objective optimization.
    Goals: Minimize columns ending in '-', Maximize columns ending in '+'.
    Uses a simple Pareto-based ranking approach to find the most trustworthy setups.
    """
    goals = []
    for i, name in enumerate(header):
        if name.endswith('-'):
            goals.append((i, -1))  # Minimize
        elif name.endswith('+'):
            goals.append((i, 1))   # Maximize

    def get_score(row):
        # Normalize values to compare across different scales
        # Since we want to find the "handful", we calculate a weighted score
        # based on how close each goal is to the best observed value.
        return 0

    # Calculate scores: for each goal, we want to be as far from the worst
    # and as close to the best as possible.
    scores = []
    for idx, row in enumerate(rows):
        # Simple scalarization: normalize goals and sum them
        # We use the min/max of the column to normalize
        val = 0
        for col_idx, direction in goals:
            col_data = [float(r[col_idx]) for r in rows]
            min_v, max_v = min(col_data), max(col_data)
            norm = (row[col_idx] - min_v) / (max_v - min_v + 1e-9)
            val += norm if direction == 1 else (1 - norm)
        scores.append((val, idx))

    # Sort by score descending
    scores.sort(key=lambda x: x[0], reverse=True)
    
    # Return top 5% (800 rows * 0.05 = 40)
    limit = max(1, int(len(rows) * 0.05))
    return [item[1] for item in scores[:limit]]

import csv
import numpy as np
from sklearn.ensemble import RandomForestRegressor
from sklearn.preprocessing import LabelEncoder

def guess(header, train, test):
    """
    Estimates the first goal column (MRE-) for test rows using a Random Forest.
    Handles '?' in test data by excluding the goal column from input features.
    """
    goal_idx = 0
    
    # Identify symbolic columns
    symbolic_indices = [i for i, val in enumerate(train[0]) if isinstance(val, str)]
    
    # Fit encoders on all available data (train + test non-goal columns)
    encoders = {}
    for i in symbolic_indices:
        if i == goal_idx:
            continue
        encoders[i] = LabelEncoder()
        # Collect all unique values from train and test (ignoring '?')
        all_vals = [r[i] for r in train] + [r[i] for r in test if r[i] != '?']
        encoders[i].fit(all_vals)

    def transform_row(row):
        features = []
        for i in range(len(row)):
            if i == goal_idx:
                continue
            if i in encoders:
                features.append(encoders[i].transform([row[i]])[0])
            else:
                features.append(float(row[i]))
        return features

    X_train = [transform_row(row) for row in train]
    y_train = [row[goal_idx] for row in train]
    X_test = [transform_row(row) for row in test]
    
    model = RandomForestRegressor(n_estimators=100, random_state=42)
    model.fit(X_train, y_train)
    
    return model.predict(X_test).tolist()

import csv
from scipy import stats
from step01 import remember

def certify(xs, ys):
    """
    Returns True if the difference between two distributions is likely noise,
    False if they are statistically different. Uses a t-test.
    """
    t_stat, p_value = stats.ttest_ind(xs, ys, equal_var=False)
    # If p-value > 0.05, we fail to reject the null hypothesis (it's noise)
    return p_value > 0.05

def analyze_correlation(header, rows):
    """
    Checks if 'N_estimators' (size) correlates with 'ACC+' (accuracy).
    """
    # Find indices
    size_idx = header.index('N_estimators')
    acc_idx = header.index('ACC+')
    
    # Split data into small and large setups based on median size
    sizes = [row[size_idx] for row in rows]
    median_size = sum(sizes) / len(sizes)
    
    small_acc = [row[acc_idx] for row in rows if row[size_idx] <= median_size]
    large_acc = [row[acc_idx] for row in rows if row[size_idx] > median_size]
    
    is_noise = certify(small_acc, large_acc)
    
    print(f"Analysis of Accuracy vs Setup Size:")
    print(f"Small setups mean accuracy: {sum(small_acc)/len(small_acc):.4f}")
    print(f"Large setups mean accuracy: {sum(large_acc)/len(large_acc):.4f}")
    
    if is_noise:
        print("Conclusion: The difference is likely just noise.")
    else:
        print("Conclusion: The difference is statistically significant.")
    return not is_noise
