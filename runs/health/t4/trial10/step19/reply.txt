import csv
from sklearn.ensemble import IsolationForest
from sklearn.preprocessing import LabelEncoder

def spot(header, rows):
    """
    Identifies rows that contain input values that are 'wrong' or anomalous
    by treating them as outliers in the input space using an Isolation Forest.
    Returns the indices of the rows, most suspicious first.
    """
    # Identify input columns (those not ending in - or +)
    input_indices = [i for i, name in enumerate(header) if not (name.endswith('-') or name.endswith('+'))]
    
    # Prepare feature matrix
    X = []
    encoders = {}
    
    # Fit encoders for symbolic columns
    for i in input_indices:
        if isinstance(rows[0][i], str):
            le = LabelEncoder()
            # Collect all values to fit the encoder
            col_vals = [str(r[i]) for r in rows]
            le.fit(col_vals)
            encoders[i] = le
            
    for row in rows:
        features = []
        for i in input_indices:
            if i in encoders:
                features.append(encoders[i].transform([str(row[i])])[0])
            else:
                features.append(float(row[i]))
        X.append(features)
    
    # Use Isolation Forest to detect anomalies.
    # A lower score indicates a more anomalous/suspicious row.
    clf = IsolationForest(contamination=0.05, random_state=42)
    clf.fit(X)
    scores = clf.decision_function(X)
    
    # Sort indices by score (lowest score = most suspicious)
    indexed_scores = sorted(enumerate(scores), key=lambda x: x[1])
    
    return [idx for idx, score in indexed_scores]

if __name__ == "__main__":
    filename = 'Health-ClosedIssues0000.n1000.train.csv'
    rows = []
    try:
        with open(filename, 'r') as f:
            reader = csv.reader(f)
            header = next(reader)
            for row in reader:
                processed_row = []
                for val in row:
                    try:
                        processed_row.append(float(val))
                    except ValueError:
                        processed_row.append(val)
                rows.append(processed_row)
        
        suspicious_indices = spot(header, rows)
        
        print("Analysis complete. The following rows contain settings that are")
        print("highly unusual compared to the rest of the dataset (most suspicious first):")
        print(suspicious_indices[:10])
        
    except Exception as e:
        print(f"Error: {e}")
        exit(1)
