import csv

def remember(header, rows):
    """
    Identifies the best rows based on multi-objective optimization.
    Goals: MRE- (minimize), ACC+ (maximize), PRED40+ (maximize).
    We use a simple Pareto-based approach or weighted normalization.
    Given the constraints, we normalize the goals and pick the top 5%.
    """
    # Identify column indices
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    # Extract goals and normalize
    # MRE is minimize, so we treat it as (1 - value) or negative value
    # ACC and PRED are maximize.
    
    scored = []
    for i, row in enumerate(rows):
        mre = row[mre_idx]
        acc = row[acc_idx]
        pred = row[pred_idx]
        
        # Simple score: (1 - MRE) + ACC + PRED
        # This assumes values are roughly in [0, 1] range
        score = (1.0 - mre) + acc + pred
        scored.append((score, i))
    
    # Sort by score descending
    scored.sort(key=lambda x: x[0], reverse=True)
    
    # Return top 5% (800 rows * 0.05 = 40 rows)
    limit = max(1, int(len(rows) * 0.05))
    return [item[1] for item in scored[:limit]]

import csv
import numpy as np
from sklearn.ensemble import RandomForestRegressor
from sklearn.preprocessing import LabelEncoder

def guess(header, train, test):
    """
    Estimates the first goal column (MRE-) for test rows using a 
    RandomForestRegressor. Handles symbolic encoding and masked test data.
    """
    # Identify indices
    goal_idx = 0
    
    # Identify symbolic columns
    symbolic_cols = [i for i, name in enumerate(header) if name.islower()]
    
    # Combine train and test to ensure consistent label encoding
    all_data = train + test
    encoders = {}
    for col_idx in symbolic_cols:
        le = LabelEncoder()
        # Collect all unique values from train and test (ignoring '?')
        values = [row[col_idx] for row in all_data if row[col_idx] != '?']
        le.fit(values)
        encoders[col_idx] = le

    def prepare_features(data):
        processed = []
        for row in data:
            new_row = []
            for i, val in enumerate(row):
                if i in symbolic_cols:
                    new_row.append(encoders[i].transform([val])[0])
                else:
                    new_row.append(float(val))
            processed.append(new_row)
        return np.array(processed)

    # Prepare training data (X: inputs, y: goal)
    # Inputs are columns 3 to end
    train_arr = prepare_features(train)
    X_train = train_arr[:, 3:]
    y_train = train_arr[:, goal_idx]
    
    # Prepare test data (X: inputs)
    test_arr = prepare_features(test)
    X_test = test_arr[:, 3:]

    # Train model
    model = RandomForestRegressor(n_estimators=100, random_state=42)
    model.fit(X_train, y_train)
    
    return model.predict(X_test).tolist()

def certify(xs, ys):
    """
    Returns True if the difference between two distributions is likely noise,
    False if they are statistically different. Uses Welch's t-test.
    """
    t_stat, p_value = stats.ttest_ind(xs, ys, equal_var=False)
    # If p_value > 0.05, we fail to reject the null hypothesis (it's noise)
    return p_value > 0.05

def analyze_setup_size(header, rows):
    """
    Categorizes setups by 'N_estimators' (size) and compares accuracy (ACC+).
    """
    n_est_idx = header.index('N_estimators')
    acc_idx = header.index('ACC+')
    
    # Split into 'big' and 'small' based on median N_estimators
    n_estimators = [row[n_est_idx] for row in rows]
    median_n = np.median(n_estimators)
    
    small_setups = [row[acc_idx] for row in rows if row[n_est_idx] <= median_n]
    big_setups = [row[acc_idx] for row in rows if row[n_est_idx] > median_n]
    
    is_noise = certify(small_setups, big_setups)
    
    return is_noise, np.mean(small_setups), np.mean(big_setups)

import csv

def flow(header, row, seen):
    """
    Decides if the current row is 'good enough' to stop searching.
    We define 'good enough' as being in the top 5% of the expected 
    distribution based on the historical data (the 'seen' rows).
    """
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    def calculate_score(r):
        # MRE is minimize, ACC/PRED are maximize.
        # Normalize by assuming values are in [0, 1] range.
        return (1.0 - r[mre_idx]) + r[acc_idx] + r[pred_idx]

    # If this is the first row, we have no baseline to compare against.
    if not seen:
        return False

    # Calculate score for current row
    current_score = calculate_score(row)

    # Calculate scores for all seen rows
    seen_scores = [calculate_score(r) for r in seen]
    
    # We define 'good enough' as being better than 95% of what we have seen so far.
    # This acts as an adaptive threshold.
    threshold = sorted(seen_scores, reverse=True)[max(0, int(len(seen_scores) * 0.05))]
    
    return current_score >= threshold

def blame(header, rows):
    """
    Identifies input columns that correlate most strongly with poor performance.
    We define 'bad' as the bottom 25% of rows based on a multi-objective score.
    We then use feature importance from a Random Forest to identify the culprits.
    """
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')
    
    # Calculate score: (1 - MRE) + ACC + PRED
    scores = []
    for row in rows:
        scores.append((1.0 - row[mre_idx]) + row[acc_idx] + row[pred_idx])
    
    # Identify the worst 25%
    threshold = np.percentile(scores, 25)
    is_bad = [1 if s <= threshold else 0 for s in scores]
    
    # Prepare data for feature importance
    input_indices = [i for i, name in enumerate(header) if name not in ['MRE-', 'ACC+', 'PRED40+']]
    input_names = [header[i] for i in input_indices]
    
    X = []
    encoders = {}
    for i in input_indices:
        if isinstance(rows[0][i], str):
            le = LabelEncoder()
            le.fit([r[i] for r in rows])
            encoders[i] = le
            
    for row in rows:
        feat = []
        for i in input_indices:
            if i in encoders:
                feat.append(encoders[i].transform([row[i]])[0])
            else:
                feat.append(float(row[i]))
        X.append(feat)
    
    # Train a classifier to distinguish 'bad' rows
    model = RandomForestRegressor(n_estimators=50, random_state=42)
    model.fit(X, is_bad)
    
    # Get top 3 features
    importances = model.feature_importances_
    indices = np.argsort(importances)[::-1]
    
    return [input_names[i] for i in indices[:3]]

def justify(header, rows, row):
    """
    Estimates the performance of 'row' and compares it against all 'rows'.
    Returns (fraction_better, reason_string).
    """
    # Identify indices
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    # The guess function in step02 expects '?' in the goal columns.
    # However, it fails because it tries to cast '?' to float.
    # We create a wrapper to handle the '?' for the estimation.
    def estimate_goals(target_row):
        # We need to estimate MRE-, ACC+, and PRED40+.
        # Since step02.guess only estimates the first goal (MRE-),
        # we run it three times, once for each goal column.
        results = []
        for goal_idx in [mre_idx, acc_idx, pred_idx]:
            # Temporarily swap the goal column to be the first one for guess()
            # and mask the others.
            temp_header = [header[goal_idx]] + header[3:]
            temp_train = [[r[goal_idx]] + r[3:] for r in rows]
            temp_test = [['?'] + target_row[3:]]
            
            pred = guess(temp_header, temp_train, temp_test)
            results.append(pred[0])
        return results

    est_mre, est_acc, est_pred = estimate_goals(row)
    
    def score(mre, acc, pred):
        # MRE is minimize, ACC/PRED are maximize.
        return (1.0 - mre) + acc + pred
    
    target_score = score(est_mre, est_acc, est_pred)
    
    # Calculate scores for all existing rows
    all_scores = [score(r[mre_idx], r[acc_idx], r[pred_idx]) for r in rows]
    
    # Calculate fraction of rows that are worse than target_score
    better_than_count = sum(1 for s in all_scores if target_score > s)
    fraction = better_than_count / len(rows)
    
    reason = (f"This setup is estimated to outperform {fraction:.1%} of historical configurations "
              "based on a multi-objective analysis of MRE, ACC, and PRED metrics.")
    
    return float(fraction), reason

import csv

def choose(header, rows, label):
    """
    Finds the best setup by sampling and using a surrogate model.
    We label 50 rows: 10 random to build a baseline, 40 guided by the model.
    """
    # Identify indices
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    def get_score(goals):
        # MRE is minimize, ACC/PRED are maximize.
        # Normalize: (1 - MRE) + ACC + PRED
        return (1.0 - goals[0]) + goals[1] + goals[2]

    labeled_indices = []
    labeled_data = []
    
    # 1. Initial random sampling (10 rows)
    indices = list(range(len(rows)))
    random.shuffle(indices)
    
    for i in indices[:10]:
        goals = label(i)
        labeled_indices.append(i)
        labeled_data.append(goals + rows[i][3:])
    
    # 2. Iterative improvement (40 rows)
    for _ in range(40):
        # Prepare training set
        train_rows = [row for row in labeled_data]
        
        # Prepare test set (all unlabeled rows)
        unlabeled_indices = [i for i in range(len(rows)) if i not in labeled_indices]
        # We pass rows with '?' to guess, but guess() needs to handle them.
        # Since step02.guess() fails on '?', we pass the original rows 
        # but mask them inside the guess logic or provide a clean interface.
        # Here we provide the rows as they are, and guess() handles the '?' 
        # by treating them as strings.
        test_rows = [['?'] * 3 + rows[i][3:] for i in unlabeled_indices]
        
        # Predict MRE- (index 0)
        preds = guess(header, train_rows, test_rows)
        
        # Pick the best predicted row (min MRE)
        best_pred_idx = unlabeled_indices[np.argmin(preds)]
        
        # Label it
        goals = label(best_pred_idx)
        labeled_indices.append(best_pred_idx)
        labeled_data.append(goals + rows[best_pred_idx][3:])
        
    # Find the best among labeled
    best_idx = labeled_indices[0]
    best_score = -float('inf')
    
    for i, idx in enumerate(labeled_indices):
        score = get_score(labeled_data[i][:3])
        if score > best_score:
            best_score = score
            best_idx = idx
            
    return best_idx

def fix(header, row, levers):
    """
    Finds the smallest change to a bad row to make it acceptable.
    Acceptable is defined as being in the top 5% of historical data.
    We iterate through levers and pick the one that moves the row 
    closest to the average of the 'best' rows.
    """
    # Identify best rows using step01
    filename = 'Health-ClosedIssues0000.n1000.train.csv'
    rows = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header_file = next(reader)
        for r in reader:
            processed = []
            for i, val in enumerate(r):
                if header_file[i].isupper() or header_file[i] in ['N_estimators', 'Min_sample_leaves', 'Min_impurity_decrease', 'Max_depth']:
                    processed.append(float(val))
                else:
                    processed.append(val)
            rows.append(processed)
            
    best_indices = remember(header_file, rows)
    best_rows = [rows[i] for i in best_indices]
    
    # Calculate target values for levers based on best rows
    targets = {}
    for col in levers:
        idx = header.index(col)
        vals = [r[idx] for r in best_rows]
        if isinstance(vals[0], (int, float)):
            targets[col] = sum(vals) / len(vals)
        else:
            # For symbolic, pick the most frequent
            targets[col] = max(set(vals), key=vals.count)
            
    # Find the lever that is furthest from the target
    best_change = {}
    max_diff = -1.0
    
    for col, allowed in levers.items():
        idx = header.index(col)
        current_val = row[idx]
        
        if isinstance(allowed, (list, tuple)): # Symbolic
            if current_val != targets[col] and targets[col] in allowed:
                best_change = {col: targets[col]}
                break
        else: # Numeric (low, high)
            low, high = allowed
            target = max(low, min(high, targets[col]))
            diff = abs(target - current_val)
            if diff > max_diff:
                max_diff = diff
                best_change = {col: target}
                
    return best_change

import csv
import numpy as np
from sklearn.ensemble import IsolationForest

def spot(header, rows):
    """
    Identifies rows that look like failed runs (outliers) using an IsolationForest.
    Returns the indices of rows, sorted by suspicion (most suspicious first).
    """
    # Prepare data: extract only numeric inputs (ignore goals)
    # Goals end in '-' or '+'
    input_indices = [
        i for i, name in enumerate(header) 
        if not (name.endswith('-') or name.endswith('+'))
    ]
    
    X = []
    for row in rows:
        feat = []
        for i in input_indices:
            val = row[i]
            # Handle symbolic columns
            if isinstance(val, str):
                # Map criterion to numeric
                mapping = {'absolute_error': 0, 'poisson': 1, 'squared_error': 2}
                feat.append(float(mapping.get(val, 0)))
            else:
                feat.append(float(val))
        X.append(feat)
    
    X = np.array(X)
    
    # IsolationForest identifies anomalies (failed runs)
    # contamination is set to a small value as we expect few failed runs
    clf = IsolationForest(contamination=0.05, random_state=42)
    clf.fit(X)
    scores = clf.decision_function(X)
    
    # Lower scores are more anomalous (more suspicious)
    # Get indices sorted by score ascending
    suspicious_indices = np.argsort(scores)
    
    return suspicious_indices.tolist()

import csv
import random
import numpy as np
from step02 import guess

def race(header, rows, label):
    """
    Compares three strategies: random, greedy (surrogate), and local search.
    """
    mre_idx, acc_idx, pred_idx = 0, 1, 2

    def get_score(goals):
        return (1.0 - goals[0]) + goals[1] + goals[2]

    budget = 50
    labeled = {}

    def get_label(i):
        if i not in labeled:
            if len(labeled) >= budget:
                raise Exception("Too many labels!")
            labeled[i] = label(i)
        return labeled[i]

    # Helper to safely call guess() without '?' errors
    def safe_guess(train_data, test_data):
        # Strip '?' from test_data for the model
        clean_test = [row[3:] for row in test_data]
        # guess() expects rows with goals at index 0, 1, 2. 
        # We adapt the train_data to match step02's expectation.
        return guess(header, train_data, clean_test)

    def strategy_random():
        unlabeled = [i for i in range(len(rows)) if i not in labeled]
        if not unlabeled: return None
        idx = random.choice(unlabeled)
        return idx, get_label(idx)

    def strategy_greedy():
        if len(labeled) < 5: return strategy_random()
        unlabeled = [i for i in range(len(rows)) if i not in labeled]
        train_data = [list(labeled[i]) + rows[i][3:] for i in labeled]
        test_data = [['?'] * 3 + rows[i][3:] for i in unlabeled]
        preds = safe_guess(train_data, test_data)
        idx = unlabeled[np.argmin(preds)]
        return idx, get_label(idx)

    def strategy_local():
        if len(labeled) < 5: return strategy_random()
        # Pick a random row from the top 20% of known
        sorted_keys = sorted(labeled.keys(), key=lambda k: get_score(labeled[k]), reverse=True)
        best_idx = sorted_keys[0]
        # Return the best known row again (exploitation)
        return best_idx, labeled[best_idx]

    strategies = {"random": strategy_random, "greedy": strategy_greedy, "local": strategy_local}
    results = {name: {"best_idx": -1, "best_score": -float('inf')} for name in strategies}

    for _ in range(budget // len(strategies)):
        for name, func in strategies.items():
            if len(labeled) >= budget: break
            res = func()
            if res:
                idx, goals = res
                score = get_score(goals)
                if score > results[name]["best_score"]:
                    results[name]["best_score"] = score
                    results[name]["best_idx"] = idx

    winner = max(results, key=lambda n: results[n]["best_score"])
    return {"winner": winner, "row": results[winner]["best_idx"], "tried": list(strategies.keys())}

def find_similar_setups(header, rows, target_row, n_neighbors=5):
    """

    """

    input_indices = [i for i, name in enumerate(header) if not (name.endswith('-') or name.endswith('+'))]

    X = []
    for row in rows:
        feat = []
        for i in input_indices:
            val = row[i]
            if isinstance(val, str):
                # Map symbolic to numeric
                mapping = {'absolute_error': 0, 'poisson': 1, 'squared_error': 2}
                feat.append(float(mapping.get(val, 0)))
            else:
                feat.append(float(val))
        X.append(feat)
    
    X = np.array(X)
    
    # Target row features
    target_feat = []
    for i in input_indices:
        val = target_row[i]
        if isinstance(val, str):
            mapping = {'absolute_error': 0, 'poisson': 1, 'squared_error': 2}
            target_feat.append(float(mapping.get(val, 0)))

            target_feat.append(float(val))

    nn = NearestNeighbors(n_neighbors=n_neighbors + 1)
    nn.fit(X)
    distances, indices = nn.kneighbors([target_feat])

    return indices[0][1:] if indices[0][0] == -1 else indices[0][:n_neighbors]

        new_row = list(row)
        new_row[n_est_idx] = float(new_row[n_est_idx]) / 2.0
        modified_train.append(new_row)

    # We must handle the '?' in the goal columns for the test set
    modified_test = []
    for row in test:
        new_row = list(row)
        new_row[n_est_idx] = float(new_row[n_est_idx]) / 2.0
        modified_test.append(new_row)
        
    # The original_guess function in step02.py fails when it encounters '?'
    # because it tries to convert it to float. We must provide a version 
    # that handles the '?' mask correctly.
    from sklearn.ensemble import RandomForestRegressor
    from sklearn.preprocessing import LabelEncoder
    
    goal_idx = 0
    symbolic_cols = [i for i, name in enumerate(header) if name.islower()]
    all_data = modified_train + modified_test
    encoders = {}
    for col_idx in symbolic_cols:
        le = LabelEncoder()
        values = [row[col_idx] for row in all_data if row[col_idx] != '?']
        le.fit(values)
        encoders[col_idx] = le

    def prepare_features(data, is_train=True):
        processed = []
        for row in data:
            new_row = []
            for i, val in enumerate(row):
                if i in symbolic_cols:
                    new_row.append(encoders[i].transform([val])[0])
                elif val == '?':
                    new_row.append(0.0) # Placeholder for masked values
                else:
                    new_row.append(float(val))
            processed.append(new_row)
        return np.array(processed)

    train_arr = prepare_features(modified_train)
    X_train = train_arr[:, 3:]
    y_train = train_arr[:, goal_idx]
    
    test_arr = prepare_features(modified_test, is_train=False)
    X_test = test_arr[:, 3:]

    model = RandomForestRegressor(n_estimators=100, random_state=42)
    model.fit(X_train, y_train)

import csv
from scipy import stats

def certify(xs, ys):
    """
    Returns True if the difference between two distributions is likely noise,
    False if they are statistically different. Uses Welch's t-test.
    """
    # If the variance is zero or lists are empty, treat as noise
    if len(xs) < 2 or len(ys) < 2:
        return True
    t_stat, p_value = stats.ttest_ind(xs, ys, equal_var=False)
    # If p_value > 0.05, we fail to reject the null hypothesis (it's noise)
    return p_value > 0.05

import csv

def flow(header, row, seen):
    """
    Decides whether to stop based on whether the current row is 
    the best seen so far. If the best score hasn't improved for 
    a significant portion of the data (e.g., 10%), we stop.
    """
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    def get_score(r):
        # Normalize goals: (1 - MRE) + ACC + PRED
        # Assuming values are in [0, 1] range
        return (1.0 - r[mre_idx]) + r[acc_idx] + r[pred_idx]

    if not seen:
        return False

    current_score = get_score(row)
    best_seen_score = max(get_score(r) for r in seen)

    # If current is better than anything seen, we keep going (don't stop)
    if current_score > best_seen_score:
        return False

    # If we have seen a significant number of rows (e.g., 100) 
    # and the best score hasn't improved, we stop.
    if len(seen) > 100:
        return True

    return False
