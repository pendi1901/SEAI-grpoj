import csv

def flow(header, row, seen):
    """
    Decides whether to stop based on whether the current row is 
    the best seen so far. If the best score hasn't improved for 
    a significant portion of the data (e.g., 10%), we stop.
    """
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    def get_score(r):
        # Normalize goals: (1 - MRE) + ACC + PRED
        # Assuming values are in [0, 1] range
        return (1.0 - r[mre_idx]) + r[acc_idx] + r[pred_idx]

    if not seen:
        return False

    current_score = get_score(row)
    best_seen_score = max(get_score(r) for r in seen)

    # If current is better than anything seen, we keep going (don't stop)
    if current_score > best_seen_score:
        return False

    # If we have seen a significant number of rows (e.g., 100) 
    # and the best score hasn't improved, we stop.
    if len(seen) > 100:
        return True

    return False

if __name__ == "__main__":
    filename = 'Health-ClosedIssues0000.n1000.train.csv'
    rows = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = []
            for i, val in enumerate(row):
                if header[i].isupper() or header[i] in ['N_estimators', 'Min_sample_leaves', 'Min_impurity_decrease', 'Max_depth']:
                    processed_row.append(float(val))
                else:
                    processed_row.append(val)
            rows.append(processed_row)

    seen = []
    stopped = False
    for i, row in enumerate(rows):
        if flow(header, row, seen):
            print(f"Stopping at row {i + 1}: No significant improvement in performance.")
            stopped = True
            break
        seen.append(row)
    
    if not stopped:
        print("Processed all rows; no stagnation detected.")
