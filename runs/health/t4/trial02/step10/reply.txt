import csv
import random
import numpy as np
from step02 import guess

def race(header, rows, label):
    """
    Compares three strategies: random, greedy (based on surrogate), 
    and exploitation (sampling near best).
    """
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    def get_score(goals):
        return (1.0 - goals[0]) + goals[1] + goals[2]

    budget = 50
    labeled = {}

    def get_label(i):
        if i not in labeled:
            if len(labeled) >= budget:
                raise Exception("Too many labels!")
            labeled[i] = label(i)
        return labeled[i]

    # Strategy 1: Random
    def strategy_random():
        indices = [i for i in range(len(rows)) if i not in labeled]
        if not indices: return None
        idx = random.choice(indices)
        return idx, get_label(idx)

    # Strategy 2: Greedy (Surrogate model)
    def strategy_greedy():
        if len(labeled) < 10: return strategy_random()
        unlabeled = [i for i in range(len(rows)) if i not in labeled]
        train_data = [list(labeled[i]) + rows[i][3:] for i in labeled]
        test_data = [['?'] * 3 + rows[i][3:] for i in unlabeled]
        preds = guess(header, train_data, test_data)
        idx = unlabeled[np.argmin(preds)]
        return idx, get_label(idx)

    # Strategy 3: Exploitation (Randomly pick from top 10% of known)
    def strategy_exploit():
        if len(labeled) < 10: return strategy_random()
        best_so_far = sorted(labeled.keys(), key=lambda i: get_score(labeled[i]), reverse=True)
        # Pick a neighbor of a good row (simple perturbation)
        idx = best_so_far[0]
        return idx, labeled[idx]

    strategies = {"random": strategy_random, "greedy": strategy_greedy, "exploit": strategy_exploit}
    results = {name: {"best_idx": -1, "best_score": -float('inf')} for name in strategies}

    # Run 50 iterations total, split among strategies
    for _ in range(budget):
        for name, func in strategies.items():
            if len(labeled) >= budget: break
            idx, goals = func()
            score = get_score(goals)
            if score > results[name]["best_score"]:
                results[name]["best_score"] = score
                results[name]["best_idx"] = idx

    winner = max(results, key=lambda n: results[n]["best_score"])
    return {"winner": winner, "row": results[winner]["best_idx"], "tried": list(strategies.keys())}

if __name__ == "__main__":
    filename = 'Health-ClosedIssues0000.n1000.train.csv'
    rows = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed = []
            for i, val in enumerate(row):
                if header[i].isupper() or header[i] in ['N_estimators', 'Min_sample_leaves', 'Min_impurity_decrease', 'Max_depth']:
                    processed.append(float(val))
                else:
                    processed.append(val)
            rows.append(processed)

    def label(i):
        return [rows[i][0], rows[i][1], rows[i][2]]

    outcome = race(header, rows, label)
    print(f"Winner: {outcome['winner']}")
    print(f"Best row index found: {outcome['row']}")
    print(f"Strategies tried: {', '.join(outcome['tried'])}")
