import csv

def remember(header, rows):
    """
    Identifies the best rows based on multi-objective optimization.
    Goals: MRE- (minimize), ACC+ (maximize), PRED40+ (maximize).
    We use a simple Pareto-based approach or weighted normalization.
    Given the constraints, we normalize the goals and pick the top 5%.
    """
    # Identify column indices
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    # Extract goals and normalize
    # MRE is minimize, so we treat it as (1 - value) or negative value
    # ACC and PRED are maximize.
    
    scored = []
    for i, row in enumerate(rows):
        mre = row[mre_idx]
        acc = row[acc_idx]
        pred = row[pred_idx]
        
        # Simple score: (1 - MRE) + ACC + PRED
        # This assumes values are roughly in [0, 1] range
        score = (1.0 - mre) + acc + pred
        scored.append((score, i))
    
    # Sort by score descending
    scored.sort(key=lambda x: x[0], reverse=True)
    
    # Return top 5% (800 rows * 0.05 = 40 rows)
    limit = max(1, int(len(rows) * 0.05))
    return [item[1] for item in scored[:limit]]

import csv
import numpy as np
from sklearn.ensemble import RandomForestRegressor
from sklearn.preprocessing import LabelEncoder

def guess(header, train, test):
    """
    Estimates the first goal column (MRE-) for test rows using a 
    RandomForestRegressor. Handles symbolic encoding and masked test data.
    """
    # Identify indices
    goal_idx = 0
    
    # Identify symbolic columns
    symbolic_cols = [i for i, name in enumerate(header) if name.islower()]
    
    # Combine train and test to ensure consistent label encoding
    all_data = train + test
    encoders = {}
    for col_idx in symbolic_cols:
        le = LabelEncoder()
        # Collect all unique values from train and test (ignoring '?')
        values = [row[col_idx] for row in all_data if row[col_idx] != '?']
        le.fit(values)
        encoders[col_idx] = le

    def prepare_features(data):
        processed = []
        for row in data:
            new_row = []
            for i, val in enumerate(row):
                if i in symbolic_cols:
                    new_row.append(encoders[i].transform([val])[0])
                else:
                    new_row.append(float(val))
            processed.append(new_row)
        return np.array(processed)

    # Prepare training data (X: inputs, y: goal)
    # Inputs are columns 3 to end
    train_arr = prepare_features(train)
    X_train = train_arr[:, 3:]
    y_train = train_arr[:, goal_idx]
    
    # Prepare test data (X: inputs)
    test_arr = prepare_features(test)
    X_test = test_arr[:, 3:]

    # Train model
    model = RandomForestRegressor(n_estimators=100, random_state=42)
    model.fit(X_train, y_train)
    
    return model.predict(X_test).tolist()

def certify(xs, ys):
    """
    Returns True if the difference between two distributions is likely noise,
    False if they are statistically different. Uses Welch's t-test.
    """
    t_stat, p_value = stats.ttest_ind(xs, ys, equal_var=False)
    # If p_value > 0.05, we fail to reject the null hypothesis (it's noise)
    return p_value > 0.05

def analyze_setup_size(header, rows):
    """
    Categorizes setups by 'N_estimators' (size) and compares accuracy (ACC+).
    """
    n_est_idx = header.index('N_estimators')
    acc_idx = header.index('ACC+')
    
    # Split into 'big' and 'small' based on median N_estimators
    n_estimators = [row[n_est_idx] for row in rows]
    median_n = np.median(n_estimators)
    
    small_setups = [row[acc_idx] for row in rows if row[n_est_idx] <= median_n]
    big_setups = [row[acc_idx] for row in rows if row[n_est_idx] > median_n]
    
    is_noise = certify(small_setups, big_setups)
    
    return is_noise, np.mean(small_setups), np.mean(big_setups)

import csv

def flow(header, row, seen):
    """
    Decides if the current row is 'good enough' to stop searching.
    We define 'good enough' as being in the top 5% of the expected 
    distribution based on the historical data (the 'seen' rows).
    """
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    def calculate_score(r):
        # MRE is minimize, ACC/PRED are maximize.
        # Normalize by assuming values are in [0, 1] range.
        return (1.0 - r[mre_idx]) + r[acc_idx] + r[pred_idx]

    # If this is the first row, we have no baseline to compare against.
    if not seen:
        return False

    # Calculate score for current row
    current_score = calculate_score(row)

    # Calculate scores for all seen rows
    seen_scores = [calculate_score(r) for r in seen]
    
    # We define 'good enough' as being better than 95% of what we have seen so far.
    # This acts as an adaptive threshold.
    threshold = sorted(seen_scores, reverse=True)[max(0, int(len(seen_scores) * 0.05))]
    
    return current_score >= threshold

def blame(header, rows):
    """
    Identifies input columns that correlate most strongly with poor performance.
    We define 'bad' as the bottom 25% of rows based on a multi-objective score.
    We then use feature importance from a Random Forest to identify the culprits.
    """
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')
    
    # Calculate score: (1 - MRE) + ACC + PRED
    scores = []
    for row in rows:
        scores.append((1.0 - row[mre_idx]) + row[acc_idx] + row[pred_idx])
    
    # Identify the worst 25%
    threshold = np.percentile(scores, 25)
    is_bad = [1 if s <= threshold else 0 for s in scores]
    
    # Prepare data for feature importance
    input_indices = [i for i, name in enumerate(header) if name not in ['MRE-', 'ACC+', 'PRED40+']]
    input_names = [header[i] for i in input_indices]
    
    X = []
    encoders = {}
    for i in input_indices:
        if isinstance(rows[0][i], str):
            le = LabelEncoder()
            le.fit([r[i] for r in rows])
            encoders[i] = le
            
    for row in rows:
        feat = []
        for i in input_indices:
            if i in encoders:
                feat.append(encoders[i].transform([row[i]])[0])
            else:
                feat.append(float(row[i]))
        X.append(feat)
    
    # Train a classifier to distinguish 'bad' rows
    model = RandomForestRegressor(n_estimators=50, random_state=42)
    model.fit(X, is_bad)
    
    # Get top 3 features
    importances = model.feature_importances_
    indices = np.argsort(importances)[::-1]
    
    return [input_names[i] for i in indices[:3]]
