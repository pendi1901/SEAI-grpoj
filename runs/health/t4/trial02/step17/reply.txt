import csv
import random
import numpy as np
from step02 import guess
from step01 import remember

def choose(header, rows, label):
    """
    Finds a setup that is both accurate (high score) and small (low N_estimators).
    We use a multi-objective score: (1 - MRE) + ACC + PRED - (N_estimators / 200).
    We use a surrogate model to guide the search within a 50-label budget.
    """
    mre_idx, acc_idx, pred_idx = 0, 1, 2
    n_est_idx = header.index('N_estimators')

    def get_score(row_idx, goals):
        # Normalize N_estimators to [0, 1] range for the penalty
        # Max N_estimators in data is 200
        n_est = float(rows[row_idx][n_est_idx])
        penalty = n_est / 200.0
        return (1.0 - goals[0]) + goals[1] + goals[2] - penalty

    labeled = {}
    
    def get_label(i):
        if i not in labeled:
            if len(labeled) >= 50:
                raise Exception("Too many labels!")
            labeled[i] = label(i)
        return labeled[i]

    # 1. Initial random sampling (10 rows)
    indices = list(range(len(rows)))
    random.shuffle(indices)
    for i in indices[:10]:
        get_label(i)

    # 2. Iterative improvement (40 rows)
    for _ in range(40):
        # Prepare training set
        train_data = [list(labeled[i]) + rows[i][3:] for i in labeled]
        unlabeled = [i for i in range(len(rows)) if i not in labeled]
        
        # Predict MRE- (index 0)
        test_data = [['?'] * 3 + rows[i][3:] for i in unlabeled]
        preds = guess(header, train_data, test_data)
        
        # Heuristic: pick row with best predicted MRE
        best_pred_idx = unlabeled[np.argmin(preds)]
        get_label(best_pred_idx)

    # Find the best among labeled
    best_idx = -1
    best_score = -float('inf')
    for idx, goals in labeled.items():
        score = get_score(idx, goals)
        if score > best_score:
            best_score = score
            best_idx = idx
            
    return best_idx

if __name__ == "__main__":
    filename = 'Health-ClosedIssues0000.n1000.train.csv'

        reader = csv.reader(f)

                    processed.append(float(val))
                else:
                    processed.append(val)
            rows.append(processed)

    call_count = 0
    def label(i):
        global call_count
        call_count += 1
        return [rows[i][0], rows[i][1], rows[i][2]]

    best_idx = choose(header, rows, label)
    print(f"Found optimal setup at index {best_idx}.")
    print(f"Setup details: N_estimators={rows[best_idx][header.index('N_estimators')]}, "
          f"Criterion={rows[best_idx][header.index('criterion')]}")
