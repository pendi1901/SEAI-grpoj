import csv
import step01
import step02

def justify(header, rows, row):
    """
    Estimates the performance of the target row and compares it against
    the historical distribution of all rows.
    """
    # 1. Estimate the missing goal values for the target row
    # Use step02's logic: train on all rows, predict for the target row
    # We need to handle the '?' in the row for the prediction
    train_data = [r[:] for r in rows]
    test_data = [row[:]]
    
    # Predict MRE- (index 0)
    # Note: step02.guess predicts MRE-. We also need ACC+ and PRED40+.
    # For simplicity, we use the average of the top 5% (from step01) 
    # or a simple KNN approach for all goals.
    # Here, we use the KNN logic from step02 for all goals.
    
    def get_predictions(target_row):
        # Re-use step02 logic to predict all 3 goals
        # Since step02.guess only does MRE-, we extend the logic here
        from sklearn.neighbors import KNeighborsRegressor
        from sklearn.preprocessing import LabelEncoder
        
        goal_indices = [0, 1, 2] # MRE-, ACC+, PRED40+
        results = []
        
        for g_idx in goal_indices:
            encoders = {}
            for i, col in enumerate(header):
                if not col[0].isupper():
                    encoders[i] = LabelEncoder()
                    encoders[i].fit([r[i] for r in rows])
            
            def transform(r):
                new = []
                for i, val in enumerate(r):
                    if i in goal_indices: continue
                    if i in encoders: new.append(float(encoders[i].transform([val])[0]))
                    else: new.append(float(val))
                return new
            
            X = [transform(r) for r in rows]
            y = [r[g_idx] for r in rows]
            model = KNeighborsRegressor(n_neighbors=5).fit(X, y)
            results.append(model.predict([transform(target_row)])[0])
        return results

    mre, acc, pred = get_predictions(row)
    
    # 2. Calculate utility for target and all rows
    def calc_utility(m, a, p):
        return (1.0 - m) + a + p
    
    target_utility = calc_utility(mre, acc, pred)
    all_utilities = [calc_utility(r[0], r[1], r[2]) for r in rows]
    
    # 3. Calculate fraction of rows that are worse
    better_than_count = sum(1 for u in all_utilities if target_utility > u)
    p = better_than_count / len(rows)
    
    reason = f"This setup achieves a utility score of {target_utility:.2f}, outperforming {p:.1%} of all historical configurations."
    
    return p, reason

if __name__ == "__main__":
    filename = 'Health-ClosedIssues0000.n1000.train.csv'

    # Test with the first row as a sample
    test_row = rows[0][:]
    test_row[0], test_row[1], test_row[2] = "?", "?", "?"
    
    p, reason = justify(header, rows, test_row)
    print(f"Performance percentile: {p:.4f}")
    print(f"Justification: {reason}")
