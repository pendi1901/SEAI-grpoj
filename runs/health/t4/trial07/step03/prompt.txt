import csv

def remember(header, rows):
    """
    Identifies the best rows based on multi-objective optimization.
    Goals: MRE- (minimize), ACC+ (maximize), PRED40+ (maximize).
    We use a simple Pareto-based approach or weighted normalization.
    Given the constraints, we normalize the goals and pick the top 5%.
    """
    # Identify column indices
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    # Extract goals and normalize
    # MRE is minimize, so we treat it as (1 - value) for maximization
    scores = []
    for i, row in enumerate(rows):
        mre = row[mre_idx]
        acc = row[acc_idx]
        pred = row[pred_idx]
        
        # Simple utility: lower MRE is better, higher ACC/PRED is better
        # We use (1 - MRE) + ACC + PRED as a heuristic score
        utility = (1.0 - mre) + acc + pred
        scores.append((utility, i))

    # Sort by utility descending
    scores.sort(key=lambda x: x[0], reverse=True)

    # Return top 5% (800 rows * 0.05 = 40 rows)
    limit = max(1, int(len(rows) * 0.05))
    return [item[1] for item in scores[:limit]]

import csv
from sklearn.neighbors import KNeighborsRegressor
from sklearn.preprocessing import LabelEncoder
import numpy as np
import step01

def guess(header, train, test):
    """
    Estimates the first goal column (MRE-) for test rows using K-Nearest Neighbors.
    Handles '?' in test rows by skipping the goal column during transformation.
    """
    goal_idx = 0  # MRE-
    
    # Prepare encoders for symbolic columns
    encoders = {}
    for i, col in enumerate(header):
        if not col[0].isupper():
            encoders[i] = LabelEncoder()
            all_vals = [r[i] for r in train] + [r[i] for r in test if r[i] != '?']
            encoders[i].fit(all_vals)

    def transform_row(row):
        new_row = []
        for i, val in enumerate(row):
            # Skip the goal column index
            if i == goal_idx:
                continue
            # Handle symbolic columns
            if i in encoders:
                new_row.append(float(encoders[i].transform([val])[0]))
            else:
                # Numeric columns
                new_row.append(float(val))
        return new_row

    X_train = [transform_row(row) for row in train]
    y_train = [row[goal_idx] for row in train]
    X_test = [transform_row(row) for row in test]
    
    # Use KNN to predict the MRE- value
    model = KNeighborsRegressor(n_neighbors=5)
    model.fit(X_train, y_train)
    
    return model.predict(X_test).tolist()
