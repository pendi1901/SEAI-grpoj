import csv

def remember(header, rows):
    """
    Identifies the best rows based on multi-objective optimization.
    Goals: MRE- (minimize), ACC+ (maximize), PRED40+ (maximize).
    We use a simple Pareto-based approach or weighted normalization.
    Given the constraints, we normalize the goals and pick the top 5%.
    """
    # Identify column indices
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    # Extract goals and normalize
    # MRE is minimize, so we treat it as (1 - value) for maximization
    scores = []
    for i, row in enumerate(rows):
        mre = row[mre_idx]
        acc = row[acc_idx]
        pred = row[pred_idx]
        
        # Simple utility: lower MRE is better, higher ACC/PRED is better
        # We use (1 - MRE) + ACC + PRED as a heuristic score
        utility = (1.0 - mre) + acc + pred
        scores.append((utility, i))

    # Sort by utility descending
    scores.sort(key=lambda x: x[0], reverse=True)

    # Return top 5% (800 rows * 0.05 = 40 rows)
    limit = max(1, int(len(rows) * 0.05))
    return [item[1] for item in scores[:limit]]

def guess(header, train, test):
    """
    Estimates the first goal column (MRE-) for test rows using K-Nearest Neighbors.
    Handles '?' in test rows by skipping the goal column during transformation.
    """
    goal_idx = 0  # MRE-
    
    # Prepare encoders for symbolic columns
    encoders = {}
    for i, col in enumerate(header):
        if not col[0].isupper():
            encoders[i] = LabelEncoder()
            all_vals = [r[i] for r in train] + [r[i] for r in test if r[i] != '?']
            encoders[i].fit(all_vals)

    def transform_row(row):
        new_row = []
        for i, val in enumerate(row):
            # Skip the goal column index
            if i == goal_idx:
                continue
            # Handle symbolic columns
            if i in encoders:
                new_row.append(float(encoders[i].transform([val])[0]))
            else:
                # Numeric columns
                new_row.append(float(val))
        return new_row

    X_train = [transform_row(row) for row in train]
    y_train = [row[goal_idx] for row in train]
    X_test = [transform_row(row) for row in test]
    
    # Use KNN to predict the MRE- value
    model = KNeighborsRegressor(n_neighbors=5)
    model.fit(X_train, y_train)
    
    return model.predict(X_test).tolist()

import csv
import numpy as np
from scipy import stats

def certify(xs, ys):
    """
    Returns True if the difference between two sets of data is likely noise,
    False if they are statistically different. Uses a Welch's t-test.
    """
    if len(xs) < 2 or len(ys) < 2:
        return True
    
    # Welch's t-test (equal_var=False)
    t_stat, p_value = stats.ttest_ind(xs, ys, equal_var=False)
    
    # If p-value > 0.05, we fail to reject the null hypothesis (they are the same)
    return p_value > 0.05

def analyze_correlation(header, rows):
    """
    Checks if 'N_estimators' (a proxy for setup size) correlates with 
    the performance goals (MRE-, ACC+, PRED40+).
    """
    n_idx = header.index('N_estimators')
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    # Split data into small vs large based on median N_estimators
    n_vals = [r[n_idx] for r in rows]
    median_n = np.median(n_vals)
    
    small = [r for r in rows if r[n_idx] <= median_n]
    large = [r for r in rows if r[n_idx] > median_n]
    
    # Compare ACC+ (a primary accuracy metric)
    acc_small = [r[acc_idx] for r in small]
    acc_large = [r[acc_idx] for r in large]
    
    is_noise = certify(acc_small, acc_large)
    
    return is_noise, np.mean(acc_small), np.mean(acc_large)

import csv
import step01

def flow(header, row, seen):
    """
    Decides if the current row is 'good enough' to stop the build process.
    We define 'good enough' as being in the top 5% of the expected distribution
    based on the historical data provided in step01.
    """
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    # Calculate utility for the current row
    # MRE is minimize, so (1 - MRE) is used for maximization
    current_utility = (1.0 - row[mre_idx]) + row[acc_idx] + row[pred_idx]

    # If no history, we cannot judge, so keep going
    if not seen:
        return False

    # Calculate utilities of all seen rows
    seen_utilities = []
    for r in seen:
        seen_utilities.append((1.0 - r[mre_idx]) + r[acc_idx] + r[pred_idx])

    # Threshold: Is this row better than 95% of what we have seen so far?
    # Or, if we have enough data, is it in the top 5% of the total expected range?
    # Here we use the 95th percentile of the seen rows as the stopping criterion.
    seen_utilities.sort(reverse=True)
    threshold_idx = max(0, int(len(seen_utilities) * 0.05))
    threshold = seen_utilities[threshold_idx]

    return current_utility >= threshold
