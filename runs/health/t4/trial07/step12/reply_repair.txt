import csv
import numpy as np
from sklearn.neighbors import KNeighborsRegressor
from sklearn.preprocessing import LabelEncoder

def guess(header, train, test):
    """
    Estimates the first goal column (MRE-) for test rows using K-Nearest Neighbors.
    Handles '?' in test rows by skipping the goal column during transformation.
    """
    goal_idx = 0  # MRE-
    
    # Prepare encoders for symbolic columns
    encoders = {}
    for i, col in enumerate(header):
        if not col[0].isupper():
            encoders[i] = LabelEncoder()
            # Fit on all available data, ignoring '?'
            all_vals = [r[i] for r in train] + [r[i] for r in test if r[i] != '?']
            encoders[i].fit(all_vals)

    def transform_row(row):
        new_row = []
        for i, val in enumerate(row):
            # Skip the goal column index
            if i == goal_idx:
                continue
            # Handle symbolic columns
            if i in encoders:
                # If value is '?', we need a fallback or skip. 
                # Since we are predicting, we use a default or handle it.
                # Given the context, '?' only appears in the goal column (index 0).
                new_row.append(float(encoders[i].transform([val])[0]))
            else:
                # Numeric columns
                new_row.append(float(val))
        return new_row

    X_train = [transform_row(row) for row in train]
    y_train = [row[goal_idx] for row in train]
    X_test = [transform_row(row) for row in test]
    
    # Use KNN to predict the MRE- value
    model = KNeighborsRegressor(n_neighbors=5)
    model.fit(X_train, y_train)
    
    return model.predict(X_test).tolist()

def evaluate_impact(header, rows):
    """
    Compares the performance of a standard KNN (n=5) vs a halved (n=2) forecaster.
    """
    split = int(len(rows) * 0.8)
    train_data = rows[:split]
    test_data = rows[split:]
    
    # Prepare test data by masking goals
    masked_test = [r[:] for r in test_data]
    for r in masked_test:
        r[0] = "?"
        
    def get_preds(n):
        goal_idx = 0
        encoders = {}
        for i, col in enumerate(header):
            if not col[0].isupper():
                encoders[i] = LabelEncoder()
                encoders[i].fit([r[i] for r in train_data])
        
        def transform(row):
            new = []
            for i, val in enumerate(row):
                if i == goal_idx: continue
                if i in encoders: new.append(float(encoders[i].transform([val])[0]))
                else: new.append(float(val))
            return new
            
        X_train = [transform(r) for r in train_data]
        y_train = [r[goal_idx] for r in train_data]
        model = KNeighborsRegressor(n_neighbors=n).fit(X_train, y_train)
        return model.predict([transform(r) for r in masked_test])

    preds_full = get_preds(5)
    preds_half = get_preds(2)
    
    actuals = [r[0] for r in test_data]
    
    err_full = np.mean(np.abs(np.array(preds_full) - np.array(actuals)))
    err_half = np.mean(np.abs(np.array(preds_half) - np.array(actuals)))
    
    return err_full, err_half

if __name__ == "__main__":
    filename = 'Health-ClosedIssues0000.n1000.train.csv'
    rows = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed = [float(v) if header[i][0].isupper() else v for i, v in enumerate(row)]
            rows.append(processed)
            
    err_full, err_half = evaluate_impact(header, rows)
    
    print("Impact of halving the forecaster size (n=5 to n=2):")
    print(f"Mean Absolute Error (Full): {err_full:.4f}")
    print(f"Mean Absolute Error (Halved): {err_half:.4f}")
    diff = err_half - err_full
    print(f"The forecast error increases by {diff:.4f} on average when halving the forecaster size.")
