import csv

def remember(header, rows):
    """
    Identifies the best rows based on multi-objective optimization.
    Goals: MRE- (minimize), ACC+ (maximize), PRED40+ (maximize).
    We use a simple Pareto-based approach or weighted normalization.
    Given the constraints, we normalize the goals and pick the top 5%.
    """
    # Identify column indices
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    # Extract goals and normalize
    # MRE is minimize, so we treat it as (1 - value) for maximization
    scores = []
    for i, row in enumerate(rows):
        mre = row[mre_idx]
        acc = row[acc_idx]
        pred = row[pred_idx]
        
        # Simple utility: lower MRE is better, higher ACC/PRED is better
        # We use (1 - MRE) + ACC + PRED as a heuristic score
        utility = (1.0 - mre) + acc + pred
        scores.append((utility, i))

    # Sort by utility descending
    scores.sort(key=lambda x: x[0], reverse=True)

    # Return top 5% (800 rows * 0.05 = 40 rows)
    limit = max(1, int(len(rows) * 0.05))
    return [item[1] for item in scores[:limit]]

import csv
import numpy as np
from scipy import stats

def certify(xs, ys):
    """
    Returns True if the difference between two sets of data is likely noise,
    False if they are statistically different. Uses a Welch's t-test.
    """
    if len(xs) < 2 or len(ys) < 2:
        return True
    
    # Welch's t-test (equal_var=False)
    t_stat, p_value = stats.ttest_ind(xs, ys, equal_var=False)
    
    # If p-value > 0.05, we fail to reject the null hypothesis (they are the same)
    return p_value > 0.05

def analyze_correlation(header, rows):
    """
    Checks if 'N_estimators' (a proxy for setup size) correlates with 
    the performance goals (MRE-, ACC+, PRED40+).
    """

    mre_idx = header.index('MRE-')

    pred_idx = header.index('PRED40+')

    # Split data into small vs large based on median N_estimators
    n_vals = [r[n_idx] for r in rows]
    median_n = np.median(n_vals)
    
    small = [r for r in rows if r[n_idx] <= median_n]

    acc_small = [r[acc_idx] for r in small]
    acc_large = [r[acc_idx] for r in large]
    
    is_noise = certify(acc_small, acc_large)
    
    return is_noise, np.mean(acc_small), np.mean(acc_large)

    Decides if the current row is 'good enough' to stop the build process.

    """

    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    # Calculate utility for the current row
    # MRE is minimize, so (1 - MRE) is used for maximization
    current_utility = (1.0 - row[mre_idx]) + row[acc_idx] + row[pred_idx]

    # If no history, we cannot judge, so keep going
    if not seen:
        return False

    # Calculate utilities of all seen rows
    seen_utilities = []
    for r in seen:
        seen_utilities.append((1.0 - r[mre_idx]) + r[acc_idx] + r[pred_idx])

    # Threshold: Is this row better than 95% of what we have seen so far?
    # Or, if we have enough data, is it in the top 5% of the total expected range?

    seen_utilities.sort(reverse=True)
    threshold_idx = max(0, int(len(seen_utilities) * 0.05))
    threshold = seen_utilities[threshold_idx]

    return current_utility >= threshold

import csv

def choose(header, rows, label):
    """
    Find the best row among 50 trials.
    Strategy:
    1. Randomly sample 10 rows to build a training set.
    2. Use KNN to predict utility for all rows.
    3. Iteratively label the best predicted rows until 50 labels are used.
    """
    n_rows = len(rows)
    goal_indices = [0, 1, 2]  # MRE-, ACC+, PRED40+

    def get_utility(row):
        return (1.0 - row[0]) + row[1] + row[2]

    def transform(row):
        new_row = []
        for i, val in enumerate(row):
            if i in goal_indices:
                continue
            if isinstance(val, str):
                new_row.append(float(hash(val) % 1000))
            else:
                new_row.append(float(val))
        return new_row

    labeled_indices = []
    
    # 1. Initial random sample
    for i in random.sample(range(n_rows), 10):
        label(i)
        labeled_indices.append(i)

    # 2. Iteratively pick best candidates
    for _ in range(40):
        # Train model on currently labeled rows
        X_train = [transform(rows[i]) for i in labeled_indices]
        y_train = [get_utility(rows[i]) for i in labeled_indices]
        
        model = KNeighborsRegressor(n_neighbors=min(len(labeled_indices), 5))
        model.fit(X_train, y_train)
        
        # Predict for all unlabeled
        best_u = -float('inf')
        best_i = -1
        
        for i in range(n_rows):
            if i not in labeled_indices:
                u = model.predict([transform(rows[i])])[0]
                if u > best_u:
                    best_u = u
                    best_i = i
        
        label(best_i)
        labeled_indices.append(best_i)

    # 3. Return best found
    best_idx = max(labeled_indices, key=lambda i: get_utility(rows[i]))
    return best_idx

    """
    Finds a change to a bad row that improves its utility.

    closer to the characteristics of the top 5% of rows.

        # Note: In the provided row, goals are at indices 0, 1, 2
        return (1.0 - r[0]) + r[1] + r[2]

    # We need a baseline to compare against. 
    # Since we don't have the full dataset inside the function, 
    # we rely on the fact that we want to maximize the utility.
    # We will pick the change that results in the highest utility 
    # based on the provided row's current state.
    
    best_change = {}
    best_utility = -float('inf')

    # Since we cannot predict the exact outcome without a model, 
    # we choose a value that is likely to be beneficial based on 
    # common trends (e.g., higher N_estimators often helps).
    
    for col_name, allowed in levers.items():
        if col_name not in header:
            continue
            
        idx = header.index(col_name)
        original_val = row[idx]
        
        candidates = []
        if isinstance(allowed, list):  # Symbolic
            candidates = [v for v in allowed if v != original_val]
        else:  # Numeric (low, high)

            # Try boundaries and midpoint
            candidates = [low, (low + high) / 2.0, high]
            candidates = [c for c in candidates if c != original_val]
            
        for val in candidates:
            # Heuristic: If we don't have a model, we pick the first valid change
            # that is within the allowed range.
            return {col_name: val}
                
    return best_change

def spot(header, rows):
    """
    Identifies rows that look like failed runs (outliers) using an Isolation Forest.
    Failed runs often exhibit anomalous input combinations that lead to poor 
    performance or inconsistent behavior.
    """
    # Identify input columns (exclude goals)
    goal_names = {'MRE-', 'ACC+', 'PRED40+'}
    input_indices = [i for i, name in enumerate(header) if name not in goal_names]
    
    # Prepare data for anomaly detection
    X = []
    for row in rows:
        processed = []
        for i in input_indices:
            val = row[i]
            if isinstance(val, str):
                # Simple hash for symbolic columns
                processed.append(float(hash(val) % 1000))
            else:
                processed.append(float(val))
        X.append(processed)
    
    # Isolation Forest is effective for detecting anomalies in high-dimensional data
    # We must fit the model before calling decision_function
    clf = IsolationForest(contamination=0.05, random_state=42)
    clf.fit(X)
    scores = clf.decision_function(X)
    
    # Lower scores indicate more anomalous (suspicious) rows
    # Sort indices by score ascending (most suspicious first)
    suspicious_indices = np.argsort(scores)
    
    return suspicious_indices.tolist()

def race(header, rows, label):

        indices = random.sample(range(n_rows), budget)

        return max(indices, key=lambda i: get_utility(rows[i]))

    def strategy_knn(budget):
        # Start with 5 random
        initial = random.sample(range(n_rows), 5)
        for i in initial: safe_label(i)
        
        for _ in range(budget - 5):
            labeled_list = list(labeled_indices)
            X_train = [transform(rows[i], goal_indices) for i in labeled_list]
            y_train = [get_utility(rows[i]) for i in labeled_list]
            model = KNeighborsRegressor(n_neighbors=min(len(labeled_list), 3)).fit(X_train, y_train)
            
            best_u, best_i = -float('inf'), -1
            for i in range(n_rows):
                if i not in labeled_indices:
                    u = model.predict([transform(rows[i], goal_indices)])[0]
                    if u > best_u: best_u, best_i = u, i
            if best_i != -1: safe_label(best_i)
        return max(list(labeled_indices), key=lambda i: get_utility(rows[i]))

    def strategy_greedy(budget):
        # Sample every Nth row
        indices = list(range(0, n_rows, n_rows // budget))[:budget]
        for i in indices: safe_label(i)
        return max(indices, key=lambda i: get_utility(rows[i]))

    # Budget 50 shared: 16, 17, 17
    best_random = strategy_random(16)
    best_knn = strategy_knn(17)
    best_greedy = strategy_greedy(17)
    
    results = {
        "random": (get_utility(rows[best_random]), best_random),
        "knn": (get_utility(rows[best_knn]), best_knn),
        "greedy": (get_utility(rows[best_greedy]), best_greedy)
    }
    
    winner = max(results, key=lambda k: results[k][0])
    return {"winner": winner, "row": results[winner][1], "tried": ["random", "knn", "greedy"]}

def guess(header, train, test):
    """
    Estimates the first goal column (MRE-) for test rows using K-Nearest Neighbors.
    Handles '?' in test rows by skipping the goal column during transformation.
    """
    goal_idx = 0  # MRE-
    
    # Prepare encoders for symbolic columns
    encoders = {}
    for i, col in enumerate(header):
        if not col[0].isupper():
            encoders[i] = LabelEncoder()
            # Fit on all available data, ignoring '?'
            all_vals = [r[i] for r in train] + [r[i] for r in test if r[i] != '?']
            encoders[i].fit(all_vals)

    def transform_row(row):
        new_row = []
        for i, val in enumerate(row):
            # Skip the goal column index
            if i == goal_idx:
                continue
            # Handle symbolic columns
            if i in encoders:
                # If value is '?', we need a fallback or skip. 
                # Since we are predicting, we use a default or handle it.
                # Given the context, '?' only appears in the goal column (index 0).
                new_row.append(float(encoders[i].transform([val])[0]))
            else:
                # Numeric columns
                new_row.append(float(val))
        return new_row

    X_train = [transform_row(row) for row in train]
    y_train = [row[goal_idx] for row in train]
    X_test = [transform_row(row) for row in test]
    
    # Use KNN to predict the MRE- value
    model = KNeighborsRegressor(n_neighbors=5)
    model.fit(X_train, y_train)
    
    return model.predict(X_test).tolist()

    """

    for r in masked_test:
        r[0] = "?"

        encoders = {}
        for i, col in enumerate(header):

                encoders[i].fit([r[i] for r in train_data])

                if i == goal_idx: continue
                if i in encoders: new.append(float(encoders[i].transform([val])[0]))
                else: new.append(float(val))

def compare_rules(header, rows):
    """
    Compares two hypothetical rules (e.g., small vs large setups) 
    to see if they result in statistically different performance.
    """
    # Use N_estimators as a proxy for the two "rules" or configurations
    n_idx = header.index('N_estimators')
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    # Define two groups based on a split in the data
    n_vals = [r[n_idx] for r in rows]
    median_n = np.median(n_vals)
    
    group1 = [r for r in rows if r[n_idx] <= median_n]
    group2 = [r for r in rows if r[n_idx] > median_n]
    
    # Calculate utility for both groups
    def get_utility(r):
        return (1.0 - r[mre_idx]) + r[acc_idx] + r[pred_idx]
    
    u1 = [get_utility(r) for r in group1]
    u2 = [get_utility(r) for r in group2]
    
    return certify(u1, u2), np.mean(u1), np.mean(u2)

import csv
import step01

def flow(header, row, seen):
    """
    Determines if the current row is the best seen so far.
    If the current row's utility is higher than any seen before, 
    we consider it an improvement and continue. If it is not, 
    we check if we have seen enough data to conclude that 
    improvements have plateaued.
    """
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    def get_utility(r):
        return (1.0 - r[mre_idx]) + r[acc_idx] + r[pred_idx]

    current_utility = get_utility(row)

    if not seen:
        return False

    # Calculate max utility seen so far
    max_seen = max(get_utility(r) for r in seen)

    # If this row is better than anything seen, keep going (don't stop)
    if current_utility > max_seen:
        return False

    # If we have seen a significant number of rows (e.g., 100) 
    # and no improvement has been made, stop.
    if len(seen) > 100:
        return True

    return False

def get_size(header, row):
    """Proxy for 'small enough to run every night': Minimize N_estimators and Max_depth."""
    n_idx = header.index('N_estimators')
    d_idx = header.index('Max_depth')
    return (float(row[n_idx]) / 200.0) + (float(row[d_idx]) / 20.0)

def transform(header, row):
    """Transforms a row into a numeric vector for model input."""
    vec = []
    for i, val in enumerate(row):
        if i < 3: continue  # Skip goals
        if isinstance(val, str):
            vec.append(float(hash(val) % 1000))
        else:
            vec.append(float(val))
    return vec

def choose(header, rows, label):
    """
    Finds a setup that forecasts well (high utility) and is small (low size).
    Optimizes for: Utility - (0.5 * NormalizedSize).
    """
    n_rows = len(rows)
    labeled_indices = []
    
    def get_score(row):
        return get_utility(row) - (0.5 * get_size(header, row))

    # Initial random sample
    for i in random.sample(range(n_rows), 10):
        label(i)
        labeled_indices.append(i)

    # Iteratively pick best candidates based on predicted score
    for _ in range(40):
        X_train = [transform(header, rows[i]) for i in labeled_indices]
        y_train = [get_score(rows[i]) for i in labeled_indices]
        
        model = KNeighborsRegressor(n_neighbors=min(len(labeled_indices), 5))
        model.fit(X_train, y_train)
        
        best_pred = -float('inf')

            if i not in labeled_indices:
                pred = model.predict([transform(header, rows[i])])[0]
                if pred > best_pred:
                    best_pred = pred
                    best_i = i
        
        label(best_i)
        labeled_indices.append(best_i)

    return max(labeled_indices, key=lambda i: get_score(rows[i]))

            u = 0 # Placeholder for utility estimation
            if u >= best_utility:
                best_utility = u
                best_change = {col_name: val}

import csv
import numpy as np
from sklearn.ensemble import IsolationForest

def spot(header, rows):
    """
    Identifies rows that contain input values that are 'wrong' (anomalous) 
    compared to the rest of the dataset. Uses Isolation Forest to detect 
    outliers in the input space.
    """
    # Identify input columns (exclude goals)
    goal_names = {'MRE-', 'ACC+', 'PRED40+'}
    input_indices = [i for i, name in enumerate(header) if name not in goal_names]
    
    # Prepare data for anomaly detection
    X = []
    for row in rows:
        processed = []
        for i in input_indices:
            val = row[i]
            # Handle '?' as a missing value, treat as a distinct category or mean
            if val == '?':
                processed.append(-1.0)
            elif isinstance(val, str):
                # Simple hash for symbolic columns
                processed.append(float(hash(val) % 1000))
            else:
                processed.append(float(val))
        X.append(processed)
    
    # Isolation Forest is effective for detecting anomalies in high-dimensional data
    # contamination=0.05 assumes roughly 5% of the data might be 'wrong' or extreme
    clf = IsolationForest(contamination=0.05, random_state=42)
    clf.fit(X)
