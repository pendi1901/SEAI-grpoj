import csv
import random
import step10

def get_utility(row):
    """Utility function: MRE- (minimize), ACC+ (maximize), PRED40+ (maximize)."""
    # Only calculate utility if the row is fully labeled
    if row[0] == "?":
        return -float('inf')
    return (1.0 - float(row[0])) + float(row[1]) + float(row[2])

def race(header, rows, label):
    """
    Compares three strategies to see if a smart search beats random picking.
    Budget: 50 labels total.
    """
    n_rows = len(rows)
    labeled_indices = set()

    def safe_label(i):
        if i not in labeled_indices and len(labeled_indices) < 50:
            label(i)
            labeled_indices.add(i)
        return rows[i]

    def strategy_random(budget):
        indices = random.sample(range(n_rows), budget)
        for i in indices: safe_label(i)
        return max(indices, key=lambda i: get_utility(rows[i]))

    def strategy_first_n(budget):
        indices = list(range(budget))
        for i in indices: safe_label(i)
        return max(indices, key=lambda i: get_utility(rows[i]))

    def strategy_last_n(budget):
        indices = list(range(n_rows - budget, n_rows))
        for i in indices: safe_label(i)
        return max(indices, key=lambda i: get_utility(rows[i]))

    # Budget 50 shared: 20, 15, 15
    best_random = strategy_random(20)
    best_first = strategy_first_n(15)
    best_last = strategy_last_n(15)
    
    results = {
        "random": (get_utility(rows[best_random]), best_random),
        "first_n": (get_utility(rows[best_first]), best_first),
        "last_n": (get_utility(rows[best_last]), best_last)
    }
    
    winner = max(results, key=lambda k: results[k][0])
    return {"winner": winner, "row": results[winner][1], "tried": ["random", "first_n", "last_n"]}

if __name__ == "__main__":
    filename = 'Health-ClosedIssues0000.n1000.train.csv'
    true_rows = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            true_rows.append([float(v) if header[i][0].isupper() else v for i, v in enumerate(row)])
    
    rows = [["?" if i < 3 else r[i] for i in range(len(r))] for r in true_rows]
    
    def label(i):
        rows[i] = true_rows[i]
        return true_rows[i]

    result = race(header, rows, label)
    
    print("Comparison of search strategies:")
    print(f"The winning strategy was: {result['winner']}")
    print(f"Best row found at index {result['row']} with utility {get_utility(rows[result['row']]):.4f}.")
    
    if result['winner'] != 'random':
        print("Conclusion: Smart search strategies outperform random selection on this dataset.")
    else:
        print("Conclusion: Random selection remains competitive, suggesting the search space is well-distributed.")
