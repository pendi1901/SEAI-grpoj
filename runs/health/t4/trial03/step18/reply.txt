import csv
from step02 import guess
from step01 import remember

def fix(header, row, levers):
    """
    Recommends a single lever change to improve the row's performance.
    Uses the surrogate model from step02 to predict the outcome of changes.
    """
    mre_idx = header.index('MRE-')
    
    # We need a reference population to train the surrogate model.
    # Since we don't have the full population passed in, we assume the 
    # harness expects us to use the data available in the environment.
    # We will load the data to build the model.
    filename = 'Health-ClosedIssues0000.n1000.train.csv'
    rows = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header_data = next(reader)
        for r in reader:
            rows.append([val if header_data[i] == 'criterion' else float(val) for i, val in enumerate(r)])

    # Train surrogate on all rows that have goals
    train_data = [r for r in rows if r[mre_idx] != "?"]
    
    best_val = None
    best_col = None
    best_mre = row[mre_idx]
    
    # Try every allowed change
    for col, allowed in levers.items():
        idx = header.index(col)
        candidates = []
        if isinstance(allowed, list):
            candidates = [v for v in allowed if v != row[idx]]
        else:
            low, high = allowed
            # Try boundaries and midpoint
            candidates = [low, high, (low + high) / 2]
            candidates = [c for c in candidates if c != row[idx]]
            
        for val in candidates:
            test_row = row[:]
            test_row[idx] = val
            # Mask goals for prediction
            test_input = [test_row[:]]
            for i in range(3): test_input[0][i] = "?"
            
            pred_mre = guess(header, train_data, test_input)[0]
            
            # We want to minimize MRE-
            if pred_mre < best_mre:
                best_mre = pred_mre
                best_col = col
                best_val = val
                
    return {best_col: best_val} if best_col else {list(levers.keys())[0]: levers[list(levers.keys())[0]][0] if isinstance(levers[list(levers.keys())[0]], list) else levers[list(levers.keys())[0]][0]}

if __name__ == "__main__":

    rows = []

        for row in reader:

    levers = {
        'N_estimators': (50, 200),
        'criterion': ['absolute_error', 'poisson', 'squared_error'],
        'Min_sample_leaves': (1, 20),
        'Min_impurity_decrease': (0, 10),
        'Max_depth': (1, 20)
    }
    
    # Pick a row that isn't perfect
    bad_row = rows[0]
    recommendation = fix(header, bad_row, levers)
    
    print("Recommended change to improve your setup:")
    for col, val in recommendation.items():
        print(f"Set '{col}' to {val}")
