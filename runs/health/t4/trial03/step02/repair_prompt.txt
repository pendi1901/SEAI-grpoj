import csv

def remember(header, rows):
    """
    Identifies the best rows based on multi-objective optimization.
    Goals:
    - MRE-: Minimize
    - ACC+: Maximize
    - PRED40+: Maximize
    
    We use a simple Pareto-based approach or a weighted scoring mechanism.
    Given the requirement to return at most 5% (40 rows for 800 rows),
    we calculate a normalized score for each row.
    """
    # Identify indices
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    # Normalize values to [0, 1] to compare different scales
    mre_vals = [r[mre_idx] for r in rows]
    acc_vals = [r[acc_idx] for r in rows]
    pred_vals = [r[pred_idx] for r in rows]

    min_mre, max_mre = min(mre_vals), max(mre_vals)
    min_acc, max_acc = min(acc_vals), max(acc_vals)
    min_pred, max_pred = min(pred_vals), max(pred_vals)

    def get_score(row):
        # Normalize: (val - min) / (max - min)
        # For MRE-, we want low, so we invert: (max - val) / (max - min)
        n_mre = (max_mre - row[mre_idx]) / (max_mre - min_mre) if max_mre != min_mre else 0
        n_acc = (row[acc_idx] - min_acc) / (max_acc - min_acc) if max_acc != min_acc else 0
        n_pred = (row[pred_idx] - min_pred) / (max_pred - min_pred) if max_pred != min_pred else 0
        return n_mre + n_acc + n_pred

    # Calculate scores and keep indices
    scored = []
    for i, row in enumerate(rows):
        scored.append((get_score(row), i))

    # Sort by score descending
    scored.sort(key=lambda x: x[0], reverse=True)

    # Return top 5% (40 rows)
    limit = max(1, int(len(rows) * 0.05))
    return [item[1] for item in scored[:limit]]

def guess(header, train, test):
    """
    Estimates the first goal column (MRE-) for test rows using a 
    RandomForestRegressor.
    """
    # Identify goal and input columns
    goal_idx = 0  # MRE-
    
    # Prepare training data
    X_train = []
    y_train = []
    
    # Encode symbolic columns
    encoders = {}
    symbolic_cols = [i for i, col in enumerate(header) if col == 'criterion']
    
    # Combine train and test to fit encoders
    all_data = train + test
    for col_idx in symbolic_cols:
        le = LabelEncoder()
        le.fit([str(row[col_idx]) for row in all_data])
        encoders[col_idx] = le

    def transform_row(row):
        new_row = []
        for i, val in enumerate(row):
            if i == goal_idx: continue
            if i in symbolic_cols:
                new_row.append(encoders[i].transform([str(val)])[0])
            else:
                new_row.append(float(val))
        return new_row

    X_train = [transform_row(row) for row in train]
    y_train = [row[goal_idx] for row in train]
    
    # Train model
    model = RandomForestRegressor(n_estimators=100, random_state=42)
    model.fit(X_train, y_train)
    
    # Predict
    X_test = [transform_row(row) for row in test]
    return model.predict(X_test).tolist()
