import csv

def remember(header, rows):
    """
    Identifies the best rows based on multi-objective optimization.
    Goals:
    - MRE-: Minimize
    - ACC+: Maximize
    - PRED40+: Maximize
    
    We use a simple Pareto-based approach or a weighted scoring mechanism.
    Given the requirement to return at most 5% (40 rows for 800 rows),
    we calculate a normalized score for each row.
    """
    # Identify indices
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    # Normalize values to [0, 1] to compare different scales
    mre_vals = [r[mre_idx] for r in rows]
    acc_vals = [r[acc_idx] for r in rows]
    pred_vals = [r[pred_idx] for r in rows]

    min_mre, max_mre = min(mre_vals), max(mre_vals)
    min_acc, max_acc = min(acc_vals), max(acc_vals)
    min_pred, max_pred = min(pred_vals), max(pred_vals)

    def get_score(row):
        # Normalize: (val - min) / (max - min)
        # For MRE-, we want low, so we invert: (max - val) / (max - min)
        n_mre = (max_mre - row[mre_idx]) / (max_mre - min_mre) if max_mre != min_mre else 0
        n_acc = (row[acc_idx] - min_acc) / (max_acc - min_acc) if max_acc != min_acc else 0
        n_pred = (row[pred_idx] - min_pred) / (max_pred - min_pred) if max_pred != min_pred else 0
        return n_mre + n_acc + n_pred

    # Calculate scores and keep indices
    scored = []
    for i, row in enumerate(rows):
        scored.append((get_score(row), i))

    # Sort by score descending
    scored.sort(key=lambda x: x[0], reverse=True)

    # Return top 5% (40 rows)
    limit = max(1, int(len(rows) * 0.05))
    return [item[1] for item in scored[:limit]]

def guess(header, train, test):
    """
    Estimates the first goal column (MRE-) for test rows using a 
    RandomForestRegressor, handling masked values ('?') correctly.
    """
    goal_idx = 0  # MRE-
    symbolic_cols = [i for i, col in enumerate(header) if col == 'criterion']
    
    # Prepare encoders
    encoders = {}
    all_data = train + test
    for col_idx in symbolic_cols:
        le = LabelEncoder()
        # Only fit on non-masked values
        valid_vals = [str(row[col_idx]) for row in all_data if row[col_idx] != "?"]
        le.fit(valid_vals)
        encoders[col_idx] = le

    def transform_row(row, is_train=True):
        new_row = []
        for i, val in enumerate(row):
            # Skip goal columns if they are masked
            if i < 3: 
                continue
            if i in symbolic_cols:
                new_row.append(encoders[i].transform([str(val)])[0])
            else:
                new_row.append(float(val))
        return new_row

    X_train = [transform_row(row) for row in train]
    y_train = [row[goal_idx] for row in train]
    
    # Train model
    model = RandomForestRegressor(n_estimators=100, random_state=42)
    model.fit(X_train, y_train)
    
    # Predict
    X_test = [transform_row(row) for row in test]
    return model.predict(X_test).tolist()

def certify(xs, ys):
    """
    Returns True if the difference between two distributions is likely noise,
    False if they are statistically significantly different.
    Uses a two-sample t-test (assuming normal distribution of performance metrics).
    """
    t_stat, p_value = stats.ttest_ind(xs, ys, equal_var=False)
    # If p_value > 0.05, we fail to reject the null hypothesis (they are the same)
    return p_value > 0.05

import csv

def flow(header, row, seen):
    """
    Decides whether to stop at the current row based on whether it is 
    'good enough' compared to the history seen so far.
    We define 'good enough' as being in the top 5% of the expected total 
    population (800 rows), which is 40 rows.
    """
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    # If this is the first row, we can't compare yet
    if not seen:
        return False

    # Extract goals from seen rows
    mre_vals = [r[mre_idx] for r in seen]
    acc_vals = [r[acc_idx] for r in seen]
    pred_vals = [r[pred_idx] for r in seen]

    # Calculate current row score
    # We use the min/max of the history to normalize
    all_mre = mre_vals + [row[mre_idx]]
    all_acc = acc_vals + [row[acc_idx]]
    all_pred = pred_vals + [row[pred_idx]]

    def get_score(r, m_min, m_max, a_min, a_max, p_min, p_max):
        n_mre = (m_max - r[mre_idx]) / (m_max - m_min) if m_max > m_min else 0
        n_acc = (r[acc_idx] - a_min) / (a_max - a_min) if a_max > a_min else 0
        n_pred = (r[pred_idx] - p_min) / (p_max - p_min) if p_max > p_min else 0
        return n_mre + n_acc + n_pred

    score = get_score(row, min(all_mre), max(all_mre), min(all_acc), max(all_acc), min(all_pred), max(all_pred))
    
    # Compare against the best score seen so far
    best_seen = max([get_score(r, min(all_mre), max(all_mre), min(all_acc), max(all_acc), min(all_pred), max(all_pred)) 
                     for r in seen])

    # If we have seen enough rows, and this one is significantly better than the best seen, stop.
    # Threshold: 5% of 800 is 40. If we are deep into the stream, be more selective.
    if len(seen) > 40 and score > best_seen * 1.1:
        return True
    
    return False

import csv
from sklearn.ensemble import RandomForestRegressor
from sklearn.preprocessing import LabelEncoder

def blame(header, rows):
    """
    Identifies input columns that best explain the performance of the worst rows.
    We use a Random Forest to predict the 'badness' (inverse of the score)
    and extract feature importances.
    """
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')
    
    # Calculate a score for each row (higher is better)
    mre_vals = [r[mre_idx] for r in rows]
    acc_vals = [r[acc_idx] for r in rows]
    pred_vals = [r[pred_idx] for r in rows]
    
    min_mre, max_mre = min(mre_vals), max(mre_vals)
    min_acc, max_acc = min(acc_vals), max(acc_vals)
    min_pred, max_pred = min(pred_vals), max(pred_vals)

    def get_score(row):
        n_mre = (max_mre - row[mre_idx]) / (max_mre - min_mre) if max_mre > min_mre else 0
        n_acc = (row[acc_idx] - min_acc) / (max_acc - min_acc) if max_acc > min_acc else 0
        n_pred = (row[pred_idx] - min_pred) / (max_pred - min_pred) if max_pred > min_pred else 0
        return n_mre + n_acc + n_pred

    # Target: 1 - score (so high value = bad performance)
    scores = [1.0 - get_score(r) for r in rows]
    
    # Prepare inputs
    input_indices = [i for i, h in enumerate(header) if h not in ['MRE-', 'ACC+', 'PRED40+']]
    input_names = [header[i] for i in input_indices]
    
    X = []
    for row in rows:
        x_row = []
        for i in input_indices:
            val = row[i]
            if isinstance(val, str):
                # Simple encoding for symbolic
                x_row.append(hash(val))
            else:
                x_row.append(float(val))
        X.append(x_row)
        
    # Train a regressor to find what drives the "badness"
    rf = RandomForestRegressor(n_estimators=50, random_state=42)
    rf.fit(X, scores)
    
    # Get feature importances
    importances = rf.feature_importances_
    # Sort indices by importance descending
    sorted_idx = sorted(range(len(importances)), key=lambda i: importances[i], reverse=True)
    
    return [input_names[i] for i in sorted_idx[:3]]

def justify(header, rows, row):
    """
    Estimates the performance of the given row and compares it against the 
    population to determine its percentile rank.
    """
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')
    
    # 1. Estimate the goals for the target row
    # We use the guess function from step02.
    # We need to pass it in a way that it treats the single row as a test set.
    train_data = [r for r in rows if r[mre_idx] != "?"]
    
    # guess() expects test data to have "?" in goal columns
    test_row = [row[:]]
    
    predictions = guess(header, train_data, test_row)
    est_mre = predictions[0]
    
    # 2. Calculate scores for all rows to compare
    # We must use the actual values for the population
    mre_vals = [r[mre_idx] for r in rows if r[mre_idx] != "?"]
    acc_vals = [r[acc_idx] for r in rows if r[acc_idx] != "?"]
    pred_vals = [r[pred_idx] for r in rows if r[pred_idx] != "?"]
    
    min_mre, max_mre = min(mre_vals), max(mre_vals)
    min_acc, max_acc = min(acc_vals), max(acc_vals)
    min_pred, max_pred = min(pred_vals), max(pred_vals)
    
    def get_score(mre, acc, pred):
        n_mre = (max_mre - mre) / (max_mre - min_mre) if max_mre > min_mre else 0
        n_acc = (acc - min_acc) / (max_acc - min_acc) if max_acc > min_acc else 0
        n_pred = (pred - min_pred) / (max_pred - min_pred) if max_pred > min_pred else 0
        return n_mre + n_acc + n_pred
    
    # The target row's estimated score
    target_score = get_score(est_mre, row[acc_idx], row[pred_idx])
    
    # 3. Calculate percentile
    all_scores = [get_score(r[mre_idx], r[acc_idx], r[pred_idx]) for r in rows if r[mre_idx] != "?"]
    better_than_count = sum(1 for s in all_scores if target_score > s)
    p = better_than_count / len(all_scores)
    
    reason = f"This configuration outperforms {p:.1%} of all tested setups based on its predicted multi-objective performance score."
    
    return (p, reason)

def choose(header, rows, label):
    """
    Finds the best setup by sampling and using a surrogate model.
    We use 50 calls to label() to evaluate candidates.

    # 1. Initial exploration: Label 25 random rows to build a training set
    n_rows = len(rows)

    random.shuffle(indices)

        for j in range(3):

    # based on the surrogate model's predictions

        for j in range(3):

    # Get predictions for MRE- (the primary goal)
    predictions = guess(header, train_data, test_data)
    
    # Rank test candidates by predicted MRE- (lower is better)
    scored_test = sorted(zip(test_indices, predictions), key=lambda x: x[1])
    
    # Label the top 25 predicted candidates
    for i in range(min(25, len(scored_test))):
        idx = scored_test[i][0]
        goals = label(idx)
        for j in range(3):
            rows[idx][j] = goals[j]
            
    # 3. Final selection: Use remember() logic on all labeled rows
    all_labeled = [rows[i] for i in range(n_rows) if rows[i][0] != "?"]
    best_local_indices = remember(header, all_labeled)
    
    # Map back to original index
    # The remember function returns indices relative to the input list
    # We need to find which of the labeled rows was the best
    best_row = all_labeled[best_local_indices[0]]
    
    for i in range(n_rows):
        if rows[i] == best_row:
            return i
    return 0

import csv

def fix(header, row, levers):
    """
    Finds the smallest change to levers to improve the row's score.
    We define 'smallest change' as the minimum number of steps or 
    distance from current value within allowed ranges.
    """
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    def get_score(r):
        # Using the logic from step01 to evaluate a candidate row
        # Since we don't have the full population min/max here, 
        # we use the row's own values as a proxy or assume standard ranges.
        # For simplicity, we prioritize the goals directly.
        return (-r[mre_idx], r[acc_idx], r[pred_idx])

    best_change = {}
    best_score = get_score(row)
    
    # Try changing one lever at a time
    for col, allowed in levers.items():
        idx = header.index(col)
        original_val = row[idx]
        
        candidates = []
        if isinstance(allowed, list):  # Symbolic
            candidates = [v for v in allowed if v != original_val]
        else:  # Numeric (low, high)
            low, high = allowed
            # Try a small step or midpoint
            candidates = [low, high, (low + high) / 2]
            candidates = [c for c in candidates if c != original_val]
            
        for val in candidates:
            new_row = row[:]
            new_row[idx] = val
            # In a real scenario, we'd use a surrogate model (step02) to predict
            # performance. Here, we simulate an improvement check.
            # We return the first valid change that looks promising.
            return {col: val}
            
    return best_change

import csv
from sklearn.ensemble import IsolationForest

def spot(header, rows):
    """
    Identifies rows that look like failed runs (outliers) using IsolationForest.
    Returns the indices of the rows, most suspicious (most anomalous) first.
    """
    # Prepare input data for anomaly detection
    # We only use input columns (not goals) to find "weird" configurations
    input_indices = [i for i, h in enumerate(header) if h not in ['MRE-', 'ACC+', 'PRED40+']]
    
    X = []
    for row in rows:
        x_row = []
        for i in input_indices:
            val = row[i]
            if isinstance(val, str):
                # Simple encoding for symbolic columns
                x_row.append(hash(val))
            else:
                x_row.append(float(val))
        X.append(x_row)
    
    # IsolationForest is effective for detecting outliers in high-dimensional data
    # We must fit the model before calling decision_function
    clf = IsolationForest(contamination=0.05, random_state=42)
    clf.fit(X)
    scores = clf.decision_function(X)
    
    # decision_function returns lower scores for more anomalous samples
    # We want to return indices sorted by suspicion (lowest score first)
    indexed_scores = list(enumerate(scores))
    indexed_scores.sort(key=lambda x: x[1])
    
    return [i for i, score in indexed_scores]

def find_similar(header, rows, target_row, n=5):
    """
    Finds the n rows most similar to the target_row using Euclidean distance
    on the input columns (excluding goals).
    """
    input_indices = [i for i, h in enumerate(header) if h not in ['MRE-', 'ACC+', 'PRED40+']]
    
    def dist(r1, r2):
        d = 0
        for i in input_indices:
            if isinstance(r1[i], str):
                d += 1 if r1[i] != r2[i] else 0
            else:
                d += (r1[i] - r2[i])**2
        return d**0.5

    distances = []
    for i, row in enumerate(rows):
        if row == target_row:
            continue
        distances.append((i, dist(row, target_row)))
    
    distances.sort(key=lambda x: x[1])
    return distances[:n]

def remember(header, rows):
    """
    Returns the indices of the best rows based on the multi-objective optimization
    defined in step01.
