import csv
import random
from step01 import remember
from step07 import choose

def race(header, rows, label):
    """
    Compares three strategies: random, greedy (best of sample), and 
    a hybrid approach. Budget is 50 labels total.
    """
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    def get_utility(row):
        return -row[mre_idx] + row[acc_idx] + row[pred_idx]

    # Strategy 1: Random
    def strategy_random(budget):
        indices = random.sample(range(len(rows)), budget)
        best_u = -float('inf')
        best_i = -1
        for i in indices:
            row = label(i)
            u = get_utility(row)
            if u > best_u:
                best_u, best_i = u, i
        return best_i, best_u

    # Strategy 2: Greedy (Sample 50, pick best)
    def strategy_greedy(budget):
        # Using the logic from step07
        best_i = choose(header, rows, label)
        return best_i, get_utility(label(best_i))

    # Strategy 3: First-improvement (Stop at first "good" row)
    def strategy_first(budget):
        best_u = -float('inf')
        best_i = -1
        for i in range(budget):
            row = label(i)
            u = get_utility(row)
            if u > best_u:
                best_u, best_i = u, i
            if u > 0.5: # Heuristic threshold
                break
        return best_i, best_u

    # Allocate budget: 16, 17, 17 labels
    res_random = strategy_random(16)
    res_greedy = strategy_greedy(17)
    res_first = strategy_first(17)

    results = {
        "random": res_random,
        "greedy": res_greedy,
        "first": res_first
    }

    winner = max(results, key=lambda k: results[k][1])
    
    return {
        "winner": winner,
        "row": results[winner][0],
        "tried": ["random", "greedy", "first"]
    }

if __name__ == "__main__":
    filename = 'Health-ClosedIssues0000.n1000.train.csv'
    rows = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = [float(val) if header[i] not in ['criterion'] else val 
                             for i, val in enumerate(row)]
            rows.append(processed_row)

    def label(i):
        return rows[i]

    result = race(header, rows, label)
    print(f"Race Results: The winner is '{result['winner']}' strategy.")
    print(f"Best row found at index {result['row']}.")
    print(f"Strategies compared: {', '.join(result['tried'])}.")
