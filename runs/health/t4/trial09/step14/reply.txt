import csv

def flow(header, row, seen):
    """
    Decides whether to stop the process by checking if the current row 
    is significantly better than all previous rows.
    
    We define 'improvement' as a new maximum utility. If the current row
    is the best seen so far, we continue. If we have seen a large number of 
    rows without a new best, we stop.
    """
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    def get_utility(r):
        return -r[mre_idx] + r[acc_idx] + r[pred_idx]

    current_utility = get_utility(row)

    if not seen:
        return False

    # Find the best utility seen so far
    best_utility = max(get_utility(s) for s in seen)

    # If current row is a new best, keep going (return False)
    if current_utility > best_utility:
        return False

    # If we have seen at least 100 rows and haven't improved in the last 50,
    # we assume the best has been found and stop (return True).
    if len(seen) > 100:
        recent_rows = seen[-50:]
        if all(get_utility(s) < current_utility for s in recent_rows):
            return True

    return False

if __name__ == "__main__":
    filename = 'Health-ClosedIssues0000.n1000.train.csv'
    rows = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = []
            for i, val in enumerate(row):
                if header[i] in ['MRE-', 'ACC+', 'PRED40+', 'N_estimators', 
                                 'Min_sample_leaves', 'Min_impurity_decrease', 'Max_depth']:
                    processed_row.append(float(val))
                else:
                    processed_row.append(val)
            rows.append(processed_row)

    seen = []
    stopped = False
    for i, row in enumerate(rows):
        if flow(header, row, seen):
            print(f"Optimization stopped at row {i+1}.")
            print(f"The best configuration found has utility: {-row[0] + row[1] + row[2]:.4f}")
            stopped = True
            break
        seen.append(row)
    
    if not stopped:
        print("Process completed without early stopping.")
