```python
import csv
from sklearn.neighbors import KNeighborsRegressor
from sklearn.preprocessing import LabelEncoder
import numpy as np

def guess(header, train, test):
    """
    Estimates the first goal column (MRE-) for test rows using a K-Nearest Neighbors
    regressor based on the input features.
    """
    # Identify indices
    goal_idx = 0  # MRE-
    
    # Identify input columns (all except goals)
    # Goals end in '-' or '+'
    input_indices = [i for i, col in enumerate(header) if not (col.endswith('-') or col.endswith('+'))]
    
    def prepare_data(data_rows):
        X = []
        for row in data_rows:
            features = []
            for i in input_indices:
                val = row[i]
                # Simple encoding for symbolic columns (criterion)
                if isinstance(val, str):
                    # Map to numeric
                    val = hash(val) % 100
                features.append(float(val))
            X.append(features)
        return np.array(X)

    X_train = prepare_data(train)
    y_train = np.array([row[goal_idx] for row in train])
    X_test = prepare_data(test)
    
    # Use KNN to estimate the MRE- value
    model = KNeighborsRegressor(n_neighbors=5)
    model.fit(X_train, y_train)
    
    return model.predict(X_test).tolist()

if __name__ == "__main__":
    filename = 'Health-ClosedIssues0000.n1000.train.csv'
    rows = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = []
            for i, val in enumerate(row):
                # Convert numeric columns to floats, keep symbolic as strings
                if header[i] not in ['criterion']:
                    processed_row.append(float(val))
                else:
                    processed_row.append(val)
            rows.append(processed_row)

    # Split into train and a dummy test set (last 5 rows) to demonstrate
    train_data = rows[:-5]
    test_data = [[r[i] if i in [3, 4, 5, 6, 7] else "?" for i in range(len(header))] for r in rows[-5:]]
    
    predictions = guess(header, train_data, test_data)
    
    print("Forecast estimates for the first goal (MRE-):")
    for i, pred in enumerate(predictions):
        print(f"Test row {i+1}: Estimated MRE = {pred:.4f}")
```
