import csv
import random

def choose(header, rows, label):
    """
    Finds a setup that is both accurate (high utility) and small (low N_estimators).
    We use a multi-objective approach: maximize utility and minimize N_estimators.
    Budget: 50 labels.
    """
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')
    n_est_idx = header.index('N_estimators')

    # Sample 50 random rows to evaluate
    indices = random.sample(range(len(rows)), min(50, len(rows)))
    
    best_score = -float('inf')
    best_idx = -1
    
    # Collect data for normalization
    sampled_data = []
    for i in indices:
        sampled_data.append((i, label(i)))
    
    n_ests = [data[1][n_est_idx] for data in sampled_data]
    min_n, max_n = min(n_ests), max(n_ests)
    
    for i, row in sampled_data:
        mre, acc, pred, n_est = row[mre_idx], row[acc_idx], row[pred_idx], row[n_est_idx]
        
        # Utility: -MRE + ACC + PRED40
        utility = -mre + acc + pred
        
        # Cost: Normalized N_estimators (lower is better)
        cost = (n_est - min_n) / (max_n - min_n + 1e-9)
        
        # Combined score: Balance accuracy and small size
        # We weigh utility heavily but penalize large N_estimators
        score = utility - (0.2 * cost)
        
        if score > best_score:
            best_score = score
            best_idx = i
            
    return best_idx

if __name__ == "__main__":
    filename = 'Health-ClosedIssues0000.n1000.train.csv'
    rows = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = []
            for i, val in enumerate(row):
                if header[i] in ['MRE-', 'ACC+', 'PRED40+', 'N_estimators', 
                                 'Min_sample_leaves', 'Min_impurity_decrease', 'Max_depth']:
                    processed_row.append(float(val))
                else:
                    processed_row.append(val)
            rows.append(processed_row)

    def label(i):
        return rows[i]

    # Create hidden rows
    hidden_rows = []
    for r in rows:
        hidden_rows.append([r[i] if not (header[i].endswith('-') or header[i].endswith('+')) else "?" for i in range(len(r))])

    best_idx = choose(header, hidden_rows, label)
    
    print(f"Optimization complete. Best setup found at index {best_idx}.")
    print(f"MRE: {rows[best_idx][0]:.4f}, ACC: {rows[best_idx][1]:.4f}, PRED40: {rows[best_idx][2]:.4f}")
    print(f"N_estimators: {rows[best_idx][3]}")
    print("This setup balances high predictive accuracy with a smaller footprint for nightly runs.")
