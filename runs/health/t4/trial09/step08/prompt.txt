import csv

def remember(header, rows):
    """
    Identifies the best rows based on multi-objective optimization.
    Goals: MRE- (minimize), ACC+ (maximize), PRED40+ (maximize).
    We use a simple Pareto-based approach or weighted normalization.
    Given the constraints, we normalize the goals and pick the top 5%.
    """
    # Identify column indices
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    # Extract goals and normalize
    # MRE is minimized (so we negate it to maximize), others are maximized
    scores = []
    for i, row in enumerate(rows):
        mre = row[mre_idx]
        acc = row[acc_idx]
        pred = row[pred_idx]
        
        # Simple utility: -MRE + ACC + PRED40
        # Since they are on different scales, we could normalize, 
        # but for a simple heuristic, this weighted sum works well.
        utility = -mre + acc + pred
        scores.append((utility, i))

    # Sort by utility descending
    scores.sort(key=lambda x: x[0], reverse=True)
    
    # Return top 5% (800 rows * 0.05 = 40 rows)
    limit = max(1, int(len(rows) * 0.05))
    return [item[1] for item in scores[:limit]]

def guess(header, train, test):
    """
    Estimates the first goal column (MRE-) for test rows using a K-Nearest Neighbors
    regressor based on the input features.
    """
    # Identify indices
    goal_idx = 0  # MRE-
    
    # Identify input columns (all except goals)
    # Goals end in '-' or '+'
    input_indices = [i for i, col in enumerate(header) if not (col.endswith('-') or col.endswith('+'))]
    
    def prepare_data(data_rows):
        X = []
        for row in data_rows:
            features = []
            for i in input_indices:
                val = row[i]
                # Simple encoding for symbolic columns (criterion)
                if isinstance(val, str):
                    # Map to numeric
                    val = hash(val) % 100
                features.append(float(val))
            X.append(features)
        return np.array(X)

    X_train = prepare_data(train)
    y_train = np.array([row[goal_idx] for row in train])
    X_test = prepare_data(test)
    
    # Use KNN to estimate the MRE- value
    model = KNeighborsRegressor(n_neighbors=5)
    model.fit(X_train, y_train)
    
    return model.predict(X_test).tolist()

def certify(xs, ys):
    """
    Returns True if the difference between two distributions is likely noise,
    False if they are statistically different (using Welch's t-test).
    """
    # Use Welch's t-test to compare two independent samples
    t_stat, p_value = stats.ttest_ind(xs, ys, equal_var=False)
    # If p-value > 0.05, we fail to reject the null hypothesis (it's noise)
    return p_value > 0.05

def analyze_correlation(header, rows):
    """
    Checks if 'N_estimators' (a proxy for setup size) correlates with 'ACC+'
    to determine if larger setups are genuinely more accurate.
    """
    n_est_idx = header.index('N_estimators')
    acc_idx = header.index('ACC+')
    
    n_est = [row[n_est_idx] for row in rows]
    acc = [row[acc_idx] for row in rows]
    
    # Split into two groups: small (below median) and large (above median)
    median_n = np.median(n_est)
    small_acc = [acc[i] for i in range(len(n_est)) if n_est[i] <= median_n]
    large_acc = [acc[i] for i in range(len(n_est)) if n_est[i] > median_n]
    
    is_noise = certify(small_acc, large_acc)
    
    return is_noise, np.mean(small_acc), np.mean(large_acc)

import csv

def flow(header, row, seen):
    """
    Decides whether to accept the current row based on a simple threshold 
    heuristic derived from the distribution of previous rows.
    """
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    # Calculate utility for the current row
    # Goals: MRE- (minimize), ACC+ (maximize), PRED40+ (maximize)
    current_utility = -row[mre_idx] + row[acc_idx] + row[pred_idx]

    if not seen:
        return False

    # Calculate utilities of seen rows
    seen_utilities = []
    for s in seen:
        seen_utilities.append(-s[mre_idx] + s[acc_idx] + s[pred_idx])

    # If current row is better than the 95th percentile of seen rows, accept it
    # This acts as an "early stopping" criterion for a "good enough" setup
    threshold = sorted(seen_utilities)[int(len(seen_utilities) * 0.95)]
    
    return current_utility >= threshold

import csv

def blame(header, rows):
    """
    Identifies input columns that contribute most to poor performance.
    We define 'bad' rows as those in the bottom 25% of utility.
    We then train a Random Forest to predict the utility and extract feature importances.
    """
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')
    
    # Calculate utility for all rows
    utilities = []
    for row in rows:
        utilities.append(-row[mre_idx] + row[acc_idx] + row[pred_idx])
    
    # Identify input columns
    input_indices = [i for i, col in enumerate(header) if not (col.endswith('-') or col.endswith('+'))]
    input_names = [header[i] for i in input_indices]
    
    # Prepare features
    X = []
    for row in rows:
        features = []
        for i in input_indices:
            val = row[i]
            if isinstance(val, str):
                # Simple encoding for symbolic columns
                val = hash(val) % 100
            features.append(float(val))
        X.append(features)
    
    # Train a regressor to predict utility
    # Features that have high importance in predicting low utility are the culprits
    model = RandomForestRegressor(n_estimators=50, random_state=42)
    model.fit(X, utilities)
    
    # Get feature importances
    importances = model.feature_importances_
    
    # Sort features by importance
    feature_importance_pairs = sorted(zip(input_names, importances), key=lambda x: x[1], reverse=True)
    
    # Return the top 3 most influential input columns
    return [pair[0] for pair in feature_importance_pairs[:3]]

def justify(header, rows, row):
    """
    Estimates the performance of the given row and compares it to the 
    distribution of all other rows to justify its quality.
    """
    # 1. Estimate the goal values for the input row
    # We use the guess function from step02 which uses KNN
    # We need to pass a list containing the row as a test set
    estimated_mre = guess(header, rows, [row])[0]
    
    # For ACC+ and PRED40+, we can use the same KNN logic or simple averages
    # For simplicity and consistency, we reuse the guess logic
    # We create temporary headers to trick guess() into predicting other goals
    def get_pred(goal_idx):
        # Temporarily swap goal to index 0 for the guess function
        h = [header[goal_idx]] + [h for i, h in enumerate(header) if i != goal_idx]
        r_train = [[r[goal_idx]] + [r[i] for i in range(len(header)) if i != goal_idx] for r in rows]
        r_test = [[0.0] + [row[i] for i in range(len(header)) if i != goal_idx]]
        return guess(h, r_train, r_test)[0]

    est_acc = get_pred(header.index('ACC+'))
    est_pred = get_pred(header.index('PRED40+'))
    
    # 2. Calculate utility for all rows
    mre_idx, acc_idx, pred_idx = header.index('MRE-'), header.index('ACC+'), header.index('PRED40+')
    
    def calc_utility(r):
        return -r[mre_idx] + r[acc_idx] + r[pred_idx]
    
    utilities = [calc_utility(r) for r in rows]
    row_utility = -estimated_mre + est_acc + est_pred
    
    # 3. Calculate fraction of rows better than this one
    better_count = sum(1 for u in utilities if u > row_utility)
    p = 1.0 - (better_count / len(rows))
    
    reason = (f"This setup is estimated to outperform {p:.1%} of all possible configurations "
              "based on its predicted balance of error reduction and accuracy.")
    
    return p, reason

def choose(header, rows, label):
    """
    Finds the best setup by sampling 50 rows, labeling them to get their 
    true goal values, and returning the index of the best one.
    """
    # Identify indices
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    # We have a budget of 50 labels.
    # We will sample 50 random indices from the available rows.
    indices = list(range(len(rows)))
    sample_size = min(50, len(rows))
    sampled_indices = random.sample(indices, sample_size)

    best_utility = -float('inf')
    best_idx = -1

    for i in sampled_indices:
        # Get true goal values
        goals = label(i)
        mre, acc, pred = goals[mre_idx], goals[acc_idx], goals[pred_idx]
        
        # Calculate utility: -MRE + ACC + PRED40
        utility = -mre + acc + pred
        
        if utility > best_utility:
            best_utility = utility
            best_idx = i

    return best_idx
