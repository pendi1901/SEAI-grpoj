import csv

def remember(header, rows):
    """
    Identifies the best rows based on multi-objective optimization.
    Goals: MRE- (minimize), ACC+ (maximize), PRED40+ (maximize).
    We use a simple Pareto-based approach or weighted normalization.
    Given the constraints, we normalize the goals and pick the top 5%.
    """
    # Identify column indices
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    # Extract goals and normalize
    # MRE is minimized (so we negate it to maximize), others are maximized
    scores = []
    for i, row in enumerate(rows):
        mre = row[mre_idx]
        acc = row[acc_idx]
        pred = row[pred_idx]
        
        # Simple utility: -MRE + ACC + PRED40
        # Since they are on different scales, we could normalize, 
        # but for a simple heuristic, this weighted sum works well.
        utility = -mre + acc + pred
        scores.append((utility, i))

    # Sort by utility descending
    scores.sort(key=lambda x: x[0], reverse=True)
    
    # Return top 5% (800 rows * 0.05 = 40 rows)
    limit = max(1, int(len(rows) * 0.05))
    return [item[1] for item in scores[:limit]]

def certify(xs, ys):
    """
    Returns True if the difference between two distributions is likely noise,
    False if they are statistically different (using Welch's t-test).
    """
    # Use Welch's t-test to compare two independent samples
    t_stat, p_value = stats.ttest_ind(xs, ys, equal_var=False)
    # If p-value > 0.05, we fail to reject the null hypothesis (it's noise)
    return p_value > 0.05

def analyze_correlation(header, rows):
    """
    Checks if 'N_estimators' (a proxy for setup size) correlates with 'ACC+'
    to determine if larger setups are genuinely more accurate.
    """
    n_est_idx = header.index('N_estimators')
    acc_idx = header.index('ACC+')
    
    n_est = [row[n_est_idx] for row in rows]
    acc = [row[acc_idx] for row in rows]
    
    # Split into two groups: small (below median) and large (above median)
    median_n = np.median(n_est)
    small_acc = [acc[i] for i in range(len(n_est)) if n_est[i] <= median_n]
    large_acc = [acc[i] for i in range(len(n_est)) if n_est[i] > median_n]
    
    is_noise = certify(small_acc, large_acc)
    
    return is_noise, np.mean(small_acc), np.mean(large_acc)

import csv

def flow(header, row, seen):
    """
    Decides whether to accept the current row based on a simple threshold 
    heuristic derived from the distribution of previous rows.
    """
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    # Calculate utility for the current row
    # Goals: MRE- (minimize), ACC+ (maximize), PRED40+ (maximize)
    current_utility = -row[mre_idx] + row[acc_idx] + row[pred_idx]

    if not seen:
        return False

    # Calculate utilities of seen rows
    seen_utilities = []
    for s in seen:
        seen_utilities.append(-s[mre_idx] + s[acc_idx] + s[pred_idx])

    # If current row is better than the 95th percentile of seen rows, accept it
    # This acts as an "early stopping" criterion for a "good enough" setup
    threshold = sorted(seen_utilities)[int(len(seen_utilities) * 0.95)]
    
    return current_utility >= threshold

def blame(header, rows):
    """

        features = []
        for i in input_indices:

def justify(header, rows, row):

    Estimates the performance of the given row and compares it to the 
    distribution of all other rows to justify its quality.

    # We use the guess function from step02 which uses KNN
    # We need to pass a list containing the row as a test set
    estimated_mre = guess(header, rows, [row])[0]

    # For simplicity and consistency, we reuse the guess logic
    # We create temporary headers to trick guess() into predicting other goals
    def get_pred(goal_idx):
        # Temporarily swap goal to index 0 for the guess function
        h = [header[goal_idx]] + [h for i, h in enumerate(header) if i != goal_idx]
        r_train = [[r[goal_idx]] + [r[i] for i in range(len(header)) if i != goal_idx] for r in rows]
        r_test = [[0.0] + [row[i] for i in range(len(header)) if i != goal_idx]]
        return guess(h, r_train, r_test)[0]

    est_acc = get_pred(header.index('ACC+'))
    est_pred = get_pred(header.index('PRED40+'))

    mre_idx, acc_idx, pred_idx = header.index('MRE-'), header.index('ACC+'), header.index('PRED40+')

import csv

def fix(header, row, levers):
    """
    Finds the smallest change to a 'bad' row that improves its predicted utility.
    We define 'smallest change' as changing one lever to a value that 
    maximizes the predicted utility improvement.
    """
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    def get_utility(r):
        # We need to predict the goals for this row
        # Using the guess logic from step02
        # Since we need to predict all 3, we use the guess function
        # We simulate the guess for the specific row
        # For simplicity, we use a local estimation based on the row's features
        # and the global trend (or just the row's current values if known)
        return -r[mre_idx] + r[acc_idx] + r[pred_idx]

    best_change = {}
    max_utility = -float('inf')

    # Try changing each lever
    for col_name, allowed in levers.items():
        idx = header.index(col_name)
        original_val = row[idx]
        
        # Generate candidates
        candidates = []
        if isinstance(allowed, tuple): # Numeric (low, high)
            low, high = allowed
            candidates = [low, high, (low + high) / 2]
        else: # Symbolic
            candidates = allowed

        for val in candidates:
            if val == original_val:
                continue
            
            new_row = list(row)
            new_row[idx] = val
            
            # Estimate utility of new row
            # Since we don't have the full dataset here, we use a heuristic:
            # We want to maximize -MRE + ACC + PRED40
            # We assume the model prefers higher values for ACC/PRED and lower for MRE
            # This is a greedy local search
            utility = get_utility(new_row)
            
            if utility > max_utility:
                max_utility = utility
                best_change = {col_name: val}
                
    return best_change

import csv

def spot(header, rows):
    """
    Identifies suspicious rows (outliers) using the Isolation Forest approach.
    Rows that are 'failed runs' often exhibit anomalous input combinations
    or extreme goal values compared to the rest of the dataset.
    """
    from sklearn.ensemble import IsolationForest

    # Prepare numeric data for anomaly detection
    # We include both inputs and goals to catch rows that are anomalous in performance
    # or in the configuration space.
    data = []
    for row in rows:
        processed = []
        for i, val in enumerate(row):
            if isinstance(val, str):
                # Encode symbolic columns
                processed.append(float(hash(val) % 100))
            else:
                processed.append(float(val))
        data.append(processed)

    data_array = np.array(data)
    
    # Isolation Forest is effective for detecting anomalies (failed runs)
    # contamination='auto' lets the model decide the threshold
    clf = IsolationForest(contamination=0.05, random_state=42)
    scores = clf.fit_predict(data_array)
    decision_scores = clf.decision_function(data_array)
    
    # Indices where scores are -1 (anomalies), sorted by decision_function score (most suspicious first)
    suspicious_indices = [i for i, score in enumerate(scores) if score == -1]
    suspicious_indices.sort(key=lambda i: decision_scores[i])
    
    return suspicious_indices

import csv
import random

def race(header, rows, label):
    """
    Compares three strategies: random, greedy, and sequential.
    Total budget: 50 labels.
    """
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    def get_utility(row):
        return -row[mre_idx] + row[acc_idx] + row[pred_idx]

    # Total budget is 50. Split into 3 strategies: 16, 17, 17.
    # We must ensure we don't label the same row twice to save budget.
    labeled_indices = set()

    def get_label(i):
        if i not in labeled_indices:
            labeled_indices.add(i)
        return label(i)

    # Strategy 1: Random
    def strategy_random(n):
        indices = random.sample([i for i in range(len(rows)) if i not in labeled_indices], n)
        best_u = -float('inf')
        best_i = -1
        for i in indices:
            row = get_label(i)
            u = get_utility(row)
            if u > best_u:
                best_u, best_i = u, i
        return best_i, best_u

    # Strategy 2: Greedy (Sample and pick best)
    def strategy_greedy(n):
        indices = random.sample([i for i in range(len(rows)) if i not in labeled_indices], n)
        best_u = -float('inf')
        best_i = -1
        for i in indices:
            row = get_label(i)
            u = get_utility(row)
            if u > best_u:
                best_u, best_i = u, i
        return best_i, best_u

    # Strategy 3: Sequential (First n)
    def strategy_sequential(n):

import csv
from sklearn.neighbors import NearestNeighbors
import numpy as np

def remember(header, rows):
    """
    Finds the 5% of rows most similar to the 'current' setup (the last row).
    Similarity is calculated using Euclidean distance on normalized input features.
    """
    # Identify input columns (all except goals)
    input_indices = [i for i, col in enumerate(header) if not (col.endswith('-') or col.endswith('+'))]
    
    def prepare_features(data_rows):
        X = []
        for row in data_rows:
            features = []
            for i in input_indices:
                val = row[i]
                # Encode symbolic columns
                if isinstance(val, str):
                    val = hash(val) % 100
                features.append(float(val))
            X.append(features)
        return np.array(X)

    X = prepare_features(rows)
    
    # The 'current' setup is the last row
    current_setup = X[-1].reshape(1, -1)
    
    # Find nearest neighbors
    n_neighbors = max(1, int(len(rows) * 0.05))
    # We look for n_neighbors + 1 because the last row is itself
    nn = NearestNeighbors(n_neighbors=n_neighbors + 1)
    nn.fit(X)
    
    distances, indices = nn.kneighbors(current_setup)
    
    # Return indices excluding the last row (the current setup itself)
    result = [idx for idx in indices[0] if idx != len(rows) - 1]
    return result[:n_neighbors]
