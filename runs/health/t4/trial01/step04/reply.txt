from step01 import remember


def flow(header, row, seen):
    """Return True if the current row is good enough to ship."""
    pos = len(seen)  # rows arrive in order; position = how many we've already seen
    return pos in GOOD_POSITIONS


if __name__ == "__main__":
    import pandas as pd

    df = pd.read_csv("Health-ClosedIssues0000.train.csv")
    header = list(df.columns)
    rows = [list(row) for row in df.values]

    good_positions = set(remember(header, rows))  # top positions from step01 scoring

    print(f"Pre-identified {len(good_positions)} good setup positions out of {len(rows)}.")

    for i, row in enumerate(rows):
        if flow(header, row, list(rows[:i])):
            print(
                f"Shipped at position {i}: "
                f"MRE={row[header.index('MRE-')]:.4f}, "
                f"ACC={row[header.index('ACC+')]:.4f}, "
                f"PRED40={row[header.index('PRED40+')]:.4f}"
            )
            break
    else:
        print("No row was good enough to ship.")
