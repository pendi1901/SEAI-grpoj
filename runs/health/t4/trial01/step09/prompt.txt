def certify(xs, ys):
    """Return True if xs and ys are the same apart from noise."""
    n = min(len(xs), len(ys))
    if n == 0:
        return False

    diffs = [abs(xs[i] - ys[i]) for i in range(n)]
    max_diff = max(diffs)

    # Scale-dependent tolerance with absolute floor
    avg_val = (sum(xs[:n]) + sum(ys[:n])) / (2 * n)
    if abs(avg_val) < 1e-6:
        return max_diff < 0.05
    rel_tol = 0.05 * abs(avg_val)
    return max_diff < rel_tol

def blame(header, rows):
    goals = [h for h in header if h.endswith('-') or h.endswith('+')]
    n = len(rows)

    # Score every row with the same normalisation as remember().
    scores = np.zeros(n)
    for col in goals:
        vals = []
        for row in rows:
            v = float(row[header.index(col)])
            vals.append(v)
        arr = np.array(vals)
        if arr.max() == arr.min():
            scores += 0.5
        else:
            normed = (arr - arr.min()) / (arr.max() - arr.min())
            if col.endswith('-'):
                scores += (1.0 - normed)
            else:
                scores += normed

    sorted_indices = np.argsort(scores)[::-1]
    top_k = max(1, int(n * 0.05))
    good_set = set(sorted_indices[:top_k].tolist())
    bad_set = set(sorted_indices[-top_k:].tolist())

    input_cols = [c for c in header if not c.endswith('-') and not c.endswith('+')]

    blame_vals = []
    for col in input_cols:
        good_vals = []
        bad_vals = []
        for i, row in enumerate(rows):
            v = float(row[header.index(col)])
            if i in good_set:
                good_vals.append(v)
            elif i in bad_set:
                bad_vals.append(v)

        if not good_vals or not bad_vals:
            blame_vals.append((0.0, col))
            continue

        g_mean = np.mean(good_vals)
        b_mean = np.mean(bad_vals)
        diff = abs(g_mean - b_mean)

        all_vals = [float(row[header.index(col)]) for row in rows]
        arr_all = np.array(all_vals)
        col_range = float(arr_all.max() - arr_all.min()) if len(all_vals) > 1 else 1.0
        norm_diff = diff / col_range if col_range > 0 else 0.0

        blame_vals.append((norm_diff, col))

    blame_vals.sort(key=lambda x: -x[0])
    return [col for _, col in blame_vals[:3]]

def justify(header, rows, row):
    """Returns (p, reason) where p is fraction of rows that row beats."""
    
    goals = [h for h in header if h.endswith('-') or h.endswith('+')]
    numeric_cols = [c for c in header if c.isupper()]
    symbolic_cols = [c for c in header if not c.isupper() and not c.endswith('-') and not c.endswith('+')]

    def encode_row(row):
        return (tuple(float(row[header.index(c)]) for c in numeric_cols),
                tuple(row[header.index(c)] for c in symbolic_cols))

    # Predict goals for the test row using nearest neighbor on non-goal features.
    train_enc = [encode_row(r) for r in rows if not any(v == '?' for v in r)]
    test_enc = encode_row(row)

    best_match_idx = None
    best_dist = float('inf')

    for i, enc in enumerate(train_enc):
        dist = 0.0
        for c in header:
            if c.endswith('-') or c.endswith('+'):
                continue
            tv = test_enc[header.index(c)]
            trv = enc[header.index(c)]
            if isinstance(tv, str) and isinstance(trv, str):
                dist += 0.0 if tv == trv else 1.0
            elif isinstance(tv, float) and isinstance(trv, float):
                dist += abs(tv - trv) / 100.0

        if dist < best_dist:
            best_dist = dist
            best_match_idx = i

    predicted_goals = []
    for col in goals:
        v = float(train_enc[best_match_idx][header.index(col)])
        predicted_goals.append(v)

    n = len(rows)
    better_count = 0

    for i, r in enumerate(rows):
        if any(v == '?' for v in r):
            continue

        all_better = True
        for col in goals:
            actual = float(r[header.index(col)])
            pred = predicted_goals[goals.index(col)]
            if col.endswith('-'):
                # Lower is better (minimize)
                if pred <= actual:
                    continue
                else:
                    all_better = False
            else:
                # Higher is better (maximize)
                if pred >= actual:
                    continue
                else:
                    all_better = False

        if all_better:
            better_count += 1

    p = better_count / n
    reason = f"This setup beats {better_count} out of {n} other setups."

    return (p, reason)

    # Score each row by normalising its goals (same scheme as remember).

        arr = np.array([v[goals.index(col)] for v in goal_vals])

    best_score = -1e9
