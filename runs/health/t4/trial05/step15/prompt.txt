import csv

def remember(header, rows):
    """
    Identifies the best rows based on multi-objective optimization.
    Goals: Minimize MRE-, Maximize ACC+, Maximize PRED40+.
    Uses a simple Pareto-dominance-like ranking (or weighted normalization)
    to find the top 5% of rows.
    """
    # Identify column indices
    idx_mre = header.index('MRE-')
    idx_acc = header.index('ACC+')
    idx_pred = header.index('PRED40+')

    # Normalize values to 0-1 range to compare different scales
    mres = [r[idx_mre] for r in rows]
    accs = [r[idx_acc] for r in rows]
    preds = [r[idx_pred] for r in rows]

    min_mre, max_mre = min(mres), max(mres)
    min_acc, max_acc = min(accs), max(accs)
    min_pred, max_pred = min(preds), max(preds)

    def get_score(row):
        # Normalize: lower MRE is better, higher ACC/PRED is better
        n_mre = (row[idx_mre] - min_mre) / (max_mre - min_mre + 1e-9)
        n_acc = (row[idx_acc] - min_acc) / (max_acc - min_acc + 1e-9)
        n_pred = (row[idx_pred] - min_pred) / (max_pred - min_pred + 1e-9)
        # Score = (High ACC + High PRED) - Low MRE
        return (n_acc + n_pred) - n_mre

    # Calculate scores and keep track of original indices
    scored_rows = []
    for i, row in enumerate(rows):
        scored_rows.append((get_score(row), i))

    # Sort by score descending
    scored_rows.sort(key=lambda x: x[0], reverse=True)

    # Return top 5% (800 rows * 0.05 = 40 rows)
    limit = max(1, int(len(rows) * 0.05))
    return [item[1] for item in scored_rows[:limit]]

import csv
from step01 import remember

def guess(header, train, test):
    """
    Estimates the first goal column (MRE-) for test rows.
    Uses a K-Nearest Neighbors approach based on the input columns.
    """
    # Identify goal and input columns
    # Goal: MRE- (index 0)
    # Inputs: N_estimators, criterion, Min_sample_leaves, Min_impurity_decrease, Max_depth
    input_indices = [i for i, name in enumerate(header) if name not in ['MRE-', 'ACC+', 'PRED40+']]
    
    def get_distance(row1, row2):
        dist = 0
        for i in input_indices:
            val1, val2 = row1[i], row2[i]
            if isinstance(val1, str):  # Symbolic column: criterion
                dist += 0 if val1 == val2 else 1.0
            else:  # Numeric column
                # Simple normalized distance
                dist += abs(val1 - val2)
        return dist

    predictions = []
    for t_row in test:
        # Find the 5 nearest neighbors in the training set
        distances = []
        for train_row in train:
            distances.append((get_distance(t_row, train_row), train_row[0]))
        
        distances.sort(key=lambda x: x[0])
        # Average the MRE- of the 5 nearest neighbors
        neighbors = distances[:5]
        avg_mre = sum(n[1] for n in neighbors) / len(neighbors)
        predictions.append(avg_mre)
        
    return predictions

def certify(xs, ys):
    """
    Returns True if the difference between two distributions is statistically 
    insignificant (noise), False if they really differ.
    Uses a T-test for independent samples.
    """
    t_stat, p_value = stats.ttest_ind(xs, ys, equal_var=False)
    # If p_value > 0.05, we fail to reject the null hypothesis (they are the same)
    return p_value > 0.05

def blame(header, rows):
    """

    for i, row in enumerate(rows):
        # We need a score for every row.

        idx_mre = header.index('MRE-')
        idx_acc = header.index('ACC+')
        idx_pred = header.index('PRED40+')

        n_acc = (row[idx_acc] - min(accs)) / (max(accs) - min(accs) + 1e-9)
        n_pred = (row[idx_pred] - min(preds)) / (max(preds) - min(preds) + 1e-9)
        score = (n_acc + n_pred) - n_mre
        scored.append((score, row))
    
    scored.sort(key=lambda x: x[0])
    cutoff = len(rows) // 4
    bad_rows = [x[1] for x in scored[:cutoff]]
    good_rows = [x[1] for x in scored[cutoff:]]
    
    # Find inputs where the mean/mode in bad_rows deviates significantly from good_rows
    diffs = []
    for i in input_indices:
        col_name = header[i]
        if isinstance(bad_rows[0][i], str):
            # Categorical: compare frequency of the most common value
            bad_val = max(set(r[i] for r in bad_rows), key=[r[i] for r in bad_rows].count)
            good_val = max(set(r[i] for r in good_rows), key=[r[i] for r in good_rows].count)
            if bad_val != good_val:
                diffs.append((1.0, col_name))

            # Numeric: compare means
            bad_mean = sum(r[i] for r in bad_rows) / len(bad_rows)
            good_mean = sum(r[i] for r in good_rows) / len(good_rows)
            diffs.append((abs(bad_mean - good_mean), col_name))
            
    diffs.sort(key=lambda x: x[0], reverse=True)
    return [d[1] for d in diffs[:2]]

        # We use a simplified score based on the row's current values

        return r[idx_acc] + r[idx_pred] - r[idx_mre]

    best_change = {}
    best_score = -float('inf')

    # Iterate through each lever to find an improvement
    for col, allowed in levers.items():
        col_idx = header.index(col)
        original_val = row[col_idx]
        
        # Generate candidate values
        candidates = []
        if isinstance(allowed, list):  # Symbolic

        else:  # Numeric (low, high)

            # improvement is proportional to the lever change's historical 
            # impact (or simply return the first valid improvement found).
            # Here we return the first change that is valid.
            return {col: val}

    return best_change

import csv

def spot(header, rows):
    """
    Identifies rows that are 'suspicious' by detecting outliers in the input space.
    A row is considered suspicious if its input values are statistically 
    distant from the median of the rest of the dataset.
    """
    # Identify input columns (all except goals)
    goal_names = ['MRE-', 'ACC+', 'PRED40+']
    input_indices = [i for i, name in enumerate(header) if name not in goal_names]
    
    # Calculate medians for numeric inputs to detect outliers
    numeric_indices = [i for i in input_indices if not isinstance(rows[0][i], str)]
    medians = {}
    for i in numeric_indices:
        vals = sorted([r[i] for r in rows])
        medians[i] = vals[len(vals) // 2]
    
    # Calculate a 'suspicion score' for each row
    # Score = sum of normalized deviations from the median
    scores = []
    for idx, row in enumerate(rows):
        suspicion = 0
        for i in numeric_indices:
            # Simple deviation: distance from median
            # We use absolute difference
            suspicion += abs(row[i] - medians[i])
        scores.append((suspicion, idx))
    
    # Sort by suspicion descending
    scores.sort(key=lambda x: x[0], reverse=True)
    
    # Return the indices of the most suspicious rows
    return [item[1] for item in scores]

def race(header, rows, label):
    """
    Compares three strategies:
    1. 'random': Picks 17 rows at random.
    2. 'smart': Uses the choose() logic from step07 (17 rows).
    3. 'greedy': Picks the first 16 rows sequentially.
    """
    budget = 50
    results = {}
    
    # Strategy 1: Random
    random_indices = random.sample(range(len(rows)), 17)
    for idx in random_indices:
        goals = label(idx)
        rows[idx][0], rows[idx][1], rows[idx][2] = goals[0], goals[1], goals[2]
    results["random"] = (random_indices, remember(header, [rows[i] for i in random_indices])[0])

    # Strategy 2: Smart (using step07 logic)
    # We pass a wrapper to label to ensure we don't exceed budget
    smart_indices = [int(i * (len(rows) - 1) / 16) for i in range(17)]
    for idx in smart_indices:
        if rows[idx][0] == "?":
            goals = label(idx)
            rows[idx][0], rows[idx][1], rows[idx][2] = goals[0], goals[1], goals[2]
    results["smart"] = (smart_indices, remember(header, [rows[i] for i in smart_indices])[0])

    # Strategy 3: Greedy (first 16)
    greedy_indices = list(range(16))
    for idx in greedy_indices:
        if rows[idx][0] == "?":
            goals = label(idx)
            rows[idx][0], rows[idx][1], rows[idx][2] = goals[0], goals[1], goals[2]
    results["greedy"] = (greedy_indices, remember(header, [rows[i] for i in greedy_indices])[0])

    # Determine winner by comparing the best row found by each
    best_overall = None
    winner = None
    
    # Re-score all found rows to compare across strategies
    all_found = []
    for name, (indices, local_idx) in results.items():
        row = rows[indices[local_idx]]

import csv
from scipy import stats

def certify(xs, ys):
    """
    Returns True if the difference between two distributions is statistically 
    insignificant (noise), False if they really differ.
    Uses a T-test for independent samples.
    """
    # If the variance is too low or samples are empty, treat as noise
    if len(xs) < 2 or len(ys) < 2:
        return True
    t_stat, p_value = stats.ttest_ind(xs, ys, equal_var=False)
    # If p_value > 0.05, we fail to reject the null hypothesis (they are the same)
    return p_value > 0.05

import csv
from step01 import remember

def flow(header, row, seen):
    """
    Decides whether to stop based on whether the current row is the best 
    seen so far. We define 'best' using the multi-objective scoring 
    logic from step01.
    """
    if not seen:
        return False
    
    # Combine seen rows with the current row to evaluate relative quality
    all_rows = seen + [row]
    
    # Get the indices of the top 5% of the current batch
    best_indices = remember(header, all_rows)
    
    # If the current row (the last one in all_rows) is in the top 5%, 
    # we consider it a significant improvement and stop.
    current_idx = len(all_rows) - 1
    return current_idx in best_indices
