import csv

def remember(header, rows):
    """
    Identifies the best rows based on multi-objective optimization.
    Goals: Minimize MRE-, Maximize ACC+, Maximize PRED40+.
    Uses a simple Pareto-dominance-like ranking (or weighted normalization)
    to find the top 5% of rows.
    """
    # Identify column indices
    idx_mre = header.index('MRE-')
    idx_acc = header.index('ACC+')
    idx_pred = header.index('PRED40+')

    # Normalize values to 0-1 range to compare different scales
    mres = [r[idx_mre] for r in rows]
    accs = [r[idx_acc] for r in rows]
    preds = [r[idx_pred] for r in rows]

    min_mre, max_mre = min(mres), max(mres)
    min_acc, max_acc = min(accs), max(accs)
    min_pred, max_pred = min(preds), max(preds)

    def get_score(row):
        # Normalize: lower MRE is better, higher ACC/PRED is better
        n_mre = (row[idx_mre] - min_mre) / (max_mre - min_mre + 1e-9)
        n_acc = (row[idx_acc] - min_acc) / (max_acc - min_acc + 1e-9)
        n_pred = (row[idx_pred] - min_pred) / (max_pred - min_pred + 1e-9)
        # Score = (High ACC + High PRED) - Low MRE
        return (n_acc + n_pred) - n_mre

    # Calculate scores and keep track of original indices
    scored_rows = []
    for i, row in enumerate(rows):
        scored_rows.append((get_score(row), i))

    # Sort by score descending
    scored_rows.sort(key=lambda x: x[0], reverse=True)

    # Return top 5% (800 rows * 0.05 = 40 rows)
    limit = max(1, int(len(rows) * 0.05))
    return [item[1] for item in scored_rows[:limit]]

import csv

def guess(header, train, test):
    """
    Estimates the first goal column (MRE-) for test rows.
    Uses a K-Nearest Neighbors approach based on the input columns.
    """
    # Identify goal and input columns
    # Goal: MRE- (index 0)
    # Inputs: N_estimators, criterion, Min_sample_leaves, Min_impurity_decrease, Max_depth
    input_indices = [i for i, name in enumerate(header) if name not in ['MRE-', 'ACC+', 'PRED40+']]
    
    def get_distance(row1, row2):
        dist = 0
        for i in input_indices:
            val1, val2 = row1[i], row2[i]
            if isinstance(val1, str):  # Symbolic column: criterion
                dist += 0 if val1 == val2 else 1.0
            else:  # Numeric column
                # Simple normalized distance
                dist += abs(val1 - val2)
        return dist

    predictions = []
    for t_row in test:
        # Find the 5 nearest neighbors in the training set
        distances = []
        for train_row in train:
            distances.append((get_distance(t_row, train_row), train_row[0]))
        
        distances.sort(key=lambda x: x[0])
        # Average the MRE- of the 5 nearest neighbors
        neighbors = distances[:5]
        avg_mre = sum(n[1] for n in neighbors) / len(neighbors)
        predictions.append(avg_mre)
        
    return predictions

import csv
from scipy import stats

def certify(xs, ys):
    """
    Returns True if the difference between two distributions is statistically 
    insignificant (noise), False if they really differ.
    Uses a T-test for independent samples.
    """
    t_stat, p_value = stats.ttest_ind(xs, ys, equal_var=False)
    # If p_value > 0.05, we fail to reject the null hypothesis (they are the same)
    return p_value > 0.05

import csv
from step01 import remember

def blame(header, rows):
    """
    Identifies input columns that correlate with poor performance.
    'Bad' rows are defined as those in the bottom 25% of the remember() score.
    We compare the distribution of input values in bad rows vs the rest.
    """
    # Identify goal and input indices
    input_indices = [i for i, name in enumerate(header) if name not in ['MRE-', 'ACC+', 'PRED40+']]
    
    # Score all rows to find the bottom 25%
    # We reuse remember logic but invert the selection
    scored = []
    for i, row in enumerate(rows):
        # We need a score for every row. 
        # Since remember() returns top indices, we calculate scores manually:
        idx_mre = header.index('MRE-')
        idx_acc = header.index('ACC+')
        idx_pred = header.index('PRED40+')
        
        mres = [r[idx_mre] for r in rows]
        accs = [r[idx_acc] for r in rows]
        preds = [r[idx_pred] for r in rows]
        
        n_mre = (row[idx_mre] - min(mres)) / (max(mres) - min(mres) + 1e-9)
        n_acc = (row[idx_acc] - min(accs)) / (max(accs) - min(accs) + 1e-9)
        n_pred = (row[idx_pred] - min(preds)) / (max(preds) - min(preds) + 1e-9)
        score = (n_acc + n_pred) - n_mre
        scored.append((score, row))
    
    scored.sort(key=lambda x: x[0])
    cutoff = len(rows) // 4
    bad_rows = [x[1] for x in scored[:cutoff]]
    good_rows = [x[1] for x in scored[cutoff:]]
    
    # Find inputs where the mean/mode in bad_rows deviates significantly from good_rows
    diffs = []
    for i in input_indices:
        col_name = header[i]
        if isinstance(bad_rows[0][i], str):
            # Categorical: compare frequency of the most common value
            bad_val = max(set(r[i] for r in bad_rows), key=[r[i] for r in bad_rows].count)
            good_val = max(set(r[i] for r in good_rows), key=[r[i] for r in good_rows].count)
            if bad_val != good_val:
                diffs.append((1.0, col_name))
        else:
            # Numeric: compare means
            bad_mean = sum(r[i] for r in bad_rows) / len(bad_rows)
            good_mean = sum(r[i] for r in good_rows) / len(good_rows)
            diffs.append((abs(bad_mean - good_mean), col_name))
            
    diffs.sort(key=lambda x: x[0], reverse=True)
    return [d[1] for d in diffs[:2]]
