import csv
from step01 import remember
from step02 import guess

def justify(header, rows, row):
    """
    Estimates the performance of the given row and compares it against 
    the existing dataset to justify its quality.
    """
    # 1. Estimate the goals for the provided row
    # We use the guess() logic from step02
    estimated_goals = guess(header, rows, [row])
    
    # Create a copy of the row with estimated goals
    # row indices: 0:MRE-, 1:ACC+, 2:PRED40+
    row_with_goals = list(row)
    row_with_goals[0] = estimated_goals[0]
    # For ACC+ and PRED40+, we can use the average of the top 5% as a proxy 
    # or simply use the guess logic if we extended it. 
    # For simplicity, we use the mean of the dataset for the other two goals.
    idx_acc = header.index('ACC+')
    idx_pred = header.index('PRED40+')
    row_with_goals[1] = sum(r[idx_acc] for r in rows) / len(rows)
    row_with_goals[2] = sum(r[idx_pred] for r in rows) / len(rows)
    
    # 2. Calculate how many rows this row is better than
    # A row is better if its score is higher (using the remember() scoring logic)
    def get_score(r):
        idx_mre, idx_acc, idx_pred = 0, 1, 2
        mres = [x[idx_mre] for x in rows]
        accs = [x[idx_acc] for x in rows]
        preds = [x[idx_pred] for x in rows]
        
        n_mre = (r[idx_mre] - min(mres)) / (max(mres) - min(mres) + 1e-9)
        n_acc = (r[idx_acc] - min(accs)) / (max(accs) - min(accs) + 1e-9)
        n_pred = (r[idx_pred] - min(preds)) / (max(preds) - min(preds) + 1e-9)
        return (n_acc + n_pred) - n_mre

    target_score = get_score(row_with_goals)
    better_than_count = sum(1 for r in rows if target_score > get_score(r))
    p = better_than_count / len(rows)
    
    reason = f"This configuration outperforms {p:.1%} of all known setups based on its estimated performance metrics."
    
    return (p, reason)

if __name__ == "__main__":
    filename = 'Health-ClosedIssues0000.n1000.train.csv'
    rows = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = []
            for i, val in enumerate(row):
                if header[i] == 'criterion':
                    processed_row.append(val)
                else:
                    processed_row.append(float(val))
            rows.append(processed_row)
    
    # Pick a sample row and mask it
    sample = list(rows[0])
    sample[0] = sample[1] = sample[2] = "?"
    
    p, reason = justify(header, rows, sample)
    print(f"Score: {p:.4f}")
    print(f"Reason: {reason}")
    exit(0)
