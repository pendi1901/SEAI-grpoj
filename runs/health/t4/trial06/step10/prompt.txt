import csv

def remember(header, rows):
    """
    Identifies the best rows based on multi-objective optimization.
    Goals: Minimize MRE-, Maximize ACC+, Maximize PRED40+.
    Uses a simple Pareto-based ranking approach to find the 'best' setups.
    """
    # Identify indices of goals
    goals = []
    for i, name in enumerate(header):
        if name.endswith('-'):
            goals.append((i, -1))  # -1 for minimize
        elif name.endswith('+'):
            goals.append((i, 1))   # 1 for maximize

    def get_score(row):
        # Normalize goals to a 0-1 scale relative to the dataset
        # Since we need a simple heuristic, we use a weighted sum of normalized values
        # This identifies rows that perform well across all objectives
        score = 0
        for idx, direction in goals:
            col_data = [float(r[idx]) for r in rows]
            min_val, max_val = min(col_data), max(col_data)
            val = float(row[idx])
            norm = (val - min_val) / (max_val - min_val + 1e-9)
            score += norm if direction == 1 else (1 - norm)
        return score

    # Calculate scores and keep indices
    scored_rows = []
    for i, row in enumerate(rows):
        scored_rows.append((i, get_score(row)))

    # Sort by score descending
    scored_rows.sort(key=lambda x: x[1], reverse=True)

    # Return top 5% (max 40 rows for 800 rows)
    limit = max(1, int(len(rows) * 0.05))
    return [x[0] for x in scored_rows[:limit]]

def guess(header, train, test):
    """
    Estimates the first goal column (MRE-) using a Random Forest Regressor.
    Handles symbolic columns by encoding them and ignores '?' in test rows.
    """
    goal_idx = 0
    symbolic_cols = [i for i, name in enumerate(header) if name == 'criterion']
    
    # Fit encoders for symbolic columns using all available data
    encoders = {}
    for col in symbolic_cols:
        le = LabelEncoder()
        all_vals = [row[col] for row in train]
        le.fit(all_vals)
        encoders[col] = le

    def transform_row(row, is_test=False):
        new_row = []
        for i, val in enumerate(row):
            if i == goal_idx:
                continue
            if i in symbolic_cols:
                new_row.append(encoders[i].transform([val])[0])
            else:
                # Handle '?' in test rows by treating as 0.0 or skipping
                try:
                    new_row.append(float(val))
                except ValueError:
                    new_row.append(0.0)
        return new_row

    X_train = [transform_row(row) for row in train]
    y_train = [float(row[goal_idx]) for row in train]

    X_test = [transform_row(row, is_test=True) for row in test]

    # Train model
    model = RandomForestRegressor(n_estimators=100, random_state=42)
    model.fit(X_train, y_train)
    
    return model.predict(X_test).tolist()

def certify(xs, ys):
    """
    Returns True if the difference between two distributions is likely noise,
    False if they are statistically significantly different.
    Uses Welch's t-test.
    """
    t_stat, p_value = stats.ttest_ind(xs, ys, equal_var=False)
    # If p_value > 0.05, we fail to reject the null hypothesis (it's noise)
    return p_value > 0.05

def analyze_size_vs_accuracy(header, rows):
    """
    Determines if 'N_estimators' (size) correlates with 'ACC+' (accuracy).
    We split data into 'small' and 'big' based on the median N_estimators.
    """
    n_idx = header.index('N_estimators')
    acc_idx = header.index('ACC+')
    
    n_vals = [row[n_idx] for row in rows]
    median_n = np.median(n_vals)
    
    small_acc = [row[acc_idx] for row in rows if row[n_idx] <= median_n]
    big_acc = [row[acc_idx] for row in rows if row[n_idx] > median_n]
    
    is_noise = certify(small_acc, big_acc)
    
    return is_noise, np.mean(small_acc), np.mean(big_acc)

import csv

def flow(header, row, seen):
    """
    Decides if the current row is 'good enough' to ship.
    We define 'good enough' as being better than all previously seen rows
    across all objectives (Pareto-dominance).
    """
    goals = []
    for i, name in enumerate(header):
        if name.endswith('-'):
            goals.append((i, -1))  # Minimize
        elif name.endswith('+'):
            goals.append((i, 1))   # Maximize

    def dominates(row_a, row_b):
        # Returns True if row_a dominates row_b
        better_in_at_least_one = False
        for idx, direction in goals:
            val_a, val_b = row_a[idx], row_b[idx]
            if direction == 1: # Maximize
                if val_a < val_b: return False
                if val_a > val_b: better_in_at_least_one = True
            else: # Minimize
                if val_a > val_b: return False
                if val_a < val_b: better_in_at_least_one = True
        return better_in_at_least_one

    # If no previous rows, this is the best so far
    if not seen:
        return True

    # If this row dominates all previous rows, it is a new Pareto optimal point
    # We ship if it is better than everything we have seen so far.
    for prev in seen:
        if not dominates(row, prev):
            return False
            
    return True

    scores = []

        score = 0

            col_data = [float(r[i]) for r in rows]

def fix(header, row, levers):
    """
    Finds the smallest change to a bad row's settings that improves its 
    Pareto ranking without increasing the 'N_estimators' (proxy for speed).
    """
    def get_score(r):
        # Simple heuristic: sum of normalized goals
        score = 0
        for i, name in enumerate(header):
            if name.endswith('-'):
                score += (1 - r[i])
            elif name.endswith('+'):
                score += r[i]
        return score

    best_change = {}
    best_score = -1.0
    
    # Try small perturbations for each lever
    for col_name, allowed in levers.items():
        idx = header.index(col_name)
        original_val = row[idx]
        
        # Generate candidates
        candidates = []
        if isinstance(allowed, list):  # Symbolic
            candidates = [v for v in allowed if v != original_val]
        else:  # Numeric (low, high)
            low, high = allowed
            # Try a small step or boundary
            candidates = [low, high, (low + high) / 2]
            
        for val in candidates:
            # Constraint: Do not increase N_estimators (speed proxy)
            if col_name == 'N_estimators' and val > original_val:
                continue
                
            new_row = list(row)
            new_row[idx] = val
            
            # Evaluate improvement
            score = get_score(new_row)
            if score > best_score:
                best_score = score
                best_change = {col_name: val}
                
    return best_change

import csv
import numpy as np

def spot(header, rows):
    """
    Identifies suspicious rows by detecting outliers in input columns.
    Rows with values significantly outside the distribution (Z-score > 3)
    are flagged as potentially erroneous entries.
    """
    suspicious = []
    input_indices = [i for i, name in enumerate(header) if not name.endswith(('-', '+'))]
    
    # Track anomaly scores for each row
    anomaly_scores = [0.0] * len(rows)
    
    for i in input_indices:
        col_data = [r[i] for r in rows if not isinstance(r[i], str)]
        if not col_data:
            continue
            
        arr = np.array(col_data)
        mean = np.mean(arr)
        std = np.std(arr)
        
        if std == 0:
            continue
            
        for row_idx, row in enumerate(rows):
            val = row[i]
            if not isinstance(val, str):
                z_score = abs((val - mean) / std)
                if z_score > 3:
                    anomaly_scores[row_idx] += z_score
                    
    # Get indices of rows with non-zero anomaly scores, sorted by score descending
    scored_indices = [(idx, score) for idx, score in enumerate(anomaly_scores) if score > 0]
    scored_indices.sort(key=lambda x: x[1], reverse=True)
    
    return [idx for idx, score in scored_indices]
