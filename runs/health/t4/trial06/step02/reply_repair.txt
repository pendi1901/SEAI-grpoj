```python
import csv
from sklearn.ensemble import RandomForestRegressor
from sklearn.preprocessing import LabelEncoder
import numpy as np

def guess(header, train, test):
    """
    Estimates the first goal column (MRE-) using a Random Forest Regressor.
    Handles symbolic columns by encoding them and ignores '?' in test rows.
    """
    goal_idx = 0
    symbolic_cols = [i for i, name in enumerate(header) if name == 'criterion']
    
    # Fit encoders for symbolic columns using all available data
    encoders = {}
    for col in symbolic_cols:
        le = LabelEncoder()
        all_vals = [row[col] for row in train]
        le.fit(all_vals)
        encoders[col] = le

    def transform_row(row, is_test=False):
        new_row = []
        for i, val in enumerate(row):
            if i == goal_idx:
                continue
            if i in symbolic_cols:
                new_row.append(encoders[i].transform([val])[0])
            else:
                # Handle '?' in test rows by treating as 0.0 or skipping
                try:
                    new_row.append(float(val))
                except ValueError:
                    new_row.append(0.0)
        return new_row

    X_train = [transform_row(row) for row in train]
    y_train = [float(row[goal_idx]) for row in train]

    X_test = [transform_row(row, is_test=True) for row in test]

    # Train model
    model = RandomForestRegressor(n_estimators=100, random_state=42)
    model.fit(X_train, y_train)
    
    return model.predict(X_test).tolist()

if __name__ == "__main__":
    filename = 'Health-ClosedIssues0000.n1000.train.csv'
    rows = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = []
            for i, val in enumerate(row):
                if header[i] == 'criterion':
                    processed_row.append(val)
                else:
                    processed_row.append(float(val))
            rows.append(processed_row)

    # Split: 80% train, 20% test
    split = int(len(rows) * 0.8)
    train_data = rows[:split]
    # Create test data where the goal column is replaced by "?"
    test_data = []
    for row in rows[split:]:
        test_row = list(row)
        test_row[0] = "?"
        test_data.append(test_row)
    
    predictions = guess(header, train_data, test_data)
    
    print("Forecasts for the first 5 test configurations:")
    for i in range(min(5, len(predictions))):
        print(f"Setup {i+1}: Estimated MRE- = {predictions[i]:.4f}")
```
