import csv
from step02 import guess

def justify(header, rows, row):
    """
    Estimates the performance of the given row and compares it against 
    the provided dataset to justify its quality.
    """
    # 1. Identify goal indices and directions
    goal_indices = [i for i, name in enumerate(header) if name.endswith(('-', '+'))]
    
    # 2. Estimate the goal values for the target row
    # guess() returns a list of predictions for the first goal column (MRE-)
    # Since we only have one goal column (MRE-) in this specific dataset, 
    # we only get one prediction back.
    estimated_mre = guess(header, rows, [row])[0]
    
    # 3. Calculate how many rows the target row is better than
    def get_score(r):
        score = 0
        for idx in goal_indices:
            col_data = [float(row_i[idx]) for row_i in rows]
            min_v, max_v = min(col_data), max(col_data)
            val = float(r[idx])
            norm = (val - min_v) / (max_v - min_v + 1e-9)
            score += norm if header[idx].endswith('+') else (1 - norm)
        return score

    # Calculate target score using estimated MRE- and assuming average for others if needed
    # Given the dataset structure, MRE- is the primary goal.
    col_data = [float(r[goal_indices[0]]) for r in rows]
    min_v, max_v = min(col_data), max(col_data)
    norm = (estimated_mre - min_v) / (max_v - min_v + 1e-9)
    target_score = (1 - norm) # MRE- is minimize

    better_count = 0
    for r in rows:
        if get_score(r) < target_score:
            better_count += 1
            
    p = better_count / len(rows)
    reason = f"Using the biggest, deepest setup is not always optimal because this configuration is estimated to outperform only {p:.4f} of all known setups."
    
    return p, reason

if __name__ == "__main__":
    filename = 'Health-ClosedIssues0000.n1000.train.csv'
    rows = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = []
            for i, val in enumerate(row):
                if header[i] == 'criterion':
                    processed_row.append(val)
                else:
                    processed_row.append(float(val))
            rows.append(processed_row)

    # Define a "biggest, deepest" setup
    # N_estimators max is 200, Max_depth max is 12
    test_row = [0.0, 0.0, 0.0, 200.0, 'squared_error', 1.0, 1.0, 12.0]
    # Hide goals
    for i, name in enumerate(header):
        if name.endswith(('-', '+')):
            test_row[i] = "?"
            
    p, reason = justify(header, rows, test_row)
    print(reason)
