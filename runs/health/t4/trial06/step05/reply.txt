import csv
from sklearn.ensemble import RandomForestRegressor
from sklearn.preprocessing import LabelEncoder

def blame(header, rows):
    """
    Identifies input columns that correlate most strongly with poor performance.
    We define 'bad' as the bottom 25% of rows based on a simple aggregate score.
    We then use feature importance from a Random Forest to see which inputs
    predict the 'badness' of a row.
    """
    # Identify goal indices
    goal_indices = [i for i, name in enumerate(header) if name.endswith(('-', '+'))]
    input_indices = [i for i, name in enumerate(header) if not name.endswith(('-', '+'))]
    
    # Calculate a simple aggregate score (normalized) to define 'bad'
    scores = []
    for row in rows:
        score = 0
        for i in goal_indices:
            col_data = [float(r[i]) for r in rows]
            min_v, max_v = min(col_data), max(col_data)
            norm = (float(row[i]) - min_v) / (max_v - min_v + 1e-9)
            score += norm if header[i].endswith('+') else (1 - norm)
        scores.append(score)
    
    # Define 'bad' as the bottom 25%
    threshold = sorted(scores)[len(scores) // 4]
    y = [1 if s <= threshold else 0 for s in scores]
    
    # Prepare X (inputs)
    X = []
    encoders = {}
    for i in input_indices:
        if isinstance(rows[0][i], str):
            le = LabelEncoder()
            le.fit([r[i] for r in rows])
            encoders[i] = le
            
    for row in rows:
        new_row = []
        for i in input_indices:
            if i in encoders:
                new_row.append(encoders[i].transform([row[i]])[0])
            else:
                new_row.append(float(row[i]))
        X.append(new_row)
    
    # Use Random Forest to find feature importance
    from sklearn.ensemble import RandomForestClassifier
    clf = RandomForestClassifier(n_estimators=50, random_state=42)
    clf.fit(X, y)
    
    # Map importances back to column names
    importances = sorted(zip(input_indices, clf.feature_importances_), key=lambda x: x[1], reverse=True)
    return [header[i] for i, imp in importances[:2]]

if __name__ == "__main__":
    filename = 'Health-ClosedIssues0000.n1000.train.csv'
    rows = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = []
            for i, val in enumerate(row):
                if header[i] == 'criterion':
                    processed_row.append(val)
                else:
                    processed_row.append(float(val))
            rows.append(processed_row)
            
    blamed_cols = blame(header, rows)
    print("The settings most responsible for poor performance are:")
    for col in blamed_cols:
        print(f"- {col}")
