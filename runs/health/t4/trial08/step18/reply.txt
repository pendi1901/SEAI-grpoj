import csv
from step08 import fix as step08_fix

def fix(header, row, levers):
    """
    Recommends the single best change to a 'bad' row to improve its utility.
    Uses the logic from step08 to evaluate potential changes.
    """
    return step08_fix(header, row, levers)

if __name__ == "__main__":
    filename = 'Health-ClosedIssues0000.n1000.train.csv'
    rows = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for r in reader:
            processed = []
            for i, val in enumerate(r):
                if header[i][0].isupper():
                    try:
                        processed.append(float(val))
                    except ValueError:
                        processed.append(0.0)
                else:
                    processed.append(val)
            rows.append(processed)

    # Define levers based on column types
    levers = {
        'N_estimators': (50, 200),
        'criterion': ['absolute_error', 'poisson', 'squared_error'],
        'Min_sample_leaves': (1, 20),
        'Max_depth': (2, 20)
    }

    # Identify a 'bad' row (e.g., one with poor utility)
    # Utility = -MRE + ACC + PRED
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')
    
    bad_row = min(rows, key=lambda r: -r[mre_idx] + r[acc_idx] + r[pred_idx])
    
    recommendation = fix(header, bad_row, levers)
    
    print("Optimization recommendation for the provided setup:")
    for col, val in recommendation.items():
        print(f"Change '{col}' to {val} to improve performance.")
    
    exit(0)
