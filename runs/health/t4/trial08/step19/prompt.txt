import csv

def remember(header, rows):
    """
    Identifies the best rows based on multi-objective optimization.
    Goals: MRE- (minimize), ACC+ (maximize), PRED40+ (maximize).
    We use a simple Pareto-based approach or weighted normalization.
    Given the constraints, we normalize the goals and pick the top 5%.
    """
    # Identify column indices
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    # Extract goals and normalize
    # MRE is minimized, so we treat it as (1 - value) or negative for maximization
    scores = []
    for i, row in enumerate(rows):
        mre = row[mre_idx]
        acc = row[acc_idx]
        pred = row[pred_idx]
        
        # Simple utility: lower MRE is better, higher ACC/PRED is better
        # Using a simple linear combination of normalized values
        # Since we don't know the range, we use the values directly as relative indicators
        utility = (-mre) + acc + pred
        scores.append((utility, i))

    # Sort by utility descending
    scores.sort(key=lambda x: x[0], reverse=True)
    
    # Return top 5% (800 rows * 0.05 = 40 rows)
    limit = max(1, int(len(rows) * 0.05))
    return [item[1] for item in scores[:limit]]

import csv
from scipy import stats

def certify(xs, ys):
    """
    Returns True if the difference between two distributions is likely noise,
    False if they are statistically significantly different.
    Uses a two-sample t-test (assuming normal distribution of performance).
    """
    if len(xs) < 2 or len(ys) < 2:
        return True
    
    # t-test: null hypothesis is that the means are equal
    t_stat, p_value = stats.ttest_ind(xs, ys, equal_var=False)
    
    # If p_value > 0.05, we fail to reject the null hypothesis (it's likely noise)
    return p_value > 0.05

def analyze_setup_size(header, rows):
    """
    Splits data into 'big' and 'small' based on N_estimators,
    then compares their ACC+ performance.
    """
    n_idx = header.index('N_estimators')
    acc_idx = header.index('ACC+')
    
    # Split by median N_estimators
    vals = [row[n_idx] for row in rows]
    median = sum(vals) / len(vals)
    
    big = [row[acc_idx] for row in rows if row[n_idx] >= median]
    small = [row[acc_idx] for row in rows if row[n_idx] < median]
    
    is_noise = certify(big, small)
    
    print(f"Average ACC+ for big setups: {sum(big)/len(big):.4f}")
    print(f"Average ACC+ for small setups: {sum(small)/len(small):.4f}")
    
    if is_noise:
        print("Conclusion: The difference is likely just noise.")
    else:
        print("Conclusion: The difference is statistically significant.")

import csv

def flow(header, row, seen):
    """
    Decides if the current row is 'good enough' to stop searching.
    We define 'good enough' as being in the top 5% of the expected distribution
    based on the historical 'seen' data.
    """
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    def get_utility(r):
        # MRE- (minimize), ACC+ (maximize), PRED40+ (maximize)
        # Normalize by using negative MRE
        return -r[mre_idx] + r[acc_idx] + r[pred_idx]

    # If this is the first row, we can't compare yet
    if not seen:
        return False

    current_utility = get_utility(row)
    seen_utilities = [get_utility(r) for r in seen]
    
    # Threshold: Is this row better than 95% of what we have seen so far?
    # We use the 95th percentile of seen data as our stopping criterion.
    seen_utilities.sort()
    threshold_idx = int(len(seen_utilities) * 0.95)
    threshold = seen_utilities[threshold_idx]
    
    return current_utility >= threshold

def justify(header, rows, row):
    """
    Estimates the performance of the given row and compares it against 
    the historical dataset to determine its relative rank.
    """
    # Identify goal indices
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')
    
    # 1. Estimate the goals for the input row
    # We need to estimate all three goals. Since guess() only estimates MRE-,
    # we use a simple KNN approach for all goals.
    def get_all_goals(target_row):
        # We need to predict MRE-, ACC+, and PRED40+
        # For simplicity, we use the average of the 5 nearest neighbors
        # based on input features.
        input_indices = [i for i, name in enumerate(header) if not (name.endswith('-') or name.endswith('+'))]
        
        # Helper to get features
        def get_features(r):
            mapping = {'absolute_error': 0, 'poisson': 1, 'squared_error': 2}
            return [mapping.get(r[i], r[i]) if isinstance(r[i], str) else r[i] for i in input_indices]

        target_feats = get_features(target_row)
        
        # Find distances to all rows
        dists = []
        for r in rows:
            r_feats = get_features(r)
            dist = sum((a - b)**2 for a, b in zip(target_feats, r_feats))
            dists.append((dist, r))
        
        dists.sort(key=lambda x: x[0])
        neighbors = [d[1] for d in dists[:5]]
        
        est_mre = sum(n[mre_idx] for n in neighbors) / 5
        est_acc = sum(n[acc_idx] for n in neighbors) / 5
        est_pred = sum(n[pred_idx] for n in neighbors) / 5
        return est_mre, est_acc, est_pred

    est_mre, est_acc, est_pred = get_all_goals(row)
    
    # 2. Calculate utility for all rows and the target row
    def get_utility(mre, acc, pred):
        return -mre + acc + pred

    utilities = [get_utility(r[mre_idx], r[acc_idx], r[pred_idx]) for r in rows]
    target_utility = get_utility(est_mre, est_acc, est_pred)
    
    # 3. Calculate fraction of rows that are worse
    worse_count = sum(1 for u in utilities if target_utility > u)
    p = worse_count / len(rows)
    
    reason = f"This setup is estimated to outperform {p:.1%} of all historical configurations based on its predicted utility score."
    
    return (p, reason)

def fix(header, row, levers):
    """
    Finds the smallest change to a 'bad' row to improve its utility.

        col_idx = header.index(col_name)

        candidates = []
        if isinstance(allowed, list):  # Symbolic

        else:  # Numeric (low, high)
            # Try a few points in the range
            low, high = allowed
            candidates = [low, (low + high) / 2, high]

            test_row = list(row)
            test_row[col_idx] = val

            # Since guess only returns MRE-, we assume other goals stay similar

            utility = -pred_goals + row[acc_idx] + row[pred_idx]
            
            if utility > max_utility:
                max_utility = utility
                best_change = {col_name: val}

import csv

def spot(header, rows):
    """
    Identifies rows that look like failed runs (outliers) using IsolationForest.
    We focus on input columns as these define the 'setup' of the run.
    """
    # Identify input columns (all columns except goals)
    input_indices = [i for i, name in enumerate(header) 
                     if not (name.endswith('-') or name.endswith('+'))]
    
    # Prepare X matrix for anomaly detection
    X = []
    for row in rows:
        features = []
        for i in input_indices:
            val = row[i]
            if isinstance(val, str):
                # Consistent encoding with previous steps
                mapping = {'absolute_error': 0, 'poisson': 1, 'squared_error': 2}
                features.append(mapping.get(val, 0))
            else:
                features.append(val)
        X.append(features)
    
    X = np.array(X)
    
    # IsolationForest is effective for detecting anomalies in high-dimensional space
    # We must fit the model before calling decision_function
    clf = IsolationForest(contamination=0.05, random_state=42)
    clf.fit(X)
    scores = clf.decision_function(X)
    
    # Lower scores indicate more anomalous (suspicious) rows
    # Return indices sorted by suspicion (lowest score first)
    suspicious_indices = np.argsort(scores)
    
    return suspicious_indices.tolist()

    # Shared tracking

    # 1. Random: 16 labels

        get_label(i)

def guess(header, train, test):
    """
    Estimates the first goal column (MRE-) for test rows using k-Nearest Neighbors.

    knn.fit(X_train, y_train)

import csv

def flow(header, row, seen):
    """
    Decides whether to stop based on whether the current row is the best 
    seen so far. If the best utility has not improved for a significant 
    number of steps (e.g., 10% of the total expected data), we stop.
    """
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    def get_utility(r):
        # MRE- (minimize), ACC+ (maximize), PRED40+ (maximize)
        # We normalize by using negative MRE
        return -r[mre_idx] + r[acc_idx] + r[pred_idx]

    if not seen:
        return False

    current_utility = get_utility(row)
    seen_utilities = [get_utility(r) for r in seen]
    best_utility_so_far = max(seen_utilities)

    # If the current row is better than anything seen, it's an improvement.
    # If we haven't seen an improvement in a while, we stop.
    # Using a patience window of 80 (10% of 800 rows).
    patience = 80
    if len(seen) > patience:
        recent_utilities = seen_utilities[-patience:]
        if best_utility_so_far >= max(recent_utilities):
            return True

    return current_utility > best_utility_so_far

def justify(header, rows, row):
    """
    Estimates the performance of the given row and compares it against 
    the historical dataset to determine its relative rank.
    """
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')
    
    # 1. Estimate goals for the input row using KNN
    # We use the existing guess function which predicts MRE-
    # For ACC+ and PRED40+, we use the average of the entire dataset 
    # as a baseline or simple KNN-based estimation.
    # To keep it simple and consistent, we estimate all three.
    def get_all_goals(target_row):
        # Use the guess function for MRE-
        est_mre = step02_guess(header, rows, [target_row])[0]
        
        # For ACC+ and PRED40+, use the mean of the training data
        est_acc = sum(r[acc_idx] for r in rows) / len(rows)
        est_pred = sum(r[pred_idx] for r in rows) / len(rows)
        return est_mre, est_acc, est_pred

    est_mre, est_acc, est_pred = get_all_goals(row)
    
    # 2. Calculate utility
    def get_utility(mre, acc, pred):
        return -mre + acc + pred

    target_utility = get_utility(est_mre, est_acc, est_pred)
    utilities = [get_utility(r[mre_idx], r[acc_idx], r[pred_idx]) for r in rows]
    
    # 3. Calculate fraction of rows that are worse
    worse_count = sum(1 for u in utilities if target_utility > u)
    p = worse_count / len(rows)
    
    reason = "Using the biggest, deepest setup often leads to overfitting or diminishing returns, as evidenced by this configuration outperforming only " + f"{p:.1%} of historical setups."
    
    return (p, reason)

def choose(header, rows, label):
    """
    Finds a setup that forecasts well (using KNN) and is small (low N_estimators).
    We use a budget of 50 labels to explore the space.
    """
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')
    n_est_idx = header.index('N_estimators')
    
    # 1. Initial exploration: 20 random samples to build a baseline
    indices = random.sample(range(len(rows)), 20)
    for i in indices:
        vals = label(i)
        rows[i][mre_idx], rows[i][acc_idx], rows[i][pred_idx] = vals
        
    # 2. Use the remaining 30 labels to find the best balance
    # We want high utility AND low N_estimators.
    # Utility = -MRE + ACC + PRED
    # We define a multi-objective score: Utility - (N_estimators / 200)
    # This penalizes large N_estimators.
    
    def get_score(row):
        util = -row[mre_idx] + row[acc_idx] + row[pred_idx]
        # Normalize N_estimators (range approx 10-200)
        size_penalty = row[n_est_idx] / 200.0
        return util - size_penalty

    # Train a predictor on the 20 samples
    train_data = [rows[i] for i in indices]
    
    # Predict on 30 random candidates
    candidates = random.sample([i for i in range(len(rows)) if i not in indices], 30)
    preds = guess(header, train_data, [rows[i] for i in candidates])
    
    # Select the best candidate based on predicted utility and size
    best_idx = -1
    best_score = -float('inf')
    
    for i, pred_mre in zip(candidates, preds):
        # Estimate other goals as mean of training
        est_acc = np.mean([r[acc_idx] for r in train_data])
        est_pred = np.mean([r[pred_idx] for r in train_data])
        
        # Calculate score
        score = (-pred_mre + est_acc + est_pred) - (rows[i][n_est_idx] / 200.0)
        
        if score > best_score:
            best_score = score
            best_idx = i
            
    # Label the best one to confirm
    label(best_idx)
    
    return best_idx
