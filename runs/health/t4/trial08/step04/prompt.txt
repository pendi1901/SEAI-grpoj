import csv

def remember(header, rows):
    """
    Identifies the best rows based on multi-objective optimization.
    Goals: MRE- (minimize), ACC+ (maximize), PRED40+ (maximize).
    We use a simple Pareto-based approach or weighted normalization.
    Given the constraints, we normalize the goals and pick the top 5%.
    """
    # Identify column indices
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    # Extract goals and normalize
    # MRE is minimized, so we treat it as (1 - value) or negative for maximization
    scores = []
    for i, row in enumerate(rows):
        mre = row[mre_idx]
        acc = row[acc_idx]
        pred = row[pred_idx]
        
        # Simple utility: lower MRE is better, higher ACC/PRED is better
        # Using a simple linear combination of normalized values
        # Since we don't know the range, we use the values directly as relative indicators
        utility = (-mre) + acc + pred
        scores.append((utility, i))

    # Sort by utility descending
    scores.sort(key=lambda x: x[0], reverse=True)
    
    # Return top 5% (800 rows * 0.05 = 40 rows)
    limit = max(1, int(len(rows) * 0.05))
    return [item[1] for item in scores[:limit]]

def guess(header, train, test):
    """
    Estimates the first goal column (MRE-) for test rows using k-Nearest Neighbors.
    """
    # Identify goal index (first column)
    goal_idx = 0
    
    # Identify input columns (all columns except goals)
    # Goals are columns where name ends in '-' or '+'
    input_indices = [i for i, name in enumerate(header) if not (name.endswith('-') or name.endswith('+'))]
    
    # Prepare training data
    X_train = []
    y_train = []
    for row in train:
        # Convert symbolic columns to numeric for KNN
        features = []
        for i in input_indices:
            val = row[i]
            if isinstance(val, str):
                # Simple encoding for criterion: absolute_error=0, poisson=1, squared_error=2
                mapping = {'absolute_error': 0, 'poisson': 1, 'squared_error': 2}
                features.append(mapping.get(val, 0))
            else:
                features.append(val)
        X_train.append(features)
        y_train.append(row[goal_idx])
        
    # Prepare test data
    X_test = []
    for row in test:
        features = []
        for i in input_indices:
            val = row[i]
            if isinstance(val, str):
                mapping = {'absolute_error': 0, 'poisson': 1, 'squared_error': 2}
                features.append(mapping.get(val, 0))
            else:
                features.append(val)
        X_test.append(features)
        
    # Fit model
    knn = KNeighborsRegressor(n_neighbors=5)
    knn.fit(X_train, y_train)
    
    return knn.predict(X_test).tolist()

import csv
from scipy import stats

def certify(xs, ys):
    """
    Returns True if the difference between two distributions is likely noise,
    False if they are statistically significantly different.
    Uses a two-sample t-test (assuming normal distribution of performance).
    """
    if len(xs) < 2 or len(ys) < 2:
        return True
    
    # t-test: null hypothesis is that the means are equal
    t_stat, p_value = stats.ttest_ind(xs, ys, equal_var=False)
    
    # If p_value > 0.05, we fail to reject the null hypothesis (it's likely noise)
    return p_value > 0.05

def analyze_setup_size(header, rows):
    """
    Splits data into 'big' and 'small' based on N_estimators,
    then compares their ACC+ performance.
    """
    n_idx = header.index('N_estimators')
    acc_idx = header.index('ACC+')
    
    # Split by median N_estimators
    vals = [row[n_idx] for row in rows]
    median = sum(vals) / len(vals)
    
    big = [row[acc_idx] for row in rows if row[n_idx] >= median]
    small = [row[acc_idx] for row in rows if row[n_idx] < median]
    
    is_noise = certify(big, small)
    
    print(f"Average ACC+ for big setups: {sum(big)/len(big):.4f}")
    print(f"Average ACC+ for small setups: {sum(small)/len(small):.4f}")
    
    if is_noise:
        print("Conclusion: The difference is likely just noise.")
    else:
        print("Conclusion: The difference is statistically significant.")
