import csv
import random
from step02 import guess

def fix(header, row, levers):
    """
    Finds the smallest change to a 'bad' row to improve its utility.
    We define 'smallest change' as changing only one lever to a value 
    that maximizes the predicted utility.
    """
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    def get_utility(r):
        return -r[mre_idx] + r[acc_idx] + r[pred_idx]

    # Load data to train a simple predictor
    filename = 'Health-ClosedIssues0000.n1000.train.csv'
    rows = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        next(reader)
        for r in reader:
            processed = []
            for i, val in enumerate(r):
                if header[i][0].isupper():
                    processed.append(float(val))
                else:
                    processed.append(val)
            rows.append(processed)

    best_change = {}
    max_utility = -float('inf')

    # Try changing each lever one by one
    for col_name, allowed in levers.items():
        col_idx = header.index(col_name)
        original_val = row[col_idx]
        
        candidates = []
        if isinstance(allowed, list):  # Symbolic
            candidates = [v for v in allowed if v != original_val]
        else:  # Numeric (low, high)
            # Try a few points in the range
            low, high = allowed
            candidates = [low, (low + high) / 2, high]
            candidates = [v for v in candidates if v != original_val]

        for val in candidates:
            test_row = list(row)
            test_row[col_idx] = val
            
            # Predict goals using KNN from step02
            # We use the whole dataset as training
            pred_goals = guess(header, rows, [test_row])[0]
            
            # Since guess only returns MRE-, we assume other goals stay similar 
            # or use a simple heuristic for the sake of the fix
            # Here we estimate utility based on the predicted MRE-
            utility = -pred_goals + row[acc_idx] + row[pred_idx]
            
            if utility > max_utility:
                max_utility = utility
                best_change = {col_name: val}

    return best_change

if __name__ == "__main__":

            rows.append(processed)

    # Define levers based on column types
    levers = {
        'N_estimators': (50, 200),
        'criterion': ['absolute_error', 'poisson', 'squared_error'],
        'Min_sample_leaves': (1, 20),
        'Max_depth': (2, 20)
    }

    # Pick a 'bad' row (e.g., one with high MRE)
    bad_row = min(rows, key=lambda r: -r[0] + r[1] + r[2])
    
    recommendation = fix(header, bad_row, levers)
    print(f"To improve performance, change: {recommendation}")
    exit(0)
