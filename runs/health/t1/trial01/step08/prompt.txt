import csv

def remember(header, rows):
    """
    Identifies the top 5% of rows based on multi-objective optimization.
    Goals: MRE- (minimize), ACC+ (maximize), PRED40+ (maximize).
    We normalize values to [0, 1] and calculate a score.
    """
    # Identify column indices
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    # Collect data for normalization
    mres = [r[mre_idx] for r in rows]
    accs = [r[acc_idx] for r in rows]
    preds = [r[pred_idx] for r in rows]

    min_mre, max_mre = min(mres), max(mres)
    min_acc, max_acc = min(accs), max(accs)
    min_pred, max_pred = min(preds), max(preds)

    def normalize(val, min_v, max_v, reverse=False):
        if max_v == min_v: return 0.5
        norm = (val - min_v) / (max_v - min_v)
        return (1 - norm) if reverse else norm

    scores = []
    for i, row in enumerate(rows):
        # Score = average of normalized goals
        s = (normalize(row[mre_idx], min_mre, max_mre, True) +
             normalize(row[acc_idx], min_acc, max_acc, False) +
             normalize(row[pred_idx], min_pred, max_pred, False)) / 3
        scores.append((s, i))

    # Sort by score descending
    scores.sort(key=lambda x: x[0], reverse=True)
    
    # Return top 5% (max 40 rows for 800 records)
    limit = max(1, len(rows) // 20)
    return [item[1] for item in scores[:limit]]

def guess(header, train, test):
    """
    Estimates the first goal (MRE-) for test rows using a K-Nearest Neighbors approach.
    Calculates distance based on input columns (N_estimators, criterion, 
    Min_sample_leaves, Min_impurity_decrease, Max_depth).
    """
    # Identify indices
    goal_idx = 0
    input_indices = [i for i, name in enumerate(header) if name not in ['MRE-', 'ACC+', 'PRED40+']]
    
    # Map symbolic values to numeric for distance calculation
    criterion_map = {'absolute_error': 0, 'poisson': 1, 'squared_error': 2}
    
    def get_features(row):
        features = []
        for i in input_indices:
            val = row[i]
            if header[i] == 'criterion':
                features.append(criterion_map.get(val, 0))
            else:
                features.append(float(val))
        return features

    # Pre-process training data
    train_features = [get_features(r) for r in train]
    train_goals = [r[goal_idx] for r in train]
    
    # Simple Euclidean distance-based estimation (K=5)
    predictions = []
    for t_row in test:
        t_feat = get_features(t_row)
        distances = []
        for i, tr_feat in enumerate(train_features):
            dist = sum((a - b) ** 2 for a, b in zip(t_feat, tr_feat)) ** 0.5
            distances.append((dist, train_goals[i]))
        
        # Sort by distance and average the top 5 neighbors
        distances.sort(key=lambda x: x[0])
        k = 5
        avg_mre = sum(d[1] for d in distances[:k]) / k
        predictions.append(avg_mre)
        
    return predictions

import csv

def certify(xs, ys):
    """
    Determines if two distributions are statistically similar using a 
    non-parametric approach (Mann-Whitney U test logic).
    Returns True if they are the same apart from noise, False if they differ.
    """
    n1, n2 = len(xs), len(ys)
    if n1 == 0 or n2 == 0:
        return True
    
    # Combine and rank
    combined = sorted([(x, 0) for x in xs] + [(y, 1) for y in ys])
    ranks = {}
    for i, (val, group) in enumerate(combined):
        ranks.setdefault(val, []).append(i + 1)
    
    # Assign average rank for ties
    avg_ranks = {val: sum(r) / len(r) for val, r in ranks.items()}
    
    rank_sum_x = sum(avg_ranks[x] for x in xs)
    
    # U statistic
    u1 = rank_sum_x - (n1 * (n1 + 1)) / 2
    u2 = (n1 * n2) - u1
    u = min(u1, u2)
    
    # Normal approximation for large samples (n > 20)
    mu = (n1 * n2) / 2
    sigma = ((n1 * n2 * (n1 + n2 + 1)) / 12) ** 0.5
    
    if sigma == 0:
        return True
        
    z = abs(u - mu) / sigma
    # Critical value for 95% confidence (z=1.96)
    return z < 1.96

import csv

def flow(header, row, seen):
    """
    Decides whether to stop at the current row based on a threshold strategy.
    We define a 'good' row as one that ranks in the top 5% of the expected 
    distribution of all possible rows.
    """
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    # We estimate the global distribution based on the 'seen' data.
    # If we haven't seen enough, we keep looking.
    if len(seen) < 20:
        return False

    def normalize(val, min_v, max_v, reverse=False):
        if max_v == min_v: return 0.5
        norm = (val - min_v) / (max_v - min_v)
        return (1 - norm) if reverse else norm

    # Extract goals from seen data
    mres = [r[mre_idx] for r in seen]
    accs = [r[acc_idx] for r in seen]
    preds = [r[pred_idx] for r in seen]

    # Calculate current row score
    s = (normalize(row[mre_idx], min(mres), max(mres), True) +
         normalize(row[acc_idx], min(accs), max(accs), False) +
         normalize(row[pred_idx], min(preds), max(preds), False)) / 3

    # Calculate scores of seen rows
    seen_scores = []
    for r in seen:
        seen_scores.append((normalize(r[mre_idx], min(mres), max(mres), True) +
                            normalize(r[acc_idx], min(accs), max(accs), False) +
                            normalize(r[pred_idx], min(preds), max(preds), False)) / 3)
    
    # If the current row is better than 95% of what we've seen, we stop.
    threshold = sorted(seen_scores)[int(len(seen_scores) * 0.95)]
    return s >= threshold

import csv

def blame(header, rows):
    """
    Identifies input columns that correlate most strongly with poor performance.
    A row is 'bad' if its goals are significantly worse than the average.
    We calculate the correlation between input settings and a 'badness' score.
    """
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')
    
    input_indices = [i for i, name in enumerate(header) if name not in ['MRE-', 'ACC+', 'PRED40+']]
    
    # Calculate a 'badness' score for each row
    # Normalize goals: MRE (lower is better), ACC/PRED (higher is better)
    mres = [r[mre_idx] for r in rows]
    accs = [r[acc_idx] for r in rows]
    preds = [r[pred_idx] for r in rows]
    
    def norm(val, vals, reverse=False):
        mn, mx = min(vals), max(vals)
        if mx == mn: return 0.5
        n = (val - mn) / (mx - mn)
        return (1 - n) if reverse else n

    badness = []
    for r in rows:
        # Score = 1 - (average of normalized goals)
        score = (norm(r[mre_idx], mres, True) + norm(r[acc_idx], accs) + norm(r[pred_idx], preds)) / 3
        badness.append(1 - score)
    
    # Find correlations between input columns and badness
    correlations = {}
    for idx in input_indices:
        col_vals = [r[idx] for r in rows]
        # Handle symbolic columns by mapping to numeric
        if isinstance(col_vals[0], str):
            unique = list(set(col_vals))
            col_vals = [unique.index(v) for v in col_vals]
        
        # Simple correlation: average badness for high values vs low values
        mid = sum(col_vals) / len(col_vals)
        low_bad = [badness[i] for i, v in enumerate(col_vals) if v <= mid]
        high_bad = [badness[i] for i, v in enumerate(col_vals) if v > mid]
        
        avg_low = sum(low_bad) / len(low_bad) if low_bad else 0
        avg_high = sum(high_bad) / len(high_bad) if high_bad else 0
        
        correlations[header[idx]] = abs(avg_high - avg_low)
    
    # Return columns sorted by their impact on badness
    sorted_cols = sorted(correlations.items(), key=lambda x: x[1], reverse=True)
    return [item[0] for item in sorted_cols]

import csv

def justify(header, rows, row):
    """
    Estimates the performance of 'row' using K-Nearest Neighbors,
    then calculates what fraction of 'rows' it outperforms.
    """
    # Identify indices
    goal_indices = [i for i, name in enumerate(header) if name.endswith('-') or name.endswith('+')]
    input_indices = [i for i, name in enumerate(header) if i not in goal_indices]
    
    # Map symbolic values to numeric for distance calculation
    criterion_map = {'absolute_error': 0, 'poisson': 1, 'squared_error': 2}
    
    def get_features(r):
        features = []
        for i in input_indices:
            val = r[i]
            if header[i] == 'criterion':
                features.append(criterion_map.get(val, 0))
            else:
                features.append(float(val))
        return features

    # Estimate goals for the target row using K=5
    target_feat = get_features(row)
    distances = []
    for r in rows:
        dist = sum((a - b) ** 2 for a, b in zip(target_feat, get_features(r))) ** 0.5
        distances.append((dist, [r[i] for i in goal_indices]))
    
    distances.sort(key=lambda x: x[0])
    k = 5
    estimated_goals = []
    for g_idx in range(len(goal_indices)):
        avg = sum(d[1][g_idx] for d in distances[:k]) / k
        estimated_goals.append(avg)
    
    # Define "better" logic: minimize '-' columns, maximize '+' columns
    def is_better(goals_a, goals_b):
        # Returns True if goals_a is better than goals_b
        for i, idx in enumerate(goal_indices):
            name = header[idx]
            if name.endswith('-'):
                if goals_a[i] > goals_b[i]: return False
            else:
                if goals_a[i] < goals_b[i]: return False
        return True

    # Calculate fraction of rows outperformed
    better_count = 0
    for r in rows:
        actual_goals = [r[i] for i in goal_indices]
        if is_better(estimated_goals, actual_goals):
            better_count += 1
            
    p = better_count / len(rows)
    reason = f"This configuration is estimated to outperform {p:.1%} of all recorded setups based on its predicted performance metrics."
    
    return p, reason

def choose(header, rows, label):
    """
    Finds the best setup by sampling 50 rows.
    We use a simple strategy: sample 50 rows, query their true goals,
    and return the index of the one with the best multi-objective score.
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith('-') or name.endswith('+')]
    
    # Normalize goals for scoring
    def get_score(goals):
        scores = []
        for i, val in enumerate(goals):
            col_name = header[goal_indices[i]]
            # We don't have global min/max here, so we use the sampled data
            # to normalize. Since we only have 50, we use them as the population.
            scores.append((val, col_name))
        return scores

    # Sample 50 indices
    indices = random.sample(range(len(rows)), min(50, len(rows)))
    
    labeled_data = []
    for i in indices:
        goals = label(i)
        labeled_data.append((i, goals))
    
    # Normalize goals across the 50 samples
    def normalize(val, min_v, max_v, reverse):
        if max_v == min_v: return 0.5
        norm = (val - min_v) / (max_v - min_v)
        return (1 - norm) if reverse else norm

    # Extract goal values for normalization
    goal_vals = {idx: [d[1][i] for d in labeled_data] for i, idx in enumerate(goal_indices)}
    mins = {idx: min(vals) for idx, vals in goal_vals.items()}
    maxs = {idx: max(vals) for idx, vals in goal_vals.items()}
    
    best_idx = -1
    best_score = -1.0
    
    for i, goals in labeled_data:
        norm_scores = []
        for j, g_idx in enumerate(goal_indices):
            name = header[g_idx]
            reverse = name.endswith('-')
            norm_scores.append(normalize(goals[j], mins[g_idx], maxs[g_idx], reverse))
        
        avg_score = sum(norm_scores) / len(norm_scores)
        if avg_score > best_score:
            best_score = avg_score
            best_idx = i
            
    return best_idx
