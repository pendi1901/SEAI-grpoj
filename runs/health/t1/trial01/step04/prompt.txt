import csv


def remember(header, rows):
    # Goal columns: '-' suffix -> minimise, '+' suffix -> maximise
    goal_cols = [c for c in header if c.endswith('-') or c.endswith('+')]
    goal_map = {}  # col -> 'min' or 'max'
    for c in goal_cols:
        goal_map[c] = 'min' if c.endswith('-') else 'max'

    n = len(rows)
    best = []
    worst = []
    for c in goal_cols:
        vals = [float(r[header.index(c)]) for r in rows]
        best.append(min(vals))
        worst.append(max(vals))

    scores = []
    for i, row in enumerate(rows):
        s = 0.0
        for j, c in enumerate(goal_cols):
            v = float(row[header.index(c)])
            if goal_map[c] == 'min':
                s += (v - worst[j]) / (best[j] - worst[j])
            else:
                s += (best[j] - v) / (best[j] - worst[j])
        scores.append(s / len(goal_cols))

    # pick top rows, at most 5% of total
    k = max(1, int(n * 0.05))
    order = sorted(range(n), key=lambda i: scores[i], reverse=True)
    return list(order[:k])

import csv


def guess(header, train, test):
    """Estimate the first goal column for each test row using nearest-neighbour on training data."""
    # Identify columns: upper-case = numeric, lower-case = symbolic
    is_numeric = [c[0].isupper() for c in header]

    # Find the first goal column (one ending in '-' or '+')
    goal_col_idx = None
    for i, c in enumerate(header):
        if c.endswith('-') or c.endswith('+'):
            goal_col_idx = i
            break

    def to_float(cell):
        try:
            return float(cell)
        except (ValueError, TypeError):
            return 0.0

    # Precompute training distances using only non-goal numeric columns
    train_data = []
    for row in train:
        vals = [to_float(row[i]) if is_numeric[i] else row[i] for i in range(len(header))]
        train_data.append(vals)

    def distance(a, b):
        d = 0.0
        for i in range(len(header)):
            if not is_numeric[i]:
                continue
            av = a[i]
            bv = b[i]
            if isinstance(av, float) and isinstance(bv, float):
                d += (av - bv) ** 2
        return d

    def predict(row):
        best_dist = float('inf')
        best_val = None
        for i in range(len(train)):
            td = train_data[i]
            if row[goal_col_idx] == "?":
                # skip goal column in distance
                continue
            dist = 0.0
            for j in range(len(header)):
                if not is_numeric[j]:
                    continue
                if j == goal_col_idx:
                    continue
                av = td[j]
                bv = row[j]
                if isinstance(av, float) and isinstance(bv, float):
                    diff = av - bv
                    dist += diff * diff
            if dist < best_dist:
                best_dist = dist
                best_val = to_float(row[goal_col_idx])
        return best_val

    results = [predict(row) for row in test]
    return results

def certify(xs, ys):
    """Return True if xs and ys are statistically indistinguishable (same apart from noise)."""
    n_x = len(xs)
    n_y = len(ys)

    mean_x = sum(xs) / n_x
    mean_y = sum(ys) / n_y

    var_x = sum((v - mean_x) ** 2 for v in xs) / (n_x - 1) if n_x > 1 else 0.0
    var_y = sum((v - mean_y) ** 2 for v in ys) / (n_y - 1) if n_y > 1 else 0.0

    se = ((var_x / n_x) + (var_y / n_y)) ** 0.5

    diff = abs(mean_x - mean_y)

    if se == 0:
        return diff == 0
    return diff < 2 * se
