import csv

def remember(header, rows):
    """
    Identifies the top 5% of rows based on multi-objective optimization.
    Goals: MRE- (minimize), ACC+ (maximize), PRED40+ (maximize).
    We normalize values to [0, 1] and calculate a score.
    """
    # Identify column indices
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    # Collect data for normalization
    mres = [r[mre_idx] for r in rows]
    accs = [r[acc_idx] for r in rows]
    preds = [r[pred_idx] for r in rows]

    min_mre, max_mre = min(mres), max(mres)
    min_acc, max_acc = min(accs), max(accs)
    min_pred, max_pred = min(preds), max(preds)

    def normalize(val, min_v, max_v, reverse=False):
        if max_v == min_v: return 0.5
        norm = (val - min_v) / (max_v - min_v)
        return (1 - norm) if reverse else norm

    scores = []
    for i, row in enumerate(rows):
        # Score = average of normalized goals
        s = (normalize(row[mre_idx], min_mre, max_mre, True) +
             normalize(row[acc_idx], min_acc, max_acc, False) +
             normalize(row[pred_idx], min_pred, max_pred, False)) / 3
        scores.append((s, i))

    # Sort by score descending
    scores.sort(key=lambda x: x[0], reverse=True)
    
    # Return top 5% (max 40 rows for 800 records)
    limit = max(1, len(rows) // 20)
    return [item[1] for item in scores[:limit]]

import csv

def guess(header, train, test):
    """
    Estimates the first goal (MRE-) for test rows using a K-Nearest Neighbors approach.
    Calculates distance based on input columns (N_estimators, criterion, 
    Min_sample_leaves, Min_impurity_decrease, Max_depth).
    """
    # Identify indices
    goal_idx = 0
    input_indices = [i for i, name in enumerate(header) if name not in ['MRE-', 'ACC+', 'PRED40+']]
    
    # Map symbolic values to numeric for distance calculation
    criterion_map = {'absolute_error': 0, 'poisson': 1, 'squared_error': 2}
    
    def get_features(row):
        features = []
        for i in input_indices:
            val = row[i]
            if header[i] == 'criterion':
                features.append(criterion_map.get(val, 0))
            else:
                features.append(float(val))
        return features

    # Pre-process training data
    train_features = [get_features(r) for r in train]
    train_goals = [r[goal_idx] for r in train]
    
    # Simple Euclidean distance-based estimation (K=5)
    predictions = []
    for t_row in test:
        t_feat = get_features(t_row)
        distances = []
        for i, tr_feat in enumerate(train_features):
            dist = sum((a - b) ** 2 for a, b in zip(t_feat, tr_feat)) ** 0.5
            distances.append((dist, train_goals[i]))
        
        # Sort by distance and average the top 5 neighbors
        distances.sort(key=lambda x: x[0])
        k = 5
        avg_mre = sum(d[1] for d in distances[:k]) / k
        predictions.append(avg_mre)
        
    return predictions

import csv

def certify(xs, ys):
    """
    Determines if two distributions are statistically similar using a 
    non-parametric approach (Mann-Whitney U test logic).
    Returns True if they are the same apart from noise, False if they differ.
    """
    n1, n2 = len(xs), len(ys)
    if n1 == 0 or n2 == 0:
        return True
    
    # Combine and rank
    combined = sorted([(x, 0) for x in xs] + [(y, 1) for y in ys])
    ranks = {}
    for i, (val, group) in enumerate(combined):
        ranks.setdefault(val, []).append(i + 1)
    
    # Assign average rank for ties
    avg_ranks = {val: sum(r) / len(r) for val, r in ranks.items()}
    
    rank_sum_x = sum(avg_ranks[x] for x in xs)
    
    # U statistic
    u1 = rank_sum_x - (n1 * (n1 + 1)) / 2
    u2 = (n1 * n2) - u1
    u = min(u1, u2)
    
    # Normal approximation for large samples (n > 20)
    mu = (n1 * n2) / 2
    sigma = ((n1 * n2 * (n1 + n2 + 1)) / 12) ** 0.5
    
    if sigma == 0:
        return True
        
    z = abs(u - mu) / sigma
    # Critical value for 95% confidence (z=1.96)
    return z < 1.96

def analyze_correlation(header, rows):
    # Setup: "Big" vs "Small" based on N_estimators
    n_idx = header.index('N_estimators')
    acc_idx = header.index('ACC+')

    median_n = sum(vals) / len(vals)

    big_setups = [r[acc_idx] for r in rows if r[n_idx] > median_n]
