import csv


def remember(header, rows):
    # Goal columns: '-' suffix -> minimise, '+' suffix -> maximise
    goal_cols = [c for c in header if c.endswith('-') or c.endswith('+')]
    goal_map = {}  # col -> 'min' or 'max'
    for c in goal_cols:
        goal_map[c] = 'min' if c.endswith('-') else 'max'

    n = len(rows)
    best = []
    worst = []
    for c in goal_cols:
        vals = [float(r[header.index(c)]) for r in rows]
        best.append(min(vals))
        worst.append(max(vals))

    scores = []
    for i, row in enumerate(rows):
        s = 0.0
        for j, c in enumerate(goal_cols):
            v = float(row[header.index(c)])
            if goal_map[c] == 'min':
                s += (v - worst[j]) / (best[j] - worst[j])
            else:
                s += (best[j] - v) / (best[j] - worst[j])
        scores.append(s / len(goal_cols))

    # pick top rows, at most 5% of total
    k = max(1, int(n * 0.05))
    order = sorted(range(n), key=lambda i: scores[i], reverse=True)
    return list(order[:k])

import csv


def guess(header, train, test):
    """Estimate the first goal column for each test row using nearest-neighbour on training data."""
    # Identify columns: upper-case = numeric, lower-case = symbolic
    is_numeric = [c[0].isupper() for c in header]

    # Find the first goal column (one ending in '-' or '+')
    goal_col_idx = None
    for i, c in enumerate(header):
        if c.endswith('-') or c.endswith('+'):
            goal_col_idx = i
            break

    def to_float(cell):
        try:
            return float(cell)
        except (ValueError, TypeError):
            return 0.0

    # Precompute training distances using only non-goal numeric columns
    train_data = []
    for row in train:
        vals = [to_float(row[i]) if is_numeric[i] else row[i] for i in range(len(header))]
        train_data.append(vals)

    def distance(a, b):
        d = 0.0
        for i in range(len(header)):
            if not is_numeric[i]:
                continue
            av = a[i]
            bv = b[i]
            if isinstance(av, float) and isinstance(bv, float):
                d += (av - bv) ** 2
        return d

    def predict(row):
        best_dist = float('inf')
        best_val = None
        for i in range(len(train)):
            td = train_data[i]
            if row[goal_col_idx] == "?":
                # skip goal column in distance
                continue
            dist = 0.0
            for j in range(len(header)):
                if not is_numeric[j]:
                    continue
                if j == goal_col_idx:
                    continue
                av = td[j]
                bv = row[j]
                if isinstance(av, float) and isinstance(bv, float):
                    diff = av - bv
                    dist += diff * diff
            if dist < best_dist:
                best_dist = dist
                best_val = to_float(row[goal_col_idx])
        return best_val

    results = [predict(row) for row in test]
    return results

def certify(xs, ys):
    """Return True if xs and ys are statistically indistinguishable (same apart from noise)."""
    n_x = len(xs)
    n_y = len(ys)

    mean_x = sum(xs) / n_x
    mean_y = sum(ys) / n_y

    var_x = sum((v - mean_x) ** 2 for v in xs) / (n_x - 1) if n_x > 1 else 0.0
    var_y = sum((v - mean_y) ** 2 for v in ys) / (n_y - 1) if n_y > 1 else 0.0

    se = ((var_x / n_x) + (var_y / n_y)) ** 0.5

    diff = abs(mean_x - mean_y)

    if se == 0:
        return diff == 0
    return diff < 2 * se

def flow(header, row, seen):
    """Return True if `row` is good enough to ship; False otherwise."""
    
    goal_cols = [c for c in header if c.endswith('-') or c.endswith('+')]
    
    def to_float(cell):
        try:
            return float(cell)
        except (ValueError, TypeError):
            return 0.0
    
    n_seen = len(seen)
    if n_seen < 10:
        # Need more data before making a decision
        return False
    
    # Track best and worst for each goal column from previously seen rows
    best_vals = []
    worst_vals = []
    for c in goal_cols:
        vals = [to_float(r[header.index(c)]) for r in seen]
        if vals:
            best_vals.append(min(vals))
            worst_vals.append(max(vals))
    
    # Evaluate current row against these ranges
    cur_vals = [to_float(row[header.index(c)]) for c in goal_cols]
    
    scores = []
    for i, (c, gv) in enumerate(zip(goal_cols, best_vals)):
        cv = cur_vals[i]
        if c.endswith('-'):
            # Minimize: lower is better; score 1.0 when equal to best
            denom = gv - worst_vals[i]
            if denom <= 0:
                scores.append(1.0)
            else:
                scores.append(max(0.0, 1.0 - (cv - gv) / denom))
        else:
            # Maximize: higher is better; score 1.0 when equal to best
            denom = best_vals[i] - worst_vals[i]
            if denom <= 0:
                scores.append(1.0)
            else:
                scores.append(max(0.0, 1.0 - (best_vals[i] - cv) / denom))
    
    avg_score = sum(scores) / len(scores)
    return avg_score >= 0.95

def blame(header, rows):
    """Return names of input columns that best explain why the worst rows are bad."""

    goal_cols = [c for c in header if c.endswith('-') or c.endswith('+')]

    def to_float(cell):
        try:
            return float(cell)
        except (ValueError, TypeError):
            return 0.0

    # --- score every row on its goals ----------------------------------------
    goal_dirs = {}
    for c in goal_cols:
        goal_dirs[c] = 'min' if c.endswith('-') else 'max'

    n = len(rows)
    best_vals = []
    worst_vals = []
    for c in goal_cols:
        vals = [to_float(r[header.index(c)]) for r in rows]
        best_vals.append(min(vals))
        worst_vals.append(max(vals))

    row_scores = []
    for i, row in enumerate(rows):
        s = 0.0
        for j, c in enumerate(goal_cols):
            v = to_float(row[header.index(c)])
            if goal_dirs[c] == 'min':
                denom = best_vals[j] - worst_vals[j]
                if denom <= 0:
                    s += 1.0
                else:
                    s += (v - worst_vals[j]) / denom
            else:
                denom = best_vals[j] - worst_vals[j]
                if denom <= 0:
                    s += 1.0
                else:
                    s += (best_vals[j] - v) / denom
        row_scores.append(s / len(goal_cols))

    # --- split into worst ~5% vs the rest ------------------------------------
    k = max(1, int(n * 0.05))
    order = sorted(range(n), key=lambda i: row_scores[i], reverse=True)
    worst_indices = set(order[n - k:])
    best_indices = set(order[: n - k])

    # --- for each input column check if its value differs between groups -----
    is_numeric = [c[0].isupper() for c in header]
    result = []

    for i, col in enumerate(header):
        if not is_numeric[i]:
            continue
        if col in goal_cols:
            continue

        worst_vals_list = [to_float(rows[j][i]) for j in worst_indices]
        best_vals_list  = [to_float(rows[j][i]) for j in best_indices]

        mean_worst = sum(worst_vals_list) / len(worst_vals_list) if worst_vals_list else 0.0
        mean_best  = sum(best_vals_list)  / len(best_vals_list)  if best_vals_list  else 0.0

        all_vals = [to_float(rows[j][i]) for j in range(n)]
        overall_mean = sum(all_vals) / n
        overall_std = (sum((v - overall_mean) ** 2 for v in all_vals) / n) ** 0.5 if n > 1 else 0.0

        diff = abs(mean_worst - mean_best)
        if overall_std > 0 and diff > 0.3 * overall_std:
            result.append(col)

    return result

def justify(header, rows, row):
    """Compare a test setup (with ? goals) against all training setups."""

    goal_cols = [c for c in header if c.endswith('-') or c.endswith('+')]
    is_numeric = [c[0].isupper() for c in header]

    def to_float(cell):
        try:
            return float(cell)
        except (ValueError, TypeError):
            return 0.0

    n = len(rows)

    # Compute best/worst per goal column from training data
    best_vals = []
    worst_vals = []
    for gcol in goal_cols:
        vals = [to_float(r[header.index(gcol)]) for r in rows]
        best_vals.append(min(vals))
        worst_vals.append(max(vals))

    # Estimate goals for test row using nearest-neighbour on non-goal numeric columns
    goal_estimates = []
    for gcol in goal_cols:
        best_dist = float('inf')
        best_val = None
        for i, train_row in enumerate(rows):
            dist = 0.0
            for j in range(len(header)):
                if not is_numeric[j]:
                    continue
                if j == header.index(gcol):
                    continue
                tv = to_float(train_row[j])
                rv = to_float(row[j])
                diff = tv - rv
                dist += diff * diff
            if dist < best_dist:
                best_dist = dist
                best_val = to_float(train_row[header.index(gcol)])
        goal_estimates.append(best_val)

    # Score a row on its goals using normalized scoring (best=1, worst=0)
    def score_row(vals):
        s = 0.0
        for j, gcol in enumerate(goal_cols):
            v = vals[j]
            if gcol.endswith('-'):
                denom = best_vals[j] - worst_vals[j]
                if denom <= 0:
                    return 1.0
                else:
                    s += (v - worst_vals[j]) / denom
            else:
                denom = best_vals[j] - worst_vals[j]
                if denom <= 0:
                    return 1.0
                else:
                    s += (best_vals[j] - v) / denom
        return s / len(goal_cols)

    test_score = score_row(goal_estimates)

    # Count how many training rows have lower scores (i.e., are worse)
    beaten_count = 0
    for i, train_row in enumerate(rows):
        vals = [to_float(train_row[header.index(gcol)]) for gcol in goal_cols]
        train_score = score_row(vals)
        if test_score > train_score:
            beaten_count += 1

    p = beaten_count / n

    reason = f"This setup is better than {p:.0%} of all training setups."
    return (p, reason)
