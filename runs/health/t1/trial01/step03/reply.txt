def certify(xs, ys):
    """Return True if xs and ys are statistically indistinguishable (same apart from noise)."""
    n_x = len(xs)
    n_y = len(ys)

    mean_x = sum(xs) / n_x
    mean_y = sum(ys) / n_y

    var_x = sum((v - mean_x) ** 2 for v in xs) / (n_x - 1) if n_x > 1 else 0.0
    var_y = sum((v - mean_y) ** 2 for v in ys) / (n_y - 1) if n_y > 1 else 0.0

    se = ((var_x / n_x) + (var_y / n_y)) ** 0.5

    diff = abs(mean_x - mean_y)

    if se == 0:
        return diff == 0
    return diff < 2 * se
