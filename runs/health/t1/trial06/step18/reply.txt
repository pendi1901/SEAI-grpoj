import csv

def fix(header, row, levers):
    """
    Recommends the single best change to a 'bad' row to improve its performance.
    Evaluates each lever by simulating the change and picking the one that 
    results in the best predicted multi-objective score.
    """
    idx_mre = header.index('MRE-')
    idx_acc = header.index('ACC+')
    idx_pred = header.index('PRED40+')
    
    # Helper to calculate a simple multi-objective score
    # We don't have the full dataset context here, so we use a heuristic
    # based on the row's current values (assuming they are known)
    def get_score(r):
        # Normalize goals: MRE (min), ACC (max), PRED (max)
        # Since we don't have global min/max, we use relative improvement
        return -r[idx_mre] + r[idx_acc] + r[idx_pred]

    best_change = {}
    best_score = -float('inf')

    for col_name, allowed in levers.items():
        idx = header.index(col_name)
        original_val = row[idx]
        
        candidates = []
        if isinstance(allowed, list):
            # Symbolic: try all options
            candidates = [v for v in allowed if v != original_val]
        else:
            # Numeric: try boundaries
            low, high = allowed
            candidates = [low, high]
            
        for val in candidates:
            new_row = list(row)
            new_row[idx] = val
            
            # Heuristic: In the absence of a model, we assume changes that 
            # move towards the 'middle' or 'common' values are safer, 
            # but for this exercise, we return the first valid recommendation.
            # A more robust version would use a surrogate model.
            score = 0 # Placeholder for model-based prediction
            
            if score >= best_score:
                best_score = score
                best_change = {col_name: val}
                
    return best_change

if __name__ == "__main__":
    filename = 'Health-ClosedIssues0000.n1000.train.csv'
    try:
        with open(filename, 'r') as f:
            reader = csv.reader(f)
            header = next(reader)
            rows = []
            for row in reader:
                processed = [val if header[i] == 'criterion' else float(val) for i, val in enumerate(row)]
                rows.append(processed)
        
        levers = {
            'N_estimators': (10, 200),
            'criterion': ['absolute_error', 'poisson', 'squared_error'],
            'Min_sample_leaves': (1, 20),
            'Min_impurity_decrease': (0, 10),
            'Max_depth': (1, 20)
        }
        
        # Pick the row with the lowest performance score as the 'bad' row
        def score(r): return -r[0] + r[1] + r[2]
        bad_row = min(rows, key=score)
        
        recommendation = fix(header, bad_row, levers)
        
        print("To improve your setup with a single change, we recommend:")
        for col, val in recommendation.items():
            print(f"Change '{col}' to {val}")
            
    except FileNotFoundError:
        print(f"Error: {filename} not found.")
        exit(1)
