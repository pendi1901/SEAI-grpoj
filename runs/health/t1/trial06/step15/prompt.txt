import csv

def remember(header, rows):
    """
    Identifies the top 5% of rows based on multi-objective optimization.
    Goals: MRE- (minimize), ACC+ (maximize), PRED40+ (maximize).
    We normalize values to [0, 1] and use a simple weighted sum (Pareto-like ranking).
    """
    n = len(rows)
    limit = max(1, n // 20)  # 5% limit
    
    # Identify column indices
    idx_mre = header.index('MRE-')
    idx_acc = header.index('ACC+')
    idx_pred = header.index('PRED40+')
    
    # Extract columns for normalization
    mres = [r[idx_mre] for r in rows]
    accs = [r[idx_acc] for r in rows]
    preds = [r[idx_pred] for r in rows]
    
    min_mre, max_mre = min(mres), max(mres)
    min_acc, max_acc = min(accs), max(accs)
    min_pred, max_pred = min(preds), max(preds)
    
    def normalize(val, min_v, max_v, reverse=False):
        if max_v == min_v: return 0.0
        norm = (val - min_v) / (max_v - min_v)
        return (1.0 - norm) if reverse else norm

    scores = []
    for i, row in enumerate(rows):
        # Score = (1 - MRE_norm) + ACC_norm + PRED_norm
        s = (normalize(row[idx_mre], min_mre, max_mre, True) +
             normalize(row[idx_acc], min_acc, max_acc) +
             normalize(row[idx_pred], min_pred, max_pred))
        scores.append((s, i))
    
    # Sort by score descending
    scores.sort(key=lambda x: x[0], reverse=True)
    
    return [item[1] for item in scores[:limit]]

import csv

def guess(header, train, test):
    """
    Estimates the first goal column (MRE-) for test rows using a 
    k-Nearest Neighbors approach based on Euclidean distance of 
    numeric input features.
    """
    # Identify goal and input columns
    goal_idx = 0  # MRE-
    # Inputs are all columns except the goals (first 3 columns)
    input_indices = [i for i, name in enumerate(header) if i >= 3]
    
    # Normalize numeric inputs for distance calculation
    def get_stats(data):
        stats = []
        for i in input_indices:
            col = [r[i] for r in data if isinstance(r[i], (int, float))]
            if col:
                stats.append((min(col), max(col)))
            else:
                stats.append((0, 1))
        return stats

    stats = get_stats(train)

    def distance(row1, row2):
        d = 0
        for idx, (min_v, max_v) in zip(input_indices, stats):
            v1, v2 = row1[idx], row2[idx]
            if isinstance(v1, str):  # Symbolic comparison
                d += 0 if v1 == v2 else 1
            else:  # Numeric normalization
                range_v = max_v - min_v
                d += ((v1 - v2) / range_v)**2 if range_v > 0 else 0
        return d**0.5

    predictions = []
    for t_row in test:
        # Find 5 nearest neighbors
        distances = []
        for train_row in train:
            distances.append((distance(t_row, train_row), train_row[goal_idx]))
        
        distances.sort(key=lambda x: x[0])
        # Return average of top 5 neighbors
        neighbors = distances[:5]
        avg_mre = sum(n[1] for n in neighbors) / len(neighbors)
        predictions.append(avg_mre)
        
    return predictions

import csv

def certify(xs, ys):
    """
    Returns True if the distributions of xs and ys are statistically 
    indistinguishable (same apart from noise), False otherwise.
    Uses a simple A12 effect size test (Vargha-Delaney).
    """
    if not xs or not ys:
        return True
    
    n_x = len(xs)
    n_y = len(ys)
    
    # Count how many times x > y, x < y, and x == y
    greater = 0
    less = 0
    for x in xs:
        for y in ys:
            if x > y:
                greater += 1
            elif x < y:
                less += 1
                
    # A12 = (count(x > y) + 0.5 * count(x == y)) / (n_x * n_y)
    a12 = (greater + 0.5 * (n_x * n_y - greater - less)) / (n_x * n_y)
    
    # If A12 is close to 0.5, they are likely the same (noise).
    # Thresholds: 0.56, 0.64, 0.71 are small, medium, large effects.
    return 0.44 <= a12 <= 0.56

def analyze(header, rows):
    """
    Compares 'ACC+' (accuracy) for 'big' vs 'small' setups.
    'Big' defined as N_estimators > median.
    """
    idx_acc = header.index('ACC+')
    idx_n = header.index('N_estimators')
    
    n_values = [r[idx_n] for r in rows]
    median_n = sorted(n_values)[len(n_values) // 2]
    
    small = [r[idx_acc] for r in rows if r[idx_n] <= median_n]
    big = [r[idx_acc] for r in rows if r[idx_n] > median_n]
    
    is_same = certify(small, big)
    
    return is_same, len(small), len(big)

import csv

def flow(header, row, seen):
    """
    Decides whether to stop at the current row based on a threshold.
    We define 'good enough' as being in the top 5% of the expected 
    performance range based on the history seen so far.
    """
    if not seen:
        return False

    # Identify goal indices
    idx_mre = header.index('MRE-')
    idx_acc = header.index('ACC+')
    idx_pred = header.index('PRED40+')

    # Collect all seen goals to establish a baseline
    mres = [r[idx_mre] for r in seen]
    accs = [r[idx_acc] for r in seen]
    preds = [r[idx_pred] for r in seen]

    # Calculate current row score
    # Normalize against the history seen so far
    def norm(val, col_data, reverse=False):
        min_v, max_v = min(col_data), max(col_data)
        if max_v == min_v: return 0.5
        res = (val - min_v) / (max_v - min_v)
        return (1.0 - res) if reverse else res

    score = (norm(row[idx_mre], mres, True) + 
             norm(row[idx_acc], accs) + 
             norm(row[idx_pred], preds)) / 3.0

    # Threshold: Stop if score is in the top 5% of observed scores
    # We use a heuristic: if score > 0.9 (very high relative to history)
    return score >= 0.9

import csv

def blame(header, rows):
    """
    Identifies which input columns correlate most strongly with poor performance.
    'Bad' rows are defined as the bottom 10% based on a multi-objective score.
    We calculate the correlation (or difference in means) between input values
    and the 'badness' of the rows.
    """
    # Identify goal indices
    idx_mre = header.index('MRE-')
    idx_acc = header.index('ACC+')
    idx_pred = header.index('PRED40+')
    
    # Identify input indices
    input_indices = [i for i, name in enumerate(header) if name not in ['MRE-', 'ACC+', 'PRED40+']]
    
    # Calculate a score for each row to define 'badness'
    # Lower score = worse performance
    def get_score(row):
        # Normalize goals: MRE (min), ACC (max), PRED (max)
        # We use simple relative ranking here
        return -row[idx_mre] + row[idx_acc] + row[idx_pred]

    scored_rows = sorted([(get_score(r), r) for r in rows])
    
    # Define 'bad' as the bottom 10%
    n_bad = max(1, len(rows) // 10)
    bad_rows = [r for s, r in scored_rows[:n_bad]]
    all_rows = rows
    
    # For each input, calculate the difference in mean/distribution between bad and all
    # We look for inputs where the bad rows have significantly different values
    impacts = []
    for i in input_indices:
        col_name = header[i]
        vals = [r[i] for r in all_rows if isinstance(r[i], (int, float))]
        bad_vals = [r[i] for r in bad_rows if isinstance(r[i], (int, float))]
        
        if not vals or not bad_vals:
            continue
            
        mean_all = sum(vals) / len(vals)
        mean_bad = sum(bad_vals) / len(bad_vals)
        
        # The 'blame' score is the absolute difference in means
        # normalized by the range of the column
        diff = abs(mean_bad - mean_all)
        impacts.append((diff, col_name))
    
    # Sort by impact descending
    impacts.sort(key=lambda x: x[0], reverse=True)
    
    # Return the top 2 most impactful input columns
    return [name for diff, name in impacts[:2]]

import csv

def justify(header, rows, row):
    """
    Returns (p, reason) where p is the fraction of rows that 'row' is better than,
    and reason is a human-readable explanation.
    """
    # Identify goal indices
    goal_indices = [i for i, name in enumerate(header) if name.endswith(('-', '+'))]
    
    # Estimate goal values for the target row using the mean of the entire dataset
    # (Simple baseline estimation since the row has '?' goals)
    estimates = []
    for g_idx in goal_indices:
        vals = [r[g_idx] for r in rows if isinstance(r[g_idx], (int, float))]
        estimates.append(sum(vals) / len(vals) if vals else 0.0)
    
    # Define 'better' logic: 
    # A row is better if it dominates or has a higher aggregate score
    def get_score(r_goals):
        score = 0
        for i, g_idx in enumerate(goal_indices):
            val = r_goals[i]
            name = header[g_idx]
            # Normalize: 0 to 1
            col_vals = [r[g_idx] for r in rows if isinstance(r[g_idx], (int, float))]
            min_v, max_v = min(col_vals), max(col_vals)
            norm = (val - min_v) / (max_v - min_v + 1e-9)
            if name.endswith('-'):
                score += (1.0 - norm)
            else:
                score += norm
        return score / len(goal_indices)

    target_score = get_score(estimates)
    
    better_count = 0
    for r in rows:
        r_goals = [r[i] for i in goal_indices]
        if get_score(r_goals) < target_score:
            better_count += 1
            
    p = better_count / len(rows)
    reason = f"This configuration outperforms {p:.1%} of all historical setups based on its estimated multi-objective performance score."
    
    return p, reason

import random

def choose(header, rows, label):
    """
    Finds the best row by sampling 50 rows and evaluating their goals.
    Uses a simple multi-objective score to determine the 'best' row.
    """
    n = len(rows)
    # We have 50 labels available.
    budget = 50
    
    # Randomly sample indices to evaluate
    indices = random.sample(range(n), min(n, budget))
    
    best_idx = -1
    best_score = -float('inf')
    
    # Identify goal indices
    goal_indices = [i for i, name in enumerate(header) if name.endswith(('-', '+'))]
    
    # Pre-calculate min/max for normalization based on the sampled data
    # Since we don't know the full range, we use the samples as a proxy
    sampled_data = []
    for i in indices:
        goals = label(i)
        sampled_data.append((i, goals))
    
    # Calculate normalization bounds from the 50 samples
    bounds = {}
    for g_idx in goal_indices:
        vals = [s[1][goal_indices.index(g_idx)] for s in sampled_data]
        bounds[g_idx] = (min(vals), max(vals))
        
    for i, goals in sampled_data:
        score = 0
        for j, g_idx in enumerate(goal_indices):
            val = goals[j]
            min_v, max_v = bounds[g_idx]
            norm = (val - min_v) / (max_v - min_v + 1e-9)
            if header[g_idx].endswith('-'):
                score += (1.0 - norm)
            else:
                score += norm
        
        if score > best_score:
            best_score = score
            best_idx = i
            
    return best_idx

import csv

def fix(header, row, levers):
    """
    Finds the smallest change to a 'bad' row to improve its performance.
    'Smallest change' is defined as the minimum Euclidean distance in 
    normalized input space.
    """
    idx_mre = header.index('MRE-')
    idx_acc = header.index('ACC+')
    idx_pred = header.index('PRED40+')
    
    def get_score(r):
        # Simple multi-objective score: minimize MRE, maximize ACC and PRED
        # We assume numeric values for goals
        return -r[idx_mre] + r[idx_acc] + r[idx_pred]

    best_change = {}
    min_dist = float('inf')

    for col_name, allowed in levers.items():
        idx = header.index(col_name)
        original_val = row[idx]
        
        candidates = []
        if isinstance(allowed, (list, tuple)) and not isinstance(allowed[0], (int, float)):
            # Symbolic
            candidates = [v for v in allowed if v != original_val]
        else:
            # Numeric (low, high)
            low, high = allowed
            # Try a small step or boundary values
            candidates = [low, high]
            
        for val in candidates:
            new_row = list(row)
            new_row[idx] = val
            
            # Calculate distance (normalized)
            dist = 0
            if isinstance(val, (int, float)):
                dist = abs(val - original_val) / (high - low + 1e-9)
            else:
                dist = 1.0
                
            if dist < min_dist:
                min_dist = dist
                best_change = {col_name: val}
                
    return best_change

import csv

def spot(header, rows):
    """
    Identifies suspicious rows that might be 'failed runs'.
    A row is considered suspicious if its input values are extreme outliers 
    relative to the rest of the dataset, or if it contains values that 
    violate logical constraints (e.g., Min_sample_leaves > 100).
    """
    # Identify input columns (those not ending in - or +)
    input_indices = [i for i, name in enumerate(header) if not (name.endswith('-') or name.endswith('+'))]
    
    # Calculate statistics for numeric inputs to find outliers
    stats = {}
    for i in input_indices:
        col_name = header[i]
        if col_name == 'criterion':
            continue
        vals = [r[i] for r in rows if isinstance(r[i], (int, float))]
        if vals:
            mean = sum(vals) / len(vals)
            std = (sum((x - mean)**2 for x in vals) / len(vals))**0.5
            stats[i] = (mean, std)

    suspicion_scores = []
    for row_idx, row in enumerate(rows):
        score = 0
        
        # 1. Check for statistical outliers (Z-score > 3)
        for i in input_indices:
            if i in stats:
                mean, std = stats[i]
                if std > 0:
                    z = abs(row[i] - mean) / std
                    if z > 3:
                        score += z
        
        # 2. Domain-specific sanity checks
        # Example: Min_sample_leaves is usually small; large values might be errors
        idx_leaves = header.index('Min_sample_leaves')
        if row[idx_leaves] > 50:
            score += 5
            
        # Example: Max_depth is usually small
        idx_depth = header.index('Max_depth')
        if row[idx_depth] > 100:
            score += 5

        if score > 0:
            suspicion_scores.append((score, row_idx))
            
    # Sort by suspicion score descending
    suspicion_scores.sort(key=lambda x: x[0], reverse=True)
    
    return [item[1] for item in suspicion_scores]

def race(header, rows, label):

    Compares three strategies:
    1. 'random': Picks 16 random rows.
    2. 'greedy': Picks 17 rows by exploring the neighborhood of the best found so far.
    3. 'smart': Picks 17 rows by sampling and focusing on the most promising region.
    Total budget: 50 labels.
    """
    budget = 50

    goal_indices = [i for i, name in enumerate(header) if name.endswith(('-', '+'))]
    
    def get_score(goals):
        score = 0
        for i, g_idx in enumerate(goal_indices):
            val = goals[i]
            # Normalize based on a rough global estimate (0 to 1)
            # Since we don't have full data, we use a heuristic range
            if header[g_idx].endswith('-'):
                score += (1.0 - min(1.0, max(0.0, val)))
            else:
                score += min(1.0, max(0.0, val))
        return score / len(goal_indices)

                raise Exception("Budget exceeded")

        return labeled[i]

    best_r = {"idx": -1, "score": -1}

        s = get_score(get_label(i))
        if s > best_r["score"]:
            best_r = {"idx": i, "score": s}
    results["random"] = best_r

    # Strategy 2: Greedy (Local search)
    best_g = {"idx": random.randint(0, len(rows) - 1), "score": -1}
    for _ in range(17):
        s = get_score(get_label(best_g["idx"]))
        best_g["score"] = s
        # Look at neighbors

            ns = get_score(get_label(neighbor))
            if ns > best_g["score"]:
                best_g = {"idx": neighbor, "score": ns}
    results["greedy"] = best_g

    # Strategy 3: Smart (Sample and focus)
    samples = random.sample(range(len(rows)), 10)
    best_s = {"idx": -1, "score": -1}
    for i in samples:
        s = get_score(get_label(i))

        if ns > best_s["score"]:
            best_s = {"idx": neighbor, "score": ns}
    results["smart"] = best_s

    winner = max(results, key=lambda k: results[k]["score"])
    return {"winner": winner, "row": results[winner]["idx"], "tried": tried}

import csv

def remember(header, rows):
    """
    Identifies the top 5% of rows based on multi-objective optimization.
    Goals: MRE- (minimize), ACC+ (maximize), PRED40+ (maximize).
    We normalize values to [0, 1] and use a simple weighted sum.
    """
    n = len(rows)
    limit = max(1, n // 20)
    
    idx_mre = header.index('MRE-')
    idx_acc = header.index('ACC+')
    idx_pred = header.index('PRED40+')
    
    # Extract columns for normalization
    mres = [r[idx_mre] for r in rows]
    accs = [r[idx_acc] for r in rows]
    preds = [r[idx_pred] for r in rows]
    
    min_mre, max_mre = min(mres), max(mres)
    min_acc, max_acc = min(accs), max(accs)
    min_pred, max_pred = min(preds), max(preds)
    
    def normalize(val, min_v, max_v, reverse=False):
        if max_v == min_v: return 0.0
        norm = (val - min_v) / (max_v - min_v)
        return (1.0 - norm) if reverse else norm

    scores = []
    for i, row in enumerate(rows):
        # Score = (1 - MRE_norm) + ACC_norm + PRED_norm
        s = (normalize(row[idx_mre], min_mre, max_mre, True) +
             normalize(row[idx_acc], min_acc, max_acc) +
             normalize(row[idx_pred], min_pred, max_pred))
        scores.append((s, i))
    
    # Sort by score descending
    scores.sort(key=lambda x: x[0], reverse=True)
    
    return [item[1] for item in scores[:limit]]

import csv

def certify(xs, ys):
    """
    Returns True if the distributions of xs and ys are statistically 
    indistinguishable (same apart from noise), False otherwise.
    Uses the Vargha-Delaney A12 effect size test.
    """
    if not xs or not ys:
        return True
    
    n_x = len(xs)
    n_y = len(ys)
    
    greater = 0
    less = 0
    for x in xs:
        for y in ys:
            if x > y:
                greater += 1
            elif x < y:
                less += 1
                
    a12 = (greater + 0.5 * (n_x * n_y - greater - less)) / (n_x * n_y)
    
    # A12 threshold: 0.56 is the standard small effect size threshold.
    # If A12 is between 0.44 and 0.56, the difference is negligible.
    return 0.44 <= a12 <= 0.56

    c1, c2 = crit_names[0], crit_names[1]

import csv

def flow(header, row, seen):
    """
    Decides whether to stop at the current row.
    We stop if the current row is significantly better than the best row 
    seen so far, or if we have processed enough data that further 
    improvement is statistically unlikely (diminishing returns).
    """
    if not seen:
        return False

    # Identify goal indices
    goal_indices = [i for i, name in enumerate(header) if name.endswith(('-', '+'))]
    
    def get_score(r):
        score = 0
        for g_idx in goal_indices:
            # We need to normalize based on the 'seen' data
            col_vals = [s[g_idx] for s in seen]
            min_v, max_v = min(col_vals), max(col_vals)
            norm = (r[g_idx] - min_v) / (max_v - min_v + 1e-9)
            if header[g_idx].endswith('-'):
                score += (1.0 - norm)
            else:
                score += norm
        return score / len(goal_indices)

    # Calculate scores for all seen rows
    seen_scores = [get_score(s) for s in seen]
    best_seen = max(seen_scores)
    current_score = get_score(row)
    
    # Stop if current row is in the top 1% of all seen so far
    # or if we have seen enough rows (e.g., 200) and performance has plateaued
    if current_score > best_seen * 1.05:
        return True
    
    if len(seen) > 200 and current_score < best_seen * 0.95:
        return True
        
    return False
