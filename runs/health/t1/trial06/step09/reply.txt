```python
import csv

def spot(header, rows):
    """
    Identifies suspicious rows that might be 'failed runs'.
    A row is considered suspicious if its input values are extreme outliers 
    relative to the rest of the dataset, or if it contains values that 
    violate logical constraints (e.g., Min_sample_leaves > 100).
    """
    # Identify input columns (those not ending in - or +)
    input_indices = [i for i, name in enumerate(header) if not (name.endswith('-') or name.endswith('+'))]
    
    # Calculate statistics for numeric inputs to find outliers
    stats = {}
    for i in input_indices:
        col_name = header[i]
        if col_name == 'criterion':
            continue
        vals = [r[i] for r in rows if isinstance(r[i], (int, float))]
        if vals:
            mean = sum(vals) / len(vals)
            std = (sum((x - mean)**2 for x in vals) / len(vals))**0.5
            stats[i] = (mean, std)

    suspicion_scores = []
    for row_idx, row in enumerate(rows):
        score = 0
        
        # 1. Check for statistical outliers (Z-score > 3)
        for i in input_indices:
            if i in stats:
                mean, std = stats[i]
                if std > 0:
                    z = abs(row[i] - mean) / std
                    if z > 3:
                        score += z
        
        # 2. Domain-specific sanity checks
        # Example: Min_sample_leaves is usually small; large values might be errors
        idx_leaves = header.index('Min_sample_leaves')
        if row[idx_leaves] > 50:
            score += 5
            
        # Example: Max_depth is usually small
        idx_depth = header.index('Max_depth')
        if row[idx_depth] > 100:
            score += 5

        if score > 0:
            suspicion_scores.append((score, row_idx))
            
    # Sort by suspicion score descending
    suspicion_scores.sort(key=lambda x: x[0], reverse=True)
    
    return [item[1] for item in suspicion_scores]

if __name__ == "__main__":
    filename = 'Health-ClosedIssues0000.n1000.train.csv'
    try:
        with open(filename, 'r') as f:
            reader = csv.reader(f)
            header = next(reader)
            rows = []
            for row in reader:
                processed = []
                for i, val in enumerate(row):
                    if header[i] == 'criterion':
                        processed.append(val)
                    else:
                        processed.append(float(val))
                rows.append(processed)
        
        suspicious_indices = spot(header, rows)
        
        print(f"Identified {len(suspicious_indices)} potentially failed/erroneous runs.")
        print("Top 5 most suspicious row indices:")
        for idx in suspicious_indices[:5]:
            print(f"Row {idx}: {rows[idx]}")
            
    except FileNotFoundError:
        print(f"Error: {filename} not found.")
        exit(1)
```
