import csv

def remember(header, rows):
    """
    Identifies the top 5% of rows based on multi-objective optimization.
    Goals: MRE- (minimize), ACC+ (maximize), PRED40+ (maximize).
    We normalize values to [0, 1] and use a simple weighted sum (Pareto-like ranking).
    """
    n = len(rows)
    limit = max(1, n // 20)  # 5% limit
    
    # Identify column indices
    idx_mre = header.index('MRE-')
    idx_acc = header.index('ACC+')
    idx_pred = header.index('PRED40+')
    
    # Extract columns for normalization
    mres = [r[idx_mre] for r in rows]
    accs = [r[idx_acc] for r in rows]
    preds = [r[idx_pred] for r in rows]
    
    min_mre, max_mre = min(mres), max(mres)
    min_acc, max_acc = min(accs), max(accs)
    min_pred, max_pred = min(preds), max(preds)
    
    def normalize(val, min_v, max_v, reverse=False):
        if max_v == min_v: return 0.0
        norm = (val - min_v) / (max_v - min_v)
        return (1.0 - norm) if reverse else norm

    scores = []
    for i, row in enumerate(rows):
        # Score = (1 - MRE_norm) + ACC_norm + PRED_norm
        s = (normalize(row[idx_mre], min_mre, max_mre, True) +
             normalize(row[idx_acc], min_acc, max_acc) +
             normalize(row[idx_pred], min_pred, max_pred))
        scores.append((s, i))
    
    # Sort by score descending
    scores.sort(key=lambda x: x[0], reverse=True)
    
    return [item[1] for item in scores[:limit]]

if __name__ == "__main__":
    filename = 'Health-ClosedIssues0000.n1000.train.csv'
    rows = []
    try:
        with open(filename, 'r') as f:
            reader = csv.reader(f)
            header = next(reader)
            for row in reader:
                # Convert numeric columns to floats, keep symbolic as strings
                processed_row = []
                for i, val in enumerate(row):
                    if header[i] in ['criterion']:
                        processed_row.append(val)
                    else:
                        processed_row.append(float(val))
                rows.append(processed_row)
        
        best_indices = remember(header, rows)
        
        print(f"Found {len(best_indices)} top-performing setups.")
        print("Indices of the most trustworthy setups:")
        print(best_indices)
        print("\nTop 3 examples:")
        for i in best_indices[:3]:
            print(f"Row {i}: {rows[i]}")
            
    except FileNotFoundError:
        print(f"Error: {filename} not found.")
        exit(1)
