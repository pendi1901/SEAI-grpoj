```python
import csv
import random

def choose(header, rows, label):
    """
    Finds the best row by sampling 50 rows and evaluating their goals.
    Uses a simple multi-objective score to determine the 'best' row.
    """
    n = len(rows)
    # We have 50 labels available.
    budget = 50
    
    # Randomly sample indices to evaluate
    indices = random.sample(range(n), min(n, budget))
    
    best_idx = -1
    best_score = -float('inf')
    
    # Identify goal indices
    goal_indices = [i for i, name in enumerate(header) if name.endswith(('-', '+'))]
    
    # Pre-calculate min/max for normalization based on the sampled data
    # Since we don't know the full range, we use the samples as a proxy
    sampled_data = []
    for i in indices:
        goals = label(i)
        sampled_data.append((i, goals))
    
    # Calculate normalization bounds from the 50 samples
    bounds = {}
    for g_idx in goal_indices:
        vals = [s[1][goal_indices.index(g_idx)] for s in sampled_data]
        bounds[g_idx] = (min(vals), max(vals))
        
    for i, goals in sampled_data:
        score = 0
        for j, g_idx in enumerate(goal_indices):
            val = goals[j]
            min_v, max_v = bounds[g_idx]
            norm = (val - min_v) / (max_v - min_v + 1e-9)
            if header[g_idx].endswith('-'):
                score += (1.0 - norm)
            else:
                score += norm
        
        if score > best_score:
            best_score = score
            best_idx = i
            
    return best_idx

if __name__ == "__main__":
    filename = 'Health-ClosedIssues0000.n1000.train.csv'
    try:
        with open(filename, 'r') as f:
            reader = csv.reader(f)
            header = next(reader)
            rows = []
            true_goals = []
            for row in reader:
                processed = []
                goals = []
                for i, val in enumerate(row):
                    if header[i].endswith(('-', '+')):
                        goals.append(float(val))
                        processed.append("?")
                    elif header[i] == 'criterion':
                        processed.append(val)
                    else:
                        processed.append(float(val))
                rows.append(processed)
                true_goals.append(goals)
        
        def label(i):
            return true_goals[i]
            
        best_idx = choose(header, rows, label)
        print(f"The best configuration found is at index {best_idx}.")
        print(f"Row data: {rows[best_idx]}")
        print(f"True goals: {true_goals[best_idx]}")
        
    except FileNotFoundError:
        print(f"Error: {filename} not found.")
        exit(1)
```
