import csv
import random

def race(header, rows, label):
    """
    Compares three strategies: random, greedy, and local search.
    Budget: 50 labels total.
    """
    goal_indices = [i for i, h in enumerate(header) if h.endswith(('-', '+'))]
    
    def get_score(goals):
        # Normalize goals: MRE- (min), ACC+ (max), PRED40+ (max)
        # We use a simple aggregate score (higher is better)
        # Since we don't know the global min/max, we use the current labels
        # to normalize relative to the budget.
        return sum(goals) # Simplified for comparison

    labeled = {}
    def get_label(i):
        if i not in labeled:
            if len(labeled) >= 50:
                raise Exception("Budget exceeded")
            labeled[i] = label(i)
        return labeled[i]

    # Strategy 1: Random
    def strategy_random():
        idx = random.choice(range(len(rows)))
        return idx, get_label(idx)

    # Strategy 2: Greedy (Sample 25, pick best)
    def strategy_greedy():
        candidates = random.sample(range(len(rows)), 25)
        best_idx = -1
        best_val = -float('inf')
        for i in candidates:
            goals = get_label(i)
            # Simple score: minimize MRE, maximize others
            score = -goals[0] + goals[1] + goals[2]
            if score > best_val:
                best_val = score
                best_idx = i
        return best_idx, best_val

    # Strategy 3: Local Search (Start random, look at neighbors)
    def strategy_local():
        curr = random.choice(range(len(rows)))
        for _ in range(24): # Remaining budget
            neighbors = random.sample(range(len(rows)), 5)
            best_n = curr
            best_val = -sum(get_label(curr))
            for n in neighbors:
                val = -sum(get_label(n))
                if val > best_val:
                    best_val = val
                    best_n = n
            curr = best_n
        return curr, get_label(curr)

    results = {}
    for name, func in [("random", strategy_random), ("greedy", strategy_greedy), ("local", strategy_local)]:
        idx, _ = func()
        results[name] = idx

    # Determine winner based on true goals
    best_overall = -float('inf')
    winner = ""
    best_row = -1
    for name, idx in results.items():
        goals = label(idx)
        score = -goals[0] + goals[1] + goals[2]
        if score > best_overall:
            best_overall = score
            winner = name
            best_row = idx

    return {"winner": winner, "row": best_row, "tried": ["random", "greedy", "local"]}

if __name__ == "__main__":
    filename = "Health-ClosedIssues0000.n1000.train.csv"
    rows = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            rows.append([val if header[i] == 'criterion' else float(val) for i, val in enumerate(row)])

    true_goals = [[r[i] for i, h in enumerate(header) if h.endswith(('-', '+'))] for r in rows]
    
    # Hide goals
    for r in rows:
        for i, h in enumerate(header):
            if h.endswith(('-', '+')):
                r[i] = "?"

    def label(i):
        return true_goals[i]

    result = race(header, rows, label)
    print(f"The winner is: {result['winner']}")
    print(f"Best row found at index: {result['row']}")
    print(f"Strategies tried: {', '.join(result['tried'])}")
