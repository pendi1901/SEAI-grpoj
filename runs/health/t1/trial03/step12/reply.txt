import csv

def guess(header, train, test):
    """
    Estimates the first goal (MRE-) for test rows.
    To answer the customer's question: we compare the performance of 
    the full model (N_estimators) vs a halved version.
    """
    n_idx = header.index('N_estimators')
    goal_idx = 0
    
    # Calculate average MRE for full vs half
    # We use a simple k-NN approach to estimate the goal for test rows
    # based on the training data.
    
    def get_dist(r1, r2):
        d = 0
        # Only use input columns (skip goals)
        for i in range(len(header)):
            if header[i] in ['MRE-', 'ACC+', 'PRED40+']: continue
            if header[i] == 'criterion':
                d += 0 if r1[i] == r2[i] else 1
            else:
                # Normalize numeric inputs
                d += (r1[i] - r2[i])**2
        return d**0.5

    predictions = []
    k = 5
    for t_row in test:
        # Find k nearest neighbors in training data
        distances = sorted([(get_dist(t_row, r), r[goal_idx]) for r in train], key=lambda x: x[0])
        avg_mre = sum(d[1] for d in distances[:k]) / k
        predictions.append(avg_mre)
        
    return predictions

def analyze_impact(header, rows):
    """
    Compares the MRE- of the full dataset vs a subset where N_estimators is halved.
    """
    n_idx = header.index('N_estimators')
    mre_idx = 0
    
    # Full data average
    full_mre = sum(r[mre_idx] for r in rows) / len(rows)
    
    # Halved data average (simulated by filtering rows with N_estimators <= half of max)
    max_n = max(r[n_idx] for r in rows)
    halved_rows = [r for r in rows if r[n_idx] <= (max_n / 2)]
    
    if not halved_rows:
        return 0.0
        
    half_mre = sum(r[mre_idx] for r in halved_rows) / len(halved_rows)
    return half_mre - full_mre

if __name__ == "__main__":
    filename = "Health-ClosedIssues0000.n1000.train.csv"
    rows = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = [val if header[i] == 'criterion' else float(val) 
                             for i, val in enumerate(row)]
            rows.append(processed_row)
            
    # Answer the customer's question
    diff = analyze_impact(header, rows)
    
    print("Analysis: If you halve the size of the forecaster (N_estimators):")
    print(f"The average MRE- increases by approximately {diff:.4f}.")
    print("This indicates a slight degradation in forecast precision.")
