import csv

def remember(header, rows):
    """
    Identifies the top 5% of rows based on multi-objective optimization.
    Goals: MRE- (minimize), ACC+ (maximize), PRED40+ (maximize).
    We use a simple Pareto-like ranking: normalize values and sum scores.
    """
    n = len(rows)
    limit = max(1, n // 20)  # 5% of rows, at least 1
    
    # Identify column indices
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')
    
    # Extract columns for normalization
    mres = [r[mre_idx] for r in rows]
    accs = [r[acc_idx] for r in rows]
    preds = [r[pred_idx] for r in rows]
    
    min_mre, max_mre = min(mres), max(mres)
    min_acc, max_acc = min(accs), max(accs)
    min_pred, max_pred = min(preds), max(preds)
    
    def get_score(row):
        # Normalize: (val - min) / (max - min)
        # For MRE-, we want low values, so we invert: (max - val) / (max - min)
        s_mre = (max_mre - row[mre_idx]) / (max_mre - min_mre) if max_mre > min_mre else 0
        s_acc = (row[acc_idx] - min_acc) / (max_acc - min_acc) if max_acc > min_acc else 0
        s_pred = (row[pred_idx] - min_pred) / (max_pred - min_pred) if max_pred > min_pred else 0
        return s_mre + s_acc + s_pred

    # Calculate scores and keep track of original indices
    scored = []
    for i, row in enumerate(rows):
        scored.append((get_score(row), i))
    
    # Sort by score descending
    scored.sort(key=lambda x: x[0], reverse=True)
    
    # Return the indices of the top 5%
    return [item[1] for item in scored[:limit]]

import csv

def guess(header, train, test):
    """
    Estimates the first goal column (MRE-) for test rows using a 
    k-Nearest Neighbors approach based on Euclidean distance of 
    input features.
    """
    # Identify goal and input columns
    goal_idx = 0
    input_indices = [i for i, h in enumerate(header) if h not in ['MRE-', 'ACC+', 'PRED40+']]
    
    # Normalize numeric inputs for distance calculation
    # Only consider columns that are numeric (not 'criterion')
    numeric_inputs = [i for i in input_indices if header[i] != 'criterion']
    
    # Pre-calculate min/max for normalization
    mins = [min(row[i] for row in train) for i in numeric_inputs]
    maxs = [max(row[i] for row in train) for i in numeric_inputs]
    
    def normalize(val, i):
        idx = numeric_inputs.index(i)
        if maxs[idx] == mins[idx]: return 0
        return (val - mins[idx]) / (maxs[idx] - mins[idx])

    def distance(row1, row2):
        d = 0
        for i in input_indices:
            if header[i] == 'criterion':
                d += 0 if row1[i] == row2[i] else 1
            else:
                d += (normalize(row1[i], i) - normalize(row2[i], i))**2
        return d**0.5

    predictions = []
    k = 5  # Number of neighbors
    for t_row in test:
        # Find k nearest neighbors
        distances = []
        for train_row in train:
            distances.append((distance(t_row, train_row), train_row[goal_idx]))
        distances.sort(key=lambda x: x[0])
        
        # Average the goal value of the k nearest neighbors
        avg_mre = sum(d[1] for d in distances[:k]) / k
        predictions.append(avg_mre)
        
    return predictions

import csv

def certify(xs, ys):
    """
    Returns True if the distributions of xs and ys are statistically 
    indistinguishable (noise), False if they differ significantly.
    Uses a simple A12 effect size test (non-parametric).
    """
    if not xs or not ys:
        return True
    
    def count_greater(list1, list2):
        count = 0
        for x in list1:
            for y in list2:
                if x > y:
                    count += 1
                elif x == y:
                    count += 0.5
        return count

    n, m = len(xs), len(ys)
    a12 = count_greater(xs, ys) / (n * m)
    
    # A12 threshold: 0.56 is a small effect size, 
    # 0.64 is medium, 0.71 is large.
    # If A12 is close to 0.5, they are likely the same (noise).
    return abs(a12 - 0.5) < 0.14

def analyze_correlation(header, rows):
    """
    Compares 'N_estimators' (size) against 'ACC+' (accuracy).
    Splits data into small and large setups and compares accuracy.
    """
    n_idx = header.index('N_estimators')
    acc_idx = header.index('ACC+')
    
    vals = [r[n_idx] for r in rows]
    median = sum(vals) / len(vals)
    
    small = [r[acc_idx] for r in rows if r[n_idx] <= median]
    large = [r[acc_idx] for r in rows if r[n_idx] > median]
    
    is_noise = certify(small, large)
    
    print(f"Analysis of setup size vs accuracy:")
    print(f"Small setups (N <= {median:.1f}) mean ACC: {sum(small)/len(small):.4f}")
    print(f"Large setups (N > {median:.1f}) mean ACC: {sum(large)/len(large):.4f}")
    
    if is_noise:
        print("Conclusion: The difference is likely just noise.")
    else:
        print("Conclusion: The difference is statistically significant.")
    return is_noise

import csv

def flow(header, row, seen):
    """
    Decides whether to accept the current row based on a Pareto-like 
    threshold derived from previously seen data.
    """
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    # If this is the first row, we have no baseline, so keep looking
    if not seen:
        return False

    # Calculate current row's normalized score relative to history
    mres = [r[mre_idx] for r in seen]
    accs = [r[acc_idx] for r in seen]
    preds = [r[pred_idx] for r in seen]

    min_mre, max_mre = min(mres), max(mres)
    min_acc, max_acc = min(accs), max(accs)
    min_pred, max_pred = min(preds), max(preds)

    def get_score(r):
        s_mre = (max_mre - r[mre_idx]) / (max_mre - min_mre) if max_mre > min_mre else 0
        s_acc = (r[acc_idx] - min_acc) / (max_acc - min_acc) if max_acc > min_acc else 0
        s_pred = (r[pred_idx] - min_pred) / (max_pred - min_pred) if max_pred > min_pred else 0
        return s_mre + s_acc + s_pred

    # Threshold: Accept if the current row is in the top 5% of observed scores
    scores = [get_score(r) for r in seen]
    current_score = get_score(row)
    
    # We define "good enough" as being better than the 95th percentile of seen data
    scores.sort(reverse=True)
    limit = max(1, len(scores) // 20)
    threshold = scores[limit - 1]

    return current_score >= threshold

import csv

def blame(header, rows):
    """
    Identifies which input columns correlate most strongly with poor performance.
    We define 'bad' rows as those in the bottom 25% of a multi-objective score.
    We then calculate the correlation between input settings and the score.
    """
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')
    
    input_cols = [i for i, h in enumerate(header) if h not in ['MRE-', 'ACC+', 'PRED40+']]
    
    # Calculate a score for each row (higher is better)
    mres = [r[mre_idx] for r in rows]
    accs = [r[acc_idx] for r in rows]
    preds = [r[pred_idx] for r in rows]
    
    min_mre, max_mre = min(mres), max(mres)
    min_acc, max_acc = min(accs), max(accs)
    min_pred, max_pred = min(preds), max(preds)
    
    def get_score(r):
        s_mre = (max_mre - r[mre_idx]) / (max_mre - min_mre) if max_mre > min_mre else 0
        s_acc = (r[acc_idx] - min_acc) / (max_acc - min_acc) if max_acc > min_acc else 0
        s_pred = (r[pred_idx] - min_pred) / (max_pred - min_pred) if max_pred > min_pred else 0
        return s_mre + s_acc + s_pred

    scores = [get_score(r) for r in rows]
    
    # Identify the 'bad' rows (bottom 25%)
    threshold = sorted(scores)[len(scores) // 4]
    bad_rows = [rows[i] for i, s in enumerate(scores) if s <= threshold]
    
    # For each input column, calculate how much its values in bad rows 
    # deviate from the global average.
    damages = []
    for i in input_cols:
        col_vals = [r[i] for r in rows if isinstance(r[i], (int, float))]
        if not col_vals: continue
        
        avg = sum(col_vals) / len(col_vals)
        bad_vals = [r[i] for r in bad_rows if isinstance(r[i], (int, float))]
        if not bad_vals: continue
        
        bad_avg = sum(bad_vals) / len(bad_vals)
        # Damage is the absolute difference between bad row average and global average
        damages.append((abs(bad_avg - avg), header[i]))
        
    damages.sort(reverse=True)
    return [d[1] for d in damages[:3]]

import csv

def justify(header, rows, row):
    """
    Estimates the performance of 'row' by comparing it to its k-nearest neighbors
    in 'rows', then calculates the fraction of 'rows' that are worse than 'row'.
    """
    # Identify indices
    goal_indices = [i for i, h in enumerate(header) if h.endswith(('-', '+'))]
    input_indices = [i for i, h in enumerate(header) if not h.endswith(('-', '+'))]
    
    # Normalize numeric inputs for distance
    numeric_inputs = [i for i in input_indices if header[i] != 'criterion']
    mins = [min(r[i] for r in rows) for i in numeric_inputs]
    maxs = [max(r[i] for r in rows) for i in numeric_inputs]
    
    def get_dist(r1, r2):
        d = 0
        for idx, i in enumerate(numeric_inputs):
            rng = maxs[idx] - mins[idx]
            if rng > 0:
                d += ((r1[i] - r2[i]) / rng) ** 2
        for i in input_indices:
            if header[i] == 'criterion' and r1[i] != r2[i]:
                d += 1
        return d**0.5

    # Estimate goal values for the target row using k-NN
    k = 10
    distances = sorted([(get_dist(row, r), r) for r in rows], key=lambda x: x[0])
    neighbors = [d[1] for d in distances[:k]]
    
    estimated_goals = []
    for g_idx in goal_indices:
        estimated_goals.append(sum(n[g_idx] for n in neighbors) / k)
    
    # Compare estimated row against all rows
    def is_better(r_goals, target_goals):
        # A row is better if it dominates or has a better aggregate score
        # Here we use a simple normalized score: lower MRE is better, higher others are better
        def score(goals):
            s = 0
            for i, g_idx in enumerate(goal_indices):
                col = header[g_idx]
                vals = [r[g_idx] for r in rows]
                mn, mx = min(vals), max(vals)
                norm = (goals[i] - mn) / (mx - mn) if mx > mn else 0
                s += (1 - norm) if col.endswith('-') else norm
            return s
        return score(target_goals) > score(r_goals)

    better_count = sum(1 for r in rows if is_better(r, estimated_goals))
    p = better_count / len(rows)
    
    reason = f"This setup outperforms {p:.1%} of all historical configurations based on its estimated multi-objective performance."
    return p, reason

    labeled_data = []

        labeled_data.append((goals, i))

    # Goals: MRE- (min), ACC+ (max), PRED40+ (max)
    # Normalize each goal column based on the 50 samples

        score = 0

    for goals, original_idx in labeled_data:
        s = get_score(goals)
        if s > best_score:
            best_score = s
            best_idx = original_idx

import csv

def fix(header, row, levers):
    """
    Finds the smallest change to a 'bad' row's settings that improves its 
    multi-objective score. We define 'smallest change' as the minimum 
    Euclidean distance in the normalized lever space.
    """
    goal_indices = [i for i, h in enumerate(header) if h.endswith(('-', '+'))]
    lever_indices = {header.index(name): name for name in levers}
    
    # Simple heuristic: try small perturbations for numeric, 
    # and all possible values for symbolic.
    recommendations = {}
    
    for idx, name in lever_indices.items():
        allowed = levers[name]
        if isinstance(allowed, (list, tuple)) and not isinstance(allowed[0], (int, float)):
            # Symbolic: try all other options
            for val in allowed:
                if val != row[idx]:
                    recommendations[name] = val
                    break
        else:
            # Numeric: try moving towards the midpoint of the range
            low, high = allowed
            mid = (low + high) / 2
            if row[idx] < mid:
                recommendations[name] = min(high, row[idx] + (high - low) * 0.1)
            else:
                recommendations[name] = max(low, row[idx] - (high - low) * 0.1)
                
    return recommendations

import csv

def spot(header, rows):
    """
    Identifies rows that look like 'failed runs'.
    A failed run is defined here as a row where the input parameters are 
    at extreme, potentially invalid boundaries, or where the goal values 
    are inconsistent with the input settings (outliers).
    """
    # Identify indices
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')
    
    # We look for rows that are statistical outliers in the goal space
    # relative to their input settings.
    # A simple approach: calculate the Z-score of the goals.
    # Rows with very high MRE and very low ACC/PRED are suspicious.
    
    def get_score(row):
        # Normalize goals: MRE- (min), ACC+ (max), PRED40+ (max)
        # We use a simple aggregate: (MRE_norm + (1-ACC_norm) + (1-PRED_norm))
        # Higher score = more suspicious
        return row[mre_idx] - row[acc_idx] - row[pred_idx]

    scores = []
    for i, row in enumerate(rows):
        scores.append((get_score(row), i))
    
    # Sort by score descending (most suspicious first)
    scores.sort(key=lambda x: x[0], reverse=True)
    
    # Return the indices of the top 5% most suspicious rows
    limit = max(1, len(rows) // 20)
    return [item[1] for item in scores[:limit]]

import csv

def remember(header, rows):
    """
    Finds the 5% of rows that are most similar to the 'best' performing rows
    in the dataset. We define 'best' as those with the highest aggregate 
    normalized score across the goal columns.
    """
    goal_indices = [i for i, h in enumerate(header) if h.endswith(('-', '+'))]
    input_indices = [i for i, h in enumerate(header) if not h.endswith(('-', '+'))]
    
    # 1. Calculate scores for all rows to identify the "gold standard"
    def get_score(row):
        s = 0
        for i in goal_indices:
            col = header[i]
            vals = [r[i] for r in rows if isinstance(r[i], (int, float))]
            if not vals: continue
            mn, mx = min(vals), max(vals)
            norm = (row[i] - mn) / (mx - mn) if mx > mn else 0
            s += (1 - norm) if col.endswith('-') else norm
        return s

    scores = [(get_score(r), i) for i, r in enumerate(rows)]
    # Top 5% of rows are our "target" setups
    limit = max(1, len(rows) // 20)
    best_rows = sorted(scores, key=lambda x: x[0], reverse=True)[:limit]
    best_indices = [x[1] for x in best_rows]

    # 2. Find rows most similar to these best setups using Euclidean distance
    # on input features.
    def get_dist(r1, r2):
        d = 0
        for i in input_indices:
            if header[i] == 'criterion':
                d += 0 if r1[i] == r2[i] else 1
            else:
                # Simple normalization for distance
                vals = [r[i] for r in rows if isinstance(r[i], (int, float))]
                mn, mx = min(vals), max(vals)
                if mx > mn:
                    d += ((r1[i] - r2[i]) / (mx - mn))**2
        return d**0.5

    # Find rows that are not in the top 5% but are closest to them
    candidates = []
    for i, row in enumerate(rows):
        if i in best_indices: continue
        # Distance to the nearest 'best' row
        min_d = min(get_dist(row, rows[b]) for b in best_indices)
        candidates.append((min_d, i))
    
    candidates.sort(key=lambda x: x[0])
    
    # Return the indices of the best rows + the closest neighbors to fill the 5%
    result = best_indices + [x[1] for x in candidates[:limit]]
    return result[:limit]
