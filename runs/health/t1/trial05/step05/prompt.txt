import csv

def remember(header, rows):
    """
    Identifies the top 5% of rows based on multi-objective optimization.
    Goals: MRE- (minimize), ACC+ (maximize), PRED40+ (maximize).
    We normalize values to [0, 1] and use a simple weighted sum (or Pareto-like ranking).
    """
    n = len(rows)
    if n == 0:
        return []

    # Identify goal indices
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    # Collect values for normalization
    mres = [r[mre_idx] for r in rows]
    accs = [r[acc_idx] for r in rows]
    preds = [r[pred_idx] for r in rows]

    min_mre, max_mre = min(mres), max(mres)
    min_acc, max_acc = min(accs), max(accs)
    min_pred, max_pred = min(preds), max(preds)

    def normalize(val, min_v, max_v, reverse=False):
        if max_v == min_v: return 0.0
        norm = (val - min_v) / (max_v - min_v)
        return 1.0 - norm if reverse else norm

    scores = []
    for i, row in enumerate(rows):
        # Calculate score: lower MRE is better, higher ACC/PRED is better
        s1 = normalize(row[mre_idx], min_mre, max_mre, reverse=True)
        s2 = normalize(row[acc_idx], min_acc, max_acc)
        s3 = normalize(row[pred_idx], min_pred, max_pred)
        # Average the normalized objectives
        scores.append((s1 + s2 + s3) / 3.0)

    # Get top 5% indices
    limit = max(1, int(n * 0.05))
    indexed_scores = list(enumerate(scores))
    indexed_scores.sort(key=lambda x: x[1], reverse=True)
    
    return [x[0] for x in indexed_scores[:limit]]

import csv

def guess(header, train, test):
    """
    Estimates the first goal column (MRE-) for test rows using a 
    k-Nearest Neighbors approach based on Euclidean distance of 
    numeric input features.
    """
    mre_idx = 0
    # Identify input columns (those that are not goals)
    input_indices = [i for i, h in enumerate(header) if not (h.endswith('-') or h.endswith('+'))]
    
    # Normalize numeric inputs for distance calculation
    # We only care about numeric inputs for distance
    numeric_inputs = [i for i in input_indices if header[i] not in ['criterion']]
    
    # Precompute min/max for normalization
    mins = []
    maxs = []
    for i in numeric_inputs:
        vals = [r[i] for r in train]
        mins.append(min(vals))
        maxs.append(max(vals))
    
    def get_dist(row1, row2):
        d2 = 0
        for idx, i in enumerate(numeric_inputs):
            if maxs[idx] > mins[idx]:
                d2 += ((row1[i] - row2[i]) / (maxs[idx] - mins[idx]))**2
        # Add penalty for different symbolic values
        crit_idx = header.index('criterion')
        if row1[crit_idx] != row2[crit_idx]:
            d2 += 1.0
        return d2**0.5

    predictions = []
    k = 5
    for t_row in test:
        # Find k nearest neighbors in training set
        distances = []
        for train_row in train:
            distances.append((get_dist(t_row, train_row), train_row[mre_idx]))
        
        distances.sort(key=lambda x: x[0])
        # Average the MRE of the k nearest neighbors
        avg_mre = sum(d[1] for d in distances[:k]) / k
        predictions.append(avg_mre)
        
    return predictions

import csv

def certify(xs, ys):
    """
    Returns True if the distributions of xs and ys are statistically 
    indistinguishable (the difference is noise), False otherwise.
    Uses a simple A12 effect size test (Vargha-Delaney).
    """
    n_x = len(xs)
    n_y = len(ys)
    if n_x == 0 or n_y == 0:
        return True

    # Count how many times x > y, x < y, and x == y
    greater = 0
    less = 0
    for x in xs:
        for y in ys:
            if x > y:
                greater += 1
            elif x < y:
                less += 1
    
    # A12 = (greater + 0.5 * equal) / (n_x * n_y)
    equal = (n_x * n_y) - greater - less
    a12 = (greater + 0.5 * equal) / (n_x * n_y)
    
    # If A12 is close to 0.5, the difference is likely noise.
    # Thresholds for small/medium/large effect sizes are typically 0.56, 0.64, 0.71.
    # We return False if the effect size is significant (e.g., > 0.6 or < 0.4).
    return 0.4 <= a12 <= 0.6

import csv

def flow(header, row, seen):
    """
    Decides whether to stop at the current row based on a simple thresholding 
    strategy. We define 'good enough' as being in the top 5% of the 
    distribution of the goals seen so far.
    """
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    # If this is the first row, we can't evaluate it against others yet
    if not seen:
        return False

    # Calculate normalized score for the current row
    # We need the range of values seen so far to normalize
    all_rows = seen + [row]
    mres = [r[mre_idx] for r in all_rows]
    accs = [r[acc_idx] for r in all_rows]
    preds = [r[pred_idx] for r in all_rows]

    def get_norm(val, vals, reverse=False):
        min_v, max_v = min(vals), max(vals)
        if max_v == min_v: return 0.0
        norm = (val - min_v) / (max_v - min_v)
        return 1.0 - norm if reverse else norm

    s1 = get_norm(row[mre_idx], mres, reverse=True)
    s2 = get_norm(row[acc_idx], accs)
    s3 = get_norm(row[pred_idx], preds)
    current_score = (s1 + s2 + s3) / 3.0

    # Calculate scores for all seen rows
    seen_scores = []
    for r in seen:
        ss1 = get_norm(r[mre_idx], mres, reverse=True)
        ss2 = get_norm(r[acc_idx], accs)
        ss3 = get_norm(r[pred_idx], preds)
        seen_scores.append((ss1 + ss2 + ss3) / 3.0)

    # If current row is better than 95% of what we've seen, stop
    threshold = 0.95
    better_than = sum(1 for s in seen_scores if current_score > s)
    if len(seen) > 20 and (better_than / len(seen)) >= threshold:
        return True
    
    return False
