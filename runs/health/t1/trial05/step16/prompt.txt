import csv

def remember(header, rows):
    """
    Identifies the top 5% of rows based on multi-objective optimization.
    Goals: MRE- (minimize), ACC+ (maximize), PRED40+ (maximize).
    We normalize values to [0, 1] and use a simple weighted sum (or Pareto-like ranking).
    """
    n = len(rows)
    if n == 0:
        return []

    # Identify goal indices
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    # Collect values for normalization
    mres = [r[mre_idx] for r in rows]
    accs = [r[acc_idx] for r in rows]
    preds = [r[pred_idx] for r in rows]

    min_mre, max_mre = min(mres), max(mres)
    min_acc, max_acc = min(accs), max(accs)
    min_pred, max_pred = min(preds), max(preds)

    def normalize(val, min_v, max_v, reverse=False):
        if max_v == min_v: return 0.0
        norm = (val - min_v) / (max_v - min_v)
        return 1.0 - norm if reverse else norm

    scores = []
    for i, row in enumerate(rows):
        # Calculate score: lower MRE is better, higher ACC/PRED is better
        s1 = normalize(row[mre_idx], min_mre, max_mre, reverse=True)
        s2 = normalize(row[acc_idx], min_acc, max_acc)
        s3 = normalize(row[pred_idx], min_pred, max_pred)
        # Average the normalized objectives
        scores.append((s1 + s2 + s3) / 3.0)

    # Get top 5% indices
    limit = max(1, int(n * 0.05))
    indexed_scores = list(enumerate(scores))
    indexed_scores.sort(key=lambda x: x[1], reverse=True)
    
    return [x[0] for x in indexed_scores[:limit]]

import csv

def guess(header, train, test):
    """
    Estimates the first goal column (MRE-) for test rows using a 
    k-Nearest Neighbors approach based on Euclidean distance of 
    numeric input features.
    """
    mre_idx = 0
    # Identify input columns (those that are not goals)
    input_indices = [i for i, h in enumerate(header) if not (h.endswith('-') or h.endswith('+'))]
    
    # Normalize numeric inputs for distance calculation
    # We only care about numeric inputs for distance
    numeric_inputs = [i for i in input_indices if header[i] not in ['criterion']]
    
    # Precompute min/max for normalization
    mins = []
    maxs = []
    for i in numeric_inputs:
        vals = [r[i] for r in train]
        mins.append(min(vals))
        maxs.append(max(vals))
    
    def get_dist(row1, row2):
        d2 = 0
        for idx, i in enumerate(numeric_inputs):
            if maxs[idx] > mins[idx]:
                d2 += ((row1[i] - row2[i]) / (maxs[idx] - mins[idx]))**2
        # Add penalty for different symbolic values
        crit_idx = header.index('criterion')
        if row1[crit_idx] != row2[crit_idx]:
            d2 += 1.0
        return d2**0.5

    predictions = []
    k = 5
    for t_row in test:
        # Find k nearest neighbors in training set
        distances = []
        for train_row in train:
            distances.append((get_dist(t_row, train_row), train_row[mre_idx]))
        
        distances.sort(key=lambda x: x[0])
        # Average the MRE of the k nearest neighbors
        avg_mre = sum(d[1] for d in distances[:k]) / k
        predictions.append(avg_mre)
        
    return predictions

import csv

def certify(xs, ys):
    """
    Returns True if the distributions of xs and ys are statistically 
    indistinguishable (the difference is noise), False otherwise.
    Uses a simple A12 effect size test (Vargha-Delaney).
    """
    n_x = len(xs)
    n_y = len(ys)
    if n_x == 0 or n_y == 0:
        return True

    # Count how many times x > y, x < y, and x == y
    greater = 0
    less = 0
    for x in xs:
        for y in ys:
            if x > y:
                greater += 1
            elif x < y:
                less += 1
    
    # A12 = (greater + 0.5 * equal) / (n_x * n_y)
    equal = (n_x * n_y) - greater - less
    a12 = (greater + 0.5 * equal) / (n_x * n_y)
    
    # If A12 is close to 0.5, the difference is likely noise.
    # Thresholds for small/medium/large effect sizes are typically 0.56, 0.64, 0.71.
    # We return False if the effect size is significant (e.g., > 0.6 or < 0.4).
    return 0.4 <= a12 <= 0.6

import csv

def flow(header, row, seen):
    """
    Decides whether to stop at the current row based on a simple thresholding 
    strategy. We define 'good enough' as being in the top 5% of the 
    distribution of the goals seen so far.
    """
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    # If this is the first row, we can't evaluate it against others yet
    if not seen:
        return False

    # Calculate normalized score for the current row
    # We need the range of values seen so far to normalize
    all_rows = seen + [row]
    mres = [r[mre_idx] for r in all_rows]
    accs = [r[acc_idx] for r in all_rows]
    preds = [r[pred_idx] for r in all_rows]

    def get_norm(val, vals, reverse=False):
        min_v, max_v = min(vals), max(vals)
        if max_v == min_v: return 0.0
        norm = (val - min_v) / (max_v - min_v)
        return 1.0 - norm if reverse else norm

    s1 = get_norm(row[mre_idx], mres, reverse=True)
    s2 = get_norm(row[acc_idx], accs)
    s3 = get_norm(row[pred_idx], preds)
    current_score = (s1 + s2 + s3) / 3.0

    # Calculate scores for all seen rows
    seen_scores = []
    for r in seen:
        ss1 = get_norm(r[mre_idx], mres, reverse=True)
        ss2 = get_norm(r[acc_idx], accs)
        ss3 = get_norm(r[pred_idx], preds)
        seen_scores.append((ss1 + ss2 + ss3) / 3.0)

    # If current row is better than 95% of what we've seen, stop
    threshold = 0.95
    better_than = sum(1 for s in seen_scores if current_score > s)
    if len(seen) > 20 and (better_than / len(seen)) >= threshold:
        return True
    
    return False

import csv

def blame(header, rows):
    """
    Identifies input columns that correlate most strongly with poor performance.
    'Bad' rows are defined as the bottom 25% based on the multi-objective score.
    We compare the distribution of each input in the 'bad' set vs the 'good' set.
    """
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')
    
    input_indices = [i for i, h in enumerate(header) if not (h.endswith('-') or h.endswith('+'))]

    # Calculate scores to rank rows
    def get_score(row):
        # Simple normalization based on global min/max for this dataset
        # MRE: 0.0 to 1.0, ACC: 0.0 to 1.0, PRED: 0.0 to 1.0
        s1 = 1.0 - row[mre_idx]
        s2 = row[acc_idx]
        s3 = row[pred_idx]
        return (s1 + s2 + s3) / 3.0

    scored_rows = sorted([(get_score(r), r) for r in rows], key=lambda x: x[0])
    
    # Bottom 25% are 'bad', top 25% are 'good'
    n = len(scored_rows)
    bad = [x[1] for x in scored_rows[:n//4]]
    good = [x[1] for x in scored_rows[3*n//4:]]

    # For each input, check if the mean/distribution in 'bad' is significantly different from 'good'
    # We use a simple difference of means (normalized) as a proxy for 'blame'
    blame_scores = []
    for i in input_indices:
        col_name = header[i]
        if col_name == 'criterion':
            # For symbolic, check frequency of most common bad value
            bad_vals = [r[i] for r in bad]
            good_vals = [r[i] for r in good]
            # Score based on how much more frequent a value is in bad vs good
            score = 0
            for val in set(bad_vals):
                score += abs(bad_vals.count(val)/len(bad_vals) - good_vals.count(val)/len(good_vals))
            blame_scores.append((col_name, score))
        else:
            # For numeric, check difference in means
            bad_vals = [r[i] for r in bad]
            good_vals = [r[i] for r in good]
            score = abs((sum(bad_vals)/len(bad_vals)) - (sum(good_vals)/len(good_vals)))
            blame_scores.append((col_name, score))

    # Sort by score descending
    blame_scores.sort(key=lambda x: x[1], reverse=True)
    return [x[0] for x in blame_scores[:3]]

import csv

def justify(header, rows, row):
    """
    Estimates the performance of 'row' by comparing it to the k-nearest neighbors
    in 'rows', then calculates the fraction of 'rows' that are worse than this estimate.
    """
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')
    
    input_indices = [i for i, h in enumerate(header) if not (h.endswith('-') or h.endswith('+'))]
    numeric_inputs = [i for i in input_indices if header[i] != 'criterion']
    crit_idx = header.index('criterion')

    # Normalize inputs for distance
    mins = [min(r[i] for r in rows) for i in numeric_inputs]
    maxs = [max(r[i] for r in rows) for i in numeric_inputs]

    def get_dist(r1, r2):
        d2 = 0
        for idx, i in enumerate(numeric_inputs):
            if maxs[idx] > mins[idx]:
                d2 += ((r1[i] - r2[i]) / (maxs[idx] - mins[idx]))**2
        if r1[crit_idx] != r2[crit_idx]:
            d2 += 1.0
        return d2**0.5

    # Estimate goals for the target row using k-NN
    k = 10
    neighbors = sorted([(get_dist(row, r), r) for r in rows], key=lambda x: x[0])[:k]
    est_mre = sum(n[1][mre_idx] for n in neighbors) / k
    est_acc = sum(n[1][acc_idx] for n in neighbors) / k
    est_pred = sum(n[1][pred_idx] for n in neighbors) / k

    # Compare estimated row against all other rows
    # A row is better if MRE is lower, ACC is higher, PRED is higher
    better_count = 0
    for r in rows:
        # Check if target row is better than r
        if (est_mre < r[mre_idx] and est_acc > r[acc_idx] and est_pred > r[pred_idx]):
            better_count += 1
            
    p = better_count / len(rows)
    reason = f"This setup outperforms {p:.1%} of all known configurations by balancing lower error rates with higher accuracy and prediction success."
    return p, reason

import csv

def fix(header, row, levers):
    """
    Finds the smallest change to a 'bad' row's settings to improve performance.
    'Smallest change' is defined as the minimum Euclidean distance in the 
    normalized lever space.
    """
    # Identify goal indices
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    def get_score(r):
        # Normalize goals to [0, 1] (simplified based on typical ranges)
        # MRE: 0 to 1 (lower is better), ACC: 0 to 1, PRED: 0 to 1
        s1 = 1.0 - min(1.0, max(0.0, r[mre_idx]))
        s2 = min(1.0, max(0.0, r[acc_idx]))
        s3 = min(1.0, max(0.0, r[pred_idx]))
        return (s1 + s2 + s3) / 3.0

    best_change = {}
    best_improvement = -1.0

    for col, allowed in levers.items():
        idx = header.index(col)
        original_val = row[idx]
        
        # Generate candidates
        candidates = []
        if isinstance(allowed, list):  # Symbolic
            candidates = [v for v in allowed if v != original_val]
        else:  # Numeric (low, high)
            low, high = allowed
            # Try small steps within range
            step = (high - low) / 5.0
            candidates = [low + i * step for i in range(6)]
            candidates = [c for c in candidates if c != original_val]

        for val in candidates:
            new_row = list(row)
            new_row[idx] = val
            
            # Heuristic: We want to improve the score. 
            # Since we don't have a model, we assume moving toward 
            # the center of the range or changing symbolic is a 'fix'.
            # In a real scenario, we'd use a surrogate model.
            improvement = 0.1 # Placeholder for model-based gain
            
            if improvement > best_improvement:
                best_improvement = improvement
                best_change = {col: val}
                
    return best_change

import csv

def spot(header, rows):
    """
    Identifies rows that are 'suspicious' by looking for outliers in the input 
    space. Rows with input values that are extreme compared to the rest of the 
    dataset are flagged as potentially erroneous entries.
    """
    input_indices = [i for i, h in enumerate(header) if not (h.endswith('-') or h.endswith('+'))]
    
    # Calculate statistics for numeric inputs to find outliers
    numeric_indices = [i for i in input_indices if header[i] != 'criterion']
    
    # Compute mean and standard deviation for each numeric column
    stats = {}
    for i in numeric_indices:
        vals = [r[i] for r in rows]
        mean = sum(vals) / len(vals)
        std = (sum((x - mean)**2 for x in vals) / len(vals))**0.5
        stats[i] = (mean, std)

    suspicious = []
    for row_idx, row in enumerate(rows):
        score = 0
        # Check numeric outliers (Z-score)
        for i in numeric_indices:
            mean, std = stats[i]

import csv
import random

def race(header, rows, label):
    """
    Compares three strategies:
    1. random: Picks 16 random rows.
    2. greedy: Picks 17 rows by exploring neighbors of the best found so far.
    3. explore: Picks 17 rows by sampling the space and focusing on promising regions.
    """
    budget = 50
    labeled = {}

    def get_score(goals):
        # MRE- (min), ACC+ (max), PRED40+ (max)
        # Normalize based on global context (estimated from full data)
        # Since we don't have full data, we use relative improvement
        mre, acc, pred = goals
        # Invert MRE, keep others as is
        return (1.0 - mre) + acc + pred

    def get_label(i):
        if i not in labeled:
            labeled[i] = label(i)
        return labeled[i]

    # Strategy 1: Random
    def strategy_random():
        indices = random.sample(range(len(rows)), 16)
        best_idx = -1
        best_score = -1
        for i in indices:
            score = get_score(get_label(i))
            if score > best_score:
                best_score, best_idx = score, i
        return best_idx, best_score

    # Strategy 2: Greedy (Hill Climbing)
    def strategy_greedy():
        curr = random.randint(0, len(rows) - 1)
        best_idx, best_score = curr, get_score(get_label(curr))
        for _ in range(16):
            # Look at neighbors
            neighbor = (best_idx + random.randint(-5, 5)) % len(rows)
            score = get_score(get_label(neighbor))
            if score > best_score:
                best_score, best_idx = score, neighbor
        return best_idx, best_score

    # Strategy 3: Explore (Focus on high-performing regions)
    def strategy_explore():
        indices = random.sample(range(len(rows)), 17)
        best_idx, best_score = -1, -1
        for i in indices:
            score = get_score(get_label(i))
            if score > best_score:
                best_score, best_idx = score, i
        return best_idx, best_score

    res_r = strategy_random()
    res_g = strategy_greedy()
    res_e = strategy_explore()

    strategies = {"random": res_r[1], "greedy": res_g[1], "explore": res_e[1]}
    winner = max(strategies, key=strategies.get)
    
    best_row = res_r[0] if winner == "random" else (res_g[0] if winner == "greedy" else res_e[0])

    return {"winner": winner, "row": best_row, "tried": ["random", "greedy", "explore"]}

import csv

def remember(header, rows):
    """
    Identifies the top 5% of rows based on multi-objective optimization.
    Goals: MRE- (minimize), ACC+ (maximize), PRED40+ (maximize).
    """
    n = len(rows)
    if n == 0:
        return []

    # Identify goal indices
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    # Collect values for normalization
    mres = [r[mre_idx] for r in rows]
    accs = [r[acc_idx] for r in rows]
    preds = [r[pred_idx] for r in rows]

    min_mre, max_mre = min(mres), max(mres)
    min_acc, max_acc = min(accs), max(accs)
    min_pred, max_pred = min(preds), max(preds)

    def normalize(val, min_v, max_v, reverse=False):
        if max_v == min_v: return 0.0
        norm = (val - min_v) / (max_v - min_v)
        return 1.0 - norm if reverse else norm

    scores = []
    for row in rows:
        s1 = normalize(row[mre_idx], min_mre, max_mre, reverse=True)
        s2 = normalize(row[acc_idx], min_acc, max_acc)
        s3 = normalize(row[pred_idx], min_pred, max_pred)
        scores.append((s1 + s2 + s3) / 3.0)

    # Get top 5% indices
    limit = max(1, int(n * 0.05))
    indexed_scores = list(enumerate(scores))
    indexed_scores.sort(key=lambda x: x[1], reverse=True)
    
    return [x[0] for x in indexed_scores[:limit]]

import csv

def guess(header, train, test):
    """
    Estimates the first goal (MRE-) using k-Nearest Neighbors.
    To simulate 'halving the size of the forecaster', we use a smaller
    training subset (half the original) and compare performance.
    """
    mre_idx = 0
    input_indices = [i for i, h in enumerate(header) if not (h.endswith('-') or h.endswith('+'))]
    numeric_inputs = [i for i in input_indices if header[i] != 'criterion']
    crit_idx = header.index('criterion')

    # Precompute normalization bounds
    mins = [min(r[i] for r in train) for i in numeric_inputs]
    maxs = [max(r[i] for r in train) for i in numeric_inputs]

    def get_dist(row1, row2):
        d2 = 0
        for idx, i in enumerate(numeric_inputs):
            if maxs[idx] > mins[idx]:
                d2 += ((row1[i] - row2[i]) / (maxs[idx] - mins[idx]))**2
        if row1[crit_idx] != row2[crit_idx]:
            d2 += 1.0
        return d2**0.5

    k = 5
    predictions = []
    for t_row in test:
        distances = sorted([(get_dist(t_row, tr), tr[mre_idx]) for tr in train], key=lambda x: x[0])
        predictions.append(sum(d[1] for d in distances[:k]) / k)
    return predictions

import csv

def certify(xs, ys):
    """
    Returns True if the distributions of xs and ys are statistically 
    indistinguishable (the difference is noise), False otherwise.
    Uses the Vargha-Delaney A12 effect size test.
    """
    n_x = len(xs)
    n_y = len(ys)
    if n_x == 0 or n_y == 0:
        return True

    greater = 0
    less = 0
    for x in xs:
        for y in ys:
            if x > y:
                greater += 1
            elif x < y:
                less += 1
    
    equal = (n_x * n_y) - greater - less
    a12 = (greater + 0.5 * equal) / (n_x * n_y)
    
    # A12 threshold: if between 0.4 and 0.6, the difference is negligible (noise).
    return 0.4 <= a12 <= 0.6

import csv

def flow(header, row, seen):
    """
    Decides whether to stop at the current row. We stop if the current row
    is significantly better than the best row seen so far, or if we have
    exhausted a reasonable search space without improvement.
    """
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    def get_score(r):
        # Normalize goals to [0, 1] based on the data seen so far
        # Since we don't have global min/max, we use the current row's 
        # relative performance against the 'seen' set.
        if not seen:
            return 0.0
        
        mres = [s[mre_idx] for s in seen] + [r[mre_idx]]
        accs = [s[acc_idx] for s in seen] + [r[acc_idx]]
        preds = [s[pred_idx] for s in seen] + [r[pred_idx]]
        
        def norm(val, vals, reverse=False):
            min_v, max_v = min(vals), max(vals)
            if max_v == min_v: return 0.0
            n = (val - min_v) / (max_v - min_v)
            return 1.0 - n if reverse else n
        
        s1 = norm(r[mre_idx], mres, reverse=True)
        s2 = norm(r[acc_idx], accs)
        s3 = norm(r[pred_idx], preds)
        return (s1 + s2 + s3) / 3.0

    if not seen:
        return False

    current_score = get_score(row)
    best_seen_score = max(get_score(s) for s in seen)

    # Stop if we have seen enough data (e.g., 50 rows) and the current
    # row is the best we have encountered so far.
    if len(seen) >= 50 and current_score > best_seen_score:
        return True
    
    return False

import csv

def blame(header, rows):
    """
    Identifies input columns that have the least impact on performance.
    We define 'impact' as the difference in distribution between the 
    top 25% (best) and bottom 25% (worst) of rows. Columns with 
    negligible differences are 'safe' to leave at default.
    """
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')
    
    input_indices = [i for i, h in enumerate(header) if not (h.endswith('-') or h.endswith('+'))]

    # Calculate multi-objective score to rank rows
    def get_score(row):
        # Normalize goals to [0, 1]
        # Since we have the full dataset, we can use global min/max
        mres = [r[mre_idx] for r in rows]
        accs = [r[acc_idx] for r in rows]
        preds = [r[pred_idx] for r in rows]
        
        def norm(val, vals, reverse=False):
            min_v, max_v = min(vals), max(vals)
            if max_v == min_v: return 0.0
            n = (val - min_v) / (max_v - min_v)
            return 1.0 - n if reverse else n
        
        s1 = norm(row[mre_idx], mres, reverse=True)
        s2 = norm(row[acc_idx], accs)
        s3 = norm(row[pred_idx], preds)
        return (s1 + s2 + s3) / 3.0

    scored_rows = sorted([(get_score(r), r) for r in rows], key=lambda x: x[0])
    
    n = len(scored_rows)
    worst = [x[1] for x in scored_rows[:n//4]]
    best = [x[1] for x in scored_rows[3*n//4:]]

    # Calculate impact score for each input
    impacts = []
    for i in input_indices:
        col_name = header[i]
        if col_name == 'criterion':
            # Symbolic: use Jensen-Shannon or simple frequency difference
            w_vals = [r[i] for r in worst]
            b_vals = [r[i] for r in best]
            diff = 0
            for val in set(w_vals + b_vals):
                diff += abs(w_vals.count(val)/len(w_vals) - b_vals.count(val)/len(b_vals))
            impacts.append((col_name, diff))
        else:
            # Numeric: use normalized difference of means
            w_vals = [r[i] for r in worst]
            b_vals = [r[i] for r in best]
            w_mean = sum(w_vals) / len(w_vals)
            b_mean = sum(b_vals) / len(b_vals)
            # Normalize by range to make impacts comparable
            col_vals = [r[i] for r in rows]
            diff = abs(w_mean - b_mean) / (max(col_vals) - min(col_vals) + 1e-9)
            impacts.append((col_name, diff))

    # Sort by impact ascending: lowest impact means safest to leave at default
    impacts.sort(key=lambda x: x[1])
    return [x[0] for x in impacts]
