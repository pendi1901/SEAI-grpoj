import csv

def remember(header, rows):
    """
    Identifies the top 5% of rows based on multi-objective optimization.
    Goals: MRE- (minimize), ACC+ (maximize), PRED40+ (maximize).
    We normalize values to [0, 1] and calculate a score.
    """
    n = len(rows)
    if n == 0:
        return []

    # Identify column indices
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    # Extract columns for normalization
    mres = [r[mre_idx] for r in rows]
    accs = [r[acc_idx] for r in rows]
    preds = [r[pred_idx] for r in rows]

    min_mre, max_mre = min(mres), max(mres)
    min_acc, max_acc = min(accs), max(accs)
    min_pred, max_pred = min(preds), max(preds)

    def normalize(val, min_v, max_v, reverse=False):
        if max_v == min_v: return 0.0
        norm = (val - min_v) / (max_v - min_v)
        return (1.0 - norm) if reverse else norm

    scores = []
    for i, row in enumerate(rows):
        # Score = average of normalized goals
        s = (normalize(row[mre_idx], min_mre, max_mre, True) +
             normalize(row[acc_idx], min_acc, max_acc, False) +
             normalize(row[pred_idx], min_pred, max_pred, False)) / 3.0
        scores.append((s, i))

    # Sort by score descending
    scores.sort(key=lambda x: x[0], reverse=True)

    # Return top 5% (or at least 1)
    limit = max(1, int(n * 0.05))
    return [item[1] for item in scores[:limit]]

import csv

def guess(header, train, test):
    """
    Estimates the first goal (MRE-) for each test row using a K-Nearest Neighbors 
    approach based on Euclidean distance of input features.
    """
    # Identify goal and input columns
    goal_idx = 0  # MRE-
    input_indices = [i for i, name in enumerate(header) if name not in ['MRE-', 'ACC+', 'PRED40+']]
    
    def get_dist(row1, row2):
        d = 0
        for i in input_indices:
            val1, val2 = row1[i], row2[i]
            if isinstance(val1, str):
                d += 0 if val1 == val2 else 1
            else:
                # Normalize numeric distance
                d += (val1 - val2) ** 2
        return d ** 0.5

    predictions = []
    for t_row in test:
        # Find the 5 nearest neighbors in the training set
        distances = []
        for train_row in train:
            distances.append((get_dist(t_row, train_row), train_row[goal_idx]))
        
        distances.sort(key=lambda x: x[0])
        
        # Average the goal value of the 5 nearest neighbors
        k = 5
        neighbors = distances[:k]
        avg_mre = sum(n[1] for n in neighbors) / k
        predictions.append(avg_mre)
        
    return predictions

import csv

def certify(xs, ys):
    """
    Determines if two distributions are significantly different using the 
    A12 effect size statistic. Returns True if they are the same (noise), 
    False if they really differ.
    """
    if not xs or not ys:
        return True

    def a12(lst1, lst2):
        m, n = len(lst1), len(lst2)
        count = 0
        for x in lst1:
            for y in lst2:
                if x > y: count += 1
                elif x == y: count += 0.5
        return count / (m * n)

    # A12 threshold: 0.56 is a small effect size, 0.64 medium, 0.71 large.
    # We use 0.6 as a conservative threshold for "really differ".
    score = a12(xs, ys)
    return 0.4 <= score <= 0.6

import csv

def flow(header, row, seen):
    """
    Decides if the current row is 'good enough' to stop searching.
    We define 'good enough' as being in the top 5% of the expected 
    distribution of the goals based on the history seen so far.
    """
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    # If this is the first row, we can't evaluate it against history
    if not seen:
        return False

    # Collect all seen values for normalization
    mres = [r[mre_idx] for r in seen]
    accs = [r[acc_idx] for r in seen]
    preds = [r[pred_idx] for r in seen]

    # Calculate current row's normalized score relative to history
    def get_norm(val, vals, reverse=False):
        min_v, max_v = min(vals), max(vals)
        if max_v == min_v: return 0.5
        norm = (val - min_v) / (max_v - min_v)
        return (1.0 - norm) if reverse else norm

    score = (get_norm(row[mre_idx], mres, True) +
             get_norm(row[acc_idx], accs, False) +
             get_norm(row[pred_idx], preds, False)) / 3.0

    # Calculate scores of all seen rows
    seen_scores = []
    for r in seen:
        s = (get_norm(r[mre_idx], mres, True) +
             get_norm(r[acc_idx], accs, False) +
             get_norm(r[pred_idx], preds, False)) / 3.0
        seen_scores.append(s)

    # If current score is better than 95% of seen, stop
    threshold = sorted(seen_scores)[int(len(seen_scores) * 0.95)]
    return score >= threshold

import csv

def blame(header, rows):
    """
    Identifies input columns that correlate most strongly with poor performance.
    We define 'bad' as the bottom 25% of rows based on a multi-objective score.
    We then calculate the correlation (or difference in means) between input 
    settings and the 'bad' status.
    """
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')
    
    input_indices = [i for i, name in enumerate(header) if name not in ['MRE-', 'ACC+', 'PRED40+']]
    
    # Score rows to find the worst ones
    def get_score(row):
        # Simple normalization based on min/max of the whole dataset
        mres = [r[mre_idx] for r in rows]
        accs = [r[acc_idx] for r in rows]
        preds = [r[pred_idx] for r in rows]
        
        def norm(v, vals, rev):
            mn, mx = min(vals), max(vals)
            if mx == mn: return 0.5
            n = (v - mn) / (mx - mn)
            return 1.0 - n if rev else n
            
        return (norm(row[mre_idx], mres, True) + 
                norm(row[acc_idx], accs, False) + 
                norm(row[pred_idx], preds, False)) / 3.0

    scored = sorted([(get_score(r), r) for r in rows])
    worst = scored[:len(scored) // 4]
    best = scored[-(len(scored) // 4):]
    
    # Compare mean/distribution of inputs in worst vs best
    impact = []
    for i in input_indices:
        name = header[i]
        w_vals = [r[1][i] for r in worst]
        b_vals = [r[1][i] for r in best]
        
        if isinstance(w_vals[0], str):
            # For symbolic, check frequency difference
            w_freq = w_vals.count(w_vals[0]) / len(w_vals)
            b_freq = b_vals.count(w_vals[0]) / len(b_vals)
            diff = abs(w_freq - b_freq)
        else:
            # For numeric, check difference in means
            diff = abs((sum(w_vals)/len(w_vals)) - (sum(b_vals)/len(b_vals)))
            
        impact.append((diff, name))
    
    impact.sort(reverse=True)
    return [name for _, name in impact[:3]]

import csv

def justify(header, rows, row):
    """
    Estimates the performance of 'row' by comparing its input features to the 
    historical 'rows' using a K-Nearest Neighbors approach, then calculates 
    the percentile rank of the estimated performance.
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith(('-', '+'))]
    input_indices = [i for i, name in enumerate(header) if i not in goal_indices]

    def get_dist(r1, r2):
        d = 0
        for i in input_indices:
            v1, v2 = r1[i], r2[i]
            if isinstance(v1, str):
                d += 0 if v1 == v2 else 1
            else:
                d += (v1 - v2) ** 2
        return d ** 0.5

    def get_score(r):
        # Normalize goals: MRE- (min), ACC+ (max), PRED40+ (max)
        scores = []
        for idx in goal_indices:
            vals = [x[idx] for x in rows]
            mn, mx = min(vals), max(vals)
            norm = (r[idx] - mn) / (mx - mn) if mx > mn else 0
            scores.append(1.0 - norm if header[idx].endswith('-') else norm)
        return sum(scores) / len(scores)

    # Estimate goals for the target row using 5-NN
    distances = sorted([(get_dist(row, r), r) for r in rows])
    k = 5
    neighbors = distances[:k]
    
    estimated_goals = []
    for idx in goal_indices:
        avg = sum(n[1][idx] for n in neighbors) / k
        estimated_goals.append(avg)
    
    # Create a temporary row with estimated goals to score it
    row_with_goals = list(row)
    for i, idx in enumerate(goal_indices):
        row_with_goals[idx] = estimated_goals[i]
    
    target_score = get_score(row_with_goals)
    
    # Compare against all rows
    all_scores = [get_score(r) for r in rows]
    better_than = [s for s in all_scores if target_score > s]
    p = len(better_than) / len(rows)
    
    reason = f"This configuration is estimated to outperform {p:.1%} of historical setups based on its input parameters."
    return p, reason

import csv
import random

def choose(header, rows, label):
    """
    Finds the best setup by sampling 50 rows, labeling them, 
    and returning the index of the best performing one.
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith(('-', '+'))]
    
    def get_score(goals):
        # Normalize goals: MRE- (min), ACC+ (max), PRED40+ (max)
        # Since we don't have the full distribution, we use the labeled values
        # to rank the 50 sampled rows relative to each other.
        scores = []
        for i, val in enumerate(goals):
            # For simplicity in ranking, we treat the sampled set as the population
            # A goal ending in '-' is minimized, '+' is maximized.
            if header[goal_indices[i]].endswith('-'):
                scores.append(-val)
            else:
                scores.append(val)
        return sum(scores)

    # Sample 50 random indices
    indices = random.sample(range(len(rows)), min(50, len(rows)))
    
    best_idx = -1
    best_score = float('-inf')
    
    for i in indices:
        goals = label(i)
        score = get_score(goals)
        if score > best_score:
            best_score = score
            best_idx = i
            
    return best_idx

def fix(header, row, levers):
    """
    Finds the smallest change to a 'bad' row that improves its score.
    'Smallest change' is defined by the number of levers adjusted.
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith(('-', '+'))]
    
    def get_score(r):
        # Simple heuristic: sum of normalized goals
        # Since we don't have the full dataset context here, we use a 
        # relative improvement metric based on the provided row structure.
        # Higher is better.
        score = 0
        for i in goal_indices:
            val = r[i]
            if header[i].endswith('-'):
                score -= val
            else:
                score += val
        return score

    # We try changing one lever at a time to find an improvement
    best_change = {}
    
    for col_name, allowed in levers.items():
        idx = header.index(col_name)
        original_val = row[idx]
        
        # Generate a candidate value
        if isinstance(allowed, tuple):  # Numeric (low, high)
            low, high = allowed
            # Try a small step or a random value within range
            new_val = low + (high - low) * 0.5
        else:  # Symbolic
            new_val = [v for v in allowed if v != original_val][0]
            
        return {col_name: new_val}

    return best_change

import csv

def spot(header, rows):
    """
    Identifies rows that are 'suspicious' by detecting outliers in the input space.
    A row is considered suspicious if its input values are statistically 
    unlikely compared to the distribution of inputs in the rest of the dataset.
    """
    input_indices = [i for i, name in enumerate(header) if not name.endswith(('-', '+'))]
    
    # Calculate mean and standard deviation for numeric inputs to detect outliers
    stats = {}
    for i in input_indices:
        col_data = [r[i] for r in rows if isinstance(r[i], (int, float))]
        if col_data:
            mean = sum(col_data) / len(col_data)
            std = (sum((x - mean) ** 2 for x in col_data) / len(col_data)) ** 0.5
            stats[i] = (mean, std if std > 0 else 1.0)

    suspicion_scores = []
    for row_idx, row in enumerate(rows):
        score = 0
        for i in input_indices:
            val = row[i]
            if isinstance(val, (int, float)) and i in stats:
                mean, std = stats[i]
                # Z-score: how many standard deviations away from the mean
                z = abs(val - mean) / std
                score += z
            elif isinstance(val, str):
                # For symbolic, check if it's a rare category
                col_vals = [r[i] for r in rows]
                freq = col_vals.count(val) / len(col_vals)
                if freq < 0.05:  # Rare category
                    score += 3.0
        
        suspicion_scores.append((score, row_idx))

    # Sort by suspicion score descending
    suspicion_scores.sort(key=lambda x: x[0], reverse=True)
    
    # Return indices of the top 10 most suspicious rows
    return [item[1] for item in suspicion_scores[:10]]

import csv
import random

def race(header, rows, label):
    """
    Compares three strategies: random, greedy, and heuristic.
    Budget is 50 labels total, shared across all strategies.
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith(('-', '+'))]
    
    def get_score(goals):
        # MRE- (min), ACC+ (max), PRED40+ (max)
        # Normalize: (1 - MRE) + ACC + PRED
        return (1.0 - goals[0]) + goals[1] + goals[2]

    labeled = {}
    def get_label(i):
        if i not in labeled:
            if len(labeled) >= 50:
                raise Exception("Budget exceeded")
            labeled[i] = label(i)
        return labeled[i]

    # Strategies share the 50-label budget

import csv

def remember(header, rows):
    """
    Identifies the top 5% of rows based on multi-objective optimization.
    Goals: MRE- (minimize), ACC+ (maximize), PRED40+ (maximize).
    """
    n = len(rows)
    if n == 0:
        return []

    # Identify goal column indices
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    # Extract goals for normalization
    mres = [r[mre_idx] for r in rows]
    accs = [r[acc_idx] for r in rows]
    preds = [r[pred_idx] for r in rows]

    min_mre, max_mre = min(mres), max(mres)
    min_acc, max_acc = min(accs), max(accs)
    min_pred, max_pred = min(preds), max(preds)

    def normalize(val, min_v, max_v, reverse=False):
        if max_v == min_v: return 0.0
        norm = (val - min_v) / (max_v - min_v)
        return (1.0 - norm) if reverse else norm

    scores = []
    for i, row in enumerate(rows):
        # Score = average of normalized goals
        s = (normalize(row[mre_idx], min_mre, max_mre, True) +
             normalize(row[acc_idx], min_acc, max_acc, False) +
             normalize(row[pred_idx], min_pred, max_pred, False)) / 3.0
        scores.append((s, i))

    # Sort by score descending
    scores.sort(key=lambda x: x[0], reverse=True)

    # Return top 5% (or at least 1)
    limit = max(1, int(n * 0.05))
    return [item[1] for item in scores[:limit]]

def guess(header, train, test):
    """
    Estimates the first goal (MRE-) for each test row.
    To answer the customer's question about halving the forecaster size,
    we compare the performance of a model trained on the full 'train' set
    vs a model trained on a subset (halved N_estimators).
    """
    # Identify indices
    goal_idx = 0
    n_est_idx = header.index('N_estimators')
    input_indices = [i for i, name in enumerate(header) if name not in ['MRE-', 'ACC+', 'PRED40+']]
    
    def get_dist(row1, row2):
        d = 0
        for i in input_indices:
            v1, v2 = row1[i], row2[i]
            if isinstance(v1, str):
                d += 0 if v1 == v2 else 1
            else:
                d += ((v1 - v2) / 200.0) ** 2  # Normalized distance
        return d ** 0.5

    def predict(model_data, target_row):
        # 5-NN prediction
        distances = sorted([(get_dist(target_row, r), r[goal_idx]) for r in model_data])
        return sum(n[1] for n in distances[:5]) / 5.0

    # Full model
    full_preds = [predict(train, t) for t in test]
    
    # Halved model: filter training data to rows with N_estimators <= median

    median_n = sum(n_ests) / len(n_ests)
    halved_train = [r for r in train if r[n_est_idx] <= median_n]
    
    halved_preds = [predict(halved_train, t) for t in test]
    
    # Return the halved model predictions as requested
    return halved_preds

import csv

def certify(xs, ys):
    """
    Determines if two distributions are significantly different using the 
    A12 effect size statistic. Returns True if they are the same (noise), 
    False if they really differ.
    """
    if not xs or not ys:
        return True

    def a12(lst1, lst2):
        m, n = len(lst1), len(lst2)
        count = 0
        for x in lst1:
            for y in lst2:
                if x > y: count += 1
                elif x == y: count += 0.5
        return count / (m * n)

    # A12 threshold: 0.56 is small, 0.64 medium, 0.71 large.
    # We use 0.6 as a conservative threshold for "really differ".
    score = a12(xs, ys)
    return 0.4 <= score <= 0.6

import csv

def flow(header, row, seen):
    """
    Decides if the current row is the best seen so far.
    We stop if the current row's multi-objective score is the best 
    among all rows seen so far, and we have seen at least 50 rows 
    to ensure the baseline is stable.
    """
    if not seen:
        return False
    
    goal_indices = [i for i, name in enumerate(header) if name.endswith(('-', '+'))]
    
    def get_score(r):
        scores = []
        for idx in goal_indices:
            # We normalize based on the current 'seen' set to evaluate relative quality
            vals = [s[idx] for s in seen + [r]]
            mn, mx = min(vals), max(vals)
            norm = (r[idx] - mn) / (mx - mn) if mx > mn else 0
            scores.append(1.0 - norm if header[idx].endswith('-') else norm)
        return sum(scores) / len(scores)

    # Only stop if we have a reasonable sample size
    if len(seen) < 50:
        return False

    current_score = get_score(row)
    seen_scores = [get_score(s) for s in seen]
    
    # Stop if this is the best score encountered so far
    return current_score >= max(seen_scores)

import csv

def blame(header, rows):
    """
    Identifies input columns that correlate most strongly with poor performance.
    We define 'bad' as the bottom 25% of rows based on a multi-objective score.
    The columns that have the least impact on the difference between 'bad' 
    and 'good' rows are the ones that can be left at factory defaults.
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith(('-', '+'))]
    input_indices = [i for i, name in enumerate(header) if not name.endswith(('-', '+'))]
    
    # Score rows to find the worst ones
    def get_score(row):
        scores = []
        for idx in goal_indices:
            vals = [r[idx] for r in rows]
            mn, mx = min(vals), max(vals)
            norm = (row[idx] - mn) / (mx - mn) if mx > mn else 0.5
            scores.append(1.0 - norm if header[idx].endswith('-') else norm)
        return sum(scores) / len(scores)

    scored = sorted([(get_score(r), r) for r in rows])
    worst = [r[1] for r in scored[:len(scored) // 4]]
    best = [r[1] for r in scored[-(len(scored) // 4):]]
    
    # Calculate impact: difference in distribution between worst and best
    impact = []
    for i in input_indices:
        name = header[i]
        w_vals = [r[i] for r in worst]
        b_vals = [r[i] for r in best]
        
        if isinstance(w_vals[0], str):
            # For symbolic, check frequency difference
            w_freq = w_vals.count(w_vals[0]) / len(w_vals)
            b_freq = b_vals.count(w_vals[0]) / len(b_vals)
            diff = abs(w_freq - b_freq)
        else:
            # For numeric, check difference in means normalized by range
            all_vals = [r[i] for r in rows]
            diff = abs((sum(w_vals)/len(w_vals)) - (sum(b_vals)/len(b_vals)))
            diff /= (max(all_vals) - min(all_vals) + 1e-9)
            
        impact.append((diff, name))
    
    # Sort by impact ascending: lowest impact means safest to leave at default
    impact.sort()
    return [name for _, name in impact]
