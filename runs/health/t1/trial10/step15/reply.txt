import csv

def blame(header, rows):
    """
    Identifies input columns that correlate most strongly with poor performance.
    We define 'bad' as the bottom 25% of rows based on a multi-objective score.
    The columns that have the least impact on the difference between 'bad' 
    and 'good' rows are the ones that can be left at factory defaults.
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith(('-', '+'))]
    input_indices = [i for i, name in enumerate(header) if not name.endswith(('-', '+'))]
    
    # Score rows to find the worst ones
    def get_score(row):
        scores = []
        for idx in goal_indices:
            vals = [r[idx] for r in rows]
            mn, mx = min(vals), max(vals)
            norm = (row[idx] - mn) / (mx - mn) if mx > mn else 0.5
            scores.append(1.0 - norm if header[idx].endswith('-') else norm)
        return sum(scores) / len(scores)

    scored = sorted([(get_score(r), r) for r in rows])
    worst = [r[1] for r in scored[:len(scored) // 4]]
    best = [r[1] for r in scored[-(len(scored) // 4):]]
    
    # Calculate impact: difference in distribution between worst and best
    impact = []
    for i in input_indices:
        name = header[i]
        w_vals = [r[i] for r in worst]
        b_vals = [r[i] for r in best]
        
        if isinstance(w_vals[0], str):
            # For symbolic, check frequency difference
            w_freq = w_vals.count(w_vals[0]) / len(w_vals)
            b_freq = b_vals.count(w_vals[0]) / len(b_vals)
            diff = abs(w_freq - b_freq)
        else:
            # For numeric, check difference in means normalized by range
            all_vals = [r[i] for r in rows]
            diff = abs((sum(w_vals)/len(w_vals)) - (sum(b_vals)/len(b_vals)))
            diff /= (max(all_vals) - min(all_vals) + 1e-9)
            
        impact.append((diff, name))
    
    # Sort by impact ascending: lowest impact means safest to leave at default
    impact.sort()
    return [name for _, name in impact]

if __name__ == "__main__":
    filename = 'Health-ClosedIssues0000.n1000.train.csv'
    try:
        with open(filename, 'r') as f:
            reader = csv.reader(f)
            header = next(reader)
            rows = []
            for row in reader:
                processed = [float(val) if header[i] not in ['criterion'] else val 
                             for i, val in enumerate(row)]
                rows.append(processed)
        
        # The customer wants to know what they can leave at default.
        # These are the inputs with the least impact on performance.
        safe_settings = blame(header, rows)
        
        print("Settings that can be left at factory defaults (least impact on performance):")
        for s in safe_settings:
            print(f"- {s}")
        exit(0)
    except FileNotFoundError:
        print(f"Error: {filename} not found.")
        exit(1)
