import csv

def fix(header, row, levers):
    """
    Finds the single best lever change to improve the performance of a 'bad' row.
    Performance is measured by normalizing goals (MRE-, ACC+, PRED40+) and 
    calculating a weighted score.
    """
    goal_indices = [i for i, name in enumerate(header) if '-' in name or '+' in name]
    
    def get_score(r):
        # We don't have global min/max here, so we use a simple directional score
        # MRE- (minimize), ACC+ (maximize), PRED40+ (maximize)
        s = 0
        for i in goal_indices:
            val = r[i]
            if header[i].endswith('-'):
                s -= val
            else:
                s += val
        return s

    best_change = {}
    best_improvement = -float('inf')
    
    for col_name, allowed in levers.items():
        idx = header.index(col_name)
        original_val = row[idx]
        
        candidates = []
        if isinstance(allowed, list):
            # Symbolic: try all alternatives
            candidates = [v for v in allowed if v != original_val]
        else:
            # Numeric: try boundaries and midpoint
            low, high = allowed
            candidates = [low, (low + high) / 2, high]
            candidates = [c for c in candidates if abs(c - original_val) > 1e-5]
            
        for cand in candidates:
            new_row = list(row)
            new_row[idx] = cand
            improvement = get_score(new_row) - get_score(row)
            if improvement > best_improvement:
                best_improvement = improvement
                best_change = {col_name: cand}
                
    return best_change

if __name__ == "__main__":
    filename = 'Health-ClosedIssues0000.n1000.train.csv'
    rows = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = []
            for i, val in enumerate(row):
                if header[i] == 'criterion':
                    processed_row.append(val)
                else:
                    processed_row.append(float(val))
            rows.append(processed_row)

    # Define levers
    levers = {
        'N_estimators': (50, 200),
        'criterion': ['absolute_error', 'poisson', 'squared_error'],
        'Min_sample_leaves': (1, 20),
        'Max_depth': (1, 20)
    }
    
    # Pick a 'bad' row (e.g., the one with the lowest aggregate score)
    def score_row(r):
        return sum(r[i] for i, h in enumerate(header) if h.endswith('+')) - \
               sum(r[i] for i, h in enumerate(header) if h.endswith('-'))
    
    bad_row = min(rows, key=score_row)
    recommendation = fix(header, bad_row, levers)
    
    print("Optimization recommendation for your setup:")
    for col, val in recommendation.items():
        print(f"Change '{col}' to {val} to improve performance.")
