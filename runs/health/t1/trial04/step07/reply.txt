import csv
import random

def choose(header, rows, label):
    """
    Finds the best setup by sampling 50 rows, labeling them, 
    and returning the index of the best performing one.
    """
    n = len(rows)
    # We have a budget of 50 labels.
    # We pick 50 indices to evaluate.
    indices = random.sample(range(n), min(n, 50))
    
    goal_indices = [i for i, name in enumerate(header) if '-' in name or '+' in name]
    
    best_idx = -1
    best_score = -float('inf')
    
    # Pre-calculate min/max for normalization based on the sampled data
    # Since we only see 50, we use them as our reference population.
    sampled_goals = []
    for i in indices:
        sampled_goals.append(label(i))
        
    # Transpose to get columns
    cols = list(zip(*sampled_goals))
    
    # Calculate min/max for each goal to normalize
    mins = []
    maxs = []
    for c in cols:
        mins.append(min(c))
        maxs.append(max(c))
        
    def get_score(goals):
        s = 0
        for i, g_idx in enumerate(goal_indices):
            val = goals[i]
            rng = maxs[i] - mins[i]
            norm = (val - mins[i]) / rng if rng > 0 else 0
            # MRE- (minimize) -> 1 - norm
            # ACC+, PRED40+ (maximize) -> norm
            if header[g_idx].endswith('-'):
                s += (1 - norm)
            else:
                s += norm
        return s

    for i, goals in zip(indices, sampled_goals):
        score = get_score(goals)
        if score > best_score:
            best_score = score
            best_idx = i
            
    return best_idx

if __name__ == "__main__":
    filename = 'Health-ClosedIssues0000.n1000.train.csv'
    rows = []
    true_goals = []
    
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = []
            goals = []
            for i, val in enumerate(row):
                if '-' in header[i] or '+' in header[i]:
                    goals.append(float(val))
                    processed_row.append("?")
                elif header[i] == 'criterion':
                    processed_row.append(val)
                else:
                    processed_row.append(float(val))
            rows.append(processed_row)
            true_goals.append(goals)

    def label(i):
        return true_goals[i]

    best_idx = choose(header, rows, label)
    
    print(f"Search complete. Best setup found at index {best_idx}.")
    print(f"Setup details: {rows[best_idx]}")
    print(f"True goal values: {true_goals[best_idx]}")
