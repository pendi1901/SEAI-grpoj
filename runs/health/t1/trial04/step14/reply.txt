import csv

def flow(header, row, seen):
    """
    Decides whether to stop based on whether the current row is a 'plateau' 
    breaker or if we have reached a point of diminishing returns.
    
    A row is 'good' if it is in the top 5% of the 'seen' data.
    We stop if we have seen enough data (e.g., 50 rows) and the current 
    row is not significantly better than the best seen so far.
    """
    if len(seen) < 50:
        return False

    goal_indices = [i for i, name in enumerate(header) if '-' in name or '+' in name]
    
    def get_score(r):
        s = 0
        for i in goal_indices:
            # Normalize goals: MRE- (min), ACC+ (max), PRED40+ (max)
            # Since we don't have global context, we use the 'seen' data as a baseline
            vals = [seen_row[i] for seen_row in seen]
            mn, mx = min(vals), max(vals)
            norm = (r[i] - mn) / (mx - mn) if mx > mn else 0
            if header[i].endswith('-'):
                s += (1 - norm)
            else:
                s += norm
        return s

    best_score = max(get_score(s) for s in seen)
    current_score = get_score(row)
    
    # If the current row is not within 99% of the best score, 
    # and we have a decent sample size, we stop to save costs.
    if current_score < (0.99 * best_score):
        return True
        
    return False

if __name__ == "__main__":
    filename = 'Health-ClosedIssues0000.n1000.train.csv'
    rows = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = []
            for i, val in enumerate(row):
                if header[i] == 'criterion':
                    processed_row.append(val)
                else:
                    processed_row.append(float(val))
            rows.append(processed_row)

    seen = []
    stopped_at = -1
    for i, row in enumerate(rows):
        if i > 50 and flow(header, row, seen):
            stopped_at = i
            break
        seen.append(row)

    if stopped_at != -1:
        print(f"Optimization stopped at row {stopped_at + 1}.")
        print("Reason: Diminishing returns detected; current setup is not improving significantly.")
    else:
        print("Processed all rows: No clear plateau found.")
