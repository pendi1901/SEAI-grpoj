import csv

def remember(header, rows):
    """
    Identifies the top 5% of rows based on multi-objective optimization.
    Goals: MRE- (minimize), ACC+ (maximize), PRED40+ (maximize).
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith(('-', '+'))]
    
    # Filter out rows that have hidden values ("?")
    valid_rows = []
    for i, row in enumerate(rows):
        if all(not isinstance(cell, str) or cell != "?" for cell in row):
            valid_rows.append((i, row))
            
    if not valid_rows:
        return [0]

    # Normalize goals to [0, 1]
    goals_data = [[r[i] for i in goal_indices] for _, r in valid_rows]
    mins = [min(g[j] for g in goals_data) for j in range(len(goal_indices))]
    maxs = [max(g[j] for g in goals_data) for j in range(len(goal_indices))]
    
    def get_score(row):
        score = 0
        for j, g_idx in enumerate(goal_indices):
            val = row[g_idx]
            rng = maxs[j] - mins[j]
            norm = (val - mins[j]) / rng if rng > 0 else 0
            if header[g_idx].endswith('-'):
                score -= norm
            else:
                score += norm
        return score

    scored = []
    for i, row in valid_rows:
        scored.append((get_score(row), i))
    
    # Sort descending by score
    scored.sort(key=lambda x: x[0], reverse=True)
    
    # Return top 5%
    limit = max(1, len(rows) // 20)
    return [item[1] for item in scored[:limit]]

if __name__ == "__main__":
    filename = "Health-ClosedIssues0000.n1000.train.csv"
    try:
        rows = []
        with open(filename, 'r') as f:
            reader = csv.reader(f)
            header = next(reader)
            for row in reader:
                processed = []
                for i, val in enumerate(row):
                    if header[i] == 'criterion':
                        processed.append(val)
                    else:
                        try:
                            processed.append(float(val))
                        except ValueError:
                            processed.append(val)
                rows.append(processed)
        
        best_indices = remember(header, rows)
        
        print("Top performing setups identified:")
        for idx in best_indices:
            print(f"Index {idx}: {rows[idx]}")
            
    except FileNotFoundError:
        print(f"Error: {filename} not found.")
        exit(1)
