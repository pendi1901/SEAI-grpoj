import csv

def justify(header, rows, row):
    """
    Estimates the performance of 'row' by comparing it against all 'rows'
    using a k-Nearest Neighbors approach to estimate the missing goals,
    then calculating the fraction of rows that this row outperforms.
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith(('-', '+'))]
    input_indices = [i for i, name in enumerate(header) if not name.endswith(('-', '+'))]
    
    # 1. Estimate the goals for the target row using k-NN
    def get_dist(r1, r2):
        d = 0
        for i in input_indices:
            if header[i] == 'criterion':
                d += 1.0 if r1[i] != r2[i] else 0.0
            else:
                d += (r1[i] - r2[i])**2
        return d**0.5

    distances = sorted([(get_dist(row, r), r) for r in rows], key=lambda x: x[0])
    k = 5
    neighbors = distances[:k]
    
    estimated_goals = []
    for g_idx in goal_indices:
        avg = sum(n[1][g_idx] for n in neighbors) / k
        estimated_goals.append(avg)

    # 2. Compare estimated row against all rows
    def is_better(r_goals, target_goals):
        # A row is better if it is closer to the ideal for all goals
        # We use a simple dominance check: better if it beats the target on average
        score_r = 0
        score_t = 0
        for i, g_idx in enumerate(goal_indices):
            # Normalize/compare: lower is better for '-', higher for '+'
            direction = -1 if header[g_idx].endswith('-') else 1
            if (r_goals[i] * direction) > (target_goals[i] * direction):
                score_r += 1
            else:
                score_t += 1
        return score_r > score_t

    better_count = 0
    for r in rows:
        r_goals = [r[g] for g in goal_indices]
        if is_better(r_goals, estimated_goals):
            better_count += 1
            
    p = better_count / len(rows)
    reason = f"This setup outperforms {int(p * 100)}% of the known configurations based on its estimated performance metrics."
    return (p, reason)

if __name__ == "__main__":
    filename = "Health-ClosedIssues0000.n1000.train.csv"
    try:
        with open(filename, 'r') as f:
            reader = csv.reader(f)
            header = next(reader)
            rows = []
            for row in reader:
                processed = [float(val) if header[i] not in ['criterion'] else val 
                             for i, val in enumerate(row)]
                rows.append(processed)
        
        # Test with the last row as the "target"
        target_row = list(rows[-1])
        for i in range(len(target_row)):
            if header[i].endswith(('-', '+')):
                target_row[i] = "?"
        
        p, reason = justify(header, rows, target_row)
        print(f"Performance Justification: {reason}")
        print(f"Fraction of rows outperformed: {p:.4f}")
        
    except FileNotFoundError:
        print(f"Error: {filename} not found.")
        exit(1)
