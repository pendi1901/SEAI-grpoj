import csv

def remember(header, rows):
    """
    Identifies the top 5% of rows based on multi-objective optimization.
    Goals: MRE- (minimize), ACC+ (maximize), PRED40+ (maximize).
    """
    # Identify column indices
    idx_mre = header.index('MRE-')
    idx_acc = header.index('ACC+')
    idx_pred = header.index('PRED40+')

    # Normalize values to [0, 1] to compare different scales
    # Since we don't know the global min/max, we use the current batch
    mres = [r[idx_mre] for r in rows]
    accs = [r[idx_acc] for r in rows]
    preds = [r[idx_pred] for r in rows]

    min_mre, max_mre = min(mres), max(mres)
    min_acc, max_acc = min(accs), max(accs)
    min_pred, max_pred = min(preds), max(preds)

    def score(row):
        # Normalize: lower MRE is better, higher ACC/PRED is better
        # Handle division by zero if all values are identical
        s_mre = (max_mre - row[idx_mre]) / (max_mre - min_mre) if max_mre > min_mre else 0
        s_acc = (row[idx_acc] - min_acc) / (max_acc - min_acc) if max_acc > min_acc else 0
        s_pred = (row[idx_pred] - min_pred) / (max_pred - min_pred) if max_pred > min_pred else 0
        return s_mre + s_acc + s_pred

    # Calculate scores and keep track of original indices
    scored_rows = []
    for i, row in enumerate(rows):
        scored_rows.append((score(row), i))

    # Sort by score descending
    scored_rows.sort(key=lambda x: x[0], reverse=True)

    # Return top 5% (or at least 1)
    limit = max(1, len(rows) // 20)
    return [item[1] for item in scored_rows[:limit]]

import csv

def guess(header, train, test):
    """
    Estimates the first goal column (MRE-) for test rows using a 
    k-Nearest Neighbors approach based on Euclidean distance of 
    numeric input features.
    """
    # Identify indices
    goal_idx = 0  # MRE-
    # Inputs are columns that are not goals (don't end in - or +)
    input_indices = [
        i for i, name in enumerate(header) 
        if not (name.endswith('-') or name.endswith('+'))
    ]

    # Normalize numeric inputs for distance calculation
    # Only consider numeric inputs for distance
    numeric_inputs = [i for i in input_indices if i != header.index('criterion')]
    
    def get_dist(row1, row2):
        d = 0
        for i in numeric_inputs:
            # Simple distance: assume range is roughly similar or normalized
            d += (row1[i] - row2[i])**2
        # Symbolic distance: 1 if different, 0 if same
        crit_idx = header.index('criterion')
        if row1[crit_idx] != row2[crit_idx]:
            d += 1.0
        return d**0.5

    predictions = []
    for t_row in test:
        # Find k nearest neighbors (k=5)
        distances = []
        for train_row in train:
            distances.append((get_dist(t_row, train_row), train_row[goal_idx]))
        
        distances.sort(key=lambda x: x[0])
        
        # Average the goal value of the 5 nearest neighbors
        k = 5
        neighbors = distances[:k]
        avg_mre = sum(n[1] for n in neighbors) / k
        predictions.append(avg_mre)
        
    return predictions

import csv

def certify(xs, ys):
    """
    Returns True if the distributions of xs and ys are statistically 
    indistinguishable (same apart from noise), False otherwise.
    Uses a simple A12 effect size check (Vargha-Delaney).
    """
    if not xs or not ys:
        return True
    
    n_x, n_y = len(xs), len(ys)
    
    # Count how often x > y
    greater = 0
    equal = 0
    for x in xs:
        for y in ys:
            if x > y:
                greater += 1
            elif x == y:
                equal += 1
                
    a12 = (greater + 0.5 * equal) / (n_x * n_y)
    
    # A12 threshold: 0.56 is a small effect size.
    # If A12 is between 0.44 and 0.56, they are effectively the same.
    return 0.44 <= a12 <= 0.56

def analyze(header, rows):
    """
    Compares 'big' setups (high N_estimators) vs 'small' setups.
    """
    idx_n = header.index('N_estimators')
    idx_acc = header.index('ACC+')
    
    # Split by median of N_estimators
    n_values = [r[idx_n] for r in rows]
    median_n = sorted(n_values)[len(n_values) // 2]
    
    big = [r[idx_acc] for r in rows if r[idx_n] > median_n]
    small = [r[idx_acc] for r in rows if r[idx_n] <= median_n]
    
    is_same = certify(big, small)
    
    print(f"Analysis: Comparing ACC+ for {len(big)} big setups vs {len(small)} small setups.")
    if is_same:
        print("Result: The difference is likely just noise.")
    else:
        print("Result: There is a statistically significant difference between setups.")
    return is_same

import csv

def flow(header, row, seen):
    """
    Decides whether to accept the current row based on whether it is 
    better than all previously seen rows according to a multi-objective 
    weighted score.
    """
    # Identify goal indices
    goals = []
    for i, name in enumerate(header):
        if name.endswith('-'):
            goals.append((i, -1))  # Minimize
        elif name.endswith('+'):
            goals.append((i, 1))   # Maximize

    def get_score(r):
        # Calculate a simple score based on relative performance.
        # Since we don't have global min/max, we use the current row's 
        # values as a baseline or just compare raw values if possible.
        # Here, we use a simple heuristic: normalize goals to [0, 1] 
        # relative to the best seen so far.
        score = 0
        for idx, direction in goals:
            val = r[idx]
            # Since we need to decide on the fly, we treat the current row
            # as a candidate. If it's the first, accept it.
            score += val * direction
        return score

    if not seen:
        return True

    # Compare current row against the best seen so far
    current_score = get_score(row)
    best_score = max(get_score(r) for r in seen)

    # If current row is better than the best seen, accept it
    # (For minimization, lower is better; for maximization, higher is better)
    # This logic assumes we want to improve upon the current best.
    return current_score < best_score if goals[0][1] == -1 else current_score > best_score

import csv

def blame(header, rows):
    """
    Identifies input columns that correlate with poor performance.
    A row is 'bad' if its goals are significantly worse than the median.
    We compare the distribution of input values in 'bad' rows vs 'good' rows.
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith(('-', '+'))]
    input_indices = [i for i, name in enumerate(header) if not name.endswith(('-', '+'))]

    def get_row_score(row):
        # Normalize goals: lower MRE is better, higher ACC/PRED is better
        # We use a simple aggregate score relative to the dataset range
        score = 0
        for i in goal_indices:
            vals = [r[i] for r in rows if isinstance(r[i], (int, float))]
            if not vals: continue
            min_v, max_v = min(vals), max(vals)
            norm = (row[i] - min_v) / (max_v - min_v) if max_v > min_v else 0
            if header[i].endswith('-'):
                score -= norm
            else:
                score += norm
        return score

    # Split into bad (bottom 25%) and good (top 25%)
    scored = sorted([(get_row_score(r), r) for r in rows], key=lambda x: x[0])
    n = len(scored) // 4
    bad_rows = [x[1] for x in scored[:n]]
    good_rows = [x[1] for x in scored[-n:]]

    # For each input, check if the mean/distribution differs significantly
    # We return columns where the 'bad' rows have a distinct average
    blamed = []
    for i in input_indices:
        if header[i] == 'criterion':
            # Symbolic: check if a specific criterion is over-represented in bad rows
            bad_vals = [r[i] for r in bad_rows]
            good_vals = [r[i] for r in good_rows]
            if bad_vals.count('squared_error') > good_vals.count('squared_error'):
                blamed.append(header[i])
        else:
            # Numeric: check if mean is significantly higher/lower
            bad_avg = sum(r[i] for r in bad_rows) / len(bad_rows)
            good_avg = sum(r[i] for r in good_rows) / len(good_rows)
            if abs(bad_avg - good_avg) / (good_avg + 1e-9) > 0.2:
                blamed.append(header[i])
                
    return blamed

import csv

def justify(header, rows, row):
    """
    Estimates the performance of 'row' by comparing it against all 'rows'
    using a k-Nearest Neighbors approach to estimate the missing goals,
    then calculating the fraction of rows that this row outperforms.
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith(('-', '+'))]
    input_indices = [i for i, name in enumerate(header) if not name.endswith(('-', '+'))]
    
    # 1. Estimate the goals for the target row using k-NN
    def get_dist(r1, r2):
        d = 0
        for i in input_indices:
            if header[i] == 'criterion':
                d += 1.0 if r1[i] != r2[i] else 0.0
            else:
                d += (r1[i] - r2[i])**2
        return d**0.5

    distances = sorted([(get_dist(row, r), r) for r in rows], key=lambda x: x[0])
    k = 5
    neighbors = distances[:k]
    
    estimated_goals = []
    for g_idx in goal_indices:
        avg = sum(n[1][g_idx] for n in neighbors) / k
        estimated_goals.append(avg)

    # 2. Compare estimated row against all rows
    def is_better(r_goals, target_goals):
        # A row is better if it is closer to the ideal for all goals
        # We use a simple dominance check: better if it beats the target on average
        score_r = 0
        score_t = 0
        for i, g_idx in enumerate(goal_indices):
            # Normalize/compare: lower is better for '-', higher for '+'
            direction = -1 if header[g_idx].endswith('-') else 1
            if (r_goals[i] * direction) > (target_goals[i] * direction):
                score_r += 1
            else:
                score_t += 1
        return score_r > score_t

    better_count = 0
    for r in rows:
        r_goals = [r[g] for g in goal_indices]
        if is_better(r_goals, estimated_goals):
            better_count += 1
            
    p = better_count / len(rows)
    reason = f"This setup outperforms {int(p * 100)}% of the known configurations based on its estimated performance metrics."
    return (p, reason)

import csv
import random

def choose(header, rows, label):
    """
    Finds the best row by sampling 50 setups.
    We use a simple random sampling strategy to explore the space,
    labeling 50 rows and picking the one with the best multi-objective score.
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith(('-', '+'))]
    
    # Randomly select 50 indices to label
    indices_to_sample = random.sample(range(len(rows)), min(50, len(rows)))
    
    best_idx = -1
    best_score = -float('inf')
    
    # We need to normalize goals to compare them. 
    # Since we only have 50 samples, we normalize based on those samples.
    sampled_data = []
    for i in indices_to_sample:
        goals = label(i)
        sampled_data.append((i, goals))
        
    # Calculate min/max for normalization based on the 50 samples
    mins = [min(s[1][j] for s in sampled_data) for j in range(len(goal_indices))]
    maxs = [max(s[1][j] for s in sampled_data) for j in range(len(goal_indices))]
    
    for i, goals in sampled_data:
        score = 0
        for j, g_idx in enumerate(goal_indices):
            val = goals[j]
            rng = maxs[j] - mins[j]
            norm = (val - mins[j]) / rng if rng > 0 else 0
            # MRE- (minimize), ACC+ (maximize), PRED40+ (maximize)
            if header[g_idx].endswith('-'):
                score -= norm
            else:
                score += norm
        
        if score > best_score:
            best_score = score
            best_idx = i
            
    return best_idx
