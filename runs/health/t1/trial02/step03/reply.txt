import csv

def certify(xs, ys):
    """
    Returns True if the distributions of xs and ys are statistically 
    indistinguishable (same apart from noise), False otherwise.
    Uses a simple A12 effect size check (Vargha-Delaney).
    """
    if not xs or not ys:
        return True
    
    n_x, n_y = len(xs), len(ys)
    
    # Count how often x > y
    greater = 0
    equal = 0
    for x in xs:
        for y in ys:
            if x > y:
                greater += 1
            elif x == y:
                equal += 1
                
    a12 = (greater + 0.5 * equal) / (n_x * n_y)
    
    # A12 threshold: 0.56 is a small effect size.
    # If A12 is between 0.44 and 0.56, they are effectively the same.
    return 0.44 <= a12 <= 0.56

def analyze(header, rows):
    """
    Compares 'big' setups (high N_estimators) vs 'small' setups.
    """
    idx_n = header.index('N_estimators')
    idx_acc = header.index('ACC+')
    
    # Split by median of N_estimators
    n_values = [r[idx_n] for r in rows]
    median_n = sorted(n_values)[len(n_values) // 2]
    
    big = [r[idx_acc] for r in rows if r[idx_n] > median_n]
    small = [r[idx_acc] for r in rows if r[idx_n] <= median_n]
    
    is_same = certify(big, small)
    
    print(f"Analysis: Comparing ACC+ for {len(big)} big setups vs {len(small)} small setups.")
    if is_same:
        print("Result: The difference is likely just noise.")
    else:
        print("Result: There is a statistically significant difference between setups.")
    return is_same

if __name__ == "__main__":
    filename = "Health-ClosedIssues0000.n1000.train.csv"
    rows = []
    try:
        with open(filename, 'r') as f:
            reader = csv.reader(f)
            header = next(reader)
            for row in reader:
                processed_row = []
                for i, val in enumerate(row):
                    if header[i] == 'criterion':
                        processed_row.append(val)
                    else:
                        processed_row.append(float(val))
                rows.append(processed_row)
        
        analyze(header, rows)
        
    except FileNotFoundError:
        print(f"Error: {filename} not found.")
        exit(1)
