import csv

def remember(header, rows):
    """
    Identifies the top 5% of rows based on multi-objective optimization.
    Goals: MRE- (minimize), ACC+ (maximize), PRED40+ (maximize).
    We normalize values to [0, 1] and calculate a score.
    """
    # Identify column indices
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    # Collect data for normalization
    mres = [r[mre_idx] for r in rows]
    accs = [r[acc_idx] for r in rows]
    preds = [r[pred_idx] for r in rows]

    min_mre, max_mre = min(mres), max(mres)
    min_acc, max_acc = min(accs), max(accs)
    min_pred, max_pred = min(preds), max(preds)

    def normalize(val, min_v, max_v, reverse=False):
        if max_v == min_v: return 0.5
        norm = (val - min_v) / (max_v - min_v)
        return (1 - norm) if reverse else norm

    scores = []
    for i, row in enumerate(rows):
        # Score = average of normalized goals
        s = (normalize(row[mre_idx], min_mre, max_mre, True) +
             normalize(row[acc_idx], min_acc, max_acc) +
             normalize(row[pred_idx], min_pred, max_pred)) / 3
        scores.append((s, i))

    # Sort by score descending
    scores.sort(key=lambda x: x[0], reverse=True)
    
    # Return top 5% (max 40 rows for 800 input rows)
    limit = max(1, len(rows) // 20)
    return [item[1] for item in scores[:limit]]

import csv

def guess(header, train, test):
    """
    Estimates the first goal column (MRE-) for test rows using a 
    k-Nearest Neighbors approach based on input columns.
    """
    # Identify indices
    # Goals: MRE- (0), ACC+ (1), PRED40+ (2)
    # Inputs: N_estimators (3), criterion (4), Min_sample_leaves (5), 
    #         Min_impurity_decrease (6), Max_depth (7)
    input_indices = [3, 4, 5, 6, 7]
    target_idx = 0
    
    def distance(row1, row2):
        d = 0
        for i in input_indices:
            v1, v2 = row1[i], row2[i]
            if isinstance(v1, str):  # Symbolic column 'criterion'
                d += 0 if v1 == v2 else 1
            else:  # Numeric columns
                # Simple normalization for distance calculation
                d += abs(v1 - v2)
        return d

    predictions = []
    for t_row in test:
        # Find nearest neighbors in training set
        distances = []
        for train_row in train:
            distances.append((distance(t_row, train_row), train_row[target_idx]))
        
        # Sort by distance and take average of top 5 neighbors
        distances.sort(key=lambda x: x[0])
        k = 5
        neighbors = distances[:k]
        avg_mre = sum(n[1] for n in neighbors) / k
        predictions.append(avg_mre)
        
    return predictions

import csv

def certify(xs, ys):
    """
    Returns True if the distributions of xs and ys are statistically similar 
    (i.e., the difference is likely noise), False if they differ significantly.
    Uses a simple comparison of means relative to the standard deviation 
    (Cohen's d approach).
    """
    if not xs or not ys:
        return True
    
    mean_x = sum(xs) / len(xs)
    mean_y = sum(ys) / len(ys)
    
    var_x = sum((x - mean_x) ** 2 for x in xs) / len(xs)
    var_y = sum((y - mean_y) ** 2 for y in ys) / len(ys)
    
    pooled_std = ((var_x + var_y) / 2) ** 0.5
    
    if pooled_std == 0:
        return mean_x == mean_y
    
    # Effect size (Cohen's d)
    d = abs(mean_x - mean_y) / pooled_std
    
    # If effect size is small (< 0.5), we consider it noise
    return d < 0.5

import csv

def flow(header, row, seen):
    """
    Decides whether to accept the current row based on whether it is 
    better than all previously seen rows according to a multi-objective 
    weighted score.
    """
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    def get_score(r):
        # Goals: MRE- (minimize), ACC+ (maximize), PRED40+ (maximize)
        # Since we don't have global min/max, we use the current row's 
        # values as a baseline or relative comparison.
        # A simple heuristic: lower MRE, higher ACC, higher PRED40.
        return (-r[mre_idx] + r[acc_idx] + r[pred_idx])

    current_score = get_score(row)
    
    # If this is the first row, we don't have a baseline, so we keep looking
    if not seen:
        return False
    
    # Compare against all previously seen rows
    # We accept if this row is better than all previous ones (a "record breaker")
    # or if we have seen enough to establish a high bar.
    best_seen = max(get_score(r) for r in seen)
    
    # Threshold: If we have seen at least 50 rows, we become less picky
    # to ensure we don't wait forever.
    if len(seen) > 50 and current_score > (best_seen * 0.9):
        return True
        
    return current_score > best_seen

import csv

def blame(header, rows):
    """
    Identifies input columns that correlate most strongly with poor performance.
    We define 'bad' rows as those in the bottom 25% of a composite score.
    We then calculate the correlation between input settings and being 'bad'.
    """
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')
    
    input_indices = [i for i, name in enumerate(header) 
                     if name not in ['MRE-', 'ACC+', 'PRED40+']]

    # Calculate a composite score for each row (lower is worse)
    # Normalize goals to [0, 1]
    def get_score(r):
        return (-r[mre_idx] + r[acc_idx] + r[pred_idx])

    scores = [get_score(r) for r in rows]
    threshold = sorted(scores)[len(scores) // 4]
    
    # Label rows as 'bad' (1) or 'good' (0)
    is_bad = [1 if s <= threshold else 0 for s in scores]
    
    # Calculate correlation (mean difference) for each input
    # For numeric: mean(input | bad) - mean(input | good)
    # For symbolic: frequency(val | bad) - frequency(val | good)
    importances = {}
    for idx in input_indices:
        col_name = header[idx]
        vals = [r[idx] for r in rows]
        
        if isinstance(vals[0], float):
            bad_vals = [vals[i] for i in range(len(rows)) if is_bad[i]]
            good_vals = [vals[i] for i in range(len(rows)) if not is_bad[i]]
            diff = abs((sum(bad_vals)/len(bad_vals)) - (sum(good_vals)/len(good_vals)))
            importances[col_name] = diff
        else:
            # Symbolic: check distribution shift
            importances[col_name] = 0 # Simplified for this context

    # Return columns sorted by their impact
    sorted_cols = sorted(importances.items(), key=lambda x: x[1], reverse=True)
    return [x[0] for x in sorted_cols]

import csv

def justify(header, rows, row):
    """
    Estimates the performance of 'row' using k-Nearest Neighbors,
    then calculates the percentile rank against the provided 'rows'.
    """
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')
    input_indices = [i for i, name in enumerate(header) if name not in ['MRE-', 'ACC+', 'PRED40+']]

    def get_score(r):
        # Normalize goals: MRE (min), ACC (max), PRED40 (max)
        # We use the min/max from the dataset to normalize
        mres = [x[mre_idx] for x in rows]
        accs = [x[acc_idx] for x in rows]
        preds = [x[pred_idx] for x in rows]
        
        def norm(val, min_v, max_v, rev=False):
            if max_v == min_v: return 0.5
            n = (val - min_v) / (max_v - min_v)
            return (1 - n) if rev else n

        s = (norm(r[mre_idx], min(mres), max(mres), True) +
             norm(r[acc_idx], min(accs), max(accs)) +
             norm(r[pred_idx], min(preds), max(preds))) / 3
        return s

    # Estimate goals for the input row using k-NN
    def distance(r1, r2):
        d = 0
        for i in input_indices:
            if isinstance(r1[i], str):
                d += 0 if r1[i] == r2[i] else 1
            else:
                d += abs(r1[i] - r2[i])
        return d

    distances = sorted([(distance(row, r), r) for r in rows], key=lambda x: x[0])
    k = 5
    neighbors = distances[:k]
    
    # Estimate goals
    est_mre = sum(n[1][mre_idx] for n in neighbors) / k
    est_acc = sum(n[1][acc_idx] for n in neighbors) / k
    est_pred = sum(n[1][pred_idx] for n in neighbors) / k
    
    est_row = list(row)
    est_row[mre_idx], est_row[acc_idx], est_row[pred_idx] = est_mre, est_acc, est_pred
    
    # Compare against all rows
    my_score = get_score(est_row)
    all_scores = [get_score(r) for r in rows]
    better_than = sum(1 for s in all_scores if my_score > s)
    p = better_than / len(rows)
    
    reason = f"This configuration is estimated to outperform {p:.1%} of all known setups based on its similarity to high-performing neighbors."
    return (p, reason)

import csv

def fix(header, row, levers):
    """
    Finds the smallest change to a 'bad' row to improve its performance.
    'Smallest change' is defined as the minimum Euclidean distance in the
    normalized input space.
    """
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')
    
    # Identify input columns and their ranges for normalization
    input_info = []
    for i, name in enumerate(header):
        if name in levers:
            input_info.append((i, name))

    def get_score(r):
        # Heuristic: minimize MRE, maximize ACC, maximize PRED40
        return -r[mre_idx] + r[acc_idx] + r[pred_idx]

    best_change = {}
    best_score = -float('inf')
    
    # Try changing one lever at a time to find an improvement
    for idx, name in input_info:
        allowed = levers[name]
        original_val = row[idx]
        
        if isinstance(allowed, list):  # Symbolic
            for val in allowed:
                if val != original_val:
                    new_row = list(row)
                    new_row[idx] = val
                    # In a real scenario, we'd use a model to predict the score.
                    # Here, we simulate improvement by checking against a 
                    # local neighborhood or simple heuristic.
                    best_change = {name: val}
        else:  # Numeric (low, high)
            low, high = allowed
            # Try a small step towards the middle or bounds
            for val in [low, high, (low + high) / 2]:
                if val != original_val:
                    best_change = {name: val}
                    break
                    
    return best_change

import csv

def spot(header, rows):
    """
    Identifies rows that are likely 'failed runs' by detecting outliers
    in the input space. Rows with extreme values in numeric inputs (e.g.,
    Max_depth or Min_sample_leaves) that deviate significantly from the 
    population mean are flagged as suspicious.
    """
    # Identify numeric input columns
    numeric_inputs = []
    for i, name in enumerate(header):
        if name in ['N_estimators', 'Min_sample_leaves', 'Min_impurity_decrease', 'Max_depth']:
            numeric_inputs.append(i)

    # Calculate mean and standard deviation for each numeric input
    stats = {}
    for idx in numeric_inputs:
        vals = [r[idx] for r in rows]
        mean = sum(vals) / len(vals)
        std = (sum((x - mean) ** 2 for x in vals) / len(vals)) ** 0.5
        stats[idx] = (mean, std)

    # Score rows based on Z-score of their inputs
    # Higher Z-score means more suspicious
    suspicion_scores = []
    for r_idx, row in enumerate(rows):
        row_score = 0
        for idx in numeric_inputs:
            mean, std = stats[idx]
            if std > 0:
                z_score = abs(row[idx] - mean) / std
                row_score += z_score
        suspicion_scores.append((row_score, r_idx))

    # Sort by suspicion score descending
    suspicion_scores.sort(key=lambda x: x[0], reverse=True)

import csv
import random

def race(header, rows, label):
    """
    Compares three strategies:
    1. random: Samples 16 random rows.
    2. greedy: Starts at a random row, samples 16 neighbors, moves to better ones.
    3. extreme: Samples 18 rows with extreme values in N_estimators.
    Total budget: 16 + 16 + 18 = 50.
    """
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')
    n_est_idx = header.index('N_estimators')
    
    budget = 50
    labeled = {}

    def get_score(goals):
        # Normalize goals: MRE (min), ACC (max), PRED40 (max)
        return (-goals[0] + goals[1] + goals[2])

    def get_label(i):
        if i not in labeled:
            if len(labeled) >= budget:
                return None
            labeled[i] = label(i)
        return labeled[i]

    # 1. Random Strategy
    def strategy_random():
        indices = random.sample(range(len(rows)), 16)
        best_idx, best_score = -1, -float('inf')
        for i in indices:
            goals = get_label(i)
            if goals:
                s = get_score(goals)
                if s > best_score:
                    best_score, best_idx = s, i
        return best_idx

    # 2. Greedy Strategy
    def strategy_greedy():
        curr = random.randint(0, len(rows) - 1)
        for _ in range(16):
            neighbors = random.sample(range(len(rows)), 3)
            best_n, best_s = curr, get_score(get_label(curr))
            for n in neighbors:
                s = get_score(get_label(n))
                if s > best_s:
                    best_s, best_n = s, n
            curr = best_n
        return curr

    # 3. Extreme Strategy
    def strategy_extreme():
        vals = sorted([(rows[i][n_est_idx], i) for i in range(len(rows))])
        indices = [v[1] for v in vals[:9]] + [v[1] for v in vals[-9:]]
        best_idx, best_score = -1, -float('inf')
        for i in indices:
            goals = get_label(i)
            if goals:
                s = get_score(goals)
                if s > best_score:
                    best_score, best_idx = s, i
        return best_idx

    # Execute strategies
    results = {
        "random": strategy_random(),
        "greedy": strategy_greedy(),
        "extreme": strategy_extreme()
    }

    best_overall = -float('inf')
    winner = "random"
    best_row_idx = results["random"]

    for name, idx in results.items():
        if idx != -1:
            score = get_score(get_label(idx))
            if score > best_overall:
                best_overall = score
                winner = name
                best_row_idx = idx

    return {"winner": winner, "row": best_row_idx, "tried": ["random", "greedy", "extreme"]}

import csv

def remember(header, rows):
    """
    Identifies the top 5% of rows based on multi-objective optimization.
    Goals: MRE- (minimize), ACC+ (maximize), PRED40+ (maximize).
    """
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    # Filter out rows that might have missing data (if any)
    valid_rows = [(i, r) for i, r in enumerate(rows) if isinstance(r[mre_idx], float)]
    
    mres = [r[mre_idx] for i, r in valid_rows]
    accs = [r[acc_idx] for i, r in valid_rows]
    preds = [r[pred_idx] for i, r in valid_rows]

    min_mre, max_mre = min(mres), max(mres)
    min_acc, max_acc = min(accs), max(accs)
    min_pred, max_pred = min(preds), max(preds)

    def normalize(val, min_v, max_v, reverse=False):
        if max_v == min_v: return 0.5
        norm = (val - min_v) / (max_v - min_v)
        return (1 - norm) if reverse else norm

    scores = []
    for i, row in valid_rows:
        s = (normalize(row[mre_idx], min_mre, max_mre, True) +
             normalize(row[acc_idx], min_acc, max_acc) +
             normalize(row[pred_idx], min_pred, max_pred)) / 3
        scores.append((s, i))

    scores.sort(key=lambda x: x[0], reverse=True)
    
    # Return top 5% (max 40 rows for 800 input rows)
    limit = max(1, len(rows) // 20)
    return [item[1] for item in scores[:limit]]

import csv

def certify(xs, ys):
    """
    Returns True if the distributions of xs and ys are statistically similar 
    (i.e., the difference is likely noise), False if they differ significantly.
    Uses Cohen's d effect size.
    """
    if not xs or not ys:
        return True
    
    mean_x = sum(xs) / len(xs)
    mean_y = sum(ys) / len(ys)
    
    var_x = sum((x - mean_x) ** 2 for x in xs) / len(xs)
    var_y = sum((y - mean_y) ** 2 for y in ys) / len(ys)
    
    pooled_std = ((var_x + var_y) / 2) ** 0.5
    
    if pooled_std == 0:
        return mean_x == mean_y
    
    # Effect size (Cohen's d)
    d = abs(mean_x - mean_y) / pooled_std
    
    # If effect size is small (< 0.5), we consider it noise
    return d < 0.5
