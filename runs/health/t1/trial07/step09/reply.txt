import csv

def spot(header, rows):
    """
    Identifies rows that are likely 'failed runs' by detecting outliers
    in the input space. Rows with extreme values in numeric inputs (e.g.,
    Max_depth or Min_sample_leaves) that deviate significantly from the 
    population mean are flagged as suspicious.
    """
    # Identify numeric input columns
    numeric_inputs = []
    for i, name in enumerate(header):
        if name in ['N_estimators', 'Min_sample_leaves', 'Min_impurity_decrease', 'Max_depth']:
            numeric_inputs.append(i)

    # Calculate mean and standard deviation for each numeric input
    stats = {}
    for idx in numeric_inputs:
        vals = [r[idx] for r in rows]
        mean = sum(vals) / len(vals)
        std = (sum((x - mean) ** 2 for x in vals) / len(vals)) ** 0.5
        stats[idx] = (mean, std)

    # Score rows based on Z-score of their inputs
    # Higher Z-score means more suspicious
    suspicion_scores = []
    for r_idx, row in enumerate(rows):
        row_score = 0
        for idx in numeric_inputs:
            mean, std = stats[idx]
            if std > 0:
                z_score = abs(row[idx] - mean) / std
                row_score += z_score
        suspicion_scores.append((row_score, r_idx))

    # Sort by suspicion score descending
    suspicion_scores.sort(key=lambda x: x[0], reverse=True)
    
    # Return indices of the top 5% most suspicious rows
    limit = max(1, len(rows) // 20)
    return [item[1] for item in suspicion_scores[:limit]]

if __name__ == "__main__":
    filename = "Health-ClosedIssues0000.n1000.train.csv"
    rows = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = []
            for i, val in enumerate(row):
                if header[i] in ['MRE-', 'ACC+', 'PRED40+', 'N_estimators', 
                                 'Min_sample_leaves', 'Min_impurity_decrease', 'Max_depth']:
                    processed_row.append(float(val))
                else:
                    processed_row.append(val)
            rows.append(processed_row)
    
    suspicious_indices = spot(header, rows)
    
    print("Analysis: Identifying potentially failed/erroneous runs.")
    print(f"Found {len(suspicious_indices)} rows with suspicious input configurations.")
    print("Top 5 most suspicious row indices:", suspicious_indices[:5])
    print("\nThese rows contain input values that deviate significantly from the norm,")
    print("suggesting they may be outliers or failed experimental configurations.")
