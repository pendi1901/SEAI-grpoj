import csv

def remember(header, rows):
    """
    Identifies the top 5% of rows based on multi-objective optimization.
    Goals: MRE- (minimize), ACC+ (maximize), PRED40+ (maximize).
    We normalize values to [0, 1] and calculate a score.
    """
    # Identify column indices
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    # Collect data for normalization
    mres = [r[mre_idx] for r in rows]
    accs = [r[acc_idx] for r in rows]
    preds = [r[pred_idx] for r in rows]

    min_mre, max_mre = min(mres), max(mres)
    min_acc, max_acc = min(accs), max(accs)
    min_pred, max_pred = min(preds), max(preds)

    def normalize(val, min_v, max_v, reverse=False):
        if max_v == min_v: return 0.5
        norm = (val - min_v) / (max_v - min_v)
        return (1 - norm) if reverse else norm

    scores = []
    for i, row in enumerate(rows):
        # Score = average of normalized goals
        s = (normalize(row[mre_idx], min_mre, max_mre, True) +
             normalize(row[acc_idx], min_acc, max_acc) +
             normalize(row[pred_idx], min_pred, max_pred)) / 3
        scores.append((s, i))

    # Sort by score descending
    scores.sort(key=lambda x: x[0], reverse=True)
    
    # Return top 5% (max 40 rows for 800 input rows)
    limit = max(1, len(rows) // 20)
    return [item[1] for item in scores[:limit]]

import csv

def guess(header, train, test):
    """
    Estimates the first goal column (MRE-) for test rows using a 
    k-Nearest Neighbors approach based on input columns.
    """
    # Identify indices
    # Goals: MRE- (0), ACC+ (1), PRED40+ (2)
    # Inputs: N_estimators (3), criterion (4), Min_sample_leaves (5), 
    #         Min_impurity_decrease (6), Max_depth (7)
    input_indices = [3, 4, 5, 6, 7]
    target_idx = 0
    
    def distance(row1, row2):
        d = 0
        for i in input_indices:
            v1, v2 = row1[i], row2[i]
            if isinstance(v1, str):  # Symbolic column 'criterion'
                d += 0 if v1 == v2 else 1
            else:  # Numeric columns
                # Simple normalization for distance calculation
                d += abs(v1 - v2)
        return d

    predictions = []
    for t_row in test:
        # Find nearest neighbors in training set
        distances = []
        for train_row in train:
            distances.append((distance(t_row, train_row), train_row[target_idx]))
        
        # Sort by distance and take average of top 5 neighbors
        distances.sort(key=lambda x: x[0])
        k = 5
        neighbors = distances[:k]
        avg_mre = sum(n[1] for n in neighbors) / k
        predictions.append(avg_mre)
        
    return predictions

import csv

def certify(xs, ys):
    """
    Returns True if the distributions of xs and ys are statistically similar 
    (i.e., the difference is likely noise), False if they differ significantly.
    Uses a simple comparison of means relative to the standard deviation 
    (Cohen's d approach).
    """
    if not xs or not ys:
        return True
    
    mean_x = sum(xs) / len(xs)
    mean_y = sum(ys) / len(ys)
    
    var_x = sum((x - mean_x) ** 2 for x in xs) / len(xs)
    var_y = sum((y - mean_y) ** 2 for y in ys) / len(ys)
    
    pooled_std = ((var_x + var_y) / 2) ** 0.5
    
    if pooled_std == 0:
        return mean_x == mean_y
    
    # Effect size (Cohen's d)
    d = abs(mean_x - mean_y) / pooled_std
    
    # If effect size is small (< 0.5), we consider it noise
    return d < 0.5

import csv

def flow(header, row, seen):
    """
    Decides whether to accept the current row based on whether it is 
    better than all previously seen rows according to a multi-objective 
    weighted score.
    """
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    def get_score(r):
        # Goals: MRE- (minimize), ACC+ (maximize), PRED40+ (maximize)
        # Since we don't have global min/max, we use the current row's 
        # values as a baseline or relative comparison.
        # A simple heuristic: lower MRE, higher ACC, higher PRED40.
        return (-r[mre_idx] + r[acc_idx] + r[pred_idx])

    current_score = get_score(row)
    
    # If this is the first row, we don't have a baseline, so we keep looking
    if not seen:
        return False
    
    # Compare against all previously seen rows
    # We accept if this row is better than all previous ones (a "record breaker")
    # or if we have seen enough to establish a high bar.
    best_seen = max(get_score(r) for r in seen)
    
    # Threshold: If we have seen at least 50 rows, we become less picky
    # to ensure we don't wait forever.
    if len(seen) > 50 and current_score > (best_seen * 0.9):
        return True
        
    return current_score > best_seen
