import csv

def certify(xs, ys):
    """
    Returns True if the distributions of xs and ys are statistically similar 
    (i.e., the difference is likely noise), False if they differ significantly.
    Uses Cohen's d effect size.
    """
    if not xs or not ys:
        return True
    
    mean_x = sum(xs) / len(xs)
    mean_y = sum(ys) / len(ys)
    
    var_x = sum((x - mean_x) ** 2 for x in xs) / len(xs)
    var_y = sum((y - mean_y) ** 2 for y in ys) / len(ys)
    
    pooled_std = ((var_x + var_y) / 2) ** 0.5
    
    if pooled_std == 0:
        return mean_x == mean_y
    
    # Effect size (Cohen's d)
    d = abs(mean_x - mean_y) / pooled_std
    
    # If effect size is small (< 0.5), we consider it noise
    return d < 0.5

if __name__ == "__main__":
    filename = "Health-ClosedIssues0000.n1000.train.csv"
    rows = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = []
            for i, val in enumerate(row):
                if header[i] in ['MRE-', 'ACC+', 'PRED40+', 'N_estimators', 
                                 'Min_sample_leaves', 'Min_impurity_decrease', 'Max_depth']:
                    processed_row.append(float(val))
                else:
                    processed_row.append(val)
            rows.append(processed_row)

    # The customer asks about the difference between two rules.
    # We compare the 'criterion' rule: 'absolute_error' vs 'squared_error'
    # as a proxy for "built-in rules" affecting the ACC+ goal.
    crit_idx = header.index('criterion')
    acc_idx = header.index('ACC+')
    
    abs_error_acc = [r[acc_idx] for r in rows if r[crit_idx] == 'absolute_error']
    sq_error_acc = [r[acc_idx] for r in rows if r[crit_idx] == 'squared_error']
    
    is_noise = certify(abs_error_acc, sq_error_acc)
    
    print("Analysis: Does the choice of criterion rule make a real difference?")
    if is_noise:
        print("Result: The difference between rules is statistically insignificant (noise).")
        print("You can stop arguing about it; the choice does not meaningfully impact accuracy.")
    else:
        print("Result: There is a statistically significant difference between the rules.")
        print("The choice of rule does matter for your outcomes.")
        
    print(f"\nMean Accuracy (absolute_error): {sum(abs_error_acc)/len(abs_error_acc):.4f}")
    print(f"Mean Accuracy (squared_error):  {sum(sq_error_acc)/len(sq_error_acc):.4f}")
