import csv

def remember(header, rows):
    """
    Identifies the top 5% of rows based on multi-objective optimization.
    Goals: MRE- (minimize), ACC+ (maximize), PRED40+ (maximize).
    We normalize the values and use a simple weighted sum (Pareto-like approach).
    """
    n = len(rows)
    if n == 0:
        return []
    
    # Identify column indices
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')
    
    # Calculate scores
    # Since we want to minimize MRE and maximize ACC/PRED, 
    # we use (1 - MRE) + ACC + PRED as a proxy for "trustworthiness"
    scores = []
    for i, row in enumerate(rows):
        mre = row[mre_idx]
        acc = row[acc_idx]
        pred = row[pred_idx]
        
        # Score calculation: higher is better
        score = (1.0 - mre) + acc + pred
        scores.append((score, i))
    
    # Sort by score descending
    scores.sort(key=lambda x: x[0], reverse=True)
    
    # Return top 5% (or at least 1)
    limit = max(1, n // 20)
    return [item[1] for item in scores[:limit]]

import csv

def guess(header, train, test):
    """
    Estimates the first goal column (MRE-) for test rows using 
    a K-Nearest Neighbors approach based on input columns.
    """
    # Identify goal and input columns
    # Goal: MRE- (index 0)
    # Inputs: N_estimators, criterion, Min_sample_leaves, Min_impurity_decrease, Max_depth
    input_indices = [i for i, name in enumerate(header) if name not in ['MRE-', 'ACC+', 'PRED40+']]
    goal_idx = 0
    
    def distance(row1, row2):
        d = 0
        for i in input_indices:
            val1, val2 = row1[i], row2[i]
            if isinstance(val1, str):
                d += 0 if val1 == val2 else 1
            else:
                # Normalize numeric distance
                d += abs(val1 - val2)
        return d

    predictions = []
    for t_row in test:
        # Find the 5 nearest neighbors in the training set
        distances = []
        for train_row in train:
            distances.append((distance(t_row, train_row), train_row[goal_idx]))
        
        distances.sort(key=lambda x: x[0])
        
        # Average the goal value of the 5 nearest neighbors
        k = 5
        neighbors = distances[:k]
        avg_mre = sum(n[1] for n in neighbors) / k
        predictions.append(avg_mre)
        
    return predictions

import csv

def certify(xs, ys):
    """
    Returns True if the distributions of xs and ys are statistically similar
    (i.e., the difference is likely just noise), False if they differ.
    Uses a simple A12 effect size test (non-parametric).
    """
    if not xs or not ys:
        return True
    
    def a12(lst1, lst2):
        m, n = len(lst1), len(lst2)
        greater = 0
        for x in lst1:
            for y in lst2:
                if x > y: greater += 1
                elif x == y: greater += 0.5
        return greater / (m * n)

    # A12 threshold: 0.56 is a small effect size, 0.64 medium, 0.71 large.
    # If the effect size is small, we consider them "the same apart from noise".
    score = a12(xs, ys)
    return abs(0.5 - score) < 0.14

import csv

def flow(header, row, seen):
    """
    Decides if the current row is 'good enough' to stop searching.
    We define 'good enough' as being in the top 5% of the potential 
    distribution based on the goals: MRE- (min), ACC+ (max), PRED40+ (max).
    """
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    def get_score(r):
        # Normalize goals: (1 - MRE) + ACC + PRED
        # Higher is better.
        return (1.0 - r[mre_idx]) + r[acc_idx] + r[pred_idx]

    # If this is the first row, we can't know if it's the best yet.
    if not seen:
        return False

    current_score = get_score(row)
    seen_scores = [get_score(r) for r in seen]
    
    # We stop if the current row is better than 95% of what we have seen so far.
    # This acts as an early stopping criterion for an online optimization.
    threshold = sorted(seen_scores)[int(len(seen_scores) * 0.95)]
    
    return current_score >= threshold

import csv

def blame(header, rows):
    """
    Identifies input columns that correlate with 'bad' performance.
    A row is 'bad' if its goals (MRE-, ACC+, PRED40+) are poor.
    We calculate a 'badness' score for each row and correlate it with input settings.
    """
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')
    
    input_cols = [i for i, name in enumerate(header) if name not in ['MRE-', 'ACC+', 'PRED40+']]
    
    # Calculate badness: High MRE, Low ACC, Low PRED
    # Normalize: MRE + (1-ACC) + (1-PRED)
    def get_badness(row):
        return row[mre_idx] + (1.0 - row[acc_idx]) + (1.0 - row[pred_idx])

    # Sort rows by badness and take the worst 20%
    rows_with_badness = sorted([(get_badness(r), r) for r in rows], key=lambda x: x[0], reverse=True)
    worst_rows = [r for _, r in rows_with_badness[:len(rows)//5]]
    
    # For each input column, calculate the average value in the worst rows
    # vs the average in the whole dataset. High deviation suggests the setting
    # is contributing to the 'badness'.
    impacts = []
    for i in input_cols:
        col_name = header[i]
        if isinstance(worst_rows[0][i], str):
            # For symbolic, check frequency of specific values
            continue
        else:
            avg_all = sum(r[i] for r in rows) / len(rows)
            avg_worst = sum(r[i] for r in worst_rows) / len(worst_rows)
            # Impact is the absolute difference in means
            impacts.append((abs(avg_worst - avg_all), col_name))
            
    # Return names of inputs sorted by impact
    impacts.sort(key=lambda x: x[0], reverse=True)
    return [name for _, name in impacts]
