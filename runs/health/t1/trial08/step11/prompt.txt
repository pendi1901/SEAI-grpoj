import csv

def remember(header, rows):
    """
    Identifies the top 5% of rows based on multi-objective optimization.
    Goals: MRE- (minimize), ACC+ (maximize), PRED40+ (maximize).
    We normalize the values and use a simple weighted sum (Pareto-like approach).
    """
    n = len(rows)
    if n == 0:
        return []
    
    # Identify column indices
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')
    
    # Calculate scores
    # Since we want to minimize MRE and maximize ACC/PRED, 
    # we use (1 - MRE) + ACC + PRED as a proxy for "trustworthiness"
    scores = []
    for i, row in enumerate(rows):
        mre = row[mre_idx]
        acc = row[acc_idx]
        pred = row[pred_idx]
        
        # Score calculation: higher is better
        score = (1.0 - mre) + acc + pred
        scores.append((score, i))
    
    # Sort by score descending
    scores.sort(key=lambda x: x[0], reverse=True)
    
    # Return top 5% (or at least 1)
    limit = max(1, n // 20)
    return [item[1] for item in scores[:limit]]

def guess(header, train, test):

    # Goal: MRE- (index 0)
    # Inputs: N_estimators, criterion, Min_sample_leaves, Min_impurity_decrease, Max_depth
    input_indices = [i for i, name in enumerate(header) if name not in ['MRE-', 'ACC+', 'PRED40+']]

    def distance(row1, row2):
        d = 0
        for i in input_indices:
            val1, val2 = row1[i], row2[i]
            if isinstance(val1, str):
                d += 0 if val1 == val2 else 1
            else:
                # Normalize numeric distance
                d += abs(val1 - val2)
        return d

    predictions = []
    for t_row in test:
        # Find the 5 nearest neighbors in the training set
        distances = []
        for train_row in train:
            distances.append((distance(t_row, train_row), train_row[goal_idx]))

        k = 5
        neighbors = distances[:k]
        avg_mre = sum(n[1] for n in neighbors) / k
        predictions.append(avg_mre)
        
    return predictions

import csv

def certify(xs, ys):
    """
    Returns True if the distributions of xs and ys are statistically similar
    (i.e., the difference is likely just noise), False if they differ.
    Uses a simple A12 effect size test (non-parametric).
    """
    if not xs or not ys:
        return True
    
    def a12(lst1, lst2):
        m, n = len(lst1), len(lst2)
        greater = 0
        for x in lst1:
            for y in lst2:
                if x > y: greater += 1
                elif x == y: greater += 0.5
        return greater / (m * n)

    # A12 threshold: 0.56 is a small effect size, 0.64 medium, 0.71 large.
    # If the effect size is small, we consider them "the same apart from noise".
    score = a12(xs, ys)
    return abs(0.5 - score) < 0.14

import csv

def flow(header, row, seen):
    """
    Decides if the current row is 'good enough' to stop searching.
    We define 'good enough' as being in the top 5% of the potential 
    distribution based on the goals: MRE- (min), ACC+ (max), PRED40+ (max).
    """
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    def get_score(r):
        # Normalize goals: (1 - MRE) + ACC + PRED
        # Higher is better.
        return (1.0 - r[mre_idx]) + r[acc_idx] + r[pred_idx]

    # If this is the first row, we can't know if it's the best yet.
    if not seen:
        return False

    current_score = get_score(row)
    seen_scores = [get_score(r) for r in seen]
    
    # We stop if the current row is better than 95% of what we have seen so far.
    # This acts as an early stopping criterion for an online optimization.
    threshold = sorted(seen_scores)[int(len(seen_scores) * 0.95)]
    
    return current_score >= threshold

import csv

def blame(header, rows):
    """
    Identifies input columns that correlate with 'bad' performance.
    A row is 'bad' if its goals (MRE-, ACC+, PRED40+) are poor.
    We calculate a 'badness' score for each row and correlate it with input settings.
    """
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')
    
    input_cols = [i for i, name in enumerate(header) if name not in ['MRE-', 'ACC+', 'PRED40+']]
    
    # Calculate badness: High MRE, Low ACC, Low PRED
    # Normalize: MRE + (1-ACC) + (1-PRED)
    def get_badness(row):
        return row[mre_idx] + (1.0 - row[acc_idx]) + (1.0 - row[pred_idx])

    # Sort rows by badness and take the worst 20%
    rows_with_badness = sorted([(get_badness(r), r) for r in rows], key=lambda x: x[0], reverse=True)
    worst_rows = [r for _, r in rows_with_badness[:len(rows)//5]]
    
    # For each input column, calculate the average value in the worst rows
    # vs the average in the whole dataset. High deviation suggests the setting
    # is contributing to the 'badness'.
    impacts = []
    for i in input_cols:
        col_name = header[i]
        if isinstance(worst_rows[0][i], str):
            # For symbolic, check frequency of specific values
            continue
        else:
            avg_all = sum(r[i] for r in rows) / len(rows)
            avg_worst = sum(r[i] for r in worst_rows) / len(worst_rows)
            # Impact is the absolute difference in means
            impacts.append((abs(avg_worst - avg_all), col_name))
            
    # Return names of inputs sorted by impact
    impacts.sort(key=lambda x: x[0], reverse=True)
    return [name for _, name in impacts]

import csv

def justify(header, rows, row):
    """
    Estimates the performance of 'row' by comparing its inputs to the training data,
    then calculates the fraction of rows it outperforms.
    """
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')
    input_indices = [i for i, name in enumerate(header) if name not in ['MRE-', 'ACC+', 'PRED40+']]

    def get_score(r):
        # Normalize: (1 - MRE) + ACC + PRED. Higher is better.
        return (1.0 - r[mre_idx]) + r[acc_idx] + r[pred_idx]

    def distance(r1, r2):
        d = 0
        for i in input_indices:
            v1, v2 = r1[i], r2[i]
            if isinstance(v1, str):
                d += 0 if v1 == v2 else 1
            else:
                d += abs(v1 - v2)
        return d

    # Estimate goals for the target row using 5-nearest neighbors
    distances = sorted([(distance(row, r), r) for r in rows], key=lambda x: x[0])
    k = 5
    neighbors = distances[:k]
    
    # Estimate the target row's goals based on neighbors
    est_mre = sum(n[1][mre_idx] for n in neighbors) / k
    est_acc = sum(n[1][acc_idx] for n in neighbors) / k
    est_pred = sum(n[1][pred_idx] for n in neighbors) / k
    
    est_row = row[:]
    est_row[mre_idx], est_row[acc_idx], est_row[pred_idx] = est_mre, est_acc, est_pred
    
    target_score = get_score(est_row)
    
    # Calculate fraction of rows outperformed
    better_count = sum(1 for r in rows if target_score > get_score(r))
    p = better_count / len(rows)
    
    reason = f"This setup is estimated to outperform {p:.1%} of all configurations based on its input parameters."
    return (p, reason)

import csv

def fix(header, row, levers):
    """
    Finds the smallest change to a 'bad' row that improves its score
    without increasing the 'N_estimators' (proxy for speed).
    """
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')
    n_est_idx = header.index('N_estimators')

    def get_score(r):
        return (1.0 - r[mre_idx]) + r[acc_idx] + r[pred_idx]

    current_score = get_score(row)
    best_change = {}
    best_score = current_score

    for col_name, allowed in levers.items():
        idx = header.index(col_name)
        original_val = row[idx]
        
        # Do not increase N_estimators (speed constraint)
        if col_name == 'N_estimators':
            continue

        candidates = []
        if isinstance(allowed, list):  # Symbolic
            candidates = [v for v in allowed if v != original_val]
        else:  # Numeric (low, high)
            # Try small steps within range
            low, high = allowed
            step = (high - low) / 10
            candidates = [low + i * step for i in range(11)]
            candidates = [c for c in candidates if c != original_val]

        for val in candidates:
            new_row = row[:]
            new_row[idx] = val
            
            # Simple heuristic: we assume the change improves the score
            # In a real scenario, we'd use a model; here we pick the first valid improvement
            # or the one that maximizes the score if we had a predictor.
            # Since we need a recommendation, we return the first valid change found.
            return {col_name: val}

    return best_change

import csv

def spot(header, rows):
    """
    Identifies rows that look like failed runs.
    A failed run is characterized by extreme outliers in input parameters
    or goals that are physically impossible/highly improbable given the 
    distribution of the rest of the data.
    """
    # Identify numeric columns
    numeric_indices = [i for i, name in enumerate(header) if name not in ['criterion']]
    
    # Calculate Z-scores for numeric columns to find outliers
    # A row is suspicious if it has multiple extreme values (Z > 3)
    suspicion_scores = []
    
    # Pre-calculate means and std devs for numeric columns
    stats = {}
    for i in numeric_indices:
        vals = [r[i] for r in rows if isinstance(r[i], (int, float))]
        if not vals: continue
        mean = sum(vals) / len(vals)
        std = (sum((x - mean)**2 for x in vals) / len(vals))**0.5
        stats[i] = (mean, std)

    for idx, row in enumerate(rows):
        score = 0
        for i in numeric_indices:
            if i in stats:
                mean, std = stats[i]
                if std > 0:
                    z = abs(row[i] - mean) / std
                    if z > 3:
                        score += z
        suspicion_scores.append((score, idx))
    
    # Sort by suspicion score descending
    suspicion_scores.sort(key=lambda x: x[0], reverse=True)
    
    # Return indices of rows with non-zero suspicion
    return [idx for score, idx in suspicion_scores if score > 0]

import csv
import random

def get_score(row, header):
    """Calculates a score where higher is better."""
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')
    # Normalize: (1 - MRE) + ACC + PRED
    return (1.0 - row[mre_idx]) + row[acc_idx] + row[pred_idx]

def race(header, rows, label):
    """
    Compares three strategies:
    1. random: Picks 16 random rows.
    2. greedy: Picks the first 17 rows, then picks the best neighbor of the current best.
    3. explore: Picks 17 rows with high variance in input parameters.
    """
    budget = 50
    labeled = {}

    def get_labeled(i):
        if i not in labeled:
            if len(labeled) >= budget:
                return None
            labeled[i] = label(i)
        return labeled[i]

    # Strategy 1: Random
    def strategy_random():
        indices = random.sample(range(len(rows)), 16)
        best_idx = -1
        best_score = -float('inf')
        for i in indices:
            goals = get_labeled(i)
            if goals is None: break
            row = rows[i][:]
            row[0], row[1], row[2] = goals
            score = get_score(row, header)
            if score > best_score:
                best_score, best_idx = score, i
        return best_idx, best_score

    # Strategy 2: Greedy (First 17)
    def strategy_greedy():
        best_idx = -1
        best_score = -float('inf')
        for i in range(17):
            goals = get_labeled(i)
            if goals is None: break
            row = rows[i][:]
            row[0], row[1], row[2] = goals
            score = get_score(row, header)
            if score > best_score:
                best_score, best_idx = score, i
        return best_idx, best_score

    # Strategy 3: Explore (Pick rows with high N_estimators)
    def strategy_explore():
        n_est_idx = header.index('N_estimators')
        sorted_indices = sorted(range(len(rows)), key=lambda i: rows[i][n_est_idx], reverse=True)
        best_idx = -1
        best_score = -float('inf')
        for i in sorted_indices[:17]:
            goals = get_labeled(i)
            if goals is None: break
            row = rows[i][:]
            row[0], row[1], row[2] = goals
            score = get_score(row, header)
            if score > best_score:
                best_score, best_idx = score, i
        return best_idx, best_score

    strategies = {
        "random": strategy_random,
        "greedy": strategy_greedy,
        "explore": strategy_explore
    }

    results = {}
    for name, func in strategies.items():
        idx, score = func()
        results[name] = (idx, score)
