import csv
import random

def get_score(row, header):
    """
    Calculates a score where higher is better.
    Goals: MRE- (minimize), ACC+ (maximize), PRED40+ (maximize).
    """
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')
    # Normalize: (1 - MRE) + ACC + PRED
    return (1.0 - row[mre_idx]) + row[acc_idx] + row[pred_idx]

def race(header, rows, label):
    """
    Compares three strategies for finding the best row within a 50-label budget.
    """
    budget = 50
    labeled = {}

    def get_labeled(i):
        if i not in labeled:
            if len(labeled) >= budget:
                return None
            labeled[i] = label(i)
        return labeled[i]

    def evaluate(indices):
        best_idx = -1
        best_score = -float('inf')
        for i in indices:
            goals = get_labeled(i)
            if goals is None: break
            row = rows[i][:]
            row[0], row[1], row[2] = goals
            score = get_score(row, header)
            if score > best_score:
                best_score, best_idx = score, i
        return best_idx, best_score

    # 1. Random: Baseline
    def strategy_random():
        indices = random.sample(range(len(rows)), 16)
        return evaluate(indices)

    # 2. High-N: Focus on high N_estimators (often correlates with performance)
    def strategy_high_n():
        n_est_idx = header.index('N_estimators')
        sorted_indices = sorted(range(len(rows)), key=lambda i: rows[i][n_est_idx], reverse=True)
        return evaluate(sorted_indices[:17])

    # 3. Low-Depth: Focus on low Max_depth (often correlates with simplicity/generalization)
    def strategy_low_depth():
        depth_idx = header.index('Max_depth')
        sorted_indices = sorted(range(len(rows)), key=lambda i: rows[i][depth_idx])
        return evaluate(sorted_indices[:17])

    strategies = {
        "random": strategy_random,
        "high_n": strategy_high_n,
        "low_depth": strategy_low_depth
    }

    results = {}
    for name, func in strategies.items():
        idx, score = func()
        results[name] = (idx, score)

    winner = max(results, key=lambda n: results[n][1])
    return {"winner": winner, "row": results[winner][0], "tried": list(strategies.keys())}

if __name__ == "__main__":
    filename = 'Health-ClosedIssues0000.n1000.train.csv'
    rows = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed = [val if header[i] == 'criterion' else float(val) for i, val in enumerate(row)]
            rows.append(processed)

    true_goals = [[r[0], r[1], r[2]] for r in rows]
    def label(i): return true_goals[i]

    masked_rows = [r[:] for r in rows]
    for r in masked_rows: r[0], r[1], r[2] = "?", "?", "?"

    result = race(header, masked_rows, label)
    
    print("Race complete: Comparing search strategies.")
    print(f"Winner: {result['winner']} strategy found the best setup.")
    print(f"Strategies tried: {', '.join(result['tried'])}.")
    if result['winner'] != 'random':
        print("Conclusion: Targeted search strategies outperform random sampling on this dataset.")
    else:
        print("Conclusion: Random sampling remains competitive with targeted strategies.")
