import csv

def remember(header, rows):
    """
    Identifies the top 5% of rows based on multi-objective optimization.
    Goals: MRE- (minimize), ACC+ (maximize), PRED40+ (maximize).
    We normalize the values and use a simple weighted sum to rank them.
    """
    n = len(rows)
    if n == 0:
        return []

    # Identify column indices
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    # Extract values for normalization
    mres = [row[mre_idx] for row in rows]
    accs = [row[acc_idx] for row in rows]
    preds = [row[pred_idx] for row in rows]

    min_mre, max_mre = min(mres), max(mres)
    min_acc, max_acc = min(accs), max(accs)
    min_pred, max_pred = min(preds), max(preds)

    def normalize(val, min_v, max_v):
        if max_v == min_v: return 0.5
        return (val - min_v) / (max_v - min_v)

    scores = []
    for i, row in enumerate(rows):
        # Normalize: MRE (lower is better -> 1 - norm), ACC/PRED (higher is better -> norm)
        s_mre = 1 - normalize(row[mre_idx], min_mre, max_mre)
        s_acc = normalize(row[acc_idx], min_acc, max_acc)
        s_pred = normalize(row[pred_idx], min_pred, max_pred)
        
        # Aggregate score
        total = s_mre + s_acc + s_pred
        scores.append((total, i))

    # Sort by score descending
    scores.sort(key=lambda x: x[0], reverse=True)
    
    # Return top 5% (or at least 1)
    limit = max(1, int(n * 0.05))
    return [item[1] for item in scores[:limit]]

import csv

def guess(header, train, test):
    """
    Estimates the first goal (MRE-) for test rows using a K-Nearest Neighbors 
    approach based on the input columns (N_estimators, criterion, 
    Min_sample_leaves, Min_impurity_decrease, Max_depth).
    """
    # Identify indices
    goal_idx = 0  # MRE-
    input_indices = [i for i, name in enumerate(header) if name not in ['MRE-', 'ACC+', 'PRED40+']]
    
    def distance(row1, row2):
        d = 0
        for i in input_indices:
            val1, val2 = row1[i], row2[i]
            if isinstance(val1, str):
                d += 0 if val1 == val2 else 1
            else:
                # Normalize numeric inputs roughly by range (assuming standard scales)
                d += abs(val1 - val2) / 100.0
        return d

    predictions = []
    for t_row in test:
        # Find 3 nearest neighbors in training data
        neighbors = sorted(train, key=lambda r: distance(r, t_row))[:3]
        # Average the MRE- of the neighbors
        avg_mre = sum(n[goal_idx] for n in neighbors) / len(neighbors)
        predictions.append(avg_mre)
        
    return predictions

import csv

def certify(xs, ys):
    """
    Determines if two distributions are statistically different using 
    a simple A12 effect size test. Returns True if they are effectively 
    the same (small effect size), False if they differ significantly.
    """
    n_x = len(xs)
    n_y = len(ys)
    if n_x == 0 or n_y == 0:
        return True

    # A12 test: probability that a random value from xs is greater than ys
    greater = 0
    equal = 0
    for x in xs:
        for y in ys:
            if x > y:
                greater += 1
            elif x == y:
                equal += 1
    
    a12 = (greater + 0.5 * equal) / (n_x * n_y)
    
    # If A12 is close to 0.5, they are the same. 
    # Thresholds for small effect size (Vargha-Delaney) are typically 0.56.
    return 0.44 <= a12 <= 0.56

def analyze_setup_size(header, rows):
    """
    Splits data into 'big' (high N_estimators) and 'small' (low N_estimators)
    and compares their ACC+ performance.
    """
    n_idx = header.index('N_estimators')
    acc_idx = header.index('ACC+')
    
    # Calculate median to split into big/small
    estimators = [row[n_idx] for row in rows]
    median = sum(estimators) / len(estimators)
    
    small = [row[acc_idx] for row in rows if row[n_idx] <= median]
    big = [row[acc_idx] for row in rows if row[n_idx] > median]
    
    is_same = certify(small, big)
    
    return is_same, len(small), len(big)

import csv

def flow(header, row, seen):
    """
    Decides whether to stop at the current row based on whether it is 
    'good enough'. We define 'good enough' as being in the top 5% 
    of the potential distribution based on the goals MRE-, ACC+, and PRED40+.
    """
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    # If this is the first row, we have no baseline to compare against.
    if not seen:
        return False

    # Extract goals from seen rows to establish a baseline
    mres = [r[mre_idx] for r in seen]
    accs = [r[acc_idx] for r in seen]
    preds = [r[pred_idx] for r in seen]

    # Calculate current row's normalized score relative to history
    def normalize(val, vals):
        min_v, max_v = min(vals), max(vals)
        if max_v == min_v: return 0.5
        return (val - min_v) / (max_v - min_v)

    s_mre = 1 - normalize(row[mre_idx], mres + [row[mre_idx]])
    s_acc = normalize(row[acc_idx], accs + [row[acc_idx]])
    s_pred = normalize(row[pred_idx], preds + [row[pred_idx]])
    
    score = s_mre + s_acc + s_pred

    # Calculate scores of seen rows to determine the 95th percentile threshold
    seen_scores = []
    for r in seen:
        ss_mre = 1 - normalize(r[mre_idx], mres + [row[mre_idx]])
        ss_acc = normalize(r[acc_idx], accs + [row[acc_idx]])
        ss_pred = normalize(r[pred_idx], preds + [row[pred_idx]])
        seen_scores.append(ss_mre + ss_acc + ss_pred)
    
    seen_scores.sort(reverse=True)
    threshold = seen_scores[max(0, int(len(seen_scores) * 0.05))]

    # Stop if the current row is better than the 95th percentile of seen rows
    return score >= threshold

import csv

def blame(header, rows):
    """
    Identifies input columns that correlate with poor performance.
    We define 'bad' rows as those in the bottom 25% of the aggregate score.
    We then compare the distribution of input values in 'bad' rows vs 'good' rows.
    """
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')
    input_indices = [i for i, name in enumerate(header) if name not in ['MRE-', 'ACC+', 'PRED40+']]

    # Calculate aggregate scores
    mres = [r[mre_idx] for r in rows]
    accs = [r[acc_idx] for r in rows]
    preds = [r[pred_idx] for r in rows]
    
    def norm(v, vals):
        mn, mx = min(vals), max(vals)
        return (v - mn) / (mx - mn) if mx > mn else 0.5

    scores = []
    for r in rows:
        s = (1 - norm(r[mre_idx], mres)) + norm(r[acc_idx], accs) + norm(r[pred_idx], preds)
        scores.append(s)

    # Split into bottom 25% (bad) and top 25% (good)
    combined = sorted(zip(scores, rows), key=lambda x: x[0])
    n = len(combined)
    bad = [x[1] for x in combined[:n // 4]]
    good = [x[1] for x in combined[3 * n // 4:]]

    # Find inputs where the mean/mode in 'bad' differs most from 'good'
    diffs = []
    for i in input_indices:
        name = header[i]
        if isinstance(bad[0][i], float):
            bad_val = sum(r[i] for r in bad) / len(bad)
            good_val = sum(r[i] for r in good) / len(good)
            diff = abs(bad_val - good_val)
        else:
            # For symbolic, compare frequency of the most common value
            bad_vals = [r[i] for r in bad]
            good_vals = [r[i] for r in good]
            mode_bad = max(set(bad_vals), key=bad_vals.count)
            diff = abs(bad_vals.count(mode_bad) / len(bad_vals) - 
                       good_vals.count(mode_bad) / len(good_vals))
        diffs.append((diff, name))

    # Return top 2 inputs that contribute most to the difference
    diffs.sort(key=lambda x: x[0], reverse=True)
    return [d[1] for d in diffs[:2]]

import csv

def justify(header, rows, row):
    """
    Estimates the performance of 'row' by comparing it to the distribution 
    of existing rows. Returns (p, reason) where p is the fraction of rows 
    that 'row' outperforms, and reason is a human-readable explanation.
    """
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')
    input_indices = [i for i, name in enumerate(header) if name not in ['MRE-', 'ACC+', 'PRED40+']]

    # Estimate goals for the target row using 3-Nearest Neighbors
    def distance(r1, r2):
        d = 0
        for i in input_indices:
            if isinstance(r1[i], str):
                d += 0 if r1[i] == r2[i] else 1
            else:
                d += abs(r1[i] - r2[i]) / 100.0
        return d

    neighbors = sorted(rows, key=lambda r: distance(r, row))[:3]
    est_mre = sum(n[mre_idx] for n in neighbors) / len(neighbors)
    est_acc = sum(n[acc_idx] for n in neighbors) / len(neighbors)
    est_pred = sum(n[pred_idx] for n in neighbors) / len(neighbors)

    # Calculate how many rows this setup is better than
    # Better = lower MRE, higher ACC, higher PRED40
    better_count = 0
    for r in rows:
        if (est_mre < r[mre_idx] and est_acc > r[acc_idx] and est_pred > r[pred_idx]):
            better_count += 1
    
    p = better_count / len(rows)
    reason = (f"This configuration is estimated to outperform {p:.1%} of all known setups "
              "by balancing lower error rates with higher accuracy and prediction thresholds.")
    
    return p, reason

def choose(header, rows, label):
    """
    Finds the best setup by sampling 50 rows, labeling them, 
    and selecting the one with the best multi-objective score.
    """
    n = len(rows)
    # We have a budget of 50 labels.
    indices = list(range(n))
    # Randomly sample 50 indices to evaluate
    sample_indices = random.sample(indices, min(n, 50))
    
    labeled_data = []
    for i in sample_indices:
        goals = label(i)
        # Reconstruct the row with real goals
        row = rows[i][:]
        row[0], row[1], row[2] = goals
        labeled_data.append((row, i))
    
    # Identify goal indices
    mre_idx, acc_idx, pred_idx = 0, 1, 2
    
    # Normalize goals for scoring
    mres = [r[0][mre_idx] for r in labeled_data]
    accs = [r[0][acc_idx] for r in labeled_data]
    preds = [r[0][pred_idx] for r in labeled_data]
    
    def norm(val, vals):
        mn, mx = min(vals), max(vals)
        return (val - mn) / (mx - mn) if mx > mn else 0.5
    
    best_score = -1
    best_idx = -1
    
    for row, original_idx in labeled_data:
        s_mre = 1 - norm(row[mre_idx], mres)
        s_acc = norm(row[acc_idx], accs)
        s_pred = norm(row[pred_idx], preds)
        score = s_mre + s_acc + s_pred
        
        if score > best_score:
            best_score = score
            best_idx = original_idx
            
    return best_idx

import csv

def fix(header, row, levers):
    """
    Finds the smallest change to a 'bad' row to improve its performance.
    'Smallest change' is defined as the minimum distance in the lever space.
    """
    def get_score(r):
        # Goals: MRE- (min), ACC+ (max), PRED40+ (max)
        # We use a simple heuristic: lower MRE, higher ACC, higher PRED40
        return -r[0] + r[1] + r[2]

    # We need a reference to evaluate if a change is 'better'.
    # Since we don't have the full dataset inside the function, we assume
    # the caller provides a row that needs improvement.
    # We perform a local search over the levers.
    
    best_change = {}
    best_val = None
    
    # For this implementation, we try changing one lever at a time
    # to a value that is likely to improve the score based on the 
    # provided levers.
    for col, allowed in levers.items():
        idx = header.index(col)
        current_val = row[idx]
        
        if isinstance(allowed, list):  # Symbolic
            for val in allowed:
                if val != current_val:
                    return {col: val}
        else:  # Numeric (low, high)
            low, high = allowed
            # Try moving towards the midpoint or boundaries
            mid = (low + high) / 2
            if current_val < mid:
                return {col: min(current_val + (high - low) * 0.1, high)}
            else:
                return {col: max(current_val - (high - low) * 0.1, low)}
                
    return best_change

import csv

def spot(header, rows):
    """
    Identifies rows that are 'suspicious' (outliers).
    We define suspicious rows as those that have extreme input values 
    (e.g., values far from the mean/median) or rows that are 
    statistically inconsistent with the rest of the dataset.
    """
    input_indices = [i for i, name in enumerate(header) if name not in ['MRE-', 'ACC+', 'PRED40+']]
    
    # Calculate means and standard deviations for numeric inputs
    stats = {}
    for i in input_indices:
        col_data = [r[i] for r in rows if isinstance(r[i], (int, float))]
        if col_data:
            mean = sum(col_data) / len(col_data)
            std = (sum((x - mean) ** 2 for x in col_data) / len(col_data)) ** 0.5
            stats[i] = (mean, std if std > 0 else 1.0)

    # Score rows based on how many standard deviations they are from the mean
    # Higher score = more suspicious
    suspicion_scores = []
    for idx, row in enumerate(rows):
        score = 0
        for i in input_indices:
            if i in stats:
                val = row[i]
                if isinstance(val, (int, float)):
                    mean, std = stats[i]
                    score += abs((val - mean) / std)
                else:
                    # For symbolic, check if it's a rare category
                    col_vals = [r[i] for r in rows]
                    freq = col_vals.count(val) / len(col_vals)
                    if freq < 0.05:  # Rare category
                        score += 2.0
        suspicion_scores.append((score, idx))

    # Sort by suspicion score descending
    suspicion_scores.sort(key=lambda x: x[0], reverse=True)
    
    # Return the indices of the top 5% most suspicious rows
    limit = max(1, int(len(rows) * 0.05))
    return [item[1] for item in suspicion_scores[:limit]]

import csv
import random

def race(header, rows, label):
    """
    Compares three strategies: random, greedy (hill climbing), and 
    a simple 'smart' sampler that avoids previously seen areas.
    The 50-label budget is shared across all strategies.
    """
    budget = 50
    labeled = {}

    def get_goals(i):
        if i not in labeled:
            if len(labeled) >= budget:
                raise Exception("Budget exceeded")
            labeled[i] = label(i)
        return labeled[i]

    def score(row_idx):
        goals = get_goals(row_idx)
        mre, acc, pred = goals
        # Goals: MRE- (minimize), ACC+ (maximize), PRED40+ (maximize)
        # We use a simple additive score: -MRE + ACC + PRED40
        return -mre + acc + pred

    # Strategy 1: Random
    def strategy_random():
        best_idx = -1
        best_val = -float('inf')
        for _ in range(16):
            idx = random.randint(0, len(rows) - 1)
            s = score(idx)
            if s > best_val:
                best_val, best_idx = s, idx
        return best_idx

    # Strategy 2: Greedy (Hill Climbing)
    def strategy_greedy():
        curr = random.randint(0, len(rows) - 1)
        # Use remaining budget for local search
        for _ in range(16):
            # Check 2 neighbors
            neighbors = random.sample(range(len(rows)), 2)
            best_n = max(neighbors, key=score)
            if score(best_n) > score(curr):
                curr = best_n
        return curr

    # Strategy 3: Exploration (Sample and pick best)
    def strategy_explore():
        # Use remaining budget
        indices = random.sample(range(len(rows)), 18)
        return max(indices, key=score)

    # Execute strategies
    res_random = strategy_random()
    res_greedy = strategy_greedy()
    res_explore = strategy_explore()

    results = {
        "random": res_random,
        "greedy": res_greedy,
        "explore": res_explore
    }

    best_strategy = max(results, key=lambda k: score(results[k]))
    
    return {
        "winner": best_strategy,
        "row": results[best_strategy],
        "tried": ["random", "greedy", "explore"]
    }

import csv

def remember(header, rows):
    """
    Identifies the top 5% of rows based on multi-objective optimization.
    Goals: MRE- (minimize), ACC+ (maximize), PRED40+ (maximize).
    """
    n = len(rows)
    if n == 0:
        return []

    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    # Extract values for normalization
    mres = [row[mre_idx] for row in rows]
    accs = [row[acc_idx] for row in rows]
    preds = [row[pred_idx] for row in rows]

    min_mre, max_mre = min(mres), max(mres)
    min_acc, max_acc = min(accs), max(accs)
    min_pred, max_pred = min(preds), max(preds)

    def normalize(val, min_v, max_v):
        if max_v == min_v: return 0.5
        return (val - min_v) / (max_v - min_v)

    scores = []
    for i, row in enumerate(rows):
        # Normalize: MRE (lower is better -> 1 - norm), ACC/PRED (higher is better -> norm)
        s_mre = 1 - normalize(row[mre_idx], min_mre, max_mre)
        s_acc = normalize(row[acc_idx], min_acc, max_acc)
        s_pred = normalize(row[pred_idx], min_pred, max_pred)
        
        # Aggregate score
        total = s_mre + s_acc + s_pred
        scores.append((total, i))

    # Sort by score descending
    scores.sort(key=lambda x: x[0], reverse=True)
    
    # Return top 5% (or at least 1)
    limit = max(1, int(n * 0.05))
    return [item[1] for item in scores[:limit]]

def guess(header, train, test):

    approach. To simulate 'halving the size of the forecaster', we reduce the 
    training set size by 50% and compare the performance.
    """
    goal_idx = 0  # MRE-
    input_indices = [i for i, name in enumerate(header) if name not in ['MRE-', 'ACC+', 'PRED40+']]
    
    def distance(row1, row2):
        d = 0
        for i in input_indices:
            val1, val2 = row1[i], row2[i]
            if isinstance(val1, str):
                d += 0 if val1 == val2 else 1
            else:
                d += abs(val1 - val2) / 100.0
        return d

    reduced_train = train[::2]

        neighbors = sorted(reduced_train, key=lambda r: distance(r, t_row))[:3]
        avg_mre = sum(n[goal_idx] for n in neighbors) / len(neighbors)
        predictions.append(avg_mre)
        
    return predictions

import csv

def certify(xs, ys):
    """
    Determines if two distributions are statistically different using 
    the A12 effect size test. Returns True if they are effectively 
    the same (small effect size), False if they differ significantly.
    """
    n_x = len(xs)
    n_y = len(ys)
    if n_x == 0 or n_y == 0:
        return True

    greater = 0
    equal = 0
    for x in xs:
        for y in ys:
            if x > y:
                greater += 1
            elif x == y:
                equal += 1
    
    a12 = (greater + 0.5 * equal) / (n_x * n_y)
    
    # Vargha-Delaney threshold: 0.56 is the standard for a small effect size.
    # If A12 is between 0.44 and 0.56, the distributions are effectively the same.
    return 0.44 <= a12 <= 0.56

def compare_rules(header, rows):
    """
    Compares two common rules for selecting configurations:
    Rule A: Favor 'absolute_error' criterion.

    rule_b = [row[acc_idx] for row in rows if row[crit_idx] == 'squared_error']
    
    return certify(rule_a, rule_b), rule_a, rule_b

import csv

def flow(header, row, seen):
    """
    Returns True if the current row is 'good enough' to stop the process.
    We define 'good enough' as being in the top 5% of the distribution 
    of all rows seen so far, based on a multi-objective score.
    """
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    # If we haven't seen enough data, keep going
    if len(seen) < 20:
        return False

    # Collect all data to normalize
    all_rows = seen + [row]
    mres = [r[mre_idx] for r in all_rows]
    accs = [r[acc_idx] for r in all_rows]
    preds = [r[pred_idx] for r in all_rows]

    def normalize(val, vals):
        min_v, max_v = min(vals), max(vals)
        if max_v == min_v: return 0.5
        return (val - min_v) / (max_v - min_v)

    # Calculate score for the current row
    s_mre = 1 - normalize(row[mre_idx], mres)
    s_acc = normalize(row[acc_idx], accs)
    s_pred = normalize(row[pred_idx], preds)
    current_score = s_mre + s_acc + s_pred

    # Calculate scores for all seen rows
    seen_scores = []
    for r in seen:
        ss_mre = 1 - normalize(r[mre_idx], mres)
        ss_acc = normalize(r[acc_idx], accs)
        ss_pred = normalize(r[pred_idx], preds)
        seen_scores.append(ss_mre + ss_acc + ss_pred)
    
    seen_scores.sort(reverse=True)
    
    # Stop if current row is better than the 95th percentile of seen rows
    threshold_idx = max(0, int(len(seen_scores) * 0.05))
    threshold = seen_scores[threshold_idx]
    
    return current_score >= threshold

import csv

def blame(header, rows):
    """
    Identifies input columns that have the least impact on performance.
    We define 'bad' rows as those in the bottom 25% of the aggregate score.
    We compare the distribution of input values in 'bad' rows vs 'good' rows.
    Columns where the distributions are statistically similar (small difference)
    are candidates for factory defaults.
    """
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')
    input_indices = [i for i, name in enumerate(header) if name not in ['MRE-', 'ACC+', 'PRED40+']]

    # Calculate aggregate scores
    mres = [r[mre_idx] for r in rows]
    accs = [r[acc_idx] for r in rows]
    preds = [r[pred_idx] for r in rows]
    
    def norm(v, vals):
        mn, mx = min(vals), max(vals)
        return (v - mn) / (mx - mn) if mx > mn else 0.5

    scores = []
    for r in rows:
        s = (1 - norm(r[mre_idx], mres)) + norm(r[acc_idx], accs) + norm(r[pred_idx], preds)
        scores.append(s)

    # Split into bottom 25% (bad) and top 25% (good)
    combined = sorted(zip(scores, rows), key=lambda x: x[0])
    n = len(combined)
    bad = [x[1] for x in combined[:n // 4]]
    good = [x[1] for x in combined[3 * n // 4:]]

    # Find inputs where the mean/mode in 'bad' is most similar to 'good'
    similarities = []
    for i in input_indices:
        name = header[i]
        if isinstance(bad[0][i], float):
            bad_val = sum(r[i] for r in bad) / len(bad)
            good_val = sum(r[i] for r in good) / len(good)
            # Normalize difference by range
            col_vals = [r[i] for r in rows]
            diff = abs(bad_val - good_val) / (max(col_vals) - min(col_vals) + 1e-9)
        else:
            bad_vals = [r[i] for r in bad]
            good_vals = [r[i] for r in good]
            # Compare frequency of the most common value
            mode_bad = max(set(bad_vals), key=bad_vals.count)
            diff = abs(bad_vals.count(mode_bad) / len(bad_vals) - 
                       good_vals.count(mode_bad) / len(good_vals))
        
        # Lower diff means the setting doesn't distinguish bad from good
        similarities.append((diff, name))

    # Return inputs with the smallest differences
    similarities.sort(key=lambda x: x[0])
    return [s[1] for s in similarities]

import csv

def justify(header, rows, row):
    """
    Estimates the performance of 'row' by comparing it to the distribution 
    of existing rows. Returns (p, reason) where p is the fraction of rows 
    that 'row' outperforms.
    """
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')
    input_indices = [i for i, name in enumerate(header) if name not in ['MRE-', 'ACC+', 'PRED40+']]

    # Estimate goals for the target row using 3-Nearest Neighbors
    def distance(r1, r2):
        d = 0
        for i in input_indices:
            if isinstance(r1[i], str):
                d += 0 if r1[i] == r2[i] else 1
            else:
                # Normalize numeric distance by range
                col_vals = [r[i] for r in rows]
                rng = max(col_vals) - min(col_vals) + 1e-9
                d += abs(r1[i] - r2[i]) / rng
        return d

    neighbors = sorted(rows, key=lambda r: distance(r, row))[:3]
    est_mre = sum(n[mre_idx] for n in neighbors) / len(neighbors)
    est_acc = sum(n[acc_idx] for n in neighbors) / len(neighbors)
    est_pred = sum(n[pred_idx] for n in neighbors) / len(neighbors)

    # Better = lower MRE, higher ACC, higher PRED40
    # We define 'better' as having a higher aggregate normalized score
    def get_score(mre, acc, pred):
        mres = [r[mre_idx] for r in rows]
        accs = [r[acc_idx] for r in rows]
        preds = [r[pred_idx] for r in rows]
        
        def norm(v, vals):
            mn, mx = min(vals), max(vals)
            return (v - mn) / (mx - mn + 1e-9)
        
        return (1 - norm(mre, mres)) + norm(acc, accs) + norm(pred, preds)

    target_score = get_score(est_mre, est_acc, est_pred)
    
    better_count = 0
    for r in rows:
        r_score = get_score(r[mre_idx], r[acc_idx], r[pred_idx])
        if target_score > r_score:
            better_count += 1
    
    p = better_count / len(rows)
    reason = ("Choosing the biggest, deepest setup often leads to overfitting or diminishing returns, "
              "as evidenced by this configuration outperforming only {:.1%} of known setups.").format(p)
    
    return p, reason

    def get_goals(i):
        if i not in labeled:
            if len(labeled) >= budget:
                raise Exception("Budget exceeded")
            labeled[i] = label(i)
        return labeled[i]

    # Sample 50 random rows to evaluate
    n = len(rows)
    sample_indices = random.sample(range(n), min(n, budget))

    best_score = -float('inf')

        return (val - mn) / (mx - mn + 1e-9)

        goals = get_goals(i)

        # Performance score (higher is better)

        s_acc = norm(goals[1], accs)

        total_score = perf - size_penalty
        
        if total_score > best_score:
            best_score = total_score
            best_idx = i

import csv

def fix(header, row, levers):
    """
    Recommends a single change to a 'bad' row to improve its performance.
    It evaluates potential changes by picking the one that maximizes the 
    improvement in the aggregate goal score (MRE-, ACC+, PRED40+).
    """
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    def get_score(r):
        # Goals: MRE- (minimize), ACC+ (maximize), PRED40+ (maximize)
        # We use a simple heuristic: -MRE + ACC + PRED40
        return -r[mre_idx] + r[acc_idx] + r[pred_idx]

    best_change = {}
    best_improvement = -float('inf')

    # We simulate the impact of changing each lever.
    # Since we don't have a model, we assume the 'best' direction is 
    # towards the median of the dataset (a common heuristic for optimization).
    for col, allowed in levers.items():
        idx = header.index(col)
        current_val = row[idx]
        
        candidates = []
        if isinstance(allowed, list):  # Symbolic
            candidates = [v for v in allowed if v != current_val]
        else:  # Numeric (low, high)
            low, high = allowed

def spot(header, rows):

    of the training data. Returns the indices of these rows, sorted by 
    suspicion score (descending).
    """

                     if not (name.endswith('-') or name.endswith('+'))]
    
    # Calculate bounds for numeric inputs
    bounds = {}
    for i in input_indices:
        col_data = [r[i] for r in rows if isinstance(r[i], (int, float))]
        if col_data:
            mn, mx = min(col_data), max(col_data)
            # Define a buffer (e.g., 20% of range) to identify "far outside"
            rng = mx - mn
            if rng > 0:
                bounds[i] = (mn - 0.2 * rng, mx + 0.2 * rng)
            else:
                bounds[i] = (mn, mx)

            if isinstance(val, (int, float)) and i in bounds:

                if val < mn or val > mx:

            elif isinstance(val, str):
                # Check for unknown symbolic values if necessary

                pass
        
        if score > 0:
            suspicious.append((score, idx))

    # Sort by suspicion score descending
    suspicious.sort(key=lambda x: x[0], reverse=True)
