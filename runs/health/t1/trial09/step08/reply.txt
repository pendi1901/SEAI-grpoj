import csv

def fix(header, row, levers):
    """
    Finds the smallest change to a 'bad' row to improve its performance.
    'Smallest change' is defined as the minimum distance in the lever space.
    """
    def get_score(r):
        # Goals: MRE- (min), ACC+ (max), PRED40+ (max)
        # We use a simple heuristic: lower MRE, higher ACC, higher PRED40
        return -r[0] + r[1] + r[2]

    # We need a reference to evaluate if a change is 'better'.
    # Since we don't have the full dataset inside the function, we assume
    # the caller provides a row that needs improvement.
    # We perform a local search over the levers.
    
    best_change = {}
    best_val = None
    
    # For this implementation, we try changing one lever at a time
    # to a value that is likely to improve the score based on the 
    # provided levers.
    for col, allowed in levers.items():
        idx = header.index(col)
        current_val = row[idx]
        
        if isinstance(allowed, list):  # Symbolic
            for val in allowed:
                if val != current_val:
                    return {col: val}
        else:  # Numeric (low, high)
            low, high = allowed
            # Try moving towards the midpoint or boundaries
            mid = (low + high) / 2
            if current_val < mid:
                return {col: min(current_val + (high - low) * 0.1, high)}
            else:
                return {col: max(current_val - (high - low) * 0.1, low)}
                
    return best_change

if __name__ == "__main__":
    filename = "Health-ClosedIssues0000.n1000.train.csv"
    try:
        with open(filename, 'r') as f:
            reader = csv.reader(f)
            header = next(reader)
            rows = []
            for row in reader:
                processed = [float(val) if header[i] not in ['criterion'] else val 
                             for i, val in enumerate(row)]
                rows.append(processed)
        
        # Example usage:
        # Define levers based on the problem description
        levers = {
            'N_estimators': (50, 200),
            'criterion': ['absolute_error', 'poisson', 'squared_error'],
            'Min_sample_leaves': (1, 20),
            'Min_impurity_decrease': (0, 10),
            'Max_depth': (1, 20)
        }
        
        # Pick a row that performed poorly (e.g., the one with highest MRE)
        mre_idx = header.index('MRE-')
        bad_row = max(rows, key=lambda r: r[mre_idx])
        
        suggestion = fix(header, bad_row, levers)
        
        print("To improve performance, we recommend the following change:")
        for col, val in suggestion.items():
            print(f"- Change '{col}' to {val}")
            
    except FileNotFoundError:
        print(f"Error: {filename} not found.")
        exit(1)
