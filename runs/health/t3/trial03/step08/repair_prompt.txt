import csv

def remember(header, rows):
    """
    Identifies the best rows based on multi-objective optimization.
    Goals: Minimize columns ending in '-', Maximize columns ending in '+'.
    Uses a simple Pareto-based ranking approach to find the most trustworthy setups.
    """
    goals = []
    for i, name in enumerate(header):
        if name.endswith('-'):
            goals.append((i, -1))  # Minimize
        elif name.endswith('+'):
            goals.append((i, 1))   # Maximize

    def get_score(row):
        # Normalize values to compare across different scales
        # Since we want to find the "best", we calculate a weighted sum 
        # of the goal columns.
        score = 0
        for idx, direction in goals:
            val = row[idx]
            score += val * direction
        return score

    # Calculate scores for all rows
    scored_rows = []
    for i, row in enumerate(rows):
        scored_rows.append((get_score(row), i))

    # Sort by score descending (higher is better for our combined metric)
    scored_rows.sort(key=lambda x: x[0], reverse=True)

    # Return top 5% (max 40 rows for 800 data rows)
    limit = max(1, len(rows) // 20)
    return [item[1] for item in scored_rows[:limit]]

import csv
from scipy import stats

def certify(xs, ys):
    """
    Performs a Welch's t-test to determine if the difference between two 
    groups (xs and ys) is statistically significant or just noise.
    Returns True if the difference is not significant (p > 0.05),
    False if they really differ (p <= 0.05).
    """
    if len(xs) < 2 or len(ys) < 2:
        return True
    
    t_stat, p_value = stats.ttest_ind(xs, ys, equal_var=False)
    
    # If p_value > 0.05, we fail to reject the null hypothesis (they are the same)
    return p_value > 0.05

import csv

def flow(header, row, seen):
    """
    Decides if the current row is 'good enough' to stop searching.
    We use a simple threshold strategy: if the current row is better than 
    all previously seen rows (Pareto-wise), we accept it if it exceeds 
    a calculated quality threshold based on the distribution of seen rows.
    """
    goals = []
    for i, name in enumerate(header):
        if name.endswith('-'):
            goals.append((i, -1))
        elif name.endswith('+'):
            goals.append((i, 1))

    def get_score(r):
        # Normalize score: higher is better
        score = 0
        for idx, direction in goals:
            score += r[idx] * direction
        return score

    current_score = get_score(row)
    
    if not seen:
        return False
    
    # Calculate scores of all seen rows
    seen_scores = [get_score(s) for s in seen]
    max_seen = max(seen_scores)
    
    # If this row is better than anything seen so far, 
    # we consider it a candidate. We stop if it's significantly 
    # better than the average of what we've seen.
    if current_score > max_seen:
        avg_seen = sum(seen_scores) / len(seen_scores)
        if current_score > avg_seen * 1.1:  # 10% improvement threshold
            return True
            
    return False

def blame(header, rows):
    """
    Identifies which input columns are most predictive of poor performance.
    We define 'bad' as the bottom 25% of rows based on a combined goal score.
    We then train a Random Forest to predict the 'badness' and report feature importances.
    """
    # Identify goal indices and input indices
    goal_indices = [i for i, h in enumerate(header) if h.endswith('-') or h.endswith('+')]
    input_indices = [i for i, h in enumerate(header) if not (h.endswith('-') or h.endswith('+'))]
    
    # Calculate a normalized score for each row to define 'badness'
    # Minimize '-' (multiply by -1), Maximize '+' (multiply by 1)
    def get_score(row):
        score = 0
        for i in goal_indices:
            direction = -1 if header[i].endswith('-') else 1
            score += row[i] * direction
        return score

    scores = [get_score(row) for row in rows]
    # Define bad rows as those with the lowest 25% of scores
    threshold = sorted(scores)[len(scores) // 4]
    is_bad = [1 if s <= threshold else 0 for s in scores]

    # Prepare input data (X)
    X = []
    encoders = {}
    for i in input_indices:
        if isinstance(rows[0][i], str):
            le = LabelEncoder()
            col_vals = [row[i] for row in rows]
            le.fit(col_vals)
            encoders[i] = le

    for row in rows:
        features = []
        for i in input_indices:
            if i in encoders:
                features.append(encoders[i].transform([row[i]])[0])
            else:
                features.append(row[i])
        X.append(features)

    # Train a regressor to find which inputs correlate with 'bad' outcomes
    model = RandomForestRegressor(n_estimators=50, random_state=42)
    model.fit(X, is_bad)

    # Get feature importances
    importances = model.feature_importances_
    indexed_importances = sorted(zip(importances, input_indices), reverse=True)
    
    # Return the names of the top 3 most influential input columns
    top_features = [header[idx] for _, idx in indexed_importances[:3]]
    return top_features

def justify(header, rows, row):
    """
    Estimates the performance of a given row and compares it to the dataset.
    Returns (p, reason) where p is the fraction of rows worse than the input,
    and reason is a human-readable explanation.
    """
    goal_indices = [i for i, h in enumerate(header) if h.endswith('-') or h.endswith('+')]
    input_indices = [i for i, h in enumerate(header) if not (h.endswith('-') or h.endswith('+'))]
    
    # Encode symbolic data
    encoders = {}
    for i in input_indices:
        if isinstance(rows[0][i], str):
            le = LabelEncoder()
            le.fit([r[i] for r in rows])
            encoders[i] = le

    def transform(r):
        return [encoders[i].transform([r[i]])[0] if i in encoders else r[i] for i in input_indices]

    # Train models to predict each goal
    X = [transform(r) for r in rows]
    predictions = []
    for g_idx in goal_indices:
        y = [r[g_idx] for r in rows]
        model = RandomForestRegressor(n_estimators=50, random_state=42)
        model.fit(X, y)
        predictions.append(model.predict([transform(row)])[0])

    # Calculate score: minimize '-' (negate), maximize '+' (keep)
    def get_score(goals):
        score = 0
        for i, val in enumerate(goals):
            direction = -1 if header[goal_indices[i]].endswith('-') else 1
            score += val * direction
        return score

    target_score = get_score(predictions)
    all_scores = [get_score([r[i] for i in goal_indices]) for r in rows]
    
    # Fraction of rows worse than target
    worse_count = sum(1 for s in all_scores if target_score > s)
    p = worse_count / len(rows)
    
    reason = f"This configuration is estimated to outperform {p:.1%} of all known setups based on its predicted goal metrics."
    return p, reason
