import csv
import random

def fix(header, row, levers):
    """
    Finds the smallest change to a 'bad' row that improves its score
    without violating constraints.
    """
    goal_indices = [i for i, h in enumerate(header) if h.endswith('-') or h.endswith('+')]
    
    def get_score(r):
        score = 0
        for i in goal_indices:
            val = r[i]
            direction = -1 if header[i].endswith('-') else 1
            score += val * direction
        return score

    # Current score of the bad row
    current_score = get_score(row)
    best_change = {}
    best_score = current_score

    # Try changing one lever at a time
    for col_name, allowed in levers.items():
        col_idx = header.index(col_name)
        
        # Generate candidates
        candidates = []
        if isinstance(allowed, list):  # Symbolic
            candidates = [v for v in allowed if v != row[col_idx]]
        else:  # Numeric (low, high)
            low, high = allowed
            # Try a few steps within range
            step = (high - low) / 5
            for i in range(1, 6):
                val = low + i * step
                if abs(val - row[col_idx]) > 1e-6:
                    candidates.append(val)

        for val in candidates:
            new_row = list(row)
            new_row[col_idx] = val
            new_score = get_score(new_row)
            
            # We want to improve the score
            if new_score > best_score:
                best_score = new_score
                best_change = {col_name: val}
                
    return best_change

if __name__ == "__main__":
    filename = 'Health-ClosedIssues0000.n1000.train.csv'
    rows = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = []
            for i, val in enumerate(row):
                try:
                    processed_row.append(float(val))
                except ValueError:
                    processed_row.append(val)
            rows.append(processed_row)

    # Define levers based on column types
    levers = {
        'N_estimators': (50, 200),
        'criterion': ['absolute_error', 'poisson', 'squared_error'],
        'Min_sample_leaves': (1, 20),
        'Min_impurity_decrease': (0, 10),
        'Max_depth': (1, 20)
    }

    # Pick a 'bad' row (e.g., the one with the lowest score)
    goal_indices = [i for i, h in enumerate(header) if h.endswith('-') or h.endswith('+')]
    def get_score(r):
        score = 0
        for i in goal_indices:
            direction = -1 if header[i].endswith('-') else 1
            score += r[i] * direction
        return score

    scores = [(get_score(r), i) for i, r in enumerate(rows)]
    bad_idx = min(scores)[1]
    bad_row = rows[bad_idx]

    recommendation = fix(header, bad_row, levers)

    print("Analysis for improving configuration:")
    if recommendation:
        col, val = list(recommendation.items())[0]
        print(f"To improve performance, change '{col}' to {val}.")
    else:
        print("No simple single-lever change found to improve performance.")
