import csv
from sklearn.ensemble import RandomForestRegressor
from sklearn.preprocessing import LabelEncoder

def justify(header, rows, row):
    """
    Estimates the performance of a given row and compares it to the dataset.
    Returns (p, reason) where p is the fraction of rows worse than the input,
    and reason is a human-readable explanation.
    """
    goal_indices = [i for i, h in enumerate(header) if h.endswith('-') or h.endswith('+')]
    input_indices = [i for i, h in enumerate(header) if not (h.endswith('-') or h.endswith('+'))]
    
    # Encode symbolic data
    encoders = {}
    for i in input_indices:
        if isinstance(rows[0][i], str):
            le = LabelEncoder()
            le.fit([str(r[i]) for r in rows])
            encoders[i] = le

    def transform(r):
        return [encoders[i].transform([str(r[i])])[0] if i in encoders else r[i] for i in input_indices]

    # Train models to predict each goal
    X = [transform(r) for r in rows]
    predictions = []
    for g_idx in goal_indices:
        y = [r[g_idx] for r in rows]
        model = RandomForestRegressor(n_estimators=50, random_state=42)
        model.fit(X, y)
        predictions.append(model.predict([transform(row)])[0])

    # Calculate score: minimize '-' (negate), maximize '+' (keep)
    def get_score(goals):
        score = 0
        for i, val in enumerate(goals):
            direction = -1 if header[goal_indices[i]].endswith('-') else 1
            score += val * direction
        return score

    target_score = get_score(predictions)
    all_scores = [get_score([r[i] for i in goal_indices]) for r in rows]
    
    # Fraction of rows worse than target
    worse_count = sum(1 for s in all_scores if target_score > s)
    p = worse_count / len(rows)
    
    reason = "Simply using the largest settings often leads to diminishing returns or overfitting, as evidenced by this configuration outperforming only {:.1%} of the observed setups.".format(p)
    return p, reason

if __name__ == "__main__":
    filename = 'Health-ClosedIssues0000.n1000.train.csv'
    rows = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = [float(val) if val.replace('.', '', 1).isdigit() or (val.startswith('-') and val[1:].replace('.', '', 1).isdigit()) else val for val in row]
            rows.append(processed_row)

    # Create a "biggest, deepest" setup
    # N_estimators (idx 3), Max_depth (idx 7)
    # Based on data range: N_estimators up to 200, Max_depth up to 20
    test_row = [0.0] * len(header)
    for i in range(len(header)):
        if header[i] == 'N_estimators': test_row[i] = 200.0
        elif header[i] == 'Max_depth': test_row[i] = 20.0
        elif header[i] == 'Min_sample_leaves': test_row[i] = 1.0
        elif header[i] == 'Min_impurity_decrease': test_row[i] = 0.0
        elif header[i] == 'criterion': test_row[i] = 'squared_error'
        elif header[i].endswith('-') or header[i].endswith('+'): test_row[i] = "?"
        else: test_row[i] = 0.0

    p, reason = justify(header, rows, test_row)
    print(reason)
