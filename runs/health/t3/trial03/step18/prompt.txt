import csv

def remember(header, rows):
    """
    Identifies the best rows based on multi-objective optimization.
    Goals: Minimize columns ending in '-', Maximize columns ending in '+'.
    Uses a simple Pareto-based ranking approach to find the most trustworthy setups.
    """
    goals = []
    for i, name in enumerate(header):
        if name.endswith('-'):
            goals.append((i, -1))  # Minimize
        elif name.endswith('+'):
            goals.append((i, 1))   # Maximize

    def get_score(row):
        # Normalize values to compare across different scales
        # Since we want to find the "best", we calculate a weighted sum 
        # of the goal columns.
        score = 0
        for idx, direction in goals:
            val = row[idx]
            score += val * direction
        return score

    # Calculate scores for all rows
    scored_rows = []
    for i, row in enumerate(rows):
        scored_rows.append((get_score(row), i))

    # Sort by score descending (higher is better for our combined metric)
    scored_rows.sort(key=lambda x: x[0], reverse=True)

    # Return top 5% (max 40 rows for 800 data rows)
    limit = max(1, len(rows) // 20)
    return [item[1] for item in scored_rows[:limit]]

def guess(header, train, test):

    """

    goal_idx = 0
    
    # Prepare data
    def process_data(data):
        X = []
        y = []
        for row in data:
            # Inputs are all columns except the goals
            # Goals are columns ending in '-' or '+'
            inputs = []
            for i, val in enumerate(row):
                if not (header[i].endswith('-') or header[i].endswith('+')):
                    inputs.append(val)
            X.append(inputs)
            if len(row) > goal_idx and isinstance(row[goal_idx], (int, float)):
                y.append(row[goal_idx])
        return X, y

    # Encode symbolic columns
    # Identify which columns are symbolic (not numeric)
    symbolic_cols = [i for i, h in enumerate(header) if not (h.endswith('-') or h.endswith('+')) and isinstance(train[0][i], str)]
    
    # Combine train and test for consistent encoding
    all_data = train + test
    encoders = {}
    for col_idx in symbolic_cols:
        le = LabelEncoder()
        col_values = [row[col_idx] for row in all_data]
        le.fit(col_values)
        encoders[col_idx] = le
        
    def transform(data):
        transformed = []
        for row in data:
            new_row = []
            for i, val in enumerate(row):
                if not (header[i].endswith('-') or header[i].endswith('+')):
                    if i in encoders:
                        new_row.append(encoders[i].transform([val])[0])
                    else:
                        new_row.append(val)
            transformed.append(new_row)
        return transformed

    X_train, y_train = transform(train), [row[goal_idx] for row in train]
    X_test = transform(test)
    
    # Train model
    model = RandomForestRegressor(n_estimators=100, random_state=42)
    model.fit(X_train, y_train)

import csv
from scipy import stats

def certify(xs, ys):
    """
    Performs a Welch's t-test to determine if the difference between two 
    groups (xs and ys) is statistically significant or just noise.
    Returns True if the difference is not significant (p > 0.05),
    False if they really differ (p <= 0.05).
    """
    if len(xs) < 2 or len(ys) < 2:
        return True
    
    t_stat, p_value = stats.ttest_ind(xs, ys, equal_var=False)
    
    # If p_value > 0.05, we fail to reject the null hypothesis (they are the same)
    return p_value > 0.05

import csv

def flow(header, row, seen):
    """
    Decides if the current row is 'good enough' to stop searching.
    We use a simple threshold strategy: if the current row is better than 
    all previously seen rows (Pareto-wise), we accept it if it exceeds 
    a calculated quality threshold based on the distribution of seen rows.
    """
    goals = []
    for i, name in enumerate(header):
        if name.endswith('-'):
            goals.append((i, -1))
        elif name.endswith('+'):
            goals.append((i, 1))

    def get_score(r):
        # Normalize score: higher is better
        score = 0
        for idx, direction in goals:
            score += r[idx] * direction
        return score

    current_score = get_score(row)
    
    if not seen:
        return False
    
    # Calculate scores of all seen rows
    seen_scores = [get_score(s) for s in seen]
    max_seen = max(seen_scores)
    
    # If this row is better than anything seen so far, 
    # we consider it a candidate. We stop if it's significantly 
    # better than the average of what we've seen.
    if current_score > max_seen:
        avg_seen = sum(seen_scores) / len(seen_scores)
        if current_score > avg_seen * 1.1:  # 10% improvement threshold
            return True
            
    return False

def blame(header, rows):
    """
    Identifies which input columns are most predictive of poor performance.
    We define 'bad' as the bottom 25% of rows based on a combined goal score.
    We then train a Random Forest to predict the 'badness' and report feature importances.
    """
    # Identify goal indices and input indices
    goal_indices = [i for i, h in enumerate(header) if h.endswith('-') or h.endswith('+')]
    input_indices = [i for i, h in enumerate(header) if not (h.endswith('-') or h.endswith('+'))]
    
    # Calculate a normalized score for each row to define 'badness'
    # Minimize '-' (multiply by -1), Maximize '+' (multiply by 1)
    def get_score(row):
        score = 0
        for i in goal_indices:
            direction = -1 if header[i].endswith('-') else 1
            score += row[i] * direction
        return score

    scores = [get_score(row) for row in rows]
    # Define bad rows as those with the lowest 25% of scores
    threshold = sorted(scores)[len(scores) // 4]
    is_bad = [1 if s <= threshold else 0 for s in scores]

    # Prepare input data (X)
    X = []
    encoders = {}
    for i in input_indices:
        if isinstance(rows[0][i], str):
            le = LabelEncoder()
            col_vals = [row[i] for row in rows]

            encoders[i] = le

    for row in rows:
        features = []
        for i in input_indices:
            if i in encoders:
                features.append(encoders[i].transform([row[i]])[0])
            else:
                features.append(row[i])
        X.append(features)

    # Train a regressor to find which inputs correlate with 'bad' outcomes
    model = RandomForestRegressor(n_estimators=50, random_state=42)
    model.fit(X, is_bad)

    # Get feature importances
    importances = model.feature_importances_
    indexed_importances = sorted(zip(importances, input_indices), reverse=True)
    
    # Return the names of the top 3 most influential input columns
    top_features = [header[idx] for _, idx in indexed_importances[:3]]

def justify(header, rows, row):
    """
    Estimates the performance of a given row and compares it to the dataset.
    Returns (p, reason) where p is the fraction of rows worse than the input,
    and reason is a human-readable explanation.
    """
    goal_indices = [i for i, h in enumerate(header) if h.endswith('-') or h.endswith('+')]
    input_indices = [i for i, h in enumerate(header) if not (h.endswith('-') or h.endswith('+'))]
    
    # Encode symbolic data
    encoders = {}
    for i in input_indices:
        if isinstance(rows[0][i], str):
            le = LabelEncoder()
            le.fit([r[i] for r in rows])
            encoders[i] = le

    def transform(r):
        return [encoders[i].transform([r[i]])[0] if i in encoders else r[i] for i in input_indices]

    # Train models to predict each goal
    X = [transform(r) for r in rows]
    predictions = []
    for g_idx in goal_indices:
        y = [r[g_idx] for r in rows]
        model = RandomForestRegressor(n_estimators=50, random_state=42)
        model.fit(X, y)
        predictions.append(model.predict([transform(row)])[0])

    # Calculate score: minimize '-' (negate), maximize '+' (keep)
    def get_score(goals):
        score = 0
        for i, val in enumerate(goals):
            direction = -1 if header[goal_indices[i]].endswith('-') else 1
            score += val * direction
        return score

    target_score = get_score(predictions)
    all_scores = [get_score([r[i] for i in goal_indices]) for r in rows]
    
    # Fraction of rows worse than target
    worse_count = sum(1 for s in all_scores if target_score > s)
    p = worse_count / len(rows)
    
    reason = f"This configuration is estimated to outperform {p:.1%} of all known setups based on its predicted goal metrics."
    return p, reason

            score += val * direction
        return score

        goals = label(i)
        score = get_score(goals)
        if score > best_score:
            best_score = score

import csv

def spot(header, rows):
    """
    Identifies rows that look like failed runs.
    A failed run often results in extreme outliers in the goal columns
    or inconsistent input values. We use Z-score analysis on the goal
    columns to find rows that are statistically anomalous.
    """
    goal_indices = [i for i, h in enumerate(header) if h.endswith('-') or h.endswith('+')]
    
    # Calculate anomaly score for each row based on goal metrics
    # We look for rows that are extreme outliers (Z-score > 3)
    anomaly_scores = []
    
    for i, row in enumerate(rows):
        score = 0
        for g_idx in goal_indices:
            val = row[g_idx]
            # Get column stats
            col_vals = [r[g_idx] for r in rows if isinstance(r[g_idx], (int, float))]
            if not col_vals:
                continue
            mean = sum(col_vals) / len(col_vals)
            std = (sum((x - mean)**2 for x in col_vals) / len(col_vals))**0.5
            if std > 0:
                z = abs(val - mean) / std
                score += z
        anomaly_scores.append((score, i))
    
    # Sort by anomaly score descending
    anomaly_scores.sort(key=lambda x: x[0], reverse=True)
    
    # Return indices of rows with high anomaly scores (top 5%)
    limit = max(1, len(rows) // 20)
    return [item[1] for item in anomaly_scores[:limit]]

import csv
import random

def race(header, rows, label):
    """
    Compares three strategies:
    1. 'random': Picks 16 random rows.
    2. 'greedy': Picks 17 rows by sampling, then greedily exploring neighbors.
    3. 'smart': Picks 17 rows by sampling, then focusing on the best-performing region.
    Total labels: 16 + 17 + 17 = 50.
    """
    goal_indices = [i for i, h in enumerate(header) if h.endswith('-') or h.endswith('+')]
    
    def get_score(goals):
        score = 0
        for i, val in enumerate(goals):
            direction = -1 if header[goal_indices[i]].endswith('-') else 1
            score += val * direction
        return score

    results = {}
    
    # Strategy 1: Random
    random_indices = random.sample(range(len(rows)), 16)
    best_r = -1
    best_s = float('-inf')
    for i in random_indices:
        s = get_score(label(i))
        if s > best_s:
            best_s, best_r = s, i

import csv

def remember(header, rows):
    """
    Identifies the best rows by calculating a normalized score for each row.
    Goals ending in '-' are minimized, goals ending in '+' are maximized.
    Returns the indices of the top 5% of rows.
    """
    goal_indices = [i for i, h in enumerate(header) if h.endswith('-') or h.endswith('+')]
    
    # Calculate scores for all rows
    # We normalize each goal column to [0, 1] to ensure equal weighting
    scores = []
    for i, row in enumerate(rows):
        row_score = 0
        for g_idx in goal_indices:
            col_vals = [r[g_idx] for r in rows if isinstance(r[g_idx], (int, float))]
            min_val = min(col_vals)
            max_val = max(col_vals)
            val = row[g_idx]
            
            # Normalize
            if max_val - min_val == 0:
                norm = 0
            else:
                norm = (val - min_val) / (max_val - min_val)
            
            # If goal ends in '-', we want low values (1 - norm)
            # If goal ends in '+', we want high values (norm)
            if header[g_idx].endswith('-'):
                row_score += (1 - norm)
            else:
                row_score += norm
        scores.append((row_score, i))
    
    # Sort by score descending
    scores.sort(key=lambda x: x[0], reverse=True)
    
    # Return top 5%
    limit = max(1, len(rows) // 20)
    return [item[1] for item in scores[:limit]]

def certify(xs, ys):
    """
    Performs a Welch's t-test to determine if the difference between two 
    groups (xs and ys) is statistically significant or just noise.
    Returns True if the difference is not significant (p > 0.05),
    False if they really differ (p <= 0.05).
    """
    if len(xs) < 2 or len(ys) < 2:
        return True
    
    t_stat, p_value = stats.ttest_ind(xs, ys, equal_var=False)
    
    # If p_value > 0.05, we fail to reject the null hypothesis (they are the same)
    return p_value > 0.05

import csv

def flow(header, row, seen):
    """
    Decides if the current row is 'good enough' to stop searching.
    We stop if the current row is a Pareto-optimal improvement over all
    previously seen rows and represents a significant gain (e.g., top 1% 
    of the potential search space).
    """
    goal_indices = [i for i, h in enumerate(header) if h.endswith('-') or h.endswith('+')]
    
    def get_score(r):
        # Normalize goal values to a [0, 1] scale for comparison
        # Since we don't have global min/max, we use the seen data
        score = 0
        for i in goal_indices:
            val = r[i]
            # Simple direction-based scoring
            if header[i].endswith('-'):
                score -= val
            else:
                score += val
        return score

    if not seen:
        return False

    current_score = get_score(row)
    seen_scores = [get_score(s) for s in seen]
    max_seen = max(seen_scores)

    # If this row is better than anything seen so far, check if it's 
    # a significant improvement (e.g., better than the best seen + a small margin)
    # This acts as an early-stopping criterion for optimization.
    if current_score > max_seen:
        # If we have enough data, check if it's in the top tier
        if len(seen) > 50:
            # Stop if we find a configuration that is significantly better 
            # than the average of the best 10% of what we've seen.
            top_10_pct = sorted(seen_scores, reverse=True)[:len(seen)//10]
            if current_score > (sum(top_10_pct) / len(top_10_pct)) * 1.05:
                return True
            
    return False

    """
    Finds a setup that is both high-performing and 'small'.

    We sample 50 rows, score them based on a weighted combination of 
    performance (goals) and efficiency (smallness), and return the best.
    """
    goal_indices = [i for i, h in enumerate(header) if h.endswith('-') or h.endswith('+')]
    n_est_idx = header.index('N_estimators')
    max_depth_idx = header.index('Max_depth')

    # Normalize goals to [0, 1] for scoring
    def get_performance_score(goals):
        score = 0
        for i, val in enumerate(goals):
            # For this specific dataset, we know the range of goals roughly
            # MRE- (0 to 1), ACC+ (0 to 1), PRED40+ (0 to 1)
            if header[goal_indices[i]].endswith('-'):

                score += val

        n_est = row[n_est_idx]

        return (norm_n + norm_d) / 2.0

    # Sample 50 rows
