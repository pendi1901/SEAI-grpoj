import csv

def remember(header, rows):
    """
    Identifies rows that are 'unusually trustworthy' by finding the Pareto frontier
    or high-performing candidates based on the goals.
    Goals: MRE- (minimize), ACC+ (maximize), PRED40+ (maximize).
    """
    # Identify indices of goal columns
    goals = []
    for i, name in enumerate(header):
        if name.endswith('-'):
            goals.append((i, -1))  # -1 for minimization
        elif name.endswith('+'):
            goals.append((i, 1))   # 1 for maximization

    # Calculate a normalized score for each row
    # We normalize each goal column to [0, 1] range
    scores = []
    for i in range(len(goals)):
        col_idx, direction = goals[i]
        values = [row[col_idx] for row in rows]
        min_v, max_v = min(values), max(values)
        rng = max_v - min_v if max_v != min_v else 1.0
        
        for r_idx, row in enumerate(rows):
            val = (row[col_idx] - min_v) / rng
            if direction == -1:
                val = 1.0 - val
            if len(scores) <= r_idx:
                scores.append(val)
            else:
                scores[r_idx] += val

    # Get indices of top 5% (or at least 1)
    num_to_return = max(1, len(rows) // 20)
    indexed_scores = list(enumerate(scores))
    indexed_scores.sort(key=lambda x: x[1], reverse=True)
    
    return [x[0] for x in indexed_scores[:num_to_return]]

import csv
from scipy.stats import pearsonr

def certify(xs, ys):
    """
    Returns True if the correlation between xs and ys is statistically significant
    (p-value < 0.05), suggesting the relationship is real rather than noise.
    """
    if len(xs) < 2 or len(ys) < 2:
        return False
    
    corr, p_value = pearsonr(xs, ys)
    
    # If p_value is small, we reject the null hypothesis (that there is no correlation)
    # Thus, the relationship is likely real.
    return p_value < 0.05

import csv

def flow(header, row, seen):
    """
    Decides if the current row is 'good enough' to stop searching.
    We define 'good enough' as being in the top 5% of the best possible 
    performance observed so far across all goal columns.
    """
    goals = []
    for i, name in enumerate(header):
        if name.endswith('-'):
            goals.append((i, -1))
        elif name.endswith('+'):
            goals.append((i, 1))

    def get_score(r):
        score = 0
        for i, (col_idx, direction) in enumerate(goals):
            val = r[col_idx]
            # Normalize based on the range of the current 'seen' set
            # If it's the first row, we treat it as a baseline
            if not seen:
                return 1.0
            
            all_vals = [s[col_idx] for s in seen]
            min_v, max_v = min(all_vals), max(all_vals)
            rng = max_v - min_v if max_v != min_v else 1.0
            
            norm = (val - min_v) / rng
            if direction == -1:
                norm = 1.0 - norm
            score += norm
        return score / len(goals)

    # If we haven't seen enough, keep looking
    if len(seen) < 20:
        return False

    current_score = get_score(row)
    seen_scores = [get_score(s) for s in seen]
    
    # Stop if this row is better than 95% of what we have seen
    threshold = sorted(seen_scores)[int(len(seen_scores) * 0.95)]
    return current_score >= threshold

import csv

def blame(header, rows):
    """
    Identifies input columns that correlate most strongly with poor performance.
    'Bad' rows are defined as those in the bottom 25% of aggregate performance.
    We calculate the correlation between input values and the 'badness' score.
    """
    # Identify goal columns
    goals = []
    for i, name in enumerate(header):
        if name.endswith('-'):
            goals.append((i, -1))
        elif name.endswith('+'):
            goals.append((i, 1))

    # Calculate aggregate performance score for each row
    def get_score(row):
        score = 0
        for i, (col_idx, direction) in enumerate(goals):
            # We use the row's own values relative to the dataset range
            col_vals = [r[col_idx] for r in rows if isinstance(r[col_idx], (int, float))]
            min_v, max_v = min(col_vals), max(col_vals)
            rng = max_v - min_v if max_v != min_v else 1.0
            norm = (row[col_idx] - min_v) / rng
            if direction == -1:
                norm = 1.0 - norm
            score += norm
        return score / len(goals)

    # Identify input columns (not goals)
    input_indices = [i for i, name in enumerate(header) 
                     if not (name.endswith('-') or name.endswith('+'))]
    
    # Calculate scores and find the worst 25%
    scores = [(get_score(r), r) for r in rows]
    scores.sort(key=lambda x: x[0])
    worst_rows = [x[1] for x in scores[:len(rows) // 4]]
    
    # For each input column, check if its values in 'worst' rows deviate 
    # significantly from the global average (simple proxy for 'blame')
    blame_scores = []
    for idx in input_indices:
        col_vals = [r[idx] for r in rows if not isinstance(r[idx], str)]
        if not col_vals: continue
        
        avg = sum(col_vals) / len(col_vals)
        worst_vals = [r[idx] for r in worst_rows if not isinstance(r[idx], str)]
        if not worst_vals: continue
        
        worst_avg = sum(worst_vals) / len(worst_vals)
        # The 'blame' is the absolute difference between worst-row average and global average
        blame_scores.append((header[idx], abs(worst_avg - avg)))

    # Sort by impact and return names
    blame_scores.sort(key=lambda x: x[1], reverse=True)
    return [x[0] for x in blame_scores]

import csv
import random

def choose(header, rows, label):
    """
    Finds the best row by sampling 50 configurations.
    Uses a simple random search strategy to explore the space,
    labels them, and returns the index of the best performing one.
    """
    goal_indices = [i for i, h in enumerate(header) if h.endswith('-') or h.endswith('+')]
    
    def get_score(row_goals):
        score = 0
        for i, g_idx in enumerate(goal_indices):
            # Since we don't have the full dataset distribution for normalization 
            # inside the function, we use the provided row_goals directly.
            # We assume the goal values are already comparable or normalized.
            val = row_goals[i]
            if header[g_idx].endswith('-'):
                score -= val
            else:
                score += val
        return score

    # We have 50 tries. We pick 50 random indices.
    num_rows = len(rows)
    tries = min(50, num_rows)
    indices = random.sample(range(num_rows), tries)
    
    best_idx = -1
    best_score = float('-inf')
    
    for idx in indices:
        goals = label(idx)
        score = get_score(goals)
        if score > best_score:
            best_score = score
            best_idx = idx
            
    return best_idx
