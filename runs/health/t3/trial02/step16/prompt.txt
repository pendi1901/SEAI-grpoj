import csv

def remember(header, rows):
    """
    Identifies rows that are 'unusually trustworthy' by finding the Pareto frontier
    or high-performing candidates based on the goals.
    Goals: MRE- (minimize), ACC+ (maximize), PRED40+ (maximize).
    """
    # Identify indices of goal columns
    goals = []
    for i, name in enumerate(header):
        if name.endswith('-'):
            goals.append((i, -1))  # -1 for minimization
        elif name.endswith('+'):
            goals.append((i, 1))   # 1 for maximization

    # Calculate a normalized score for each row
    # We normalize each goal column to [0, 1] range
    scores = []
    for i in range(len(goals)):
        col_idx, direction = goals[i]
        values = [row[col_idx] for row in rows]
        min_v, max_v = min(values), max(values)
        rng = max_v - min_v if max_v != min_v else 1.0
        
        for r_idx, row in enumerate(rows):
            val = (row[col_idx] - min_v) / rng
            if direction == -1:
                val = 1.0 - val
            if len(scores) <= r_idx:
                scores.append(val)
            else:
                scores[r_idx] += val

    # Get indices of top 5% (or at least 1)
    num_to_return = max(1, len(rows) // 20)
    indexed_scores = list(enumerate(scores))
    indexed_scores.sort(key=lambda x: x[1], reverse=True)
    
    return [x[0] for x in indexed_scores[:num_to_return]]

from scipy.stats import pearsonr

def certify(xs, ys):
    """
    Returns True if the correlation between xs and ys is statistically significant
    (p-value < 0.05), suggesting the relationship is real rather than noise.
    """
    if len(xs) < 2 or len(ys) < 2:
        return False
    
    corr, p_value = pearsonr(xs, ys)
    
    # If p_value is small, we reject the null hypothesis (that there is no correlation)
    # Thus, the relationship is likely real.
    return p_value < 0.05

import csv

def flow(header, row, seen):
    """
    Decides if the current row is 'good enough' to stop searching.
    We define 'good enough' as being in the top 5% of the best possible 
    performance observed so far across all goal columns.
    """
    goals = []
    for i, name in enumerate(header):
        if name.endswith('-'):
            goals.append((i, -1))
        elif name.endswith('+'):
            goals.append((i, 1))

    def get_score(r):
        score = 0
        for i, (col_idx, direction) in enumerate(goals):
            val = r[col_idx]
            # Normalize based on the range of the current 'seen' set
            # If it's the first row, we treat it as a baseline
            if not seen:
                return 1.0
            
            all_vals = [s[col_idx] for s in seen]
            min_v, max_v = min(all_vals), max(all_vals)
            rng = max_v - min_v if max_v != min_v else 1.0
            
            norm = (val - min_v) / rng
            if direction == -1:
                norm = 1.0 - norm
            score += norm
        return score / len(goals)

    # If we haven't seen enough, keep looking
    if len(seen) < 20:
        return False

    current_score = get_score(row)
    seen_scores = [get_score(s) for s in seen]
    
    # Stop if this row is better than 95% of what we have seen
    threshold = sorted(seen_scores)[int(len(seen_scores) * 0.95)]
    return current_score >= threshold

import csv

def blame(header, rows):
    """
    Identifies input columns that correlate most strongly with poor performance.
    'Bad' rows are defined as those in the bottom 25% of aggregate performance.
    We calculate the correlation between input values and the 'badness' score.
    """
    # Identify goal columns
    goals = []
    for i, name in enumerate(header):
        if name.endswith('-'):
            goals.append((i, -1))
        elif name.endswith('+'):
            goals.append((i, 1))

    # Calculate aggregate performance score for each row
    def get_score(row):
        score = 0
        for i, (col_idx, direction) in enumerate(goals):
            # We use the row's own values relative to the dataset range
            col_vals = [r[col_idx] for r in rows if isinstance(r[col_idx], (int, float))]
            min_v, max_v = min(col_vals), max(col_vals)
            rng = max_v - min_v if max_v != min_v else 1.0
            norm = (row[col_idx] - min_v) / rng
            if direction == -1:
                norm = 1.0 - norm
            score += norm
        return score / len(goals)

    # Identify input columns (not goals)
    input_indices = [i for i, name in enumerate(header) 
                     if not (name.endswith('-') or name.endswith('+'))]
    
    # Calculate scores and find the worst 25%
    scores = [(get_score(r), r) for r in rows]
    scores.sort(key=lambda x: x[0])
    worst_rows = [x[1] for x in scores[:len(rows) // 4]]
    
    # For each input column, check if its values in 'worst' rows deviate 
    # significantly from the global average (simple proxy for 'blame')
    blame_scores = []
    for idx in input_indices:
        col_vals = [r[idx] for r in rows if not isinstance(r[idx], str)]
        if not col_vals: continue
        
        avg = sum(col_vals) / len(col_vals)
        worst_vals = [r[idx] for r in worst_rows if not isinstance(r[idx], str)]
        if not worst_vals: continue
        
        worst_avg = sum(worst_vals) / len(worst_vals)
        # The 'blame' is the absolute difference between worst-row average and global average
        blame_scores.append((header[idx], abs(worst_avg - avg)))

    # Sort by impact and return names
    blame_scores.sort(key=lambda x: x[1], reverse=True)
    return [x[0] for x in blame_scores]

def choose(header, rows, label):

    Finds the best row by sampling 50 configurations.

    labels them, and returns the index of the best performing one.

        score = 0
        for i, g_idx in enumerate(goal_indices):
            # Since we don't have the full dataset distribution for normalization 
            # inside the function, we use the provided row_goals directly.
            # We assume the goal values are already comparable or normalized.
            val = row_goals[i]
            if header[g_idx].endswith('-'):
                score -= val
            else:
                score += val
        return score

    # We have 50 tries. We pick 50 random indices.

    tries = min(50, num_rows)
    indices = random.sample(range(num_rows), tries)

    best_score = float('-inf')
    
    for idx in indices:
        goals = label(idx)
        score = get_score(goals)
        if score > best_score:
            best_score = score
            best_idx = idx
            
    return best_idx

import csv

def fix(header, row, levers):
    """
    Finds the smallest change to a 'bad' row's settings that improves performance.
    We use a simple heuristic: evaluate the impact of changing each lever 
    by checking the average performance of rows in the dataset that match 
    the new lever value.
    """
    # Identify goal indices and directions
    goals = []
    for i, name in enumerate(header):
        if name.endswith('-'):
            goals.append((i, -1))
        elif name.endswith('+'):
            goals.append((i, 1))

    def get_row_score(r):
        score = 0
        for i, direction in goals:
            val = r[i]
            score += (val * direction)
        return score

    # Find the best possible value for each lever based on historical data
    best_changes = {}
    
    for col_name, allowed in levers.items():
        if col_name not in header:
            continue
        idx = header.index(col_name)
        
        # Find all rows that have a specific value for this lever
        # and calculate their average performance
        candidates = []
        if isinstance(allowed, list):
            candidates = allowed
        else:
            # For numeric, sample a few points in the range
            low, high = allowed
            candidates = [low + (high - low) * i / 4 for i in range(5)]
            
        best_val = None
        max_avg_score = float('-inf')
        
        for val in candidates:
            # Filter rows that match this value (with tolerance for floats)
            if isinstance(val, float):
                matches = [r for r in rows if abs(r[idx] - val) < (high - low) * 0.1]
            else:
                matches = [r for r in rows if r[idx] == val]
            
            if matches:
                avg_score = sum(get_row_score(r) for r in matches) / len(matches)
                if avg_score > max_avg_score:
                    max_avg_score = avg_score
                    best_val = val
        
        if best_val is not None and best_val != row[idx]:
            best_changes[col_name] = best_val

    # Return the single best change found
    if best_changes:
        # Return the first recommended change
        k = list(best_changes.keys())[0]
        return {k: best_changes[k]}
    
    return {}

import csv

def spot(header, rows):
    """
    Identifies rows that are 'suspicious' by detecting outliers in the input space.
    Rows with input values that are statistically far from the mean (z-score > 3)
    are flagged as potentially erroneous entries.
    """
    input_indices = [i for i, name in enumerate(header) 
                     if not (name.endswith('-') or name.endswith('+'))]
    
    # Only consider numeric input columns for outlier detection
    numeric_inputs = []
    for idx in input_indices:
        vals = [r[idx] for r in rows if isinstance(r[idx], (int, float))]
        if vals:
            numeric_inputs.append((idx, vals))

    # Calculate z-scores for each numeric input
    suspicion_scores = [0.0] * len(rows)
    for idx, vals in numeric_inputs:
        mean = sum(vals) / len(vals)
        variance = sum((x - mean) ** 2 for x in vals) / len(vals)
        std = variance ** 0.5
        if std == 0:
            continue
            
        for r_idx, row in enumerate(rows):
            val = row[idx]
            if isinstance(val, (int, float)):
                z_score = abs(val - mean) / std
                # If z-score is high, it's suspicious
                if z_score > 3:
                    suspicion_scores[r_idx] += z_score

    # Sort rows by suspicion score descending
    indexed_scores = sorted(enumerate(suspicion_scores), key=lambda x: x[1], reverse=True)
    
    # Return indices of rows with non-zero suspicion
    return [i for i, score in indexed_scores if score > 0]

import csv

def race(header, rows, label):
    """
    Compares three strategies:
    1. 'random': Picks rows at random.
    2. 'first_n': Picks the first 16 rows sequentially.
    3. 'last_n': Picks the last 17 rows sequentially.
    Total budget: 50 labels.
    """
    goal_indices = [i for i, h in enumerate(header) if h.endswith('-') or h.endswith('+')]
    
    def get_score(row_goals):
        score = 0
        for i, g_idx in enumerate(goal_indices):
            val = row_goals[i]
            # Normalize: MRE- (minimize), ACC+ (maximize), PRED40+ (maximize)
            # We use a simple heuristic: lower is better for -, higher for +
            if header[g_idx].endswith('-'):
                score -= val
            else:
                score += val
        return score

    strategies = {
        "random": random.sample(range(len(rows)), 17),
        "first_n": list(range(16)),
        "last_n": list(range(len(rows) - 17, len(rows)))
    }

    best_overall_score = float('-inf')
    winner = None
    best_row_idx = -1

    for name, indices in strategies.items():
        for idx in indices:
            goals = label(idx)

import csv

def remember(header, rows):
    """
    Identifies the best rows by calculating a normalized score for each row
    based on the goal columns (MRE-, ACC+, PRED40+).
    Returns the indices of the top 5% of rows.
    """
    goals = []
    for i, name in enumerate(header):
        if name.endswith('-'):
            goals.append((i, -1))  # Minimize
        elif name.endswith('+'):
            goals.append((i, 1))   # Maximize

    # Calculate normalized scores
    scores = [0.0] * len(rows)
    for col_idx, direction in goals:
        values = [row[col_idx] for row in rows]
        min_v, max_v = min(values), max(values)
        rng = max_v - min_v if max_v != min_v else 1.0
        
        for r_idx, row in enumerate(rows):
            val = (row[col_idx] - min_v) / rng
            if direction == -1:
                val = 1.0 - val
            scores[r_idx] += val

    # Get indices of top 5%
    num_to_return = max(1, len(rows) // 20)
    indexed_scores = sorted(enumerate(scores), key=lambda x: x[1], reverse=True)
    
    return [x[0] for x in indexed_scores[:num_to_return]]

def certify(xs, ys):
    """
    Returns True if the two sets of results are statistically indistinguishable 
    (p-value >= 0.05), implying the choice of rule makes no real difference.
    Returns False if they are statistically different (p-value < 0.05).
    """
    if len(xs) < 2 or len(ys) < 2:
        return True
    
    # Perform an independent two-sample t-test
    _, p_value = ttest_ind(xs, ys, equal_var=False)
    
    # If p_value >= 0.05, we fail to reject the null hypothesis (no difference)
    return p_value >= 0.05

import csv

def flow(header, row, seen):
    """
    Decides when to stop searching by monitoring the improvement of the 
    Pareto frontier. If the current row does not improve the best-observed 
    performance significantly after a burn-in period, we stop.
    """
    goals = []
    for i, name in enumerate(header):
        if name.endswith('-'):
            goals.append((i, -1))
        elif name.endswith('+'):
            goals.append((i, 1))

    def get_score(r):
        score = 0
        for i, (col_idx, direction) in enumerate(goals):
            val = r[col_idx]
            # Normalize based on the range of the current 'seen' set
            all_vals = [s[col_idx] for s in seen] + [val]
            min_v, max_v = min(all_vals), max(all_vals)
            rng = max_v - min_v if max_v != min_v else 1.0
            norm = (val - min_v) / rng
            if direction == -1:
                norm = 1.0 - norm
            score += norm
        return score / len(goals)

    # Burn-in period: wait for at least 50 rows to establish a baseline
    if len(seen) < 50:
        return False

    # Calculate the best score seen so far
    seen_scores = [get_score(s) for s in seen]
    best_score = max(seen_scores)
    
    # Current row score
    current_score = get_score(row)
    
    # If the current row is the best we've ever seen, keep going.
    # If we haven't seen an improvement in the last 50 rows, stop.
    if current_score > best_score:
        return False
    
    # Check if we've plateaued (no improvement in the last 50 rows)
    recent_scores = seen_scores[-50:]
    if best_score > max(recent_scores):
        return True
        
    return False

import csv

def blame(header, rows):
    """
    Identifies input columns that are 'safe' to leave at default.
    We define 'safe' as those input columns whose variance in the 
    'best' rows is significantly lower than their variance in the 
    'worst' rows, or those that show no correlation with performance.
    
    Returns a list of input column names that have the least impact 
    on performance (i.e., they are 'blameless').
    """
    # Identify goal columns
    goals = []
    for i, name in enumerate(header):
        if name.endswith('-'):
            goals.append((i, -1))
        elif name.endswith('+'):
            goals.append((i, 1))

    # Calculate aggregate performance score for each row
    def get_score(row):
        score = 0
        for i, (col_idx, direction) in enumerate(goals):
            col_vals = [r[col_idx] for r in rows if isinstance(r[col_idx], (int, float))]
            min_v, max_v = min(col_vals), max(col_vals)
            rng = max_v - min_v if max_v != min_v else 1.0
            norm = (row[col_idx] - min_v) / rng
            if direction == -1:
                norm = 1.0 - norm
            score += norm
        return score / len(goals)

    # Identify input columns
    input_indices = [i for i, name in enumerate(header) 
                     if not (name.endswith('-') or name.endswith('+'))]
    
    # Sort rows by performance
    scores = [(get_score(r), r) for r in rows]
    scores.sort(key=lambda x: x[0])
    
    # Compare variance of inputs in top 25% vs bottom 25%
    n = len(rows) // 4
    best_rows = [x[1] for x in scores[-n:]]
    worst_rows = [x[1] for x in scores[:n]]
    
    stability = []
    for idx in input_indices:
        # Only analyze numeric inputs for stability
        col_vals = [r[idx] for r in rows if isinstance(r[idx], (int, float))]
        if not col_vals: continue
        
        best_vals = [r[idx] for r in best_rows if isinstance(r[idx], (int, float))]
        worst_vals = [r[idx] for r in worst_rows if isinstance(r[idx], (int, float))]
        
        if len(best_vals) < 2 or len(worst_vals) < 2: continue
        
        # Calculate variance
        var_best = sum((x - sum(best_vals)/len(best_vals))**2 for x in best_vals) / len(best_vals)
        var_worst = sum((x - sum(worst_vals)/len(worst_vals))**2 for x in worst_vals) / len(worst_vals)
        
        # If variance is low in best rows, the setting is likely stable/safe
        stability.append((header[idx], var_best))

    # Return columns with the lowest variance in the best-performing rows
    stability.sort(key=lambda x: x[1])
    return [x[0] for x in stability]
