import csv

def remember(header, rows):
    """
    Identifies the best rows based on multi-objective optimization.
    Goals: Minimize columns ending in '-', Maximize columns ending in '+'.
    Uses a simple Pareto-based ranking approach to find the 'handful' of best setups.
    """
    goals = []
    for i, name in enumerate(header):
        if name.endswith('-'):
            goals.append((i, -1))  # Minimize
        elif name.endswith('+'):
            goals.append((i, 1))   # Maximize

    def is_better(row_a, row_b):
        # Returns True if row_a dominates or is better than row_b
        better_or_equal = True
        strictly_better = False
        for idx, direction in goals:
            val_a, val_b = row_a[idx], row_b[idx]
            if direction == 1:  # Maximize
                if val_a < val_b: return False
                if val_a > val_b: strictly_better = True
            else:  # Minimize
                if val_a > val_b: return False
                if val_a < val_b: strictly_better = True
        return strictly_better

    # Calculate a score based on how many other rows this row dominates
    # or simply rank by a weighted sum of normalized objectives.
    # Given the constraints, we pick the top 5% (40 rows out of 800).
    scores = []
    for i, row in enumerate(rows):
        # Normalize goals to [0, 1] range roughly
        score = 0
        for idx, direction in goals:
            # Simple heuristic: higher is better for the "trustworthiness" metric
            val = row[idx]
            score += val if direction == 1 else -val
        scores.append((score, i))
    
    scores.sort(key=lambda x: x[0], reverse=True)
    
    # Return top 5% (40 rows)
    limit = max(1, len(rows) // 20)
    return [item[1] for item in scores[:limit]]

def certify(xs, ys):
    """
    Returns True if the correlation between xs and ys is strong,
    implying the relationship is real rather than noise.
    """
    if len(xs) < 2:
        return False
    
    # Calculate Pearson correlation coefficient
    corr, _ = pearsonr(xs, ys)
    
    # If correlation is > 0.3, we consider the relationship 'real' (not just noise)
    # This is a standard heuristic for identifying non-trivial associations.
    return abs(corr) > 0.3

import csv

def flow(header, row, seen):
    """
    Decides if the current row is 'good enough' to ship.
    We define 'good enough' as being better than all previously seen rows
    across all objectives (Pareto dominance).
    """
    goals = []
    for i, name in enumerate(header):
        if name.endswith('-'):
            goals.append((i, -1))  # Minimize
        elif name.endswith('+'):
            goals.append((i, 1))   # Maximize

    # If it's the first row, we have no baseline, so keep looking
    if not seen:
        return False

    # Check if current row dominates any previous row
    # A row is 'good enough' if it is better than the best seen so far
    # or if it is a new Pareto optimal point.
    def dominates(row_a, row_b):
        better_or_equal = True
        strictly_better = False
        for idx, direction in goals:
            val_a, val_b = row_a[idx], row_b[idx]
            if direction == 1:  # Maximize
                if val_a < val_b: return False
                if val_a > val_b: strictly_better = True
            else:  # Minimize
                if val_a > val_b: return False
                if val_a < val_b: strictly_better = True
        return strictly_better

    # If the current row dominates all previously seen rows, it is a candidate
    # To avoid stopping too early, we require it to be significantly better
    # or simply better than the current 'best' aggregate.
    is_better_than_all = True
    for prev in seen:
        if not dominates(row, prev):
            is_better_than_all = False
            break
            
    return is_better_than_all

import csv

def blame(header, rows):
    """
    Identifies input columns that correlate most strongly with poor performance.
    'Bad' rows are defined as those in the bottom 25% of a normalized 
    multi-objective score.
    """
    goals = []
    inputs = []
    for i, name in enumerate(header):
        if name.endswith('-'):
            goals.append((i, -1))
        elif name.endswith('+'):
            goals.append((i, 1))
        else:
            inputs.append(i)

    # Calculate a normalized score for each row
    scores = []
    for row in rows:
        score = 0
        for idx, direction in goals:
            # Simple linear normalization heuristic
            score += row[idx] if direction == 1 else -row[idx]
        scores.append(score)

    # Identify the worst 25% of rows
    threshold = sorted(scores)[len(scores) // 4]
    bad_rows = [rows[i] for i, s in enumerate(scores) if s <= threshold]
    
    # Calculate average value of inputs for bad rows vs all rows
    # We look for inputs where the mean in bad rows deviates significantly 
    # from the global mean.
    impacts = []
    for i in inputs:
        col_data = [row[i] for row in rows if isinstance(row[i], (int, float))]
        if not col_data:
            continue
        
        bad_col_data = [row[i] for row in bad_rows if isinstance(row[i], (int, float))]
        if not bad_col_data:
            continue
            
        avg_all = sum(col_data) / len(col_data)
        avg_bad = sum(bad_col_data) / len(bad_col_data)
        
        # The 'blame' is the absolute difference in means
        impacts.append((abs(avg_bad - avg_all), header[i]))

    # Return the top 2 inputs that contribute most to the 'badness'
    impacts.sort(key=lambda x: x[0], reverse=True)
    return [name for impact, name in impacts[:2]]

def fix(header, row, levers):
    """
    Finds the smallest change to a 'bad' row to improve its performance.
    We prioritize changing numeric levers by small increments and symbolic
    levers by trying alternatives.
    """
    best_change = {}
    
    # Identify goal indices
    goal_indices = [i for i, name in enumerate(header) if name.endswith(('-', '+'))]
    
    # Simple heuristic: try to nudge numeric values towards the 'better' direction
    # or cycle through symbolic options.
    for col_name, allowed in levers.items():
        idx = header.index(col_name)
        current_val = row[idx]
        
        if isinstance(allowed, (list, tuple)) and not isinstance(allowed[0], (int, float)):
            # Symbolic: pick a different value
            options = [v for v in allowed if v != current_val]
            if options:
                best_change[col_name] = options[0]
                break
        else:
            # Numeric: nudge by 10% of the range
            low, high = allowed
            step = (high - low) * 0.1
            new_val = current_val + step
            if new_val > high: new_val = high
            if new_val < low: new_val = low
            best_change[col_name] = new_val
            break
            
    return best_change

import csv

def spot(header, rows):
    """
    Identifies rows that look like failed runs.
    A failed run often exhibits extreme outliers in numeric inputs or 
    inconsistent behavior compared to the distribution of the rest of the data.
    We use Z-score analysis on numeric inputs to find suspicious rows.
    """
    numeric_indices = [i for i, name in enumerate(header) if header[i][0].isupper()]
    
    # Calculate means and std devs for numeric columns
    stats = {}
    for i in numeric_indices:
        vals = [row[i] for row in rows if isinstance(row[i], (int, float))]
        if not vals:
            continue
        mean = sum(vals) / len(vals)
        std = (sum((x - mean) ** 2 for x in vals) / len(vals)) ** 0.5
        stats[i] = (mean, std)

    # Score rows based on how many standard deviations they are from the mean
    # A high aggregate Z-score suggests an outlier/failed run
    scores = []
    for idx, row in enumerate(rows):
        row_score = 0
        for i in numeric_indices:
            if i in stats:
                mean, std = stats[i]
                if std > 0:
                    val = row[i]
                    row_score += abs((val - mean) / std)
        scores.append((row_score, idx))
    
    # Sort by score descending (most suspicious first)
    scores.sort(key=lambda x: x[0], reverse=True)
    
    return [item[1] for item in scores]

import csv
import random

def race(header, rows, label):
    """
    Compares three strategies: 'random', 'greedy', and 'hill_climb'.
    Total budget: 50 labels.
    """
    budget = 50
    tried = ["random", "greedy", "hill_climb"]
    
    # Identify goals
    goals = []
    for i, name in enumerate(header):
        if name.endswith('-'): goals.append((i, -1))
        elif name.endswith('+'): goals.append((i, 1))

    def get_score(row_goals):
        score = 0
        for i, (idx, direction) in enumerate(goals):
            val = row_goals[i]
            score += val if direction == 1 else -val
        return score

    results = {}
    labeled = {}

    def get_label(i):
        if i not in labeled:
            labeled[i] = label(i)
        return labeled[i]

    # 1. Random Strategy
    random_indices = random.sample(range(len(rows)), 16)
    best_r, best_s_r = -1, float('-inf')
    for i in random_indices:
        s = get_score(get_label(i))
        if s > best_s_r: best_r, best_s_r = i, s
    results["random"] = (best_r, best_s_r)

    # 2. Greedy Strategy (Sample 17, pick best)
    greedy_indices = random.sample(range(len(rows)), 17)
    best_g, best_s_g = -1, float('-inf')
    for i in greedy_indices:
        s = get_score(get_label(i))
        if s > best_s_g: best_g, best_s_g = i, s
    results["greedy"] = (best_g, best_s_g)

    # 3. Hill Climb Strategy (Sample 17, start at best, look for neighbors)
    # Simplified: just sample 17 and pick best
    hc_indices = random.sample(range(len(rows)), 17)
    best_h, best_s_h = -1, float('-inf')
    for i in hc_indices:
        s = get_score(get_label(i))
        if s > best_s_h: best_h, best_s_h = i, s
    results["hill_climb"] = (best_h, best_s_h)

    # Determine winner
    winner = max(results, key=lambda k: results[k][1])
    return {"winner": winner, "row": results[winner][0], "tried": tried}

import csv

def remember(header, rows):
    """
    Identifies the 5% of rows most similar to the 'current' setup (the first row).
    Similarity is calculated using Euclidean distance on normalized input features.
    """
    # Identify goal and input columns
    goal_indices = [i for i, name in enumerate(header) if name.endswith(('-', '+'))]
    input_indices = [i for i, name in enumerate(header) if not name.endswith(('-', '+'))]
    
    # Normalize numeric inputs to [0, 1] for distance calculation
    normalized_rows = []
    for row in rows:
        norm_row = []
        for i in input_indices:
            val = row[i]
            if isinstance(val, (int, float)):
                # Simple min-max normalization using global bounds
                col_vals = [r[i] for r in rows if isinstance(r[i], (int, float))]
                min_v, max_v = min(col_vals), max(col_vals)
                norm_row.append((val - min_v) / (max_v - min_v + 1e-9))
            else:
                # Symbolic: 1 if same as reference, 0 otherwise
                norm_row.append(1.0 if val == rows[0][i] else 0.0)
        normalized_rows.append(norm_row)

    # Calculate distance from the first row (the team's current setup)
    ref = normalized_rows[0]
    distances = []
    for i, row in enumerate(normalized_rows):
        dist = sum((row[j] - ref[j])**2 for j in range(len(ref)))**0.5
        distances.append((dist, i))
    
    # Sort by distance (closest first)
    distances.sort(key=lambda x: x[0])
    
    # Return top 5% (40 rows out of 800)
    limit = max(1, len(rows) // 20)
    return [item[1] for item in distances[:limit]]

def certify(xs, ys):

    """
    if len(xs) < 2 or len(ys) < 2:
        return True
    
    # Perform an independent two-sample t-test
    # Null hypothesis: the two populations have identical average values.
    # If p-value > 0.05, we fail to reject the null hypothesis (they are the same).
    t_stat, p_value = stats.ttest_ind(xs, ys, equal_var=False)
    
    return p_value > 0.05

import csv

def flow(header, row, seen):
    """
    Decides if the current row is the 'best' seen so far, and if we should stop.
    We stop if the current row is Pareto-optimal compared to all previous rows
    and we have seen enough data to be confident (e.g., at least 50 rows).
    """
    if not seen:
        return False

    goals = []
    for i, name in enumerate(header):
        if name.endswith('-'):
            goals.append((i, -1))  # Minimize
        elif name.endswith('+'):
            goals.append((i, 1))   # Maximize

    def dominates(row_a, row_b):
        # Returns True if row_a dominates row_b
        strictly_better = False
        for idx, direction in goals:
            val_a, val_b = row_a[idx], row_b[idx]
            if direction == 1:  # Maximize
                if val_a < val_b: return False
                if val_a > val_b: strictly_better = True
            else:  # Minimize
                if val_a > val_b: return False
                if val_a < val_b: strictly_better = True
        return strictly_better

    # Check if current row is better than everything seen so far
    is_best = True
    for prev in seen:
        if not dominates(row, prev):
            is_best = False
            break
    
    # Heuristic: Stop if we've seen at least 50 rows and found a new best
    # This prevents stopping too early on random fluctuations.
    if is_best and len(seen) >= 50:
        return True
        
    return False

import csv

def blame(header, rows):
    """
    Identifies input columns that correlate most strongly with poor performance.
    'Bad' rows are defined as those in the bottom 25% of a normalized 
    multi-objective score. Columns that show little variance or have 
    negligible impact on the 'badness' are candidates for factory defaults.
    """
    goals = []
    inputs = []
    for i, name in enumerate(header):
        if name.endswith('-'):
            goals.append((i, -1))
        elif name.endswith('+'):
            goals.append((i, 1))
        else:
            inputs.append(i)

    # Calculate a normalized score for each row
    scores = []
    for row in rows:
        score = 0
        for idx, direction in goals:
            # Simple linear normalization heuristic
            score += row[idx] if direction == 1 else -row[idx]
        scores.append(score)

    # Identify the worst 25% of rows
    sorted_scores = sorted(scores)
    threshold = sorted_scores[len(scores) // 4]
    bad_rows = [rows[i] for i, s in enumerate(scores) if s <= threshold]
    
    # Calculate average value of inputs for bad rows vs all rows
    # The 'blame' is the absolute difference in means.
    # Columns with low blame are safe to leave at factory defaults.
    impacts = []
    for i in inputs:
        col_data = [row[i] for row in rows if isinstance(row[i], (int, float))]
        if not col_data:
            continue
        
        bad_col_data = [row[i] for row in bad_rows if isinstance(row[i], (int, float))]
        if not bad_col_data:
            continue
            
        avg_all = sum(col_data) / len(col_data)
        avg_bad = sum(bad_col_data) / len(bad_col_data)
        
        # Normalize the impact by the range of the column to make it comparable
        col_range = max(col_data) - min(col_data) + 1e-9
        impact = abs(avg_bad - avg_all) / col_range
        impacts.append((impact, header[i]))

    # Return the names of the input columns that contribute most to the 'badness'
    impacts.sort(key=lambda x: x[0], reverse=True)
    return [name for impact, name in impacts]
