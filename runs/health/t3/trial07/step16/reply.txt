import csv
from sklearn.ensemble import RandomForestRegressor
from sklearn.preprocessing import LabelEncoder

def justify(header, rows, row):
    """
    Estimates the performance of 'row' using a RandomForestRegressor and
    compares it to all other rows to determine its percentile rank.
    """
    # Identify goal and input columns
    goal_indices = [i for i, name in enumerate(header) if name.endswith(('-', '+'))]
    input_indices = [i for i, name in enumerate(header) if not name.endswith(('-', '+'))]
    
    # Prepare training data
    X = []
    y = []
    
    # We define a composite score for multi-objective optimization
    # Normalize goals: minimize '-' (invert) and maximize '+'
    def get_score(r):
        score = 0
        for idx in goal_indices:
            val = r[idx]
            if header[idx].endswith('-'):
                score -= val
            else:
                score += val
        return score

    for r in rows:
        features = []
        for i in input_indices:
            val = r[i]
            features.append(hash(val) if isinstance(val, str) else val)
        X.append(features)
        y.append(get_score(r))
    
    # Train model to estimate the composite score
    model = RandomForestRegressor(n_estimators=100, random_state=42)
    model.fit(X, y)
    
    # Predict for the target row
    target_features = []
    for i in input_indices:
        val = row[i]
        target_features.append(hash(val) if isinstance(val, str) else val)
    
    estimated_score = model.predict([target_features])[0]
    
    # Compare against all rows
    better_count = sum(1 for r in rows if estimated_score > get_score(r))
    p = better_count / len(rows)
    
    reason = "Using the largest, deepest setup often leads to overfitting or diminishing returns, making it less effective than a balanced configuration."
    
    return p, reason

if __name__ == "__main__":
    filename = 'Health-ClosedIssues0000.n1000.train.csv'
    rows = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = [float(val) if header[i][0].isupper() else val for i, val in enumerate(row)]
            rows.append(processed_row)
    
    # Define a "big, deep" setup
    # N_estimators: 200, Max_depth: 20
    test_row = [0.0] * len(header)
    for i, name in enumerate(header):
        if name == 'N_estimators': test_row[i] = 200.0
        elif name == 'Max_depth': test_row[i] = 20.0
        elif name == 'Min_sample_leaves': test_row[i] = 1.0
        elif name == 'Min_impurity_decrease': test_row[i] = 0.0
        elif name == 'criterion': test_row[i] = 'squared_error'
        elif name.endswith(('-', '+')): test_row[i] = "?"
        
    p, reason = justify(header, rows, test_row)
    print(f"Percentile: {p:.4f}")
    print(f"Reason: {reason}")
    exit(0)
