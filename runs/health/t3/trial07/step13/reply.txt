import csv
from scipy import stats

def certify(xs, ys):
    """
    Returns True if the two distributions are statistically indistinguishable 
    (i.e., they are the same apart from noise), False if they really differ.
    Uses a t-test to compare means.
    """
    if len(xs) < 2 or len(ys) < 2:
        return True
    
    # Perform an independent two-sample t-test
    # Null hypothesis: the two populations have identical average values.
    # If p-value > 0.05, we fail to reject the null hypothesis (they are the same).
    t_stat, p_value = stats.ttest_ind(xs, ys, equal_var=False)
    
    return p_value > 0.05

if __name__ == "__main__":
    filename = 'Health-ClosedIssues0000.n1000.train.csv'
    rows = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = [float(val) if header[i][0].isupper() else val for i, val in enumerate(row)]
            rows.append(processed_row)

    # The customer asks about the "two built-in rules". 
    # In the context of this dataset, the 'criterion' column has 3 values:
    # 'absolute_error', 'poisson', 'squared_error'.
    # We compare the primary goal (MRE-) across the two most common rules.
    
    mre_idx = 0
    crit_idx = header.index('criterion')
    
    # Filter rows by criterion
    group_a = [row[mre_idx] for row in rows if row[crit_idx] == 'absolute_error']
    group_b = [row[mre_idx] for row in rows if row[crit_idx] == 'squared_error']
    
    are_same = certify(group_a, group_b)
    
    print("Analyzing if the choice of 'criterion' rule makes a significant difference...")
    if are_same:
        print("Result: The choice of rule does not make a statistically significant difference.")
        print("Advice: You can stop arguing about it; the performance impact is negligible.")
    else:
        print("Result: The choice of rule leads to statistically different outcomes.")
        print("Advice: The choice matters; you should continue to evaluate which rule is best.")
    
    exit(0)
