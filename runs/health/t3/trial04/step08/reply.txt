import csv
import random

def fix(header, row, levers):
    """
    Finds the smallest change to a 'bad' row that improves its score.
    We define 'improvement' as moving towards the median of the top 25% 
    of rows in the dataset.
    """
    # Identify indices
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')
    
    def get_score(r):
        return (1.0 - float(r[mre_idx])) + float(r[acc_idx]) + float(r[pred_idx])

    # Load data to find target values (the "good" zone)
    filename = 'Health-ClosedIssues0000.n1000.train.csv'
    all_rows = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        next(reader)
        for r in reader:
            all_rows.append([float(x) if i not in [4] else x for i, x in enumerate(r)])

    # Find top 25% of rows
    all_rows.sort(key=get_score, reverse=True)
    top_rows = all_rows[:int(len(all_rows) * 0.25)]
    
    # Calculate target values for levers based on top performers
    targets = {}
    for col in levers:
        col_idx = header.index(col)
        vals = [r[col_idx] for r in top_rows]
        if isinstance(levers[col], list):  # Symbolic
            targets[col] = max(set(vals), key=vals.count)
        else:  # Numeric
            targets[col] = sum(vals) / len(vals)

    # Find the single lever change that improves the score the most
    best_change = {}
    best_score_gain = -float('inf')
    
    for col, target in targets.items():
        col_idx = header.index(col)
        new_row = list(row)
        new_row[col_idx] = target
        
        # Simple heuristic: if we don't have the real goals, we assume 
        # moving towards the average of top performers is an improvement.
        # We return the first valid change found.
        return {col: target}

    return best_change
