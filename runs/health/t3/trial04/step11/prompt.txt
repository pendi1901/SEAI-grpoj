import csv

def remember(header, rows):
    """
    Identifies the best rows based on multi-objective optimization.
    Goals: MRE- (minimize), ACC+ (maximize), PRED40+ (maximize).
    We use a simple Pareto-based ranking or weighted normalization.
    Given the constraints, we normalize the goals and pick the top 5%.
    """
    # Identify indices
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    # Extract goals and normalize
    # MRE is minimize, so we use (1 - normalized_value) to treat as maximize
    data = []
    for i, row in enumerate(rows):
        mre = row[mre_idx]
        acc = row[acc_idx]
        pred = row[pred_idx]
        
        # Simple score: Higher is better
        # We assume values are within reasonable ranges [0, 1]
        score = (1.0 - mre) + acc + pred
        data.append((score, i))

    # Sort by score descending
    data.sort(key=lambda x: x[0], reverse=True)

    # Return top 5% (800 rows * 0.05 = 40 rows)
    limit = max(1, int(len(rows) * 0.05))
    return [item[1] for item in data[:limit]]

import csv

def guess(header, train, test):
    """
    Estimates the first goal column (MRE-) for test rows using a 
    RandomForestRegressor with OneHotEncoding for symbolic columns.
    """
    df_train = pd.DataFrame(train, columns=header)
    df_test = pd.DataFrame(test, columns=header)

    # Identify goal columns and input columns
    goal_cols = [col for col in header if col.endswith('-') or col.endswith('+')]
    target_col = goal_cols[0]  # MRE-
    input_cols = [col for col in header if col not in goal_cols]

    # Identify numeric and categorical columns
    numeric_features = [col for col in input_cols if df_train[col].dtype != 'object']
    categorical_features = [col for col in input_cols if df_train[col].dtype == 'object']

    # Create a preprocessor
    preprocessor = ColumnTransformer(
        transformers=[
            ('num', 'passthrough', numeric_features),
            ('cat', OneHotEncoder(handle_unknown='ignore'), categorical_features)
        ])

    # Create a pipeline
    model = Pipeline(steps=[
        ('preprocessor', preprocessor),
        ('regressor', RandomForestRegressor(n_estimators=100, random_state=42))
    ])

    # Train model
    X_train = df_train[input_cols]
    y_train = df_train[target_col]
    model.fit(X_train, y_train)

    # Predict
    predictions = model.predict(df_test[input_cols])
    return predictions.tolist()

def certify(xs, ys):
    """
    Returns True if the correlation between xs and ys is high (suggesting 
    the relationship is real), False if it is low (suggesting noise).
    Using a threshold of 0.3 for moderate correlation.
    """
    if len(xs) < 2:
        return False
    corr, _ = pearsonr(xs, ys)
    # If correlation is significant (e.g., > 0.3), we consider it 'real'
    return abs(corr) > 0.3

                perf -= val  # Minimize
            else:
                perf += val  # Maximize
        
        sizes.append(size)
        performances.append(perf)

import csv

def flow(header, row, seen):
    """
    Decides if the current row is 'good enough' to ship.
    We define 'good enough' as being in the top 5% of the expected 
    distribution of performance based on historical data (seen).
    """
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    def get_score(r):
        # Normalize goals: (1 - MRE) + ACC + PRED40
        # We assume values are within [0, 1]
        return (1.0 - float(r[mre_idx])) + float(r[acc_idx]) + float(r[pred_idx])

    # If this is the first row, we have no baseline
    if not seen:
        return False

    # Calculate score for current row
    current_score = get_score(row)
    
    # Calculate scores for all seen rows
    seen_scores = [get_score(r) for r in seen]
    
    # Threshold: Is this row better than 95% of what we have seen so far?
    # We use a dynamic threshold based on the current distribution.
    seen_scores.sort(reverse=True)
    top_5_percent_idx = max(0, int(len(seen_scores) * 0.05))
    threshold = seen_scores[top_5_percent_idx]
    
    return current_score >= threshold

def blame(header, rows):
    """
    Identifies which input columns are most predictive of poor performance.
    We define 'bad' as the bottom 25% of rows based on an aggregate score.
    We use a Random Forest Classifier to identify feature importance.
    """
    df = pd.DataFrame(rows, columns=header)
    
    goal_cols = [c for c in header if c.endswith('-') or c.endswith('+')]
    input_cols = [c for c in header if c not in goal_cols]
    
    # Calculate aggregate performance score
    # Normalize: (1 - MRE) + ACC + PRED40
    def get_score(row):
        mre = row[header.index('MRE-')]
        acc = row[header.index('ACC+')]
        pred = row[header.index('PRED40+')]
        return (1.0 - mre) + acc + pred

    scores = [get_score(row) for row in rows]
    df['score'] = scores
    
    # Define 'bad' as the bottom 25%
    threshold = df['score'].quantile(0.25)
    df['is_bad'] = (df['score'] <= threshold).astype(int)
    
    # Identify feature types
    numeric_features = [c for c in input_cols if df[c].dtype != 'object']
    categorical_features = [c for c in input_cols if df[c].dtype == 'object']
    
    # Preprocessor handles string conversion via OneHotEncoder
    preprocessor = ColumnTransformer(
        transformers=[
            ('num', 'passthrough', numeric_features),
            ('cat', OneHotEncoder(handle_unknown='ignore'), categorical_features)
        ])
    
    # Use Classifier to predict 'is_bad'
    model = Pipeline(steps=[
        ('preprocessor', preprocessor),
        ('classifier', RandomForestClassifier(n_estimators=50, random_state=42))
    ])
    
    model.fit(df[input_cols], df['is_bad'])
    
    # Extract feature importances
    importances = model.named_steps['classifier'].feature_importances_
    
    # Map importances back to original input column names
    cat_names = []
    if categorical_features:
        cat_names = model.named_steps['preprocessor'].named_transformers_['cat'].get_feature_names_out(categorical_features)
    
    all_feature_names = numeric_features + list(cat_names)
    
    importance_map = {col: 0.0 for col in input_cols}
    for name, imp in zip(all_feature_names, importances):
        for col in input_cols:
            if name.startswith(col):
                importance_map[col] += imp
                break
                
    # Return columns sorted by importance
    sorted_cols = sorted(importance_map.items(), key=lambda x: x[1], reverse=True)
    return [col for col, imp in sorted_cols if imp > 0]

import csv
import pandas as pd
from sklearn.ensemble import RandomForestRegressor
from sklearn.preprocessing import OneHotEncoder
from sklearn.compose import ColumnTransformer
from sklearn.pipeline import Pipeline

def justify(header, rows, row):
    """
    Estimates the performance of the given row and compares it to all rows.
    Returns (fraction_better, reason_string).
    """
    df = pd.DataFrame(rows, columns=header)
    goal_cols = [c for c in header if c.endswith('-') or c.endswith('+')]
    input_cols = [c for c in header if c not in goal_cols]

    # Helper to calculate aggregate score
    def get_score(r):
        score = 0
        for col in goal_cols:
            val = float(r[header.index(col)])
            if col.endswith('-'):
                score -= val
            else:
                score += val
        return score

    # Identify feature types for the preprocessor
    numeric_features = [c for c in input_cols if df[c].dtype != 'object']
    categorical_features = [c for c in input_cols if df[c].dtype == 'object']
    
    preprocessor = ColumnTransformer(
        transformers=[
            ('num', 'passthrough', numeric_features),
            ('cat', OneHotEncoder(handle_unknown='ignore'), categorical_features)
        ])

    # Estimate each goal column individually
    est_row = list(row)
    for goal in goal_cols:
        model = Pipeline(steps=[
            ('preprocessor', preprocessor),
            ('regressor', RandomForestRegressor(n_estimators=50, random_state=42))
        ])
        model.fit(df[input_cols], df[goal])
        
        # Prepare input row for prediction
        input_df = pd.DataFrame([row], columns=header)[input_cols]
        est_val = model.predict(input_df)[0]
        est_row[header.index(goal)] = est_val
    
    est_score = get_score(est_row)
    all_scores = [get_score(r) for r in rows]
    
    # Fraction of rows that are worse than our estimated row
    better_than_count = sum(1 for s in all_scores if est_score > s)
    p = better_than_count / len(rows)
    
    reason = f"Based on our predictive model, this configuration outperforms {p:.1%} of all historical setups."
    return (p, reason)

def choose(header, rows, label):
    """
    Finds the best row by sampling 50 setups.
    We use a simple random sampling strategy to explore the space,
    labeling 50 rows and picking the one with the best aggregate score.
    """
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    def get_score(goals):
        # goals = [MRE-, ACC+, PRED40+]
        # Score = (1 - MRE) + ACC + PRED40
        return (1.0 - goals[0]) + goals[1] + goals[2]

    # Indices of rows we can sample
    indices = list(range(len(rows)))
    # Sample 50 indices
    sample_size = min(50, len(indices))
    sampled_indices = random.sample(indices, sample_size)

    best_idx = -1
    best_score = -float('inf')

    for i in sampled_indices:
        # Get true values from the oracle
        goals = label(i)
        score = get_score(goals)
        
        if score > best_score:
            best_score = score
            best_idx = i

    return best_idx

import csv

def spot(header, rows):
    """
    Identifies rows that look like failed runs.
    A failed run often results in extreme outliers in the goal columns
    (e.g., MRE- near 1.0 or 0.0, ACC+ near 0.0).
    We calculate a 'suspicion score' based on how far a row's goals
    deviate from the population mean.
    """
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    # Calculate means for the goals to establish a baseline
    mres = [r[mre_idx] for r in rows]
    accs = [r[acc_idx] for r in rows]
    preds = [r[pred_idx] for r in rows]
    
    m_mre = sum(mres) / len(mres)
    m_acc = sum(accs) / len(accs)
    m_pred = sum(preds) / len(preds)

    suspicion = []
    for i, row in enumerate(rows):
        # Suspicion: Euclidean distance from the mean of goals
        # Failed runs often have MRE near 1.0 (bad) and ACC/PRED near 0.0 (bad)
        d_mre = (row[mre_idx] - m_mre) ** 2
        d_acc = (row[acc_idx] - m_acc) ** 2
        d_pred = (row[pred_idx] - m_pred) ** 2
        
        # Weight MRE higher as it is a primary error metric
        score = (d_mre * 2.0) + d_acc + d_pred
        suspicion.append((score, i))

    # Sort by suspicion score descending
    suspicion.sort(key=lambda x: x[0], reverse=True)
    
    return [item[1] for item in suspicion]
