```python
import csv
import pandas as pd
from sklearn.ensemble import RandomForestRegressor
from sklearn.preprocessing import OneHotEncoder
from sklearn.compose import ColumnTransformer
from sklearn.pipeline import Pipeline

def get_score(header, row):
    """Calculates aggregate score: (1 - MRE) + ACC + PRED40."""
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')
    return (1.0 - float(row[mre_idx])) + float(row[acc_idx]) + float(row[pred_idx])

def justify(header, rows, row):
    """
    Estimates the performance of the given row and compares it to all rows.
    Returns (fraction_better, reason_string).
    """
    df = pd.DataFrame(rows, columns=header)
    goal_cols = [c for c in header if c.endswith('-') or c.endswith('+')]
    input_cols = [c for c in header if c not in goal_cols]

    # Identify feature types
    numeric_features = [c for c in input_cols if not isinstance(rows[0][header.index(c)], str)]
    categorical_features = [c for c in input_cols if isinstance(rows[0][header.index(c)], str)]
    
    preprocessor = ColumnTransformer(
        transformers=[
            ('num', 'passthrough', numeric_features),
            ('cat', OneHotEncoder(handle_unknown='ignore'), categorical_features)
        ])

    # Estimate each goal column individually
    est_row = list(row)
    for goal in goal_cols:
        model = Pipeline(steps=[
            ('preprocessor', preprocessor),
            ('regressor', RandomForestRegressor(n_estimators=50, random_state=42))
        ])
        model.fit(df[input_cols], df[goal])
        
        input_df = pd.DataFrame([row], columns=header)[input_cols]
        est_val = model.predict(input_df)[0]
        est_row[header.index(goal)] = est_val
    
    est_score = get_score(header, est_row)
    all_scores = [get_score(header, r) for r in rows]
    
    # Fraction of rows that are worse than our estimated row
    better_than_count = sum(1 for s in all_scores if est_score > s)
    p = better_than_count / len(rows)
    
    reason = "Bigger and deeper setups often lead to overfitting or diminishing returns, as shown by this configuration outperforming only {:.1%} of historical setups.".format(p)
    return (p, reason)

if __name__ == "__main__":
    filename = 'Health-ClosedIssues0000.n1000.train.csv'
    rows = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = []
            for i, val in enumerate(row):
                if i == 4:
                    processed_row.append(val)
                else:
                    processed_row.append(float(val))
            rows.append(processed_row)

    # Create a "biggest, deepest" setup
    # N_estimators: 200, Max_depth: 20, Min_sample_leaves: 1, Min_impurity_decrease: 0, criterion: 'squared_error'
    test_row = [0.0, 0.0, 0.0, 200.0, 'squared_error', 1.0, 0.0, 20.0]
    goal_indices = [header.index('MRE-'), header.index('ACC+'), header.index('PRED40+')]
    for idx in goal_indices:
        test_row[idx] = "?"
            
    p, reason = justify(header, rows, test_row)
    print(reason)
```
