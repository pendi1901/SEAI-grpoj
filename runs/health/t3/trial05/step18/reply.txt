```python
import csv

def fix(header, row, levers):
    """
    Recommends a single change to a 'bad' row to improve performance.
    It evaluates potential changes by comparing the row's current goal 
    metrics against the average of the dataset (or a simple heuristic).
    """
    goal_indices = [i for i, h in enumerate(header) if h.endswith('-') or h.endswith('+')]
    
    def get_score(r):
        # Simple heuristic: normalize goals to 0-1 relative to the row's values
        # We want to minimize MRE- and maximize ACC+/PRED40+
        score = 0
        for i in goal_indices:
            val = r[i]
            if header[i].endswith('-'):
                score -= val
            else:
                score += val
        return score

    best_change = {}
    best_score = -float('inf')
    
    # Try changing each lever one by one
    for col_name, allowed in levers.items():
        col_idx = header.index(col_name)
        original_val = row[col_idx]
        
        candidates = []
        if isinstance(allowed, list):
            # Symbolic: try all allowed values
            candidates = [v for v in allowed if v != original_val]
        else:
            # Numeric: try low, mid, high
            low, high = allowed
            candidates = [low, (low + high) / 2, high]
            candidates = [v for v in candidates if v != original_val]
            
        for val in candidates:
            new_row = list(row)
            new_row[col_idx] = val
            
            # Evaluate the change
            score = get_score(new_row)
            if score > best_score:
                best_score = score
                best_change = {col_name: val}
            
    return best_change

if __name__ == "__main__":
    filename = 'Health-ClosedIssues0000.n1000.train.csv'
    rows = []
    try:
        with open(filename, 'r') as f:
            reader = csv.reader(f)
            header = next(reader)
            for row in reader:
                processed_row = []
                for i, val in enumerate(row):
                    try:
                        processed_row.append(float(val))
                    except ValueError:
                        processed_row.append(val)
                rows.append(processed_row)
        
        # Define levers based on dataset
        levers = {
            'N_estimators': (50, 200),
            'criterion': ['absolute_error', 'poisson', 'squared_error'],
            'Min_sample_leaves': (1, 20),
            'Max_depth': (2, 20)
        }
        
        # Pick a sample row (e.g., the first one)
        bad_row = rows[0]
        recommendation = fix(header, bad_row, levers)
        
        print("Recommendation for improvement:")
        for col, val in recommendation.items():
            print(f"Change '{col}' to {val}")
        exit(0)
    except Exception as e:
        print(f"Error: {e}")
        exit(1)
```
