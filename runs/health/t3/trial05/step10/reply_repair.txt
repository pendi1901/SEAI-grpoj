import csv
import random

def race(header, rows, label):
    """
    Compares three strategies: random, greedy, and hill-climbing.
    Total budget is 50 labels.
    """
    goal_indices = [i for i, h in enumerate(header) if h.endswith('-') or h.endswith('+')]
    
    labeled = {}
    def get_label(i):
        if i not in labeled:
            if len(labeled) >= 50:
                raise Exception("Budget exceeded")
            labeled[i] = label(i)
        return labeled[i]

    def get_score(row_idx):
        goals = get_label(row_idx)
        score = 0
        for j, g_idx in enumerate(goal_indices):
            val = goals[j]
            # Normalize: MRE- (minimize), ACC+/PRED40+ (maximize)
            # Since we don't have global min/max, we use raw values 
            # assuming consistent scale or relative comparison.
            if header[g_idx].endswith('-'):
                score -= val
            else:
                score += val
        return score

    # Budget: 50 labels total. 
    # We split into 3 strategies: 16, 17, 17 labels.
    
    # 1. Random Strategy (16 labels)
    def strategy_random():
        best_idx = -1
        best_score = -float('inf')
        for _ in range(16):
            idx = random.randint(0, len(rows) - 1)
            score = get_score(idx)
            if score > best_score:
                best_score, best_idx = score, idx
        return best_idx, best_score

    # 2. Greedy Strategy (17 labels)
    def strategy_greedy():
        samples = random.sample(range(len(rows)), 17)
        best_idx = max(samples, key=get_score)
        return best_idx, get_score(best_idx)

    # 3. Hill Climbing Strategy (17 labels)
    def strategy_hill():
        curr = random.randint(0, len(rows) - 1)
        # 1 initial + 16 steps
        for _ in range(16):
            neighbor = random.randint(0, len(rows) - 1)
            if get_score(neighbor) > get_score(curr):
                curr = neighbor
        return curr, get_score(curr)

    results = {
        "random": strategy_random(),
        "greedy": strategy_greedy(),
        "hill": strategy_hill()
    }

    winner = max(results, key=lambda k: results[k][1])
    return {
        "winner": winner,
        "row": results[winner][0],
        "tried": ["random", "greedy", "hill"]
    }

if __name__ == "__main__":
    filename = 'Health-ClosedIssues0000.n1000.train.csv'
    rows = []
    try:
        with open(filename, 'r') as f:
            reader = csv.reader(f)
            header = next(reader)
            for row in reader:
                processed_row = []
                for i, val in enumerate(row):
                    try:
                        processed_row.append(float(val))
                    except ValueError:
                        processed_row.append(val)
                rows.append(processed_row)

        def label(i):
            return [rows[i][j] for j, h in enumerate(header) if h.endswith('-') or h.endswith('+')]

        result = race(header, rows, label)
        print(f"The winner is: {result['winner']} (found best row at index {result['row']})")
        print(f"Strategies compared: {', '.join(result['tried'])}")
        exit(0)
    except Exception as e:
        print(f"Error: {e}")
        exit(1)
