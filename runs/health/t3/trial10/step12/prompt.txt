import csv

def remember(header, rows):
    """
    Identifies the best rows based on multi-objective optimization.
    Goals: MRE- (minimize), ACC+ (maximize), PRED40+ (maximize).
    We use a simple Pareto-based approach or weighted normalization.
    Given the constraints, we normalize the goals and pick the top 5%.
    """
    # Identify column indices
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    # Extract values for normalization
    mres = [r[mre_idx] for r in rows]
    accs = [r[acc_idx] for r in rows]
    preds = [r[pred_idx] for r in rows]

    min_mre, max_mre = min(mres), max(mres)
    min_acc, max_acc = min(accs), max(accs)
    min_pred, max_pred = min(preds), max(preds)

    def normalize(val, min_v, max_v, reverse=False):
        if max_v == min_v: return 0
        norm = (val - min_v) / (max_v - min_v)
        return (1 - norm) if reverse else norm

    scores = []
    for i, row in enumerate(rows):
        # Score = (1 - normalized_MRE) + normalized_ACC + normalized_PRED
        s = (normalize(row[mre_idx], min_mre, max_mre, True) +
             normalize(row[acc_idx], min_acc, max_acc) +
             normalize(row[pred_idx], min_pred, max_pred))
        scores.append((s, i))

    # Sort by score descending
    scores.sort(key=lambda x: x[0], reverse=True)
    
    # Return top 5% (800 rows * 0.05 = 40 rows)
    limit = max(1, int(len(rows) * 0.05))
    return [item[1] for item in scores[:limit]]

import csv

def flow(header, row, seen):
    """
    Decides if the current row is 'good enough' to ship.
    We define 'good enough' as being in the top 5% of the expected distribution
    based on the historical data (seen).
    """
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    # If this is the first row, we have no baseline, so keep looking.
    if not seen:
        return False

    # Extract goals from seen rows
    mres = [r[mre_idx] for r in seen]
    accs = [r[acc_idx] for r in seen]
    preds = [r[pred_idx] for r in seen]

    # Calculate current row's relative performance
    # We use a simple heuristic: is this row better than 95% of what we've seen?
    # A row is better if MRE is lower, ACC is higher, and PRED40 is higher.
    
    def is_better(r1, r2):
        # Returns True if r1 is better than r2
        return (r1[mre_idx] <= r2[mre_idx] and 
                r1[acc_idx] >= r2[acc_idx] and 
                r1[pred_idx] >= r2[pred_idx])

    # Count how many seen rows are worse than the current row
    better_count = sum(1 for s in seen if is_better(row, s))
    
    # If this row is better than 95% of the seen rows, ship it.
    # We require at least 20 rows to have been seen to make a statistically 
    # meaningful decision.
    if len(seen) >= 20:
        if better_count / len(seen) >= 0.95:
            return True
            
    return False

def blame(header, rows):
    """
    Identifies input columns that best explain poor performance.
    We define 'bad' as the bottom 25% of rows based on a composite score.
    We then train a Random Forest to predict the score and use feature 
    importances to identify the culprits.
    """
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')
    
    goal_indices = [i for i, h in enumerate(header) if h.endswith('-') or h.endswith('+')]
    input_indices = [i for i in range(len(header)) if i not in goal_indices]
    input_names = [header[i] for i in input_indices]

    # Calculate a composite score for each row (lower is worse)
    # Normalize goals: MRE (min), ACC (max), PRED (max)
    mres = [r[mre_idx] for r in rows]
    accs = [r[acc_idx] for r in rows]
    preds = [r[pred_idx] for r in rows]
    
    def norm(val, vals, reverse=False):
        mn, mx = min(vals), max(vals)
        if mx == mn: return 0
        n = (val - mn) / (mx - mn)
        return (1 - n) if reverse else n

    scores = []
    for r in rows:
        s = (norm(r[mre_idx], mres, True) + norm(r[acc_idx], accs) + norm(r[pred_idx], preds))
        scores.append(s)

    # Prepare data for Random Forest
    X = []
    encoders = {}
    for i in input_indices:
        if isinstance(rows[0][i], str):
            le = LabelEncoder()
            col_data = [r[i] for r in rows]
            le.fit(col_data)
            encoders[i] = le
    
    for r in rows:
        row_x = []
        for i in input_indices:
            val = r[i]
            if i in encoders:
                row_x.append(encoders[i].transform([val])[0])
            else:
                row_x.append(val)
        X.append(row_x)

    # Train model to predict the composite score
    model = RandomForestRegressor(n_estimators=50, random_state=42)
    model.fit(X, scores)
    
    # Get feature importances
    importances = model.feature_importances_
    # Sort indices by importance
    sorted_indices = sorted(range(len(importances)), key=lambda i: importances[i], reverse=True)
    
    # Return the top 3 most influential input columns
    return [input_names[i] for i in sorted_indices[:3]]

    """

        return [encoders[i].transform([r[i]])[0] if i in encoders else r[i] for i in input_indices]

    X_train = [transform_row(r) for r in rows]

    estimated_goals = []

        y_train = [r[g_idx] for r in rows]
        model = RandomForestRegressor(n_estimators=50, random_state=42)

        estimated_goals.append(model.predict([transform_row(row)])[0])
    
    # Calculate how many rows are worse than our estimated row

    def is_better(r_est, r_actual):
        # r_est is [MRE, ACC, PRED]
        # r_actual is the full row
        mre_better = r_est[0] <= r_actual[goal_indices[0]]
        acc_better = r_est[1] >= r_actual[goal_indices[1]]
        pred_better = r_est[2] >= r_actual[goal_indices[2]]
        return mre_better and acc_better and pred_better

    better_count = sum(1 for r in rows if is_better(estimated_goals, r))
    p = better_count / len(rows)
    
    reason = f"This setup is estimated to outperform {p:.1%} of all historical configurations based on its input parameters."

import csv

def choose(header, rows, label):
    """
    Finds the best row by using an active learning approach:
    1. Sample 20 random rows to build an initial model.
    2. Use the model to predict performance for all unknown rows.
    3. Use the remaining 30 queries to sample rows with high predicted utility.
    """
    goal_indices = [i for i, h in enumerate(header) if h.endswith('-') or h.endswith('+')]
    input_indices = [i for i in range(len(header)) if i not in goal_indices]
    
    # Helper to encode data
    encoders = {}
    for i in input_indices:
        if isinstance(rows[0][i], str):
            le = LabelEncoder()
            le.fit([r[i] for r in rows])
            encoders[i] = le

    def get_features(row):
        return [encoders[i].transform([row[i]])[0] if i in encoders else row[i] for i in input_indices]

    def get_score(goals):
        # Normalize goals: MRE- (min), ACC+ (max), PRED40+ (max)
        # We use a simple weighted sum of normalized values
        mre, acc, pred = goals
        # Since we don't have global min/max easily, we use relative values
        # MRE is minimized, others maximized.
        return -mre + acc + pred

    queried_indices = []
    
    # 1. Initial random sample
    initial_indices = random.sample(range(len(rows)), 20)
    for i in initial_indices:
        label(i)
        queried_indices.append(i)

    # 2. Iterative improvement
    for _ in range(30):
        # Train model on currently known data
        X = [get_features(rows[i]) for i in queried_indices]
        y = [get_score(label(i)) for i in queried_indices]
        
        model = RandomForestRegressor(n_estimators=50, random_state=42)
        model.fit(X, y)
        
        # Predict scores for all unqueried rows
        best_idx = -1
        best_pred = -float('inf')
        
        for i in range(len(rows)):
            if i not in queried_indices:
                pred = model.predict([get_features(rows[i])])[0]
                if pred > best_pred:
                    best_pred = pred
                    best_idx = i
        
        label(best_idx)
        queried_indices.append(best_idx)

    # Return the index of the best row found
    final_scores = {i: get_score(label(i)) for i in queried_indices}
    return max(final_scores, key=final_scores.get)

import csv

def fix(header, row, levers):
    """
    Finds the smallest change to a bad row's settings to improve performance
    without increasing complexity (assuming N_estimators is a proxy for speed).
    """
    # Identify indices
    goal_indices = [i for i, h in enumerate(header) if h.endswith('-') or h.endswith('+')]
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')
    n_est_idx = header.index('N_estimators')
    
    # Helper to evaluate if a row is "better"
    def get_score(r):
        # Simple heuristic: minimize MRE, maximize ACC and PRED
        # We use placeholders for unknown goals if necessary, but here we assume
        # we are comparing against the provided row's current state.
        return -r[mre_idx] + r[acc_idx] + r[pred_idx]

    best_change = {}
    
    # Try changing one lever at a time
    for col_name, allowed in levers.items():
        col_idx = header.index(col_name)
        current_val = row[col_idx]
        
        # For numeric, try a small step within range
        if isinstance(allowed, tuple):
            low, high = allowed
            # Try moving towards the middle or a random valid point
            # Here we try a small perturbation
            step = (high - low) * 0.1
            for direction in [-1, 1]:
                new_val = max(low, min(high, current_val + (direction * step)))
                if new_val != current_val:
                    best_change[col_name] = new_val
        
        # For symbolic, try a different value
        elif isinstance(allowed, list):
            for val in allowed:
                if val != current_val:
                    best_change[col_name] = val
                    break
                    
    # Constraint: Do not increase N_estimators (speed proxy)
    if 'N_estimators' in best_change:
        if best_change['N_estimators'] > row[n_est_idx]:
            del best_change['N_estimators']
            
    return best_change

import csv

def spot(header, rows):
    """
    Identifies rows that look like failed runs.
    A failed run often manifests as extreme outliers in input parameters
    or inconsistent behavior relative to the rest of the dataset.
    We use Z-score analysis on numeric inputs to find suspicious rows.
    """
    # Identify input indices (non-goal columns)
    goal_indices = [i for i, h in enumerate(header) if h.endswith('-') or h.endswith('+')]
    input_indices = [i for i in range(len(header)) if i not in goal_indices]
    
    # Calculate Z-scores for numeric inputs
    # A row is suspicious if it has extreme values (high Z-score)
    suspicion_scores = [0.0] * len(rows)
    
    for i in input_indices:
        # Only process numeric columns
        if isinstance(rows[0][i], (int, float)):
            col_vals = [r[i] for r in rows]
            mu = sum(col_vals) / len(col_vals)
            sigma = (sum((x - mu) ** 2 for x in col_vals) / len(col_vals)) ** 0.5
            
            if sigma > 0:
                for row_idx, row in enumerate(rows):
                    z = abs(row[i] - mu) / sigma
                    suspicion_scores[row_idx] += z
                    
    # Sort indices by suspicion score descending
    indexed_scores = list(enumerate(suspicion_scores))
    indexed_scores.sort(key=lambda x: x[1], reverse=True)
    
    return [idx for idx, score in indexed_scores if score > 0]

import csv
import random

def race(header, rows, label):
    """
    Compares three strategies: 'random', 'greedy', and 'hill_climbing'.
    Total budget: 50 labels.
    """
    goal_indices = [i for i, h in enumerate(header) if h.endswith('-') or h.endswith('+')]
    
    def get_score(row_idx):
        goals = label(row_idx)
        # Normalize: MRE- (min), ACC+ (max), PRED40+ (max)
        # Since we don't have global stats, we use a simple weighted sum
        # MRE is index 0, ACC index 1, PRED index 2
        return -goals[0] + goals[1] + goals[2]

    budget = 50
    queried = {}

    def get_val(i):
        if i not in queried:
            queried[i] = get_score(i)
        return queried[i]

    # Strategy 1: Random
    random_indices = random.sample(range(len(rows)), 16)
    for i in random_indices:
        get_val(i)
    
    # Strategy 2: Greedy (pick best from random sample)
    best_random = max(random_indices, key=get_val)

    # Strategy 3: Hill Climbing (start from best random, move to neighbors)
    current = best_random
    for _ in range(17):
        if len(queried) >= budget: break
        # Look at 5 random neighbors
        neighbors = random.sample(range(len(rows)), 5)
        best_neighbor = max(neighbors, key=lambda x: get_val(x) if x in queried else -float('inf'))
        if get_val(best_neighbor) > get_val(current):
            current = best_neighbor
        else:
            # Explore new
            next_idx = random.choice(range(len(rows)))
            if get_val(next_idx) > get_val(current):
                current = next_idx

    # Strategy 4: Simple Search (remaining budget)
    while len(queried) < budget:
        idx = random.choice(range(len(rows)))
        get_val(idx)

    # Determine winner
    results = {
        "random": best_random,
        "greedy": best_random,
        "hill_climbing": current
    }
    
    best_overall = max(results, key=lambda s: get_val(results[s]))
    
    return {
        "winner": best_overall,
        "row": results[best_overall],
        "tried": ["random", "greedy", "hill_climbing"]
    }

import csv
import math

def remember(header, rows):
    """
    Finds the 5% of rows most similar to the 'current' setup (the first row).
    Similarity is calculated using Euclidean distance on normalized input columns.
    """
    goal_indices = [i for i, h in enumerate(header) if h.endswith('-') or h.endswith('+')]
    input_indices = [i for i in range(len(header)) if i not in goal_indices]
    
    # Normalize inputs for distance calculation
    normalized_rows = []
    for r in rows:
        norm_row = []
        for i in input_indices:
            val = r[i]
            if isinstance(val, (int, float)):
                # Simple min-max normalization
                col_vals = [row[i] for row in rows if isinstance(row[i], (int, float))]
                mn, mx = min(col_vals), max(col_vals)
                norm_row.append((val - mn) / (mx - mn) if mx > mn else 0)
            else:
                # Symbolic: 1 if match, 0 if not
                norm_row.append(1.0 if val == rows[0][i] else 0.0)
        normalized_rows.append(norm_row)

    # Compare all rows to the first row (the team's current setup)
    target = normalized_rows[0]
    distances = []
    for i, row in enumerate(normalized_rows):
        dist = math.sqrt(sum((a - b) ** 2 for a, b in zip(target, row)))
        distances.append((dist, i))
    
    # Sort by distance (closest first)
    distances.sort(key=lambda x: x[0])
    
    # Return top 5% (800 rows * 0.05 = 40 rows)
    limit = max(1, int(len(rows) * 0.05))
    return [item[1] for item in distances[:limit]]
