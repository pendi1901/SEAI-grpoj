```python
import csv
from sklearn.ensemble import RandomForestRegressor
from sklearn.preprocessing import LabelEncoder

def guess(header, train, test):
    """
    Predicts the first goal column (MRE-) using a Random Forest Regressor.
    """
    goal_idx = 0
    goal_indices = [i for i, h in enumerate(header) if h.endswith('-') or h.endswith('+')]
    feature_indices = [i for i in range(len(header)) if i not in goal_indices]
    
    encoders = {}
    def encode_data(data):
        encoded = []
        for row in data:
            new_row = []
            for i in feature_indices:
                val = row[i]
                if isinstance(val, str):
                    if i not in encoders:
                        col_vals = [r[i] for r in train]
                        encoders[i] = LabelEncoder().fit(col_vals)
                    new_row.append(encoders[i].transform([val])[0])
                else:
                    new_row.append(val)
            encoded.append(new_row)
        return encoded

    X_train = encode_data(train)
    y_train = [row[goal_idx] for row in train]
    X_test = encode_data(test)
    
    # Train model
    model = RandomForestRegressor(n_estimators=100, random_state=42)
    model.fit(X_train, y_train)
    
    return model.predict(X_test).tolist()

if __name__ == "__main__":
    filename = 'Health-ClosedIssues0000.n1000.train.csv'
    rows = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = []
            for i, val in enumerate(row):
                try:
                    processed_row.append(float(val))
                except ValueError:
                    processed_row.append(val)
            rows.append(processed_row)

    n_est_idx = header.index('N_estimators')
    mre_idx = header.index('MRE-')

    # Baseline: Predict MRE- using original N_estimators
    train_data = rows
    test_data = [r[:] for r in rows]
    for row in test_data:
        for i, h in enumerate(header):
            if h.endswith('-') or h.endswith('+'):
                row[i] = "?"
    
    baseline_preds = guess(header, train_data, test_data)
    
    # Experiment: Halve N_estimators
    test_data_halved = [r[:] for r in rows]
    for row in test_data_halved:
        row[n_est_idx] = row[n_est_idx] / 2.0
        for i, h in enumerate(header):
            if h.endswith('-') or h.endswith('+'):
                row[i] = "?"
                
    halved_preds = guess(header, train_data, test_data_halved)
    
    # Calculate average impact
    diffs = [h - b for h, b in zip(halved_preds, baseline_preds)]
    avg_impact = sum(diffs) / len(diffs)
    
    print("Analysis: Impact of halving the forecaster size (N_estimators).")
    print(f"Average change in MRE-: {avg_impact:.4f}")
    print("If positive, the error (MRE) increases, meaning performance degrades.")
    exit(0)
```
