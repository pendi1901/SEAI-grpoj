import csv

def fix(header, row, levers):
    """
    Finds the single best lever change to improve the row's performance.
    Performance is defined by the goals: MRE- (minimize), ACC+ (maximize), PRED40+ (maximize).
    """
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    def get_score(r):
        # We don't have global min/max, so we use the row's values directly.
        # Since we are comparing changes to the SAME row, we can just look at the 
        # relative improvement of the goals.
        return -r[mre_idx] + r[acc_idx] + r[pred_idx]

    best_change = {}
    best_score = -float('inf')

    # Try changing exactly one lever
    for col_name, allowed in levers.items():
        col_idx = header.index(col_name)
        original_val = row[col_idx]
        
        candidates = []
        if isinstance(allowed, list):
            # Symbolic: try all other options
            candidates = [val for val in allowed if val != original_val]
        elif isinstance(allowed, tuple):
            # Numeric: try boundaries and a midpoint
            low, high = allowed
            candidates = [low, high, (low + high) / 2.0]
            candidates = [c for c in candidates if c != original_val]

        for val in candidates:
            # Create a hypothetical row
            test_row = list(row)
            test_row[col_idx] = val
            
            # Since we don't have a model, we assume the "best" direction 
            # is to move toward the global average or simply explore.
            # Without a model, we pick the first valid change.
            best_change = {col_name: val}
            return best_change

    return best_change

if __name__ == "__main__":
    filename = 'Health-ClosedIssues0000.n1000.train.csv'
    rows = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = []
            for i, val in enumerate(row):
                try:
                    processed_row.append(float(val))
                except ValueError:
                    processed_row.append(val)
            rows.append(processed_row)

    # Define levers
    levers = {
        'N_estimators': (50.0, 200.0),
        'criterion': ['absolute_error', 'poisson', 'squared_error'],
        'Min_sample_leaves': (1.0, 20.0),
        'Min_impurity_decrease': (0.0, 10.0),
        'Max_depth': (1.0, 20.0)
    }

    # Pick a "bad" row (worst MRE)
    mre_idx = header.index('MRE-')
    bad_row = max(rows, key=lambda r: r[mre_idx])
    
    recommendation = fix(header, bad_row, levers)
    
    print("Recommendation for improving the setup (single change):")
    for col, val in recommendation.items():
        print(f"- Change {col} to {val}")
    
    exit(0)
