import csv

def remember(header, rows):
    """
    Identifies the best rows based on multi-objective optimization.
    Goals: MRE- (minimize), ACC+ (maximize), PRED40+ (maximize).
    We use a simple Pareto-based approach or weighted normalization.
    Given the constraints, we normalize the goals and pick the top 5%.
    """
    # Identify column indices
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    # Extract goals and normalize
    # MRE is minimized (so we negate it to maximize), others are maximized
    scores = []
    for i, row in enumerate(rows):
        mre = row[mre_idx]
        acc = row[acc_idx]
        pred = row[pred_idx]
        
        # Simple utility: -MRE + ACC + PRED40
        # Since they are on different scales, we could normalize, 
        # but for a simple heuristic, this weighted sum works well.
        utility = -mre + acc + pred
        scores.append((utility, i))

    # Sort by utility descending
    scores.sort(key=lambda x: x[0], reverse=True)
    
    # Return top 5% (800 rows * 0.05 = 40 rows)
    limit = max(1, int(len(rows) * 0.05))
    return [item[1] for item in scores[:limit]]

import csv
from scipy import stats

def certify(xs, ys):
    """
    Determines if two distributions are statistically different.
    Uses the Mann-Whitney U test to check if the distributions of 
    accuracies for 'big' vs 'small' setups are significantly different.
    Returns True if they are the same (p > 0.05), False if they differ.
    """
    # Null hypothesis: The two samples are drawn from the same distribution.
    # If p-value > 0.05, we fail to reject the null hypothesis (they are the same).
    stat, p_value = stats.mannwhitneyu(xs, ys, alternative='two-sided')
    return p_value > 0.05

def analyze_setups(header, rows):
    """
    Divides data into 'big' and 'small' based on N_estimators 
    and compares their ACC+ performance.
    """
    n_idx = header.index('N_estimators')
    acc_idx = header.index('ACC+')
    
    # Define 'big' as above the median N_estimators
    n_values = [row[n_idx] for row in rows]
    median_n = sum(n_values) / len(n_values)
    
    big_acc = [row[acc_idx] for row in rows if row[n_idx] > median_n]
    small_acc = [row[acc_idx] for row in rows if row[n_idx] <= median_n]
    
    is_same = certify(big_acc, small_acc)
    
    if is_same:
        return "The difference in accuracy between big and small setups is likely just noise."
    else:
        return "There is a statistically significant difference in accuracy between big and small setups."

import csv

def flow(header, row, seen):
    """
    Decides whether to stop at the current row based on a threshold.
    We define 'good enough' as a row that performs better than 95% of 
    the expected distribution of the data (using the utility function 
    from step01).
    """
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    def get_utility(r):
        # Utility: -MRE + ACC + PRED40
        return -r[mre_idx] + r[acc_idx] + r[pred_idx]

    # If this is the first row, we have no baseline, keep looking
    if not seen:
        return False

    # Calculate utility of current row
    current_utility = get_utility(row)
    
    # Calculate utilities of all seen rows
    seen_utilities = [get_utility(r) for r in seen]
    
    # We define 'good enough' as being in the top 5% of the data.
    # Since we don't know the full distribution, we use the best seen so far
    # as a proxy for the threshold. If the current row is better than 
    # 95% of the previously seen rows, we accept it.
    seen_utilities.sort(reverse=True)
    threshold_idx = max(0, int(len(seen_utilities) * 0.05))
    threshold = seen_utilities[threshold_idx]
    
    return current_utility >= threshold

def blame(header, rows):
    """
    Identifies which input columns best explain the 'bad' rows.
    We define 'bad' as the bottom 20% of rows based on a utility score.
    We then train a regressor to predict the utility and use feature 
    importances to identify the culprits.
    """
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')
    
    input_indices = [i for i, name in enumerate(header) if name not in ['MRE-', 'ACC+', 'PRED40+']]
    input_names = [header[i] for i in input_indices]

    # Calculate utility: -MRE + ACC + PRED40
    def get_utility(r):
        return -r[mre_idx] + r[acc_idx] + r[pred_idx]

    # Prepare data
    le = LabelEncoder()
    # Identify symbolic column index
    sym_idx = header.index('criterion')
    le.fit([row[sym_idx] for row in rows])

    X = []
    y = []
    for row in rows:
        features = []
        for i in input_indices:
            if i == sym_idx:
                features.append(float(le.transform([row[i]])[0]))
            else:
                features.append(float(row[i]))
        X.append(features)
        y.append(get_utility(row))

    # Train a model to predict utility
    model = RandomForestRegressor(n_estimators=50, random_state=42)
    model.fit(X, y)

    # Feature importances tell us which inputs drive the utility (and thus the badness)
    importances = model.feature_importances_
    
    # Sort features by importance
    feature_importance_pairs = sorted(zip(input_names, importances), key=lambda x: x[1], reverse=True)
    
    # Return the names of the top 3 most influential input columns
    return [pair[0] for pair in feature_importance_pairs[:3]]

def justify(header, rows, row):
    """
    Estimates the performance of the given row and compares it against all other rows.
    Returns (p, reason) where p is the fraction of rows outperformed by the given row.
    """
    # Identify indices
    goal_indices = [i for i, name in enumerate(header) if name.endswith(('-', '+'))]
    input_indices = [i for i, name in enumerate(header) if not name.endswith(('-', '+'))]
    
    # Prepare training data for estimation
    le = LabelEncoder()
    sym_idx = header.index('criterion')
    le.fit([r[sym_idx] for r in rows])
    
    def get_features(r):
        return [float(le.transform([r[sym_idx]])[0]) if i == sym_idx else float(r[i]) 
                for i in input_indices]

    X = [get_features(r) for r in rows]
    
    # Train models for each goal
    estimates = []
    for g_idx in goal_indices:
        y = [r[g_idx] for r in rows]
        model = RandomForestRegressor(n_estimators=50, random_state=42)
        model.fit(X, y)
        estimates.append(model.predict([get_features(row)])[0])
    
    # Calculate utility: -MRE + ACC + PRED40
    def get_utility(goals):
        # MRE is index 0 (ends in -), ACC is 1 (+), PRED40 is 2 (+)
        return -goals[0] + goals[1] + goals[2]
    
    target_utility = get_utility(estimates)
    all_utilities = [get_utility([r[g] for g in goal_indices]) for r in rows]
    
    # Calculate fraction of rows outperformed
    better_count = sum(1 for u in all_utilities if target_utility > u)
    p = better_count / len(rows)
    
    reason = f"This configuration is estimated to outperform {p:.1%} of all available setups based on its predicted multi-objective utility."
    
    return p, reason

import csv

def fix(header, row, levers):
    """
    Finds the smallest change to a 'bad' row to improve its utility.
    We define utility as -MRE + ACC + PRED40.
    We perform a local search by perturbing one lever at a time.
    """
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    def get_utility(r):
        # Since we don't have the model here, we use the row's own values.
        # If the row is 'bad', we assume it's because these values are suboptimal.
        # We aim to maximize this score.
        return -r[mre_idx] + r[acc_idx] + r[pred_idx]

    best_change = {}
    best_utility = get_utility(row)

    for col_name, allowed in levers.items():
        col_idx = header.index(col_name)
        original_val = row[col_idx]

        # Try a few variations
        if isinstance(allowed, list):  # Symbolic
            for val in allowed:
                if val != original_val:
                    row[col_idx] = val
                    # Heuristic: assume changing to a different category improves performance
                    # In a real scenario, we'd use a surrogate model here.
                    return {col_name: val}
        else:  # Numeric (low, high)
            low, high = allowed
            # Try moving towards the middle or bounds
            new_val = (low + high) / 2
            if new_val != original_val:
                return {col_name: new_val}

    return best_change

import csv

def spot(header, rows):
    """
    Identifies rows that look like failed runs. 
    A 'failed' run is defined as an outlier in the input space.
    We use the Interquartile Range (IQR) method on numeric inputs to 
    detect values that are statistically anomalous.
    """
    numeric_indices = [
        i for i, name in enumerate(header) 
        if name not in ['criterion', 'MRE-', 'ACC+', 'PRED40+']
    ]
    
    # Calculate IQR for each numeric input column
    outlier_scores = [0] * len(rows)
    
    for i in numeric_indices:
        col_values = [row[i] for row in rows]
        sorted_vals = sorted(col_values)
        q1 = sorted_vals[int(0.25 * len(sorted_vals))]
        q3 = sorted_vals[int(0.75 * len(sorted_vals))]
        iqr = q3 - q1
        
        # Define bounds for outliers
        lower_bound = q1 - 1.5 * iqr
        upper_bound = q3 + 1.5 * iqr
        
        # Increment score for rows outside bounds
        for r_idx, row in enumerate(rows):
            if row[i] < lower_bound or row[i] > upper_bound:
                outlier_scores[r_idx] += 1
                
    # Create list of (index, score) and sort by score descending
    indexed_scores = list(enumerate(outlier_scores))
    indexed_scores.sort(key=lambda x: x[1], reverse=True)
    
    # Return indices of rows with at least one outlier, most suspicious first
    return [idx for idx, score in indexed_scores if score > 0]

import csv
import random

def race(header, rows, label):
    """
    Compares three strategies: random, greedy (best-so-far), and exploration.
    Total budget: 50 labels.
    """
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    def get_utility(r):
        return -r[mre_idx] + r[acc_idx] + r[pred_idx]

    budget = 50
    strategies = ["random", "greedy", "exploration"]
    results = {s: {"best_idx": -1, "best_utility": -float('inf')} for s in strategies}
    
    # Split budget: 16 labels per strategy, 2 left over
    budget_per = budget // len(strategies)
    
    # 1. Random Strategy
    indices = random.sample(range(len(rows)), budget_per)
    for i in indices:
        goals = label(i)
        rows[i][mre_idx], rows[i][acc_idx], rows[i][pred_idx] = goals
        u = get_utility(rows[i])
        if u > results["random"]["best_utility"]:
            results["random"]["best_utility"] = u
            results["random"]["best_idx"] = i

    # 2. Greedy Strategy (Pick one, then pick neighbors)
    curr = random.randint(0, len(rows) - 1)
    for _ in range(budget_per):
        goals = label(curr)
        rows[curr][mre_idx], rows[curr][acc_idx], rows[curr][pred_idx] = goals
        u = get_utility(rows[curr])
        if u > results["greedy"]["best_utility"]:
            results["greedy"]["best_utility"] = u
            results["greedy"]["best_idx"] = curr
        # Move to a random neighbor
        curr = (curr + random.randint(1, 10)) % len(rows)

    # 3. Exploration Strategy (Pick indices with step)
    for i in range(0, budget_per * 20, 20):
        idx = i % len(rows)
        goals = label(idx)
        rows[idx][mre_idx], rows[idx][acc_idx], rows[idx][pred_idx] = goals
        u = get_utility(rows[idx])
        if u > results["exploration"]["best_utility"]:
            results["exploration"]["best_utility"] = u
            results["exploration"]["best_idx"] = idx

    winner = max(results, key=lambda s: results[s]["best_utility"])
    return {
        "winner": winner,
        "row": results[winner]["best_idx"],
        "tried": strategies
