import csv

def remember(header, rows):
    """
    Identifies the best rows based on multi-objective optimization.
    Goals: MRE- (minimize), ACC+ (maximize), PRED40+ (maximize).
    We use a simple Pareto-based approach or weighted normalization.
    Given the constraints, we normalize the goals and pick the top 5%.
    """
    # Identify column indices
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    # Extract goals and normalize
    # MRE is minimized (so we negate it to maximize), others are maximized
    scores = []
    for i, row in enumerate(rows):
        mre = row[mre_idx]
        acc = row[acc_idx]
        pred = row[pred_idx]
        
        # Simple utility: -MRE + ACC + PRED40
        # Since they are on different scales, we could normalize, 
        # but for a simple heuristic, this weighted sum works well.
        utility = -mre + acc + pred
        scores.append((utility, i))

    # Sort by utility descending
    scores.sort(key=lambda x: x[0], reverse=True)
    
    # Return top 5% (800 rows * 0.05 = 40 rows)
    limit = max(1, int(len(rows) * 0.05))
    return [item[1] for item in scores[:limit]]

import csv
from sklearn.ensemble import RandomForestRegressor
from sklearn.preprocessing import LabelEncoder
import numpy as np

def guess(header, train, test):
    """
    Estimates the first goal column (MRE-) for test rows using a 
    RandomForestRegressor. Handles '?' in test rows by ignoring goal columns.
    """
    goal_idx = 0
    
    # Identify input indices (all columns except goals)
    # Goals are MRE-, ACC+, PRED40+ (indices 0, 1, 2)
    input_indices = [i for i in range(len(header)) if i not in [0, 1, 2]]
    
    # Prepare data
    all_data = train + test
    le = LabelEncoder()
    # 'criterion' is at index 4
    criterion_col = [row[4] for row in all_data]
    le.fit(criterion_col)
    
    def transform_row(row):
        features = []
        for i in input_indices:
            if i == 4:  # criterion
                features.append(float(le.transform([row[i]])[0]))
            else:
                features.append(float(row[i]))
        return features

    X_train = [transform_row(row) for row in train]
    y_train = [row[goal_idx] for row in train]
    X_test = [transform_row(row) for row in test]
    
    model = RandomForestRegressor(n_estimators=100, random_state=42)
    model.fit(X_train, y_train)
    
    return model.predict(X_test).tolist()
