import csv
from sklearn.ensemble import RandomForestRegressor
from sklearn.preprocessing import LabelEncoder

def blame(header, rows):
    """
    Identifies which input columns have the least impact on performance.
    Settings that have low feature importance in predicting the utility 
    can be left at factory defaults without hurting performance.
    """
    # Identify indices
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')
    
    input_indices = [i for i, name in enumerate(header) if name not in ['MRE-', 'ACC+', 'PRED40+']]
    input_names = [header[i] for i in input_indices]
    sym_idx = header.index('criterion')

    # Utility: -MRE + ACC + PRED40
    def get_utility(r):
        return -r[mre_idx] + r[acc_idx] + r[pred_idx]

    # Prepare data
    le = LabelEncoder()
    le.fit([row[sym_idx] for row in rows])

    X = []
    y = []
    for row in rows:
        features = []
        for i in input_indices:
            if i == sym_idx:
                features.append(float(le.transform([row[i]])[0]))
            else:
                features.append(float(row[i]))
        X.append(features)
        y.append(get_utility(row))

    # Train model to find feature importance
    model = RandomForestRegressor(n_estimators=50, random_state=42)
    model.fit(X, y)

    # Sort features by importance (ascending)
    # The least important features are the ones that can be left at defaults
    importances = model.feature_importances_
    feature_importance_pairs = sorted(zip(input_names, importances), key=lambda x: x[1])
    
    # Return the names of the input columns that have the least impact
    return [pair[0] for pair in feature_importance_pairs]

if __name__ == "__main__":
    filename = 'Health-ClosedIssues0000.n1000.train.csv'
    rows = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = []
            for i, val in enumerate(row):
                if header[i] == 'criterion':
                    processed_row.append(val)
                else:
                    processed_row.append(float(val))
            rows.append(processed_row)
    
    safe_settings = blame(header, rows)
    
    print("Customer Inquiry: Which settings can I leave at factory default?")
    print("Answer: The following settings have the least impact on performance and are safe to leave at defaults:")
    for i, name in enumerate(safe_settings):
        print(f"{i+1}. {name}")
