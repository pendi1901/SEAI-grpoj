import pandas as pd


def remember(header, rows):
    """Return positions (0-based) of rows whose forecasts are unusually trustworthy."""
    # Goals to minimise: MRE-
    # Goals to maximise: ACC+, PRED40+
    goals = ["MRE-", "ACC+", "PRED40+"]

    best_scores = {}
    for g in goals:
        vals = []
        for r in rows:
            v = float(r[header.index(g)])
            if g.endswith("-"):
                # lower is better -> score = value (smaller = closer to 0)
                vals.append(v)
            else:
                # higher is better -> score = -value (so larger original => smaller score)
                vals.append(-v)

        best_scores[g] = min(vals)

    scored_rows = []
    for i, r in enumerate(rows):
        s = 0.0
        for g in goals:
            v = float(r[header.index(g)])
            if g.endswith("-"):
                s += (v - best_scores[g])
            else:
                s += (-v - best_scores[g])
        scored_rows.append((i, s))

    scored_rows.sort(key=lambda x: x[1])
    n = len(rows)
    top_k = max(1, int(n * 0.05))
    return [pos for pos, _ in scored_rows[:top_k]]

import pandas as pd


def certify(xs, ys):
    """Return True if xs and ys are the same apart from noise."""
    n = min(len(xs), len(ys))
    diffs = [xs[i] - ys[i] for i in range(n)]
    mean_diff = sum(diffs) / n
    std_diff = (sum((d - mean_diff) ** 2 for d in diffs) / max(n - 1, 1)) ** 0.5
    if std_diff < 1e-6:
        return True
    z = abs(mean_diff) / std_diff
    return z < 2.0

import pandas as pd
from sklearn.tree import DecisionTreeRegressor


def flow(header, row, seen):
    goals = ["MRE-", "ACC+", "PRED40+"]
    goal_cols = [c for c in header if c in goals]
    input_cols = [c for c in header if c not in goals]

    # Predict current row's MRE- using a model trained on seen rows
    X_train = []
    y_train = []
    best_mre_so_far = None
    for s_row in seen:
        x = [float(s_row[header.index(c)]) for c in input_cols]
        y = float(s_row[header.index(goal_cols[0])])  # MRE-
        X_train.append(x)
        y_train.append(y)
        if best_mre_so_far is None or y < best_mre_so_far:
            best_mre_so_far = y

    model = DecisionTreeRegressor(max_depth=5, random_state=42)
    model.fit(X_train, y_train)

    x_new = [float(row[header.index(c)]) for c in input_cols]
    pred = float(model.predict([x_new])[0])

    # First row: can't decide yet
    if not seen:
        return False

    # Stop when prediction is significantly better than any seen so far
    if best_mre_so_far is None:
        return False

    improvement_ratio = (best_mre_so_far - pred) / max(best_mre_so_far, 1e-6)
    return improvement_ratio >= 0.5

import pandas as pd


def blame(header, rows):
    """Return names of input columns that best explain why the worst rows are bad."""
    goals = ["MRE-", "ACC+", "PRED40+"]
    goal_cols = [c for c in header if c in goals]
    input_cols = [c for c in header if c not in goals and c != "?"]

    # Score each row: lower is better (for -), higher is better (for +)
    scored = []
    bests = {}
    for g in goals:
        vals = []
        for r in rows:
            v = float(r[header.index(g)])
            if g.endswith("-"):
                vals.append(v)
            else:
                vals.append(-v)
        bests[g] = min(vals)

    row_scores = []
    for i, r in enumerate(rows):
        s = 0.0
        for g in goals:
            v = float(r[header.index(g)])
            if g.endswith("-"):
                s += (v - bests[g])
            else:
                s += (-v - bests[g])
        row_scores.append((i, s))

    # Worst rows are those with highest score (most negative performance)
    row_scores.sort(key=lambda x: x[1], reverse=True)
    worst = [pos for pos, _ in row_scores[:max(1, len(rows) // 5)]]

    # For each input column, compute average value across worst rows vs all rows
    col_impact = {}
    for c in input_cols:
        worst_vals = []
        all_vals = []
        for r in rows:
            v = float(r[header.index(c)])
            if header.index(c) < len(header):
                all_vals.append(v)
                if any(pos == i for pos, _ in row_scores[:max(1, len(rows) // 5)]):
                    worst_vals.append(v)

        mean_worst = sum(worst_vals) / max(len(worst_vals), 1)
        mean_all = sum(all_vals) / max(len(all_vals), 1)
        col_impact[c] = (mean_worst, mean_all)

    # Rank columns by absolute difference in means
    ranked = sorted(col_impact.items(), key=lambda x: abs(x[1][0] - x[1][1]), reverse=True)
    return [c for c, _ in ranked[:5]]

def justify(header, rows, row):
    """Return (p, reason) where p is the fraction of `rows` that my predicted setup beats."""
    goals = ["MRE-", "ACC+", "PRED40+"]
    goal_cols = [c for c in header if c in goals]
    input_cols = [c for c in header if c not in goals and c != "?"]

    # Precompute best scores from the reference rows (same convention as step01.py)
    bests = {}
    for g in goals:
        vals = []
        for r in rows:
            v = float(r[header.index(g)])
            if g.endswith("-"):
                vals.append(v)
            else:
                vals.append(-v)
        bests[g] = min(vals)

    # Train a separate regressor per goal column on the reference data
    models = {}
    for g in goals:
        X_g, y_g = [], []
        for r in rows:
            x = [float(r[header.index(c)]) for c in input_cols]
            y = float(r[header.index(g)])
            X_g.append(x)
            y_g.append(y)
        model = DecisionTreeRegressor(max_depth=5, random_state=42)
        model.fit(X_g, y_g)
        models[g] = model

    # Predict the hidden goal cells of `row` from its input features
    x_new = [float(row[header.index(c)]) for c in input_cols]
    predicted_goals = {}
    for g in goals:
        predicted_goals[g] = float(models[g].predict([x_new])[0])

    # Score a set of goal values against the precomputed bests (lower is better)
    def score(goal_vals):
        s = 0.0
        for g in goals:
            v = goal_vals[g]
            if g.endswith("-"):
                s += (v - bests[g])
            else:
                s += (-v - bests[g])
        return s

    my_score = score(predicted_goals)

    # Score every reference row and count how many are worse than mine
    scores = [score({g: float(r[header.index(g)]) for g in goals}) for r in rows]
    n_worse = sum(1 for s in scores if s > my_score)
    p = n_worse / len(rows) if len(rows) else 0.0

    reason = f"Better than {p:.0%} of the other setups."
    return (p, reason)

def choose(header, rows, label):
    """Try up to 50 different row positions and return the best one."""
    goals = ["MRE-", "ACC+", "PRED40+"]
    
    # Precompute best scores from step01 convention (lower is better)

    for g in goals:
        vals = []
        for r in rows:
            v = float(r[header.index(g)])
            if g.endswith("-"):
                vals.append(v)
            else:
                vals.append(-v)
        bests[g] = min(vals)

    def score_row(goal_vals):
        s = 0.0
        for g in goals:
            v = goal_vals[g]
            if g.endswith("-"):
                s += (v - bests[g])
            else:
                s += (-v - bests[g])
        return s

    # Probe a diverse set of rows: first few, then evenly spaced
    n_rows = len(rows)
    probe_indices = list(range(min(50, n_rows)))  # start with first 50
    
    if n_rows > 100:
        # Add more spread-out probes beyond the first 50
        step = max(2, n_rows // 50)
        for i in range(step, n_rows, step):
            probe_indices.append(i)
    
    best_pos = None
    best_score = float('inf')
    
    for idx in probe_indices:
        goals_dict = label(idx)
        s = score_row(goals_dict)
        if s < best_score:
            best_score = s
            best_pos = idx
    
    return best_pos

import pandas as pd


def spot(header, rows):
    """Return positions (0-based) of rows whose input values look wrong."""
    goals = ["MRE-", "ACC+", "PRED40+"]
    goal_cols = [c for c in header if c in goals]
    input_cols = [c for c in header if c not in goals and c != "?"]

    # Precompute best scores from the full dataset (step01 convention)
    bests = {}
    for g in goals:
        vals = []
        for r in rows:
            v = float(r[header.index(g)])
            if g.endswith("-"):
                vals.append(v)
            else:
                vals.append(-v)
        bests[g] = min(vals)

    # Score each row (lower is better)
    def score_row(row):
        s = 0.0
        for g in goals:
            v = float(row[header.index(g)])
            if g.endswith("-"):
                s += (v - bests[g])
            else:
                s += (-v - bests[g])
        return s

    # Score each row and find the worst ones
    scored = [(i, score_row(r)) for i, r in enumerate(rows)]
    scored.sort(key=lambda x: x[1], reverse=True)

    # Now check which rows have suspicious input values
    # A row is suspicious if its inputs are outliers compared to other rows with similar goals
    suspicious = []

    for i, s in scored:
        row = rows[i]
        # Check each input column for consistency
        for c in input_cols:
            col_idx = header.index(c)
            v = float(row[col_idx])

            # Get other rows with similar goal scores (within 10% of this row's score)
            similar_rows = [j for j, sj in scored if abs(sj - s) < abs(s) * 0.1 + 0.5]

            if len(similar_rows) >= 3:
                # Compute mean and std of this column among similar rows
                col_vals = []
                for j in similar_rows:
                    col_vals.append(float(rows[j][col_idx]))
                mean_val = sum(col_vals) / len(col_vals)
                std_val = (sum((x - mean_val) ** 2 for x in col_vals) / max(len(col_vals) - 1, 1)) ** 0.5

                # If this row's value is far from the group mean (more than 2 std), it's suspicious
                if std_val > 0 and abs(v - mean_val) > 2 * std_val:
                    suspicious.append(i)
                    break  # Only need one suspicious column per row

    return sorted(suspicious, reverse=True)
