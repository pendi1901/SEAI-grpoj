```python
import csv
from sklearn.ensemble import RandomForestRegressor
from sklearn.preprocessing import LabelEncoder
import numpy as np

def guess(header, train, test):
    """
    Estimates the first goal column (MRE-) for test rows using a 
    RandomForestRegressor. Handles '?' in test rows by ignoring goal columns.
    """
    goal_idx = 0
    
    # Identify input indices (all columns except goals)
    # Goals are MRE-, ACC+, PRED40+ (indices 0, 1, 2)
    input_indices = [i for i in range(len(header)) if i not in [0, 1, 2]]
    
    # Prepare data
    all_data = train + test
    le = LabelEncoder()
    # 'criterion' is at index 4
    criterion_col = [row[4] for row in all_data]
    le.fit(criterion_col)
    
    def transform_row(row):
        features = []
        for i in input_indices:
            if i == 4:  # criterion
                features.append(float(le.transform([row[i]])[0]))
            else:
                features.append(float(row[i]))
        return features

    X_train = [transform_row(row) for row in train]
    y_train = [row[goal_idx] for row in train]
    X_test = [transform_row(row) for row in test]
    
    model = RandomForestRegressor(n_estimators=100, random_state=42)
    model.fit(X_train, y_train)
    
    return model.predict(X_test).tolist()

if __name__ == "__main__":
    filename = 'Health-ClosedIssues0000.n1000.train.csv'
    rows = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = []
            for i, val in enumerate(row):
                # Convert numeric columns to floats, keep symbolic as strings
                if header[i] == 'criterion':
                    processed_row.append(val)
                else:
                    processed_row.append(float(val))
            rows.append(processed_row)
    
    # Split into train and a dummy test set (last 10 rows)
    train_data = rows[:-10]
    test_data = [r[:] for r in rows[-10:]]
    
    # Mask the goals in test data
    for r in test_data:
        r[0] = "?"
        r[1] = "?"
        r[2] = "?"
        
    estimates = guess(header, train_data, test_data)
    
    print("Forecast estimation complete.")
    print(f"Estimated MRE- for the last 10 configurations:")
    for i, est in enumerate(estimates):
        print(f"Row {i+1}: {est:.4f}")
```
