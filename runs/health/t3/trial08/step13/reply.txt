import csv
from scipy import stats

def certify(xs, ys):
    """
    Returns True if the difference between two distributions is likely noise,
    False if they are statistically significantly different.
    Uses the Mann-Whitney U test as it does not assume normal distribution.
    """
    # A p-value threshold of 0.05 is standard for statistical significance.
    # If p > 0.05, we fail to reject the null hypothesis (they are the same).
    stat, p_value = stats.mannwhitneyu(xs, ys, alternative='two-sided')
    return p_value > 0.05

def get_score(header, row):
    """Calculates a normalized aggregate score for a row."""
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')
    
    # Normalize: (1-MRE) + ACC + PRED
    mre = float(row[mre_idx])
    acc = float(row[acc_idx])
    pred = float(row[pred_idx])
    return (1.0 - mre) + acc + pred

if __name__ == "__main__":
    filename = 'Health-ClosedIssues0000.n1000.train.csv'
    try:
        with open(filename, 'r') as f:
            reader = csv.reader(f)
            header = next(reader)
            rows = []
            for row in reader:
                processed_row = []
                for val in row:
                    try:
                        processed_row.append(float(val))
                    except ValueError:
                        processed_row.append(val)
                rows.append(processed_row)

        # The customer asks about the "two built-in rules".
        # Based on the context of the previous steps, we compare 
        # the performance of 'absolute_error' vs 'squared_error' 
        # as the two primary criterion options in the dataset.
        
        crit_idx = header.index('criterion')
        
        abs_err_scores = [get_score(header, r) for r in rows if r[crit_idx] == 'absolute_error']
        sq_err_scores = [get_score(header, r) for r in rows if r[crit_idx] == 'squared_error']
        
        is_noise = certify(abs_err_scores, sq_err_scores)
        
        print("Comparing 'absolute_error' vs 'squared_error' criteria:")
        if is_noise:
            print("The difference in performance is statistically insignificant.")
            print("Conclusion: You can stop arguing; the choice makes no real difference.")
        else:
            print("The difference in performance is statistically significant.")
            print("Conclusion: The choice of criterion does matter.")
            
    except FileNotFoundError:
        print(f"Error: {filename} not found.")
        exit(1)
