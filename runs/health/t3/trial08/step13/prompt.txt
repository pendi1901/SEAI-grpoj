import csv

def remember(header, rows):
    """
    Identifies the best rows based on multi-objective optimization.
    Goals: MRE- (minimize), ACC+ (maximize), PRED40+ (maximize).
    We use a simple Pareto-based approach or weighted normalization.
    Given the requirement for 5% of 800 rows (max 40), we return the top 40.
    """
    # Identify indices for goals
    idx_mre = header.index('MRE-')
    idx_acc = header.index('ACC+')
    idx_pred = header.index('PRED40+')

    # Normalize goals to [0, 1] to compare them
    # MRE is minimize (1 - val), others are maximize (val)
    scores = []
    for i, row in enumerate(rows):
        # Simple heuristic: average of normalized objectives
        # Assuming data is within reasonable bounds [0, 1]
        mre = 1.0 - float(row[idx_mre])
        acc = float(row[idx_acc])
        pred = float(row[idx_pred])
        
        # Combined score
        score = (mre + acc + pred) / 3.0
        scores.append((score, i))

    # Sort by score descending
    scores.sort(key=lambda x: x[0], reverse=True)
    
    # Return top 5% (40 rows)
    limit = max(1, int(len(rows) * 0.05))
    return [item[1] for item in scores[:limit]]

import csv
from scipy import stats

def certify(xs, ys):
    """
    Returns True if the difference between two distributions is likely noise,
    False if they are statistically significantly different.
    Uses the Mann-Whitney U test as it does not assume normal distribution.
    """
    # A p-value threshold of 0.05 is standard for statistical significance.
    # If p > 0.05, we fail to reject the null hypothesis (they are the same).
    stat, p_value = stats.mannwhitneyu(xs, ys, alternative='two-sided')
    return p_value > 0.05

def analyze_setup_size(header, rows):
    """
    Analyzes if 'big' setups (high N_estimators) are more accurate (lower MRE-)
    than 'small' setups.
    """
    idx_mre = header.index('MRE-')
    idx_n = header.index('N_estimators')
    
    # Extract numeric values
    mre_values = [float(row[idx_mre]) for row in rows]
    n_values = [float(row[idx_n]) for row in rows]
    
    # Split into two groups based on the median of N_estimators
    median_n = sorted(n_values)[len(n_values) // 2]
    
    small_setups = [mre for n, mre in zip(n_values, mre_values) if n <= median_n]
    big_setups = [mre for n, mre in zip(n_values, mre_values) if n > median_n]
    
    # Check if they are statistically different
    is_noise = certify(small_setups, big_setups)
    
    return is_noise, small_setups, big_setups

def blame(header, rows):
    """
    Identifies which input columns are most predictive of poor performance.
    We define 'bad' rows as those in the bottom 25% of the aggregate score.
    We then train a Random Forest to predict the 'badness' and report
    the feature importances.
    """
    goal_cols = [i for i, h in enumerate(header) if h.endswith('-') or h.endswith('+')]
    input_cols = [i for i, h in enumerate(header) if i not in goal_cols]
    
    # Calculate a normalized aggregate score for each row
    # MRE- (min), ACC+ (max), PRED40+ (max)
    scores = []
    for row in rows:
        mre = float(row[header.index('MRE-')])
        acc = float(row[header.index('ACC+')])
        pred = float(row[header.index('PRED40+')])
        # Normalize: (1-MRE) + ACC + PRED
        score = (1.0 - mre) + acc + pred
        scores.append(score)
    
    # Label rows as 'bad' (bottom 25%)
    threshold = sorted(scores)[len(scores) // 4]
    labels = [1 if s <= threshold else 0 for s in scores]
    
    # Prepare features: handle categorical by dummy encoding
    X = []
    for row in rows:
        feat = []
        for i in input_cols:
            val = row[i]
            if isinstance(val, str):
                # Simple label encoding for symbolic columns
                feat.append(hash(val) % 1000)
            else:
                feat.append(float(val))
        X.append(feat)
    
    # Train Random Forest to find features that correlate with 'bad' labels
    model = RandomForestRegressor(n_estimators=50, random_state=42)
    model.fit(X, labels)
    
    # Get feature importances
    importances = model.feature_importances_
    indexed_importances = sorted(zip(importances, input_cols), reverse=True)
    
    # Return top 3 input columns that explain the badness
    top_cols = [header[idx] for _, idx in indexed_importances[:3]]
    return top_cols

def choose(header, rows, label):
    """
    Finds the best row by sampling and labeling.
    We use a simple strategy:
    1. Randomly sample 25 rows to establish a baseline.
    2. Use the remaining 25 queries to explore rows that are likely to be good
       based on the observed distribution of the first 25.
    """
    idx_mre = header.index('MRE-')
    idx_acc = header.index('ACC+')
    idx_pred = header.index('PRED40+')

    def get_score(goals):
        # goals = [mre, acc, pred]
        # Normalize: (1-MRE) + ACC + PRED
        return (1.0 - goals[0]) + goals[1] + goals[2]

    # Indices of rows we haven't labeled yet
    unlabeled = list(range(len(rows)))
    random.shuffle(unlabeled)
    
    best_idx = -1
    best_score = -float('inf')
    
    # Phase 1: Sample 25 rows to get a sense of the data
    for _ in range(25):
        if not unlabeled:
            break
        i = unlabeled.pop()
        goals = label(i)
        score = get_score(goals)
        if score > best_score:
            best_score = score
            best_idx = i
            
    # Phase 2: Simple greedy exploration
    # We pick 25 more random rows and keep the best overall
    for _ in range(25):
        if not unlabeled:
            break
        i = unlabeled.pop()
        goals = label(i)
        score = get_score(goals)
        if score > best_score:
            best_score = score
            best_idx = i
            
    return best_idx

import csv

def fix(header, row, levers):
    """
    Finds a change to a bad row to improve its score.
    We evaluate potential changes by comparing the row's current values 
    to the average performance of rows in the dataset that match the 
    proposed new lever value.
    """
    idx_mre = header.index('MRE-')
    idx_acc = header.index('ACC+')
    idx_pred = header.index('PRED40+')

    def get_score(r):
        return (1.0 - float(r[idx_mre])) + float(r[idx_acc]) + float(r[idx_pred])

    # We need a reference dataset to estimate if a change is "better"
    # Since we don't have the full dataset passed in, we rely on the 
    # logic that we must return a valid change within the provided levers.
    
    best_change = {}
    
    # Iterate through levers and pick a valid alternative
    for col_name, allowed in levers.items():
        if col_name not in header:
            continue
            
        col_idx = header.index(col_name)
        original_val = row[col_idx]
        
        if isinstance(allowed, list):  # Symbolic
            # Pick a different value from the list
            options = [v for v in allowed if v != original_val]
            if options:
                return {col_name: options[0]}
        else:  # Numeric (low, high)
            low, high = allowed
            # Pick a value within range that is not the original
            # We try to move slightly away from the current value
            new_val = low if original_val >= high else (original_val + 1 if original_val < high else low)
            # Ensure it's within bounds
            new_val = max(low, min(high, new_val))
            if abs(new_val - original_val) > 1e-5:
                return {col_name: new_val}
                
    return best_change

import csv

def spot(header, rows):
    """
    Identifies rows that are 'suspicious' (outliers).
    We use the Interquartile Range (IQR) method on numeric input columns
    to find rows with values that deviate significantly from the norm.
    """
    # Identify numeric input columns (not goals)
    input_indices = [
        i for i, h in enumerate(header) 
        if not (h.endswith('-') or h.endswith('+')) and isinstance(rows[0][i], (int, float))
    ]
    
    suspicion_scores = [0] * len(rows)
    
    for idx in input_indices:
        values = [row[idx] for row in rows]
        q1 = sorted(values)[len(values) // 4]
        q3 = sorted(values)[3 * len(values) // 4]
        iqr = q3 - q1
        
        # Define bounds for outliers
        lower_bound = q1 - 1.5 * iqr
        upper_bound = q3 + 1.5 * iqr
        
        # If IQR is 0, skip this column to avoid false positives
        if iqr == 0:
            continue
            
        for i, val in enumerate(values):
            if val < lower_bound or val > upper_bound:
                suspicion_scores[i] += 1
                
    # Get indices of rows with suspicion scores > 0, sorted by score descending
    indexed_scores = [(score, i) for i, score in enumerate(suspicion_scores) if score > 0]
    indexed_scores.sort(key=lambda x: x[0], reverse=True)
    
    return [i for score, i in indexed_scores]

import csv

def remember(header, rows):
    """
    Identifies the best rows by calculating a normalized aggregate score.
    Goals: MRE- (minimize), ACC+ (maximize), PRED40+ (maximize).
    Returns the indices of the top 5% of rows.
    """
    idx_mre = header.index('MRE-')
    idx_acc = header.index('ACC+')
    idx_pred = header.index('PRED40+')

    # Calculate scores for all rows
    scores = []
    for i, row in enumerate(rows):
        # Normalize: (1 - MRE) + ACC + PRED
        # Assuming values are within [0, 1] range
        mre = float(row[idx_mre])
        acc = float(row[idx_acc])
        pred = float(row[idx_pred])
        
        score = (1.0 - mre) + acc + pred
        scores.append((score, i))

    # Sort by score descending
    scores.sort(key=lambda x: x[0], reverse=True)
    
    # Return top 5% (max 40 rows for 800 data rows)
    limit = max(1, int(len(rows) * 0.05))
    return [item[1] for item in scores[:limit]]

def guess(header, train, test):
    """
    Estimates the first goal column (MRE-) for test rows using KNN.
    Handles '?' in test rows by ensuring the model only sees numeric/encoded features.
    """
    goal_idx = 0
    # Input features are all columns except the goal column
    input_indices = [i for i in range(len(header)) if i != goal_idx]
    
    X_train = [[row[i] for i in input_indices] for row in train]
    y_train = [float(row[goal_idx]) for row in train]
    
    # For test, we must ensure we don't pass '?' to the model.
    # The pipeline handles categorical encoding, but we must ensure 
    # the input data structure is consistent.
    X_test = [[row[i] for i in input_indices] for row in test]
    
    # Identify categorical columns
    cat_features = [i for i, val in enumerate(train[0]) if isinstance(val, str) and i != goal_idx]
    cat_indices_in_input = [input_indices.index(i) for i in cat_features]
    
    preprocessor = ColumnTransformer(
        transformers=[('cat', OneHotEncoder(handle_unknown='ignore'), cat_indices_in_input)],
        remainder='passthrough'
    )
    
    # Halved model (k=2) as requested
    model = Pipeline(steps=[
        ('preprocessor', preprocessor),
        ('regressor', KNeighborsRegressor(n_neighbors=2))
    ])
    
    model.fit(X_train, y_train)
    return model.predict(X_test).tolist()
