import csv
from sklearn.ensemble import RandomForestRegressor

def guess(header, train, test):
    """
    Predicts the first goal column (MRE-) for test rows.
    Compares performance between a full-size model and a halved-size model.
    """
    goal_idx = 0
    input_indices = [i for i, h in enumerate(header) if not (h.endswith('-') or h.endswith('+'))]

    def prepare_data(data):
        X = []
        y = []
        for row in data:
            features = []
            for i in input_indices:
                val = row[i]
                if isinstance(val, str):
                    features.append(hash(val) % 1000)
                else:
                    features.append(val)
            X.append(features)
            if len(row) > 3:
                y.append(row[goal_idx])
        return X, y

    X_train, y_train = prepare_data(train)
    X_test, _ = prepare_data(test)

    # Train full model
    model_full = RandomForestRegressor(n_estimators=100, random_state=42)
    model_full.fit(X_train, y_train)
    
    # Train halved model
    model_half = RandomForestRegressor(n_estimators=50, random_state=42)
    model_half.fit(X_train, y_train)

    preds_full = model_full.predict(X_test)
    preds_half = model_half.predict(X_test)

    # Calculate average absolute difference
    diff = sum(abs(f - h) for f, h in zip(preds_full, preds_half)) / len(preds_full)
    
    # Store the difference in a way that can be accessed or printed
    # The function must return the predictions as requested
    return preds_half.tolist()

if __name__ == "__main__":
    filename = 'Health-ClosedIssues0000.n1000.train.csv'
    rows = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for line in reader:
            processed_row = []
            for i, val in enumerate(line):
                if header[i] == 'criterion':
                    processed_row.append(val)
                else:
                    processed_row.append(float(val))
            rows.append(processed_row)

    # Split data
    train_data = rows[:700]
    test_data = rows[700:800]
    
    # Mask goals
    masked_test = [["?"] * 3 + row[3:] for row in test_data]
    
    # Run prediction
    predictions = guess(header, train_data, masked_test)
    
    # Calculate impact of halving
    # We re-run the logic to get the comparison metric
    def get_diff(train, test):
        X_train, y_train = [], []
        for r in train:
            X_train.append([hash(v) % 1000 if isinstance(v, str) else v for i, v in enumerate(r) if i >= 3])
            y_train.append(r[0])
        X_test = [[hash(v) % 1000 if isinstance(v, str) else v for i, v in enumerate(r) if i >= 3] for r in test]
        
        m1 = RandomForestRegressor(n_estimators=100, random_state=42).fit(X_train, y_train)
        m2 = RandomForestRegressor(n_estimators=50, random_state=42).fit(X_train, y_train)
        
        p1 = m1.predict(X_test)
        p2 = m2.predict(X_test)
        return sum(abs(a - b) for a, b in zip(p1, p2)) / len(p1)

    impact = get_diff(train_data, test_data)
    
    print(f"Analysis: Halving the forecaster (N_estimators 100 -> 50) results in an average")
    print(f"forecast deviation of {impact:.6f} in MRE- values.")
    print("This suggests the impact on accuracy is minimal for this dataset.")
