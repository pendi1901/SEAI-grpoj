import csv

def remember(header, rows):
    """
    Identifies rows that are 'unusually trustworthy' by finding the Pareto frontier
    or high-performing candidates based on the goals.
    Goals: MRE- (minimize), ACC+ (maximize), PRED40+ (maximize).
    """
    # Identify column indices
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    # Normalize scores to a 0-1 range to compare them
    # Since we want low MRE, high ACC, high PRED, we define a utility score
    # Utility = (1 - MRE_norm) + ACC_norm + PRED_norm
    
    mre_vals = [r[mre_idx] for r in rows]
    acc_vals = [r[acc_idx] for r in rows]
    pred_vals = [r[pred_idx] for r in rows]
    
    min_mre, max_mre = min(mre_vals), max(mre_vals)
    min_acc, max_acc = min(acc_vals), max(acc_vals)
    min_pred, max_pred = min(pred_vals), max(pred_vals)
    
    def norm(val, mi, ma):
        return (val - mi) / (ma - mi) if ma > mi else 0

    scores = []
    for i, row in enumerate(rows):
        s = (1 - norm(row[mre_idx], min_mre, max_mre)) + \
            norm(row[acc_idx], min_acc, max_acc) + \
            norm(row[pred_idx], min_pred, max_pred)
        scores.append((s, i))
    
    # Sort by score descending
    scores.sort(key=lambda x: x[0], reverse=True)
    
    # Return top 5% (800 rows * 0.05 = 40)
    limit = max(1, int(len(rows) * 0.05))
    return [item[1] for item in scores[:limit]]

import csv
from scipy import stats

def certify(xs, ys):
    """
    Returns True if the difference between two distributions is statistically 
    insignificant (noise), False if they are significantly different.
    Uses Welch's t-test to compare the means of two independent samples.
    """
    # If samples are too small, assume noise (True)
    if len(xs) < 2 or len(ys) < 2:
        return True
    
    # Welch's t-test (equal_var=False)
    t_stat, p_value = stats.ttest_ind(xs, ys, equal_var=False)
    
    # If p-value > 0.05, we fail to reject the null hypothesis (they are the same)
    return p_value > 0.05

import csv

def flow(header, row, seen):
    """
    Decides if the current row is 'good enough' to stop searching.
    We define 'good enough' as being in the top 5% of the performance 
    distribution observed so far, based on a normalized utility score.
    """
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')

    def get_score(r):
        # Utility = (1 - MRE) + ACC + PRED
        # Since we don't have global min/max, we use the current row's 
        # values relative to the 'seen' history.
        mre = r[mre_idx]
        acc = r[acc_idx]
        pred = r[pred_idx]
        return (1.0 - mre) + acc + pred

    if not seen:
        return False

    current_score = get_score(row)
    seen_scores = [get_score(r) for r in seen]
    
    # Threshold: Is this row better than 95% of what we have seen?
    # We require at least 20 rows to have a meaningful distribution.
    if len(seen) < 20:
        return False
    
    seen_scores.sort(reverse=True)
    # Top 5% index
    threshold_idx = max(0, int(len(seen_scores) * 0.05))
    threshold_score = seen_scores[threshold_idx]
    
    return current_score >= threshold_score

import csv

def blame(header, rows):
    """
    Identifies input columns that correlate most strongly with poor performance.
    Performance is defined by a utility score: (1 - MRE) + ACC + PRED.
    We compare the distribution of input values in the worst 20% of rows
    against the rest to find which inputs are 'damaging'.
    """
    mre_idx = header.index('MRE-')
    acc_idx = header.index('ACC+')
    pred_idx = header.index('PRED40+')
    
    # Calculate utility for each row
    def get_utility(r):
        return (1.0 - r[mre_idx]) + r[acc_idx] + r[pred_idx]

    scored_rows = []
    for r in rows:
        scored_rows.append((get_utility(r), r))
    
    # Sort by utility ascending (worst first)
    scored_rows.sort(key=lambda x: x[0])
    
    # Define worst 20%
    n_worst = max(1, len(rows) // 5)
    worst_rows = [r for _, r in scored_rows[:n_worst]]
    
    # Identify input columns (exclude goals)
    input_indices = [i for i, h in enumerate(header) if not (h.endswith('-') or h.endswith('+'))]
    
    # For each input, calculate how much the mean in the worst set deviates 
    # from the global mean (normalized by standard deviation)
    impacts = []
    for i in input_indices:
        col_vals = [r[i] for r in rows if isinstance(r[i], (int, float))]
        if not col_vals:
            continue
            
        global_mean = sum(col_vals) / len(col_vals)
        worst_vals = [r[i] for r in worst_rows if isinstance(r[i], (int, float))]
        if not worst_vals:
            continue
            
        worst_mean = sum(worst_vals) / len(worst_vals)
        
        # The 'blame' score is the absolute difference in means
        impacts.append((abs(worst_mean - global_mean), header[i]))
    
    # Sort by impact descending
    impacts.sort(key=lambda x: x[0], reverse=True)
    
    return [name for _, name in impacts[:3]]

def justify(header, rows, row):
    """
    Estimates the performance of the given row and compares it to the dataset.
    Returns (p, reason) where p is the fraction of rows outperformed by the input.
    """
    # Identify indices
    goal_indices = [i for i, h in enumerate(header) if h.endswith('-') or h.endswith('+')]
    input_indices = [i for i, h in enumerate(header) if not (h.endswith('-') or h.endswith('+'))]
    
    # Prepare training data for the goals
    X_train = [[r[i] for i in input_indices] for r in rows]
    # Convert symbolic inputs to numeric
    for i, idx in enumerate(input_indices):
        if isinstance(rows[0][idx], str):
            mapping = {val: j for j, val in enumerate(set(r[idx] for r in rows))}
            for r_idx in range(len(X_train)):
                X_train[r_idx][i] = mapping[rows[r_idx][idx]]
            # Update row input
            row[idx] = mapping.get(row[idx], 0)

    # Train models for each goal
    predictions = []
    for g_idx in goal_indices:
        y_train = [r[g_idx] for r in rows]
        model = RandomForestRegressor(n_estimators=50, random_state=42)
        model.fit(X_train, y_train)
        predictions.append(model.predict([[row[i] for i in input_indices]])[0])
    
    # Utility function: minimize '-' goals, maximize '+' goals
    def get_utility(goals):
        u = 0.0
        for i, g_idx in enumerate(goal_indices):
            col_name = header[g_idx]
            val = goals[i]
            # Normalize to 0-1 range based on column min/max
            col_vals = [r[g_idx] for r in rows]
            mi, ma = min(col_vals), max(col_vals)
            norm = (val - mi) / (ma - mi) if ma > mi else 0
            if col_name.endswith('-'):
                u += (1.0 - norm)
            else:
                u += norm
        return u / len(goal_indices)

    # Compare against all rows
    target_utility = get_utility(predictions)
    better_count = 0
    for r in rows:
        row_utility = get_utility([r[i] for i in goal_indices])
        if target_utility > row_utility:
            better_count += 1
            
    p = better_count / len(rows)
    reason = f"This configuration is estimated to outperform {p:.1%} of all possible setups based on its predicted performance metrics."
    return (p, reason)

import csv

def get_utility(header, row):
    """Calculates a normalized utility score for a row."""
    goals = []
    for i, h in enumerate(header):
        if h.endswith('-') or h.endswith('+'):
            goals.append((h, row[i]))
    
    # Since we don't have global min/max, we use a heuristic utility
    # MRE- (minimize), ACC+ (maximize), PRED40+ (maximize)
    # We map to 0-1 based on typical ranges observed in the data
    mre = row[0]
    acc = row[1]
    pred = row[2]
    
    # Utility = (1 - MRE) + ACC + PRED
    return (1.0 - min(1.0, max(0.0, mre))) + min(1.0, max(0.0, acc)) + min(1.0, max(0.0, pred))

    """
    Finds the smallest change to settings that improves utility.
    We perform a local search in the lever space.
    """
    best_change = {}

    for col_name, allowed in levers.items():

        candidates = []
        if isinstance(allowed, list):  # Symbolic
            candidates = [v for v in allowed if v != original_val]
        else:  # Numeric (low, high)
            low, high = allowed
            candidates = [low, high, (low + high) / 2]
            
        for val in candidates:
            new_row = list(row)
            new_row[idx] = val
            
            # Simple heuristic: if we don't have a model, we assume

    for col_name, allowed in levers.items():

            return {col_name: allowed[0]}
        else:
            return {col_name: allowed[0]}
            
    return best_change

import csv

def spot(header, rows):
    """
    Identifies rows that are 'suspicious' by detecting outliers in the input space.
    Rows with input values that are extreme (e.g., far from the mean/median) 
    relative to the rest of the dataset are flagged as potential errors.
    """
    input_indices = [i for i, h in enumerate(header) if not (h.endswith('-') or h.endswith('+'))]
    
    # Calculate statistics for each input column to find outliers
    # We use Z-score approach: (val - mean) / std_dev
    suspicion_scores = [0.0] * len(rows)
    
    for idx in input_indices:
        col_vals = [r[idx] for r in rows if isinstance(r[idx], (int, float))]
        if not col_vals:
            continue
            
        mean = sum(col_vals) / len(col_vals)
        variance = sum((x - mean) ** 2 for x in col_vals) / len(col_vals)
        std_dev = variance ** 0.5
        
        if std_dev == 0:
            continue
            
        for i, row in enumerate(rows):
            val = row[idx]
            if isinstance(val, (int, float)):
                z_score = abs(val - mean) / std_dev
                # If Z-score is very high, it's an outlier
                if z_score > 3:
                    suspicion_scores[i] += z_score
                    
    # Create list of (index, score)
    indexed_scores = list(enumerate(suspicion_scores))
    # Filter out rows with 0 suspicion
    suspicious = [item for item in indexed_scores if item[1] > 0]
    # Sort by score descending
    suspicious.sort(key=lambda x: x[1], reverse=True)
    
    return [item[0] for item in suspicious]

import csv

def get_utility(header, goals):
    """Calculates a normalized utility score for a set of goals."""
    u = 0.0
    goal_indices = [i for i, h in enumerate(header) if h.endswith('-') or h.endswith('+')]
    for i, g_idx in enumerate(goal_indices):
        col_name = header[g_idx]
        val = goals[i]
        # Since we don't have global min/max, we assume normalized 0-1 range
        # based on the problem context.
        if col_name.endswith('-'):
            u += (1.0 - min(1.0, max(0.0, val)))
        else:
            u += min(1.0, max(0.0, val))
    return u / len(goal_indices)

def race(header, rows, label):
    """
    Compares three strategies: random, greedy (first improvement), and 
    exploration (sampling then picking best).
    """
    budget = 50
    tried_indices = set()
    
    def get_labeled_utility(idx):
        if idx not in tried_indices:
            if len(tried_indices) >= budget:
                return -1.0
            tried_indices.add(idx)
        return get_utility(header, label(idx))

    # Strategy 1: Random
    random_indices = random.sample(range(len(rows)), 16)
    best_random_idx = -1
    best_random_u = -1.0
    for idx in random_indices:
        u = get_labeled_utility(idx)
        if u > best_random_u:
            best_random_u = u
            best_random_idx = idx

    # Strategy 2: Greedy (Hill climbing)
    greedy_idx = random.randint(0, len(rows) - 1)
    best_greedy_u = get_labeled_utility(greedy_idx)
    for _ in range(16):
        neighbor = random.randint(0, len(rows) - 1)
        u = get_labeled_utility(neighbor)
        if u > best_greedy_u:
            best_greedy_u = u
            greedy_idx = neighbor
    
    # Strategy 3: Exploration (Sample 18, pick best)
    explore_indices = random.sample(range(len(rows)), 18)
    best_explore_idx = -1
    best_explore_u = -1.0
    for idx in explore_indices:
        u = get_labeled_utility(idx)
        if u > best_explore_u:
            best_explore_u = u
            best_explore_idx = idx

    results = {
        "random": (best_random_u, best_random_idx),
        "greedy": (best_greedy_u, greedy_idx),
        "exploration": (best_explore_u, best_explore_idx)
    }
    
    winner = max(results, key=lambda k: results[k][0])
    
    return {
        "winner": winner,
        "row": results[winner][1],
        "tried": ["random", "greedy", "exploration"]
    }

    mre = row[mre_idx]
    acc = row[acc_idx]
    pred = row[pred_idx]

def remember(header, rows):
    """
    Finds the 5% of rows that are most similar to the 'best' performing row
    found in the dataset, based on input configuration similarity.
    """
    # 1. Find the best row
    utilities = [(get_utility(header, r), i) for i, r in enumerate(rows)]
    best_idx = max(utilities, key=lambda x: x[0])[1]
    best_row = rows[best_idx]
    
    # 2. Define input indices
    input_indices = [i for i, h in enumerate(header) if not (h.endswith('-') or h.endswith('+'))]
    
    # 3. Calculate similarity (Euclidean distance in input space)
    def similarity(r1, r2):
        dist = 0
        for i in input_indices:
            v1, v2 = r1[i], r2[i]
            if isinstance(v1, (int, float)):
                dist += (v1 - v2) ** 2
            else:
                dist += 0 if v1 == v2 else 1
        return dist

    # 4. Rank rows by similarity to the best row
    similarities = []
    for i, row in enumerate(rows):
        if i == best_idx: continue
        similarities.append((similarity(best_row, row), i))
    
    similarities.sort(key=lambda x: x[0])
    
    # 5. Return top 5%
    limit = max(1, int(len(rows) * 0.05))
    return [idx for _, idx in similarities[:limit]]

def certify(xs, ys):
    """
    Returns True if the difference between two distributions is statistically 
    insignificant (noise), False if they are significantly different.
    Uses Welch's t-test to compare the means of two independent samples.
    """
    # If samples are too small, assume noise (True)
    if len(xs) < 2 or len(ys) < 2:
        return True
    
    # Welch's t-test (equal_var=False)
    t_stat, p_value = stats.ttest_ind(xs, ys, equal_var=False)
    
    # If p-value > 0.05, we fail to reject the null hypothesis (they are the same)
    return p_value > 0.05

def blame(header, rows):
    """
    Identifies input columns that have the least impact on performance.
    If a column's distribution in the 'worst' rows is statistically similar 
    to the global distribution, it can be left at factory defaults.
    """
    input_indices = [i for i, h in enumerate(header) if not (h.endswith('-') or h.endswith('+'))]
    
    # Calculate utility and sort to find the worst 20%
    scored_rows = sorted([(get_utility(header, r), r) for r in rows], key=lambda x: x[0])
    n_worst = max(1, len(rows) // 5)
    worst_rows = [r for _, r in scored_rows[:n_worst]]
    
    # For each input, calculate the difference in mean/distribution
    # between the worst rows and the global set.
    # Low impact means the setting doesn't correlate with being "bad".
    impacts = []
    for i in input_indices:
        col_vals = [r[i] for r in rows if isinstance(r[i], (int, float))]
        if not col_vals: continue
        
        global_mean = sum(col_vals) / len(col_vals)
        worst_vals = [r[i] for r in worst_rows if isinstance(r[i], (int, float))]
        if not worst_vals: continue
        
        worst_mean = sum(worst_vals) / len(worst_vals)
        
        # Impact is the normalized deviation
        std_dev = (sum((x - global_mean)**2 for x in col_vals) / len(col_vals))**0.5
        score = abs(worst_mean - global_mean) / (std_dev if std_dev > 0 else 1)
        impacts.append((score, header[i]))
    
    # Sort by impact ascending: lowest impact means safest to leave at default
    impacts.sort(key=lambda x: x[0])
    
    return [name for _, name in impacts]

import csv

def spot(header, rows):
    """
    Identifies rows that contain input values far outside the observed range 
    (outliers) for each numeric input column. Returns the indices of these 
    rows, sorted by the total 'suspicion' score (sum of Z-scores).
    """
    # Identify input columns (those not ending in - or +)
    input_indices = [i for i, h in enumerate(header) if not (h.endswith('-') or h.endswith('+'))]
    
    # Store suspicion scores for each row
    suspicion_scores = [0.0] * len(rows)
    
    for idx in input_indices:
        # Extract numeric values for this column
        col_vals = [r[idx] for r in rows if isinstance(r[idx], (int, float))]
        if not col_vals:
            continue
            
        # Calculate mean and standard deviation
        n = len(col_vals)
        mean = sum(col_vals) / n
        variance = sum((x - mean) ** 2 for x in col_vals) / n
        std_dev = variance ** 0.5
        
        # If std_dev is 0, all values are the same; no outliers possible
        if std_dev == 0:
            continue
            
        # Flag values that are more than 3 standard deviations from the mean
        for i, row in enumerate(rows):
            val = row[idx]
            if isinstance(val, (int, float)):
                z_score = abs(val - mean) / std_dev
                if z_score > 3:
                    suspicion_scores[i] += z_score
                    
    # Create list of (index, score) for rows with non-zero suspicion
    indexed_scores = [(i, s) for i, s in enumerate(suspicion_scores) if s > 0]
    
    # Sort by score descending (most suspicious first)
    indexed_scores.sort(key=lambda x: x[1], reverse=True)
    
    return [item[0] for item in indexed_scores]
