import csv

def remember(header, rows):
    """
    Identifies the best rows based on multi-objective optimization.
    Goals: Minimize columns ending in '-', Maximize columns ending in '+'.
    Uses a simple Pareto-dominance approach to find the 'best' setups.
    """
    goals = []
    for i, name in enumerate(header):
        if name.endswith('-'):
            goals.append((i, -1))  # Minimize
        elif name.endswith('+'):
            goals.append((i, 1))   # Maximize

    def get_score(row):
        # Normalize values to compare different scales
        # Since we want to find 'unusually trustworthy', we look for 
        # rows that perform well across all goal columns.
        scores = []
        for idx, direction in goals:
            val = row[idx]
            # Simple heuristic: higher is better for +, lower for -
            scores.append(val * direction)
        return sum(scores)

    # Calculate scores and keep track of original indices
    indexed_rows = []
    for i, row in enumerate(rows):
        indexed_rows.append((i, get_score(row)))

    # Sort by score descending
    indexed_rows.sort(key=lambda x: x[1], reverse=True)

    # Return top 5% (800 rows * 0.05 = 40 rows)
    limit = max(1, int(len(rows) * 0.05))
    return [item[0] for item in indexed_rows[:limit]]

    Estimates the first goal column (MRE-) for test rows using a

def certify(xs, ys):
    """
    Returns True if the correlation between two sets of numbers is high,
    suggesting the relationship is real rather than noise.
    """
    if len(xs) < 2 or len(ys) < 2:
        return False
    corr, _ = pearsonr(xs, ys)
    # If correlation is strong (e.g., > 0.5), we consider the trend real
    return abs(corr) > 0.5

import csv

def flow(header, row, seen):
    """
    Decides if the current row is 'good enough' to stop searching.
    We define 'good enough' as being in the top 5% of the performance 
    distribution based on the goals observed so far.
    """
    goals = []
    for i, name in enumerate(header):
        if name.endswith('-'):
            goals.append((i, -1))
        elif name.endswith('+'):
            goals.append((i, 1))

    def get_score(r):
        score = 0
        for idx, direction in goals:
            score += r[idx] * direction
        return score

    # Calculate score for current row
    current_score = get_score(row)

    # If this is the first row, we can't compare yet
    if not seen:
        return False

    # Compare against all previously seen rows
    # We stop if the current row is better than 95% of what we've seen
    seen_scores = [get_score(r) for r in seen]
    better_than_count = sum(1 for s in seen_scores if current_score > s)
    
    # Threshold: if better than 95% of history, or if we have enough history
    # to be confident (e.g., at least 20 rows seen)
    if len(seen) >= 20:
        if better_than_count / len(seen) >= 0.95:
            return True
            
    return False

def blame(header, rows):
    """
    Identifies which input columns contribute most to poor performance.
    We define 'bad' as the bottom 20% of rows based on a weighted goal score.
    We then train a classifier to predict the 'badness' and extract feature importances.
    """
    df = pd.DataFrame(rows, columns=header)
    
    goal_cols = [c for c in header if c.endswith('-') or c.endswith('+')]
    input_cols = [c for c in header if c not in goal_cols]
    
    # Calculate a normalized score for each row to identify 'bad' ones
    # Minimize '-' (invert) and Maximize '+'
    scores = []
    for _, row in df.iterrows():
        s = 0
        for col in goal_cols:
            val = row[col]
            if col.endswith('-'):
                s -= val
            else:
                s += val
        scores.append(s)
    
    df['score'] = scores
    # Threshold for 'bad' (bottom 20%)
    threshold = df['score'].quantile(0.2)
    df['is_bad'] = (df['score'] <= threshold).astype(int)
    
    # Prepare features
    X = df[input_cols].copy()
    for col in input_cols:
        if X[col].dtype == 'object':
            le = LabelEncoder()
            X[col] = le.fit_transform(X[col].astype(str))
        else:
            # Fill numeric NaNs with mean, ensuring we don't touch object columns
            X[col] = X[col].fillna(X[col].mean())
            
    y = df['is_bad']
    
    # Use Random Forest to find which inputs correlate with being 'bad'
    clf = RandomForestClassifier(n_estimators=50, random_state=42)
    clf.fit(X, y)
    
    importances = sorted(zip(input_cols, clf.feature_importances_), key=lambda x: x[1], reverse=True)
    
    # Return the top 3 inputs that explain the bad performance
    return [name for name, score in importances[:3]]

import csv

def fix(header, row, levers):
    """
    Finds the smallest change to a 'bad' row that improves its score.
    We evaluate potential changes by comparing them against the average
    performance of the dataset.
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith('-') or name.endswith('+')]
    
    def get_score(r):
        score = 0
        for i, name in enumerate(header):
            if name.endswith('-'):
                score -= r[i]
            elif name.endswith('+'):
                score += r[i]
        return score

    best_change = {}
    best_score = get_score(row)
    
    # Try changing one lever at a time
    for col_name, allowed in levers.items():
        if col_name not in header:
            continue
        idx = header.index(col_name)
        original_val = row[idx]
        
        candidates = []
        if isinstance(allowed, (list, tuple)) and not isinstance(allowed[0], (int, float)) and not isinstance(allowed, tuple):
            # Symbolic
            for val in allowed:
                if val != original_val:
                    candidates.append(val)
        else:
            # Numeric (low, high)
            low, high = allowed
            # Try boundaries and midpoint
            for val in [low, high, (low + high) / 2]:
                if val != original_val:
                    candidates.append(val)
        
        for val in candidates:
            new_row = list(row)
            new_row[idx] = val
            
            # Simple heuristic: if the new configuration score is better, accept it
            current_val = get_score(new_row)
            if current_val > best_score:
                best_score = current_val
                best_change = {col_name: val}
                
    return best_change

import csv

def spot(header, rows):
    """
    Identifies suspicious rows (outliers) based on the assumption that 
    failed runs often exhibit extreme input values or inconsistent 
    parameter combinations. We use a Z-score approach on numeric inputs 
    to find rows that deviate significantly from the norm.
    """
    # Identify input columns (those not ending in - or +)
    input_indices = [
        i for i, name in enumerate(header) 
        if not name.endswith('-') and not name.endswith('+')
    ]
    
    # Only consider numeric input columns for Z-score calculation
    numeric_input_indices = [
        i for i in input_indices if isinstance(rows[0][i], (int, float))
    ]
    
    # Calculate mean and std for each numeric input column
    stats = {}
    for i in numeric_input_indices:
        vals = [row[i] for row in rows]
        mean = sum(vals) / len(vals)
        std = (sum((x - mean) ** 2 for x in vals) / len(vals)) ** 0.5
        stats[i] = (mean, std if std > 0 else 1.0)
        
    # Score rows based on the sum of absolute Z-scores of their inputs
    # Higher score means more suspicious
    row_scores = []
    for idx, row in enumerate(rows):
        score = 0
        for i in numeric_input_indices:
            mean, std = stats[i]
            z_score = abs((row[i] - mean) / std)
            score += z_score
        row_scores.append((idx, score))
        
    # Sort by score descending (most suspicious first)
    row_scores.sort(key=lambda x: x[1], reverse=True)
    
    return [item[0] for item in row_scores]

import csv
import random

def race(header, rows, label):
    """
    Compares three strategies: random, greedy (first-improvement), 
    and extreme (sampling from the tails of the input distribution).
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith('-') or name.endswith('+')]
    directions = [-1 if header[i].endswith('-') else 1 for i in goal_indices]

    def get_score(goals):
        return sum(val * direction for val, direction in zip(goals, directions))

    budget = 50
    tried_indices = set()
    results = {}

    def get_label(i):
        if i not in tried_indices:
            if len(tried_indices) >= budget:
                raise ValueError("Budget exceeded")
            tried_indices.add(i)
        return label(i)

    # Strategy 1: Random
    def strategy_random():
        best_idx, best_score = -1, float('-inf')
        for _ in range(16):
            idx = random.randint(0, len(rows) - 1)
            score = get_score(get_label(idx))
            if score > best_score:
                best_score, best_idx = score, idx
        return best_idx, best_score

    # Strategy 2: Greedy (Hill Climbing)
    def strategy_greedy():
        curr = random.randint(0, len(rows) - 1)
        best_idx, best_score = curr, get_score(get_label(curr))
        for _ in range(15):
            neighbor = random.randint(0, len(rows) - 1)
            score = get_score(get_label(neighbor))
            if score > best_score:
                best_score, best_idx = score, neighbor
        return best_idx, best_score

    # Strategy 3: Extreme (Focus on high-variance inputs)
    def strategy_extreme():
        best_idx, best_score = -1, float('-inf')
        # Sample from the first 10 and last 10 rows (assuming sorted or structured data)
        indices = list(range(0, 10)) + list(range(len(rows) - 10, len(rows)))

import csv
import math

def remember(header, rows):
    """
    Identifies the 5% of rows most similar to the 'team's current setup' 
    (defined as the first row in the dataset) that also perform well.
    """
    # Identify goal indices and directions
    goal_indices = [i for i, name in enumerate(header) if name.endswith('-') or name.endswith('+')]
    input_indices = [i for i, name in enumerate(header) if i not in goal_indices]
    
    # The team's current setup is the first row
    current_setup = rows[0]
    
    def get_score(row):
        score = 0
        for i, name in enumerate(header):
            if name.endswith('-'):
                score -= row[i]
            elif name.endswith('+'):
                score += row[i]
        return score

    def get_similarity(row):
        # Euclidean distance on numeric inputs
        dist = 0
        for i in input_indices:
            val1 = row[i]
            val2 = current_setup[i]
            if isinstance(val1, (int, float)) and isinstance(val2, (int, float)):
                dist += (val1 - val2) ** 2
            else:
                dist += 1 if val1 != val2 else 0
        return math.sqrt(dist)

    # Score rows based on a balance of performance and similarity to current setup
    # We want rows that are similar (low distance) and perform well (high score)
    scored_rows = []
    for i, row in enumerate(rows):
        perf = get_score(row)
        sim = get_similarity(row)
        # Combine: higher performance is good, lower distance is good
        # We use -sim to make it a maximization problem
        combined = perf - sim
        scored_rows.append((i, combined))
    
    # Sort by combined score descending
    scored_rows.sort(key=lambda x: x[1], reverse=True)
    
    # Return top 5%
    limit = max(1, int(len(rows) * 0.05))
    return [item[0] for item in scored_rows[:limit]]

import csv

def certify(xs, ys):
    """
    Returns True if the two sets of numbers are statistically indistinguishable
    (i.e., the difference is likely just noise), False if they differ significantly.
    Uses a t-test for independent samples.
    """
    if len(xs) < 2 or len(ys) < 2:
        return True
    
    # Perform a two-sample t-test
    t_stat, p_value = stats.ttest_ind(xs, ys, equal_var=False)
    
    # If p_value > 0.05, we fail to reject the null hypothesis, 
    # meaning there is no statistically significant difference.
    return p_value > 0.05

import csv

def flow(header, row, seen):
    """
    Decides whether to stop searching based on whether the current row
    is significantly better than the best seen so far.
    We stop if the current row is the best we have seen in a long time 
    (plateau detection).
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith('-') or name.endswith('+')]
    
    def get_score(r):
        score = 0
        for i in goal_indices:
            val = r[i]
            if header[i].endswith('-'):
                score -= val
            else:
                score += val
        return score

    if not seen:
        return False

    # Calculate scores for all seen rows
    seen_scores = [get_score(r) for r in seen]
    best_seen = max(seen_scores)
    current_score = get_score(row)

    # If we have seen enough data (e.g., 50 rows), check for improvement.
    # If the current row is not better than the best seen in the last 50 rows,
    # we assume we have reached a plateau and stop.
    if len(seen) >= 50:
        recent_scores = seen_scores[-50:]
        if current_score <= max(recent_scores):
            return True
            
    return False
