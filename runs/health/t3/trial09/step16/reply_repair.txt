```python
import csv
import pandas as pd
import numpy as np
from sklearn.ensemble import RandomForestRegressor
from sklearn.preprocessing import LabelEncoder

def justify(header, rows, row):
    """
    Estimates the performance of a 'big/deep' setup and compares it to the dataset.
    Returns (fraction_better, reason_string).
    """
    df = pd.DataFrame(rows, columns=header)
    goal_cols = [c for c in header if c.endswith('-') or c.endswith('+')]
    input_cols = [c for c in header if c not in goal_cols]

    # Prepare training data and encode symbolic columns
    df_encoded = df.copy()
    encoders = {}
    for col in input_cols:
        if df_encoded[col].dtype == 'object':
            le = LabelEncoder()
            df_encoded[col] = le.fit_transform(df_encoded[col].astype(str))
            encoders[col] = le
        else:
            df_encoded[col] = df_encoded[col].astype(float)

    X_train = df_encoded[input_cols]
    
    # Prepare target row
    row_df = pd.DataFrame([row], columns=header)
    X_test = row_df[input_cols].copy()
    for col in input_cols:
        if col in encoders:
            X_test[col] = encoders[col].transform(X_test[col].astype(str))
        else:
            X_test[col] = X_test[col].astype(float)

    # Estimate goals
    estimated_goals = {}
    for goal in goal_cols:
        y_train = df_encoded[goal]
        model = RandomForestRegressor(n_estimators=100, random_state=42)
        model.fit(X_train, y_train)
        estimated_goals[goal] = model.predict(X_test)[0]

    # Calculate score for target row
    def get_score(r_goals):
        s = 0
        for col in goal_cols:
            val = r_goals[col]
            s += val * (-1 if col.endswith('-') else 1)
        return s

    row_score = get_score(estimated_goals)

    # Calculate scores for all historical rows
    all_scores = []
    for r in rows:
        r_goals = {col: r[header.index(col)] for col in goal_cols}
        all_scores.append(get_score(r_goals))
    
    better_count = sum(1 for s in all_scores if row_score > s)
    fraction = better_count / len(rows)
    
    reason = "Using the biggest, deepest setup often leads to overfitting or diminishing returns, making it perform worse than more balanced configurations."
    return (float(fraction), reason)

if __name__ == "__main__":
    filename = "Health-ClosedIssues0000.n1000.train.csv"
    rows = []
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = []
            for val in row:
                try:
                    processed_row.append(float(val))
                except ValueError:
                    processed_row.append(val)
            rows.append(processed_row)

    # Create a "big/deep" test row: Max N_estimators, Max depth
    test_row = [0.0] * len(header)
    for i, name in enumerate(header):
        if name == 'N_estimators': test_row[i] = 200.0
        elif name == 'Max_depth': test_row[i] = 20.0
        elif name == 'Min_sample_leaves': test_row[i] = 1.0
        elif name == 'Min_impurity_decrease': test_row[i] = 0.0
        elif name == 'criterion': test_row[i] = 'squared_error'
        elif name.endswith('-') or name.endswith('+'): test_row[i] = "?"

    p, reason = justify(header, rows, test_row)
    print(f"Performance Score: {p:.4f}")
    print(f"Justification: {reason}")
```
