import csv

def flow(header, row, seen):
    """
    Decides whether to stop searching based on whether the current row
    is significantly better than the best seen so far.
    We stop if the current row is the best we have seen in a long time 
    (plateau detection).
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith('-') or name.endswith('+')]
    
    def get_score(r):
        score = 0
        for i in goal_indices:
            val = r[i]
            if header[i].endswith('-'):
                score -= val
            else:
                score += val
        return score

    if not seen:
        return False

    # Calculate scores for all seen rows
    seen_scores = [get_score(r) for r in seen]
    best_seen = max(seen_scores)
    current_score = get_score(row)

    # If we have seen enough data (e.g., 50 rows), check for improvement.
    # If the current row is not better than the best seen in the last 50 rows,
    # we assume we have reached a plateau and stop.
    if len(seen) >= 50:
        recent_scores = seen_scores[-50:]
        if current_score <= max(recent_scores):
            return True
            
    return False

if __name__ == "__main__":
    filename = "Health-ClosedIssues0000.n1000.train.csv"
    rows = []
    
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = []
            for val in row:
                try:
                    processed_row.append(float(val))
                except ValueError:
                    processed_row.append(val)
            rows.append(processed_row)

    seen = []
    stopped_at = -1
    for i, row in enumerate(rows):
        if flow(header, row, seen):
            stopped_at = i + 1
            break
        seen.append(row)

    if stopped_at != -1:
        print(f"Optimization stopped at row {stopped_at} because no significant improvement was detected.")
    else:
        print("Processed all rows: No plateau detected.")
