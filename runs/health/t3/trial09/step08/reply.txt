import csv
import random

def fix(header, row, levers):
    """
    Finds the smallest change to a 'bad' row that improves its score
    without increasing complexity (N_estimators).
    """
    goal_indices = [i for i, name in enumerate(header) if name.endswith('-') or name.endswith('+')]
    
    def get_score(r):
        score = 0
        for i, name in enumerate(header):
            if name.endswith('-'):
                score -= r[i]
            elif name.endswith('+'):
                score += r[i]
        return score

    # Current score (assuming row is fully populated for calculation)
    best_score = get_score(row)
    best_change = {}

    # Try changing one lever at a time
    for col_name, allowed in levers.items():
        idx = header.index(col_name)
        original_val = row[idx]
        
        # Generate candidates
        candidates = []
        if isinstance(allowed, tuple):  # Numeric (low, high)
            low, high = allowed
            # Try a few steps within range
            step = (high - low) / 5
            for i in range(1, 6):
                val = low + i * step
                if val != original_val:
                    candidates.append(val)
        else:  # Symbolic
            for val in allowed:
                if val != original_val:
                    candidates.append(val)
        
        for val in candidates:
            new_row = list(row)
            new_row[idx] = val
            
            # Heuristic: If we improve score, consider it
            # We assume the model performance improves if score increases
            if get_score(new_row) > best_score:
                best_score = get_score(new_row)
                best_change = {col_name: val}
                
    return best_change
