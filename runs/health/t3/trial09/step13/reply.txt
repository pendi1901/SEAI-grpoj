import csv
from scipy import stats

def certify(xs, ys):
    """
    Returns True if the two sets of numbers are statistically indistinguishable
    (i.e., the difference is likely just noise), False if they differ significantly.
    Uses a t-test for independent samples.
    """
    if len(xs) < 2 or len(ys) < 2:
        return True
    
    # Perform a two-sample t-test
    t_stat, p_value = stats.ttest_ind(xs, ys, equal_var=False)
    
    # If p_value > 0.05, we fail to reject the null hypothesis, 
    # meaning there is no statistically significant difference.
    return p_value > 0.05

if __name__ == "__main__":
    filename = "Health-ClosedIssues0000.n1000.train.csv"
    rows = []
    header = []
    
    with open(filename, 'r') as f:
        reader = csv.reader(f)
        header = next(reader)
        for row in reader:
            processed_row = []
            for val in row:
                try:
                    processed_row.append(float(val))
                except ValueError:
                    processed_row.append(val)
            rows.append(processed_row)

    # The customer asks about the "two built-in rules". 
    # In this dataset, 'criterion' is the primary rule-based parameter.
    # We compare the performance (MRE-) of 'absolute_error' vs 'squared_error'.
    mre_idx = header.index('MRE-')
    crit_idx = header.index('criterion')
    
    abs_error_results = [r[mre_idx] for r in rows if r[crit_idx] == 'absolute_error']
    sq_error_results = [r[mre_idx] for r in rows if r[crit_idx] == 'squared_error']
    
    is_same = certify(abs_error_results, sq_error_results)
    
    print("Analysis of rule choice (absolute_error vs squared_error):")
    if is_same:
        print("The choice between these rules makes no statistically significant difference.")
        print("You can stop arguing about it.")
    else:
        print("The choice between these rules results in a statistically significant difference.")
        print("The rule choice does matter for your performance.")
