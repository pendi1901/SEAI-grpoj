import csv

def remember(header, rows):
    """
    Identifies the best rows based on multi-objective optimization.
    Goals: Minimize columns ending in '-', Maximize columns ending in '+'.
    Uses a simple Pareto-dominance approach to find the 'best' setups.
    """
    goals = []
    for i, name in enumerate(header):
        if name.endswith('-'):
            goals.append((i, -1))  # Minimize
        elif name.endswith('+'):
            goals.append((i, 1))   # Maximize

    def get_score(row):
        # Normalize values to compare different scales
        # Since we want to find 'unusually trustworthy', we look for 
        # rows that perform well across all goal columns.
        scores = []
        for idx, direction in goals:
            val = row[idx]
            # Simple heuristic: higher is better for +, lower for -
            scores.append(val * direction)
        return sum(scores)

    # Calculate scores and keep track of original indices
    indexed_rows = []
    for i, row in enumerate(rows):
        indexed_rows.append((i, get_score(row)))

    # Sort by score descending
    indexed_rows.sort(key=lambda x: x[1], reverse=True)

    # Return top 5% (800 rows * 0.05 = 40 rows)
    limit = max(1, int(len(rows) * 0.05))
    return [item[0] for item in indexed_rows[:limit]]

import csv

def guess(header, train, test):
    """
    Estimates the first goal column (MRE-) for test rows using a 
    RandomForestRegressor. Handles symbolic columns by encoding them.
    """
    df_train = pd.DataFrame(train, columns=header)
    df_test = pd.DataFrame(test, columns=header)

    # Identify goal columns
    goal_cols = [name for name in header if name.endswith('-') or name.endswith('+')]
    target_col = goal_cols[0]
    
    # Features are all columns except the goals
    features = [name for name in header if name not in goal_cols]
    
    # Encode symbolic columns
    # We must ensure all columns are numeric for sklearn
    for col in header:
        if df_train[col].dtype == 'object':
            le = LabelEncoder()
            # Combine train and test to ensure all categories are covered
            combined = pd.concat([df_train[col].astype(str), df_test[col].astype(str)], axis=0)
            le.fit(combined)
            df_train[col] = le.transform(df_train[col].astype(str))
            df_test[col] = le.transform(df_test[col].astype(str))
        else:
            # Ensure numeric columns are float
            df_train[col] = df_train[col].astype(float)
            df_test[col] = df_test[col].astype(float)

    X_train = df_train[features]
    y_train = df_train[target_col]
    X_test = df_test[features]

    # Train model
    model = RandomForestRegressor(n_estimators=100, random_state=42)
    model.fit(X_train, y_train)
    
    # Predict
    predictions = model.predict(X_test)
    return predictions.tolist()

import csv

from scipy.stats import pearsonr

def certify(xs, ys):
    """
    Returns True if the correlation between two sets of numbers is high,
    suggesting the relationship is real rather than noise.
    """
    if len(xs) < 2 or len(ys) < 2:
        return False
    corr, _ = pearsonr(xs, ys)
    # If correlation is strong (e.g., > 0.5), we consider the trend real
    return abs(corr) > 0.5

def analyze_setup_size_vs_accuracy(header, rows):
    """
    Analyzes if 'big' setups (N_estimators) correlate with better accuracy (ACC+).
    """
    df = pd.DataFrame(rows, columns=header)

    # ACC+ is the goal to maximize
    xs = df['N_estimators'].tolist()

    else:
        return "The relationship between setup size and accuracy appears to be noise; no significant correlation found."

import csv

def flow(header, row, seen):
    """
    Decides if the current row is 'good enough' to stop searching.
    We define 'good enough' as being in the top 5% of the performance 
    distribution based on the goals observed so far.
    """
    goals = []
    for i, name in enumerate(header):
        if name.endswith('-'):
            goals.append((i, -1))
        elif name.endswith('+'):
            goals.append((i, 1))

    def get_score(r):
        score = 0
        for idx, direction in goals:
            score += r[idx] * direction
        return score

    # Calculate score for current row
    current_score = get_score(row)

    # If this is the first row, we can't compare yet
    if not seen:
        return False

    # Compare against all previously seen rows
    # We stop if the current row is better than 95% of what we've seen
    seen_scores = [get_score(r) for r in seen]
    better_than_count = sum(1 for s in seen_scores if current_score > s)
    
    # Threshold: if better than 95% of history, or if we have enough history
    # to be confident (e.g., at least 20 rows seen)
    if len(seen) >= 20:
        if better_than_count / len(seen) >= 0.95:
            return True
            
    return False

def blame(header, rows):
    """
    Identifies which input columns contribute most to poor performance.
    We define 'bad' as the bottom 20% of rows based on a weighted goal score.
    We then train a classifier to predict the 'badness' and extract feature importances.
    """
    df = pd.DataFrame(rows, columns=header)
    
    goal_cols = [c for c in header if c.endswith('-') or c.endswith('+')]
    input_cols = [c for c in header if c not in goal_cols]
    
    # Calculate a normalized score for each row to identify 'bad' ones
    # Minimize '-' (invert) and Maximize '+'
    scores = []
    for _, row in df.iterrows():
        s = 0
        for col in goal_cols:
            val = row[col]
            if col.endswith('-'):
                s -= val
            else:
                s += val
        scores.append(s)
    
    df['score'] = scores
    # Threshold for 'bad' (bottom 20%)
    threshold = df['score'].quantile(0.2)
    df['is_bad'] = (df['score'] <= threshold).astype(int)
    
    # Prepare features
    X = df[input_cols].copy()
    for col in input_cols:
        if X[col].dtype == 'object':
            le = LabelEncoder()
            X[col] = le.fit_transform(X[col].astype(str))
        else:
            # Fill numeric NaNs with mean, ensuring we don't touch object columns
            X[col] = X[col].fillna(X[col].mean())
            
    y = df['is_bad']
    
    # Use Random Forest to find which inputs correlate with being 'bad'
    clf = RandomForestClassifier(n_estimators=50, random_state=42)
    clf.fit(X, y)
    
    importances = sorted(zip(input_cols, clf.feature_importances_), key=lambda x: x[1], reverse=True)
    
    # Return the top 3 inputs that explain the bad performance
    return [name for name, score in importances[:3]]
